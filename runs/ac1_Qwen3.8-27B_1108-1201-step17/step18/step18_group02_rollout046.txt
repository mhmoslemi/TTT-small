def away_frank_wolfe(y, g, n):
    """
    y: current point (sum to 1, non-negative)
    g: FW vertex (sum to 1, non-negative)
    Returns: new y after FW or Away step
    """
    # FW step with optimal alpha (line search)
    # f(y + alpha*(g-y)) for alpha in [0, 1]
    # For convex f, the optimal alpha is 2/(t+2) or we can do backtracking
    
    # For away step:
    # Find v_A: the basis vertex e_i that minimizes <e_i, g - y> = g_i - y_i
    # i.e., the one where y_i is largest relative to g_i
    # Actually, v_A should be chosen to maximize the decrease
    # Standard choice: v_A = e_i where i = argmax_j (y_j - g_j) / y_j (or just argmax y_j where g_j = 0)
    
    # If g == y, no step
    if np.allclose(g, y):
        return y
    
    # Choose away vertex: the one with max y_i among those where g_i is small
    # Simple heuristic: argmax y_i where g_i < y_i / 2
    candidates = [i for i in range(n) if y[i] > 0 and g[i] < y[i] * 0.5]
    if not candidates:
        # Fall back to FW step
        return y  # caller handles alpha
    
    # Pick the one with largest y_i
    v_A_idx = max(candidates, key=lambda i: y[i])
    
    # Away direction: d = (g - e_{v_A}) / (1 - 1) ... 
    # Actually: away point = (y - alpha_A * e_{v_A}) / (1 - alpha_A)
    # where alpha_A = y[v_A_idx] (to make component v_A_idx zero)
    # Then FW step from the away point toward g
    
    # More standard: 
    # y_away = (y - y[v_A] * e_{v_A}) / (1 - y[v_A])
    # Then try: (1-t) * y_away + t * g for t in [0, 1]
    # Or equivalently: y + t * (g - e_{v_A}) * (some scaling)
    
    # Let me use a simpler formulation:
    # The away step direction is: d_away = (g - e_{v_A}) 
    # We want to find alpha such that y + alpha * d_away stays in simplex
    # y_i + alpha * (g_i - [i == v_A_idx]) >= 0 for all i
    # For i == v_A_idx: y[v_A] - alpha >= 0, so alpha <= y[v_A]
    # For other i: y[i] + alpha * (g[i] - 0) >= 0, always true if alpha >= 0
    # Also need sum to be 1: sum(y) + alpha * (sum(g) - 1) = 1 + alpha*(1-1) = 1. Good.
    # So max alpha_away = y[v_A_idx]
    
    # But we also want to minimize the objective. Let's try alpha = y[v_A_idx] (full away step)
    # and also try the FW step, and pick the better one.
    
    y_away = y.copy()
    y_away[v_A_idx] = 0
    s = y_away.sum()
    if s < 1e-15:
        return y
    y_away = y_away / s
    
    return y_away  # Return the away point; caller will then do FW step from here

**Direct subgradient method for the normalized problem:**

Minimize f(a) = max_k (a*a)[k] subject to sum(a) = 1, a >= 0.

The subgradient of f at a is: (a*a_argmax) where (a*a_argmax)[i] = sum_{j: i+j = k*} a[j] for the argmax k*.

Actually, the subgradient of max_k (a*a)[k] at a is the set of convex combinations of the subgradients of each (a*a)[k] that achieves the max. The subgradient of (a*a)[k] at a is: the vector with (a*a)[k]_i = a[i+k] + a[i-k] (approximately, with boundary effects).

More precisely: d/da_i [(a*a)[k]] = a[i-k] + a[k-i] (with appropriate boundary handling).

So the subgradient of f(a) = max_k (a*a)[k] is:
sg_i = sum_{k in argmax} w_k * (a[i-k] + a[k-i]) for some weights w_k >= 0, sum w_k = 1.

A simple subgradient method:
1. Compute ac = autocorr(a)
2. Find k* = argmax ac
3. Compute sg[i] = a[i-k*] + a[k*-i] (with boundary: a[j] = 0 if j < 0 or j >= n)
4. Update: a = a - eta * sg (projected onto {a >= 0, sum(a) = 1})
5. Decrease eta

This is O(n) per iteration (plus O(n log n) for the autocorrelation).

But this is very slow to converge. The LP-FW is much better because it finds the optimal direction at each step.

**Hybrid approach**: Use subgradient to guide the LP-FW constraint selection. The subgradient tells us which lag is most important, so we definitely include that constraint in the LP.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def build_constraint_matrix(y, constraint_list, n):
    """Build the constraint matrix for the LP: row k has y[i] at column (k-i)."""
    K = len(constraint_list)

            valid = (g_idx >= 0) & (g_idx < n)
            if np.any(valid):
                coeffs[jj, g_idx[valid]] = y[idx[valid]]
    return coeffs

def subgradient_descent(a0, time_limit, verbose=True):
    """Nesterov-accelerated subgradient descent on max autocorrelation."""
    n = len(a0)
    y = norm_sum1(a0).astype(np.float64)
    y_nesterov = y.copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()

    t0 = time.monotonic()
    c0 = max(fast_autocorr(y).max() * 0.5, 0.01)
    gamma = 1.0
    beta = 0.0
    rng = np.random.default_rng(42)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y_nesterov)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))

        # Subgradient: for k_star, sg[i] = y_nesterov[i - k_star] + y_nesterov[k_star - i] (with boundaries)
        sg = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            g1_idx = idx - k_star
            g2_idx = k_star - idx
            valid1 = (g1_idx >= 0) & (g1_idx < n)
            valid2 = (g2_idx >= 0) & (g2_idx < n)
            if np.any(valid1):
                sg[idx[valid1]] += y_nesterov[g1_idx[valid1]]
            if np.any(valid2):
                sg[idx[valid2]] += y_nesterov[g2_idx[valid2]]

        # Step size
        step = gamma / np.sqrt(t)
        y_new = y_nesterov - step * sg
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / s

        # Nesterov update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_nesterov_new = y_new + (beta_new - 1.0) / (beta + 1.0) * (y_new - y_nesterov)
        y_nesterov_new = np.maximum(y_nesterov_new, 0.0)
        s2 = y_nesterov_new.sum()
        if s2 > 1e-10:
            y_nesterov_new = y_nesterov_new / s2
        else:
            y_nesterov_new = y_new.copy()

        y = y_new
        y_nesterov = y_nesterov_new
        beta = beta_new

        m_check = fast_autocorr(y).max()
        if m_check < best_m - 1e-14:
            best_m = m_check
            best_a = y.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad iter={t} ({elapsed:.0f}s): J={J:.6f}")

    # Also check y_nesterov
    m_fn = fast_autocorr(y_nesterov).max()
    if m_fn < best_m:
        best_m = m_fn
        best_a = norm_sum1(y_nesterov)

    return best_a, best_m

    a = norm_sum1(a0).astype(np.float64)

        i_end = min(n - 1, k_max)
        if i_start > i_end:
            break

        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_max - idx]
        if contrib.max() < 1e-20:
            break

        src = idx[np.argmax(contrib)]
        src_val = a[src]

        if len(candidates) == 0:
            candidates = np.arange(n)

        # Score: prefer positions that reduce peak
        best_dst = None

            amt = src_val * frac
            if amt < 1e-15:

            for _ in range(3):

                if dst == src:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_trial_m - 1e-16:

                    best_dst = (dst, frac)

        if best_dst is not None:
            dst, frac = best_dst
            amt = src_val * frac
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_m = best_trial_m
            best_a = a.copy()
        else:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_candidates(n):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    for freq in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.005)
            candidates.append(a / a.sum())

    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())

    a = np.cos(np.pi * (x - 0.5)) ** 2
    candidates.append(a / a.sum())
    candidates.append(np.ones(n) / n)

    # Alternating patterns
    for period in [2, 3, 4]:
        a = np.ones(n)
        a[period - 1::period] = 0.3
        candidates.append(a / a.sum())

    return candidates


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with AFW, subgradient, and multi-phase approach."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick parametric search ===
    print(f"\n=== Phase 1: Parametric search ===")
    n_best = len(best_seq)
    t1_end = t_global + min(time_budget * 0.04, 50.0)
    if time.monotonic() < t1_end:
        candidates = parametric_candidates(n_best)
        for i, cand in enumerate(candidates):
            if time.monotonic() > t1_end:
                break
            sc = evaluate_safe(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"    Param {i}: {sc:.6f}")

    # Also try blends
    rng0 = np.random.default_rng(555)
    for w in [0.3, 0.5, 0.7]:
        cand = w * best_seq + (1 - w) * np.ones(n_best) / n_best
        cand = norm_sum1(cand)
        check_and_update(cand, f"blend_{w}")

    # === Phase 2: Main AFW optimization ===
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AFW (n={n_best}) ===")
    t2 = min(time_budget * 0.45, 600.0)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(t2, remaining - 100.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = afw_optimize(a_in, t2, solver, max_K=2500, verbose=True)
        check_and_update(a_out, f"afw_n{n_best}")

    # === Phase 3: Subgradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Subgradient (n={n_best}) ===")
    t3 = min(remaining * 0.12, 120.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_descent(a_in, t3, verbose=True)
        check_and_update(a_out, f"subgrad_n{n_best}")

    # === Phase 4: AFW restart from perturbed best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: AFW restart (n={n_best}) ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 60:
        rng_p = np.random.default_rng(314)
        pert = rng_p.normal(0, 0.015, n_best) * best_seq
        a_pert = best_seq + pert
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        a_out, _ = afw_optimize(a_pert, t4, solver, max_K=2000, verbose=True)
        check_and_update(a_out, f"afw_restart_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length ===")
    target_lengths = []
    for delta in [300, -300, 500, -500]:
        nl = n_best + delta
        if 2500 <= nl <= 6000:
            target_lengths.append(nl)

    seen = set()
    unique_lens = []
    for ln in target_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lens.append(ln)

    for new_n in unique_lens[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"  Trying n={new_n}")
        a_interp = interpolate_sequence(best_seq, new_n)
        t5a = min(remaining * 0.3, 150.0)
        if t5a > 40:
            a_out, _ = afw_optimize(a_interp, t5a, solver, max_K=2000, verbose=False)
            check_and_update(a_out, f"afw_n{new_n}")

    # === Phase 6: Final AFW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final AFW (n={n_best}) ===")
    if remaining > 120 and n_best <= 5500:
        t6 = min(remaining * 0.5, 300.0)
        if t6 > 60:
            a_in = norm_sum1(best_seq)
            a_out, _ = afw_optimize(a_in, t6, solver, max_K=3000, verbose=True)
            check_and_update(a_out, f"afw_final_n{n_best}")

    # === Phase 7: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Coord polish (n={n_best}) ===")
        t7 = min(remaining * 0.6, 80.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_polish(a_in, t7, rng_seed=999, verbose=False)
            check_and_update(a_out, f"coord_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
