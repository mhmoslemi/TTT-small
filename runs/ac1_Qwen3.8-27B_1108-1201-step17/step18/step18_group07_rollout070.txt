def spectral_step(a, amt=0.2):
    n=len(a); N=_next_pow2(2*n)
    A=np.fft.rfft(a,N); mag=np.abs(A); ph=np.angle(A)
    mean=mag.mean()
    if mean <=1e-300: return normalize(a)
    ratio=mag/mean
    gamma=max(0.05, 1.0-amt)
    mag_new=mean*np.power(np.maximum(ratio,0.0), gamma)
    Anew=mag_new*np.exp(1j*ph)
    an=np.fft.irfft(Anew,N)[:n]
    an=np.maximum(an,0.0)
    s=an.sum()
    if s<=1e-300: return normalize(a)
    an/=s
    mix=0.5*amt
    out=(1.0-mix)*a + mix*an
    return normalize(out)

- L-BFGS-B polishing in softmax/log-space on a smooth surrogate objective.

- Length interpolation variation and elite candidate final validation.

import numpy as np

    from scipy.optimize import minimize as scipy_minimize

def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return np.array([1.0], dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 1e-15 or not np.isfinite(s):
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    return x / s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    N = _next_pow2(m)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:m]


def score_unit(a):
    """Score assuming a is non-negative and sum(a) == 1."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")
    c = fast_conv(a)
    val = float(2.0 * n * np.max(c))
    if not np.isfinite(val):
        return float("inf")
    return val


def corr_from_weights(w, a):
    """
    G[j] = sum_k w[k] a[k-j], with a[m]=0 outside [0,n-1].
    This is the unnormalized gradient building block for self-convolution peaks.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 3 * n
    N = _next_pow2(need)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g = np.fft.irfft(W * np.conj(A), N)[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    return np.maximum(g, 0.0)


def smooth_components(a, beta):
    a = np.asarray(a, dtype=np.float64)
    c = fast_conv(a)
    maxc = float(np.max(c))
    beta = max(1e-12, float(beta))
    expc = np.exp(beta * (c - maxc))
    s = float(np.sum(expc))
    if s <= 1e-300 or not np.isfinite(s):
        w = np.ones_like(c, dtype=np.float64) / float(c.size)
        f = maxc
    else:
        w = expc / s
        f = maxc + float(np.log(s) / beta)
    G = corr_from_weights(w, a)
    return c, f, w, G


def apply_mirror_g(a, g, eta):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    if eta <= 0.0:
        return normalize(a)
    g = np.where(np.isfinite(g), g, 0.0)
    loga = np.log(np.maximum(a, 1e-300))
    loga = loga - float(eta) * g
    loga = np.where(np.isfinite(loga), loga, -700.0)
    loga = loga - float(np.max(loga))
    an = np.exp(loga)
    an = np.where(np.isfinite(an), an, 0.0)
    an = np.maximum(an, 0.0)
    s = float(np.sum(an))
    if s <= 1e-300:
        return normalize(a)
    return an / s


def score_after_move_buf(c, a, s_idx, d_idx, eps, buf):
    """
    Exact score after moving eps mass from s_idx to d_idx, using current c=conv(a,a).
    buf is an output buffer of length len(c).
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    m = int(c.size)
    if eps <= 0.0 or s_idx == d_idx or n == 0:
        return float(2.0 * n * np.max(c))

    buf[:] = 0.0
    two_eps = 2.0 * float(eps)

    # Linear term: 2 * eps * (shift(a,d) - shift(a,s))
    buf[d_idx:d_idx + n] += two_eps * a
    buf[s_idx:s_idx + n] -= two_eps * a

    # Quadratic sparse terms.
    eps2 = float(eps) * float(eps)
    q = 2 * int(d_idx)
    if q < m:
        buf[q] += eps2
    q = 2 * int(s_idx)
    if q < m:
        buf[q] += eps2
    q = int(d_idx) + int(s_idx)
    if q < m:
        buf[q] -= 2.0 * eps2

    np.add(c, buf, out=buf)
    val = float(2.0 * n * np.max(buf))
    return val if np.isfinite(val) else float("inf")


def spectral_step(a, amt=0.2):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return a
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    ph = np.angle(A)
    mean = float(np.mean(mag))
    if mean <= 1e-300:
        return normalize(a)
    ratio = mag / mean
    gamma = max(0.05, 1.0 - float(amt))
    mag_new = mean * np.power(np.maximum(ratio, 0.0), gamma)
    A_new = mag_new * np.exp(1j * ph)
    an = np.fft.irfft(A_new, N)[:n]
    an = np.where(np.isfinite(an), an, 0.0)
    an = np.maximum(an, 0.0)
    s = float(np.sum(an))
    if s <= 1e-300:
        return normalize(a)
    an = an / s
    mix = 0.5 * float(amt)
    out = (1.0 - mix) * a + mix * an
    return normalize(out)


def make_uniform(n):
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    return np.ones(n, dtype=np.float64) / float(n)


def make_power(n, p):
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.power(np.maximum(x, 1e-9), float(p)) + 0.05
    return normalize(a)


def make_exp(n, lam):
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(float(lam) * (x - 0.5))
    return normalize(a)


def make_segments(n, levels, power=1.0):
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    if levels <= 1:
        return make_uniform(n)
    levels = int(levels)
    counts = [n // levels] * levels
    for i in range(n % levels):
        counts[i] += 1
    a = np.empty(n, dtype=np.float64)
    pos = 0
    for lvl in range(levels):
        val = float(levels - lvl) ** float(power)
        a[pos:pos + counts[lvl]] = val
        pos += counts[lvl]
    a += 0.1
    return normalize(a)


def make_gaussian(n, center, width):
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    w = max(1e-6, float(width))
    a = np.exp(-0.5 * ((x - float(center)) / w) ** 2)
    a += 0.01
    return normalize(a)


def make_random_smooth(n, seed):
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    rng = np.random.default_rng(int(seed))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) * 0.7
    for f in range(1, 16):
        coeff = rng.normal(0.0, 0.12 / np.sqrt(float(f)))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        a += coeff * np.sin(2.0 * np.pi * float(f) * x + phase)
    a = np.maximum(a, 0.01)
    return normalize(a)


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([1.0], dtype=np.float64)
    old_n = int(f.size)
    if old_n == 0:
        return make_uniform(new_n)
    if old_n == new_n:
        return normalize(f)
    s = float(np.sum(np.maximum(f, 0.0)))
    if s <= 1e-15:
        return make_uniform(new_n)
    f = f / s
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    nf = np.interp(new_x, old_x, f)
    nf = np.maximum(nf, 0.0)
    ss = float(np.sum(nf))
    if ss <= 1e-15:
        return make_uniform(new_n)
    return nf / ss


def add_elite(elite, a, sc, max_keep=10):
    if a is None:
        return elite
    try:
        sc = float(sc)
    except Exception:
        return elite
    if not np.isfinite(sc):
        return elite
    if len(elite) >= max_keep:
        min_idx = 0
        min_val = float(elite[0][0])
        for i in range(1, len(elite)):
            v = float(elite[i][0])
            if v < min_val:
                min_val = v
                min_idx = i
        if sc >= min_val - 1e-9:
            return elite
        elite.pop(min_idx)
    elite.append((sc, np.asarray(a, dtype=np.float64).copy()))
    return elite


def alloc_with_reserve(remaining, alloc, min_reserve=250.0, frac_reserve=0.18):
    remaining = max(0.0, float(remaining))
    reserve = max(float(min_reserve), remaining * float(frac_reserve))
    avail = remaining - reserve
    if avail <= 5.0:
        return 0.0
    return min(float(alloc), float(avail))


def pair_polish(a0, time_limit, beta=120.0, max_attempts=800, verbose=False):
    a = normalize(a0)
    n = int(a.size)
    best = a.copy()
    best_sc = score_unit(a)
    current_sc = best_sc
    t0 = time.monotonic()
    deadline = t0 + float(time_limit)

    rng = np.random.default_rng(24681357)
    stall = 0
    beta_cur = float(beta)
    n_src = min(40, n)
    n_dest = min(240, n)

    while time.monotonic() < deadline:
        c = fast_conv(a)
        if c.size == 0:
            break
        cur_val = float(2.0 * n * np.max(c))
        if not np.isfinite(cur_val):
            break
        current_sc = cur_val

        maxc = float(np.max(c))
        expc = np.exp(beta_cur * (c - maxc))
        ss = float(np.sum(expc))
        if ss <= 1e-300 or not np.isfinite(ss):
            w = np.ones_like(c, dtype=np.float64) / float(c.size)
        else:
            w = expc / ss
        G = corr_from_weights(w, a)

        src_score = a * G
        src_score = np.where(np.isfinite(src_score), src_score, 0.0)
        dst_order = np.argsort(G)
        src_idx = np.argsort(src_score, kind="mergesort")[::-1][:n_src]

        base_dest = dst_order[:n_dest]
        if n > 80:
            lo_start = n // 5
            lo_stop = n // 2
            if lo_stop > lo_start:
                mid = dst_order[lo_start:lo_stop]
                pick = rng.choice(mid, size=min(50, len(mid)), replace=False)
                dest_idx = np.unique(np.concatenate((base_dest, np.asarray(pick, dtype=int))))
            else:
                dest_idx = base_dest
        else:
            dest_idx = base_dest
        try_dests = dest_idx[:120]

        eval_buf = np.zeros(c.size, dtype=np.float64)
        improved = False
        attempts = 0
        max_att = int(max_attempts)

        # Direct single-move local search using exact O(n) convolution updates.
        for si in src_idx:
            s_pos = int(si)
            if a[s_pos] < 1e-12:
                continue
            g_s = G[s_pos]
            for di in try_dests:
                d_pos = int(di)
                if d_pos == s_pos:
                    continue
                if G[d_pos] >= g_s:
                    continue
                for frac in (0.02, 0.05, 0.1, 0.2, 0.35, 0.55):
                    attempts += 1
                    if attempts > max_att:
                        break
                    amt = a[s_pos] * frac
                    if amt <= 1e-300:
                        continue
                    sc_try = score_after_move_buf(c, a, s_pos, d_pos, amt, eval_buf)
                    if sc_try < current_sc - 1e-10:
                        # Local refinement around the accepted fraction.
                        best_frac = frac
                        best_trial_sc = sc_try
                        for f2 in (0.5, 2.0, 0.25):
                            fvar = frac * f2
                            if fvar <= 0.0 or fvar >= 1.0:
                                continue
                            amt2 = a[s_pos] * fvar
                            if amt2 <= 1e-300:
                                continue
                            sc2 = score_after_move_buf(c, a, s_pos, d_pos, amt2, eval_buf)
                            if sc2 < best_trial_sc:
                                best_trial_sc = sc2
                                best_frac = fvar
                        an = a.copy()
                        amt_best = a[s_pos] * best_frac
                        an[s_pos] = a[s_pos] - amt_best
                        an[d_pos] = a[d_pos] + amt_best
                        an = np.maximum(an, 0.0)
                        ssum = float(np.sum(an))
                        if ssum > 1e-300:
                            an = an / ssum
                        a = an
                        current_sc = best_trial_sc
                        improved = True
                        if current_sc < best_sc:
                            best_sc = current_sc
                            best = a.copy()
                        break
                if improved or attempts > max_att:
                    break
            if improved or attempts > max_att:
                break

        # Coordinated mirror step from smoothed gradient, if single moves failed.
        if not improved:
            g = G - np.mean(G)
            scale = float(np.max(np.abs(g)))
            if np.isfinite(scale) and scale > 1e-18:
                g = g / scale
                for eta_try in (0.25, 0.1, 0.5):
                    an = apply_mirror_g(a, g, eta_try)
                    sc_try = score_unit(an)
                    if sc_try < current_sc - 1e-10:
                        a = an
                        current_sc = sc_try
                        improved = True
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best = a.copy()
                        break

        # Spectral flatten attempt.
        if not improved:
            asp = spectral_step(a, 0.14)
            sc_try = score_unit(asp)
            if sc_try < current_sc - 1e-10:
                a = asp
                current_sc = sc_try
                improved = True
                if sc_try < best_sc:
                    best_sc = sc_try
                    best = a.copy()

        # Random single move using fast exact scoring.
        if not improved and positive_present and try_dests.size > 0:
            s_pos = int(positive[rng.integers(positive.size)])
            d_pos = int(try_dests[rng.integers(try_dests.size)])
            if s_pos != d_pos:
                for frac in (0.015, 0.04, 0.09):
                    amt = a[s_pos] * frac
                    if amt <= 0.0:
                        continue
                    sc_try = score_after_move_buf(c, a, s_pos, d_pos, amt, eval_buf)
                    if sc_try < current_sc - 1e-10:
                        an = a.copy()
                        an[s_pos] = a[s_pos] - amt
                        an[d_pos] = a[d_pos] + amt
                        an = np.maximum(an, 0.0)
                        ssum = float(np.sum(an))
                        if ssum > 1e-300:
                            an = an / ssum
                        a = an
                        current_sc = sc_try
                        improved = True
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best = a.copy()
                        break

        if improved:
            stall = 0
            beta_cur = min(2000.0, beta_cur * 1.02)
            if verbose and stage % 250 == 0:
                print(f"    pair_polish improved: {best_sc:.8f}")
            continue

        # No improvement: change scale and perturb.
        stall += 1
        beta_cur *= 1.35
        if beta_cur > 2500.0:
            beta_cur = 25.0

        if stall > 6 or stall % 4 == 0:
            noise = rng.normal(0.0, 0.012, n)
            a = normalize(0.7 * best + 0.3 * a + noise)
            current_sc = score_unit(a)
            if current_sc < best_sc:
                best_sc = current_sc
                best = a.copy()

        stall = 0

    return best, best_sc


def smooth_mirror_optimize(a0, time_limit, beta_start=5.0, beta_end=600.0, eta_start=0.45):
    a = normalize(a0)
    n = int(a.size)
    best = a.copy()
    best_sc = score_unit(a)
    current_sc = best_sc
    t0 = time.monotonic()
    deadline = t0 + float(time_limit)
    eta = float(eta_start)
    stall = 0
    it = 0
    rng = np.random.default_rng(987654321)

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - t0
        prog = min(1.0, elapsed / max(float(time_limit), 1e-9))
        beta = float(beta_start) * (float(beta_end) / float(beta_start)) ** prog

        c, f, w, G = smooth_components(a, beta)
        g = G - np.mean(G)
        scale = float(np.max(np.abs(g)))
        if not np.isfinite(scale) or scale < 1e-18:
            a = normalize(best + rng.normal(0.0, 0.01, n))
            current_sc = score_unit(a)
            eta = max(0.05, eta * 0.7)
            stall = 0
            continue
        g = g / scale
        g = np.clip(g, -4.0, 4.0)

        an = None
        sc_try = None
        for factor in (1.0, 0.5, 0.25):
            trial = apply_mirror_g(a, g, eta * factor)
            trial_sc = score_unit(trial)
            if trial_sc < current_sc - 1e-10:
                an = trial
                sc_try = trial_sc
                break

        if an is not None:
            a = an
            current_sc = sc_try
            if sc_try < best_sc:
                best_sc = sc_try
                best = a.copy()
            stall = 0
            eta = min(1.8, eta * 1.04)
        else:
            stall += 1
            eta = max(0.05, eta * 0.72)
            if stall > 25:
                if stall > 90:
                    a = best.copy()
                    current_sc = best_sc
                else:
                    a = normalize(0.6 * best + 0.4 * (a + rng.normal(0.0, 0.02, n)))
                    current_sc = score_unit(a)
                eta = max(0.2, eta * 1.4)
                stall = 0

        if it % 250 == 0:
            asp = spectral_step(a, 0.13)
            sc_try = score_unit(asp)
            if sc_try < current_sc - 1e-10:
                a = asp
                current_sc = sc_try
                if sc_try < best_sc:
                    best_sc = sc_try
                    best = a.copy()

    return best, best_sc


def softmax_w(w):
    w = np.asarray(w, dtype=np.float64)
    if w.size == 0:
        return np.array([1.0], dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w = w - float(np.max(w))
    y = np.exp(w)
    s = float(np.sum(y))
    if s <= 1e-300 or not np.isfinite(s):
        y = np.exp(w - 700.0)
        s = float(np.sum(y))
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(w.size, dtype=np.float64) / float(w.size)
    return y / s


def scipy_smooth_obj(x):
    if time.monotonic() > _SCIPY_DEADLINE:
        return 1e100, np.zeros_like(x, dtype=np.float64)

    x = np.asarray(x, dtype=np.float64)
    a = softmax_w(x)
    n = int(a.size)
    if n == 0:
        return 1e100, np.zeros_like(x, dtype=np.float64)

    c = fast_conv(a)
    if not np.all(np.isfinite(c)):
        return 1e100, np.zeros_like(x, dtype=np.float64)

    maxc = float(np.max(c))
    b = max(1e-12, float(_SCIPY_BETA))
    expc = np.exp(b * (c - maxc))
    s = float(np.sum(expc))
    if s <= 1e-300 or not np.isfinite(s):
        return 1e100, np.zeros_like(x, dtype=np.float64)

    val = maxc + float(np.log(s) / b)
    w = expc / s
    G = corr_from_weights(w, a)
    grad_a = 2.0 * G
    mu = float(np.dot(grad_a, a))
    grad_w = a * (grad_a - mu)
    grad_w -= float(np.mean(grad_w))
    grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)

    scale = float(n) * float(n)
    val *= scale
    grad_w = grad_w * scale

    if not np.isfinite(val):
        val = 1e100
    return float(val), grad_w.astype(np.float64, copy=False)


def l_bfgs_polish(a0, time_limit, beta=100.0, maxfun=None):
    a = normalize(a0)
    n = int(a.size)
    if scipy_minimize is None or n == 0 or n > 12000 or time_limit <= 5.0:
        return a, score_unit(a)

    best = a.copy()
    best_sc = score_unit(a)
    t0 = time.monotonic()
    deadline = t0 + float(time_limit)

    if maxfun is None:
        maxfun = int(max(900, min(15000, float(time_limit) * 160)))

    global _SCIPY_BETA, _SCIPY_DEADLINE
    _SCIPY_BETA = float(beta)
    _SCIPY_DEADLINE = deadline

    w0 = np.log(np.maximum(a, 1e-12))
    w0 = w0 - float(np.mean(w0))
    w0 = np.clip(w0, -35.0, 35.0)

    bounds = [(-35.0, 35.0)] * n
    options = {
        "maxiter": 300,
        "maxfun": int(maxfun),
        "ftol": 1e-15,
        "gtol": 1e-6,
        "disp": False,
    }

    try:
        res = scipy_minimize(scipy_smooth_obj, w0, jac=True, method="L-BFGS-B",
                             bounds=bounds, options=options)
        if time.monotonic() < deadline:
            an = normalize(softmax_w(np.asarray(res.x, dtype=np.float64)))
            sc = score_unit(an)
            if np.isfinite(sc) and sc < best_sc:
                best = an
                best_sc = sc
    except Exception:
        pass

    return best, best_sc


def propose_candidate(budget_s=1800.0):

            parent = normalize(p)
            if parent.size > 9000:
                parent = interpolate_sequence(parent, 6000)

            best_sc = sc
            best = sd.copy()
        elite = add_elite(elite, sd, sc)

    if parent is not None:
        small_seeds = seeds[:10]
        for alpha in (0.2, 0.5, 0.8):
            for sd in small_seeds:
                if sd.size != parent.size:
                    continue
                bl = normalize(alpha * parent + (1.0 - alpha) * sd)
                sc = score_unit(bl)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best = bl.copy()
                elite = add_elite(elite, bl, sc)

    # Quick global-ish smoothing in log-space.
    remaining = deadline - time.monotonic()
    t_lim = alloc_with_reserve(remaining, 70.0, 350.0, 0.2)
    if scipy_minimize is not None and t_lim > 15.0:
        cand, sc = l_bfgs_polish(best, t_lim, beta=80.0)
        elite = add_elite(elite, cand, sc)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
            print(f"quick l-bfgs score={best_sc:.8f}")

    # Fast exact local polish.
    remaining = deadline - time.monotonic()
    t_lim = alloc_with_reserve(remaining, 200.0, 400.0, 0.2)
    if t_lim > 20.0:
        cand, sc = pair_polish(best, t_lim, beta=150.0, max_attempts=700)
        elite = add_elite(elite, cand, sc)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
            print(f"quick pair polish score={best_sc:.8f}")

    # Smooth mirror descent.
    remaining = deadline - time.monotonic()
    t_lim = alloc_with_reserve(remaining, 450.0, 450.0, 0.2)
    if t_lim > 30.0:
        cand, sc = smooth_mirror_optimize(best, t_lim, beta_start=4.0, beta_end=500.0, eta_start=0.45)
        elite = add_elite(elite, cand, sc)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
            print(f"mirror descent score={best_sc:.8f}")

    # L-BFGS polishing at several smoothing temperatures.
    if scipy_minimize is not None:
        for bval in (50.0, 200.0, 700.0):
            remaining = deadline - time.monotonic()
            t_lim = alloc_with_reserve(remaining, 110.0, 350.0, 0.2)
            if t_lim < 20.0:
                break
            cand, sc = l_bfgs_polish(best, t_lim, beta=bval)
            elite = add_elite(elite, cand, sc)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand.copy()
                print(f"l-bfgs beta={bval:.0f} score={best_sc:.8f}")

    # Polish a few distinct elite candidates.
    polished = 0
    for item in list(elite):
        if polished >= 3:
            break
        sc_el = item[0]
        arr = item[1]
        if arr.size != best.size:
            continue
        diff = float(np.max(np.abs(arr - best)))
        if diff < 1e-7:
            continue
        remaining = deadline - time.monotonic()
        t_lim = alloc_with_reserve(remaining, 70.0, 350.0, 0.2)
        if t_lim < 15.0:
            break
        cand, sc = pair_polish(arr, t_lim, beta=250.0, max_attempts=500)
        polished += 1
        elite = add_elite(elite, cand, sc)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
            print(f"elite polish score={best_sc:.8f}")

    # Length variation.
    for delta in (-600, -300, 300, 600, 1000):
        remaining = deadline - time.monotonic()
        t_lim = alloc_with_reserve(remaining, 80.0, 300.0, 0.2)
        if t_lim < 30.0:
            break
        new_n = int(best.size) + int(delta)
        if new_n < 1500 or new_n > 9000:
            continue
        cand = interpolate_sequence(best, new_n)
        sc = score_unit(cand)
        elite = add_elite(elite, cand, sc)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
            print(f"length {new_n} score={best_sc:.8f}")
            remaining = deadline - time.monotonic()
            t_lim2 = alloc_with_reserve(remaining, 50.0, 250.0, 0.2)
            if t_lim2 > 10.0:
                cand2, sc2 = pair_polish(cand, t_lim2, beta=250.0, max_attempts=450)
                elite = add_elite(elite, cand2, sc2)
                if np.isfinite(sc2) and sc2 < best_sc:
                    best_sc = sc2
                    best = cand2.copy()
                    print(f"length {new_n} polished score={best_sc:.8f}")

    # Final intensive pair polish.
    remaining = deadline - time.monotonic()
    t_lim = min(remaining * 0.45, 550.0)
    if t_lim > 20.0:
        cand, sc = pair_polish(best, t_lim, beta=300.0, max_attempts=900)
        elite = add_elite(elite, cand, sc)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
            print(f"final pair polish score={best_sc:.8f}")

    # Final L-BFGS polish.
    remaining = deadline - time.monotonic()
    if scipy_minimize is not None:
        t_lim = min(remaining * 0.3, 260.0)
        if t_lim > 20.0:
            cand, sc = l_bfgs_polish(best, t_lim, beta=450.0)
            elite = add_elite(elite, cand, sc)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand.copy()
                print(f"final l-bfgs score={best_sc:.8f}")

    # Final mirror descent squeeze.
    remaining = deadline - time.monotonic()
    t_lim = min(remaining * 0.5, 300.0)
    if t_lim > 20.0:
        cand, sc = smooth_mirror_optimize(best, t_lim, beta_start=20.0, beta_end=1500.0, eta_start=0.3)
        elite = add_elite(elite, cand, sc)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
            print(f"final mirror score={best_sc:.8f}")

    # Spend whatever remains in fast local polish.
    remaining = deadline - time.monotonic()
    t_lim = max(0.0, remaining - 10.0)
    if t_lim > 10.0:
        cand, sc = pair_polish(best, t_lim, beta=600.0, max_attempts=1100)
        elite = add_elite(elite, cand, sc)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
            print(f"very final pair score={best_sc:.8f}")

    if best is None or best.size == 0 or not np.isfinite(best_sc):
        best = make_uniform(4000)
        best_sc = score_unit(best)

    best = normalize(best)

    # Choose the best according to the official evaluator over retained elites.
    final_cands = [x[1] for x in elite]
    final_cands.append(best)
    final_best = best.copy()
    final_score = best_sc
    for arr in final_cands:
        try:
            arr = np.asarray(arr, dtype=np.float64)
            lst = [float(x) for x in arr]
            sc = float(evaluate_sequence(lst))
            if np.isfinite(sc) and sc < final_score:
                final_score = sc
                final_best = arr.copy()
        except Exception:
            pass

    if not np.isfinite(final_score):
        final_best = best
        final_score = best_sc

    out = [float(x) for x in np.asarray(final_best, dtype=np.float64)]
    print(f"FINAL internal={best_sc:.8f} official={final_score:.8f} n={len(out)} time={time.monotonic()-t_start:.1f}s")
    return out
