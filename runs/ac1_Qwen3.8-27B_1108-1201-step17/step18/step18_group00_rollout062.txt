Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))^2` for non-negative sequences. Key insights:

1. The convolution `np.convolve(a,a)` peak is NOT always at lag 0 (unlike autocorrelation) - for uniform sequences it peaks at the middle
2. The gradient of `conv[k]` w.r.t. `a_j` is simply `2*a[k-j]` (a shifted copy of a)
3. LP-FW linearizes one argument of the bilinear form, giving a solvable LP subproblem
4. A smoothed objective (log-sum-exp of conv values) enables use of L-BFGS with softmax parameterization

My approach uses: fast LP-FW with scipy's HiGHS solver, smoothed L-BFGS with increasing temperature, targeted coordinate descent, and multi-length exploration.

"""

2. LP-based Frank-Wolfe with Nesterov acceleration (scipy linprog)
3. Smoothed L-BFGS with softmax parameterization and increasing beta

import numpy as np

from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft


def fast_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_obj(a):
    """Compute 2*n*max_conv for sum-normalized a."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def solve_lp_fw_subproblem(y, active_set, n):
    """Solve the LP subproblem for LP-FW using scipy linprog."""
    constraint_list = sorted(active_set)
    K = len(constraint_list)
    if K == 0:
        return None

    # Build constraint matrix
    A_rows = []
    for k in constraint_list:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        row = np.zeros(n + 1)
        idx = np.arange(i_start, i_end)
        row[idx] = y[k_int - idx]
        row[n] = -1.0  # -M coefficient
        A_rows.append(row)

    if not A_rows:
        return None

    A_ub = np.array(A_rows)
    b_ub = np.zeros(len(A_rows))

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimize M (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds: g_i >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def lp_fw_nesterov(a0, time_limit, initial_K=400, max_K=3000, verbose=True):
    """LP-FW with Nesterov acceleration and adaptive active set."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()

    # Initialize active set with top conv values
    ac = fast_conv(x)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 500
    iter_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        # Nesterov momentum
        mu_t = (np.sqrt(1.0 + 4.0 * t * t) - 1.0) / 2.0
        y = x + mu_t * (x - x_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute conv of y
        ac = fast_conv(y)
        M_curr = ac.max()

        # Update best
        ac_x = fast_conv(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set: add violated constraints
        violations = np.where(ac > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = ac[violations] - np.partition(ac, -max_K)[max_K - 1] if len(ac) > max_K else ac[violations]
            top_v = violations[np.argsort(ac[violations])[::-1][:50]]
            for k in top_v:
                active_set.add(int(k))

        # Trim active set if too large
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(list(active_set)[i]) for i in keep)

        # Solve LP subproblem
        g_dir = solve_lp_fw_subproblem(y, active_set, n)
        if g_dir is None:
            # Fallback: use uniform
            g_dir = np.ones(n) / n

        # Line search
        yf = rfft(y, 1 << (2 * n - 1).bit_length())
        gf = rfft(g_dir, 1 << (2 * n - 1).bit_length())
        fft_size = 1 << (2 * n - 1).bit_length()

        default_alpha = 2.0 / (t + 2.0)
        alphas = [max(0.01, min(1.0, default_alpha)),
                  max(0.01, min(1.0, default_alpha * 0.5)),
                  max(0.01, min(1.0, default_alpha * 2.0)),
                  max(0.01, min(1.0, 1.0 / (t + 1.0)))]
        if t > 100:
            alphas.append(max(0.01, min(1.0, default_alpha * 0.3)))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * x + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        # Update
        x_prev = x.copy()
        x = (1.0 - best_alpha) * x + best_alpha * g_dir
        x = np.maximum(x, 0.0)
        s_x = x.sum()
        if s_x < 1e-15:
            x = x_prev.copy()
        else:
            x = x / s_x

        iter_count += 1

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW-N iter={:5d} ({:.1f}s): J={:.6f}, |S|={}, alpha={:.4f}".format(
                t, elapsed, J, len(active_set), best_alpha))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW-N done ({:.1f}s): J={:.6f}, iters={}".format(elapsed, J, iter_count))
    return best_a

def targeted_coord_descent(a0, time_limit, verbose=False):

    a = norm_sum1(a0)

    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        # Find top peaks
        n_peaks = min(20, len(conv))
        top_k = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            # Find positions contributing to this peak
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            max_prod_idx = int(idx[np.argmax(products)])
            max_prod_val = products.max()
            if max_prod_val < 1e-20:
                continue

            # Score all positions by how much they contribute to top peaks
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            # Find safe destination (low score)
            safe_mask = scores < np.percentile(scores, 20)
            if not safe_mask.any():
                safe_mask = np.ones(n, dtype=bool)
            safe_indices = np.where(safe_mask)[0]
            if len(safe_indices) == 0:
                safe_indices = np.arange(n)
            
            # Pick destination with lowest score among safe
            dst = int(safe_indices[np.argmin(scores[safe_indices])])
            if dst == max_prod_idx:
                dst = int(safe_indices[np.argmin(scores[safe_indices])])
                if dst == max_prod_idx:
                    continue

            # Try moving mass
            src_val = a[max_prod_idx]
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:

                if amt < 1e-15:

                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 100 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord ({:.1f}s): J={:.6f}, stall={}".format(elapsed, 2.0 * n * best_m, stall))

    return best_a


def parametric_init(n, parent=None):
    """Generate diverse parametric initializations and return the best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Sine powers
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1(np.sin(np.pi * (i_arr + 0.5) / n) ** p))

    # Raised cosine
    for p in [1.0, 2.0, 3.0]:
        candidates.append(norm_sum1((0.5 * (1 + np.cos(2 * np.pi * (i_arr + 0.5) / n))) ** p))

    # Power laws
    for p in [0.3, 0.5, 1.0, 2.0]:
        candidates.append(norm_sum1((i_arr + 1) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    # Bimodal
    for c in [0.25, 0.35, 0.45]:
        for s in [0.05, 0.1, 0.15, 0.25]:
            a = np.exp(-((x - c) ** 2) / (2 * s * s)) + np.exp(-((x - (1 - c)) ** 2) / (2 * s * s))
            candidates.append(norm_sum1(a))

    # Single Gaussian
    for c in [0.3, 0.5, 0.7]:
        for s in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2 * s * s))))

    # Cosine modulation
    for freq in [1, 2, 3]:
        candidates.append(norm_sum1(np.maximum(1 + 0.5 * np.cos(2 * np.pi * freq * x), 0.01)))

    # If we have a parent, add it
    if parent is not None and len(parent) == n:
        candidates.append(norm_sum1(parent))

    # Evaluate all
    best_a = candidates[0]
    best_m = fast_conv(best_a).max()
    for a in candidates:
        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a

    return best_a


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def iterated_local_search(a0, time_limit, n_restarts=5, verbose=False):
    """Iterated local search with random perturbations."""
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_conv(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb
        noise_level = 0.05 / (1.0 + 0.3 * r)
        pert = rng.normal(0, noise_level, n) * best_a
        a_r = best_a + pert
        a_r = np.maximum(a_r, 0)
        s_r = a_r.sum()
        if s_r < 1e-15:
            a_r = best_a.copy()
        else:
            a_r = a_r / s_r

        # Quick local optimization (short LP-FW)
        t_per = min(60.0, (time_limit - (time.monotonic() - t0)) / max(1, n_restarts - r))
        a_opt = lp_fw_nesterov(a_r, t_per, initial_K=200, max_K=1500, verbose=False)
        
        m_opt = fast_conv(a_opt).max()
        if m_opt < best_m - 1e-12:
            best_m = m_opt
            best_a = a_opt.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print("    ILS restart {}: J={:.6f} ({:.1f}s)".format(r, 2.0 * n * best_m, elapsed))

    return best_a


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total time budget: {:.1f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        best_seq = parametric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={:.6f}".format(len(best_seq), best_score))

    def try_update(bseq, bscore, a, label):
        if a is None:
            return bseq, bscore
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return bseq, bscore
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < bscore:
            elapsed = time.monotonic() - t_global
            print("*** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
            return a_norm, sc
        return bseq, bscore

    n0 = len(best_seq)

    # Phase 1: Parametric refinement
    print("\n=== Phase 1: Parametric init (n={}) ===".format(n0))
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 5:
        a_param = parametric_init(n0, parent)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param")

    # Phase 2: LP-FW with Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW Nesterov (n={}) ===".format(n_best))
    t2 = min(remaining * 0.28, 400.0)
    if t2 > 30:
        a_out = lp_fw_nesterov(norm_sum1(best_seq), t2, initial_K=500, max_K=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n{}".format(n_best))

    # Phase 3: Smoothed L-BFGS
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Smoothed L-BFGS (n={}) ===".format(n_best))
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        try:
            a_out = smoothed_lbfgs(norm_sum1(best_seq), t3, betas=[10, 30, 80, 150], verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_n{}".format(n_best))
        except Exception as e:
            print("    L-BFGS error: {}".format(str(e)))

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Coord descent (n={}) ===".format(n_best))
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_out = targeted_coord_descent(norm_sum1(best_seq), t4, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n{}".format(n_best))

    # Phase 5: Iterated local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: ILS (n={}) ===".format(n_best))
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_out = iterated_local_search(norm_sum1(best_seq), t5, n_restarts=4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ils_n{}".format(n_best))

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 9000]

    for new_n in longer_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n=== Phase 6: Scale to n={} ===".format(new_n))
        a_long = interpolate_seq(best_seq, new_n)
        best_seq, best_score = try_update(best_seq, best_score, a_long, "interp_n{}".format(new_n))

        t6a = min(remaining * 0.35, 200.0)
        if t6a > 30:
            a_out = lp_fw_nesterov(a_long, t6a, initial_K=300, max_K=2000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_long_n{}".format(new_n))

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.3, 100.0)
            if t6b > 10:
                a_out = targeted_coord_descent(norm_sum1(best_seq), t6b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_long_n{}".format(new_n))

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final polish (n={}) ===".format(n_best))
    if remaining > 60:
        t7 = min(remaining * 0.5, 250.0)
        if t7 > 30:
            a_out = lp_fw_nesterov(norm_sum1(best_seq), t7, initial_K=600, max_K=4000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_n{}".format(n_best))

    # Phase 8: Final coord
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.6, 100.0)
        if t8 > 10:
            a_out = targeted_coord_descent(norm_sum1(best_seq), t8, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_coord")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = parametric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score={:.6f}".format(best_score))

    return [float(x) for x in best_seq]
