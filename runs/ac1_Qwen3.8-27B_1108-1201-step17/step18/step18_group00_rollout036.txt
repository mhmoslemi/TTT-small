import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_max_gradient(a, T, fft_size=None):
    """Compute smooth max approximation and its gradient."""
    n = len(a)
    if fft_size is None:
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
    
    c = fast_autocorr(a)
    # Smooth max
    c_scaled = c / T
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    log_sum = c_max + np.log(exp_c.sum())
    f_smooth = T * log_sum
    p = exp_c / exp_c.sum()  # softmax weights
    
    # Gradient: grad_i = 2 * sum_k p[k] * a[k-i]
    # = 2 * cross_corr(a, p)[i] = 2 * IFFT(FFT(p) * conj(FFT(a)))[:n]
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    grad = 2.0 * np.fft.irfft(np.conj(A) * P, fft_size)[:n]
    
    return f_smooth, grad, p, c

def mirror_descent_smooth(a0, time_limit, T_start=1.0, T_end=0.01, 
                          c0=1.0, verbose=True):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    
    # FFT size for gradient computation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    t0 = time.monotonic()
    a_prev = a.copy()
    
    # Total iterations (estimate)
    n_iters = 50000
    t = 0
    stall = 0
    
    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Temperature schedule (geometric cooling)
        progress = min(t / n_iters, 1.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov lookahead
        beta = 2.0 / (t + 2.0)
        if t == 1:
            y = a.copy()
        else:
            y = a + (1.0 - beta) / beta * (a - a_prev)
            y = np.maximum(y, 1e-30)
            y = y / y.sum()
        
        # Compute smooth objective and gradient at y
        f_smooth, grad, p, c_y = smooth_max_gradient(y, T, fft_size)

        M_y = c_y.max()
        if M_y < best_obj - 1e-15:

            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Step size: eta = c0 / sqrt(t)
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent step (exponential map)
        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a_prev = a
        a = a_new

            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_obj
            print("  MSD t=" + str(t) + " (" + str(elapsed) + "s): obj=" + repr(obj) + ", T=" + repr(T))

import numpy as np

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  LP solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def get_fft_size(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(elapsed) + "s)")
    return best_seq, best_score


def smooth_mirror_descent(a0, time_limit, T_start_tau=1.0, T_end_tau=0.0005,
                          c0=1.5, n_max_iters=300000, verbose=True, rng_seed=42):
    """Mirror descent with log-sum-exp smooth max, Nesterov momentum, adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-20)
    a = a / a.sum()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    fft_size = get_fft_size(n)
    t0 = time.monotonic()
    a_prev = a.copy()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 8000
    restart_count = 0

    for t in range(1, n_max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            restart_count += 1
            if restart_count > 20:
                break
            noise_level = 0.02 / (1.0 + 0.1 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r.copy()
            a_prev = a_r.copy()
            stall = 0
            c0 = max(c0 * 0.95, 0.3)
            if verbose:
                elapsed = time.monotonic() - t0
                obj = 2.0 * n * best_m
                print("    MSD restart " + str(restart_count) + " (" + str(elapsed) + "s): obj=" + repr(obj))
            continue

        progress = min((t - 1) / max(n_max_iters - 1, 1), 1.0)
        tau = T_start_tau * (T_end_tau / T_start_tau) ** progress
        T = tau / n

        beta = 2.0 / (t + 2.0)
        if t == 1 or (t == restart_count * max_stall + 1 if restart_count > 0 else False):
            y = a.copy()
        else:
            y = a + ((1.0 - beta) / beta) * (a - a_prev)
            y = np.maximum(y, 1e-20)
            y = y / y.sum()

        c_y = fast_autocorr(y)
        M_y = c_y.max()

        if M_y < best_m - 1e-14:
            best_m = M_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        c_scaled = c_y / T
        c_max_val = c_scaled.max()
        exp_c = np.exp(c_scaled - c_max_val)
        p = exp_c / exp_c.sum()

        A_fft = np.fft.rfft(y, fft_size)
        P_fft = np.fft.rfft(p, fft_size)
        grad = 2.0 * np.fft.irfft(np.conj(A_fft) * P_fft, fft_size)[:n]

        grad_scale = np.abs(grad).max()
        if grad_scale < 1e-30:
            grad_scale = 1.0

        eta = c0 / (np.sqrt(t) + 1.0)
        eta = min(eta, 0.5 / (grad_scale + 1e-30))

        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        a_prev = a
        a = a_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_m
            print("    MSD t=" + str(t) + " (" + str(elapsed) + "s): obj=" + repr(obj) + ", tau=" + repr(tau) + ", stall=" + str(stall))

    elapsed = time.monotonic() - t0
    obj = 2.0 * n * best_m
    if verbose:
        print("    MSD done (" + str(elapsed) + "s): obj=" + repr(obj))
    return best_a, best_m


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw_improved(a0, time_limit, solver, initial_K=200, max_K=1500,
                   add_per_iter=20, verbose=True):
    """LP-FW with margin-based active set and bisection line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 1500
    iter_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: no improve for " + str(no_improve) + " iters")
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(constraint_list[i]) for i in range(min(initial_K, len(constraint_list))))
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = get_fft_size(n)
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Bisection line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = fast_autocorr(y).max()
        lo, hi = 0.0, 1.0
        for _ in range(12):
            mid = (lo + hi) / 2.0
            trial = (1.0 - mid) * y + mid * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                hi = mid
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M - 1e-15:
                best_trial_M = M_trial
                best_alpha = mid
                lo = mid
            else:
                hi = mid

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new
        iter_count += 1

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(elapsed) + "s): obj=" + repr(J) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(elapsed) + "s): obj=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    """Coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 800
    step_scales = [0.003, 0.007, 0.015, 0.03, 0.06, 0.12]
    fft_size = get_fft_size(n)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(40, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute impact score for each position
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 200:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 300 == 0:

def parametric_init(n, best_init=None, best_score=float('inf')):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy() if best_init is not None else np.ones(n) / n
    best_val = best_score

    candidates = []

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    for alpha in [0.5, 1.0, 1.5]:
        for beta in [0.5, 1.0, 1.5]:
            candidates.append(norm_sum1((i_arr + 1.0) ** alpha * (n - i_arr) ** beta))

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.15, 0.25, 0.4]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))))

    for p in [1.0, 2.0, 3.0]:
        candidates.append(norm_sum1(x ** p + (1.0 - x) ** p))

    for freq in [1, 2, 3]:
        candidates.append(norm_sum1(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01)))

    for rate in [1, 2, 5]:
        candidates.append(norm_sum1(np.exp(-rate * x) + np.exp(rate * (x - 1.0))))

    rng = np.random.default_rng(99)
    for _ in range(40):
        candidates.append(norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0)))

    for a in candidates:
        v = fast_autocorr(a).max()
        if v < best_val:
            best_val = v
            best_a = a.copy()

    return best_a, best_val

    print("\n=== Phase 3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40 and n_best <= 8000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t3, solver,
                                  initial_K=200, max_K=1500,
                                  add_per_iter=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(n_best), t_global)

    # Phase 4: Second mirror descent round (with different temperature)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Smooth Mirror Descent round 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 40:
        a_out, _ = smooth_mirror_descent(norm_sum1(best_seq), t4,
                                         T_start_tau=0.3, T_end_tau=0.0001,
                                         c0=1.0, n_max_iters=150000,
                                         verbose=True, rng_seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "msd2_n" + str(n_best), t_global)

    # Phase 5: Try slightly different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length exploration ===")
    test_lengths = [n_best - 300, n_best + 300, n_best + 600]
    test_lengths = [l for l in test_lengths if 3000 <= l <= 9000]

    for new_n in test_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.30, 150.0)
        if t5a > 40:
            a_out, _ = smooth_mirror_descent(a_long, t5a,
                                             T_start_tau=0.5, T_end_tau=0.0003,
                                             c0=1.2, n_max_iters=100000,
                                             verbose=False, rng_seed=77)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "msd_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 7500:
            t5b = min(remaining * 0.30, 150.0)
            if t5b > 40:
                a_out, _ = lp_fw_improved(norm_sum1(best_seq), t5b, solver,
                                          initial_K=200, max_K=1200,
                                          add_per_iter=15, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # Phase 6: Final LP-FW on best found
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 80 and n_best <= 8000:
        t6 = min(remaining * 0.30, 250.0)
        if t6 > 40:
            a_out, _ = lp_fw_improved(norm_sum1(best_seq), t6, solver,
                                      initial_K=300, max_K=2000,
                                      add_per_iter=25, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final coordinate polish (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t7 = min(remaining * 0.50, 200.0)
        if t7 > 20:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t7, rng_seed=314, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 8: One more MSD round with very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final MSD (low T) ===")
    if remaining > 30:
        t8 = min(remaining * 0.60, 150.0)
        if t8 > 20:
            a_out, _ = smooth_mirror_descent(norm_sum1(best_seq), t8,
                                             T_start_tau=0.05, T_end_tau=0.00005,
                                             c0=0.5, n_max_iters=100000,
                                             verbose=False, rng_seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "msd_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(elapsed) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
