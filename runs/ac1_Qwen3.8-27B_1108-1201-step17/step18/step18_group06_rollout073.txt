        # Periodic perturbation to escape local minima
        if no_improve > 500 and step % 200 == 0:
            strength = min(0.02, 0.005 * (1 + no_improve / 1000.0))
            pert = rng.normal(0, strength, n)
            a_p = normalize1(a + pert * a)
            J_p = 2.0 * n * fast_autocorr(a_p).max()
            if J_p < best_J:
                best_a = a_p.copy()
                best_J = J_p
                a = a_p
                x = np.log(a + 1e-30)
                cur_lr = lr * 0.5
            no_improve = 0

        if verbose and step % 200 == 0:
            el = time.monotonic() - t0
            print(f"    ADAM s={step} ({el:.0f}s): J={best_J:.6f} M={M:.8f} b={beta_cur:.1f} lr={cur_lr:.6f} ni={no_improve}")

    return best_a, best_J


def peak_shave(a0, time_limit, n_top=5, verbose=False, rng_seed=0):
    """Direct coordinate moves to reduce top convolution peaks."""
    n = len(a0)
    a = normalize1(a0).copy()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Get top peaks
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for ki in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(ki)
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Find destination with low contribution to top peaks
            cost = np.zeros(n)
            for k2 in top_k:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                v2 = np.arange(lo2, hi2 + 1)
                if len(v2) > 0:
                    cost[v2] += a[k2i - v2]

            # Pick low-cost, low-value destination
            order = np.argsort(cost + 5.0 * a)[:max(50, n // 4)]
            dst = order[np.argmin(a[order])]
            if dst == src:
                dst = int(rng.integers(n))
                if dst == src:
                    continue

            found = False
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-14:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = normalize1(a_t)
                M_t = fast_autocorr(a_t).max()
                if M_t < M * (1 - 1e-8):
                    a = a_t
                    improved = True
                    found = True
                    break

        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.005, 0.04)
                a[i] -= amt
                a[j] += amt
                a = normalize1(a)
                M_new = fast_autocorr(a).max()
                if M_new < fast_autocorr(best_a).max():
                    pass  # will update on next loop

    return best_a, best_M


def spectral_refine(a0, time_limit, alpha0=0.3, n_iters_cap=3000, verbose=False):
    """Gerchberg-Saxton spectral flattening with nonnegativity projection."""
    n = len(a0)
    a = normalize1(a0).copy()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters_cap):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        alpha = alpha0 + (1.0 - alpha0) * (it / max(1, n_iters_cap))
        # Flattening: pull P toward its mean
        ratio = P / np.maximum(P_mean, 1e-30)
        P_target = P_mean * np.power(ratio, 1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_ratio = np.sqrt(P_target / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.8, 1.25)
        A_new = A * mag_ratio

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped blend
            blend = 0.2
            a_trial = normalize1((1 - blend) * a + blend * a_new)
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_M:
                best_M = M_trial
                best_a = a_trial.copy()
                a = a_trial

    return best_a, best_M


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.full(new_n, 1.0 / new_n)
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.full(new_n, 1.0 / new_n)
    return fi / ss


def propose_candidate(budget_s=1800.0):
    global height_sequence_1
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = eval_J(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            el = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={el:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            ps = eval_J(parent)
            print(f"Parent: n={len(parent)}, score={ps:.6f}")
            best_seq = normalize1(parent)
            best_score = ps
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_ramp(4000, p=1.0)
        best_score = eval_J(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 0: Quick evaluation of parametric families at parent length ===
    n0 = len(best_seq)
    print(f"\n=== Phase 0: Parametric eval (n={n0}) ===")
    candidates = []
    for p_val in [0.7, 0.8, 0.9, 1.0, 1.1, 1.3, 1.5]:
        candidates.append((f"ramp_p{p_val}", make_ramp(n0, p=p_val)))
    candidates.append(("cos_ramp", make_cos_ramp(n0, alpha=0.4)))
    candidates.append(("exp_ramp1.5", make_exp_ramp(n0, rate=1.5)))
    candidates.append(("exp_ramp2.0", make_exp_ramp(n0, rate=2.0)))

    for lbl, cand in candidates:
        try_update(cand, f"p0_{lbl}")

    # === Phase 1: Adam on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam (n={len(best_seq)}) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_in = normalize1(best_seq)
        a_out, _ = Adam_optimize(a_in, t1, beta_start=5.0, beta_end=30.0,
                                  lr=0.004, verbose=True, rng_seed=42)
        try_update(a_out, "adam1")

    # === Phase 2: Peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Peak shave (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_in = normalize1(best_seq)
        a_out, _ = peak_shave(a_in, t2, n_top=5, verbose=False, rng_seed=1)
        try_update(a_out, "peak1")

    # === Phase 3: Spectral refine ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = normalize1(best_seq)
        a_out, _ = spectral_refine(a_in, t3, alpha0=0.3, verbose=False)
        try_update(a_out, "spectral1")

    # === Phase 4: Adam with different params on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Adam v2 (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 20:
        a_in = normalize1(best_seq)
        a_out, _ = Adam_optimize(a_in, t4, beta_start=10.0, beta_end=40.0,
                                  lr=0.003, verbose=True, rng_seed=77)
        try_update(a_out, "adam2")

    # === Phase 5: Peak shave again ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Peak shave v2 ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = normalize1(best_seq)
        a_out, _ = peak_shave(a_in, t5, n_top=3, verbose=False, rng_seed=123)
        try_update(a_out, "peak2")

    # === Phase 6: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length (base n={n_best}) ===")
    for delta in [500, -500, 1000, -1000, 2000]:
        new_n = n_best + delta
        if new_n < 2000 or new_n > 12000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_seq(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        sub_t = min(remaining * 0.30, 200.0)
        if sub_t > 15:
            a_out, _ = Adam_optimize(a_long, sub_t, beta_start=5.0, beta_end=25.0,
                                      lr=0.004, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"adam_n{new_n}")

    # === Phase 7: Final Adam polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Adam (n={n_best}) ===")
    t7 = min(remaining * 0.40, 300.0)
    if t7 > 15:
        a_in = normalize1(best_seq)
        a_out, _ = Adam_optimize(a_in, t7, beta_start=8.0, beta_end=35.0,
                                  lr=0.002, verbose=True, rng_seed=999)
        try_update(a_out, "adam_final")

    # === Phase 8: Final peak shave + spectral ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8a = min(remaining * 0.45, 150.0)
        t8b = remaining * 0.25
        print(f"\n=== Phase 8: Final combined (n={n_best}) ===")
        if t8a > 10:
            a_in = normalize1(best_seq)
            a_out, _ = peak_shave(a_in, t8a, n_top=4, rng_seed=555)
            try_update(a_out, "peak_final")
        if t8b > 10:
            a_in = normalize1(best_seq)
            a_out, _ = spectral_refine(a_in, t8b, alpha0=0.4, n_iters_cap=2000)
            try_update(a_out, "spectral_final")
            # Quick adam after spectral
            rem2 = time_budget - (time.monotonic() - t_global)
            if rem2 > 10:
                a_out2, _ = Adam_optimize(normalize1(best_seq), rem2 * 0.7,
                                           beta_start=10.0, beta_end=40.0,
                                           lr=0.002, verbose=False, rng_seed=888)
                try_update(a_out2, "adam_post_spec")

    # Final validation
    if best_seq is not None:
        fs = eval_J(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        best_seq = make_ramp(4000, 1.0)
        best_score = eval_J(best_seq)

    return [float(x) for x in best_seq]
