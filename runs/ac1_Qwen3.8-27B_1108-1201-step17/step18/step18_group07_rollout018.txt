def fast_conv(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def logsumexp_obj(a, beta):

    shifted = beta * (c - max_c)

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def logsumexp_obj(a, beta):
    """Compute log-sum-exp approximation of max(conv(a,a))."""
    c = fast_conv(a)
    max_c = c.max()
    if max_c < 1e-30:
        return 0.0
    shifted = beta * (c - max_c)
    shifted = np.maximum(shifted, -500.0)
    log_sum = np.log(np.exp(shifted).sum())
    return max_c + log_sum / beta


def compute_grad_lse(a, c, beta, top_k=30):
    """Compute subgradient of log-sum-exp objective.
    
    d/d(a[j]) c[k] = 2*a[k-j] if 0 <= k-j < n, else 0.
    Gradient of LSE is weighted average of individual gradients.
    """
    n = len(a)
    k_max = min(top_k, len(c))
    order = np.argsort(c)[::-1][:k_max]
    vals = c[order]
    max_v = vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (vals - max_v) / max_v
        exponents = np.maximum(exponents, -500.0)
        w = np.exp(exponents)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / ws

    grad = np.zeros(n)
    for wi, k_idx in zip(w, order):
        if wi < 1e-14:
            continue
        k = int(k_idx)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start >= j_end:
            continue
        j_range = np.arange(j_start, j_end)
        a_idx = k - j_range
        grad[j_start:j_end] += 2.0 * wi * a[a_idx]
    return grad


def mirror_descent_step(a, grad, eta):
    """Mirror descent step on the simplex (entropy-regularized).
    a_new[i] = a[i] * exp(-eta * g[i]) / normalizer
    """
    log_ratio = -eta * grad
    # Numerical stability
    lr_max = log_ratio.max()
    if lr_max > 500:
        log_ratio = log_ratio - lr_max
    log_ratio = np.maximum(log_ratio, -500.0)
    a_new = a * np.exp(log_ratio)
    a_new = np.maximum(a_new, 1e-300)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def backtracking_mirror_descent(a, beta, top_k, max_iter, time_limit, t0,
                                 eta_init=0.05, rho=0.5, c_armijo=1e-4):
    """Mirror descent with backtracking line search on log-sum-exp objective."""
    n = len(a)
    a_cur = a.copy()
    f_cur = logsumexp_obj(a_cur, beta)
    best_a = a_cur.copy()
    best_f = f_cur
    eta = eta_init

        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(a_cur)
        g = compute_grad_lse(a_cur, c, beta, top_k=top_k)

        # Backtracking line search
        accepted = False
        eta_trial = eta
        for bt in range(20):
            a_trial = mirror_descent_step(a_cur, g, eta_trial)
            f_trial = logsumexp_obj(a_trial, beta)

            # Armijo condition

            # Simpler: just check if objective decreased

                break

        f_cur = f_trial

def spectral_rebalance(a, strength=0.3, iterations=5):
    """Spectral rebalancing: reduce dynamic range of |FFT(a)|.
    More aggressive than simple flattening - targets the top frequency bins.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    for _ in range(iterations):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        mag_mean = mag.mean()
        if mag_mean < 1e-30:
            break

        ratio = mag / (mag_mean + 1e-30)
        # Compress: raise low, lower high
        mag_new = mag_mean * np.power(ratio, 1.0 - strength)
        mag_new = np.maximum(mag_new, 1e-30)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Blend
        a = (1.0 - strength * 0.4) * a + strength * 0.4 * a_new
        a = norm_simplex(a)

    return a


def spectral_targeted(a, c, num_targets=5, strength=0.2):
    """Targeted spectral correction: reduce peaks at specific time-domain positions.
    
    Identifies which frequency components contribute to the worst peaks and
    selectively attenuates them.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    # For each target peak position k*, the contribution from frequency f is
    # |A(f)|^2 * exp(2j*pi*f*k*/fft_size). We want to reduce this sum.
    # Simple approach: for peaks above threshold, reduce |A| at frequencies
    # where the phase aligns constructively with the peak position.
    peak_threshold = c.max() * 0.9
    peak_positions = np.where(c >= peak_threshold)[0]
    peak_positions = peak_positions[:num_targets]

    if len(peak_positions) == 0:
        return a

    # Average phase alignment for the peak positions
    freqs = np.fft.rfftfreq(fft_size)
    phase_sum = np.zeros(len(A))
    for k in peak_positions:
        phase_sum += np.cos(2 * np.pi * freqs * int(k))

    # Reduce amplitude where phase_sum is positive (constructive for peaks)
    attenuation = 1.0 - strength * 0.5 * np.maximum(phase_sum / len(peak_positions), 0)
    attenuation = np.maximum(attenuation, 0.5)

    A_new = A * attenuation
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

    return norm_simplex(0.7 * a + 0.3 * a_new)

    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2)) / (2 * width ** 2)
    a += 0.05
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_golden(n):
    """Based on golden ratio distribution - low discrepancy sequence."""
    phi = (1 + np.sqrt(5)) / 2
    x = np.arange(n) * (1.0 / phi) % 1.0
    # Create a smooth bump centered at the golden ratio positions
    a = np.zeros(n)
    for pos in x[:min(n, 20)]:
        a += np.exp(-0.5 * ((x - pos) / 0.05) ** 2)
    a += 0.01
    return norm_simplex(a)


def make_init_rand_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def optimize_single(a0, time_limit, t0, verbose=False):
    """Full optimization pipeline on a single sequence.
    
    Uses annealed log-sum-exp mirror descent with periodic spectral rebalancing.
    """
    n = len(a0)
    a = norm_simplex(a0)
    c = fast_conv(a)
    best_a = a.copy()
    best_m = c.max()
    best_obj = obj_from_conv(c, n)

    if verbose:
        print(f"    Start: n={n}, M={best_m:.8f}, obj={best_obj:.6f}")

    # Annealed beta schedule
    beta_schedule = [10, 30, 100, 300, 600, 1000, 2000]
    iters_per_phase = 2000
    eta_start = 0.03

    for phase, beta in enumerate(beta_schedule):
        if time.monotonic() - t0 > time_limit:
            break

        # Adjust eta based on beta (smaller eta for larger beta)
        eta = eta_start / (1.0 + 0.1 * (beta / 100.0))

        top_k = min(20 + 5 * phase, 50)

        a_new, f_new = backtracking_mirror_descent(
            a, beta, top_k, max_iter=iters_per_phase,
            time_limit=time_limit, t0=t0, eta_init=eta
        )

        c_new = fast_conv(a_new)
        M_new = c_new.max()

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            best_obj = obj_from_conv(c_new, n)
            a = a_new
            if verbose:
                print(f"    Phase {phase} (beta={beta}): M={best_m:.8f}, obj={best_obj:.6f}")
        else:
            a = a_new  # still use the improved smoothed version

        # Periodic spectral rebalancing
        if time.monotonic() - t0 < time_limit:
            a_spec = spectral_rebalance(a, strength=0.15 + 0.05 * phase, iterations=3)
            c_spec = fast_conv(a_spec)
            if c_spec.max() < best_m - 1e-15:
                best_m = c_spec.max()
                best_a = a_spec.copy()
                best_obj = obj_from_conv(c_spec, n)
                a = a_spec
                if verbose:
                    print(f"    Spectral: M={best_m:.8f}, obj={best_obj:.6f}")

        # Targeted spectral correction
        if time.monotonic() - t0 < time_limit:
            c_curr = fast_conv(a)
            a_targ = spectral_targeted(a, c_curr, num_targets=3, strength=0.15)
            c_targ = fast_conv(a_targ)
            if c_targ.max() < best_m - 1e-15:
                best_m = c_targ.max()
                best_a = a_targ.copy()
                best_obj = obj_from_conv(c_targ, n)
                a = a_targ

    return best_a, best_obj


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_init_cosine(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Optimize parent directly with log-sum-exp ===
    print(f"\n=== Phase 1: Log-sum-exp on parent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = optimize_single(a_in, t1, t_global, verbose=True)
        try_update(a_out, f"lse_parent_n{n0}")

    # === Phase 2: Try diverse initializations at parent length ===
    print(f"\n=== Phase 2: Diverse inits (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2_total = min(remaining * 0.25, 400.0)
    if t2_total > 30:
        inits = [
            (make_init_bowl(n0, 2.5), "bowl2.5"),
            (make_init_bowl(n0, 3.5), "bowl3.5"),
            (make_init_dual_peak(n0, 0.2, 0.8, 0.15), "dual0.2/0.8"),
            (make_init_dual_peak(n0, 0.15, 0.85, 0.1), "dual0.15/0.85"),
            (make_init_sinc(n0, 8.0), "sinc8"),
            (make_init_sinc(n0, 12.0), "sinc12"),
            (make_init_rand_smooth(n0, seed=7), "rand7"),
            (make_init_rand_smooth(n0, seed=13), "rand13"),
            (make_init_cosine(n0, mode=2), "cos2"),
        ]
        t_per = t2_total / len(inits)
        for init, lbl in inits:
            remaining_now = time_budget - (time.monotonic() - t_global)
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 10:
                break
            a_out, _ = optimize_single(init, t_avail, t_global, verbose=False)
            try_update(a_out, f"lse_{lbl}_n{n0}")

    # === Phase 3: Try different lengths ===
    print(f"\n=== Phase 3: Length variation ===")
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t3_total = min(remaining * 0.20, 300.0)

    candidate_lengths = []
    for delta in [500, 1000, -500, -1000, 2000, -2000]:
        nn = n_best + delta
        if 1500 <= nn <= 10000:
            candidate_lengths.append(nn)

    if t3_total > 30 and candidate_lengths:
        # Deduplicate and sort by proximity to n_best
        candidate_lengths = list(dict.fromkeys(candidate_lengths))
        candidate_lengths.sort(key=lambda x: abs(x - n_best))
        candidate_lengths = candidate_lengths[:4]

        t_per = t3_total / len(candidate_lengths)
        for new_n in candidate_lengths:
            remaining_now = time_budget - (time.monotonic() - t_global)
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 10:
                break
            print(f"  Trying n={new_n}...")
            a_interp = interpolate_sequence(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")

            a_out, _ = optimize_single(a_interp, t_avail, t_global, verbose=False)
            try_update(a_out, f"lse_n{new_n}")

    # === Phase 4: Intensive optimization on current best ===
    print(f"\n=== Phase 4: Intensive LSE on best ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = optimize_single(a_in, t4, t_global, verbose=True)
        try_update(a_out, f"intensive_n{n_best}")

    # === Phase 5: Spectral polishing ===
    print(f"\n=== Phase 5: Spectral polishing ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        a_in = norm_simplex(best_seq)
        t5 = min(remaining * 0.15, 150.0)
        if t5 > 5:
            # Iterative spectral with decreasing strength
            a_cur = a_in.copy()
            for strength in [0.3, 0.2, 0.15, 0.1, 0.05, 0.03]:
                if time.monotonic() - t_global > time_budget - 30:
                    break
                a_cur = spectral_rebalance(a_cur, strength=strength, iterations=5)
                c_cur = fast_conv(a_cur)
                if c_cur.max() < fast_conv(best_seq).max():
                    try_update(a_cur, f"spec_s{strength}")

            # Targeted correction
            c_cur = fast_conv(a_cur)
            for tgt_n in [5, 3, 2]:
                a_t = spectral_targeted(a_cur, c_cur, num_targets=tgt_n, strength=0.1)
                try_update(a_t, f"spec_targ{tgt_n}")
                c_cur = fast_conv(a_t)

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        a_in = norm_simplex(best_seq)
        t6 = min(remaining * 0.25, 200.0)
        if t6 > 5:
            a_out, _ = coordinate_descent_peaks(a_in, t6, t_global, max_iter=100000)
            try_update(a_out, f"cd_n{n_best}")

            # Another round with perturbation
            remaining_now = time_budget - (time.monotonic() - t_global)
            if remaining_now > 10:
                rng = np.random.default_rng(2024)
                a_pert = norm_simplex(a_out + rng.normal(0, 0.005, n_best))
                t6b = min(remaining_now * 0.5, 100.0)
                if t6b > 5:
                    a_out2, _ = coordinate_descent_peaks(a_pert, t6b, t_global, max_iter=50000)
                    try_update(a_out2, f"cd_pert_n{n_best}")

    # === Phase 7: Very high beta refinement ===
    print(f"\n=== Phase 7: High-beta refinement ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        a_in = norm_simplex(best_seq)
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 5:
            # Direct high-beta optimization
            a_cur = a_in.copy()
            for beta in [2000, 5000, 10000]:
                if time.monotonic() - t_global > time_budget - 20:
                    break
                g = compute_grad_lse(a_cur, fast_conv(a_cur), beta, top_k=10)
                eta = 0.005 / (1.0 + beta / 5000.0)
                for step in range(200):
                    if time.monotonic() - t_global > time_budget - 20:
                        break
                    a_new = mirror_descent_step(a_cur, g, eta)
                    c_new = fast_conv(a_new)
                    if c_new.max() < fast_conv(a_cur).max() - 1e-15:
                        a_cur = a_new
                        try_update(a_cur, f"hb_beta{beta}")
                        g = compute_grad_lse(a_cur, fast_conv(a_cur), beta, top_k=10)
                    else:
                        eta *= 0.7

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_cosine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
