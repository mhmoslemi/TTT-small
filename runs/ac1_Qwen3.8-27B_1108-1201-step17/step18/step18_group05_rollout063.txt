def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_gradient_topk(a, ac, top_k, weights=None):
    """Compute gradient of max(ac) approximation using top-k lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    if weights is None:
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 5.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    else:
        w = weights[:k_max]
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_gradient_optimize(a0, time_limit, top_k=80, c0=3.0, beta_mom=0.9,
                                tau_init=10.0, tau_final=500.0, eta_init=1.0,
                                restart_every=5000, rng_seed=42, verbose=False):
    """Improved Nesterov accelerated gradient with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    v = np.zeros(n)  # momentum
    eta = eta_init
    stall_count = 0
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        progress = min(1.0, n_iters / 20000.0)
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(x)
        ac_max = ac.max()

        # Softmax weights for active lags
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(np.minimum(ac_shifted, 50))
        p = exp_ac / exp_ac.sum()

        # Compute gradient using top-k approximation
        k_max = min(top_k, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_p = p[top_idx]
        p_sum = top_p.sum()
        if p_sum > 1e-30:
            top_p = top_p / p_sum
        else:
            top_p = np.ones(k_max) / k_max

        grad = np.zeros(n)
        for wi, k in zip(top_p, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        # Nesterov step
        v = beta_mom * v - eta * grad
        y = x + v
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Evaluate
        m_new = fast_autocorr(y).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = y.copy()
            stall_count = 0
            eta = min(eta * 1.01, 5.0)
        else:
            stall_count += 1
            if stall_count > 500:
                eta *= 0.8
            if stall_count > 2000:
                # Restart with perturbation
                pert = rng.normal(0, 0.02, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-10)
                x_r = x_r / x_r.sum()
                x = x_r
                v = np.zeros(n)
                stall_count = 0
                eta = eta_init
                top_k = min(top_k + 20, 200)
            else:
                x = y

        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NestGrad iter={n_iters} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, tau={tau:.1f}, stall={stall_count}")

    return best_a, best_m


def targeted_pair_reduction(a0, time_limit, n_top_lags=10, n_trials_per_lag=5,
                            rng_seed=789, verbose=False):
    """Targeted reduction of peak lags by moving mass from contributing pairs."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 50
    fractions = [0.001, 0.005, 0.01, 0.02, 0.003, 0.008, 0.015, 0.03, 0.05]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_lags = np.argsort(ac)[::-1][:n_top_lags]

        # Build sensitivity map: which positions appear in top lags
        sens = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            idx = np.arange(i_start, i_end + 1)
            sens[idx] += a[ki - idx] * 0.5  # Each position contributes half (paired)

        improved = False

        for k in top_lags:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[ki - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src_pos = idx[np.argmax(contrib)]
            src_val = a[src_pos]
            if src_val < 1e-12:
                continue

            # Find safe destinations (low sensitivity)
            safe_mask = sens < sens.mean()
            safe_positions = np.where(safe_mask)[0]
            if len(safe_positions) < 3:
                safe_positions = np.argsort(sens)[:n // 4]

            # Try a few destinations and fractions
            rng.shuffle(safe_positions)
            for dst in safe_positions[:n_trials_per_lag * 2]:
                if dst == src_pos or improved:
                    continue
                for frac in fractions:
                    amt = src_val * frac * (ac[ki] / max_contrib)
                    if amt < 1e-14 or amt > src_val * 0.5:
                        continue
                    a_trial = a.copy()
                    a_trial[src_pos] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            # Random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        else:
            stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    TPR ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    if solver is None:
        return a0.copy(), fast_autocorr(a0).max()

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = min(2 * n - 1, 3000)
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 15 == 0:
            K = min(K + 30, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def fine_tune(a0, time_limit, eta_init=0.05, rng_seed=999, verbose=False):
    """Very fine-tuning with small steps and high-temperature softmax."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    eta = eta_init
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        ac = fast_autocorr(a)
        ac_max = ac.max()
        # Use moderate temperature for fine-tuning
        tau = 50.0
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(np.minimum(ac_shifted, 50))
        p = exp_ac / exp_ac.sum()

        k_max = min(100, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_p = p[top_idx]
        p_sum = top_p.sum()
        if p_sum > 1e-30:
            top_p = top_p / p_sum
        else:
            top_p = np.ones(k_max) / k_max

        grad = np.zeros(n)
        for wi, k in zip(top_p, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        # Projected gradient step (no momentum for stability)
        a_trial = a - eta * (grad - grad.mean())
        a_new = project_simplex(a_trial)
        m_new = fast_autocorr(a_new).max()

        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.005, 0.5)
        else:
            stall += 1
            if stall > 200:
                eta *= 0.7
            if stall > 500:
                # Tiny perturbation
                if a.max() > 1e-6:
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * 0.001
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        m_p = fast_autocorr(a).max()
                        if m_p < best_m:
                            best_m = m_p
                            best_a = a.copy()
                            stall = 0
            else:
                a = a_new

    return best_a, best_m


def propose_candidate(budget_s=1800.0):

    t_global = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
