from scipy.sparse import csr_matrix

_SOLVER_CACHE = None


def _get_solver():

def _fft_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fsz = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsz)
    return np.fft.irfft(A * A, fsz)[:size]


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _interp_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f_norm)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return out / ss


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = _fft_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))


def _score_norm1(a):
    """Score assuming a is already normalized to sum=1."""
    n = len(a)
    ac = _fft_autocorr(a)
    return float(2.0 * n * ac.max())


def _parametric_init(n, t_limit, base_a, base_score):

    best_val = base_score
    i_arr = np.arange(n, dtype=np.float64)

    def _try(a, label):

        a = _normalize(a)
        v = _score_norm1(a)
        if v < best_val:

            best_a = a.copy()

    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:

        for beta in [0.5, 1.0, 1.5, 2.0]:

    for c in [0.25, 0.35, 0.5, 0.65, 0.75]:

        _try(np.exp(rate * x), f"er{rate}")

    for w in [0.3, 0.5, 0.7]:

        _try(w * bn + (1 - w) * lin, f"blend{w}")

            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_nest = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_nest = y_new + beta_t * (y_new - y)
            x_nest = np.maximum(x_nest, 0.0)
            sx = x_nest.sum()
            if sx < 1e-15:
                x_nest = y_new.copy()
            else:
                x_nest = x_nest / sx

        y = y_new
        iter_ct += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, avg_solve={solve_time_total/max(t,1):.2f}s")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_ct}")
    return best_a, best_m


def _fft_gradient(y, ac, beta, top_k=None):
    """Compute gradient of soft-max of autocorrelation using FFT."""
    n = len(y)
    size = 2 * n - 1
    if top_k is not None and top_k < size:
        top_idx = np.argsort(ac)[::-1][:top_k]
        p = np.zeros(size)
        vals = ac[top_idx]
        exp_vals = np.exp(beta * (vals - vals.max()))
        p[top_idx] = exp_vals
        p = p / p.sum()
    else:
        ac_shifted = ac - ac.max()
        exp_vals = np.exp(beta * ac_shifted)
        p = exp_vals / exp_vals.sum()

    # grad[j] = 2 * sum_{k=j}^{j+n-1} p[k] * y[k-j]
    # = 2 * sum_{l=0}^{n-1} p[j+l] * y[l]
    # Use FFT: result = conv(flip(y), p) at positions n-1, n, ..., 2n-2
    fsz = 1 << (size - 1).bit_length()
    y_flip = y[::-1]
    Yf = np.fft.rfft(y_flip, fsz)
    Pf = np.fft.rfft(p, fsz)
    conv_result = np.fft.irfft(Yf * Pf, fsz)[:size]
    # grad[j] = 2 * conv_result[n-1+j] for j=0..n-1
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def _amd_optimize(a0, t_limit, beta_start=3.0, beta_end=30.0, c0=3.0,
                  top_k_init=200, top_k_max=800, restart_every=3000, verbose=False):
    n = len(a0)
    x = _normalize(a0).copy()
    x_prev = x.copy()
    best_a = x.copy()
    best_m = _fft_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    stall = 0
    cur_top_k = top_k_init

    for t in range(1, 20000000):
        if time.monotonic() - t0 > t_limit:
            break

        # Adaptive beta: increase over time
        frac = min(1.0, t / max(1, int(t_limit * 2)))
        beta = beta_start + (beta_end - beta_start) * frac

        # Adaptive top_k: grow over time
        cur_top_k = min(top_k_init + int(frac * (top_k_max - top_k_init)), 2 * n - 1)

        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = _fft_autocorr(y)
        grad = _fft_gradient(y, ac, beta, top_k=cur_top_k)

        # Mirror descent step
        eta = c0 / (1.0 + np.sqrt(t) * 0.8)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = _fft_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert_type = int(rng.integers(4))
            if pert_type == 0:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.5, 2.5))
                pert_seq = _normalize((i_arr + 1.0) ** p)
            elif pert_type == 1:
                xa = np.linspace(0, 1, n)
                c = float(rng.uniform(0.3, 0.7))
                sig = float(rng.uniform(0.1, 0.35))
                pert_seq = _normalize(np.exp(-((xa - c) ** 2) / (2 * sig * sig)))
            elif pert_type == 2:
                xa = np.linspace(0, 1, n)
                c1 = float(rng.uniform(0.2, 0.4))
                c2 = float(rng.uniform(0.6, 0.8))
                sig = float(rng.uniform(0.08, 0.2))
                pert_seq = _normalize(np.exp(-((xa - c1) ** 2) / (2 * sig ** 2)) +
                                      np.exp(-((xa - c2) ** 2) / (2 * sig ** 2)))
            else:
                pert = rng.normal(0, 0.02, n) * best_a
                pert_seq = _normalize(best_a + pert)

            mix_w = float(rng.uniform(0.15, 0.45))
            x = _normalize((1 - mix_w) * best_a + mix_w * pert_seq)
            x_prev = x.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, tk={cur_top_k}")

    return best_a, best_m


def _coord_descent(a0, t_limit, rng_seed=42, verbose=False):
    n = len(a0)
    a = _normalize(a0).copy()
    best_a = a.copy()
    best_m = _fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fsz = 1 << (2 * n - 2).bit_length()

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < t_limit:
        ac = _fft_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0

        # Find top lags
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:15]:
            if time.monotonic() - t0 > t_limit:
                break
            k_i = int(k)
            js = max(0, k_i - n + 1)
            je = min(n - 1, k_i)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_i - idx]
            n_pairs = min(6, len(idx))
            pair_idx = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_idx:
                if time.monotonic() - t0 > t_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score destination positions
                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]

                scores[src] = np.inf
                # Avoid the peak region
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    scores[s2:e2 + 1] += 1.0

                # Find best destination (lowest score, non-trivial)
                valid = scores < np.inf
                if valid.sum() < 5:
                    valid = np.ones(n, dtype=bool)
                    valid[src] = False
                dst = int(np.argmin(np.where(valid, scores, np.inf)))

                if a[dst] > 0.4 * src_val:
                    continue

                for frac in [0.03, 0.06, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = _normalize(a_trial)
                    M_trial = _fft_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

        if not improved:
            stall += 1
            if stall > max_stall:
                break
            if stall % 15 == 0:
                n_moves = int(rng.integers(3, 12))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.005, 0.03))
                        a[i] -= a[i] * frac
                        a[j] += a[i] * frac * 0
                        # Redo properly
                        a = _normalize(a)
                        M_new = _fft_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                if verbose and stall % 150 == 0:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = _normalize(p)
            best_score = _score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 5000
        i_arr = np.arange(n_init, dtype=np.float64)
        best_seq = _normalize((i_arr + 1.0) ** 0.7)
        best_score = _score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def _check(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = _score(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric init (30s) ===
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 35.0)
    print(f"\n=== Phase 1: Parametric (n={n0}, {t1:.0f}s) ===")
    if t1 > 5:
        a_p, v_p = _parametric_init(n0, t1, best_seq, 2.0 * n0 * _fft_autocorr(best_seq).max())
        _check(a_p, "parametric")

    # === Phase 2: LP-Frank-Wolfe main (500s) ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 550.0)
    print(f"\n=== Phase 2: LP-FW (n={n_best}, {t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = _lp_frank_wolfe(best_seq, t2, solver,
                                    K_init=400, K_max=4500, verbose=True)
        _check(a_out, f"lp1_n{n_best}")

    # === Phase 3: AMD with FFT gradient (300s) ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.18, 300.0)
    print(f"\n=== Phase 3: AMD-FFT (n={n_best}, {t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = _amd_optimize(best_seq, t3, beta_start=3.0, beta_end=35.0,
                                 c0=4.0, top_k_init=200, top_k_max=700,
                                 restart_every=3000, verbose=True)
        _check(a_out, f"amd_n{n_best}")

    # === Phase 4: Coordinate descent (200s) ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.14, 200.0)
    print(f"\n=== Phase 4: CoordDescent (n={n_best}, {t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = _coord_descent(best_seq, t4, rng_seed=77, verbose=True)
        _check(a_out, f"coord_d_n{n_best}")

    # === Phase 5: LP-FW second round (200s) ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.14, 220.0)
    print(f"\n=== Phase 5: LP-FW round 2 (n={n_best}, {t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = _lp_frank_wolfe(best_seq, t5, solver,
                                    K_init=500, K_max=5000, verbose=True)
        _check(a_out, f"lp2_n{n_best}")

    # === Phase 6: Try different lengths ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length exploration ===")

    length_cands = []
    for d in [300, -300, 600, -600, 1000]:
        nn = n_best + d
        if 3000 <= nn <= 7000:
            length_cands.append(nn)
    for rn in [4000, 5000, 6000]:
        if rn != n_best and 3000 <= rn <= 7000:
            length_cands.append(rn)
    length_cands = sorted(set(length_cands))

    for nn in length_cands[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={nn} (remaining={remaining:.0f}s)")
        a_long = _interp_to_length(best_seq, nn)

        t6a = min(remaining * 0.45, 120.0)
        if t6a > 15:
            a_out, _ = _amd_optimize(a_long, t6a, beta_start=3.0, beta_end=25.0,
                                     c0=3.5, top_k_init=150, top_k_max=500,
                                     restart_every=2500, verbose=False)
            _check(a_out, f"amd_n{nn}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and nn <= 5500:
            t6b = min(remaining * 0.5, 130.0)
            if t6b > 25:
                a_out, _ = _lp_frank_wolfe(a_long, t6b, solver,
                                            K_init=350, K_max=3000, verbose=False)
                _check(a_out, f"lp_n{nn}")

    # === Phase 7: Final AMD polish ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.25, 150.0)
    print(f"\n=== Phase 7: Final AMD (n={n_best}, {t7:.0f}s) ===")
    if t7 > 10:
        a_out, _ = _amd_optimize(best_seq, t7, beta_start=10.0, beta_end=50.0,
                                 c0=2.5, top_k_init=400, top_k_max=1000,
                                 restart_every=5000, verbose=True)
        _check(a_out, f"amd_final_n{n_best}")

    # === Phase 8: Final coordinate descent ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.40, 120.0)
    print(f"\n=== Phase 8: Final Coord (n={n_best}, {t8:.0f}s) ===")
    if t8 > 10:
        a_out, _ = _coord_descent(best_seq, t8, rng_seed=314, verbose=True)
        _check(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Final quick LP ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.50, 100.0)
    if t9 > 20 and n_best <= 6000:
        print(f"\n=== Phase 9: Final LP (n={n_best}, {t9:.0f}s) ===")
        a_out, _ = _lp_frank_wolfe(best_seq, t9, solver,
                                    K_init=600, K_max=5000, verbose=False)
        _check(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        xa = np.linspace(0, 1, n_fb)
        best_seq = _normalize(np.sin(np.pi * xa) ** 2)
        best_score = _score(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
