```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a fast "soft mirror descent"
whose full subgradient is computed in O(n log n) via FFT (no top-k truncation).

Key methods
-----------
1. Soft mirror descent (a.k.a. exponential weights / entropy regularized):
   - c = conv(a,a) via FFT
   - s = softmax(c / (cmax * temperature))          (soft active set over ALL lags)
   - grad[i] = 2 * sum_m s[m] * a[m-i] computed as a single FFT convolution
   - Nesterov momentum + mirror (log/exp) step + simplex normalize
   - adaptive temperature annealing + periodic runtime restarts
2. Parametric shape initialisation (power law, shifted beta, exponential-quadratic,
   sine*exp) so the search is seeded in diverse basins.
3. Multi-length exploration (interpolating a good parent across a few lengths).
4. Aggressive coordinate-descent polish on the best result.

All helpers are top-level; no closures / lambdas. `evaluate_sequence` is assumed
to be imported and callable.
"""

import numpy as np
import time


# ---------------------------------------------------------------------------
# core kernels
# ---------------------------------------------------------------------------
def _next_pow2(m):
    return 1 << (max(1, m) - 1).bit_length()


def fast_conv(a):
    """Linear autocorrelation conv(a,a) via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fs = _next_pow2(size)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def fast_soft_grad(a, s):
    """grad[i] = 2 * sum_m s[m]*a[m-i]  (subgradient of soft-active max)."""
    a = np.asarray(a, dtype=np.float64)
    s = np.asarray(s, dtype=np.float64)
    n = len(a)
    size = len(s) + n - 1
    fs = _next_pow2(size)
    S = np.fft.rfft(s, fs)
    B = np.fft.rfft(a[::-1], fs)          # convolution with flipped a
    conv = np.fft.irfft(S * B, fs)
    return 2.0 * conv[n - 1: 2 * n - 1]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s <= 1e-9:
        return float("inf")
    a = a / s
    return float(evaluate_sequence([float(x) for x in a]))


# ---------------------------------------------------------------------------
# shape generators (all returned normalized to sum 1)
# ---------------------------------------------------------------------------
def shape_power(n, p):
    j = np.arange(n)
    x = (j + 1.0) / (n + 1.0)
    return normalize(x ** p)


def shape_beta(n, p, q):
    j = np.arange(n)
    x = (j + 1.0) / (n + 1.0)
    u = np.minimum(1.0, j / float(n))
    return normalize(x ** p * (1.0 - u) ** q)


def shape_expquad(n, p, q):
    j = np.arange(n)
    x = j / float(n)
    return normalize(np.exp(p * x - q * x * x))


def shape_sine_exp(n, p, alpha):
    j = np.arange(n)
    x = (j + 0.5) / float(n)
    return normalize(np.sin(np.pi * x) ** p * np.exp(-alpha * x))


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return out / ss


# ---------------------------------------------------------------------------
# main optimiser: soft mirror descent
# ---------------------------------------------------------------------------
def soft_mirror_descent(a0, time_limit, tparam_init=2.0, tparam_final=120.0,
                        c0=2.0, restart_stall=4000, seed=0, verbose=False):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    fail = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        y = 0.5 * (x + x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        c = fast_conv(y)
        cmax = c.max()
        if cmax < 1e-300:
            break

        # adaptive temperature (anneal -> concentrate on the true peaks)
        elapsed = time.monotonic() - t0
        prog = min(1.0, elapsed / max(time_limit, 1e-9))
        tparam = tparam_init * (tparam_final / tparam_init) ** prog

        # soft active set over all lags
        z = c / (cmax * tparam)
        z = z - z.max()
        s = np.exp(z)
        ss = s.sum()
        if ss < 1e-300:
            s = np.zeros_like(c)
            s[int(np.argmax(c))] = 1.0
        else:
            s = s / ss

        grad = fast_soft_grad(y, s)

        eta = c0 / (np.sqrt(elapsed * 60.0) + 1.0)
        eta = min(eta, 0.5)

        log_a = np.log(y) - eta * grad
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M = fast_conv(x).max()
        if M < best_M - 1e-16:
            best_M = M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if verbose and stall % 5000 == 0:
            print(f"    SMD n={n} ({elapsed:.0f}s): J={2*n*best_M:.6f} tparam={tparam:.1f} stall={stall}")

        if stall > restart_stall:
            # restarted runtime from best with a gentle spectral perturbation
            noise = rng.normal(0.0, 0.03, n) * best_a
            xr = np.maximum(best_a + noise, 1e-12)
            xr = xr / xr.sum()
            x = xr
            x_prev = xr.copy()
            stall = 0
            fail += 1
            if fail > 15:
                break

    return best_a, best_M

# coordinate polish

def coordinate_polish(a0, time_limit, seed=123, verbose=False):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        top = np.argpartition(c, -min(40, len(c)))[-min(40, len(c)):][:20]
        improved = False

        for k in top:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n, ki + 1)
            idx = np.arange(js, je)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-22:
                continue
            src = idx[int(np.argmax(contrib))]
            sv = a[src]
            if sv < 1e-12:
                continue
            found = False
            for _ in range(6):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in (0.01, 0.005, 0.02, 0.04):
                    amt = sv * frac
                    if amt < 1e-15:
                        continue
                    at = a.copy()
                    at[src] -= amt
                    at[dst] += amt
                    at = np.maximum(at, 0.0)
                    at = at / at.sum()
                    Mt = fast_conv(at).max()
                    if Mt < best_M - 1e-16:
                        a = at
                        best_M = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if i != j2 and a[i] > 1e-10:
                frac = float(rng.uniform(0.002, 0.01))
                amt = a[i] * frac
                a[i] -= amt
                a[j2] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M = fast_conv(a).max()
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 40 == 0:
            print(f"    Coord n={n} ({time.monotonic()-t0:.0f}s): J={2*n*best_M:.6f} stall={stall}")

    return best_a, best_M

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1760.0)

    best_seq = None
    best_score = float("inf")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    # ---- load parent -------------------------------------------------------
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-8:
            parent = p.copy()
            try_update(parent, "parent")
            print(f"Parent n={len(parent)} score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = shape_power(n_base, 2.0)
        best_score = score(best_seq)
        print(f"Default init n={n_base} score={best_score:.6f}")

    # ---- choose lengths to work on ----------------------------------------
    n0 = len(best_seq)
    lengths = set()
    lengths.add(n0)
    if parent is not None:
        lengths.add(len(parent))
    for dd in (-500, 0, 500, 1000, 1500):
        cand = n0 + dd
        if 2500 <= cand <= 9000:
            lengths.add(cand)
    lengths = sorted(lengths)

    # ---- Phase A: parametric seeds (cheap) + short SMD --------------------
    print("\n=== Phase A: seed & soft-mirror-descent over lengths ===")
    for n in lengths:
        t_phase = time_budget - (time.monotonic() - t_global)
        if t_phase < 120:
            break
        # budget per length scaled by how many lengths remain
        per_len_max = 300.0
        t_avail = min(per_len_max, t_phase * max(0.25, 1.0 / len(lengths)))
        t_avail = max(60.0, t_avail)
        t_avail = min(t_avail, t_phase - 60.0)
        if t_avail < 40:
            break

        seeds = []
        if parent is not None:
            seeds.append(interpolate_sequence(parent, n))
        else:
            seeds.append(best_seq if len(best_seq) == n else shape_power(n, 2.0))
        seeds.append(shape_power(n, 1.0))
        seeds.append(shape_power(n, 2.5))
        seeds.append(shape_beta(n, 2.0, 1.0))
        seeds.append(shape_expquad(n, 3.0, 1.5))
        seeds.append(shape_sine_exp(n, 1.0, 1.5))

        # cheap pre-rank by objective
        pre = []
        for idx, sd in enumerate(seeds):
            pre.append((score(sd), idx))
        pre.sort()
        order = [pre[i][1] for i in range(min(4, len(pre)))]

        for rank, idx in enumerate(order):
            t_step = time_budget - (time.monotonic() - t_global)
            if t_step < 80:
                break
            share = t_avail / max(1, len(order) - rank)
            share = max(25.0, share)
            share = min(share, t_step - 60.0)
            if share < 20:
                break
            a_in = normalize(seeds[idx])
            a_out, _ = soft_mirror_descent(a_in, share,
                                           tparam_init=2.0, tparam_final=120.0,
                                           c0=2.0, restart_stall=4000,
                                           seed=1000 + n % 97 + rank)
            try_update(a_out, f"smd_n{n}_s{idx}")

    # ---- Phase B: deep SMD from current best (a few lengths) --------------
    print("\n=== Phase B: deep soft-mirror-descent from best ===")
    for n in lengths:
        t_phase = time_budget - (time.monotonic() - t_global)
        if t_phase < 150:
            break
        a_in = normalize(interpolate_sequence(best_seq, n))
        share = min(400.0, t_phase * 0.5)
        share = max(80.0, share)
        share = min(share, t_phase - 90.0)
        if share < 50:
            break
        a_out, _ = soft_mirror_descent(a_in, share,
                                       tparam_init=1.5, tparam_final=80.0,
                                       c0=2.5, restart_stall=6000,
                                       seed=5000 + n % 131, verbose=True)
        try_update(a_out, f"smd_deep_n{n}")

    # ---- Phase C: coordinate polish on the best ---------------------------
    print("\n=== Phase C: coordinate polish ===")
    t_phase = time_budget - (time.monotonic() - t_global)
    if t_phase > 60:
        share = min(240.0, t_phase * 0.6)
        share = max(30.0, share)
        a_out, _ = coordinate_polish(normalize(best_seq), share, seed=789, verbose=True)
        try_update(a_out, "polish1")

    # ---- Phase D: one more SMD + polish cycle -----------------------------
    t_phase = time_budget - (time.monotonic() - t_global)
    if t_phase > 100:
        share = min(200.0, t_phase * 0.5)
        share = max(40.0, share)
        a_out, _ = soft_mirror_descent(normalize(best_seq), share,
                                       tparam_init=1.2, tparam_final=60.0,
                                       c0=2.0, restart_stall=5000, seed=9117)
        try_update(a_out, "smd_final")

    t_phase = time_budget - (time.monotonic() - t_global)
    if t_phase > 45:
        share = min(120.0, t_phase - 15.0)
        share = max(20.0, share)

    # ---- final validation -------------------------------------------------
    if best_seq is None:

    fs = score(best_seq)

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, t={time.monotonic()-t_global:.0f}s")
    return [float(x) for x in best_seq]
```
