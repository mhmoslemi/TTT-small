import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate as sp_correlate

# --- Helper functions ---

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
    return float(evaluate_sequence(a_list))

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except:

    return cp.ECOS

# --- Projected Gradient (with log-sum-exp smoothing) ---
def smooth_gradient_step(a, tau, eta):
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    shifted = tau * (ac - ac_max)
    exp_ac = np.exp(shifted)
    p = exp_ac / exp_ac.sum()
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    # = 2 * cross-correlation of p and a
    # grad[j] = 2 * sum_k p[k] * a[k-j] where 0 <= k-j < n
    # This is the valid cross-correlation
    grad = 2.0 * sp_correlate(p, a, mode='valid')
    
    # Subtract mean for tangent projection
    grad_t = grad - grad.mean()
    
    a_new = a - eta * grad_t
    a_new = project_simplex(a_new)
    return a_new, ac.max()

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
