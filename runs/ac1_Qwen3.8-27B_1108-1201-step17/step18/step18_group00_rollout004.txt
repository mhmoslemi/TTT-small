import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Algorithm from Wang & Carlier (2009)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

    Compute gradient of f_tau(a) = (1/tau) * log(sum_k exp(tau * c[k]))
    where c = conv(a,a).
    
    Returns (grad_a, f_tau_value, c)
    """

    size = 2 * n - 1
    fft_size = 1 << (int(size) - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    tc = tau * c
    tc = tc - tc.max()  # for numerical stability
    p = np.exp(tc)

    p = p / p_sum

    A2 = np.fft.rfft(a, L)
    cc = np.fft.irfft(np.conj(P) * A2, L)[:n]
    grad_a = (2.0 / tau) * cc
    
    return grad_a, f_val, c
