import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def parametric_cubic(n, c1, c2, c3):
    x = np.linspace(0.0, 1.0, n)
    log_a = c1 * x + c2 * x**2 + c3 * x**3
    log_a -= log_a.max()
    a = np.exp(log_a)
    a = np.maximum(a, 1e-30)
    return a / a.sum()


def parametric_power(n, power, base_eps=0.01):
    x = np.arange(1, n + 1, dtype=np.float64) / n
    a = (x + base_eps) ** power
    return a / a.sum()


def parametric_gaussian_ramp(n, alpha, width, center):
    x = np.linspace(0.0, 1.0, n)
    base = 1.0 + alpha * x
    bump = np.exp(-((x - center) / width) ** 2)
    a = base * bump
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _obj_cubic(params, n):
    a = parametric_cubic(n, params[0], params[1], params[2])
    c = fast_autocorr(a)
    return 2.0 * n * c.max()

def topk_gradient(a, c, k=5):
    n = len(a)
    top_idx = np.argsort(c)[::-1][:k]
    peak_vals = c[top_idx]
    total = peak_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = peak_vals / total

    grad = np.zeros(n)
    for i in range(k):
        ki = int(top_idx[i])
        wi = w[i]
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    return grad

def simulated_annealing(a0, time_limit, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    c = fast_autocorr(a)
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    temp = 0.05
    cooldown = 0.9995
    min_temp = 1e-6

    iteration = 0
    while time.monotonic() - t0 < time_limit and temp > min_temp:
        iteration += 1
        c = fast_autocorr(a)
        M_curr = c.max()

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        frac = rng.uniform(0.02, 0.08)
        amt = a[i] * frac
        if amt < 1e-15:
            continue

        a_new = a.copy()
        a_new[i] -= amt
        a_new[j] += amt
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

            a = a_new
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a.copy()

        if verbose and iteration % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA iter={iteration} ({elapsed:.0f}s): J={J:.6f}, temp={temp:.6f}")

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
