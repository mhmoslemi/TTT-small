def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def score(a):

    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')

def smooth_descent(a0, time_limit, beta_start, beta_end, verbose=True):
    """Smooth gradient descent with log-sum-exp approximation."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    # Nesterov momentum
    v = np.zeros(n)
    momentum = 0.0
    
    eta = 0.1  # base learning rate
    beta = beta_start
    min_improvement = 1e-10
    stall = 0
    max_stall = 500
    
    rng = np.random.default_rng(123)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient
        grad, c = compute_smooth_grad(a, beta)
        
        # Nesterov step
        momentum = min(momentum + 0.005, 0.9)
        v = momentum * v + (1 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-15:
            v = v / v_norm
        
        # Adaptive learning rate
        eta_t = eta / (1.0 + 0.01 * t)
        
        # Mirror descent step in log-space
        log_a = np.log(np.maximum(a, 1e-30)) - eta_t * v
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        # Check improvement
        c_new = fast_autocorr(a)
        M_new = c_new.max()
        s_new = a.sum()
        sc = 2.0 * n * M_new / (s_new * s_new)
        
        if sc < best_score - min_improvement:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Anneal beta
        if t % 100 == 0:
            progress = min(1.0, t / 5000.0)
            beta = beta_start * (beta_end / beta_start) ** progress
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            # Perturb
            noise = rng.normal(0, 0.05, n)
            a = normalize(best_a + noise * best_a)
            v = np.zeros(n)
            momentum = 0.0
            beta = max(beta_start, beta * 0.5)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothDescent t={t} ({elapsed:.0f}s): score={best_score:.6f}, beta={beta:.1f}")
    
    return best_a, best_score

def lp_frank_wolfe(a0, time_limit, initial_K=300, max_K=2500, add_per_iter=40):

    import cvxpy as cp
    n = len(a0)
    a = normalize(a0)

    x = a.copy()
    
    # Initialize constraint set
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    # Pre-allocate for constraint matrix
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-12:
            best_m = ac_x.max()
            best_a = x.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix A where A[j, i] = y[k_i - i] for the i-th element
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP: minimize M s.t. A@g <= M, sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        prob = cp.Problem(cp.Minimize(M_var), constraints)

            prob.solve(solver=cp.CLARABEL, verbose=False)

            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_sum = g_val.sum()

            continue

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            for k in violations[:add_per_iter]:
                constraint_set.add(int(k))
        
        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)

            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
    
    return best_a, best_m

def pair_coordinate_descent(a0, time_limit, rng_seed=42):

    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()

        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = a.copy()
        
        improved = False
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Find top contributing pairs
            n_pairs = min(5, len(idx))
            pair_idx = np.argsort(contrib)[::-1][:n_pairs]
            
            for pi in pair_idx:
                src = int(idx[pi])
                dst = int(k_int - src)
                if src == dst or a[src] < 1e-12 or a[dst] < 1e-12:
                    continue
                
                # Try reducing the larger of a[src], a[dst] and moving to a random location
                if a[src] > a[dst]:
                    move_from = src
                    move_to_candidates = dst
                else:
                    move_from = dst
                    move_to_candidates = src
                
                # Try moving some mass to a random location
                for _ in range(3):
                    dst_rand = int(rng.integers(n))
                    if dst_rand == move_from:
                        continue
                    
                    for frac in [0.02, 0.05, 0.1]:
                        amt = a[move_from] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[move_from] -= amt
                        a_trial[dst_rand] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-12:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            
            if improved:
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
            # Random perturbation
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.01, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

def parametric_init(n, best_sequence=None):
    """Try various parametric initializations."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    
    # Beta-like
    for a_p in [0.5, 1.0, 1.5, 2.0]:
        for b_p in [0.5, 1.0, 1.5, 2.0]:
            candidates.append(np.maximum(i_arr + 1, 0.5) ** a_p * np.maximum(n - i_arr, 0.5) ** b_p)
    
    # Gaussians
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sig * sig)))
    
    # Exponentials
    for rate in [1, 2, 3, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * x))
    
    # Sine
    for freq in [1, 2, 3]:
        candidates.append(np.sin(np.pi * freq * x) ** 2)
    
    best_cand = None
    best_val = float('inf')
    for cand in candidates:
        cand_norm = normalize(cand)
        sc = score(cand_norm)
        if sc < best_val:
            best_val = sc
            best_cand = cand_norm
    
    # Also try blending with parent
    if best_sequence is not None:
        for w in [0.3, 0.5, 0.7]:
            if best_cand is not None:
                blend = normalize(w * best_sequence + (1-w) * best_cand)
                sc = score(blend)
                if sc < best_val:
                    best_val = sc
                    best_cand = blend
    
    if best_cand is not None:
        return best_cand, best_val
    return np.ones(n) / n, float('inf')

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parametric multi-family initialization
2. Smooth gradient descent with log-sum-exp annealing (FFT-based gradients)
3. LP-Frank-Wolfe with aggressive cutting-plane management
4. Pair coordinate descent with lookahead
5. Multi-length exploration with interpolation
"""
import numpy as np
import time

def fast_autocorr(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Clip to non-negative and normalize to unit sum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def compute_score(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    c = fast_autocorr(a_norm)
    return float(2.0 * n * c.max())

def compute_smooth_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*c[k])) w.r.t. a.
    Returns (grad, c) where grad[m] = 2 * sum_j a[j]*w[j+m]
    and w = softmax(beta * c).
    """
    n = len(a)
    c = fast_autocorr(a)
    # Softmax weights
    bc = beta * c
    bc -= bc.max()
    exp_bc = np.exp(bc)
    w = exp_bc / exp_bc.sum()
    # Gradient: grad[m] = 2 * sum_j a[j] * w[j+m]
    # This is cross-correlation (w * a) at lags 0..n-1
    # Using FFT: corr = ifft(conj(fft(w)) * fft(a))
    size = 2 * n - 1
    # Need fft_size >= (2n-1) + n - 1 = 3n - 2
    fft_size = 1 << (3 * n - 2).bit_length()
    Af = np.fft.rfft(a, fft_size)
    Wf = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(np.conj(Wf) * Af, fft_size)[:n]
    grad = 2.0 * corr
    return grad, c

def smooth_gradient_descent(a0, time_limit, beta_start=50.0, beta_end=3000.0, verbose=True):
    """Smooth gradient descent with mirror descent on simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    momentum = 0.0
    base_lr = 0.15
    beta = beta_start
    stall = 0
    max_stall = 800
    rng = np.random.default_rng(77)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        grad, c = compute_smooth_grad(a, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        grad = grad / gnorm

        # Nesterov momentum
        momentum = min(momentum + 0.003, 0.85)
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm > 1e-18:
            v = v / vnorm

        # Adaptive learning rate with decay
        lr = base_lr / (1.0 + 0.005 * t)

        # Mirror descent in log-space
        log_a = np.log(np.maximum(a, 1e-30)) - lr * v
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)

        else:
