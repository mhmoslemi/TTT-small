        # a[j - ks] valid when 0 <= j - ks <= n-1, i.e., ks <= j <= ks+n-1
        j_lo2 = max(0, ks)
        j_hi2 = min(n - 1, ks + n - 1)
        if j_lo2 <= j_hi2:
            grad_ac[j_lo2:j_hi2+1] += a[np.arange(j_lo2, j_hi2+1) - ks]
        
        return (S * grad_ac - 2 * M * a) / (S**3) * 2 * n
    else:
        # Multi-peak with softmax weighting
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        ac_top = ac[top_idx]
        weights = ac_top / ac_top.sum()
        
        grad_ac = np.zeros(n)
        for idx_i, k in enumerate(top_idx):
            wi = weights[idx_i]
            if wi < 1e-10:
                continue
            ks = int(k)
            j_lo1 = max(0, ks - n + 1)
            j_hi1 = min(n - 1, ks)
            if j_lo1 <= j_hi1:
                grad_ac[j_lo1:j_hi1+1] += wi * a[ks - np.arange(j_lo1, j_hi1+1)]
            j_lo2 = max(0, ks)
            j_hi2 = min(n - 1, ks + n - 1)
            if j_lo2 <= j_hi2:
                grad_ac[j_lo2:j_hi2+1] += wi * a[np.arange(j_lo2, j_hi2+1) - ks]
        
        return (S * grad_ac - 2 * M * a) / (S**3) * 2 * n


def grad_optimize_exact(a0, time_limit, top_k=3, temperature0=2.0, c0=3.0,
                        restart_every=3000, verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent with exact subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    restart_count = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g = exact_subgradient(y, ac, top_k=top_k)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        # Mirror descent step in log space
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.1, 0.02 + 0.01 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    GRAD restart t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def lp_direction(f, M, n_constraints=100, verbose=False):
    """
    Solve LP to find direction g that maximizes sum(g)
    subject to conv(f,g) <= M (using subset of constraints) and g >= 0.
    """
    n = len(f)
    
    if n > 3000:
        # For large n, use a coarser approach
        # Only use top n_constraints peaks
        fc = fast_autocorr(f)
        top_idx = np.argsort(fc)[::-1][:n_constraints]
        # Build constraint matrix with only these constraints
        A_rows = []
        for k in top_idx:
            ks = int(k)
            row = np.zeros(n)
            j_lo = max(0, ks - n + 1)
            j_hi = min(n - 1, ks)
            for j in range(j_lo, j_hi + 1):
                row[j] = f[ks - j]
            A_rows.append(row)
        A = np.array(A_rows)
    else:
        # Build full constraint matrix (sparse)
        A = lil_matrix((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                A[k, j] = f[k - j]
        A = A.tocsr()
    
    b = np.full(A.shape[0], M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception as e:
        if verbose:
            print(f"    LP error: {e}")
    
    return None


def lp_optimize(a0, time_limit, verbose=False, rng_seed=42, blend=0.3):
    """Iterative LP-based optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    for it in range(1000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        g = lp_direction(a, M, n_constraints=80)
        if g is None:
            break
        
        g = np.maximum(g, 0)
        sg = g.sum()
        if sg < 1e-15:
            break
        g = g / sg
        
        # Blend
        a_new = (1 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try smaller blend
            for b_small in [0.1, 0.05, 0.02]:
                a_try = (1 - b_small) * a + b_small * g
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                M_try = fast_autocorr(a_try).max()
                if M_try < best_m - 1e-15:
                    best_m = M_try
                    best_a = a_try.copy()
                    a = a_try
                    stall = 0
                    break
            else:
                stall += 1
                if stall > 20:
                    # Perturb
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * 0.05
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                    stall = 0
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP it={it} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def spectral_flatten(a0, time_limit, alpha0=0.5, verbose=False):
    """Modified Gerchberg-Saxton with adaptive alpha."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Adaptive alpha: start aggressive, become gentle
        alpha = alpha0 + (1.0 - alpha0) * min(1.0, it / 500.0)
        
        # Target: pull spectrum towards flat
        P_target = P_mean * (np.maximum(P, 1e-30) / P_mean) ** (1.0 - alpha)
        
        phase = np.angle(A)
        mag_old = np.abs(A)
        mag_new = np.sqrt(P_target)
        
        # Damped update
        A_new = phase + 0  # keep phase
        mag_update = mag_new / np.maximum(mag_old, 1e-30)
        mag_update = np.clip(mag_update, 0.8, 1.2)
        A_new = A * mag_update
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.2
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()
    
    return best_a, best_m


def make_cosine_taper(n):
    """a[i] = cos^2(pi*i/(2n)) - zero at right edge."""
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * x / (2.0 * n)) ** 2
    return norm_sum1(a)


def make_sine_bump(n):
    """a[i] = sin(pi*(i+1)/(n+1)) - zero at both edges."""
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 1) / (n + 1))
    return norm_sum1(a)


def make_power_law_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_mild_ramp(n, center_frac=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 + center_frac * x
    return norm_sum1(a)


def make_gaussian_bump(n, center=0.5, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return norm_sum1(a)


def make_asymmetric_gaussian(n, center=0.6, width_left=0.2, width_right=0.4):
    """Asymmetric Gaussian: different widths on left and right of center."""
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    mask_left = x <= center
    mask_right = x > center
    a[mask_left] = np.exp(-((x[mask_left] - center) / width_left) ** 2)
    a[mask_right] = np.exp(-((x[mask_right] - center) / width_right) ** 2)
    return norm_sum1(a)


def make_polynomial(n, coeffs=None):
    """a[i] = polynomial in i/n. Default: 1 + 0.5*(i/n) + 0.1*(i/n)^2"""
    if coeffs is None:
        coeffs = [1.0, 0.5, 0.1]
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for k, c in enumerate(coeffs):
        a += c * x ** k
    a = np.maximum(a, 0)
    return norm_sum1(a)
