def fast_conv(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def smooth_max_and_weights(conv, t, top_k=200):
    """Compute smooth max t^{-1}log(sum exp(t*c)) and softmax weights on top_k."""
    n_conv = len(conv)
    if n_conv <= top_k:
        idx = np.arange(n_conv)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    
    c_top = conv[idx]
    c_ref = c_top.max()
    exp_vals = np.exp(t * (c_top - c_ref))
    w_sum = exp_vals.sum()
    
    # For entries outside top_k, their contribution is exp(t*(c - c_ref)) which is very small
    # Approximate: smooth_max ≈ (1/t)*(c_ref + log(w_sum) + (n_conv - top_k)*exp(t*(c_next - c_ref)))
    # For simplicity, just use top_k
    sm_val = (1.0 / t) * (c_ref + np.log(w_sum))
    
    weights = exp_vals / w_sum  # These are d(sm_val)/d(c_top[idx]) * t
    # Actually: d((1/t)*log(sum exp(t*c_k)))/d(c_k) = exp(t*c_k)/sum_j exp(t*c_j)
    # For top_k: weight_k = exp(t*c_k)/sum_{j in top_k} exp(t*c_j)
    # But we're missing the (n_conv - top_k) other terms. For large t, they're negligible.
    
    # Full gradient vector (sparse)
    grad_c = np.zeros(n_conv)
    grad_c[idx] = weights
    
    return sm_val, grad_c, idx

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B on smooth-max objective with softmax parameterization (exploits curvature)
2. Multi-scale smoothing parameter annealing for convergence
3. Diverse multi-start initialization (structural patterns)
4. FISTA with simplex projection as robust backup
5. Targeted coordinate descent polish on active peaks
6. Multi-length exploration

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_weights(conv, t, top_k=200):
    """Compute (1/t)*log(sum exp(t*c)) and gradient weights on top_k entries."""
    n_conv = len(conv)
    if n_conv <= top_k:
        idx = np.arange(n_conv)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]

    c_top = conv[idx]
    c_ref = c_top.max()
    exp_vals = np.exp(t * (c_top - c_ref))
    w_sum = exp_vals.sum()

    sm_val = (1.0 / t) * (c_ref + np.log(w_sum))
    weights = exp_vals / w_sum

    grad_c = np.zeros(n_conv)
    grad_c[idx] = weights
    return sm_val, grad_c, idx


def compute_grad_a(a, grad_c, active_idx):
    """Compute d(obj)/d(a[i]) from convolution gradient."""
    n = len(a)
    grad_a = np.zeros(n)
    for k_i in active_idx:
        k = int(k_i)
        wk = grad_c[k]
        if wk < 1e-16:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += wk * 2.0 * a[k - idx]
    return grad_a


def softmax_from_h(h):
    h_shift = h - h.max()
    a = np.exp(h_shift)
    a = a / a.sum()
    return a


def make_h_from_a(a):
    a_safe = np.maximum(a, 1e-15)
    return np.log(a_safe / a_safe.sum())


def lbgfbs_run(a0, t_smooth, top_k, maxiter):
    """Single L-BFGS-B run with given smoothing parameter."""
    n = len(a0)
    h0 = make_h_from_a(a0)

    def obj_grad(h):
        a = softmax_from_h(h)
        conv = fast_conv(a)
        sm_val, grad_c, active_idx = smooth_max_and_weights(conv, t_smooth, top_k)
        obj = 2.0 * n * sm_val
        grad_a = compute_grad_a(a, grad_c, active_idx)
        grad_a *= 2.0 * n
        dot_ga_a = np.dot(grad_a, a)
        grad_h = a * (grad_a - dot_ga_a)
        return obj, grad_h

    t0 = time.monotonic()
    try:
        result = minimize(obj_grad, h0, method='L-BFGS-B', jac=True,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                   'maxfun': maxiter * 8, 'disp': False})
        a_out = softmax_from_h(result.x)
        a_out = np.maximum(a_out, 0.0)
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0.copy()
    elapsed = time.monotonic() - t0
    return a_out, elapsed


def coordinate_polish(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent: move mass from high-cost to low-cost positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 80:
        conv = fast_conv(a)
        # Find top peaks
        n_peaks = min(10, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_peaks]

        # Compute cost per coordinate (total contribution to top peaks)
        costs = np.zeros(n)
        for k in top_peaks:
            k_i = int(k)
            lo = max(0, k_i - n + 1)
            hi = min(n - 1, k_i)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            costs[idx] += 2.0 * a[idx] * a[k_i - idx]

        improved = False
        # Try moving mass from high-cost to low-cost positions
        src_candidates = np.argsort(costs)[::-1][:max(5, n // 200)]
        dst_candidates = np.argsort(costs)[:max(10, n // 100)]

        for src_i in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src_i)
            if a[src_i] < 1e-12:
                continue
            for dst_i in dst_candidates:
                dst_i = int(dst_i)
                if dst_i == src_i:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5]:
                    amt = a[src_i] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial /= s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

        print(f"  Trying n={new_n}: ", end="", flush=True)
        a_interp = interpolate_seq(_best_tracker['seq'], new_n)
        try_update(a_interp, f"intp{new_n}")

        t5a = min(remaining * 0.35, 150.0)
        if t5a > 15:
            a_out, e = lbgfbs_run(a_interp, t_smooth=400, top_k=100,
                                  maxiter=int(300 * t5a / 50))
            try_update(a_out, f"lbgfbs{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.25, 80.0)
            if t5b > 10:
                a_out2, _ = coordinate_polish(a_interp, t5b, seed=99)
                try_update(a_out2, f"cd{new_n}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Polish (n={len(_best_tracker['seq'])}) ===", flush=True)
    if remaining > 40:
        a_in = norm_sum1(_best_tracker['seq'])
        t6a = min(remaining * 0.5, 200.0)
        a_curr = a_in
        for t_val in [1500, 5000]:
            t_this = t6a / 2
            if time.monotonic() - t_global > time_budget - 30:
                break
            a_curr, e = lbgfbs_run(a_curr, t_smooth=t_val, top_k=50,
                                   maxiter=int(200 * min(t_this, 80) / 40))
            sc = evaluate_sequence([float(v) for v in a_curr])
            if np.isfinite(sc) and sc < _best_tracker['score']:
                _best_tracker['score'] = sc
                _best_tracker['seq'] = a_curr.copy()
                print(f"  Final L-BFGS t={t_val}: {sc:.6f}", flush=True)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t6b = min(remaining * 0.6, 100.0)
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = coordinate_polish(a_in, t6b, seed=777)
        try_update(a_out, "cd_final")

    # Final
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, "
              f"time={time.monotonic()-t_global:.0f}s", flush=True)
    else:
        n_fb = 5000
        _best_tracker['seq'] = make_hann(n_fb)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}", flush=True)

    return [float(v) for v in _best_tracker['seq']]
