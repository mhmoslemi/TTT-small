import numpy as np

from scipy.optimize import minimize_scalar

def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_value(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    s = a.sum()
    return 2.0 * n * c.max() / (s*s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def shape_power(n, p):
    """a[j] = ((j+1)/(n+1))^p"""
    j = np.arange(n)
    return ((j+1)/(n+1))**p

def shape_beta(n, p, q):
    """a[j] = ((j+1)/(n+1))^p * (1 - j/n)^q"""
    j = np.arange(n)
    x = (j+1)/(n+1)
    return x**p * (1 - j/n)**q

def shape_expquad(n, p, q):
    """a[j] = exp(p*j/n - q*(j/n)^2)"""
    j = np.arange(n)
    x = j / n
    return np.exp(p*x - q*x*x)

def mirror_descent(a0, time_limit, top_k_init=30, top_k_max=300, c0=3.0, verbose=False):
    """Mirror descent with exponential weights for minimizing max(conv)."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    top_k = top_k_init
    beta = 4.0
    stall = 0
    max_stall = 5000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute convolution
        c = fast_conv(y)
        
        # Find top-k peaks
        k_sel = min(top_k, len(c))
        top_idx = np.argpartition(c, -k_sel)[-k_sel:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]
        
        # Compute weighted subgradient
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_sel) / k_sel
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            idx = np.arange(j_start, j_end)
            grad[idx] += 2.0 * wi * y[ki - idx]
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive top_k
        if t % 2000 == 0:
            top_k = min(top_k + 5, top_k_max)
        
        # Restart
        if stall > max_stall:
            stall = 0
            top_k = min(top_k + 20, top_k_max)
            # Perturb best
            noise = np.random.default_rng(t).normal(0, 0.02, n) * best_a
            x = np.maximum(best_a + noise, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            beta = max(beta - 0.5, 2.0)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M  # (sum=1, so sum^2=1)
            print(f"  MD t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={top_k}, stall={stall}")
    
    return best_a, best_M
