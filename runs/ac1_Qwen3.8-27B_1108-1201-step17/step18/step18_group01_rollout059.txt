def fast_autocorr(a):
    """Compute linear autocorrelation (self-convolution) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def mgd_softmax(a0, time_limit, beta0, beta_end, eta0, verbose=False, seed=42):
    """
    MGD with exp-softmax subgradient and beta annealing.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = eta0
    patience = 0
    max_patience = 500
    iter_count = 0
    total_iters = 20000  # soft cap

    while time.monotonic() - t0 < time_limit and iter_count < total_iters:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            # Try a smaller patience restart
            if patience > max_patience * 2:
                break
            max_patience = int(max_patience * 0.7) + 100

        # Annealed beta
        progress = min(1.0, iter_count / total_iters)
        beta = beta0 + (beta_end - beta0) * progress

        # Compute softmax weights over convolution peaks
        # Use top portion to keep it efficient
        k_use = min(len(ac), max(50, n // 10))
        top_idx = np.argsort(ac)[::-1][:k_use]
        top_vals = ac[top_idx]
        
        # Softmax weights
        max_tv = top_vals.max()
        w = np.exp(beta * (top_vals - max_tv))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_use) / k_use
        else:
            w = w / w_sum

        # Compute subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * a[ki - idx]

        # Center the gradient
        grad = grad - grad.mean()
        gmax = np.abs(grad).max()
        if gmax < 1e-18:
            break
        grad = grad / gmax

        # Multiplicative update with backtracking
        accepted = False
        eta_try = eta
        for _bt in range(10):
            a_new = a * np.exp(-eta_try * grad)
            a_new = np.maximum(a_new, 1e-15)
            s = a_new.sum()
            if s < 1e-12:
                eta_try *= 0.5
                continue
            a_new = a_new / s
            M_new = fast_autocorr(a_new).max()
            if M_new <= M + 1e-10:
                a = a_new
                if M_new < best_m - 1e-14:
                    best_m = M_new
                    best_a = a.copy()
                accepted = True
                eta_try = min(eta_try * 1.05, 0.5)
                break
            else:
                eta_try *= 0.5

        if not accepted:
            eta = eta_try
            patience += 50
        else:
            eta = eta_try

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SM iter={iter_count} ({elapsed:.0f}s): J={J:.8f}, eta={eta:.5f}, beta={beta:.1f}")

    if verbose:
        J = 2.0 * n * best_m

def anneal_polish(a0, time_limit, verbose=False, seed=123):
    """
    Simulated annealing local search on the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 0.001 * best_m  # initial temperature
    T_final = 0.00001 * best_m
    cooling_rate = 0.9997
    stall = 0
    max_stall = 2000

    block_sizes = [3, 5, 8, 12, 20, 35]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        M = fast_autocorr(a).max()
        n_trials = 50

        for _t in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            # Choose perturbation type
            r = rng.random()
            if r < 0.5:
                # Block perturbation
                bs = int(block_sizes[rng.integers(len(block_sizes))])
                start = int(rng.integers(0, max(1, n - bs + 1)))
                end = min(n, start + bs)
                block = a[start:end].copy()
                scale = rng.uniform(0.01, 0.06) * (block.mean() + 1e-10)
                pert = rng.normal(0, scale, end - start)
                new_block = np.maximum(block + pert, 0.0)
                a_trial = a.copy()
                a_trial[start:end] = new_block
            elif r < 0.8:
                # Two-point move
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                while j == i:
                    j = int(rng.integers(n))
                frac = rng.uniform(0.005, 0.04)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] = max(0, a[i] - amt)
                a_trial[j] += amt
            else:
                # Gaussian perturbation on small subset
                k = int(rng.integers(2, min(10, n)))
                idx = rng.choice(n, size=k, replace=False)
                scale = rng.uniform(0.01, 0.05)
                pert = rng.normal(0, scale * a.mean(), k)
                a_trial = a.copy()
                a_trial[idx] = np.maximum(a[idx] + pert, 0.0)

            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-12:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()

            delta = M_trial - M
            if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-18)):
                a = a_trial
                if M_trial < best_m - 1e-15:
                    best_m = M_trial
                    best_a = a.copy()
                    stall = 0
                # Reset stall on acceptance
                if delta < 0:
                    stall = max(0, stall - 2)
            else:
                pass

        T *= cooling_rate
        if T < T_final:
            T = T_final
        stall += 1

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Anneal stall={stall} ({elapsed:.0f}s): J={J:.8f}, T={T:.2e}")

    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Anneal done ({elapsed:.0f}s): J={J:.8f}")
    return best_a


def coord_refine(a0, time_limit, verbose=False, seed=99):

    Coordinate descent refinement targeting peak-contributing positions.
    """
    n = len(a0)
    a = norm_sum1(a0)

        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for kk in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(kk)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:

            # Find best destination

            peak_impact = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_impact[idx2] += a[k2i - idx2]

            # Move to position with lowest peak impact
            peak_impact[src] = np.inf  # don't move to self
            dst = int(np.argmin(peak_impact))

            for frac in [0.005, 0.01, 0.02, 0.04, 0.06, 0.1]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-12:

                    a = a_trial
                    best_m = M_trial

            # Random escape

            if i != j and a[i] > 1e-10:

                M_new = fast_autocorr(a).max()
                if M_new < best_m:

                    best_a = a.copy()
                    stall = 0
                T = 0  # no acceptance of worse

    elapsed = time.monotonic() - t0

def parametric_init(n):
    """Generate good initial sequences from parametric families."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    # Sine squared
    candidates.append(np.sin(np.pi * x) ** 2)
    # Power functions
    for p in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        candidates.append((i_arr + 1) ** p)
        candidates.append((n - i_arr) ** p)
        candidates.append(np.abs(x - 0.5) ** p + 0.01)
    # Gaussian bumps
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sigma ** 2)))
    # Multi-bump
    for n_bumps in [2, 3, 4]:
        centers = np.linspace(0.2, 0.8, n_bumps)
        for sigma in [0.08, 0.12, 0.15]:
            f = np.zeros(n)
            for c in centers:
                f += np.exp(-((x - c) ** 2) / (2 * sigma ** 2))
            candidates.append(f)
    # Sine waves
    for freq in [1, 2, 3, 4]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.001))
        candidates.append(0.5 + 0.5 * np.sin(2 * np.pi * freq * x * 0.5))
    # Exponential decay
    for rate in [1, 2, 5]:
        candidates.append(np.exp(-rate * x) + 0.01)
        candidates.append(np.exp(rate * (x - 1)) + 0.01)
    # Raised cosine
    candidates.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # Triangular
    candidates.append(1 - 2 * np.abs(x - 0.5))

    best_a = None
    best_val = float('inf')
    for cand in candidates:
        cand = norm_sum1(cand)
        m = fast_autocorr(cand).max()
        val = 2.0 * n * m
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    return best_a, best_val

    ]), key=lambda l: abs(l - n_parent))
    lengths_to_try = [l for l in lengths_to_try if 2000 <= l <= 8000]

    # Phase 1: Quick parametric initialization for parent length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Parametric init (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        t1 = min(remaining * 0.03, 30.0)
        a_p, s_p = parametric_init(n_parent)
        sc_p = evaluate_safe(a_p)
        if np.isfinite(sc_p) and sc_p < best_score:
            best_score = sc_p
            best_seq = a_p.copy()
            print(f"  Parametric improved to {sc_p:.8f}")

    # Phase 2: MGD softmax on parent length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: MGD softmax (n={n_best}, remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.22, 300.0)
    if t2 > 30:
        a_out = mgd_softmax(norm_sum1(best_seq), t2, beta0=5.0, beta_end=50.0,
                           eta0=0.03, verbose=True, seed=1)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST MGD: {sc:.8f}")

    # Phase 3: Second MGD round with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: MGD r2 (n={n_best}, remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 30:
        a_out = mgd_softmax(norm_sum1(best_seq), t3, beta0=20.0, beta_end=100.0,
                           eta0=0.01, verbose=True, seed=2)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST MGD r2: {sc:.8f}")

    # Phase 4: Annealing polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Annealing (n={n_best}, remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out = anneal_polish(norm_sum1(best_seq), t4, verbose=True, seed=7)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Anneal: {sc:.8f}")

    # Phase 5: Coordinate refine
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord refine (n={n_best}, remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        a_out = coord_refine(norm_sum1(best_seq), t5, verbose=True, seed=3)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Coord: {sc:.8f}")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length (remaining={remaining:.0f}s) ===")
    tried_lengths = {len(best_seq)}
    for new_n in lengths_to_try:
        if new_n in tried_lengths:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        tried_lengths.add(new_n)
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        # Quick check if interpolation helps
        sc_interp = evaluate_safe(a_long)
        if np.isfinite(sc_interp) and sc_interp < best_score:
            best_score = sc_interp
            best_seq = a_long.copy()
            print(f"  Interpolation to n={new_n} already improves: {sc_interp:.8f}")

        # MGD on new length
        t6 = min(remaining * 0.18, 120.0)
        if t6 > 20:
            a_out = mgd_softmax(a_long, t6, beta0=10.0, beta_end=60.0,
                               eta0=0.02, verbose=False, seed=new_n % 100)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST n={new_n}: {sc:.8f}")

        # Quick anneal
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.10, 60.0)
            if t6b > 10:
                a_out = anneal_polish(norm_sum1(best_seq), t6b, verbose=False, seed=new_n % 200)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST anneal n={new_n}: {sc:.8f}")

    # Phase 7: Final MGD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final MGD (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 25:
        t7 = min(remaining * 0.35, 150.0)
        if t7 > 15:
            a_out = mgd_softmax(norm_sum1(best_seq), t7, beta0=30.0, beta_end=150.0,
                               eta0=0.005, verbose=False, seed=777)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final MGD: {sc:.8f}")

    # Phase 8: Final coordinate + anneal
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.30, 100.0)
        if t8 > 10:
            a_out = coord_refine(norm_sum1(best_seq), t8, verbose=False, seed=555)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final coord: {sc:.8f}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t9 = min(remaining * 0.50, 80.0)
        if t9 > 10:
            a_out = anneal_polish(norm_sum1(best_seq), t9, verbose=False, seed=888)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final anneal: {sc:.8f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5200
        best_seq, _ = parametric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.8f}")

    return [float(x) for x in best_seq]
