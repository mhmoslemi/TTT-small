"""
Minimize J = 2*n*max(conv(a,a)) / (sum(a))^2  (scale-invariant; we work with sum=1).

Approach (different from the previous LP-heavy SOTA):
  - The self-convolution (a*a) has Fourier transform F(w)^2 = |F(w)|^2 * e^{2i arg F(w)}.
    Low peak self-convolution therefore requires BOTH a flat power spectrum AND a
    (near) vanishing doubled phase, i.e. a strongly non-symmetric, "chirp-like" shape.
  - Workhorse: fast multiplicative-weight / mirror descent in the simplex using the
    exact subgradient g[i] = (sum_k w_k 2 a[k-i]) over the top-K peak lags. This is
    O(K*n + n log n) per step -> ~1e5+ steps in a few minutes.
  - Refiner: LP / Frank-Wolfe with cutting planes (cvxpy CLARABEL/PDLP).
  - Diversity: many non-symmetric parametric inits (random Fourier-phase Gibbs, two-tone,
    power laws, phase-modulated sines) + periodic restarts / annealing kicks.
  - Length exploration by interpolation, plus a final cheap coordinate polish.
"""

import numpy as np
import time
import cvxpy as cp


# ---------------------------------------------------------------------------
# core numerics
# ---------------------------------------------------------------------------
def fast_conv(a):
    n = int(len(a))
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    out = np.fft.irfft(A * A, fft_size)
    return out[:size]


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-6:
        return float("inf")
    return float(2.0 * n * fast_conv(a).max() / (s * s))


def eval_global(a):
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float("inf")


def interp_resample(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a = a / s
    old_x = np.linspace(0.0, 1.0, len(a))
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, a)
    out = np.maximum(out, 0.0)
    ts = out.sum()
    if ts < 1e-15:
        return np.ones(new_n) / new_n
    return out / ts

_SOLVER = None


def get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(40, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.6])
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _SOLVER = s
                print("  LP solver:", s)
                return s
        except Exception:
            continue
    _SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _SOLVER


# ---------------------------------------------------------------------------
# initialisation families (all deliberately non-symmetric)
# ---------------------------------------------------------------------------
def gen_candidates(n, rng):
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    cands = []

    # sinusoidal envelope (non-symmetric)
    cands.append(norm_sum1(np.sin(np.pi * x) ** 2))
    cands.append(norm_sum1(np.sin(np.pi * x * 2) ** 2))

    # power laws and reversed
    for p in [0.3, 0.6, 1.0, 1.6, 2.5]:
        cands.append(norm_sum1((i + 1.0) ** p))
        cands.append(norm_sum1((n - i) ** p))

    # beta-like (asymmetric skew)
    for al in [0.4, 0.8, 1.5]:
        for be in [0.4, 1.0, 2.5]:
            a = np.maximum(i + 1.0, 0.5) ** al * np.maximum(n - i, 0.5) ** be
            cands.append(norm_sum1(a))

    # two-tone / phase-modulated sines (builds Fourier phase -> F^2 cancellations)
    for _ in range(6):
        f1 = float(rng.uniform(0.5, 6.0))
        f2 = float(rng.uniform(1.5, 12.0))
        p1 = float(rng.uniform(-3.0, 3.0))
        p2 = float(rng.uniform(-3.0, 3.0))
        a = 1.0 + 0.55 * np.sin(2 * np.pi * f1 * x + p1) + 0.35 * np.sin(2 * np.pi * f2 * x + p2)
        cands.append(norm_sum1(np.maximum(a, 0.0)))

    # random Fourier Gibbs (non-symmetric, smooth)
    for _ in range(8):
        ph = rng.uniform(0.0, 2 * np.pi, 7)
        vv = rng.uniform(-0.9, 0.9, 7)
        s = np.zeros(n)
        for m in range(7):
            s += vv[m] * np.cos(2 * np.pi * (m + 1) * x + ph[m])
        cands.append(norm_sum1(np.exp(1.5 * s)))

    # shifted Gaussians (off-centre -> non-symmetric)
    for c in [0.2, 0.32, 0.68, 0.8]:
        for sg in [0.12, 0.22, 0.35]:
            cands.append(norm_sum1(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    # bimodal, unbalanced
    for c1, c2, w2 in [(0.15, 0.8, 0.4), (0.2, 0.85, 1.6), (0.1, 0.75, 0.7)]:
        a = np.exp(-((x - c1) ** 2) / (2 * 0.1 ** 2)) + w2 * np.exp(-((x - c2) ** 2) / (2 * 0.1 ** 2))
        cands.append(norm_sum1(a))

    return cands


def parametric_best(n, rng):
    best_a = None
    best_s = float("inf")
    for a in gen_candidates(n, rng):
        s = local_score(a)
        if s < best_s:
            best_s = s
            best_a = a
    return best_a, best_s

# multiplicative-weight / mirror descent (workhorse)
# ---------------------------------------------------------------------------
def pressure(a, top_idx, w):
    n = len(a)
    P = np.zeros(n)
    for jj in range(len(top_idx)):
        k = int(top_idx[jj])
        i0 = max(0, k - n + 1)
        i1 = min(n - 1, k)
        if i0 <= i1:
            idx = np.arange(i0, i1 + 1)
            P[idx] += 2.0 * w[jj] * a[k - idx]
    return P


def mw_optimize(a0, time_limit, top_k, eta0, rng, verbose):
    a = norm_sum1(a0)
    best_a = a.copy()
    best_s = local_score(a)
    t0 = time.monotonic()
    eta = float(eta0)
    kick = 0
    kick_every = 6000
    n = len(a)

    it = 0
    stall = 0
    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        M = float(c.max())
        s = 2.0 * n * M
        if s < best_s - 1e-12:
            best_s = s
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        k_max = min(int(top_k), len(c) - 1)
        order = np.argsort(c)[::-1][:k_max]
        tv = np.maximum(c[order], 1e-30)
        w = tv ** 2.0
        wsum = w.sum()
        if wsum <= 0:
            w = np.ones(k_max) / k_max
        else:
            w = w / wsum

        P = pressure(a, order, w)
        Pgmax = P.max()
        a_new = a * np.exp(-eta * (P - Pgmax))
        a_new = np.maximum(a_new, 1e-30)
        st = a_new.sum()
        if st < 1e-15 or not np.isfinite(st):
            a_new = a.copy()
        else:
            a_new = a_new / st
        a = a_new

        it += 1
        # gentle annealing of the step
        eta = max(8.0, eta0 * (0.5 ** (it / 40000.0)))

        # periodic kick: mix with a fresh non-symmetric shape + noise to escape basins
        if stall > kick_every:
            kick += 1
            u = rng.uniform(0.0, 1.0, n)
            mm = norm_sum1(best_a + 0.18 * (u - 0.5) + 0.10 * rng.normal(0.0, 1.0, n))
            a = norm_sum1((1.0 - 0.30) * best_a + 0.30 * mm)
            eta = min(eta0 * 1.5, eta * 1.4)
            stall = 0

        if verbose and it % 8000 == 0:
            el = time.monotonic() - t0
            print("    MW it=%d t=%.0fs J=%.6f best=%.6f eta=%.1f" % (it, el, s, best_s, eta))

    return best_a, best_s

# LP / Frank-Wolfe cutting planes (refiner)
# ---------------------------------------------------------------------------
def lp_frank_wolfe(a0, time_limit, solver, init_K, max_K, add_per_iter, verbose):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac = fast_conv(y)
    best_m = float(ac.max())
    t0 = time.monotonic()

    init_K = min(init_K, 2 * n - 1)
    top_init = np.argsort(ac)[::-1][:init_K]
    constraint_set = list(set(int(k) for k in top_init))
    failed = 0

    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_conv(y)
        Mc = float(ac.max())
        if Mc < best_m - 1e-14:
            best_m = Mc
            best_a = y.copy()

        if len(constraint_set) > max_K:
            margins = np.array([Mc - ac[k] for k in constraint_set])
            keep = np.argsort(margins)[:max_K]
            constraint_set = [constraint_set[i] for i in keep]
        K = len(constraint_set)
        if K == 0:
            break

        A = np.zeros((K, n))
        for r, k in enumerate(constraint_set):
            i0 = max(0, k - n + 1)
            i1 = min(n - 1, k)
            if i0 <= i1:
                idx = np.arange(i0, i1 + 1)
                A[r, idx] = y[k - idx]

        g = cp.Variable(n, nonneg=True)
        Mv = cp.Variable()
        cons = [cp.sum(g) == 1, A @ g <= Mv]
        prob = cp.Problem(cp.Minimize(Mv), cons)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed += 1
            if failed > 5:
                constraint_set = constraint_set[:max(50, init_K // 2)]
                failed = 0
            continue
        if prob.status not in ("optimal", "optimal_inaccurate"):
            continue
        gv = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gs = gv.sum()
        if gs < 1e-15:
            continue
        gv /= gs

        # add violated lags
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(gv, fft_size)
        cg = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]
        viol = np.where(cg > Mc + 1e-9)[0]
        if len(viol):
            amt = cg[viol] - Mc
            topv = viol[np.argsort(amt)[::-1][:add_per_iter]]
            for k in topv:
                constraint_set.append(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * gv
        y_new = np.maximum(y_new, 0.0)
        ys = y_new.sum()
        y = y_new / ys if ys > 1e-15 else y.copy()

        if verbose and t % 40 == 0:
            el = time.monotonic() - t0
            print("    LP it=%d t=%.0fs J=%.6f |S|=%d" % (t, el, 2.0 * n * best_m, len(constraint_set)))

    return best_a, 2.0 * n * best_m

# cheap coordinate polish (mass transfer from the dominant peak pair)
# ---------------------------------------------------------------------------
def coord_polish(a0, time_limit, rng):
    a = norm_sum1(a0)
    n = len(a)
    best_a = a.copy()
    best_m = float(fast_conv(a).max())
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        k = int(np.argmax(c))
        js = np.arange(max(0, k - n + 1), min(n - 1, k) + 1)
        if len(js) == 0:
            break
        prod = a[js] * a[k - js]
        j0 = js[int(np.argmax(prod))]
        i0 = k - j0
        big = int(j0) if a[j0] >= a[i0] else int(i0)
        moved = False
        for _ in range(8):
            dst = int(rng.integers(n))
            if dst == big or a[dst] > 0.5 * a[big]:
                continue
            trial = a.copy()
            for frac in (0.05, 0.12, 0.25):
                amt = a[big] * frac
                tt = a.copy()
                tt[big] -= amt
                tt[dst] += amt
                tt = np.maximum(tt, 0.0)
                ts = tt.sum()
                if ts < 1e-15:
                    continue
                tt /= ts
                nm = float(fast_conv(tt).max())
                if nm < best_m - 1e-14:
                    best_m = nm
                    best_a = tt.copy()
                    a = tt
                    moved = True
                    break
                tt[big] += amt
                tt[dst] -= amt
        if not moved:
            # small random kick
            ii = [int(rng.integers(n)) for _ in range(4)]
            a2 = a.copy()
            for s in ii:
                j = int(rng.integers(n))
                if s != j and a2[s] > 1e-10:
                    amt = a2[s] * 0.03
                    a2[s] -= amt
                    a2[j] += amt
            a2 = np.maximum(a2, 0.0)
            a2 /= a2.sum()
            nm = float(fast_conv(a2).max())
            if nm < best_m - 1e-14:
                best_m, best_a, a = nm, a2.copy(), a2
            else:
                break
    return best_a, 2.0 * n * best_m / (1.0 ** 2) if best_a.sum() >= 0.999 else local_score(best_a)
