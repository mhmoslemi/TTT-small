        step = step0 / (1.0 + 0.005 * t) ** 0.5
        scale = T * a_max * n
        if scale < 1e-30:
            scale = 1e-30
        
        log_a = np.log(np.maximum(a, 1e-30)) - step * g / scale
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate new solution
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
        
        # Damped update
        damp = 0.4 + 0.2 * progress  # increase damping over time
        a = (1.0 - damp) * a + damp * a_new
        
        # Periodic re-evaluation of best
        if t % 5000 == 0:
            c_check = fast_conv(a)
            if c_check.max() < best_m:
                best_m = c_check.max()
                best_a = a.copy()
            
            if verbose:
                print(f"      MD-LSE {label} t={t} ({elapsed:.0f}s): "
                      f"obj={2*n*best_m:.6f}, T={T:.2e}, step={step:.4f}")
        
        # Restart perturbation if stuck
        if t % 10000 == 0 and t > 10000:
            c_check = fast_conv(a)
            if abs(c_check.max() - best_m) < 1e-12:
                # Stuck - add small random perturbation
                rng = np.random.default_rng(t)
                a = norm_simplex(a + rng.normal(0, 0.005, n))
    
    # Final check
    c_final = fast_conv(a)
    if c_final.max() < best_m:
        best_m = c_final.max()
        best_a = a.copy()
    
    return best_a, 2.0 * n * best_m


def spectral_flatten(a, strength=0.3):
    """Compress dynamic range of spectral magnitude."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return a_new


def spectral_optimize(a0, time_limit, strength=0.25, verbose=False):
    """Iterative spectral flattening with adaptive step."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    c = fast_conv(a)
    best_m = c.max()
    t0 = time.monotonic()
    
    step = strength
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        a_flat = spectral_flatten(a, step)
        c_new = fast_conv(a_flat)
        M_new = c_new.max()
        
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_flat.copy()
            a = a_flat
            step = min(0.5, step * 1.03)
            stall = 0
        else:
            a = norm_simplex(0.7 * a + 0.3 * best_a)
            step = max(0.03, step * 0.8)
            stall += 1
        
        if verbose and stall % 10 == 0:
            print(f"      Spectral: step={step:.3f}, obj={2*n*best_m:.6f}")
    
    return best_a, 2.0 * n * best_m


def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions to reduce peak."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    c = fast_conv(a)
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()
        
        # Find top peaks
        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try moving to several random destinations
            for attempt in range(8):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.08, 0.15]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.02, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
    
    return best_a, 2.0 * n * best_m


def phase_perturb(a0, rng, strength=0.3):
    """Perturb the phase of the DFT, keeping amplitude."""
    n = len(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Add random phase perturbation
    phase_pert = phase + strength * rng.normal(0, 1, len(phase))
    
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0) + 0.01
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.3, pos2=0.7, width=0.15):
    x = np.linspace(0, 1, n)
    a = (np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) +
         np.exp(-((x - pos2) ** 2) / (2 * width ** 2))) + 0.05
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2 + 0.02
    return norm_simplex(a)


def make_init_smooth_random(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.1 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.01)
    return norm_simplex(a)


def make_init_asymmetric(n):
    """Asymmetric: high on one side, decaying on the other."""
    x = np.linspace(0, 1, n)
    a = np.exp(-3.0 * x) + 0.2 * np.exp(-3.0 * (1 - x))
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4000
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}, budget={time_budget:.0f}s")
    
    # === Phase 1: Mirror descent with LSE on parent/best (longest run) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MD-LSE on best (n={n0}) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        a_in = norm_simplex(best_seq)
        a_out, obj_out = mirror_descent_lse(a_in, t1, verbose=True, label="parent")
        try_update(a_out, f"mdlse_n{n0}")
        
        # Spectral polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_sp, _ = spectral_optimize(a_out, min(60.0, remaining * 0.15), verbose=False)
            try_update(a_sp, f"spectral_n{n0}")
    
    # === Phase 2: Try different initializations at same n ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Diverse inits (n={n0}) ===")
    t2_total = min(remaining * 0.25, 350.0)
    if t2_total > 60:
        inits = []
        labels = []
        
        inits.append(make_init_bowl(n0, sharpness=2.5))

        labels.append("dual")

        labels.append("smooth1")

        labels.append("sinc")
        
        inits.append(make_init_cosine(n0, mode=2))
        labels.append("cos2")
        
        t_per = t2_total / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            remaining_now = time_budget - 30 - elapsed_so_far
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 8:
                break
            print(f"    MD-LSE '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = mirror_descent_lse(init, t_avail, verbose=False, label=lbl)
            try_update(a_out, f"mdlse_{lbl}")
            
            # Quick spectral polish
            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now > 15:

    # === Phase 3: Phase perturbation for diversity ===
    remaining = time_budget - (time.monotonic() - t_global)

            a_out, _ = mirror_descent_lse(a_pert, t_avail, verbose=False, label=f"ph{i}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_sp, _ = spectral_optimize(a_out, min(30.0, remaining * 0.2), verbose=False)
                try_update(a_sp, f"spec_n{new_n}")

    for delta in [1000, 2000]:

            try_update(a_out, f"mdlse_short_n{new_n}")
    
    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    t5 = min(remaining * 0.15, 150.0)
    if t5 > 15:

        a_out, _ = coordinate_descent(a_in, t5, verbose=False)
        try_update(a_out, f"cd_n{n_best}")

    t6 = min(remaining * 0.35, 250.0)
    if t6 > 20:
        a_in = norm_simplex(best_seq)
        
        # Slight perturbation before final run
        rng6 = np.random.default_rng(777)
        a_in = norm_simplex(a_in + rng6.normal(0, 0.005, n_best))
        
        a_out, _ = mirror_descent_lse(a_in, t6, verbose=True, label="final")
        try_update(a_out, f"mdlse_final_n{n_best}")
        
        # Spectral polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sp, _ = spectral_optimize(a_out, min(40.0, remaining * 0.4), verbose=False)
            try_update(a_sp, f"spec_final_n{n_best}")
    
    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        print(f"\n=== Phase 7: Final CD (n={n_best}) ===")
        t7 = min(remaining * 0.5, 60.0)
        if t7 > 5:

            a_out, _ = coordinate_descent(a_in, t7, verbose=False)

    # Final validation
    if best_seq is not None:

        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
