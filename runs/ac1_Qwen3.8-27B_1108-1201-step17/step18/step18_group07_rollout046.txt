def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_val(a):

def lse_gradient(a, c, beta):
    """Compute gradient of LSE(beta * c) w.r.t. a where c = conv(a,a).
    
    d LSE / da_j = sum_k w_k * dc_k/da_j
    where w_k = exp(beta * c_k) / sum(exp(beta * c_k))
    and dc_k/da_j = 2*a[k-j] for 0 <= k-j < n, else 0.
    """
    n = len(a)
    c_max = c.max()
    exponents = beta * (c - c_max)
    exponents = np.maximum(exponents, -80.0)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    # Only process significant weights
    threshold = w.max() * 1e-8
    active = np.where(w > threshold)[0]
    for k in active:
        wk = w[k]
        if wk < 1e-12:
            continue
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start >= j_end:
            continue
        j = np.arange(j_start, j_end)
        grad[j] += 2.0 * wk * a[ki - j]
    return grad

    n = len(a0)
    a = norm_simplex(a0)

    t0 = time.monotonic()

def spectral_step(a, extent=0.3):
    """One step of spectral flattening."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - extent)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    blend = 0.5 * extent
    a_out = (1.0 - blend) * a + blend * a_new
    return norm_simplex(a_out)

def freq_design_init(n, mode='flat'):
    """Design initial sequence in frequency domain."""
    if mode == 'flat':
        # Flat frequency response (ideal low autocorrelation)
        A = np.ones(n // 2 + 1)
        a = np.fft.irfft(A, n)
    elif mode == 'raised_cosine':
        f = np.linspace(0, 1, n // 2 + 1)
        A = np.cos(np.pi * (f - 0.5)) ** 2
        A = np.maximum(A, 0.1)
        a = np.fft.irfft(A, n)
    elif mode == 'tapered':
        f = np.linspace(0, 1, n // 2 + 1)
        A = 1.0 - 0.5 * f
        a = np.fft.irfft(A, n)
    elif mode == 'bessel':
        from scipy.special import j0
        f = np.linspace(0, 1, n // 2 + 1)
        A = j0(np.pi * f)
        A = np.maximum(A, 0.05)
        a = np.fft.irfft(A, n)
    else:
        a = np.ones(n)
    
    a = np.real(a[:n])
    a = np.maximum(a, 0.0)
    return norm_simplex(a)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    print(f"\n=== Phase 2: Peak reduction (n={n_best}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = peak_reduction(a_in, t2, verbose=False)
        try_update(a_out, f"peakred_n{n_best}")
    
    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coordinate descent (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 10:
        a_in = norm_simplex(best_seq)
        a_out, _ = coordinate_descent(a_in, t3)
        try_update(a_out, f"cd_n{n_best}")
    
    # === Phase 4: Hybrid polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Hybrid polish (n={n_best}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = hybrid_polish(a_in, t4)
        try_update(a_out, f"hybrid_n{n_best}")
    
    # === Phase 5: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length variation ===")
    
    lengths_to_try = []
    for delta in [500, 1000, 2000]:
        lengths_to_try.append(n_best + delta)
    for delta in [500, 1000, 2000]:
        if n_best - delta >= 1000:
            lengths_to_try.append(n_best - delta)
    
    # Also try some standard lengths
    for standard_n in [3000, 4000, 5000, 6000]:
        if standard_n not in lengths_to_try and 1000 <= standard_n <= 8000:
            lengths_to_try.append(standard_n)
    
    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: abs(x - n_best))
    
    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        if new_n < 500 or new_n > 10000:
            continue
        print(f"  Trying n={new_n}...")
        a_cand = interpolate_sequence(best_seq, new_n)
        try_update(a_cand, f"interp_n{new_n}")
        
        t5 = min(remaining * 0.2, 120.0)
        if t5 > 15:
            a_out, _ = ftrl_mirrondescent(a_cand, t5)
            try_update(a_out, f"md_n{new_n}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_sp, _ = spectral_optimize(a_out, min(20.0, remaining * 0.3))
                try_update(a_sp, f"spectral_n{new_n}")
    
    # === Phase 6: Final intensive optimization on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final optimization (n={n_best}) ===")
    t6 = min(remaining * 0.35, 300.0)
    if t6 > 20:
        a_in = norm_simplex(best_seq)
        
        # Slight perturbation for diversity
        rng6 = np.random.default_rng(555)
        perturbed = norm_simplex(a_in + rng6.normal(0, 0.005, n_best))
        a_out, _ = ftrl_mirrondescent(perturbed, t6 * 0.6)
        try_update(a_out, f"md_final1_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_sp, _ = spectral_optimize(a_out, min(t6 * 0.3, remaining * 0.5))
            try_update(a_sp, f"spectral_final1_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_cd, _ = coordinate_descent(a_out, min(t6 * 0.4, remaining * 0.5))
            try_update(a_cd, f"cd_final1_n{n_best}")
    
    # === Phase 7: Spectral-only extended ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Extended spectral ({n_best}) ===")
    t7 = min(remaining * 0.30, 180.0)
    if t7 > 10:
        a_in = norm_simplex(best_seq)
        a_out, _ = spectral_optimize(a_in, t7)
        try_update(a_out, f"spectral_ext_n{n_best}")
        
        # CD after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_cd, _ = coordinate_descent(a_out, min(remaining * 0.5, 60.0))
            try_update(a_cd, f"cd_after_spec_n{n_best}")
    
    # === Phase 8: Final micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        print(f"\n=== Phase 8: Final micro-polish ===")
        t8 = min(remaining * 0.6, 80.0)
        if t8 > 5:
            a_in = norm_simplex(best_seq)
            # Very short FTRL with high beta
            a_out, _ = ftrl_mirrondescent(a_in, t8 * 0.5)
            try_update(a_out, f"md_micro_n{n_best}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 5:
                a_sp, _ = spectral_optimize(a_out, min(t8 * 0.3, remaining * 0.5))
                try_update(a_sp, f"spec_micro_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_init_bowl(n_fb, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
