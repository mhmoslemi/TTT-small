    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    y = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    gamma = c0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        if symmetric:
            y = make_symmetric(y)
        else:
            sys_ = y.sum()
            if sys_ < 1e-15:
                y = x.copy()
            else:
                y = y / sys_

        ac = fast_autocorr(y)
        grad = compute_soft_grad(y, ac, cur_top_k, cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y + 1e-30) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        if symmetric:
            a_new = make_symmetric(a_new)
        else:
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / sn

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            noise_level = 0.04 / (1.0 + 0.04 * (stall if stall > 0 else 1))
            noise_level = min(noise_level, 0.04)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            if symmetric:
                x_r = make_symmetric(x_r)
            else:
                sr = x_r.sum()
                if sr < 1e-15:
                    x_r = best_a.copy()
                else:
                    x_r = x_r / sr
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 0.3, 2.0)
            gamma = max(gamma * 0.93, 1.0)
            if verbose:
                elapsed = time.monotonic() - t0
                print("    NMD restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(2.0 * n * best_m))

    return best_a, best_m


def build_constraint_matrix(a, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        ki = int(k)
        i_start = max(0, ki - (n - 1))
        i_end = min(n - 1, ki)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000,
          add_per_iter=25, verbose=True, symmetric=False):
    n = len(a0)
    a = norm_sum1(a0)
    if symmetric:
        a = make_symmetric(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 1000

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: converged at iter " + str(t))
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        if symmetric:
            half = (n + 1) // 2
            coeffs_half = np.zeros((K, half))
            for jj, k in enumerate(constraint_list):
                ki = int(k)
                i_start = max(0, ki - (n - 1))
                i_end = min(n - 1, ki)
                if i_start >= i_end:
                    continue
                idx = np.arange(i_start, i_end)
                vals = y[ki - idx]
                for ii in range(len(idx)):
                    if getattr(lp_fw, '_force_skip', False):
                        break
                    ih = min(idx[ii], n - 1 - idx[ii])
                    if ih < half:
                        coeffs_half[jj, ih] += vals[ii]
            g_half = cp.Variable(half, nonneg=True)
            M_var = cp.Variable()
            if n % 2 == 0:
                sum_c = 2.0 * cp.sum(g_half)
            else:
                sum_c = 2.0 * cp.sum(g_half) - g_half[half - 1]
            constraints = [sum_c == 1, coeffs_half @ g_half <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue
            gh = np.maximum(np.asarray(g_half.value, dtype=np.float64), 0.0)
            g_dir = np.zeros(n)
            for i in range(half):
                g_dir[i] = gh[i]
                g_dir[n - 1 - i] = gh[i]
            if n % 2 == 1:
                g_dir[half - 1] = gh[half - 1]
        else:
            coeffs = build_constraint_matrix(y, constraint_list, n)
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                if len(active_set) > initial_K:
                    active_set = set(int(constraint_list[i]) for i in range(min(initial_K, len(constraint_list))))
                continue
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue
            g_dir = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)

        gs = g_dir.sum()
        if gs < 1e-15:
            continue
        g_dir = g_dir / gs

        size_fc = 2 * n - 1
        fft_fc = 1 << (size_fc - 1).bit_length()
        Yf = np.fft.rfft(y, fft_fc)
        Gf = np.fft.rfft(g_dir, fft_fc)
        conv_ys = np.fft.irfft(Yf * Gf, fft_fc)[:size_fc]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for kk in top_v:
                active_set.add(int(kk))

        default_alpha = 2.0 / (t + 2.0)
        alphas_to_try = [default_alpha,
                         max(0.01, default_alpha * 0.5),
                         max(0.01, default_alpha * 2.0),
                         max(0.01, 1.0 / (t + 1.0)),
                         max(0.01, 0.5 * default_alpha + 0.02)]
        alphas_to_try = [min(1.0, max(0.01, al)) for al in alphas_to_try]

        best_alpha = default_alpha
        best_trial_M = fast_autocorr((1.0 - default_alpha) * y + default_alpha * g_dir).max()
        for al in alphas_to_try:
            if al == default_alpha:
                continue
            trial = (1.0 - al) * y + al * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            if symmetric:
                trial = make_symmetric(trial)
            Mt = fast_autocorr(trial).max()
            if Mt < best_trial_M:
                best_trial_M = Mt
                best_alpha = al

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            if symmetric:
                y_new = make_symmetric(y_new)
            else:
                sy = y_new.sum()
                if sy < 1e-15:
                    y_new = y.copy()
                else:
                    y_new = y_new / sy

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def peak_shave(a0, time_limit, rng_seed=99, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:20]
        improved = False

        for ki in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(ki)
            js = max(0, k_int - (n - 1))
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            cmax = contrib.max()
            if cmax < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - (n - 1))
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - (n - 1))
                e2 = min(n - 1, k2i)
                idx2 = np.arange(s2, e2 + 1)
                scores[idx2] += a[k2i - idx2]
                scores[k2i - idx2] += a[idx2]

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])
            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.008, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                Mt = fast_autocorr(a_trial).max()
                if Mt < best_m - 1e-16:
                    a = a_trial
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 200:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mnr = fast_autocorr(a).max()
                        if Mnr < best_m:
                            best_m = Mnr
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mnr = fast_autocorr(a).max()
                        if Mnr < best_m:
                            best_m = Mnr
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            print("    Shave (" + str(elapsed) + "s): J=" + repr(2.0 * n * best_m) + " stall=" + str(stall))

    return best_a, best_m


def parametric_init(n, time_limit):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    candidates.append(norm_sum1(np.sin(np.pi * x) ** 2))
    candidates.append(norm_sum1(np.ones(n) / n))

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            c = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(norm_sum1(c))

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.5]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))))

    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1(x ** p + (1.0 - x) ** p))

    for freq in [1, 2, 3, 4]:
        for ph in [0.0, 0.5, 1.0]:
            c = np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.005)
            candidates.append(norm_sum1(c))

    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50]:
            candidates.append(norm_sum1(1.0 / (1.0 + np.exp(-s * (x - c)))))

    for rate in [0.5, 1, 2, 5, 10]:
        candidates.append(norm_sum1(np.exp(-rate * x) + np.exp(rate * (x - 1.0))))

    for w1 in [0.1, 0.2, 0.3, 0.5]:
        c = (1.0 - w1) * np.ones(n) + w1 * (i_arr / max(n - 1, 1))
        candidates.append(norm_sum1(c))
        c2 = (1.0 - w1) * np.ones(n) + w1 * (1.0 - i_arr / max(n - 1, 1))
        candidates.append(norm_sum1(c2))

    for w1 in [0.1, 0.2, 0.3, 0.5]:
        c = (1.0 - w1) * np.ones(n) + w1 * (i_arr / max(n - 1, 1)) ** 2
        c = np.maximum(c, 0.01)
        candidates.append(norm_sum1(c))
        c2 = (1.0 - w1) * np.ones(n) + w1 * (1.0 - i_arr / max(n - 1, 1)) ** 2
        c2 = np.maximum(c2, 0.01)
        candidates.append(norm_sum1(c2))

    rng = np.random.default_rng(42)
    for _ in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0))
        candidates.append(a)

    best_a = None
    best_val = float('inf')
    for c in candidates:
        v = fast_autocorr(c).max()
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, 2.0 * n * best_val


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 8000
        a_init, _ = parametric_init(n_base, 30.0)
        best_seq = a_init
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    def try_update(bs, bsc, a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")

    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: LP-FW asymmetric (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30 and n0 <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t1, solver, initial_K=350, max_K=2500, add_per_iter=30, verbose=True, symmetric=False)
        try_update(best_seq, best_score, a_out, "lp_asym_n" + str(n0))

    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Symmetric LP-FW (n=" + str(n0) + ") ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 20 and n0 <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver, initial_K=250, max_K=1800, add_per_iter=20, verbose=True, symmetric=True)
        try_update(best_seq, best_score, a_out, "lp_sym_n" + str(n0))

    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Nesterov MD asymmetric (n=" + str(n0) + ") ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t3, top_k=80, beta=5.0, c0=4.0, restart_every=2500, symmetric=False, verbose=True)
        try_update(best_seq, best_score, a_out, "nmd_n" + str(n0))

    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Nesterov MD symmetric (n=" + str(n0) + ") ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 20:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t4, top_k=70, beta=4.5, c0=3.5, restart_every=2000, symmetric=True, verbose=True)
        try_update(best_seq, best_score, a_out, "nmd_sym_n" + str(n0))

    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 5: Peak shave (n=" + str(n0) + ") ===")
    t5 = min(remaining * 0.06, 100.0)
    if t5 > 15:
        a_out, _ = peak_shave(norm_sum1(best_seq), t5, rng_seed=77, verbose=True)
        try_update(best_seq, best_score, a_out, "shave_n" + str(n0))

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    target_n = n_best + 1500
    if target_n > n_best and target_n <= 10000 and remaining > 120:
        print("\n=== Phase 6: Upscale to n=" + str(target_n) + " ===")
        a_long = interpolate_seq(best_seq, target_n)
        try_update(best_seq, best_score, a_long, "interp_n" + str(target_n))

        t6a = min(remaining * 0.25, 180.0)
        if t6a > 30:
            a_out, _ = lp_fw(a_long, t6a, solver, initial_K=300, max_K=2000, add_per_iter=25, verbose=True, symmetric=False)
            try_update(best_seq, best_score, a_out, "lp_long_n" + str(target_n))

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.30, 150.0)
            if t6b > 20:
                a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t6b, top_k=70, beta=4.5, c0=3.5, restart_every=2000, symmetric=False, verbose=True)
                try_update(best_seq, best_score, a_out, "nmd_long_n" + str(target_n))

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 8000:
        t7 = min(remaining * 0.30, 200.0)
        if t7 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t7, solver, initial_K=400, max_K=3000, add_per_iter=35, verbose=True, symmetric=False)
            try_update(best_seq, best_score, a_out, "lp_final_n" + str(n_best))

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final peak shave (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t8 = min(remaining * 0.50, 120.0)
        if t8 > 10:
            a_out, _ = peak_shave(norm_sum1(best_seq), t8, rng_seed=314, verbose=True)
            try_update(best_seq, best_score, a_out, "shave_final_n" + str(n_best))

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t9 = min(remaining * 0.6, 80.0)

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global

    else:

        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]

Wait, I used `nonlocal` in `try_update` which is a closure. The rules say "Make all helper functions top level and have no closures from function nesting. Don't use any lambda functions." Let me fix this by removing the nested function and doing the update inline.

Let me also reconsider the `build_constraint_matrix` - I was using columns indexed by `idx` but I need to make sure the LP is correct. The constraint is `(y * g)[k] <= M` which is `sum_i y[ki-i] * g[i] <= M` (with valid indices). So the coefficient of `g[i]` in row k is `y[ki-i]`.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse parametric + Fourier-based initializations
2. Nesterov-accelerated mirror descent with peak-aware soft gradient
3. Improved LP-Frank-Wolfe with backtracking line search and adaptive active set
4. Symmetric-constrained LP-FW (halves problem size)
5. Multi-scale refinement: upsample and refine at larger n
6. Peak-shaving local refinement with direct mass redistribution
"""

import numpy as np
import time
import cvxpy as cp

_solver_cache = None


def get_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _solver_cache = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _solver_cache


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(max(len(a), 1)) / max(len(a), 1)
    return a / s
