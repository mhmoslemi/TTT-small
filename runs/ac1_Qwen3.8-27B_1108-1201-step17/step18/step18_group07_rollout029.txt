Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based active set optimization
combined with spectral smoothing and FTRL refinement.

import numpy as np
import time
from scipy.optimize import linprog

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_conv_ab(a, b):
    """Fast linear convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def norm_simplex(a):
    """Normalize to probability simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_constraint_matrix(a, peak_indices):
    """Build constraint matrix for LP: rows correspond to peak constraints.
    
    Constraint: sum_j a[peak-k - j] * b[j] <= M for valid j.
    A[row, col] = a[peak - col] if 0 <= peak - col < n, else 0.
    """
    n = len(a)
    K = len(peak_indices)
    A = np.zeros((K, n))
    for i, k in enumerate(peak_indices):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            cols = np.arange(j_start, j_end)
            rows_idx = k_int - cols
            A[i, cols] = a[rows_idx]
    return A

def lp_optimize(a, time_limit, K=50, verbose=False):
    """LP-based active set optimization.
    
    At each iteration, solve:
        max sum(b) s.t. conv(a,b)[k] <= M for k in active_set, b >= 0
    Then update a toward b.
    """
    n = len(a)
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    step = 0.3
    stall = 0
    max_stall = 30
    
    iteration = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iteration += 1
        c = fast_conv(a)
        M = c.max()
        
        if M < best_m:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Get active set: top K peaks
        K_cur = min(K, len(c))
        top_idx = np.argsort(c)[::-1][:K_cur]
        
        # Build constraint matrix
        A_mat = build_constraint_matrix(a, top_idx)
        
        # Solve LP: max sum(b) s.t. A_mat @ b <= M, b >= 0
        c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
        b_ub = np.full(K_cur, M)
        
        try:
            result = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, 
                           bounds=[(0, None)] * n, method='highs')
        except Exception:
            break
        
        if not result.success:
            # Try with fewer constraints
            K_cur = max(10, K_cur // 2)
            top_idx = np.argsort(c)[::-1][:K_cur]
            A_mat = build_constraint_matrix(a, top_idx)
            b_ub = np.full(K_cur, M)
            try:
                result = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                               bounds=[(0, None)] * n, method='highs')
            except Exception:
                break
            
            if not result.success:
                break
        
        b = result.x
        b = np.maximum(b, 0.0)
        s_b = b.sum()
        if s_b < 1e-15:
            break
        b = b / s_b  # Normalize to sum=1
        
        # Update
        a_new = (1 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check if improvement
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < M:
            a = a_new
            # Increase step
            step = min(0.5, step * 1.02)
            stall = 0
        else:
            # Reduce step
            step = max(0.02, step * 0.7)
            stall += 1
        
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iteration} ({elapsed:.1f}s): M={best_m:.8f}, step={step:.3f}")
    
    return best_a, best_m

def spectral_smooth_step(a, axtent=0.3):
    """One step of spectral flattening."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - axtent)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    a_out = (1.0 - axtent * 0.5) * a + axtent * 0.5 * a_new
    return norm_simplex(a_out)

def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    step = 0.3
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        a = spectral_smooth_step(a, axtent=step)
        c = fast_conv(a)
        M = c.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            step = min(0.5, step * 1.03)
        else:
            a = norm_simplex(0.5 * a + 0.5 * best_a)
            step = max(0.05, step * 0.8)
            if step < 0.05:
                break
    
    return best_a, best_m

def ftrl_optimize(a0, time_limit, verbose=False):
    """FTRL mirror descent with annealed temperature."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    T_init = best_m * 0.5
    T_min = best_m * 0.01
    top_k_init = 50
    top_k_min = 5
    
    x = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, t / 30000.0)
        T = T_init * (T_min / T_init) ** progress
        cur_top_k = max(top_k_min, int(top_k_init * (1 - 0.8 * progress) + top_k_min))
        cur_beta = 20.0 + 60.0 * progress
        
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        c = fast_conv(y)
        
        # Softmax-weighted subgradient
        k_max = min(cur_top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = c[top_idx]
        max_v = top_vals[0]
        if max_v < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            exponents = cur_beta * (top_vals - max_v) / max_v
            exponents = np.maximum(exponents, -50.0)
            w = np.exp(exponents)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        eta = 1.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x = a_new
        a = a_new
        
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
        
        if t % 5000 == 0:
            a = spectral_smooth_step(a, axtent=0.3)
            c_check = fast_conv(a)
            if c_check.max() < best_m:
                best_m = c_check.max()
                best_a = a.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        n_top = min(10, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            for attempt in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m

def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)

def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return norm_simplex(a)

def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_simplex(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based active set optimization
combined with spectral smoothing, FTRL refinement, and coordinate descent polishing.

Key innovation: At each iteration, solve a linear program to find the optimal
direction that maximizes mass redistribution while keeping cross-correlation
with the current sequence bounded by the current peak. This provides a
principled search direction that converges faster than subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Fast linear convolution a*b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))

def build_constraint_matrix(a, peak_indices):
    """Build LP constraint matrix for cross-correlation constraints.
    
    For each peak index k, the constraint is:
        sum_j a[k-j] * b[j] <= M  (for valid j)
    So A[i, j] = a[k_i - j] if 0 <= k_i - j < n, else 0.
    """
    n = len(a)
    K = len(peak_indices)
    A = np.zeros((K, n))
    for i in range(K):
        k_int = int(peak_indices[i])
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            cols = np.arange(j_start, j_end)
            row_vals = k_int - cols
            A[i, cols] = a[row_vals]
    return A

def time_budget_remaining(t0, time_limit):
    """Helper to compute remaining time."""
    return max(0.0, time_limit - (time.monotonic() - t0))


def spectral_smooth_step(a, axtent=0.3):
    """One step of spectral flattening: compress dynamic range of |A|."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a

    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - axtent)
    mag_new = np.maximum(mag_new, 1e-30)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

    a_out = (1.0 - axtent * 0.5) * a + axtent * 0.5 * a_new
    return norm_simplex(a_out)

        max_v = top_vals[0]
        if max_v < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            exponents = cur_beta * (top_vals - max_v) / max_v
            exponents = np.maximum(exponents, -50.0)
            w = np.exp(exponents)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = 1.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x = a_new
        a = a_new

        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()

        # Periodic spectral smoothing
        if t % 5000 == 0:
            a = spectral_smooth_step(a, axtent=0.25)
            c_check = fast_conv(a)
            if c_check.max() < best_m:
                best_m = c_check.max()
                best_a = a.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        n_top = min(15, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue

            for attempt in range(8):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.15]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break

    return best_a, best_m


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_simplex(a)


def make_init_rand_smooth(n, seed=0, max_freq=15):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, max_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining LP, spectral, FTRL, and CD methods."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP-based optimization with parent (main workhorse) ===
    print(f"\n=== Phase 1: LP Active Set (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t
    if t > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = lp_optimize(a_in, t, K=50, verbose=True)
        try_update(a_out, f"lp_main_n{n0}")

    # === Phase 2: Spectral polish after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 2: Spectral Polish ===")
        t2 = min(remaining * 0.1, 60.0)
        if t2 > 10:
            a_in = norm_simplex(best_seq)
            a_out, _ = spectral_optimize(a_in, t2)
            try_update(a_out, f"spectral_n{n0}")

    # === Phase 3: LP with different K values ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP with varying K ===")
    for K_val in [30, 80, 100]:
        remaining = time_budget - 30 - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t3 = min(remaining * 0.15, 150.0)
        if t3 > 15:
            print(f"  LP K={K_val} (budget={t3:.0f}s)...")
            a_in = norm_simplex(best_seq)
            a_out, _ = lp_optimize(a_in, t3, K=K_val, verbose=False)
            try_update(a_out, f"lp_K{K_val}_n{n0}")

    # === Phase 4: Multi-initialization LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Multi-init LP (n={n_best}) ===")
    t4_total = min(remaining * 0.15, 250.0)

    if t4_total > 30:
        inits = []
        labels = []
        inits.append(make_init_bowl(n_best, sharpness=2.5))
        labels.append("bowl2.5")
        inits.append(make_init_dual_peak(n_best, 0.2, 0.8, 0.12))
        labels.append("dualpeak")
        inits.append(make_init_rand_smooth(n_best, seed=42))
        labels.append("rand42")
        inits.append(make_init_rand_smooth(n_best, seed=123))
        labels.append("rand123")

        t_per = t4_total / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - 30 - (time.monotonic() - t_global)
            if remaining < 10:
                break
            t_avail = min(t_per, remaining)
            if t_avail < 8:
                break
            print(f"  LP '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = lp_optimize(init, t_avail, K=50, verbose=False)
            try_update(a_out, f"lp_{lbl}")

    # === Phase 5: FTRL refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: FTRL refinement (n={n_best}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = ftrl_optimize(a_in, t5, verbose=False)
        try_update(a_out, f"ftrl_n{n_best}")

        # Quick spectral after FTRL
        remaining = time_budget - 30 - (time.monotonic() - t_global)
        if remaining > 10:
            a_sp, _ = spectral_optimize(a_out, min(15.0, remaining * 0.3))
            try_update(a_sp, f"spectral_ftrl_n{n_best}")

    # === Phase 6: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length variation ===")

    for delta in [500, 1000, -500, -1000]:
        new_n = n_best + delta
        if new_n < 1500 or new_n > 10000:
            continue
        remaining = time_budget - 30 - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}...")
        a_new = interpolate_sequence(best_seq, new_n)
        try_update(a_new, f"interp_n{new_n}")

        t6 = min(remaining * 0.25, 120.0)
        if t6 > 15:
            a_out, _ = lp_optimize(a_new, t6, K=40)
            try_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final intensive loop ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final optimization (n={n_best}) ===")

    # Alternate LP and spectral
    for round_idx in range(3):
        remaining = time_budget - 30 - (time.monotonic() - t_global)
        if remaining < 20:
            break
        t7 = min(remaining * 0.3, 100.0)
        if t7 > 10:
            a_in = norm_simplex(best_seq)
            a_out, _ = lp_optimize(a_in, t7, K=60, verbose=False)
            try_update(a_out, f"lp_r{round_idx}_n{n_best}")

            remaining = time_budget - 30 - (time.monotonic() - t_global)
            if remaining > 10:
                a_sp, _ = spectral_optimize(a_out, min(20.0, remaining * 0.2))
                try_update(a_sp, f"spectral_r{round_idx}_n{n_best}")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        print(f"\n=== Phase 8: Final CD (n={n_best}) ===")
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 5:
            a_in = norm_simplex(best_seq)
            a_out, _ = coordinate_descent(a_in, t8, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
