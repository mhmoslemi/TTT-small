    # So: grad_k[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    # Wait, but k can range from 0 to 2n-2. If k-i >= n or k-i < 0, then a[k-i] doesn't exist.
    # In that case, the term a[i]*a[k-i] doesn't appear in the sum, so the gradient is 0.
    # But wait, we also need to check: does a[j]*a[k-j] with j=k-i (so a[k-i]*a[i]) appear?
    # This requires 0 <= k-i < n (for j=k-i to be valid) and 0 <= i < n (which it is).
    # So the condition is: 0 <= k-i < n.
    
    # Actually I realize I need to be even more careful. The sum is:
    # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
    # The valid j range is: max(0, k-n+1) <= j <= min(n-1, k)
    
    # For a specific i, a[i] appears in the sum in two ways:
    # 1. As a[j] with j=i: contributes a[i]*a[k-i] if 0<=i<n (always) and 0<=k-i<n
    # 2. As a[k-j] with k-j=i, i.e., j=k-i: contributes a[k-i]*a[i] if 0<=k-i<n and 0<=i<n (always)
    # So the total contribution of a[i] to conv[k] is:
    # - If 0 <= k-i < n: a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i]
    #   BUT if i = k-i (i.e., k=2i), we only have ONE term, not two.
    #   Actually no: when i = k-i, j=i gives the term a[i]*a[i], and j=k-i=i gives the SAME term.
    #   So in the sum, a[i]*a[i] appears only once (with coefficient 1).
    #   d/d(a[i]) (a[i]*a[i]) = 2*a[i]
    # - If not (0 <= k-i < n): a[i] doesn't appear in the sum, gradient = 0
    
    # So the gradient is:
    # grad_k[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    # (This formula works even when i = k-i, giving 2*a[i], which is correct.)
    
    # Now, for the linearized QP:
    # Minimize t
    # Subject to: conv[k] + grad_k · (a - a_cur) <= t for k in active_lags
    #              sum(a) = 1
    #              a >= 0
    
    # This is a linear program! (Linear objective, linear constraints)
    # Let's use scipy.optimize.linprog.
    
    # Variables: a[0], ..., a[n-1], t
    # But t only appears in the objective and the inequality constraints, so we can eliminate it:
    # Minimize max_k (conv[k] + grad_k · delta) where delta = a - a_cur
    # Subject to: a >= 0, sum(a) = 1
    
    # This is still not a standard LP form. Let me keep t as a variable.
    
    # Decision variables: [a_0, ..., a_{n-1}, t] (n+1 variables)
    # Minimize: t (i.e., [0, 0, ..., 0, 1])
    # Subject to:
    #   For each k in active_lags: grad_k · a - t <= conv[k] - grad_k · a_cur
    #   sum(a) = 1: a_0 + ... + a[n-1] = 1
    #   a_i >= 0 for all i
    #   t is free (unbounded below, but the constraints will bound it)
    
    # Actually, for linprog, all variables must be >= 0 (unless we use bounds).
    # Let me use bounds: a_i in [0, inf], t in [-inf, inf]
    
    # Number of inequality constraints: len(active_lags)
    # A_ineq[:, i] = grad_k[i] for a_i
    # A_ineq[:, n] = -1 for t
    # b_ineq[k] = conv[k] - grad_k · a_cur
    
    # Number of equality constraints: 1 (sum a = 1)
    # A_eq[:, :n] = 1
    # A_eq[:, n] = 0
    # b_eq = [1]
    
    # c = [0, 0, ..., 0, 1]
    
    # This is O(n * |active|) in constraint size, which should be manageable for n up to a few thousand.
    
    grad_matrix = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        k_int = int(k)
        # grad_k[i] = 2*a[k-i] if 0 <= k-i < n
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        # For i in [i_start, i_end-1], k-i is in [0, n-1]
        k_minus_i = k_int - np.arange(i_start, i_end)
        grad_matrix[idx, i_start:i_end] = 2.0 * a[k_minus_i]
    
    # Compute b_ineq
    grad_dot_a = grad_matrix @ a
    b_ineq = conv[active_lags] - grad_dot_a
    
    # Build LP
    c_lp = np.zeros(n + 1)
    c_lp[n] = 1.0  # minimize t
    
    A_ineq = np.zeros((len(active_lags), n + 1))
    A_ineq[:, :n] = grad_matrix
    A_ineq[:, n] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(None, None)]
    
    from scipy.optimize import linprog
    result = linprog(c_lp, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        a_new = result.x[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            return a_new
    return a  # No improvement, return original


def sa_optimize(a0, time_limit, seed=42, T0=0.001, T_end=0.0001, cooling=0.999):
    """Simulated annealing optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    T = T0
    
    while time.monotonic() - t0 < time_limit:
        # Propose a move
        move_type = rng.random()
        
        if move_type < 0.4:
            # Single element adjustment
            i = int(rng.integers(n))
            delta = rng.normal(0, 0.01) * a[i]
            a_new = a.copy()
            a_new[i] = max(0, a[i] + delta)
            a_new = norm_sum1(a_new)
        elif move_type < 0.7:
            # Pair swap (move mass from i to j)
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a_new = a.copy()
                a_new[i] = max(0, a[i] - amt)
                a_new[j] = a[j] + amt
                a_new = norm_sum1(a_new)
            else:
                a_new = a.copy()
        elif move_type < 0.85:
            # Block move (shift a block of elements)
            block_len = max(10, n // 100)
            i = int(rng.integers(n - block_len))
            j = int(rng.integers(n - block_len))
            if abs(i - j) > block_len:
                a_new = a.copy()
                a_new[j:j+block_len] = a_new[i:i+block_len].copy()
                a_new = norm_sum1(a_new)
            else:
                a_new = a.copy()
        else:
            # Gaussian perturbation of a random block
            block_len = max(50, n // 20)
            i = int(rng.integers(n - block_len))
            pert = rng.normal(0, 0.05, block_len)
            a_new = a.copy()
            a_new[i:i+block_len] = np.maximum(0, a[i:i+block_len] + pert)
            a_new = norm_sum1(a_new)
        
        # Evaluate
        M_new = fast_conv(a_new).max()
        J_new = 2.0 * n * M_new
        J_cur = 2.0 * n * fast_conv(a).max()
        
        # Accept/reject
        if J_new < J_cur or rng.random() < np.exp(-(J_new - J_cur) / T):
            a = a_new
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
        
        # Cool
        T = max(T_end, T * cooling)
        
        # Safety check
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_J / (2.0 * n)


def freq_domain_smooth(a, alpha=0.1):
    """Apply frequency-domain low-pass filtering then re-normalize."""
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    # Apply Gaussian low-pass in frequency domain
    freqs = np.fft.rfftfreq(fft_size)
    sigma = n * alpha
    lp = np.exp(-freqs**2 / (2 * sigma**2))
    A_filt = A * lp
    a_filt = np.fft.irfft(A_filt, fft_size)[:n]
    a_filt = np.maximum(a_filt, 0)
    return norm_sum1(a_filt)

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_active_lags(conv, fraction=0.08, min_count=5, max_count=40):
    conv_max = conv.max()
    threshold = conv_max * (1.0 - fraction)
    active = np.where(conv >= threshold)[0]
    if len(active) < min_count:
        active = np.argsort(conv)[::-1][:max(min_count, 5)]
    if len(active) > max_count:

def lp_active_step(a, conv, active_lags):
    n = len(a)
    n_active = len(active_lags)

    grad_matrix = np.zeros((n_active, n))
    for idx, k in enumerate(active_lags):

        i_start = max(0, k_int - n + 1)

    b_ineq = conv[active_lags] - grad_dot_a

    A_ineq[:, n] = -1.0

    bounds = [(0.0, None)] * n + [(None, None)]

            a_new = np.maximum(result.x[:n], 0.0)

            # Local smooth perturbation
            block = max(20, n // 50)
            i = int(rng.integers(max(1, n - block)))
            pert = rng.normal(0, 0.02, block)
            a_new = a.copy()
            a_new[i:i + block] = np.maximum(0, a[i:i + block] + pert)
            a_new = norm_sum1(a_new)
        elif move_type < 0.88:
            # Swap two blocks
            bl = max(10, n // 100)
            i = int(rng.integers(max(1, n - bl)))
            j = int(rng.integers(max(1, n - bl)))
            if abs(i - j) > bl:
                a_new = a.copy()
                tmp = a_new[i:i + bl].copy()
                a_new[i:i + bl] = a_new[j:j + bl]
                a_new[j:j + bl] = tmp
                a_new = norm_sum1(a_new)
            else:
                a_new = a.copy()
        else:
            # Reverse a random block (structural move)
            bl = max(20, min(n // 10, int(rng.integers(20, n // 5 + 1))))
            i = int(rng.integers(max(1, n - bl)))
            a_new = a.copy()
            a_new[i:i + bl] = a_new[i:i + bl][::-1]
            a_new = norm_sum1(a_new)

        M_new = fast_conv(a_new).max()
        delta_J = M_new - cur_M

        if delta_J < 0 or (T > T_end and rng.random() < np.exp(-delta_J / T)):
            a = a_new
            cur_M = M_new
            if M_new < best_M - 1e-15:
                best_M = M_new
                best_a = a.copy()
                improve_count += 1

        # Cooling
        T = max(T_end, T * 0.9995)
        it += 1

        if verbose and it % 5000 == 0:
            J = 2.0 * n * best_M
            print(f"    SA it={it}: J={J:.6f}, T={T:.6f}, improves={improve_count}")

    return best_a, best_M


def freq_smooth(a, alpha=0.05):
    """Gaussian low-pass in frequency domain."""
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    freqs = np.fft.rfftfreq(fft_size)
    sigma = max(1.0, n * alpha)
    lp = np.exp(-freqs ** 2 / (2.0 * sigma ** 2))
    A_filt = A * lp
    a_filt = np.fft.irfft(A_filt, fft_size)[:n]
    a_filt = np.maximum(a_filt, 0.0)
    return norm_sum1(a_filt)


def multi_res_optimize(a0, time_limit, seed=42):
    """Optimize at multiple resolutions."""
    n0 = len(a0)
    best_a = a0.copy()
    best_M = fast_conv(a0).max()
    t0 = time.monotonic()

    resolutions = sorted(set([max(200, n0 // 8), max(400, n0 // 4), max(800, n0 // 2)]))
    for n_res in resolutions:
        if time.monotonic() - t0 > time_limit - 20:
            break
        t_avail = min((time_limit - 20 - (time.monotonic() - t0)) / max(1, len(resolutions) - resolutions.index(n_res)), 120.0)
        if t_avail < 15:
            break

        print(f"  Multi-res n={n_res}, t={t_avail:.0f}s")

        old_x = np.linspace(0, 1, len(best_a))
        new_x = np.linspace(0, 1, n_res)
        a_res = np.interp(new_x, old_x, best_a)
        a_res = norm_sum1(a_res)

        # Quick SA at this resolution
        a_opt, M_opt = sa_optimize(a_res, t_avail, seed=seed + n_res)

        # Interpolate back
        old_x2 = np.linspace(0, 1, n_res)
        new_x2 = np.linspace(0, 1, n0)
        a_back = np.interp(new_x2, old_x2, a_opt)
        a_back = norm_sum1(a_back)
        M_back = fast_conv(a_back).max()

        if M_back < best_M:
            best_M = M_back
            best_a = a_back

    return best_a, best_M


def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def make_diverse_inits(n, parent=None):
    inits = []
    labels = []
    if parent is not None:
        inits.append(norm_sum1(parent))
        labels.append("parent")
    
    # Sine-based
    x = np.linspace(0, 1, n)
    inits.append(norm_sum1(np.sin(np.pi * x) ** 2)); labels.append("sin2")
    inits.append(norm_sum1(np.sin(np.pi * x) ** 4)); labels.append("sin4")
    
    # Power decay
    inits.append(norm_sum1((n - x * n + 1) ** 0.3)); labels.append("dec03")
    inits.append(norm_sum1((n - x * n + 1) ** 0.7)); labels.append("dec07")
    inits.append(norm_sum1((x * n + 1) ** 0.3)); labels.append("inc03")
    
    # Sinc
    for bw in [4.0, 7.0, 10.0, 15.0]:
        a = np.sinc((x - 0.5) * bw) ** 2
        inits.append(norm_sum1(np.maximum(a, 0))); labels.append(f"sinc{bw}")
    
    # Cosine modes
    for mode in [1, 2, 3]:
        a = np.cos(np.pi * mode * (x - 0.5)) ** 2
        inits.append(norm_sum1(np.maximum(a, 0))); labels.append(f"cos{mode}")
    
    # Two-peak
    for p1, p2, w in [(0.2, 0.7, 0.1), (0.15, 0.8, 0.08), (0.3, 0.6, 0.12)]:
        a = np.exp(-((x - p1) / w) ** 2) + np.exp(-((x - p2) / (w * 1.3)) ** 2)
        inits.append(norm_sum1(a)); labels.append(f"2pk_{p1}_{p2}")
    
    # Random smooth
    for s in [7, 42, 123]:
        rng = np.random.default_rng(s)
        a = np.ones(n) * 0.5
        for f in range(1, 10):
            a += rng.normal(0, 0.1 / np.sqrt(f)) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        inits.append(norm_sum1(np.maximum(a, 0.001))); labels.append(f"rnd{s}")
    
    # Exponential
    inits.append(norm_sum1(np.exp(-1.5 * x))); labels.append("exp15")
    inits.append(norm_sum1(np.exp(-3.0 * x))); labels.append("exp30")
    
    # Centered power
    xc = np.abs(np.arange(n) - n / 2.0) + 1.0
    inits.append(norm_sum1(xc ** (-0.5))); labels.append("cctr05")
    inits.append(norm_sum1(xc ** (-1.0))); labels.append("cctr10")
    
    return inits, labels


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss
