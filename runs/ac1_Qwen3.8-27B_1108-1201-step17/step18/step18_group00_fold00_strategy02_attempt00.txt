            S_val = np.sum(fixed_seq)
            if S_val < 0.015:
                return float('inf')
            ac = np.convolve(fixed_seq, fixed_seq, mode='full')
            max_val = np.max(ac)
            J = float((2 * len(seq) * max_val) / (S_val ** 2))
            # Check is_valid for edge cases
            if np.isnan(J) or J == float('inf') or J < -0.1:
                raise ValueError("Floating point error")
            return J
        except:
            return float('inf')

    # Objective function: direct property sampling without using FFT
    def objective(arr):
        return evaluate(arr)

    # Apply coordinate optimizing with physics-style randomization
    arr_opt = seq_guess.copy()
    best_score = evaluate(arr_opt)

    initial_fitness = best_score
    no_improve_i = 0

    # Increase search intensity if needed
    while no_improve_i < 10 and ticker < time_budget:
        try:
            # Simulate beam search: don't always move in the same direction
            new_arr = np.clip(minimize(objective, arr_opt, method='COVNE',
                                      bounds=([0,]*(n_true), [1000,]*(n_true)), 
                                      options={'maxiter':200}, return_values_if_error=True)
            ticker += 10
        except:
            # If constraints are too severe, try shortening or pickling dimensions
            new_arr = arr_opt

        if evaluate(new_arr) < best_score:
            best_score = evaluate(new_arr)
            no_improve_i = 0
        else:
            no_improve_i += 1

        # Apply crossover from previous solution
        if no_improve_i > 5:
            p_values = np.linspace(0, n_true-1, n_true)
            new_arr = arr_opt.copy()
            history_fitness_ratio = best_score / initial_fitness
            new_arr = new_arr * (1.0 - history_fitness_ratio) + arr_opt * history_fitness_ratio
            
            # Geometrically sample the grid to ensure diversity
            # This is a key optimization-specific change from scratch
            
            best_score = evaluate(new_arr)
            no_improve_i = 0  # Reset improvement counter

    # Final validation before return
    if np.sum(arr_opt) < 0.015:
        raise ValueError("Final sequence has too small sum")

    return list(arr_opt)
