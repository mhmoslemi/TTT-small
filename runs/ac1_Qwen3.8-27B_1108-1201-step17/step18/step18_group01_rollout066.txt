"""
Minimize 2*n*max(convolve(a,a))/(sum a)**2 over non-negative sequences a.

The objective is scale-invariant, so we normalize sum(a)=1 and minimize
    2*n*max_k (a*a)[k].
Linear (not circular) convolution means the peak is NOT forced to the
center; asymmetric sequences (geometric / tilted ramps) can push the peak
off-center and beat the symmetric uniform value of 2.

Search strategy (all top-level, no closures/lambdas):
  Phase 1  parametric + random initialization (geometric, tilted ramp, gaussian, sine)
  Phase 2  multiplicative-weights gradient descent (mirror map on simplex)
           with softmax-weighted MULTI-LAG subgradient + adaptive step
  Phase 3  projected gradient descent on a log-sum-exp SMOOTHED max objective
           with temperature annealing + backtracking line search
  Phase 4  coord/block local search with backtracking (two-swap + block jitter)
  Phase 5  multi-length exploration: interpolate best to longer n and re-optimize
  Phase 6  final MGD + local polish
Pipeline always returns the best sequence found within the time budget.
"""

import numpy as np
import time
import math


# --------------------------------------------------------------------------- #
# Low level helpers
# --------------------------------------------------------------------------- #
def fast_autocorr(a):
    """Linear autocorrelation (full convolution a*a) via FFT, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)
    return out[:size]


def norm_sum1(a):
    """Clip to >=0 and rescale to sum=1 (all-positive fallback)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def evaluate_safe(a):

def score_of(a):
    """Fast internal score 2*n*max(conv)/sum^2 (== 2*n*max(conv) when sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return float("inf")
    return 2.0 * a.size * float(fast_autocorr(a).max()) / (s * s)


# --------------------------------------------------------------------------- #
# Gradients
# --------------------------------------------------------------------------- #
def conv_lag_gradient(a, k):
    """
    d/dc_j [conv(a,a)[k]] where conv(a,a)[k]=sum_i a[i]*a[k-i].
    = 2*a[k-j] for j that actually appear in the sum, else 0.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    k = int(k)
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[k - idx]
    return g


def softmax_weights(vals, temperature):
    vals = np.asarray(vals, dtype=np.float64)
    v = vals - vals.max()
    z = np.exp(v / temperature)
    return z / z.sum()


def multi_lag_gradient(a, top_idx, weights):
    """Weighted sum of per-lag subgradients (a valid subgradient of the max)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        if w <= 1e-14:
            continue
        g += w * conv_lag_gradient(a, k)
    return g


# --------------------------------------------------------------------------- #
# Optimizer 1: multiplicative-weights gradient descent (multi-lag)
# --------------------------------------------------------------------------- #
def mgd(a0, time_limit, top_k, eta0, verbose=False):
    n = a0.size
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    eta = eta0
    patience = 0
    max_patience = 500
    cur_top = top_k
    iters = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = float(ac.max())
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience > max_patience:
            break

        kk = min(cur_top, ac.size)
        top_idx = np.argsort(ac)[::-1][:kk]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        # softmax over the top lags -> smoothly focus on the worst peaks
        temp = max((top_vals[0] - top_vals[-1]) / max(1.0, kk * 0.4), 1e-8)
        w = softmax_weights(top_vals, temp)

        g = multi_lag_gradient(a, top_idx, w)
        g = g - g.mean()
        gmax = float(np.abs(g).max())
        if gmax < 1e-15:
            break
        g = g / gmax

        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        M_new = float(fast_autocorr(a_new).max())

        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.5
            if eta < 0.0004:
                cur_top = max(8, cur_top // 2)
                eta = 0.025
                patience = max(0, patience - 150)

        iters += 1
        if verbose and iters % 150 == 0:
            print(f"    MGD iter={iters} ({time.monotonic()-t0:5.0f}s) J={2*n*best_m:.6f} eta={eta:.5f} top={cur_top}")

    if verbose:
        print(f"    MGD done ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.6f} iters={iters}")
    return best_a


# --------------------------------------------------------------------------- #
# Optimizer 2: projected gradient on smoothed (log-sum-exp) max
# --------------------------------------------------------------------------- #
def project_simplex(v):
    """Standard exact projection onto {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if rho_candidates.size == 0:
        return np.ones(n) / n
    rho = int(rho_candidates[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, tau):
    """Gradient of tau*log(sum_k exp(conv(k)/tau)) wrt a."""
    ac = fast_autocorr(a)
    M = ac.max()
    z = np.exp((ac - M) / tau)
    p = z / z.sum()
    g = np.zeros(a.size)
    sig = np.where(p > 1e-9)[0]
    for k in sig:
        g += p[k] * conv_lag_gradient(a, int(k))
    return g, M


def pgd_smooth(a0, time_limit, verbose=False):
    n = a0.size
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    tau = max(best_m * 0.15, 1e-5)
    eta = 0.02
    patience = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = float(ac.max())
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience > 350:
            tau *= 0.5
            eta = 0.02
            patience = 0
            if tau < 1e-7:
                break
            continue

        tau_eff = max(tau, best_m * 0.02)
        g, _ = smoothed_grad(a, tau_eff)

        # backtracking line search
        a_new = None
        M_new = float("inf")
        step = eta
        for _ in range(8):
            cand = project_simplex(a - step * g)
            mc = float(fast_autocorr(cand).max())
            if mc <= M + 1e-12:
                a_new = cand
                M_new = mc
                break
            step *= 0.5
        if a_new is None:
            tau *= 0.8
            continue

        a = a_new
        eta = max(step * 1.1, 1e-7)
        tau *= 0.997

        iters += 1
        if verbose and iters % 100 == 0:
            print(f"    PGD iter={iters} ({time.monotonic()-t0:5.0f}s) J={2*n*best_m:.6f} tau={tau_eff:.2e}")

    if verbose:
        print(f"    PGD done ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.6f} iters={iters}")
    return best_a

# Optimizer 3: local search (two-swap mass moves + block jitter)
# --------------------------------------------------------------------------- #
def local_search(a0, time_limit, block_size=20, verbose=False):
    n = a0.size
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    fracs = (0.02, 0.05, 0.1, 0.2, 0.35, 0.55)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = float(ac.max())
        improved = False

        # targeted two-swap moves around the worst lag
        top_lags = np.argsort(ac)[::-1][:8]
        k_pick = int(top_lags[0])
        lo = max(0, k_pick - n + 1)
        hi = min(n - 1, k_pick)

        trials = 40
        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            r = rng.random()
            if r < 0.6 and hi >= lo:
                i = int(rng.integers(lo, hi + 1))
            else:
                i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue
            for frac in fracs:
                amt = a[i] * frac
                tr = a.copy()
                tr[i] -= amt
                tr[j] += amt
                tr = np.maximum(tr, 0.0)
                s = tr.sum()
                if s < 1e-15:
                    break
                tr = tr / s
                mt = float(fast_autocorr(tr).max())
                if mt < best_m - 1e-14:
                    a = tr
                    best_m = mt
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # block jitter escape
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            a[start:end] += rng.normal(0.0, 0.02 * (a[start:end].mean() + 1e-4), end - start)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if stall > 400:
                break
        else:
            stall = 0

    if verbose:
        print(f"    Local done ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.6f}")
    return best_a

def interpolate_seq(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = f.size
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, fn)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return out / ss


def make_parametric_candidates(n, rng):
    cands = []
    i = np.arange(n, dtype=np.float64)
    x = i / max(1.0, n - 1.0)
    # geometric (both directions) - exploit asymmetric linear-conv peak shift
    for r in (0.9995, 0.999, 0.997, 0.99, 0.97, 0.93, 0.85):
        cands.append(r ** i)
        cands.append(r ** (n - 1 - i))
    # tilted ramps
    for a in (0.3, 0.7, 1.0, 1.6, 2.5):
        cands.append((i + 1.0) ** a)
        cands.append((n - i) ** a)
    # gaussian bumps
    for c in (0.25, 0.4, 0.5, 0.6, 0.75):
        for sig in (0.1, 0.2, 0.35):
            cands.append(np.exp(-((x - c) ** 2) / (2 * sig * sig)))
    # sine powers
    for f in (1, 2, 3):
        cands.append(np.maximum(np.sin(np.pi * f * x), 0.0) ** 2 + 1e-3)
    # exponential fills
    for rate in (1.0, 3.0, 8.0):
        cands.append(np.exp(-rate * x))
        cands.append(np.exp(rate * (x - 1.0)))
    # random smooth (low-freq) sequence
    for _ in range(6):
        base = np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 6.28))
        base = np.maximum(base, 0.0) ** 2 + 0.05
        noise = np.cumsum(rng.normal(0, 0.05, n))
        noise -= noise.min()
        base = base + 0.02 * noise
        cands.append(base)
    return cands
