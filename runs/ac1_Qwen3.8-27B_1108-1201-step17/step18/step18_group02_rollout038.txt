def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def gen_init_candidates(n):
    """Generate diverse initial candidates, emphasizing asymmetric shapes."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Monotone power laws (KEY: asymmetric!)
    for p in [0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (i_arr + 1.0)**p
        candidates.append(a / a.sum())
        a = (n - i_arr)**p
        candidates.append(a / a.sum())
    
    # Exponential
    for alpha in [1, 2, 3, 5, 8, 12]:
        a = np.exp(alpha * x)
        candidates.append(a / a.sum())
        a = np.exp(alpha * (1 - x))
        candidates.append(a / a.sum())
    
    # Sin^2 (symmetric, for comparison)
    for freq in [1, 2]:
        for phase in [0, 0.3, 0.5]:
            a = np.sin(np.pi * freq * x + phase)**2
            a = np.maximum(a, 0.01)
            candidates.append(a / a.sum())
    
    # Mixed: monotone + oscillation
    for p in [0.5, 1.0]:
        for freq in [3, 5]:
            for amp in [0.1, 0.3]:
                a = ((i_arr + 1.0)**p) * (1.0 + amp * np.sin(2*np.pi*freq*x))
                a = np.maximum(a, 0.01)
                candidates.append(a / a.sum())
    
    # Uniform
    candidates.append(np.ones(n) / n
    
    return candidates
