"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. Nesterov accelerated gradient on log-sum-exp smoothed objective with adaptive beta
2. LP-Frank-Wolfe with improved active set management
3. Peak-targeted coordinate descent
4. Multi-length exploration with interpolation
"""

import numpy as np
import time
import cvxpy as cp

# ... (helper functions)

def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    # Sort in decreasing order
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def logsumexp_objective(a, beta):
    """Compute the log-sum-exp smoothed objective and its gradient."""
    b = fast_autocorr(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    # Clip to prevent overflow
    shifted = np.clip(shifted, -500, 500)
    p = np.exp(shifted)
    p = p / p.sum()
    obj = (b_max + np.log(p.sum()) / beta)  # = (1/beta) * log(sum(exp(beta*b)))
    # Gradient: (1/beta) * sum_k p_k * db_k/da
    # db_k/da_j = a_{k-j} (valid indices)
    # grad_j = (1/beta) * sum_m p_{j+m} * a_m
    a_rev = a[::-1]
    # Use FFT for the convolution
    conv_len = len(a) + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[len(a) - 1 : 2 * len(a) - 1] / beta

            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    MD restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    MD t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

    return best_a, best_max_b


def lp_fw_improved(a0, time_limit, solver, initial_K=400, max_K=3000,
                   add_per_iter=30, verbose=True):
    """Improved LP-Frank-Wolfe with better active set and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 1200
    warm_start_prob = None
    warm_start_vars = {}

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: converged at iter " + str(t))
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations to add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1), 2.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def coordinate_polish_v2(a0, time_limit, verbose=True):
    """Peak-targeted coordinate descent with adaptive strategy."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(40, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            src_val = a[src]
            partner_val = a[partner] if 0 <= partner < n else 0.0

            # Determine which to reduce
            if src_val >= partner_val:
                reduce_idx = src
                keep_idx = partner
            else:
                reduce_idx = partner
                keep_idx = src

            if a[reduce_idx] < 1e-12:
                continue

            # Find safe destination: low gradient positions
            # Compute "pressure" at each position
            pressure = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            # Exclude positions that are part of top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            cand_pressure = pressure[candidates]
            dst = int(candidates[np.argmin(cand_pressure)])

            if dst == reduce_idx:
                dst = int(candidates[np.argmin(cand_pressure[candidates != reduce_idx])])
            if dst == reduce_idx:
                continue

            # Try different fractions
            found = False
            for frac in [0.01, 0.02, 0.04, 0.08, 0.15]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 200:
                # Random exploration
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Broader random moves
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 400 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord v2 (" + str(elapsed) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def parametric_init(n, time_limit, best_init, best_score):
    """Quick parametric search for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    candidates = []
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    # Symmetric
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)
        candidates.append(x ** p + (1.0 - x) ** p)
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    # Cosine
    for freq in [1, 2, 3]:
        candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x)) / 2.0, 0.01))
    # Sine
    for p in [1, 2, 3, 4]:
        candidates.append(np.sin(np.pi * x) ** p)

    for a_raw in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(a_raw)
        b = fast_autocorr(a)
        m = b.max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Random
    rng = np.random.default_rng(42)
    for _ in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(elapsed) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: Nesterov + LP-FW + coordinate descent with multi-length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric search
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: Nesterov accelerated gradient (main optimizer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.30, 480.0)
    if t2 > 60:
        a_out, _ = nesterov_projected(norm_sum1(best_seq), t2,
                                      beta_start=15.0, beta_end=600.0,
                                      beta_growth=1.004, eta_base=0.8,
                                      top_k_ratio=0.05, restart_every=3000,
                                      verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(n_best), t_global)

    # Phase 3: Mirror descent (alternative)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Mirror descent (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 30:
        a_out, _ = mirror_descent(norm_sum1(best_seq), t3,
                                  beta=40.0, eta_start=0.03, eta_decay=0.9995,
                                  restart_every=1500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "md_n" + str(n_best), t_global)

    # Phase 4: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 40 and n_best <= 7000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t4, solver,
                                  initial_K=500, max_K=3000,
                                  add_per_iter=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(n_best), t_global)

    # Phase 5: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Coordinate v2 (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_out, _ = coordinate_polish_v2(norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(n_best), t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 300, n_best + 600, n_best + 1000]
    longer_lengths = [l for l in longer_lengths if l <= 8000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n=== Phase 6: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.30, 180.0)
        if t6a > 30:
            a_out, _ = nesterov_projected(a_long, t6a,
                                          beta_start=20.0, beta_end=500.0,
                                          beta_growth=1.005, eta_base=0.7,
                                          top_k_ratio=0.04, restart_every=2500,
                                          verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 7000:
            t6b = min(remaining * 0.40, 200.0)
            if t6b > 30:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_fw_improved(norm_sum1(a_long2), t6b, solver,
                                          initial_K=400, max_K=2500,
                                          add_per_iter=25, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

    # Phase 7: Final Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t7 = min(remaining * 0.35, 250.0)
        if t7 > 30:
            a_out, _ = nesterov_projected(norm_sum1(best_seq), t7,
                                          beta_start=50.0, beta_end=1000.0,
                                          beta_growth=1.006, eta_base=0.6,
                                          top_k_ratio=0.03, restart_every=2000,
                                          verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final_n" + str(n_best), t_global)

    # Phase 8: Final coordinate
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final coord (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t8 = min(remaining * 0.50, 150.0)
        if t8 > 15:
            a_out, _ = coordinate_polish_v2(norm_sum1(best_seq), t8, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 7000:
        print("\n=== Phase 9: Final LP burst ===")
        t9 = min(remaining * 0.70, 180.0)
        if t9 > 20:
            a_out, _ = lp_fw_improved(norm_sum1(best_seq), t9, solver,
                                      initial_K=600, max_K=3500,
                                      add_per_iter=35, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_burst", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(elapsed) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
