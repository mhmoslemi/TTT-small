Let me also add a novel element: **spectrum shaping**.

The autocorrelation peak is related to the spectrum. Specifically:
- The autocorrelation is the inverse DFT of |FFT(a)|^2 (the power spectrum).
- To minimize the peak autocorrelation, we want the power spectrum to be as flat as possible.
- But we need a ≥ 0.

So the algorithm can be:
1. Compute FFT(a) = A.
2. In frequency domain, replace |A_k|^2 with a "flattened" version: |A_k|'_2 = alpha * |A_k|^2 + (1-alpha) * mean(|A|^2).
3. Compute new A' with same phase but modified amplitude.
4. Take IFFT to get a'.
5. Project a' onto {x ≥ 0, sum(x) = 1}.
6. Blend: a = (1-t)*a + t*a'.
7. Repeat.

This is the SPA (Successive Projection Algorithm) for PAPR reduction, adapted for our problem.

The advantage of this approach is that it works directly on the quantity we want to minimize (the peak of the inverse spectrum) and tends to converge to a flat spectrum, which corresponds to low autocorrelation peaks.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with exact subgradient and log-barrier for non-negativity
2. Spectrum-shaping (SPA-type) projection in frequency domain

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size, n=fft_size)[:size]

def norm_pos(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def subgrad_max_conv(a, conv, n_active=None):
    """Exact subgradient of max_k conv(k). If n_active is given, use top-n."""
    n = len(a)
    if n_active is not None and n_active < len(conv):
        # Use top-n active peaks with equal weight
        idx = np.argsort(conv)[::-1][:n_active]
        w = 1.0 / n_active
    else:
        # Use exact: all peaks within epsilon of max
        conv_max = conv.max()
        # tolerance
        tol = 1e-12 * conv_max
        idx = np.where(conv >= conv_max - tol)[0]
        w = 1.0 / len(idx)
    
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        i_vals = np.arange(lo, hi + 1)
        j_vals = k - i_vals
        grad[i_vals] += w * a[j_vals]
    return 2.0 * grad  # factor of 2 for the quadratic

def fista_log(a0, t_limit, seed=42, c0=1.0, verbose=False):
    """FISTA in log-space with mirror descent step."""
    n = len(a0)
    x = np.log(norm_pos(a0) + 1e-15)  # log of a
    prev_x = x.copy()
    best_a = norm_pos(a0).copy()
    best_m = fast_conv(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    t = 0
    stall = 0
    patience = 5000
    
    while time.monotonic() - t0 < t_limit:
        t += 1
        beta = (t - 1.0) / (t + 1.8)
        y_log = x + beta * (x - prev_x)
        y = np.exp(y_log - y_log.max())
        y = y / y.sum()
        
        conv_y = fast_conv(y)
        g = subgrad_max_conv(y, conv_y)
        
        # Mirror descent step in log-space
        eta = c0 / (np.sqrt(t) + 1.0)
        new_log = y_log - eta * g / y  # gradient of log-barrier adjusted
        # Actually, the mirror step is: new_log = y_log - eta * g (since the mirror map is exp)
        # But we need to be careful: the gradient in the x-space is g, and the mirror map is exp.
        # The update is: new_x = y - eta * g (in x-space), then project to simplex.
        # In log-space: new_log = log(y - eta * g) approximately.
        # Better: new_x = y * exp(-eta * g) (multiplicative update)
        
        new_x = y * np.exp(-eta * g)
        new_x = np.maximum(new_x, 1e-15)
        new_x = new_x / new_x.sum()
        new_log = np.log(new_x)
        
        x, prev_x = new_log, x
        
        M = fast_conv(new_x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = new_x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > patience:
            # Restart with perturbation
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = np.log(x_r + 1e-15)
            prev_x = x.copy()
            stall = 0
            c0 *= 1.05
            patience = int(patience * 1.1)
    
    return best_a, best_m

def spa_optimize(a0, t_limit, alpha=0.3, beta_blend=0.5, seed=42, verbose=False):
    """Spectrum-shaping / SPA-type optimization."""
    n = len(a0)
    a = norm_pos(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    while time.monotonic() - t0 < t_limit:
        A = np.fft.rfft(a, fft_size)
        power = A.real**2 + A.imag**2
        
        # Flatten the power spectrum
        mean_p = power.mean()
        power_flat = alpha * power + (1.0 - alpha) * mean_p
        
        # New amplitude
        new_mag = np.sqrt(np.maximum(power_flat, 1e-30))
        phase = np.angle(A) if A.size > 1 else np.zeros_like(A)
        new_A = new_mag * np.exp(1j * phase)
        
        new_a = np.fft.irfft(new_A, fft_size, n=fft_size)[:n]
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s > 1e-15:
            new_a = new_a / s
        else:
            new_a = a.copy()
        
        # Blend
        a = (1.0 - beta_blend) * a + beta_blend * new_a
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        M = fast_conv(a).max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
    
    return best_a, best_m

def coord_polish(a0, t_limit, seed=42, verbose=False):
    """Coordinate descent: move mass from high-cost to low-cost positions."""
    n = len(a0)
    a = norm_pos(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < t_limit:
        conv = fast_conv(a)
        conv_max = conv.max()
        
        # Find top peaks
        n_peaks = min(20, 2*n-1)
        peaks = np.argsort(conv)[::-1][:n_peaks]
        
        # Compute cost for each coordinate
        cost = np.zeros(n)
        for k in peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_vals = np.arange(lo, hi + 1)
            j_vals = k - i_vals
            cost[i_vals] += a[j_vals]
        
        # Also add self-cost (edge effects)
        cost += a * len(peaks)
        
        # Find best (src, dst) pair: src has high cost, dst has low cost
        # Try top few sources
        n_try = max(10, n // 100)
        src_indices = np.argsort(cost)[::-1][:n_try]
        dst_indices = np.argsort(cost)[:n_try]
        
        improved = False
        for si in src_indices:
            si = int(si)
            if a[si] < 1e-14:
                continue
            for di in dst_indices:
                di = int(di)
                if di == si:
                    continue
                # Try moving a fraction of a[si] to a[di]
                for frac in [0.1, 0.25, 0.5, 0.75]:
                    amt = a[si] * frac
                    a_tri = a.copy()
                    a_tri[si] -= amt
                    a_tri[di] += amt
                    a_tri = np.maximum(a_tri, 0.0)
                    s = a_tri.sum()
                    if s < 1e-15:
                        continue
                    a_tri = a_tri / s
                    M = fast_conv(a_tri).max()
                    if M < best_m - 1e-14:
                        a = a_tri
                        best_m = M
                        best_a = a.copy()
                        improved = True
                        if verbose:
                            print(f"    CD: moved {frac*100:.0f}% from {si} to {di}, M={M:.2e}")
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M = fast_conv(a).max()
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_m

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_n = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_n)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss
