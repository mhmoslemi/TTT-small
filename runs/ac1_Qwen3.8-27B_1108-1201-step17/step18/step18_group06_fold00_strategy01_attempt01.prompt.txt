import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient(a, ac, n_peaks=80, temperature=1.0):
    """Compute softmax-weighted multi-peak subgradient."""
    n = len(a)
    k_max = min(n_peaks, len(ac))
    # Use softmax weighting over all lags for smoother gradient
    ac_clipped = np.minimum(ac, ac.max())
    log_ac = np.log(np.maximum(ac_clipped, 1e-30))
    exp_vals = np.exp((log_ac - log_ac.max()) / temperature)
    w = exp_vals / exp_vals.sum()
    # Focus on top peaks but with smooth falloff
    grad = np.zeros(n)
    # Compute contribution from top peaks
    top_idx = np.argsort(ac)[::-1][:k_max]
    for k in top_idx:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-10:
            continue
        # d c[ki]/d a[j] = 2*a[ki-j] for valid j
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    return grad

def spectral_flatten(a0, time_limit, alpha0=0.3, iters=5000, verbose=False):
    """Gerchberg-Saxton flat spectrum optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Progressive: start with aggressive flattening, then refine
        alpha = alpha0 + (1.0 - alpha0) * (it / max(1, iters))
        P_target = P_mean * (P / np.maximum(P_mean, 1e-30)) ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target / np.maximum(P, 1e-30))
        A_new = A * mag_new
        # But cap the modification to avoid instability
        mag_ratio = mag_new
        mag_ratio = np.clip(mag_ratio, 0.7, 1.4)
        A_new = A * mag_ratio

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped: take a smaller step
            blend = 0.3
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def peak_reduction_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Direct coordinate moves targeting peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Get top peaks
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            # Find the pair (i, ki-i) with maximum contribution
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # The position with max contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find a good destination: position not heavily involved in other top peaks
            costs = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                j2_lo = max(0, k2i - n + 1)
                j2_hi = min(n - 1, k2i)
                valid = np.arange(j2_lo, j2_hi + 1)
                if len(valid) > 0:
                    costs[valid] += a[k2i - valid]

            # Pick destination with lowest cost among positions with low value
            candidates = np.argsort(costs + 10 * a)[:n // 3]
            best_dst = candidates[np.argmin(a[candidates])]
            if best_dst == src:
                best_dst = int(rng.integers(n))
                if best_dst == src:
                    continue

            # Try moving a fraction of mass
            found = False
            for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                # Try random small move from src
                if a[src] > 1e-10:
                    dst = int(rng.integers(n))
                    if dst != src:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[src] * frac
                        a[src] -= amt
                        a[dst] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def make_power_law_ramp(n, power=0.5):
    """Increasing ramp: a[i] = (i+1)^power, normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_mild_ramp(n, center_frac=0.3):
    """1 + c*i/n shape, normalized."""
    x = np.linspace(0, 1, n)
    a = 1.0 + center_frac * x
    return norm_sum1(a)


def make_exponential_ramp(n, rate=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return norm_sum1(a)


def make_alg_ramp(n, alpha=1.5):
    """a[i] = 1/(n-i+alpha) type: increasing towards the right."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (n - i + alpha)
    return norm_sum1(a)


def make_two_phase(n, frac=0.6, alpha=0.3):
    """Flat on left, ramp on right."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    mask = x > frac
    a[mask] = 1.0 + alpha * (x[mask] - frac) / (1.0 - frac)
    return norm_sum1(a)
