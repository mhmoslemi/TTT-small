def solve_lp(a, M):
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    
    # Constraint: (a * g)[k] <= M for all k
    # (a*g)[k] = sum_i a[i] * g[k-i] for valid i
    # This is a linear function of g for each k
    
    constraints = []
    for k in range(2*n - 1):
        # (a*g)[k] = sum_{i} a[i] * g[k-i] where 0 <= i < n and 0 <= k-i < n
        # So i ranges from max(0, k-n+1) to min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        # (a*g)[k] = sum_{i=i_lo}^{i_hi} a[i] * g[k-i]
        # In terms of g: = sum_{j=k-i_hi}^{k-i_lo} a[k-j] * g[j]
        j_lo = k - i_hi
        j_hi = k - i_lo
        expr = a[i_lo:i_hi+1][::-1] @ g[j_lo:j_hi+1]

    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
