def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_n = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    out = np.interp(new_x, old_x, f_n)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return out / s
