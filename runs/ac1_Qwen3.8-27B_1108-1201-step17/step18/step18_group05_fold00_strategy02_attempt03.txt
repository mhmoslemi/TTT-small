    def spectral_init(n, n_modes=20):

            for freq in freqs:
                coef = np.random.normal(0, 0.01/np.sqrt(freq))
                base += coef * np.sin(2*np.pi*freq*x + phase)
            return np.maximum(base, 0.0)
        except:
            return np.ones(n) / n

    def interpolate_sequence(f, new_n):
        try:
            old_n = len(f)
            if old_n == new_n:
                return f.copy()
            if old_n < new_n:
                # Linearly interpolate with more points
                new_x = np.linspace(0, 1, new_n)
                old_x = np.linspace(0, 1, old_n)
                return norm_sum1(np.interp(new_x, old_x, f) * np.sqrt(new_n/len(f)))
            else:
                # Keep more energy in the shorter sequence
                new_x = np.linspace(0, 1, new_n)
                old_x = np.linspace(0, 1, old_n)
                return norm_sum1(np.interp(new_x, old_x, f) * np.sqrt(len(f)/new_n))
        except:
            return np.ones(new_n) / new_n
