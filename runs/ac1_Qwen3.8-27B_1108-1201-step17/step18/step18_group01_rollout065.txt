import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def gradient_of_lag(a, k):
    """Compute dc_k/da for lag k (0 <= k < n) of linear autocorrelation."""
    n = len(a)
    g = np.zeros(n)
    if k == 0:
        g[:] = 2.0 * a
    elif k < n:
        # j in [0, k): g[j] = a[j+k]
        g[0:k] = a[k:n]
        # j in [k, n-k): g[j] = a[j+k] + a[j-k]
        if n - k > k:
            g[k:n-k] = a[2*k:n] + a[0:n-2*k]
        # j in [n-k, n-1]: g[j] = a[j-k]
        g[n-k:] = a[0:k]
    return g


def compute_weighted_gradient(a, ac, alpha, top_k):
    """
    Compute soft-max weighted gradient of the autocorrelation peaks.
    Returns (gradient, num_active_lags).
    """
    n = len(a)
    ac_half = ac[:n]

    # Soft-max weights over all lags
    max_v = ac_half.max()
    if max_v < 1e-30:
        return np.zeros(n), 0
    exp_v = np.exp(alpha * (ac_half - max_v))
    p = exp_v / exp_v.sum()

    # Only use top significant weights (prune for efficiency)
    sorted_idx = np.argsort(p)[::-1]
    num_active = min(top_k, n)
    active_idx = sorted_idx[:num_active]
    active_p = p[active_idx]
    wsum = active_p.sum()
    if wsum < 1e-30:
        return np.zeros(n), 0
    active_p = active_p / wsum

    # Compute weighted gradient
    grad = np.zeros(n)
    for i in range(num_active):
        w = active_p[i]
        if w < 1e-12:
            continue
        k = int(active_idx[i])
        grad += w * gradient_of_lag(a, k)

    return grad, num_active

def peak_shave(a0, time_limit, num_peaks=12, verbose=False):
