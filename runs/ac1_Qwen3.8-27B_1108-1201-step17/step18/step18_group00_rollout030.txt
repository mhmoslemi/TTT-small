def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_score(a):
    """Compute 2*n*max(ac)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def evaluate_safe(a):
    try:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def smoothed_obj_grad(a, beta, n, penalty=1e-6):
    """
    Compute smoothed objective and gradient.
    Objective: (1/beta)*log(sum(exp(beta*ac_k))) + penalty*(sum(a)-1)^2
    Gradient computed via FFT cross-correlation.
    """
    a = np.maximum(a, 1e-15)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.ones(n)
    a_n = a / s  # normalized for autocorrelation computation (scale invariant)
    
    ac = fast_autocorr(a_n)
    size = 2 * n - 1
    ac_max = ac.max()
    ac_shifted = ac - ac_max
    exp_vals = np.exp(beta * ac_shifted)
    log_sum = ac_max + np.log(np.sum(exp_vals)) / beta
    w = exp_vals / np.sum(exp_vals)  # softmax weights
    
    # Gradient of smoothed max w.r.t. a_n:
    # grad[i] = 2 * sum_j w[i+j] * a_n[j] = 2 * cross_corr(w, a_n)[i]
    fft_size = 1 << (3 * n).bit_length()
    W_ext = np.zeros(fft_size)
    W_ext[:size] = w
    A_ext = np.zeros(fft_size)
    A_ext[:n] = a_n
    corr = np.fft.irfft(np.fft.rfft(W_ext) * np.conj(np.fft.rfft(A_ext)), fft_size)[:n]
    grad_ac = 2.0 * corr / s  # chain rule through normalization
    
    # Gradient of penalty term: 2*penalty*(sum(a)-1)*ones/s (approximately)
    # Since we normalize a_n = a/s, the gradient w.r.t. a is:
    # d(a_n)/d(a) = (I - a_n*ones^T)/s
    # So grad w.r.t. a = (I - a_n*ones^T)/s @ grad_ac + penalty_grad
    # Simplify: grad[i] = (grad_ac[i] - a_n[i]*sum(a_n*grad_ac))/s
    dot = np.sum(a_n * grad_ac)
    grad = (grad_ac - a_n * dot) / s
    
    # Add tiny penalty to keep sum(a) near 1
    if penalty > 0:
        grad += 2.0 * penalty * (a.sum() - 1.0) / a.sum()
    
    obj = 2.0 * n * log_sum + penalty * (a.sum() - 1.0) ** 2
    return obj, grad


def lbfgs_continuation(a0, time_limit, betas=None, verbose=True):
    """
    L-BFGS with log-sum-exp smoothing and continuation on beta.
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    if betas is None:
        betas = [5, 15, 50, 150, 500, 1500, 5000, 15000, 50000]
    
    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        time_left = time_limit - (time.monotonic() - t0)
        
        # Stop early if beta is very large and we're not improving
        if beta > 1000 and (time.monotonic() - t0) > time_limit * 0.7:
            break
        
        max_iter = min(200, max(30, int(time_left * 50)))
        
        def obj_grad(x):
            x = np.maximum(x, 1e-15)
            o, g = smoothed_obj_grad(x, beta, n, penalty=1e-8)
            return o, g
        
        res = scipy_minimize(obj_grad, a, method='L-BFGS-B',
                             bounds=[(1e-15, None)] * n,
                             options={'maxiter': max_iter, 'ftol': 1e-12,
                                      'gtol': 1e-8, 'maxls': 50})
        
        a_new = np.maximum(res.x, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            if verbose:
                print("    L-BFGS beta=" + str(beta) + ": score=" + str("%.6f" % best_score))
        
        a = a_new
        
        # Check time
        if verbose and (time.monotonic() - t0) % 10 < 2:
            print("    L-BFGS beta=" + str(beta) + " done, elapsed=" + str("%.1f" % (time.monotonic() - t0)) + "s")
    
    return best_a, best_score


def lbfgs_multistart(a0, time_limit, n_starts=3, verbose=True):
    """Run L-BFGS continuation from multiple perturbed starts."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    
    for si in range(n_starts):
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 30:
            break
        
        if si == 0:
            start = a.copy()
        else:
            noise = rng.normal(0, 0.02 / si, n) * a
            start = a + noise
            start = np.maximum(start, 1e-10)
            start = start / start.sum()
        
        al = min(time_left * 0.6 / (n_starts - si), 200.0)
        a_out, sc = lbfgs_continuation(start, al, verbose=False)
        
        sc_final = compute_score(a_out)
        if sc_final < best_score - 1e-10:
            best_score = sc_final
            best_a = a_out.copy()
            a = a_out.copy()
            if verbose:
                print("    L-BFGS multi start " + str(si) + ": score=" + str("%.6f" % best_score))
    
    return best_a, best_score


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2500,
          add_per_iter=30, verbose=True):
    """LP-based Frank-Wolfe with line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    
    no_improve = 0
    max_no_improve = 1000
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        coeffs = build_constraint_matrix(y, constraint_list, n)
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(constraint_list[i]) for i in range(min(initial_K, len(constraint_list))))
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Line search
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))
        
        default_alpha = 2.0 / (t + 2.0)
        alphas = [max(0.01, min(1.0, default_alpha)),
                  max(0.01, min(1.0, default_alpha * 0.5)),
                  max(0.01, min(1.0, default_alpha * 2.0)),
                  max(0.01, min(1.0, 1.0 / (t + 1.0)))]
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        
        y = y_new
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str("%.1f" % elapsed) + "s): J=" + str("%.6f" % J))
    
    return best_a, best_m


def mirror_descent_accel(a0, time_limit, top_k=80, beta_exp=3.0, c0=5.0,
                         restart_every=3000, verbose=True):
    """Accelerated mirror descent with Nesterov momentum and soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    restart_count = 0
    cur_top_k = top_k
    cur_beta = beta_exp
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Soft subgradient from top-k peaks
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.05 / (1.0 + 0.08 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            gamma = max(gamma * 0.9, 0.5)
            cur_top_k = min(cur_top_k + 5, 150)
            if verbose and restart_count % 3 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    MD restart " + str(restart_count) + " (" + str("%.1f" % elapsed) + "s): J=" + str("%.6f" % J))
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=True):
    """Coordinate descent: move mass from high-impact to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 800
    step_scales = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(40, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Score all positions by their impact on top autocorrelation peaks
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            
            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])
            
            if dst == src:
                continue
            
            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
            
            if not found and stall > 200:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            if stall % 100 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 300 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str("%.1f" % elapsed) + "s): J=" + str("%.6f" % J) + ", stall=" + str(stall))
    
    return best_a, best_m


def parametric_init(n, best_init, best_score):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score
    
    families = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        families.append((i_arr + 1.0) ** p)
        families.append((n - i_arr) ** p)
    
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            families.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
    
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            families.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    
    for p in [0.5, 1.0, 1.5, 2.0]:
        families.append(x ** p + (1.0 - x) ** p)
    
    for freq in [1, 2, 3, 5]:
        for ph in [0.0, np.pi / 2]:
            families.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.01))
    
    # Anti-concentration: mass at edges
    for rate in [1, 2, 5, 10]:
        families.append(np.exp(-rate * x) + np.exp(rate * (x - 1.0)))
    
    # Truncated
    for frac in [0.3, 0.5, 0.7]:
        center = n // 2
        half = int(n * frac / 2)
        f = np.zeros(n)
        f[center - half:center + half] = 1.0
        families.append(f)
    
    for f in families:
        a = norm_sum1(f)
        v = compute_score(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()
    
    # Random
    rng = np.random.default_rng(99)
    for trial in range(50):
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 4.0))
        v = compute_score(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()
    
    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = get_solver()
    print("Time budget: " + str("%.0f" % time_budget) + "s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_score = evaluate_safe(p)
            if np.isfinite(p_score):
                best_seq = norm_sum1(p)
                best_score = p_score
                print("Parent: n=" + str(len(best_seq)) + ", score=" + str("%.6f" % best_score))
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + str("%.6f" % best_score))
    
    n0 = len(best_seq)
    
    # Phase 1: Parametric search
    print("\n=== Phase 1: Parametric init (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 5:
        a_param, _ = parametric_init(n0, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param", t_global)
    
    # Phase 2: L-BFGS with continuation (MAIN NEW METHOD)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: L-BFGS continuation (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 60:
        a_out, _ = lbfgs_multistart(norm_sum1(best_seq), t2, n_starts=3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_n" + str(n_best), t_global)
    
    # Phase 3: LP-FW polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW polish (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 40 and n_best <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=400, max_K=3000, add_per_iter=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(n_best), t_global)
    
    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Coordinate descent (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t4, rng_seed=42, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(n_best), t_global)
    
    # Phase 5: Try longer sequences
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Longer sequences ===")
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 12000]
    
    for new_n in longer_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)
        sc_long = compute_score(a_long)
        if sc_long < best_score:
            best_seq = a_long.copy()
            best_score = sc_long
            print("  Interpolated: score=" + str("%.6f" % best_score))
        
        # L-BFGS on longer
        t5a = min(remaining * 0.3, 150.0)
        if t5a > 40:
            a_out, _ = lbfgs_multistart(a_long, t5a, n_starts=2, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_n" + str(new_n), t_global)
        
        # LP-FW on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 8000:
            t5b = min(remaining * 0.3, 120.0)
            if t5b > 30:
                a_out, _ = lp_fw(norm_sum1(best_seq), t5b, solver,
                                 initial_K=300, max_K=2000, add_per_iter=25, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)
    
    # Phase 6: Final L-BFGS + LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t6a = min(remaining * 0.4, 200.0)
        if t6a > 30:
            a_out, _ = lbfgs_multistart(norm_sum1(best_seq), t6a, n_starts=2, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_lbfgs", t_global)
    
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 8000:
        t6b = min(remaining * 0.4, 150.0)
        if t6b > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t6b, solver,
                             initial_K=500, max_K=3000, add_per_iter=30, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_lpfw", t_global)
    
    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7 = min(remaining * 0.7, 100.0)
        if t7 > 15:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t7, rng_seed=314, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_coord", t_global)
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + str("%.6f" % best_score) + ", n=" + str(len(best_seq)) + ", time=" + str("%.1f" % elapsed) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + str("%.6f" % best_score))
    
    return [float(x) for x in best_seq]
