def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_from_autocorr(a, ac):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * ac.max() / (s * s))


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe_v2(a0, time_limit, solver, initial_K=300, max_K=2000,
                      add_per_iter=30, verbose=True):
    """LP-FW with line search and adaptive constraint management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Initialize constraint set with top lags
    ac_y = fast_autocorr(a)
    init_idx = np.argsort(ac_y)[::-1][:min(initial_K, size)]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0
    stall_iters = 0

    # Nesterov momentum variables
    y = a.copy()
    x = a.copy()

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac_curr = fast_autocorr(y)
        M_curr = ac_curr.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            stall_iters = 0
        else:
            stall_iters += 1

        # If stalled too long, reset momentum
        if stall_iters > 500:
            y = best_a.copy()
            x = best_a.copy()
            stall_iters = 0
            # Refresh constraint set
            ac_refresh = fast_autocorr(y)
            new_idx = np.argsort(ac_refresh)[::-1][:min(initial_K, size)]
            constraint_set = set(int(k) for k in new_idx)

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_curr[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build sparse constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 5:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Line search for step size
        best_step = None
        best_obj = None
        current_obj = fast_autocorr(y).max()

        for step in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1]:
            candidate = (1.0 - step) * y + step * g_dir
            candidate = np.maximum(candidate, 0.0)
            s_cand = candidate.sum()
            if s_cand < 1e-15:
                continue
            candidate = candidate / s_cand
            obj = fast_autocorr(candidate).max()
            if best_obj is None or obj < best_obj:
                best_obj = obj
                best_step = candidate

        if best_step is None:
            best_step = y.copy()
            best_obj = current_obj

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * best_step
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW2 iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW2 done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def gradient_descent_smooth(a0, time_limit, verbose=True):
    """Smooth gradient descent with soft-max weighting and Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    x = a.copy()
    x_prev = a.copy()
    lr_schedule = 8.0
    stall = 0
    top_k = 60
    beta = 4.0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        momentum = (t - 1.0) / (t + 1.5)
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation
        Ac = fast_autocorr(y)

        # Soft-max gradient
        k_max = min(top_k, size)
        top_idx = np.argsort(Ac)[::-1][:k_max]
        top_vals = np.maximum(Ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        # Gradient: g[j] = sum_k w_k * 2 * y[k-j] (with bounds)
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        # Multiplicative update (mirror descent on simplex)
        eta = lr_schedule / (np.sqrt(t) + 1.0)
        log_a = np.log(y + 1e-30) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                # Restart with perturbation
                rng = np.random.default_rng(t)
                pert = rng.normal(0, 0.02, n) * best_a
                x = norm_sum1(best_a + pert)
                x_prev = x.copy()
                stall = 0
                top_k = min(top_k + 20, 150)
                beta = max(beta - 0.5, 2.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GradDescent ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def local_search_polish(a0, time_limit, verbose=True):
    """Targeted local search: move mass away from peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    rng = np.random.default_rng(12345)

    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0

        # Get top peak lags
        n_peaks = min(30, size)
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # For each peak, identify top contributing pairs and try to reduce them
        improved = False
        for k in top_lags[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            # Top contributing source positions
            n_pairs = min(6, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Find good destination: positions not heavily involved in peak lags
                # Score each position by how much it contributes to peak lags
                peak_involvement = np.zeros(n)
                for k2 in top_lags[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        peak_involvement[idx2] += a[k2i - idx2] * ac[k2i]
                        peak_involvement[k2i - idx2] += a[idx2] * ac[k2i]

                # Prefer moving to low-involvement positions
                dst_candidates = np.where(peak_involvement < np.median(peak_involvement))[0]
                dst_candidates = dst_candidates[dst_candidates != src]
                if len(dst_candidates) == 0:
                    dst_candidates = np.delete(np.arange(n), src)

                # Try a few random destinations plus the best-scoring one
                trial_dsts = set()
                best_dst = dst_candidates[np.argmin(peak_involvement[dst_candidates])]
                trial_dsts.add(int(best_dst))
                n_random = min(4, len(dst_candidates))
                for _ in range(n_random):
                    trial_dsts.add(int(dst_candidates[rng.integers(len(dst_candidates))]))

                for dst in trial_dsts:
                    if dst == src:
                        continue
                    # Try multiple transfer fractions
                    for frac in [0.02, 0.05, 0.08, 0.12, 0.2, 0.3]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

                if improved:
                    break

        if not improved:
            stall += 1
            if stall > max_stall:
                break

            # Random perturbation
            if stall % 50 == 0:
                n_moves = int(rng.integers(5, 20))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LocalSearch ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

        for beta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta)

    # Gaussian
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    # Bimodal
    for c1 in [0.15, 0.2, 0.25, 0.3]:
        for c2 in [0.7, 0.75, 0.8, 0.85]:
            for sigma in [0.05, 0.08, 0.12, 0.15, 0.2]:
                candidates.append(np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) +
                                np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma)))

    # Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50, 100]:
            candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))
            candidates.append(1.0 / (1.0 + np.exp(s * (x - c))))

    # Sinusoidal
    for freq in [1, 2, 3, 4]:
        for ph in [0.0, 0.25, 0.5, 0.75]:
            candidates.append(np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01))

    # Exponential
    for rate in [1, 2, 3, 5, 8, 12]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1)))

    # Product of distance from center
    for p in [1, 2, 3]:
        candidates.append(np.abs(x - 0.5) ** p + 0.1)

    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-15:
            continue
        cand = cand / s
        ac = fast_autocorr(cand)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = cand.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_val = 2.0

    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        best_seq, _ = parametric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n0 = len(best_seq)

    # Phase 1: Quick parametric search at parent length
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 10:
        print(f"\n=== Phase 1: Parametric init (n={n0}) ===")
        a_param, val_param = parametric_init(n0)
        check_and_update(a_param, "parametric")

    # Phase 2: LP-FW with moderate constraints
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, initial_K=250,
                                     max_K=1500, add_per_iter=25, verbose=True)
        check_and_update(a_out, f"lpfw_n{n_best}")

    # Phase 3: Gradient descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Gradient descent (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_descent_smooth(a_in, t3, verbose=True)
        check_and_update(a_out, f"grad_n{n_best}")

    # Phase 4: Local search polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Local search (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = local_search_polish(a_in, t4, verbose=True)
        check_and_update(a_out, f"local_n{n_best}")

    # Phase 5: Second LP-FW round (with different params)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW round 2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t5, solver, initial_K=400,
                                     max_K=2500, add_per_iter=40, verbose=False)
        check_and_update(a_out, f"lpfw2_n{n_best}")

    # Phase 6: Try nearby lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    length_candidates = []
    for delta in [100, -100, 200, -200, 300, -300, 500, -500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            length_candidates.append(new_n)

    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick gradient descent on new length
        t6a = min(remaining * 0.3, 80.0)
        if t6a > 10:
            a_out, _ = gradient_descent_smooth(a_long, t6a, verbose=False)
            check_and_update(a_out, f"grad_n{new_n}")

        # LP-FW if time allows
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 6000:
            t6b = min(remaining * 0.4, 120.0)
            if t6b > 20:
                a_in = norm_sum1(best_seq)
                a_interp = interpolate_sequence(a_in, new_n)
                a_out, _ = lp_frank_wolfe_v2(a_interp, t6b, solver,
                                             initial_K=200, max_K=1200,
                                             add_per_iter=20, verbose=False)
                check_and_update(a_out, f"lpfw_n{new_n}")

    # Phase 7: Final LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 6500:
        t7 = min(remaining * 0.35, 180.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v2(a_in, t7, solver, initial_K=500,
                                         max_K=3000, add_per_iter=50, verbose=True)
            check_and_update(a_out, f"lpfw_final_n{n_best}")

    # Phase 8: Final local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Final local search (n={n_best}) ===")
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = local_search_polish(a_in, t8, verbose=True)
            check_and_update(a_out, f"local_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
