def fast_conv(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def smooth_max(c, beta, top_k=20):
    """Smooth approximation of max(c) using log-sum-exp on top elements."""
    n = len(c)
    k = min(top_k, n)
    # Find top-k indices (use partition for speed)
    if k == n:
        idx = np.arange(n)
        vals = c.copy()
    else:
        # Use argpartition for O(n) top-k
        part_idx = np.argpartition(-c, k-1)[:k]
        idx = part_idx
        vals = c[part_idx].copy()
    max_v = vals.max()
    if max_v < 1e-30:
        return 0.0, np.ones(len(c)) / len(c)
    # Log-sum-exp on top-k
    shifted = beta * (vals - max_v)
    shifted = np.maximum(shifted, -100.0)
    log_sum = np.log(np.exp(shifted).sum())
    smooth_val = max_v + log_sum / beta
    # Compute weights (softmax on top-k)
    weights = np.exp(shifted - log_sum)
    # Full gradient: distribute weights back
    grad_w = np.zeros(n)
    grad_w[idx] = weights
    return smooth_val, grad_w

    eps = 1e-8
    
    # Logsumexp parameters
    T_total = 20000  # total steps for annealing
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Anneal beta (smoothing parameter for max approximation)
        progress = min(1.0, t / T_total)
        cur_beta = beta_start + (beta_end - beta_start) * progress
        cur_top_k = max(5, int(top_k * (1 - 0.5 * progress)))
        
        # Current a from log_a (softmax)
        max_log = log_a.max()
        a = np.exp(log_a - max_log)
        s = a.sum()
        if s < 1e-30:
            a = np.ones(n) / n
            log_a = np.full(n, -np.log(n))
        else:
            a = a / s
        
        # Compute objective and gradient
        c = fast_conv(a)
        smooth_val, grad_w = smooth_max(c, cur_beta, cur_top_k)
        grad = compute_subgradient(a, c, grad_w, n)
        
        # Adam update in log-space
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate schedule (cosine decay)
        lr_t = lr * (1.0 + np.cos(np.pi * min(t / T_total, 1.0))) / 2.0
        lr_t = max(lr_t, lr * 0.01)
        
        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Track best
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={cur_beta:.0f}")
        
        if t > 50000:  # safety cap
            break
    
    return best_a, best_m

def spectral_flatten(a, alpha=0.7, n_iter=100, time_limit=60.0, verbose=False):
    """Iterative spectral flattening: compress dynamic range of |A(f)|."""
    n = len(a)
    a = norm_simplex(a)
    best_a = a.copy()
    c = fast_conv(a)
    best_m = c.max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    cur_alpha = alpha
    stalled = 0
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        mag_mean = mag.mean()
        if mag_mean < 1e-30:
            break
        
        # Compress: mag_new = mag_mean * (mag/mag_mean)^(1-alpha)
        ratio = mag / (mag_mean + 1e-30)
        mag_new = mag_mean * ratio ** (1.0 - cur_alpha)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Blend
        a_trial = 0.5 * a + 0.5 * a_new
        a_trial = norm_simplex(a_trial)
        
        c_trial = fast_conv(a_trial)
        M_trial = c_trial.max()
        
        if M_trial < best_m - 1e-15:
            best_m = M_trial
            best_a = a_trial.copy()
            a = a_trial
            stalled = 0
            # Slightly increase compression
            cur_alpha = min(0.95, cur_alpha * 1.02)
        else:
            stalled += 1
            cur_alpha = max(0.3, cur_alpha * 0.95)
            if stalled > 50:
                break
    
    return best_a, best_m

def local_peak_repair(a0, time_limit=60.0, verbose=False):
    """Iteratively reduce the worst peak by redistributing mass."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    c = fast_conv(a)
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stalled = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stalled < max_stall:
        c = fast_conv(a)
        # Find the worst lag
        k_worst = int(np.argmax(c))
        k_val = c[k_worst]
        
        # Find top contributing pairs for this lag
        i_start = max(0, k_worst - n + 1)
        i_end = min(n, k_worst + 1)
        if i_start >= i_end:
            stalled += 1
            continue
        
        idx = np.arange(i_start, i_end)
        partners = k_worst - idx
        products = a[idx] * a[partners]
        
        # Sort by product (descending)
        order = np.argsort(-products)
        
        improved = False
        # Try to reduce the top contributing pairs
        for oi in order[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            i = idx[oi]
            p = partners[oi]
            prod = products[oi]
            if prod < 1e-10:
                continue
            
            # Try moving mass from i (or p) to a random position
            # Choose the one with more mass to move from
            if a[i] >= a[p]:
                src = i
            else:
                src = p
            
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try several fractions and several destination choices
            for frac in [0.05, 0.1, 0.15, 0.2]:
                amt = src_val * frac
                # Try a few random destinations
                for _ in range(5):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = a_trial[src] - amt
                    a_trial[dst] = a_trial[dst] + amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stalled = 0
        else:
            stalled += 1
            # Random kick to escape local minimum
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-8:
                amt = a[i1] * 0.05
                a[i1] -= amt
                a[i2] += amt
                a = norm_simplex(a)
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stalled = 0
    
    return best_a, best_m

def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)

def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)

def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_simplex(a)

def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)

def make_init_multi_peak(n, n_peaks=4, seed=7):
    """Create a sequence with multiple Gaussian peaks at random positions."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.05
    positions = rng.uniform(0.1, 0.9, n_peaks)
    positions.sort()
    widths = rng.uniform(0.03, 0.1, n_peaks)
    for pos, w in zip(positions, widths):
        a += np.exp(-((x - pos) ** 2) / (2 * w ** 2))
    return norm_simplex(a)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv(a):
    """Fast linear convolution a*a via FFT, returns full convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence score."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(max(0.0, min(1000.0, x))) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def topk_softmax_weights(c, beta, top_k):
    """Compute softmax weights on the top-k elements of c for subgradient computation."""
    n = len(c)
    k = min(top_k, n)
    if k == n:
        idx = np.arange(n)
        vals = c.copy()
    else:
        idx = np.argpartition(-c, k - 1)[:k]
        vals = c[idx].copy()
    max_v = vals.max()
    if max_v < 1e-30:
        w = np.zeros(n)
        w[idx] = 1.0 / k
        return w
    shifted = beta * (vals - max_v)
    shifted = np.maximum(shifted, -100.0)
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    if w_sum < 1e-30:
        w = np.zeros(n)
        w[idx] = 1.0 / k
        return w
    weights = exp_shifted / w_sum
    w = np.zeros(n)
    w[idx] = weights
    return w


def compute_subgradient(a, w, n):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a).

def adam_optimize(a0, time_limit, lr=0.05, beta_start=40.0, beta_end=400.0,

    n = len(a0)
    a = norm_simplex(a0)
    log_a = np.log(np.maximum(a, 1e-15))
    best_a = a.copy()
    c = fast_conv(a)
    best_m = c.max()
    t0 = time.monotonic()

    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    # Nesterov momentum
    x_nest = log_a.copy()

    # Annealing
    T_total = max(1, int(time_limit * 15000))  # estimate total steps
    T_total = min(T_total, 300000)

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Annealing schedule
        progress = min(1.0, t / T_total)
        cur_beta = beta_start + (beta_end - beta_start) * progress
        cur_top_k = max(5, int(top_k * (1.0 - 0.6 * progress)))

        # Nesterov step
        if t > 1:
            x_nest = x_nest + momentum * (x_nest - log_a)
            x_nest = np.maximum(x_nest, -50.0)  # numerical safety
            max_xn = x_nest.max()
            y = np.exp(x_nest - max_xn)
            sy = y.sum()
            if sy < 1e-30:
                y = np.exp(log_a - log_a.max())
                sy = y.sum()
            a = y / sy
        else:
            a = norm_simplex(a0)

        # Compute convolution and gradient
        c = fast_conv(a)
        w = topk_softmax_weights(c, cur_beta, cur_top_k)
        grad = compute_subgradient(a, w, n)

        # Adam update
        m_state = b1 * m_state + (1 - b1) * grad
        v_state = b2 * v_state + (1 - b2) * (grad * grad)
        m_hat = m_state / (1 - b1 ** t)
        v_hat = v_state / (1 - b2 ** t)

        # Learning rate with cosine decay
        lr_t = lr * max(0.01, (1.0 + np.cos(np.pi * progress)) / 2.0)

        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        log_a = np.clip(log_a, -50.0, 10.0)  # numerical safety

        # Track best

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()

            print(f"      Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={cur_beta:.0f}")

    best_m = c.max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    cur_alpha = alpha
    stalled = 0

    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        mag_mean = mag.mean()
        if mag_mean < 1e-30:
            break

        ratio = mag / (mag_mean + 1e-30)
        mag_new = mag_mean * ratio ** (1.0 - cur_alpha)
        mag_new = np.maximum(mag_new, 1e-30)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        a_trial = 0.5 * a + 0.5 * a_new
        a_trial = norm_simplex(a_trial)

        c_trial = fast_conv(a_trial)
        M_trial = c_trial.max()

        if M_trial < best_m - 1e-15:
            best_m = M_trial
            best_a = a_trial.copy()
            a = a_trial
            stalled = 0
            cur_alpha = min(0.95, cur_alpha * 1.03)
        else:
            stalled += 1
            cur_alpha = max(0.2, cur_alpha * 0.93)
            if stalled > 100:
                break

    return best_a, best_m


def local_peak_repair(a0, time_limit=60.0, verbose=False):
    """Iteratively reduce the worst peak by redistributing mass."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    c = fast_conv(a)
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77777)

    stalled = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stalled < max_stall:
        c = fast_conv(a)
        k_worst = int(np.argmax(c))

        i_start = max(0, k_worst - n + 1)
        i_end = min(n, k_worst + 1)
        if i_start >= i_end:
            stalled += 1
            continue

        idx = np.arange(i_start, i_end)
        partners = k_worst - idx
        products = a[idx] * a[partners]

        order = np.argsort(-products)

        improved = False
        for oi in order[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            i = idx[oi]
            p = partners[oi]
            prod = products[oi]
            if prod < 1e-12:
                continue

            if a[i] >= a[p]:
                src = i
            else:
                src = p

            src_val = a[src]
            if src_val < 1e-10:
                continue

            for frac in [0.05, 0.1, 0.15]:
                amt = src_val * frac
                for _ in range(4):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = a_trial[src] - amt
                    a_trial[dst] = a_trial[dst] + amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial.copy()
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stalled = 0
        else:
            stalled += 1
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-8:
                amt = a[i1] * 0.03
                a[i1] -= amt
                a[i2] += amt
                a = norm_simplex(a)
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stalled = 0

    return best_a, best_m


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_simplex(a)


def make_init_multi_peak(n, n_peaks=4, seed=1):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.05
    positions = np.sort(rng.uniform(0.08, 0.92, n_peaks))
    widths = rng.uniform(0.03, 0.12, n_peaks)
    for pos, w in zip(positions, widths):
        a += np.exp(-((x - pos) ** 2) / (2 * w ** 2))
    return norm_simplex(a)


def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 12):
        coeff = rng.normal(0, 0.18 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def make_init_anti_symmetric(n, mode=2):
    """Create a sequence that is approximately anti-symmetric under reflection i -> n-1-i."""
    x = np.linspace(0, 1, n)
    # cos(pi * mode * (2x - 1)) changes sign under x -> 1-x when mode is odd
    a = np.cos(np.pi * mode * (2 * x - 1)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
