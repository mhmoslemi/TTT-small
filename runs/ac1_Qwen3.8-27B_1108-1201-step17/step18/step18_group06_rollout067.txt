def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def exact_subgradient(a, ac, top_k=3):
    n = len(a)
    S = a.sum()
    M = ac.max()

            grad_ac[idx1] += a[ks - idx1]
        j_lo2 = max(0, ks)
        j_hi2 = min(n - 1, ks + n - 1)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            grad_ac[idx2] += a[idx2 - ks]
        return (S * grad_ac - 2.0 * M * a) / (S ** 3) * 2.0 * n
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        ac_top = ac[top_idx]
        sw = ac_top.sum()
        if sw < 1e-30:
            sw = 1.0
        weights = ac_top / sw
        grad_ac = np.zeros(n)
        for ii in range(k_max):
            wi = weights[ii]
            if wi < 1e-10:
                continue
            ks = int(top_idx[ii])
            j_lo1 = max(0, ks - n + 1)

            if j_lo1 <= j_hi1:
                idx1 = np.arange(j_lo1, j_hi1 + 1)

            if j_lo2 <= j_hi2:

                grad_ac[idx2] += wi * a[idx2 - ks]

def lp_direction(f, M, n_constraints=80):
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix, csr_matrix
    n = len(f)
    if n > 4000:
        fc = fast_autocorr(f)
        top_idx = np.argsort(fc)[::-1][:n_constraints]
        A = lil_matrix((len(top_idx), n))
        for row_i, k in enumerate(top_idx):
            ks = int(k)
            j_lo = max(0, ks - n + 1)
            j_hi = min(n - 1, ks)
            for j in range(j_lo, j_hi + 1):
                A[row_i, j] = f[ks - j]
        A = A.tocsr()
    else:
        A = lil_matrix((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                A[k, j] = f[k - j]
        A = A.tocsr()
    b = np.full(A.shape[0], M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60})
        if result.success:
            return np.asarray(result.x, dtype=np.float64)
    except Exception:
        pass
    return None

def make_power_law_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_mild_ramp(n, center_frac=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 + center_frac * x
    return norm_sum1(a)


def make_cosine_taper(n):
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * x / (2.0 * n)) ** 2
    return norm_sum1(a)


def make_sine_bump(n):
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 1.0) / (n + 1.0))
    return norm_sum1(a)


def make_gaussian_bump(n, center=0.5, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return norm_sum1(a)


def make_asymmetric_gauss(n, center=0.6, wl=0.2, wr=0.4):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    ml = x <= center
    mr = x > center
    if np.any(ml):
        a[ml] = np.exp(-((x[ml] - center) / wl) ** 2)
    if np.any(mr):
        a[mr] = np.exp(-((x[mr] - center) / wr) ** 2)
    return norm_sum1(a)


def update_best(best_state, a, label):
    if a is None:
        return best_state
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_state
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_state[1]:
        elapsed = time.monotonic() - best_state[2]
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return (a_norm.copy(), sc, best_state[2])
    return best_state
