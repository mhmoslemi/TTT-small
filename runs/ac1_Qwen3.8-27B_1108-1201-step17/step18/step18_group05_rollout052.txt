def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_linear_ramp(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return norm_sum1(a)

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)

def make_sine(n, freq=1.0):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.sin(np.pi * freq * x)
    a = np.maximum(a, 0.01)
    return norm_sum1(a)

def make_asymmetric(n, alpha=0.3):
    """Asymmetric shape: shifted peak."""
    x = np.linspace(-1, 1, n)
    center = -alpha
    a = np.exp(-((x - center) / 0.8) ** 2) + 0.01
    return norm_sum1(a)

def make_ramp_quadratic(n):
    """Quadratic ramp: a[i] proportional to (i+1)^2."""
    a = np.arange(1, n + 1, dtype=np.float64) ** 2
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_solver():

def fast_autocorr(a):

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        best_a_trial = a_new if 'a_new' in dir() else a
        for trial_eta in [eta, eta * 0.5, eta * 0.25, eta * 2.0]:
            a_trial = a - trial_eta * g_tangent
            a_trial = project_simplex(a_trial)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_a_trial = a_trial
                best_eta = trial_eta

        a = best_a_trial

        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall = 0
            eta = min(eta * 1.02, 3.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.85
            if stall > 400:
                pert = rng.normal(0, 0.002, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a = a_p / a_p.sum()
                stall = 200

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Anneal iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=500, top_K_max=6000):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 100, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                      0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 5 == 0:
            K = min(K + 50, top_K_max)

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=456):
    """Aggressive coordinate descent focusing on peak-reducing moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity (blame) vector
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destinations
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def two_phase_optimize(a0, time_limit, solver, verbose=True, rng_seed=0):
    """
    Two-phase optimization:
    Phase A: Nesterov multiplicative descent (fast, global)
    Phase B: Smoothed annealing (precise, local)
    """
    n = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()

    # Phase A: NMD (60% of time)
    t_a = time_limit * 0.6
    a_out, m_out = nesterov_multiplicative_descent(
        a0, t_a, top_k_init=200, top_k_max=3000,
        c0=3.0, verbose=verbose, rng_seed=rng_seed,
        nesterov_mom=0.95, weight_sharpness=3.0)
    if m_out < best_m:
        best_a = a_out
        best_m = m_out

    # Phase B: Smoothed anneal (40% of time)
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 20:
        t_b = remaining * 0.9
        a_out, m_out = smoothed_anneal(
            best_a, t_b, verbose=verbose, rng_seed=rng_seed + 100,
            tau_min=5.0, tau_max=400.0, eta_init=0.5)
        if m_out < best_m:
            best_a = a_out
            best_m = m_out

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = make_linear_ramp(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-start two-phase optimization ===
    print(f"\n=== Phase 1: Multi-start two-phase (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 600.0)

    if t_phase1 > 120:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_linear_ramp(n0))
        labels.append("ramp")
        inits.append(make_u_shape(n0, sharpness=2.0))
        labels.append("u2.0")
        inits.append(make_asymmetric(n0, alpha=0.25))
        labels.append("asym")
        inits.append(make_sine_ramp(n0))
        labels.append("sine_ramp")

        # Blend parent with ramp
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_linear_ramp(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend_pr")

        # Remove near-duplicates
        seen_sigs = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen_sigs:
                seen_sigs.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(30.0, (t_phase1 - 60) / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 120:
                break
            t_avail = min(t_per, time_budget - 120 - (time.monotonic() - t_global))
            if t_avail < 30:
                break
            print(f"  Two-phase init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = two_phase_optimize(init, t_avail, solver, verbose=False, rng_seed=idx * 17)
            try_update(a_out, f"2ph_{lbl}")

    # === Phase 2: Focused NMD on best with large K ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Deep NMD (n={n_best}) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_multiplicative_descent(
            a_in, t2, top_k_init=500, top_k_max=5000,
            c0=2.5, verbose=True, rng_seed=999,
            nesterov_mom=0.97, weight_sharpness=5.0)
        try_update(a_out, "nmd_deep")

    # === Phase 3: Smoothed anneal on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smoothed anneal (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_anneal(
            a_in, t3, verbose=True, rng_seed=300,
            tau_min=10.0, tau_max=600.0, eta_init=0.3)
        try_update(a_out, "anneal_deep")

    # === Phase 4: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length exploration ===")

    trial_lengths = []
    for delta in [500, 1000, -500, 1500, -1000, 2000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 200.0)
        if t7a > 30:
            a_out, _ = two_phase_optimize(a_long, t7a, solver, verbose=False,
                                            rng_seed=new_n % 1000)
            try_update(a_out, f"2ph_n{new_n}")

    # === Phase 5: LP Frank-Wolfe on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP Frank-Wolfe (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(
            a_in, t5, solver, verbose=True,
            top_K_init=min(1000, 2 * n_best - 1),
            top_K_max=min(6000, 2 * n_best - 1))
        try_update(a_out, "lp_main")

    # === Phase 6: Second NMD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: NMD variant (n={n_best}) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_multiplicative_descent(
            a_in, t6, top_k_init=1000, top_k_max=6000,
            c0=2.0, verbose=False, rng_seed=7777,
            nesterov_mom=0.99, weight_sharpness=8.0)
        try_update(a_out, "nmd_var")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.5, 120.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=12345)
            try_update(a_out, "coord_final")

        # Final brief anneal
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.6, 100.0)
            a_in = norm_sum1(best_seq)
            a_out, _ = smoothed_anneal(a_in, t7b, verbose=False, rng_seed=9999,
                                         tau_min=50.0, tau_max=800.0, eta_init=0.2)
            try_update(a_out, "anneal_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_linear_ramp(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
