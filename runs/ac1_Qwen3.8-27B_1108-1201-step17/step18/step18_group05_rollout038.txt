def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def compute_exact_subgrad(a, ac, top_k=1):
    n = len(a)
    k_sel = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_sel]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** 2
    w = w / w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        for j in range(max(0, k_int - n + 1), min(n, k_int + 1)):
            grad[j] += wi * a[k_int - j]
            grad[k_int - j] += wi * a[j]
    return grad


def compute_soft_subgrad(a, ac, top_k=30, beta=3.0):
    n = len(a)
    k_sel = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_sel]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_sel) / k_sel
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        idx = np.arange(j_start, j_end)
        grad[idx] += wi * a[k_int - idx]
        grad[k_int - idx] += wi * a[idx]
    return grad


def amd_optimize(a0, time_limit, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 5.0
    stall = 0
    cur_top_k = 20
    cur_mode = 'soft'
    cur_beta = 4.0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        if cur_mode == 'exact':
            g = compute_exact_subgrad(y, ac, top_k=max(1, min(5, len(ac))))
        else:
            g = compute_soft_subgrad(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            if cur_mode == 'soft':
                cur_mode = 'exact'
            else:
                cur_mode = 'soft'
                cur_top_k = min(cur_top_k + 5, 80)
                cur_beta = 2.5
            cur_top_k = min(cur_top_k + 3, 80)

        if stall > 5000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            gamma = max(gamma * 0.9, 1.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, mode={cur_mode}")

    return best_a, best_m


def lp_frank_wolfe_sym(a0, time_limit, solver, verbose=True):
    """LP-FW exploiting mirror symmetry for halved LP size."""
    n = len(a0)
    if n % 2 != 0:
        n -= 1
        a0 = a0[:n]
    m = n // 2

    a = norm_sum1(a0[:n])
    # Enforce initial symmetry
    a_sym = np.concatenate([a[:m], a[:m][::-1]])
    a = norm_sum1(a_sym)

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    K = min(100, 2 * n - 1)
    constraint_pool = set()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Build constraint pool: top K + previously active
        top_idx = np.argsort(ac)[::-1][:K]
        for idx in top_idx:
            constraint_pool.add(int(idx))
        # Also add near-max constraints
        threshold = M_curr * 0.95
        near_max = np.where(ac >= threshold)[0]
        for idx in near_max:
            if len(constraint_pool) < 800:
                constraint_pool.add(int(idx))

        # Cap the pool
        if len(constraint_pool) > 600:
            # Keep only the most important
            ac_vals = {k: ac[k] for k in constraint_pool}
            sorted_keys = sorted(ac_vals.keys(), key=lambda x: -ac_vals[x])
            constraint_pool = set(sorted_keys[:600])

        binding_keys = sorted(constraint_pool)
        K_sel = len(binding_keys)

        # Build symmetric constraint matrix T (K_sel x m)
        A_full = a  # length n, symmetric
        T = np.zeros((K_sel, m))
        for ii, k in enumerate(binding_keys):
            k_int = int(k)
            for l in range(k_int, n):
                coeff = A_full[l - k_int]
                if coeff < 1e-15:
                    continue
                if l < m:
                    T[ii, l] += coeff
                else:
                    j = n - 1 - l
                    T[ii, j] += coeff

        b_var = cp.Variable(m, nonneg=True)
        constraints = [T @ b_var <= M_curr, cp.sum(b_var) == 0.5]

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, 2 * n - 1)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 20, 2 * n - 1)
            iter_count += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-10:
            K = min(K + 20, 2 * n - 1)
            iter_count += 1
            continue

        # Build full symmetric b
        b_full = np.concatenate([b_val, b_val[::-1]])
        b_full = norm_sum1(b_full)

        # Backtracking line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 2.0), 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - alpha) * a + alpha * b_full
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * b_full
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        # Re-symmetrize
        a = np.concatenate([a[:m], a[:m][::-1]])
        a = a / a.sum()

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 5 == 0:
            K = min(K + 40, 2 * n - 1)

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-Sym iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m

def peak_shaving(a0, time_limit, verbose=True, rng_seed=100):
    """Targeted peak-shaving: reduce specific autocorrelation peaks via mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:8]

        improved = False
        for k in top_peaks:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            idx = np.arange(j_start, j_end)
            contrib = a[idx] * a[ki - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-contribution destination
            all_contrib = np.zeros(n)
            for kp in top_peaks[:4]:
                kip = int(kp)
                for j in range(max(0, kip - n + 1), min(n, kip + 1)):
                    all_contrib[j] += a[j] * a[kip - j]

            low_mask = all_contrib < 0.3 * M_curr
            candidates = np.where(low_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(all_contrib < all_contrib.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Pick a few candidates
            cand_sample = candidates[:50]
            if len(cand_sample) > 10:
                cand_sample = rng.choice(cand_sample, 10, replace=False)

            found = False
            for dst in cand_sample:
                if dst == src or found:
                    continue
                # Binary search on transfer amount
                lo, hi = 1e-6, src_val * 0.5
                for _ in range(12):
                    mid = (lo + hi) / 2
                    a_trial = a.copy()
                    a_trial[src] -= mid
                    a_trial[dst] += mid
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        lo = mid
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                    else:
                        hi = mid
                if improved:
                    found = True
                    break

            if time.monotonic() - t0 > time_limit:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0) / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

    stall = 0
    sweep_count = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:5]

        # Build sensitivity map
        sens = np.zeros(n)
        for k in top_peaks:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            idx = np.arange(j_start, j_end)
            sens[idx] += a[ki - idx]

        improved = False
        # Try moving mass from high-sensitivity to low-sensitivity positions
        hi_idx = np.where(sens > sens.mean())[0]
        lo_idx = np.where(sens < sens.mean() * 0.5)[0]

        if len(hi_idx) == 0 or len(lo_idx) == 0:
            stall += 1
            continue

        # Sample some pairs
        n_try = min(20, len(hi_idx) * len(lo_idx))
        for _ in range(n_try):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.choice(hi_idx))
            j = int(rng.choice(lo_idx))
            if i == j or a[i] < 1e-12:
                continue
            for frac in [0.001, 0.003, 0.005, 0.002, 0.01]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0) / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
        sweep_count += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PairSwap final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def smooth_anneal(a0, time_limit, verbose=True, rng_seed=300):
    """Smoothed optimization with multi-scale temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    for tau in [10.0, 30.0, 80.0, 200.0, 500.0]:
        phase_start = time.monotonic()
        remaining = time_limit - (phase_start - t0)
        if remaining < 5:
            break
        phase_limit = min(remaining * 0.25, 200.0)

        eta = 0.5
        stall = 0
        while time.monotonic() - phase_start < phase_limit and stall < 200:
            ac = fast_autocorr(a)
            ac_max = ac.max()
            shifted = tau * (ac - ac_max)
            # Clip for numerical stability
            shifted = np.clip(shifted, -50, 0)
            exp_ac = np.exp(shifted)
            p = exp_ac / exp_ac.sum()

            # Gradient via correlation
            from scipy.signal import correlate as sp_corr
            p_cum = np.concatenate([[0.0], np.cumsum(p)])
            k_lo = np.arange(n)
            k_hi = np.minimum(k_lo + n, 2 * n - 1)
            S = p_cum[k_hi] - p_cum[k_lo]
            cross = sp_corr(p, a, mode='valid')
            g = a * S + cross
            g_tangent = g - g.mean()

            a_new = project_simplex(a - eta * g_tangent)
            m_new = fast_autocorr(a_new).max()

            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                eta = min(eta * 1.01, 2.0)
            else:
                a = a_new
                stall += 1
                if stall > 30:
                    eta *= 0.85
                if stall > 100:
                    pert = rng.normal(0, 0.002, n) * a
                    a_p = np.maximum(a + pert, 0.0)
                    a = a_p / a_p.sum()
                    stall = 50

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothAnneal tau={tau} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-method ensemble."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = make_u_shape(n_base, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: AMD from multiple starts (25% time) ===
    print(f"\n=== Phase 1: AMD multi-start (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)

    if t_phase1 > 60:
        inits = [norm_sum1(best_seq)]
        labels = ["parent"]

        # Add diverse inits
        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.5))
        labels.append("u3.5")

        # Slightly perturbed parent
        rng_p = np.random.default_rng(42)
        pert = rng_p.normal(0, 0.01, n0) * best_seq
        inits.append(norm_sum1(best_seq + pert))
        labels.append("parent_pert")

        # Blend
        blend = 0.7 * norm_sum1(best_seq) + 0.3 * make_u_shape(n0, 2.0)
        inits.append(norm_sum1(blend))
        labels.append("blend")

        t_per = max(20.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: Symmetric LP-FW (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Symmetric LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_sym(a_in, t2, solver, verbose=True)
        try_update(a_out, "lp_sym_1")

    # === Phase 3: Smooth annealing (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smooth annealing (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_anneal(a_in, t3, verbose=True, rng_seed=300)
        try_update(a_out, "smooth")

    # === Phase 4: Peak shaving (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak shaving (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t4, verbose=True, rng_seed=100)
        try_update(a_out, "peak_shave")

    # === Phase 5: Second LP-FW round (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW round 2 (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_sym(a_in, t5, solver, verbose=True)
        try_update(a_out, "lp_sym_2")

    # === Phase 6: Non-symmetric LP-FW (8% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Non-sym LP-FW (n={n_best}) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_nonsym(a_in, t6, solver, verbose=True)
        try_update(a_out, "lp_nonsym")

    # === Phase 7: Length exploration (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    if remaining > 80:
        for delta in [200, -200, 400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            new_n = n_best + delta
            if new_n < 1000 or new_n > 8000:
                continue
            if new_n % 2 != 0:
                new_n -= 1
            a_interp = interpolate_sequence(best_seq, new_n)
            t7 = min(remaining * 0.4, 80.0)
            if t7 < 20:
                continue
            print(f"  n={new_n}, AMD...")
            a_out, _ = amd_optimize(a_interp, t7, verbose=False, rng_seed=new_n % 999)
            try_update(a_out, f"amd_n{new_n}")

    # === Phase 8: Final pair-swap polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.7, 120.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = pair_swap_polish(a_in, t8, verbose=True, rng_seed=500)
            try_update(a_out, "polish_1")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.5, 60.0)
            if t8b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = peak_shaving(a_in, t8b, verbose=False, rng_seed=600)
                try_update(a_out, "polish_2")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_u_shape(n_fb, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
