def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_objective(a):
    """Compute 2*n*max_conv/(sum)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, 
          add_per_iter=30, verbose=True):
    """LP-based Frank-Wolfe with backtracking line search."""
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    best_obj = eval_objective(y)
    t0 = time.monotonic()
    
    ac = fast_autocorr(y)
    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_set = list(np.sort(init_idx))
    
    iter_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        curr_obj = 2.0 * n * M_curr  # with sum=1 normalization
        
        if curr_obj < best_obj - 1e-12:
            best_obj = curr_obj
            best_a = y.copy()
        
        # Build constraint matrix
        K = len(constraint_set)
        A_mat = np.zeros((K, n))
        for jj, k in enumerate(constraint_set):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                A_mat[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Compute conv(y, g_dir) to find violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        # Add violated constraints
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.append(int(k))
            constraint_set = list(set(constraint_set))
        
        # Prune constraint set if too large
        if len(constraint_set) > max_K:
            margins = [M_curr - ac[k] for k in constraint_set]
            keep = np.argsort(margins)[:max_K]
            constraint_set = [constraint_set[i] for i in keep]
        
        # Backtracking line search
        alpha = 1.0
        for _bt in range(10):
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new < 1e-15:
                alpha *= 0.5
                continue
            y_new = y_new / s_new
            obj_new = eval_objective(y_new)
            if obj_new < curr_obj - 1e-14:
                break
            alpha *= 0.5
        
        y = y_new
        iter_count += 1
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    print(f"    LP-FW done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, iters={iter_count}")
    return best_a, best_obj

def subgradient_polish(a0, time_limit, top_k=50, lr_init=0.01, verbose=True):
    """Polish using weighted subgradient of the max function."""
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    best_obj = eval_objective(y)
    t0 = time.monotonic()
    
    lr = lr_init
    patience = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        curr_obj = 2.0 * n * M_curr
        
        if curr_obj < best_obj - 1e-13:
            best_obj = curr_obj
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 50:
            lr *= 0.5
            patience = 0
            if lr < 1e-8:
                break
        
        # Compute weighted subgradient
        # Weights: softmax over top-k autocorrelation values
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
        max_v = top_vals.max()
        weights = np.exp((top_vals - max_v) * 5.0)
        weights = weights / weights.sum()
        
        # Subgradient of c_k at y is conv(y, e_k) = shifted y
        # Weighted sum: sum_w w_i * conv(y, e_{top_idx[i]})
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            k_int = int(k)
            w = weights[i]
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start < j_end:
                idx = np.arange(j_start, j_end)
                grad[idx] += w * y[k_int - idx]
        
        # Projected gradient step
        y_new = y - lr * grad
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            continue
        y_new = y_new / s_new
        
        # Only accept if objective improves
        obj_new = eval_objective(y_new)
        if obj_new < curr_obj:
            y = y_new
        else:
            # Try smaller step
            y_new = y - lr * 0.5 * grad
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new > 1e-15:
                y_new = y_new / s_new
                obj_new = eval_objective(y_new)
                if obj_new < curr_obj:
                    y = y_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, lr={lr:.2e}")
    
    elapsed = time.monotonic() - t0
    print(f"    SubGrad done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}")
    return best_a, best_obj

def coord_polish(a0, time_limit, verbose=True):
    """Coordinate descent: transfer mass from peak-contributing positions to safe positions."""
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    best_obj = eval_objective(y)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for t in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Find the lag with maximum autocorrelation
        k_max = int(np.argmax(ac))
        
        # Find the pair (i, j) with i+j=k_max that contributes most
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = y[idx] * y[k_max - idx]
        
        if contrib.max() < 1e-20:
            break
        
        src = idx[np.argmax(contrib)]
        
        # Find a safe destination (minimal impact on top peaks)
        top_peaks = np.argsort(ac)[::-1][:10]
        safe_score = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            if s2 <= e2:
                idx2 = np.arange(s2, e2 + 1)
                safe_score[idx2] += y[k2i - idx2]
        
        # Prefer positions with low safe_score
        candidates = np.where(safe_score < safe_score.mean())[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        candidates = candidates[candidates != src]
        
        if len(candidates) == 0:
            break
        
        # Try several fractions
        improved = False
        for frac in [0.01, 0.005, 0.02, 0.001]:
            amt = y[src] * frac
            if amt < 1e-15:
                continue
            # Pick the best candidate (lowest safe_score)
            dst = candidates[np.argmin(safe_score[candidates])]
            y_trial = y.copy()
            y_trial[src] -= amt
            y_trial[dst] += amt
            y_trial = np.maximum(y_trial, 0.0)
            s_t = y_trial.sum()
            if s_t < 1e-15:
                continue
            y_trial = y_trial / s_t
            obj_trial = eval_objective(y_trial)
            if obj_trial < best_obj - 1e-15:
                y = y_trial
                best_obj = obj_trial
                best_a = y.copy()
                improved = True
                break
        
        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and y[i] > 1e-10:
                amt = y[i] * 0.005
                y[i] -= amt
                y[j] += amt
                y = np.maximum(y, 0.0)
                y = y / y.sum()
                obj_new = eval_objective(y)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = y.copy()
    
    elapsed = time.monotonic() - t0
    print(f"    Coord done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}")
    return best_a, best_obj

def generate_diverse_starts(n, rng_seed=0):
    """Generate diverse initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(rng_seed)
    candidates = []
    
    # Sine-squared
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())
    
    # Asymmetric: concentrated left
    a = np.exp(-3.0 * x)
    candidates.append(a / a.sum())
    
    # Asymmetric: concentrated right  
    a = np.exp(-3.0 * (1.0 - x))
    candidates.append(a / a.sum())
    
    # Double peak
    a = np.exp(-((x-0.3)**2)/0.02) + np.exp(-((x-0.7)**2)/0.02)
    candidates.append(a / a.sum())
    
    # Cosine taper
    a = np.cos(np.pi * (x - 0.5)) ** 2
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())
    
    # Random smoothed
    a = rng.random(n)
    a = np.convolve(a, np.ones(20)/20, mode='same')
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())
    
    # Power law
    a = (i_arr + 1.0) ** 0.5
    candidates.append(a / a.sum())
    
    # Watermelon (high in middle, low at edges)
    a = np.sin(np.pi * x) ** 0.5
    candidates.append(a / a.sum())
    
    # Triangular
    a = 1.0 - np.abs(2.0 * x - 1.0)
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())
    
    return candidates

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with backtracking line search and adaptive constraint management
2. Diverse multi-start initialization
3. Weighted subgradient polishing with adaptive learning rate
4. Coordinate descent final polish
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
import cvxpy as cp
import math

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_objective(a):
    """Compute 2*n*max_conv/(sum)^2 for normalized (sum=1) sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def norm_sum1(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def check_update(a, best_seq, best_score, t_global, label):
    """Check if a improves the best and return updated state."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = float(evaluate_sequence([float(x) for x in a_norm]))
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def build_constraint_matrix(y, constraint_set):
    """Build the A matrix for the LP: A[k,j] = y[k-j] if valid."""
    n = len(y)
    K = len(constraint_set)
    A_mat = np.zeros((K, n))
    for jj, k in enumerate(constraint_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            A_mat[jj, idx] = y[k_int - idx]
    return A_mat


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=1500,
          add_per_iter=30, refresh_every=50, verbose=True):
    """LP-based Frank-Wolfe with backtracking line search."""
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    ac = fast_autocorr(y)
    best_obj = 2.0 * n * ac.max()
    t0 = time.monotonic()

    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_set = list(np.sort(init_idx.astype(int)))

    iter_count = 0
    lp_fail = 0
    stagnation = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        curr_obj = 2.0 * n * M_curr

        if curr_obj < best_obj - 1e-13:
            best_obj = curr_obj
            best_a = y.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Refresh constraint set periodically
        if iter_count > 0 and iter_count % refresh_every == 0:
            init_idx = np.argsort(ac)[::-1][:max(initial_K, max_K // 2)]
            constraint_set = list(np.sort(init_idx.astype(int)))

        K = len(constraint_set)
        if K == 0:
            break

        A_mat = build_constraint_matrix(y, constraint_set)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail += 1
            if lp_fail > 10:
                constraint_set = list(np.argsort(ac)[::-1][:initial_K].astype(int))
                lp_fail = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail += 1
            if lp_fail > 15:
                constraint_set = list(np.argsort(ac)[::-1][:max(100, initial_K // 2)].astype(int))
                lp_fail = 0
            continue

        lp_fail = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations and add to constraint set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.append(int(k))
            constraint_set = list(set(constraint_set))

        # Prune if too large
        if len(constraint_set) > max_K:
            margins = [M_curr - ac[int(k)] for k in constraint_set]
            keep = np.argsort(margins)[:max_K]
            constraint_set = [constraint_set[i] for i in keep]

        # Backtracking line search
        alpha = 1.0
        y_new = y.copy()
        for _bt in range(8):
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            s_new = y_trial.sum()
            if s_new < 1e-15:
                alpha *= 0.5
                continue
            y_trial = y_trial / s_new
            obj_new = 2.0 * n * fast_autocorr(y_trial).max()
            if obj_new <= curr_obj + 1e-12:
                y_new = y_trial
                break
            alpha *= 0.5
        else:
            # If all backtracking failed, take a tiny step
            y_new = (1.0 - 0.01) * y + 0.01 * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new > 1e-15:
                y_new = y_new / s_new
            else:
                y_new = y.copy()

        y = y_new
        iter_count += 1

        if verbose and iter_count % 30 == 1:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    print(f"    LP-FW done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, iters={iter_count}")
    return best_a, best_obj


def subgradient_polish(a0, time_limit, top_k=80, lr_init=0.003, verbose=True):
    """Polish using weighted subgradient of the max autocorrelation."""
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    ac = fast_autocorr(y)
    best_obj = 2.0 * n * ac.max()
    t0 = time.monotonic()

    lr = lr_init
    patience = 0
    max_patience = 80

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        curr_obj = 2.0 * n * M_curr

        if curr_obj < best_obj - 1e-13:
            best_obj = curr_obj
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            lr *= 0.4
            patience = 0
            if lr < 1e-9:
                break

        # Weighted subgradient using softmax over top-k lags
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx].copy()
        # Shift for numerical stability
        top_vals -= top_vals.max()
        weights = np.exp(top_vals * 10.0)
        weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            k_int = int(top_idx[i])
            w = weights[i]
            if w < 1e-8:
                continue
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start < j_end:
                idx = np.arange(j_start, j_end)
                grad[idx] += w * y[k_int - idx]

        # Gradient step with projection
        y_new = y - lr * grad
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            continue
        y_new = y_new / s_new

        obj_new = 2.0 * n * fast_autocorr(y_new).max()
        if obj_new < curr_obj:
            y = y_new
        else:
            # Halve step
            y_new = y - lr * 0.5 * grad
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new > 1e-15:
                y_new = y_new / s_new
                obj_new2 = 2.0 * n * fast_autocorr(y_new).max()
                if obj_new2 < curr_obj:
                    y = y_new

        if verbose and t % 200 == 1:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, lr={lr:.2e}")

    elapsed = time.monotonic() - t0
    print(f"    SubGrad done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}")
    return best_a, best_obj


def coord_polish(a0, time_limit, verbose=True):
    """Coordinate descent: move mass from peak contributors to safe positions."""
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    ac = fast_autocorr(y)
    best_obj = 2.0 * n * ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for t in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        k_max = int(np.argmax(ac))

        # Find top contributing pair for the max lag
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        if i_start > i_end:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and y[i] > 1e-10:
                amt = y[i] * 0.003
                y[i] -= amt
                y[j] += amt
                y = np.maximum(y, 0.0)
                y = y / y.sum()
                obj_new = 2.0 * n * fast_autocorr(y).max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = y.copy()
            continue

        idx = np.arange(i_start, i_end + 1)
        contrib = y[idx] * y[k_max - idx]
        src = idx[np.argmax(contrib)]

        # Compute safety scores for all positions
        top_peaks = np.argsort(ac)[::-1][:12]
        safe_score = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            if s2 <= e2:
                idx2 = np.arange(s2, e2 + 1)
                safe_score[idx2] += y[k2i - idx2]

        # Find safe destination
        mask = (safe_score < np.median(safe_score)) & (np.arange(n) != src)
        candidates = np.where(mask)[0]
        if len(candidates) == 0:
            candidates = np.where(np.arange(n) != src)[0]

        improved = False
        for frac in [0.008, 0.004, 0.015, 0.002]:
            amt = y[src] * frac
            if amt < 1e-15:
                continue
            dst = candidates[np.argmin(safe_score[candidates])]
            y_trial = y.copy()
            y_trial[src] -= amt
            y_trial[dst] += amt
            y_trial = np.maximum(y_trial, 0.0)
            s_t = y_trial.sum()
            if s_t < 1e-15:
                continue
            y_trial = y_trial / s_t
            obj_trial = 2.0 * n * fast_autocorr(y_trial).max()
            if obj_trial < best_obj - 1e-15:
                y = y_trial
                best_obj = obj_trial
                best_a = y.copy()
                improved = True
                break

        if not improved and t % 50 == 0:
            # Random exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and y[i] > 1e-10:
                amt = y[i] * rng.uniform(0.002, 0.01)
                y[i] -= amt
                y[j] += amt
                y = np.maximum(y, 0.0)
                y = y / y.sum()
                obj_new = 2.0 * n * fast_autocorr(y).max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = y.copy()

    elapsed = time.monotonic() - t0
    print(f"    Coord done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}")
    return best_a, best_obj


def generate_diverse_starts(n, rng_seed=0):
    """Generate diverse initial sequences for multi-start."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(rng_seed)
    candidates = []

    # Sine-squared
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())

    # Asymmetric left
    a = np.exp(-4.0 * x)
    candidates.append(a / a.sum())

    # Asymmetric right
    a = np.exp(-4.0 * (1.0 - x))
    candidates.append(a / a.sum())

    # Double bump
    a = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    candidates.append(a / a.sum())

    # Cosine taper
    a = np.cos(np.pi * (x - 0.5)) ** 2
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())

    # Random smoothed
    a = rng.random(n)
    kernel = np.ones(30) / 30.0
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())

    # Watermelon (sharper center)
    a = np.sin(np.pi * x) ** 0.4
    candidates.append(a / a.sum())

    # Triangular
    a = 1.0 - np.abs(2.0 * x - 1.0)
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())

    # Bell curve
    a = np.exp(-((x - 0.5) ** 2) / 0.05)
    candidates.append(a / a.sum())

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
