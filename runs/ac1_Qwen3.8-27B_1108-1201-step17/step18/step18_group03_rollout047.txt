import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, best_init, best_score):
    """Try parametric families to find a good starting point."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = norm_sum1(best_init)
    best_val = 2.0 * n * fast_autocorr(best_a).max()

    def try_update(a, label):
        nonlocal best_a, best_val
        a_n = norm_sum1(a)
        ac = fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = a_n.copy()
            if time.monotonic() - t0 > 5:
                print(f"    Parametric [{label}]: J={val:.6f}")

    # Gaussian centered at midpoint
    for sigma in np.linspace(0.15, 0.55, 10):
        try_update(np.exp(-((x - 0.5) ** 2) / (2.0 * sigma * sigma)), f"gauss_{sigma:.2f}")

    # Beta-like distributions
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = np.maximum(i_arr + 0.5, 0.1) ** alpha * np.maximum(n - i_arr, 0.1) ** beta
            try_update(a, f"beta_{alpha}_{beta}")

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        try_update((i_arr + 1.0) ** p, f"pow_{p}")
        try_update((n - i_arr) ** p, f"ipow_{p}")

    # Sine-based
    for freq in [1, 2, 3]:
        for ph in [0.0, 0.25, 0.5, 0.75]:
            try_update(np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01), f"sin{freq}_{ph}")

    # Exponential decay from edges
    for rate in [1, 2, 3, 5]:
        try_update(np.exp(-rate * x) + np.exp(rate * (x - 1.0)), f"exp_edges_{rate}")

    # Truncated cosine
    for f_c in [0.5, 1.0, 1.5, 2.0]:
        try_update(np.maximum(np.cos(f_c * np.pi * (x - 0.5)), 0.01), f"cos_{f_c}")

    elapsed = time.monotonic() - t0
    print(f"  Parametric init done ({elapsed:.1f}s): best J={best_val:.6f}")
    return best_a, best_val


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=2500, add_per_iter=30, verbose=True):
    """LP-Frank-Wolfe with cutting planes and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_fw = a.copy()

    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    # Initialize with top-K lags
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x_fw)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x_fw.copy()

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K < 5:
            # Reinitialize
            constraint_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
            continue

        # Build constraint matrix: row k has a[k-i] in column i
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        # LP: min M s.t. A*g <= M, sum(g)=1, g>=0
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 3:
                constraint_set = set(int(k) for k in constraint_list[:min(initial_K, len(constraint_list))])
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov-accelerated step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x_fw = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def subgradient_polish(a0, time_limit, verbose=True):
    """Polish using Nesterov-accelerated mirror descent on logsumexp approximation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    gamma = 2.0
    beta_start = 30.0
    beta_end = 200.0
    total_iters = max(200, int(time_limit * 5))

    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Interpolate beta
        frac = t / total_iters
        beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov lookahead
        if t == 1:
            y = x.copy()
        else:
            mu = 2.0 / (t + 3.0)
            beta_t = (1.0 - mu) / (1.0 + mu)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Compute autocorrelation of y
        ac = fast_autocorr(y)

        # Compute logsumexp gradient
        ac_shifted = ac - ac.max()
        weights = np.exp(beta * ac_shifted)
        weights /= weights.sum()

        # Subgradient: grad[i] = 2 * sum_k weights[k] * y[k-i]
        grad = np.zeros(n)
        # Use convolution for efficiency
        fft_size = 1 << (2 * n - 2).bit_length()
        Wf = np.fft.rfft(weights, fft_size)
        Yf = np.fft.rfft(y, fft_size)
        # grad[i] = 2 * sum_k w[k] * y[k-i] = 2 * (w * y_flipped)[i] ... 
        # Actually: (w * y)[k] = sum_i w[i]*y[k-i], so w[k]*y[k-i] summed over k:
        # sum_k w[k]*y[k-i] = (w_rev * y)[i] where w_rev[j] = w[2n-2-j]
        # Simpler: grad = 2 * irfft(fft(w)*conj(fft(y)))[:n] ... no that's cross-correlation
        # Let's just do: grad[i] = 2 * sum_k weights[k] * y[k-i] (with y[j]=0 outside [0,n-1])
        # This is the cross-correlation of weights and y, evaluated at lag i.
        # cross_corr[i] = sum_k weights[k+i] * y[k] = (weights_rev * y)[i+n-1]
        # Hmm, let me just use the direct formula via FFT:
        # sum_k w[k]*y[k-i] = element i of the convolution w * y_flipped
        # where y_flipped[j] = y[n-1-j]
        # Actually the simplest: use the fact that this is like a "cross-correlation"
        # Let's compute it as: sum_k w[k]*y[k-i] for each i
        # = sum_k w[k+i]*y[k] (reindex with k->k+i)
        # = (w_shifted * y) at lag 0... 
        
        # Easiest: just use np.convolve for the weighted sum
        # grad[i] = 2 * sum_k weights[k] * y[k-i] (y[j]=0 if j<0 or j>=n)
        # = 2 * (weights * y_rev)[i + n - 1] where y_rev[j] = y[n-1-j]
        y_rev = y[::-1]
        cross = np.convolve(weights, y_rev, mode='full')
        grad = 2.0 * cross[n - 1:2 * n - 1]

        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y + 1e-15) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD polish t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.0f}")

    return best_a, best_m


def peak_reduction(a0, time_limit, verbose=True):
    """Targeted peak reduction: move mass away from peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    stall = 0
    max_stall = 500
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0

        # Find top peaks
        n_peaks = min(10, len(ac))
        top_peaks = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        for k in top_peaks[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break
            k_int = int(k)
            # Contributing pairs: (i, k_int - i) for valid i
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            partner_idx = k_int - idx
            contrib = a[idx] * a[partner_idx]

            # Top contributing pairs
            n_top_pairs = min(10, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_top_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                partner = partner_idx[pi]
                src_val = a[src]
                partner_val = a[partner]
                if src_val < 1e-10 and partner_val < 1e-10:
                    continue

                # Choose which element to reduce (the one with more "alternative" positions)
                # Try moving from the larger one
                if src_val >= partner_val:
                    move_from = src
                    move_to_partner = partner
                else:
                    move_from = partner
                    move_to_partner = src

                # Find good destination: a position that doesn't contribute to any top peak
                # Score each position by its contribution to top peaks
                peak_contrib = np.zeros(n)
                for k2 in top_peaks[:5]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        partner2 = k2i - idx2
                        peak_contrib[idx2] += a[partner2] * ac[k2i]
                        peak_contrib[partner2] += a[idx2] * ac[k2i]

                # Find position with lowest peak contribution, far from source
                peak_contrib[move_from] = np.inf
                # Prefer positions far from the peak region
                dst_candidates = np.argsort(peak_contrib)[:20]

                for dst in dst_candidates[:5]:
                    if dst == move_from:
                        continue
                    # Try moving a fraction
                    for frac in [0.02, 0.05, 0.1, 0.15]:
                        amt = a[move_from] * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[move_from] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-14:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        if not improved:
            stall += 1
            if stall > max_stall:
                break
            # Random perturbation every 20 stalls
            if stall % 20 == 0:
                n_moves = int(rng.integers(2, 8))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = float(rng.uniform(0.01, 0.05))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-14:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakReduce ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            stall += 1
            if stall > max_stall:
                break

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.exp(-((x - 0.5) ** 2) / (2 * 0.3 ** 2)))
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick parametric init at current length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)
    print(f"\n=== Phase 1: Parametric init (n={n0}) ===")
    t1 = min(remaining * 0.05, 40.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, best_seq, best_score)
        check_and_update(a_param, "parametric")

    # === Phase 2: Main LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 30 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, initial_K=400, max_K=2500,
                                   add_per_iter=40, verbose=True)
        check_and_update(a_out, f"lp_main_n{n_best}")

    # === Phase 3: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Subgradient polish (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_polish(a_in, t3, verbose=True)
        check_and_update(a_out, f"sgd_n{n_best}")

    # === Phase 4: Peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak reduction (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction(a_in, t4, verbose=True)
        check_and_update(a_out, f"peakred_n{n_best}")

    # === Phase 5: Pairwise improvement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pairwise improvement (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_improvement(a_in, t5, verbose=False)
        check_and_update(a_out, f"pairwise_n{n_best}")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    length_candidates = []
    for delta in [-1000, -500, 500, 1000, 1500, 2000, -1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            length_candidates.append(new_n)

    for new_n in sorted(set(length_candidates))[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_interp = interpolate_sequence(best_seq, new_n)

        # Quick LP-FW
        t6 = min(remaining * 0.4, 150.0)
        if t6 > 20 and new_n <= 7000:
            a_out, _ = lp_frank_wolfe(a_interp, t6, solver, initial_K=300,
                                       max_K=2000, add_per_iter=30, verbose=False)
            check_and_update(a_out, f"lp_n{new_n}")

        # Quick peak reduction
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t6b = min(remaining * 0.3, 80.0)
            if t6b > 10:
                a_out, _ = peak_reduction(norm_sum1(best_seq if len(best_seq) == new_n else interpolate_sequence(best_seq, new_n)),
                                          t6b, verbose=False)
                check_and_update(a_out, f"peakred_n{new_n}")

    # === Phase 7: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 60 and n_best <= 7000:
        t7 = min(remaining * 0.45, 250.0)
        if t7 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=500,
                                       max_K=3000, add_per_iter=50, verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final subgradient + peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
        t8a = min(remaining * 0.4, 100.0)
        if t8a > 15:
            a_out, _ = subgradient_polish(norm_sum1(best_seq), t8a, verbose=False)
            check_and_update(a_out, f"sgd_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.4, 80.0)
        if t8b > 10:
            a_out, _ = peak_reduction(norm_sum1(best_seq), t8b, verbose=False)
            check_and_update(a_out, f"peakred_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.exp(-((x - 0.5) ** 2) / (2 * 0.25 ** 2)))
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
