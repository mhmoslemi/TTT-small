For the spectral peak shaving, here's the key idea:

import numpy as np

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely evaluate the sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

    except Exception:
        return float('inf')


def subgradient_weighted(a, c, top_k=15, beta=25.0):
    """Compute weighted subgradient focusing on top peaks."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()

    max_v = top_vals[0] if top_vals[0] > 1e-30 else 1.0
    exponents = beta * (top_vals - max_v) / max_v
    exponents = np.maximum(exponents, -50.0)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def ftrl_optimize(a0, time_limit, verbose=False, max_restarts=3):
    """FTRL mirror descent with adaptive annealing and restarts."""
    n = len(a0)

def spectral_refine_step(a, strength=0.2):
    """Targeted spectral refinement: reduce frequencies causing peak interference."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    c = fast_conv(a)
    k_peak = int(np.argmax(c))

    freqs = np.arange(len(mag))
    cos_vals = np.cos(2.0 * np.pi * freqs * k_peak / n)

    pos_mask = cos_vals > 0.3
    neg_mask = cos_vals < -0.3

    mag_new = mag.copy()
    mag_new[pos_mask] *= (1.0 - strength * 0.5)
    if neg_mask.any():
        mag_new[neg_mask] *= (1.0 + strength * 0.3)

    mag_mean = mag.mean()
    if mag_mean > 1e-30:
        ratio = mag_new / mag
        ratio = np.maximum(ratio, 0.5)
        mag_new = mag * np.minimum(ratio, 2.0)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

    blend = min(strength, 0.5)
    a_out = (1.0 - blend) * a + blend * a_new
    return norm_simplex(a_out)

def coordinate_descent_peaked(a0, time_limit, verbose=False):

    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    stall = 0

        c = fast_conv(a)
        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]

        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Find best destination by trying several
            candidates = []
            for _ in range(20):
                candidates.append(int(rng.integers(n)))
            # Also try positions far from the peak
            dist = np.abs(np.arange(n) - src)
            far_idx = np.argsort(dist)[::-1]
            candidates.extend([int(far_idx[i]) for i in range(0, min(10, n), max(1, n // 10))])

            best_move = None
            best_move_val = best_m
            for dst in candidates:
                if dst == src or a[dst] >= src_val:
                    continue
                # Binary search on move fraction
                lo, hi = 0.001, 0.5
                best_frac = lo
                for _ in range(6):
                    mid = (lo + hi) / 2
                    amt = src_val * mid
                    a_tr = a.copy()
                    a_tr[src] -= amt
                    a_tr[dst] += amt
                    a_tr = norm_simplex(a_tr)
                    M_tr = fast_conv(a_tr).max()
                    if M_tr < best_move_val:
                        best_move_val = M_tr
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid

                if best_frac > 0.01:
                    amt = src_val * best_frac
                    a_tr = a.copy()
                    a_tr[src] -= amt
                    a_tr[dst] += amt
                    a_tr = norm_simplex(a_tr)
                    M_tr = fast_conv(a_tr).max()
                    if M_tr < best_m - 1e-14:
                        a = a_tr
                        best_m = M_tr
                        best_a = a.copy()
                        improved = True
                        break

            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # Perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            a_r += coeff * np.sin(2 * np.pi * f_r * x_r + phase_r)
        a_r = np.maximum(a_r, 0.001)
        inits.append(norm_simplex(a_r))
        labels.append("rand_smooth")

        t_per = t_phase1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            remaining_now = time_budget - 30 - elapsed_so_far
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 10:
                break
            print(f"    FTRL '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = ftrl_optimize(init, t_avail, verbose=False, max_restarts=2)
            try_update(a_out, f"ftrl_{lbl}")

            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now > 20:
                a_os, _ = oscillatory_refine(a_out, min(20.0, remaining_now * 0.3), verbose=False)
                try_update(a_os, f"osc_{lbl}")

    # === Phase 2: Intensive FTRL on best found (25% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Intensive FTRL (n={n_best}) ===")
    t2 = min(remaining * 0.25, 380.0)
    if t2 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = ftrl_optimize(a_in, t2 * 0.6, verbose=True, max_restarts=4)
        try_update(a_out, f"ftrl_intensive_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.15, 120.0)
        if t2b > 15:
            a_out2, _ = ftrl_optimize(a_out, t2b, verbose=False, max_restarts=2)
            try_update(a_out2, f"ftrl_intensive2_n{n_best}")

    # === Phase 3: Oscillatory refinement (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Oscillatory refinement (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = oscillatory_refine(a_in, t3, verbose=True)
        try_update(a_out, f"osc_n{n_best}")

        # Quick FTRL after oscillatory
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.10, 80.0)
        if t3b > 10:
            a_out2, _ = ftrl_optimize(a_out, t3b, verbose=False, max_restarts=1)
            try_update(a_out2, f"ftrl_after_osc_n{n_best}")

    # === Phase 4: Coordinate descent (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate Descent (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_in = norm_simplex(best_seq)
        a_out, _ = coordinate_descent_peaked(a_in, t4, verbose=True)
        try_update(a_out, f"cd_n{n_best}")

    # === Phase 5: Multi-length exploration (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length (base n={n_best}) ===")

    for delta in [500, 1000, -500]:
        new_n = n_best + delta
        if new_n < 2000 or new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t5 = min(remaining * 0.30, 120.0)
        if t5 > 20:
            a_out, _ = ftrl_optimize(a_long, t5, verbose=False, max_restarts=2)
            try_update(a_out, f"ftrl_n{new_n}")

    # === Phase 6: Final intensive polish (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}) ===")
    t6 = min(remaining * 0.50, 150.0)
    if t6 > 15:
        a_in = norm_simplex(best_seq)
        a_out, _ = ftrl_optimize(a_in, t6 * 0.5, verbose=True, max_restarts=3)
        try_update(a_out, f"ftrl_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out2, _ = coordinate_descent_peaked(a_out, min(remaining * 0.5, 60.0), verbose=False)
            try_update(a_out2, f"cd_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out3, _ = oscillatory_refine(a_out2, min(remaining * 0.6, 40.0), verbose=False)
            try_update(a_out3, f"osc_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
