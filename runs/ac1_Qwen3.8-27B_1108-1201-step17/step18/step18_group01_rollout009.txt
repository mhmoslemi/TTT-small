import numpy as np
from scipy.optimize import linprog
import time

def lp_direction(a, M, verbose=False):
    """Solve LP to find direction b that maximizes sum(b) subject to (a*b)_k <= M, b >= 0, sum(b) = 1."""
    n = len(a)
    # Variables: b_0, ..., b_{n-1}
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Equality constraint: sum(b) = 1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    # Inequality constraints: (a*b)_k <= M for k = 0, ..., 2n-2
    # (a*b)_k = sum_j a[j] * b[k-j] where 0 <= j < n and 0 <= k-j < n
    # For a given k, the valid j range is max(0, k-n+1) to min(n-1, k)
    
    # Build the constraint matrix (2n-1) x n
    # This is a sparse matrix (banded)
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = M * np.ones(2*n-1)
    
    # Bounds: b_i >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x
```

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lp_solve_full(a, M):
    """Solve the full LP: max sum(b) s.t. b>=0, sum(b)=1, (a*b)_k <= M."""
    n = len(a)
    c_obj = -np.ones(n)
    
    # Build constraint matrix as sparse
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n-1, k)
        if j_start <= j_end:
            rows.extend([k]*(j_end-j_start+1))
            cols.extend(range(j_start, j_end+1))
            vals.extend(a[j_start:j_end+1].tolist())
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = M * np.ones(2*n-1)
    
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)]*n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x

def lp_optimize(a0, time_limit, verbose=False):
    """Iterative LP-based optimization."""
    a = norm_sum1(a0)
    n = len(a)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    step = 0.1
    patience = 0
    max_patience = 50
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            step *= 0.5
            patience = 0
            if step < 0.01:
                break
        
        b = lp_solve_full(a, M)
        if b is None:
            break
        
        # Update
        t = step
        a_new = (1-t)*a + t*b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new > M * 1.01:
            # Too aggressive, reduce step
            step *= 0.5
            if step < 0.01:
                break
        else:
            a = a_new
            # Slowly increase step
            step = min(0.3, step * 1.05)
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  LP iter patience={patience}, M={best_M:.8f}, step={step:.4f}, t={elapsed:.0f}s")
    
    return best_a

    """Solve LP: min t s.t. b>=0, sum(b)=1, (a*b)_k <= t for all k."""
    n = len(a)
    # Variables: b_0, ..., b_{n-1}, t (n+1 variables)
    # Objective: minimize t => c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequality: (a*b)_k - t <= 0 for all k
    # (a*b)_k = sum_j a[j]*b[k-j] - t <= 0
    # Row k: a[j] for valid j in columns 0..n-1, and -1 in column n (for t)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
        rows.append(k)
        cols.append(n)  # column for t
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    bounds = [(0, None)]*n + [(None, None)]  # b_i >= 0, t free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    return res.x[:n], res.x[-1]
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based conditional gradient
method with subgradient refinement and coordinate descent polish.

Key innovation: Uses a linear program to find the optimal descent direction
at each step, solving min_t s.t. (a*b)_k <= t, b in simplex.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n+1)
    cond = u * ind > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_lp_structure(n):
    rows = np.zeros(0, dtype=np.int32)
    cols = np.zeros(0, dtype=np.int32)
    row_parts = []
    col_parts = []
    for k in range(2*n-1):
        js = max(0, k-n+1)
        je = min(n-1, k)
        cnt = je - js + 1
        row_parts.append(np.full(cnt, k, dtype=np.int32))
        col_parts.append(np.arange(js, je+1, dtype=np.int32))
        # t variable column
        row_parts.append(np.array([k], dtype=np.int32))
        col_parts.append(np.array([n], dtype=np.int32))
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    return rows, cols

def solve_lp(a, rows, cols):
    n = len(a)
    size = 2*n-1
    vals = np.ones(len(rows))
    mask = cols < n
    vals[mask] = a[cols[mask]]
    A_ub = csr_matrix((vals, (rows, cols)), shape=(size, n+1))
    b_ub = np.zeros(size)
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    return res.x[:n], res.x[-1]

def lp_optimize(a0, time_limit, verbose=True):
    a = norm_sum1(a0)
    n = len(a)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    rows, cols = build_lp_structure(n)
    step = 0.05
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
        
        b, t_val = solve_lp(a, rows, cols)
        if b is None:
            break
        
        # Line search for step size
        for s in [step, step*2, step*0.5, step*0.3]:
            a_try = (1-s)*a + s*np.maximum(b, 0)
            a_try = np.maximum(a_try, 0)
            ss = a_try.sum()
            if ss < 1e-15:
                continue
            a_try = a_try / ss
            M_try = fast_autocorr(a_try).max()
            if M_try < M:
                a = a_try
                break
        else:
            # No improvement, reduce step
            step *= 0.5
            if step < 0.005:
                # Try subgradient fallback
                top = np.argsort(ac)[::-1][:20]
                g = np.zeros(n)
                for k in top:
                    k = int(k)
                    js = max(0, k-n+1)
                    je = min(n-1, k)
                    idx = np.arange(js, je+1)
                    g[idx] += 2*a[k-idx]
                g /= len(top)
                a_try = a - 0.01 * g
                a_try = proj_simplex(a_try)
                M_try = fast_autocorr(a_try).max()
                if M_try < M:
                    a = a_try
                else:
                    break
            continue
        
        iters += 1
        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_M
            print(f"  LP iter={iters} t={elapsed:.0f}s score={score:.6f} M={best_M:.8f} step={step:.4f}")
        
        # Gradually increase step
        step = min(0.2, step * 1.1)
    
    if verbose:
        elapsed = time.monotonic() - t0
        score = 2.0 * n * best_M
        print(f"  LP done: {iters} iters, {elapsed:.0f}s, score={score:.6f}")
    return best_a

def subgrad_optimize(a0, time_limit, verbose=False):
    a = norm_sum1(a0)
    n = len(a)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 0.05
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 200:
            break
        
        top_k = min(50, len(ac))
        top = np.argsort(ac)[::-1][:top_k]
        weights = np.maximum(ac[top], 1e-20) ** 1.5
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top):
            k = int(k)
            js = max(0, k-n+1)
            je = min(n-1, k)
            idx = np.arange(js, je+1)
            g[idx] += 2*w*a[k-idx]
        
        a_try = a - eta * g
        a_try = proj_simplex(a_try)
        M_try = fast_autocorr(a_try).max()
        if M_try <= M:
            a = a_try
        else:
            eta *= 0.5
            if eta < 0.001:
                break
    
    return best_a

def coord_polish(a0, time_limit, verbose=False):
    a = norm_sum1(a0)
    n = len(a)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M = ac.max()
        top_k = min(10, len(ac))
        top = np.argsort(ac)[::-1][:top_k]
        improved = False
        
        for k in top:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k-n+1)
            je = min(n-1, k)
            idx = np.arange(js, je+1)
            contrib = a[idx] * a[k-idx]
            src = idx[np.argmax(contrib)]
            
            # Find destination: position with minimum peak contribution
            peak_influence = np.zeros(n)
            for k2 in top[:5]:
                k2 = int(k2)
                s2 = max(0, k2-n+1)
                e2 = min(n-1, k2)
                if s2 <= e2:
                    peak_influence[s2:e2+1] += 1
            
            candidates = np.where(peak_influence == 0)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(peak_influence[candidates])]
            if dst == src:
                continue
            
            for frac in [0.02, 0.05, 0.1]:
                amt = a[src] * frac
                a_try = a.copy()
                a_try[src] -= amt
                a_try[dst] += amt
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                M_try = fast_autocorr(a_try).max()
                if M_try < M - 1e-15:
                    a = a_try
                    M = M_try
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.03
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def parametric_init(n):
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_score = np.inf
    
    families = []
    # Sine powers
    for p in [1, 2, 3, 4, 6]:
        families.append(np.maximum(np.sin(np.pi * x)**p, 0.01))
    # Gaussian
    for c in [0.3, 0.4, 0.5]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            families.append(np.exp(-((x-c)**2)/(2*sigma**2)))
    # Power laws
    for p in [0.5, 1, 2, 3, 5]:
        families.append((i_arr+1)**(-p) * (n-i_arr-1)**(-p))
    # Raised cosine
    for w in [0.1, 0.2, 0.3, 0.5]:
        families.append(0.5*(1+np.cos(2*np.pi*w*x)))
    # Two-hump
    for d in [0.1, 0.2, 0.3]:
        families.append(np.exp(-10*(x-0.3-d)**2) + np.exp(-10*(x-0.7+d)**2))
    
    for fam in families:
        fam_n = norm_sum1(fam)
        score = 2.0 * n * fast_autocorr(fam_n).max()
        if score < best_score:
            best_score = score
            best_a = fam_n.copy()
    
    return best_a
