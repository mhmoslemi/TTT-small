def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(v, target_sum=1.0):
    n = len(v)
    lo = v.min() - target_sum
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.maximum(v - mid, 0.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.maximum(v - (lo + hi) / 2.0, 0.0)
    result_sum = result.sum()
    if result_sum > 1e-10:
        result = result * (target_sum / result_sum)
    return result


def cross_corr_fft(a, b):
    n = len(a)
    m = len(b)
    L = 1 << (n + m - 1).bit_length()
    A = np.fft.fft(a, L)
    B = np.fft.fft(b, L)
    return np.fft.ifft(A * np.conj(B)).real


def compute_grad(a, weights):
    n = len(a)
    return 2.0 * cross_corr_fft(a, weights)[:n]


def softmax_weights(conv, tau):
    cmax = conv.max()
    logits = (conv - cmax) / tau
    logits = np.clip(logits, -50, 50)
    w = np.exp(logits)
    w_sum = w.sum()
    if w_sum < 1e-30:
        idx = np.argmax(conv)
        w = np.zeros_like(conv)
        w[idx] = 1.0
    else:
        w = w / w_sum
    return w


def solve_dual_mw(f, M, n_mw_iter=80, eta=0.15):
    n = len(f)
    size = 2 * n - 1
    y = np.ones(size) / size
    for _ in range(n_mw_iter):
        h = cross_corr_fft(f, y)[:n]
        j_star = int(np.argmin(h))
        if h[j_star] >= 1.0 - 1e-6:
            break
        lo = j_star
        hi = min(size - 1, j_star + n - 1)
        if lo <= hi:
            segment = f[:hi - lo + 1]
            y[lo:hi + 1] *= np.exp(eta * segment)
    return y


def fista_optimize(a0, time_limit, tau_init=0.004, tau_min=0.0003, seed=42,

def freq_domain_iterate(a0, n_iter=200, alpha=0.7, verbose=False):
    n = len(a0)
    a = norm_sum1(a0).copy()
    L = 1 << (2 * n - 1).bit_length()

    for it in range(n_iter):
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        P = np.abs(A) ** 2

        P_nondc = P[1:]
        P_nondc_sum = P_nondc.sum()
        if P_nondc_sum < 1e-30:
            break

        P_flat = np.ones_like(P)
        P_flat[0] = P[0]
        P_flat[1:] = (P[1:] / P_nondc_sum) ** alpha * P_nondc_sum

        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_flat, 0)) * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

    return a


def coord_descent_polish(a0, time_limit, seed=42, verbose=False):

            idx2 = np.arange(lo2, hi2 + 1)

            n_cand = max(8, n // 40)

                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:

                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()

        if not improved:

                    if M_new < best_m:

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)
    old_vals = np.concatenate([a_norm, [0.0]])
    new_x = np.linspace(0.0, 1.0, new_n + 1)
    new_vals = np.interp(new_x, old_x, old_vals)
    a_interp = new_vals[:new_n]
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss
