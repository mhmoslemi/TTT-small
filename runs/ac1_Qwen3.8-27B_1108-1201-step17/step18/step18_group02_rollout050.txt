def gradient_topk(a, ac, k, tau):
    """Gradient of tau*log(sum of top-k exp(ac/tau)) w.r.t. a."""
    n = len(a)
    # Top-k selection
    idx = np.argpartition(ac, -k)[-k:]
    top_ac = ac[idx]
    
    # Softmax over top-k
    scaled = top_ac / tau
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    
    # Full p vector
    p_full = np.zeros(len(ac))
    p_full[idx] = p
    
    # Gradient: grad[i] = 2 * sum_j p_full[i+j] * a[j]
    # Using FFT

    grad = 2.0 * result[n-1:n-1+n]
