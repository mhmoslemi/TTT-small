def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5

        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)

def flat_spectrum_init(n, seed=7):
    """Initialize with a sequence whose DFT magnitude is close to flat."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    magnitudes = np.ones(fft_size // 2 + 1) * 0.8
    magnitudes[0] = np.sqrt(n)
    magnitudes[1] = 0.0
    for k in range(2, fft_size // 2):
        magnitudes[k] = 0.8 + 0.2 * rng.uniform(-1, 1)
    phases = rng.uniform(-np.pi, np.pi, fft_size // 2 + 1)
    A = magnitudes * np.exp(1j * phases)
    A[fft_size // 2] = magnitudes[fft_size // 2].real
    a = np.fft.irfft(A, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def smooth_soft_subgradient(a, ac, top_k=80, beta=4.0):
    """Compute weighted subgradient of max(ac) using softmax weights on top-k lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def softmax_smooth_gradient(a, tau):
    """Compute gradient of smooth approximation to max(conv(a,a))."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    if ac_max < 1e-30:
        return np.zeros(n)
    shifted = tau * (ac - ac_max)
    exp_ac = np.exp(shifted)
    p = exp_ac / exp_ac.sum()
    # Gradient of sum_k p_k * (Ca)[k] w.r.t. a
    # d/d a[j] of (Ca)[k] = a[k-j] + a[j] (symmetric)
    # So grad[j] = sum_k p_k * (a[k-j] + a[j]) for valid k,j
    # = sum_k p_k * a[k-j] + a[j] * sum_k p_k
    # = cross-correlation + a[j]
    grad = np.zeros(n)
    for j in range(n):
        k_lo = max(0, j - n + 1)
        k_hi = min(2 * n - 1, j + n - 1)
        if k_lo > k_hi:
            continue
        idx = np.arange(k_lo, k_hi + 1)
        a_idx = idx - j
        grad[j] = np.dot(p[k_lo:k_hi + 1], a[a_idx]) + a[j]
    return grad - grad.mean()

        if time.monotonic() - t0 > time_limit:

        y = np.maximum(y, 1e-15)

        else:
            y = y / sy

        ac = fast_autocorr(y)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:

def build_lp_matrix(a, K, n):
    """Build the constraint matrix T for the LP. T[k,j] = a[k-j] if valid."""
    ac = fast_autocorr(a)
    binding_idx = np.argsort(ac)[::-1][:K]
    T_rows = []
    T_cols = []
    T_vals = []
    for k in binding_idx:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            T_rows.append(len(T_rows))
            T_cols.append(j)
            T_vals.append(a[ki - j])
    return binding_idx, T_rows, T_cols, T_vals


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200,
                   top_K_max=None, two_phase=True):
    """Improved LP-based Frank-Wolfe with adaptive K, two-phase strategy, and thorough line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    max_lags = 2 * n - 1

    if top_K_max is None:
        top_K_max = max_lags
    K = min(top_K_init, max_lags)
    iter_count = 0
    phase = 1

    while time.monotonic() - t0 < time_limit:
        if two_phase and iter_count > 30 and phase == 1:
            phase = 2
            K = min(K * 2, top_K_max)

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, max_lags)
        use_all = (K_sel >= max_lags)

        if use_all:
            binding_idx = np.arange(max_lags)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build sparse constraint matrix
        T_rows = []
        T_cols = []
        T_vals = []
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_rows.append(ii)
                T_cols.append(j)
                T_vals.append(a[ki - j])

        if len(T_rows) == 0:
            break

        b_var = cp.Variable(n, nonneg=True)
        T_sparse = cp.csc_matrix((T_vals, (T_rows, T_cols)), shape=(K_sel, n))
        constraints = [T_sparse @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(0), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 25, top_K_max)
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Verify direction actually helps
        conv_check = fast_autocorr(0.5 * a + 0.5 * g_dir)
        top_lags = np.argsort(abs(ac))[::-1][:min(50, max_lags)]
        if conv_check[top_lags].max() > M_curr * 1.05:
            K = min(K + 50, top_K_max)
            continue

        # Thorough line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj

        trial_alphas = [best_alpha, 2.0 / (iter_count + 1.0)]
        for fa in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            if fa not in trial_alphas:
                trial_alphas.append(fa)

        for alpha in trial_alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m - 1e-16:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        # Adaptive K
        if iter_count % 15 == 0:
            K = min(K + 40, top_K_max)

        # Full constraint check every 50 iterations
        if iter_count % 50 == 0 and K < max_lags:
            K_check = min(K * 2, max_lags)
            K = K_check

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}, phase={phase}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=True, tau_init=20.0, tau_final=300.0,
                             eta_init=0.6, rng_seed=100, use_momentum=True):
    """Smoothed projected gradient with momentum and adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    total_iters = 0
    v = np.zeros(n)  # momentum

    while time.monotonic() - t0 < time_limit and stall < 600:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit * 0.8, 1.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        shifted = tau * (ac - ac_max)
        exp_ac = np.exp(shifted)
        p = exp_ac / exp_ac.sum()

        # Compute gradient: grad[j] = sum_k p_k * (a[k-j] + a[j])
        # Using the relation: d(conv(a,a)[k])/d(a[j]) = a[k-j] + a[j] (for valid indices)
        grad = np.zeros(n)
        for j in range(n):
            k_lo = max(0, j - n + 1)
            k_hi = min(2 * n - 1, j + n - 1)
            if k_lo <= k_hi:
                idx = np.arange(k_lo, k_hi + 1)
                a_idx = idx - j
                grad[j] = np.dot(p[k_lo:k_hi + 1], a[a_idx]) + a[j]

        g_tangent = grad - grad.mean()

        if use_momentum:
            v = 0.9 * v + 0.1 * g_tangent
            step = v
        else:
            step = g_tangent

        a_trial = a - eta * step
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.02, 3.0)
        else:
            stall += 1
            a = a_new
            if stall > 80:
                eta *= 0.75
            if stall > 200:
                pert = rng.normal(0, 0.004, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a = a_p / a_p.sum()
                v = np.zeros(n)
                stall = 100
                eta = eta_init * 0.6
            if stall > 400:
                a = best_a.copy()
                v = np.zeros(n)
                stall = 500
                eta = min(eta * 1.5, eta_init * 2)

        total_iters += 1

        if verbose and total_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Targeted coordinate descent focusing on peak autocorrelation lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 80
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-impact destination
            dest_sensitivity = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                idx2 = np.arange(s2, e2 + 1)
                dest_sensitivity[idx2] += a[k2i - idx2]

            candidates = np.where(dest_sensitivity < dest_sensitivity.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            top_cands = candidates[np.argsort(dest_sensitivity[candidates])[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.02, 0.01, 0.05, 0.03, 0.08, 0.1, 0.15, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick AMD multi-start (20% time) ===
    print(f"\n=== Phase 1: Multi-start AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.18, 250.0)

    if t_phase1 > 50:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3.0")
        inits.append(flat_spectrum_init(n0, seed=7))
        labels.append("flat_spec")
        inits.append(spectral_init(n0, n_modes=40, seed=123))
        labels.append("spectral")
        inits.append(make_bimodal(n0, width=0.25))
        labels.append("bimodal")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 40)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(12.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=80, beta=5.0, c0=5.0,
                                     restart_every=4000, verbose=False,
                                     rng_seed=100 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: Main LP-FW (28% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Main LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.28, 500.0)
    if t2 > 50 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                    top_K_init=top_K_init,
                                    top_K_max=top_K_max,
                                    two_phase=True)
        try_update(a_out, "lp_main")

    # === Phase 3: Smooth gradient (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smooth gradient (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t3, verbose=True,
                                             tau_init=15.0, tau_final=250.0,
                                             eta_init=0.7, rng_seed=200,
                                             use_momentum=True)
        try_update(a_out, "smooth1")

    # === Phase 4: LP-FW fine (18% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW fine (n={n_best}) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                    top_K_init=top_K_init,
                                    top_K_max=top_K_max,
                                    two_phase=True)
        try_update(a_out, "lp_fine")

    # === Phase 5: Coordinate polish (8% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate polish (n={n_best}) ===")
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, rng_seed=456)
        try_update(a_out, "coord1")

    # === Phase 6: Length exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    t6_total = min(remaining * 0.10, 150.0)

    trial_lengths = []
    for delta in [-500, -300, 300, 500, 700]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    if trial_lengths and t6_total > 40:
        t_per_len = t6_total / min(len(trial_lengths), 2)
        for new_n in trial_lengths[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            print(f"  Trying n={new_n}...")
            a_long = interpolate_sequence(best_seq, new_n)

            t7a = min(t_per_len * 0.6, remaining * 0.4)
            if t7a > 15:
                a_out, _ = amd_optimize(a_long, t7a, top_k=80, beta=5.0, c0=4.0,
                                         restart_every=5000, verbose=False,
                                         rng_seed=new_n % 1000)
                try_update(a_out, f"amd_n{new_n}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40 and new_n <= 5500:
                t7b = min(t_per_len * 0.4, remaining * 0.5)
                if t7b > 15:
                    a_in2 = norm_sum1(interpolate_sequence(best_seq, new_n))
                    top_K = min(2000, 2 * new_n - 1)
                    a_out, _ = lp_frank_wolfe(a_in2, t7b, solver, verbose=False,
                                               top_K_init=min(300, 2 * new_n - 1),
                                               top_K_max=top_K, two_phase=True)
                    try_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final combined optimization (remainder) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final optimization (n={n_best}, remaining={remaining:.0f}s) ===")

    if remaining > 40:
        t_final = min(remaining * 0.5, 200.0)
        if t_final > 20 and n_best <= 6000:
            a_in = norm_sum1(best_seq)
            top_K_init = min(800, 2 * n_best - 1)
            top_K_max = min(5000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t_final, solver, verbose=True,
                                       top_K_init=top_K_init,
                                       top_K_max=top_K_max, two_phase=True)
            try_update(a_out, "lp_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t_sm = min(remaining * 0.4, 100.0)
        if t_sm > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = smooth_gradient_optimize(a_in, t_sm, verbose=False,
                                                 tau_init=50.0, tau_final=400.0,
                                                 eta_init=0.3, rng_seed=500,
                                                 use_momentum=True)
            try_update(a_out, "smooth_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t_cd = min(remaining * 0.5, 60.0)
        if t_cd > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t_cd, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Ensure all values are valid
        best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
