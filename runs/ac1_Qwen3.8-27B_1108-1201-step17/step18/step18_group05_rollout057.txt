    """

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None):

    best_a = a.copy()

        M_curr = ac.max()

        base_obj = M_curr

    cur_top_k = top_k_init
    cur_beta = beta_init
    restart_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        # Polyak-like step size
        eta = gamma / (np.sqrt(t) + 1.0)

        # Multiplicative update
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive parameter adjustment
        if stall % 500 == 0 and stall > 0:
            if stall % 1000 == 0:
                cur_beta = max(cur_beta * 0.95, 2.5)
                cur_top_k = min(cur_top_k + 10, 200)
            if stall > 5000:
                # Restart with perturbation
                pert = rng.normal(0, 0.02, n) * best_a
                x_r = np.maximum(best_a + pert, 1e-10)
                x = x_r / x_r.sum()
                x_prev = x.copy()
                stall = 0
                restart_count += 1
                cur_top_k = min(cur_top_k + 20, 300)
                cur_beta = max(cur_beta - 0.5, 3.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  Nesterov t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, k={cur_top_k}, beta={cur_beta:.1f}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=True, tau_init=10.0, tau_final=200.0,
                              eta_init=0.5, rng_seed=100):
    """Smoothed projected gradient with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < 800:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = np.correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        a_new = project_simplex(a - eta * g_tangent)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            a = a_new
            stall += 1
            if stall > 100:
                eta *= 0.85
            if stall > 300:
                pert = rng.normal(0, 0.003, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a = a_p / a_p.sum()
                stall = 150

        total_iters += 1
        if verbose and total_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with sensitivity-based move selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    n_top = min(30, 2 * n - 1)

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity from top lags
        sens = np.zeros(n)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destinations
            s_sorted = np.argsort(sens)
            for dst_idx in range(min(5, n)):
                dst = s_sorted[dst_idx]
                if dst == src:
                    continue
                for frac in [0.02, 0.01, 0.05, 0.1, 0.03]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random move
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    break

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with hybrid optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = make_u_shape(n_base, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Spectral shaping (NEW) ===
    print(f"\n=== Phase 1: Spectral shaping (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.15, 200.0)

    if t_phase1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_shaping(a_in, t_phase1, verbose=True, rng_seed=42)
        try_update(a_out, "spectral_main")

        # Try from different initializations
        inits = [
            make_u_shape(n0, sharpness=1.5),
            make_u_shape(n0, sharpness=3.0),
            make_bimodal(n0, width=0.2),
            make_cosine(n0),
        ]
        if parent is not None:
            inits.append(norm_sum1(parent))

        for idx, init in enumerate(inits[:3]):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_avail = min(remaining * 0.15, 100.0)
            if t_avail < 20:
                break
            print(f"  Spectral init {idx}...")
            a_out, _ = spectral_shaping(init, t_avail, verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"spectral_{idx}")

    # === Phase 2: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 40:
        a_in = norm_sum1(best_seq)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(2000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_main")

    # === Phase 3: Nesterov subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Nesterov subgradient (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(a_in, t3, verbose=True,
                                         top_k_init=60, beta_init=6.0,
                                         c0=5.0, rng_seed=77)
        try_update(a_out, "nesterov")

    # === Phase 4: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Smoothed gradient (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t4, verbose=True,
                                             tau_init=10.0, tau_final=150.0,
                                             eta_init=0.5, rng_seed=100)
        try_update(a_out, "smooth")

    # === Phase 5: LP-FW round 2 (on best so far) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_r2")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate polish (n={n_best}) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [500, 1000, -500, 1500, -1000, 2000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.3, 150.0)
        if t7a > 30:
            a_out, _ = nesterov_subgradient(a_long, t7a * 0.5, verbose=False,
                                             top_k_init=60, beta_init=6.0,
                                             c0=5.0, rng_seed=new_n % 1000)
            try_update(a_out, f"nesterov_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 5000:
            t7b = min(remaining * 0.4, 120.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                top_K = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                           top_K_init=min(400, 2 * new_n - 1),
                                           top_K_max=top_K)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final refinement on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final refinement (n={n_best}) ===")
    if remaining > 30:
        # Spectral + LP combo
        t8a = min(remaining * 0.3, 120.0)
        if t8a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = spectral_shaping(a_in, t8a * 0.6, verbose=False, rng_seed=999)
            try_update(a_out, "spectral_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t8b = min(remaining * 0.4, 120.0)
            if t8b > 20:
                a_in = norm_sum1(best_seq)
                top_K_init = min(800, 2 * n_best - 1)
                top_K_max = min(4000, 2 * n_best - 1)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=True,
                                           top_K_init=top_K_init, top_K_max=top_K_max)
                try_update(a_out, "lp_final")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 9: Final polish ===")
        t9 = min(remaining * 0.7, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_u_shape(n_fb, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
