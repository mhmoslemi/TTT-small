def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_centered_power(n, power=0.5):
    x_c = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm_sum1(x_c ** (-power))


def make_mixed(n, pos1=0.2, pos2=0.65, width=0.1, ratio=1.0):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / (width * 1.5)) ** 2)
    return norm_sum1(a)


def make_exponential(n, rate=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm_sum1(np.exp(-rate * x))


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Default init: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    n0 = len(_best_tracker['seq'])
    print(f"Starting: n={n0}, score={_best_tracker['score']:.6f}")

    # === Phase A: Multi-start optimization on n0 (40% of budget) ===
    print(f"\n=== Phase A: Multi-start (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase_a = min(remaining * 0.40, 700.0)

    if t_phase_a > 30:
        rng_master = np.random.default_rng(12345)
        a_parent = norm_sum1(_best_tracker['seq'])

        # Build diverse initializations
        inits = []
        labels = []

        # Parent and smoothed versions
        inits.append(a_parent.copy())
        labels.append("parent")

        # Slightly smoothed parent (Gaussian blur)
        kernel_size = max(3, n0 // 500)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4) ** 2)
        kernel /= kernel.sum()
        a_smooth = np.convolve(a_parent, kernel, mode='same')
        inits.append(norm_sum1(a_smooth))
        labels.append("smoothed")

        # Slightly sharpened parent
        a_sharp = a_parent ** 1.5
        inits.append(norm_sum1(a_sharp))
        labels.append("sharp")

        # Structured perturbations
        for sp in range(3):
            rng_sp = np.random.default_rng(1000 + sp)
            a_pert = a_parent + rng_sp.normal(0, 0.02, n0) * a_parent
            inits.append(norm_sum1(a_pert))
            labels.append(f"pert{sp}")

        # Different parametric forms
        inits.append(make_decreasing(n0, 0.4))
        labels.append("dec04")
        inits.append(make_decreasing(n0, 0.8))
        labels.append("dec08")
        inits.append(make_sinc(n0, 6.0))
        labels.append("sinc6")
        inits.append(make_sinc(n0, 12.0))
        labels.append("sinc12")
        inits.append(make_cosine(n0, 1))
        labels.append("cos1")
        inits.append(make_centered_power(n0, 0.7))
        labels.append("cctr07")
        inits.append(make_mixed(n0, 0.15, 0.6, 0.08))
        labels.append("mixA")
        inits.append(make_exponential(n0, 1.2))
        labels.append("exp12")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = t_phase_a / n_inits
        print(f"  {n_inits} initializations, ~{t_per:.0f}s each")

        for idx in range(n_inits):
            if time.monotonic() - t_global > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            lbl = unique_labels[idx]
            print(f"  Optimizing '{lbl}' ({t_avail:.0f}s)")
            a_out, _ = optimize_phased(unique_inits[idx], t_avail,
                                       seed=2000 + idx, verbose=False)
            try_update(a_out, f"A_{lbl}")

    # === Phase B: Length exploration (25% of budget) ===
    print(f"\n=== Phase B: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase_b = min(remaining * 0.25, 450.0)
    n_best = len(_best_tracker['seq'])

    if t_phase_b > 40:
        candidate_ns = []
        for delta in [-500, -250, 250, 500, 750, 1000]:
            new_n = n_best + delta
            if 2000 <= new_n <= 8000:
                candidate_ns.append(new_n)

        n_per = t_phase_b / max(1, len(candidate_ns))
        for new_n in candidate_ns:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(n_per, remaining - 30)
            if t_avail < 10:
                break
            print(f"  n={new_n} ({t_avail:.0f}s)")
            a_interp = interpolate_seq(_best_tracker['seq'], new_n)
            try_update(a_interp, f"B_intp{new_n}")
            a_out, _ = optimize_phased(a_interp, t_avail, seed=5000 + new_n, verbose=False)
            try_update(a_out, f"B_{new_n}")

    # === Phase C: Ultra-fine polish on best (35% of budget) ===
    print(f"\n=== Phase C: Ultra-fine polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase_c = min(remaining * 0.70, 600.0)
    n_final = len(_best_tracker['seq'])
    print(f"  Final n={n_final}, score={_best_tracker['score']:.6f}")

    if t_phase_c > 20:
        a_in = norm_sum1(_best_tracker['seq'])
        # Run the optimizer with a very long Freeze phase
        # Override phases for final polish
        t_sub1 = t_phase_c * 0.15
        t_sub2 = t_phase_c * 0.35
        t_sub3 = t_phase_c * 0.50

        # Sub-phase C1: moderate
        print(f"  C1 ({t_sub1:.0f}s): moderate cooling")
        x = a_in.copy()
        best_x = x.copy()
        best_m = fast_conv(x).max()
        rng_c = np.random.default_rng(77777)
        log_x = np.log(x + 1e-30)
        c0 = 2.0
        iters = 0
        stall = 0
        pt0 = time.monotonic()
        while time.monotonic() - pt0 < t_sub1:
            iters += 1
            p = min(1.0, iters / 8000.0)
            n_act = int(30 + 40 * p)

            eta = c0 / (np.sqrt(iters) + 1.0)
            conv_x = fast_conv(x)

            improved = False
            e = eta
            for _ in range(15):
                log_new = log_x - e * g
                max_log = log_new.max()
                a_new = np.exp(log_new - max_log)
                a_new = np.maximum(a_new, 1e-30)
                a_new = a_new / a_new.sum()
                M_new = fast_conv(a_new).max()
                if M_new < best_m:
                    x = a_new
                    log_x = np.log(x + 1e-30)
                    if M_new < best_m - 1e-16:
                        best_m = M_new
                        best_x = x.copy()
                        stall = 0
                    improved = True
                    break
                e *= 0.6
            if not improved:
                log_new = log_x - 0.3 * (c0 / (np.sqrt(iters) + 1.0)) * g
                max_log = log_new.max()
                a_new = np.exp(log_new - max_log)
                a_new = np.maximum(a_new, 1e-30)
                a_new = a_new / a_new.sum()
