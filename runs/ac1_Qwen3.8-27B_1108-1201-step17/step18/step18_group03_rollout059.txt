Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth mirror descent with
log-sum-exp approximation, adaptive step sizes, and coordinate descent polish.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_smooth_gradient(a, beta):
    """
    Compute the gradient of the log-sum-exp approximation of max(conv(a,a)).
    Returns (convolution, softmax_weights, gradient).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Compute convolution
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    
    # Compute softmax weights
    c_scaled = beta * c
    c_scaled -= c_scaled.max()  # for numerical stability
    p = np.exp(c_scaled)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(size) / size
    else:
        p = p / p_sum
    
    # Compute cross-correlation: cross[i] = sum_k p[k] * a[k-i]
    N = 1 << (size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    A2 = np.fft.rfft(a, N)
    cross = np.fft.irfft(P * np.conj(A2), N)
    grad = 4.0 * n * cross[:n]
    
    return c, p, grad


def mirror_descent(a0, time_budget, beta_start=5.0, beta_end=80.0, 
                   step_start=0.8, step_end=0.3, verbose=True):
    """
    Smooth mirror descent for minimizing 2n*max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_c = fast_autocorr(a)
    best_M = best_c.max()
    t0 = time.monotonic()
    
    total_iter = max(100, int(time_budget * 2000))  # estimate
    iter_count = 0
    stall_count = 0
    stall_threshold = 2000
    
    for t in range(1, 10000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_budget:
            break
        
        # Adaptive beta
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Adaptive step size
        step = step_start + (step_end - step_start) * progress
        
        # Compute gradient
        c, p, grad = compute_smooth_gradient(a, beta)
        
        # Scale gradient for stability
        gmax = grad.max()
        if gmax < 1e-15:
            break
        
        # Mirror descent update: a_new = a * exp(-lr * grad / gmax)
        lr = step
        log_update = -lr * grad / gmax
        # Clip for numerical stability
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        M_new = fast_autocorr(a_new).max()

            best_M = M_new
            best_a = a_new.copy()
            stall_count = 0
            a = a_new
        else:
            a = a_new  # still update a (momentum-like)
            stall_count += 1
        
        iter_count += 1
        
        if verbose and iter_count % 5000 == 0:
            J = 2.0 * n * best_M
            print(f"    MD iter={iter_count} (t={elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall_count}")

    while time.monotonic() - t0 < time_budget:
        # Try a random move
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or a[i] < 1e-12:
            continue
        
        # Determine direction: move from larger to smaller (or random)
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        
        # Try a small transfer
        frac = float(rng.uniform(0.001, 0.01))
        amt = a[src] * frac
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = a_trial / a_trial.sum()
        
        M_trial = fast_autocorr(a_trial).max()
        if M_trial < best_M - 1e-16:
            a = a_trial
            best_M = M_trial
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            # Try larger moves
            for _ in range(100):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                frac = float(rng.uniform(0.01, 0.1))
                src = i if a[i] > a[j] else j
                dst = j if src == i else i
                if a[src] < 1e-12:
                    continue
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-16:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    stall = 0
                    break
            if stall > max_stall:
                break
    
    J = 2.0 * n * best_M
    if verbose:
        print(f"    CoordDesc done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_M
