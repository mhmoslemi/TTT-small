def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def soft_max_grad(a, tau):
    """
    Compute gradient of soft-max surrogate:
    f(a) = (1/tau) * log(sum_k exp(tau * conv[k]))
    
    Returns (f_value, grad) where grad[j] = df/da[j].
    """
    n = len(a)
    conv = fast_conv(a)
    
    # Scale to avoid overflow
    max_conv = conv.max()
    if max_conv < 1e-30:
        return 0.0, np.zeros(n)
    
    # Stable log-sum-exp
    scaled = tau * (conv - max_conv)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum()) + tau * max_conv
    f_val = log_sum / tau
    
    # Weights w[k] = exp(tau * conv[k]) / sum(exp(tau*conv[k]))
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (where a is zero-extended)
    # This is the cross-correlation of w and a.
    # Using FFT: cross_corr = ifft(FFT(w) * conj(FFT(a)))
    corr_size = len(w) + n - 1  # = 3n-2
    fft_size = 1 << (corr_size - 1).bit_length()
    
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    
    grad = 2.0 * cross
    return f_val, grad


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_local(a):
    """Compute the objective value locally (fast, no function call)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)


def parametric_init(n, rng):
    """Generate diverse parametric initializations. Returns list of sequences."""
    candidates = []
    x = np.linspace(0, 1, n)
    i = np.arange(n, dtype=np.float64)
    
    # Raised cosine
    candidates.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x))))
    # Sine
    candidates.append(normalize(np.sin(np.pi * (x + 0.5/n))))
    # Sine squared
    candidates.append(normalize(np.sin(np.pi * (x + 0.5/n))**2))
    # Cubic
    candidates.append(normalize(np.sin(np.pi * (x + 0.5/n))**4))
    # Gaussian at center
    for sigma in [0.1, 0.15, 0.2, 0.25, 0.3]:
        candidates.append(normalize(np.exp(-((x - 0.5)**2) / (2*sigma**2))))
    # Two-lobed
    for c in [0.25, 0.3, 0.35]:
        for sigma in [0.05, 0.08, 0.1]:
            a = np.exp(-((x-c)**2)/(2*sigma**2)) + np.exp(-((x-(1-c))**2)/(2*sigma**2))
            candidates.append(normalize(a))
    # Polynomial (i+1)^p * (n-i)^q
    for p in [0.5, 1, 2, 3]:
        for q in [0.5, 1, 2, 3]:
            a = (i + 1)**p * (n - i)**q
            candidates.append(normalize(a))
    # Power law
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(normalize((i+1)**p))
        candidates.append(normalize((n-i)**p))
    # Cosine with different frequencies
    for freq in [1, 2, 3, 4]:
        for phase in [0, 0.5]:
            a = np.maximum(1 + np.cos(2*np.pi*freq*x + phase*np.pi), 0.01)
            candidates.append(normalize(a))
    # Asymmetric: ramp + bell
    for rate in [1, 2, 3, 5]:
        a = np.exp(-rate * x) + 0.5 * np.exp(-2*rate*(x-0.5)**2)
        candidates.append(normalize(a))
    
    # Random
    for _ in range(30):
        alpha = rng.uniform(0.5, 3.0)
        candidates.append(normalize(rng.random(n) ** alpha))
    
    return candidates


def search_best_init(n, time_limit, parent_seq=None):
    """Find the best initialization from diverse candidates."""
    t0 = time.monotonic()

def soft_max_anneal(a0, time_limit, tau_start=5.0, tau_end=200.0, 
                    n_anneal_steps=10, lr=0.5, verbose=True):
    """
    Optimize the soft-max surrogate with geometric annealing of tau.
    Uses mirror descent (exponential weights) on the simplex.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_hard = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Could precompute the convolution sizes
    t = 0
    for step in range(n_anneal_steps):
        tau = tau_start * (tau_end / tau_start) ** (step / max(n_anneal_steps - 1, 1))
        
        # Inner loop: optimize at fixed tau
        iters = max(100, int(5000 / (1 + step * 0.5)))
        for it in range(iters):
            t += 1
            if time.monotonic() - t0 > time_limit:
                break
            
            f_val, grad = soft_max_grad(a, tau)
            
            # Step size: decrease over iterations
            eta = lr / (1.0 + 0.1 * it)
            
            # Mirror descent update on simplex
            log_a = np.log(a + 1e-30) - eta * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = a_new / a_new.sum()
            
            a = a_new
            
            # Track best (using hard max)
            conv_max = fast_conv(a).max()
            if conv_max < best_hard - 1e-15:
                best_hard = conv_max
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            hard_val = 2.0 * n * best_hard
            print(f"    tau={tau:.1f} ({elapsed:.0f}s): J_hard={hard_val:.6f}")
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a


def lp_fw(a0, time_limit, solver, init_K=200, max_K=3000, verbose=True):
    """
    LP-based Frank-Wolfe for minimizing max_k (y⊗g)[k] on the simplex.
    """
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_M = fast_conv(y).max()
    t0 = time.monotonic()
    
    # Initialize active set
    ac = fast_conv(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
    
    no_improve = 0
    max_no_improve = 1000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print(f"    LP-FW: converged after {t} iters ({time.monotonic()-t0:.0f}s)")
            break
        
        # Compute current max
        ac_y = fast_conv(y)
        M_curr = ac_y.max()
        
        if M_curr < best_M - 1e-14:
            best_M = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Build constraint matrix for active set
        constraints = sorted(active_set)
        if len(constraints) > max_K:
            # Prune: keep the most binding constraints
            margins = np.array([M_curr - ac_y[k] for k in constraints])
            keep_idx = np.argsort(margins)[:max_K]
            active_set = set(int(constraints[i]) for i in keep_idx)
            constraints = sorted(active_set)
        
        K = len(constraints)
        if K == 0:
            break
        
        # Build constraint matrix efficiently
        # coeffs[jj, i] = y[k_jj - i] if 0 <= k_jj - i < n else 0
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraints):
            start = max(0, k - n + 1)
            end = min(n, k + 1)
            if start < end:
                coeffs[jj, start:end] = y[k - (start:end)]
        
        # Solve LP: min M s.t. coeffs @ g <= M, sum(g) = 1, g >= 0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints_cp = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints_cp)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try with fewer constraints
            if len(active_set) > init_K:
                top = sorted(active_set)[:init_K]
                active_set = set(top)
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for violated constraints and add to active set
        # conv(y, g_dir)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_yg > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_curr
            # Add top violators
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                active_set.add(int(k))
        
        # Step size: use 2/(t+2) by default, try a few alternatives
        alpha_default = 2.0 / (t + 2.0)
        alphas = [alpha_default, alpha_default * 0.5, alpha_default * 2.0,
                  min(1.0, 1.0/(t+1)), min(1.0, 2.0/(t+1))]
        alphas = [max(0.005, min(1.0, al)) for al in alphas]
        
        best_alpha = alpha_default
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t <= 1:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(active_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    if verbose:

def kkt_polish(a0, time_limit, verbose=True):
    """
    KKT-based active set Newton method.
    At optimum: for active constraints S and positive coordinates P:
    2 * sum_{k in S} lambda_k * d(conv[k])/da[j] + mu = 0 for j in P
    sum(lambda) = 1
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    
    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv(a)
        M = conv.max()
        
        # Identify active constraints (conv[k] close to M)
        threshold = M * 0.9999
        active = np.where(conv >= threshold)[0]
        if len(active) < 2:
            break
        
        # Identify positive coordinates
        pos_mask = a > 1e-10
        pos_idx = np.where(pos_mask)[0]
        p = len(pos_idx)
        K = len(active)
        
        if p < 2:
            break
            
        # Build the KKT system
        # For each j in pos_idx: 2 * sum_{k in active} lambda_k * d(conv[k])/da[j] + mu = 0
        # Plus: sum(lambda) = 1
        # Total: p+1 equations, K+1 unknowns (lambda_0,...,lambda_{K-1}, mu)
        
        # We need K = p for a square system (after adding sum(lambda)=1)
        # If K > p, we need to select the most active K_p constraints
        # If K < p, we need to add more constraints
        
        # For simplicity, take the top p active constraints
        if K > p:
            active_ordered = active[np.argsort(conv[active])[::-1]][:p]
            K = p
        elif K < p:
            # Not enough active constraints; use all of them and make system underdetermined
            # In this case, set lambda = 1/K for all active, and solve for mu
            mu_val = -2.0 * (1.0/K) * np.mean([
                2.0 * sum(a[max(0,k-j):min(n,k+1)] * a[...] )  # This is getting complicated
                for j in pos_idx[:1]
            ])
            # Skip KKT for now, just do a small coordinate update
            break
        
        # Now K = p, build the system
        # A_system @ [lambda_0, ..., lambda_{p-1}, mu] = b
        # Row j (for j in pos_idx): 2 * sum_k lambda_k * d(conv[k])/da[j] + mu = 0
        # Last row: sum(lambda) = 1
        
        A_sys = np.zeros((p + 1, p + 1))
        b_sys = np.zeros(p + 1)
        b_sys[p] = 1.0  # sum(lambda) = 1
        
        for ii, j in enumerate(pos_idx):
            for kk, k in enumerate(active_ordered):
                # d(conv[k])/da[j] = 2*a[k-j] if 0<=k-j<n else 0
                diff = k - j
                if 0 <= diff < n:
                    A_sys[ii, kk] = 4.0 * a[diff]  # 2 * (derivative) = 2 * 2 * a[k-j]
                # Also check j-k
                diff2 = j - k
                if 0 <= diff2 < n:
                    A_sys[ii, kk] += 4.0 * a[diff2]
            A_sys[ii, p] = 1.0  # coefficient of mu
            b_sys[ii] = 0.0
        
        try:
            sol = np.linalg.solve(A_sys, b_sys)
        except np.linalg.LinAlgError:
            break
        
        lambdas = sol[:p]
        mu = sol[p]
        
        # Check if lambdas are all non-negative
        if np.min(lambdas) < -1e-6:
            # KKT not satisfied; try with a subset
            # Remove the most negative lambda and its constraint
            worst = np.argmin(lambdas)
            active = np.delete(active, worst)
            continue
        
        # Compute the implied M from the KKT conditions
        # M = -mu / 2 (from the stationarity condition at boundary points)
        # Actually, at the optimum, M is determined by the constraints
        # Let's just verify: M should equal conv[k] for active k
        M_check = conv[active_ordered].mean()
        
        # The KKT solution tells us the optimal multipliers
        # To get a new point, we can use the fact that at the optimum
        # the gradient of the Lagrangian is zero
        # For a practical update, we can use the LP solution
        
        # Actually, the KKT system tells us the structure but not directly
        # how to update a. Let me use a different approach:
        # Solve the equality-constrained problem where we force
        # conv[k] = M for all k in active set, sum(a) = 1, a >= 0
        # This is a non-convex system of equations
        
        # Simpler approach: just check convergence
        if verbose and outer % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * M
            print(f"    KKT iter={outer} ({elapsed:.0f}s): J={J:.6f}, K={K}")
        
        # Check convergence: if M hasn't changed much, stop
        if M < best_M - 1e-10:
            best_M = M
            best_a = a.copy()
        elif outer > 10 and abs(M - best_M) < 1e-12:
            break
    
    return best_a


def coordinate_polish(a0, time_limit, rng_seed=123, verbose=True):
    """
    Coordinate descent: for each coordinate, try moving mass to reduce the max convolution.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()
        
        # Find the most contributing pairs to the peak
        peak_k = np.argmax(conv)
        
        # For the peak, find which (i, k-i) pairs contribute most
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        if contrib.max() < 1e-15:
            stall += 1
            continue
        
        # Try to reduce the largest contributor
        src_idx = idx[np.argmax(contrib)]
        src_val = a[src_idx]
        
        if src_val < 1e-12:
            stall += 1
