        b = a

    A = np.fft.rfft(a, fft_size)

def interp_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    rs = result.sum()
    if rs < 1e-15:

def get_subgradient(a, ac, rel_tol=1e-7):
    n = len(a)

        je = min(n, k + 1)
        if js < je:
            js_arr = np.arange(js, je)
            g[js_arr] += 2.0 * a[np.abs(k - js_arr)]
    g /= len(active)
    return g, M, len(active)
