"""
Minimizes  J = 2*n*max(conv(a,a)) / (sum a)**2  for non-negative a.

KEY OBSERVATION:
  np.convolve(a,a,'full') is the *convolution*, NOT the autocorrelation.
  For a non-palindromic (e.g. increasing) sequence the peak is shifted toward
  the large end, and a *linear* ramp a[i]=i+1 gives J -> 4/3 ~= 1.333 for large n,
  far below the 1.50 target.  The previous state-of-the-art (1.504) only tried
  mild (sqrt) ramps, leaving a very large headroom on the table.

STRATEGY:
  A) Fast parametric sweep (power-law / affine / exponential ramps, several n)
     to obtain a strong, near-linear starting point (~1.33-1.35).
  B) Log-space (mirror) descent with a softmax-weighted subgradient of the
     convolution peak and a backtracking line search (keeps only strict
     improvements), with periodic re-blending to a fresh ramp to escape minima.
  C) Peak-targeted coordinate mass moves for fine local polish.
  D) Re-optimise at a couple of neighbouring lengths.
  The best sequence seen (validated against evaluate_sequence) is always kept,
  so the output is at least the best parametric ramp.
"""

import numpy as np
import time


# --------------------------------------------------------------------------- #
#  Core fast primitives
# --------------------------------------------------------------------------- #
def fast_conv(a):
    """Full linear convolution a*a via FFT.  Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def fast_J(a):
    """Exactly 2*n*max(conv)/(sum a)^2, matching evaluate_sequence logic."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.size
    m = fast_conv(a).max()
    return float(2.0 * n * m / (s * s))


def norm1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full(a.shape, 1.0 / a.size)
    return a / s


def eval_safe(a):
    try:
        lst = [float(x) for x in np.asarray(a, dtype=np.float64)]
        r = evaluate_sequence(lst)
        return float(r)
    except Exception:
        return float('inf')


# --------------------------------------------------------------------------- #
#  Parametric families
# --------------------------------------------------------------------------- #
def ramp_power(n, p):
    x = np.arange(1, n + 1, dtype=np.float64)
    return norm1(x ** p)


def ramp_affine(n, c, p=1.0):
    x = np.linspace(0.0, 1.0, n)
    return norm1(1.0 + c * (x ** p))


def ramp_exp(n, r):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(r * x)
    return norm1(a)


def interp_len(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    old = a.size
    if old == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-300:
        return np.full(new_n, 1.0 / new_n)
    an = a / s
    ox = np.linspace(0.0, 1.0, old)
    nx = np.linspace(0.0, 1.0, new_n)
    out = np.interp(nx, ox, an)
    return norm1(out)

    for p in np.arange(0.0, 2.4, 0.1):
        cands.append(ramp_power(n, p))
    for c in [0.3, 0.6, 1.0, 1.5, 2.0, 3.0, 5.0]:
        for p in [0.6, 1.0, 1.5, 2.0]:
            cands.append(ramp_affine(n, c, p))
    for r in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        cands.append(ramp_exp(n, r))
    for a in cands:
        if time.monotonic() > deadline:
            break
        j = fast_J(a)
        if j < best_j:
            best_j = j
            best_a = a.copy()
    return best_a, best_j


# --------------------------------------------------------------------------- #
#  Mirror descent with softmax subgradient + line search
# --------------------------------------------------------------------------- #
def mirror_gradient(a, ac, npeaks, temp):
    n = a.size
    top = np.argsort(ac)[::-1][:npeaks]
    ac_top = ac[top]
    m = ac_top.max()
    w = np.exp((ac_top - m) / max(temp, 1e-6))
    w = w / w.sum()
    g = np.zeros(n)
    for kk, ww in zip(top, w):
        ki = int(kk)
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        if lo > hi:
            continue
        j = np.arange(lo, hi + 1)
        pair = ki - j
        g[j] += ww * (2.0 * a[pair])
    return g


def mirror_refine(a0, start_t, deadline, rng,
                  npeaks=90, temp=2.0, eta0=1.5, verbose=False):
    a = norm1(a0)
    best_a = a.copy()
    ac = fast_conv(a)
    best_j = ac.max()          # sum==1 so J == 2*n*max ; scale-free compare
    t0 = time.monotonic()
    it = 0
    stall = 0
    while time.monotonic() < deadline:
        it += 1
        ac = fast_conv(a)
        cur_m = ac.max()
        g = mirror_gradient(a, ac, npeaks, temp)
        # backtracking line search over log-space step
        base = eta0 / (1.0 + 0.0004 * it)
        accepted = False
        la = np.log(np.maximum(a, 1e-300))
        for _ in range(6):
            la = np.log(np.maximum(a, 1e-300)) - base * g
            an = np.exp(la - la.max())
            an = np.maximum(an, 0.0)
            s = an.sum()
            if s < 1e-300:
                break
            an = an / s
            m_new = fast_conv(an).max()
            if m_new < cur_m - 1e-14:
                a = an
                if m_new < best_j:
                    best_j = m_new
                    best_a = an.copy()
                    stall = 0
                cur_m = m_new
                accepted = True
                break
            base *= 0.5
        if not accepted:
            stall += 1
            if stall % 40 == 0:
                # nudge with a fresh ramp of the same length to escape
                strength = 0.25
                rr = ramp_power(a.size, rng.uniform(0.6, 1.6))
                a = norm1((1.0 - strength) * best_a + strength * rr)
                stall = 0
                temp = max(0.8, temp - 0.1)
                npeaks = int(min(160, npeaks + 5))
        if verbose and it % 2000 == 0:
            J = 2.0 * a.size * best_j
            print(f"      mirror it={it} J={J:.6f} peaks={npeaks} "
                  f"temp={temp:.2f} stall={stall} elapsed={time.monotonic()-t0:.0f}s")
    return best_a, 2.0 * a.size * best_j


# --------------------------------------------------------------------------- #
#  Peak-targeted coordinate moves (local polish)
# --------------------------------------------------------------------------- #
def peak_polish(a0, start_t, deadline, rng, verbose=False):
    a = norm1(a0)
    best_a = a.copy()
    ac = fast_conv(a)
    best_j = ac.max()
    n = a.size
    t0 = time.monotonic()
    stall = 0
    while time.monotonic() < deadline and stall < 400:
        ac = fast_conv(a)
        M = ac.max()
        top = np.argsort(ac)[::-1][:12]
        # involvement cost map
        cost = np.zeros(n)
        for kk in top:
            ki = int(kk)
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            j = np.arange(lo, hi + 1)
            cost[j] += a[ki - j]
        improved = False
        for kk in top[:6]:
            if time.monotonic() > deadline:
                break
            ki = int(kk)
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            if lo > hi:
                continue
            j = np.arange(lo, hi + 1)
            contrib = a[j] * a[ki - j]
            src = j[int(np.argmax(contrib))]
            if a[src] < 1e-8:
                continue
            # destination: low value and low involvement
            cand = np.argsort(cost + 20.0 * a)[:max(1, n // 4)]
            dst = cand[int(np.argmin(a[cand]))]
            if dst == src:
                continue
            for frac in (0.05, 0.15, 0.3, 0.5):
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                at = a.copy()
                at[src] -= amt
                at[dst] += amt
                at = np.maximum(at, 0.0)
                s = at.sum()
                if s < 0.01:
                    continue
                at = at / s
                mnew = fast_conv(at).max()
                if mnew < best_j - 1e-15:
                    a = at
                    best_j = mnew
                    best_a = at.copy()
                    stall = 0
                    improved = True
                    break
        if not improved:
            # random perturbation to escape
            i = int(rng.integers(n)); jj = int(rng.integers(n))
            if i != jj and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.01, 0.06)
                a[i] -= amt; a[jj] += amt
                a = np.maximum(a, 0.0); a = a / a.sum()
                mnew = fast_conv(a).max()
                if mnew < best_j:
                    best_j = mnew; best_a = a.copy(); stall = 0
            else:
                stall += 1
        if verbose and stall % 50 == 0:
            print(f"      polish stall={stall} J={2.0*n*best_j:.6f} "
                  f"elapsed={time.monotonic()-t0:.0f}s")
    return best_a, 2.0 * n * best_j


# --------------------------------------------------------------------------- #
#  Top-level search
# --------------------------------------------------------------------------- #
def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    deadline = t_global + min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(12345)

    best_a = None
    best_score = float('inf')

    def nothing():  # (placeholder to satisfy readability; not used)
        pass

    # ---- load parent if available -------------------------------------- #
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-8:
            parent = p.copy()
    except Exception:
        parent = None

    def consider(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = norm1(np.asarray(a, dtype=np.float64))
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  *** NEW BEST [{label}] J={sc:.6f} n={a.size} "
                  f"t={time.monotonic()-t_global:.0f}s")

    print(f"budget={budget_s:.0f}s parent={'yes' if parent is not None else 'no'}")

    # ---- Phase A: parametric sweep at a few lengths --------------------- #
    print("== Phase A: parametric sweep ==")
    n_base = 4200 if (parent is None or parent.size <= 0) else int(parent.size)
    n_options = sorted({n_base, 3600, 4800, 5400})
    if parent is not None:
        consider(parent, "parent")

    for n in n_options:
        if time.monotonic() > deadline - 300:
            break
        sub_dead = min(deadline - 240, time.monotonic() + 45.0)
        a_s, j_s = sweep_params(n, sub_dead)
        if a_s is not None:
            consider(a_s, f"sweep_n{n}")

    if best_a is None:
        best_a = ramp_power(4200, 1.0)
        best_score = eval_safe(best_a)
    n0 = best_a.size
    print(f"  after sweep best J={best_score:.6f} n={n0}")

    # ---- Phase B: mirror refinement (main workhorse) -------------------- #
    print("== Phase B: mirror descent ==")
    remaining = deadline - time.monotonic()
    configs = [
        (90, 2.5, 1.5),
        (70, 1.5, 2.0),
        (120, 1.2, 1.8),
        (150, 1.0, 2.2),
        (100, 1.8, 1.2),
    ]
    for ci, (pk, tp, et) in enumerate(configs):
        if time.monotonic() > deadline - 120:
            break
        sub_dead = min(deadline - 120, time.monotonic() + remaining * 0.28)
        if sub_dead - time.monotonic() < 30:
            break
        a_in = norm1(best_a)
        a_out, _ = mirror_refine(a_in, time.monotonic(), sub_dead, rng,
                                 npeaks=pk, temp=tp, eta0=et, verbose=False)
        consider(a_out, f"mirror{ci}_n{a_in.size}")
        if time.monotonic() > deadline - 90:
            break

    # ---- Phase C: peak polish on the very best -------------------------- #
    print("== Phase C: peak polish ==")
    if time.monotonic() < deadline - 80:
        sub_dead = min(deadline - 80, time.monotonic() + 150.0)
        a_out, _ = peak_polish(norm1(best_a), time.monotonic(), sub_dead,
                               rng, verbose=False)
        consider(a_out, f"polish_n{best_a.size}")

    # ---- Phase D: refine length with a short mirror pass ---------------- #
    print("== Phase D: neighbouring length ==")
    if time.monotonic() < deadline - 120:
        for nn in [n0 + 600, n0 - 600, 6000]:
            if time.monotonic() > deadline - 120:
                break
            if nn < 1500 or nn > 12000:
                continue
            a_l = interp_len(best_a, nn)
            consider(a_l, f"interp_n{nn}")
            sub_dead = min(deadline - 120, time.monotonic() + 70.0)
            if sub_dead - time.monotonic() < 20:
                break
            a_out, _ = mirror_refine(a_l, time.monotonic(), sub_dead, rng,
                                     npeaks=100, temp=1.5, eta0=1.8, verbose=False)
            consider(a_out, f"mirror_n{nn}")

    # ---- Phase E: final mirror + polish on absolute best ---------------- #
    print("== Phase E: final.push ==")
    if time.monotonic() < deadline - 30:
        sub_dead = min(deadline - 30, time.monotonic() + 90.0)
        a_out, _ = mirror_refine(norm1(best_a), time.monotonic(), sub_dead, rng,
                                 npeaks=140, temp=1.0, eta0=2.0, verbose=False)
        consider(a_out, "final_mirror")
    if time.monotonic() < deadline - 20 and best_a is not None:
        sub_dead = min(deadline - 20, time.monotonic() + 40.0)
        a_out, _ = peak_polish(norm1(best_a), time.monotonic(), sub_dead, rng,
                               verbose=False)
        consider(a_out, "final_polish")

    # ---- final validation ---------------------------------------------- #
    if best_a is None:
        best_a = ramp_power(4200, 1.0)
    best_a = norm1(best_a)
    fs = eval_safe(best_a)
    if np.isfinite(fs):
        best_score = fs
    if not np.isfinite(best_score):
        best_a = ramp_power(4200, 1.0)
        best_score = eval_safe(best_a)
    print(f"FINAL J={best_score:.6f} n={best_a.size} "
          f"time={time.monotonic()-t_global:.0f}s")
    return [float(x) for x in best_a]
