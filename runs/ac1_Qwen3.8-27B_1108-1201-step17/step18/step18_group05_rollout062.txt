def norm_sum1(a):

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    x = np.linspace(0, 1, n)
    a = 1.0 + sharpness * ((2*x - 1)**2 - 0.25)**2 * 4
    # Actually let's do: peaks at x=0, x=0.5, x=1
    a = np.exp(-((x - 0)/0.15)**2) + np.exp(-((x - 0.5)/0.1)**2) + np.exp(-((x - 1)/0.15)**2)
    return norm_sum1(a)


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=300, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                K = min(K + 100, top_K_max)
                stall = 0

        iter_count += 1

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=False, tau_init=20.0, tau_final=300.0,
                              eta_init=0.5, rng_seed=100):
    """Smoothed projected gradient with adaptive temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 600.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross

        g_tangent = g - g.mean()

        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            if stall > 50:
                eta *= 0.8
            if stall > 200:
                pert = rng.normal(0, 0.005, n) * a
                a_pert = np.maximum(a + pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                    stall = 0
                    eta = eta_init * 0.5
                else:
                    a = a_new
            else:
                a = a_new

        if verbose and (stall % 500 == 0 or stall == 0 and elapsed > 10):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m

1. LP-based Frank-Wolfe with adaptive K and CLARABLE solver

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABLE, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                print(f"  LP solver: {s}")
                return s
        except Exception:

    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=400, top_K_max=None):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 100, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2, 0.1]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                K = min(K + 200, top_K_max)
                stall = 0

        iter_count += 1

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=False, tau_init=20.0, tau_final=300.0,
                              eta_init=0.5, rng_seed=100):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    eta = eta_init
    stall = 0

    while time.monotonic() - t0 < time_limit:

        progress = min(1.0, elapsed / min(time_limit, 600.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)
        m_new = fast_autocorr(a_new).max()

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            if stall > 50:
                eta *= 0.8
            if stall > 200:
                pert = rng.normal(0, 0.005, n) * a
                a_pert = np.maximum(a + pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                    stall = 0
                    eta = eta_init * 0.5
                else:
                    a = a_new
            else:
                a = a_new

        if verbose and (stall % 500 == 0 or (stall == 0 and elapsed > 10)):
            elapsed2 = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad ({elapsed2:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

def sa_optimize(a0, time_limit, verbose=False, rng_seed=42, T0=5e-6, Tf=1e-8):
    """
    Simulated annealing with mass-transfer moves.
    Uses biased source selection (from peak-contributing positions).
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    current_m = best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Precompute peak info
    ac = fast_autocorr(a)
    peak_k = int(np.argmax(ac))

    it = 0
    accepted = 0
    improved = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        frac_time = (time.monotonic() - t0) / time_limit
        T = T0 * (Tf / T0) ** frac_time

        # Periodically refresh peak info
        if it % 5000 == 0:
            ac = fast_autocorr(a)
            peak_k = int(np.argmax(ac))
            current_m = ac.max()

        # Choose source: biased towards positions contributing to the peak
        if rng.random() < 0.6:
            s = max(0, peak_k - n + 1)
            e = min(n - 1, peak_k)
            src_range = np.arange(s, e + 1)
            weights = a[src_range] * a[peak_k - src_range]
            total_w = weights.sum()
            if total_w > 1e-15:
                probs = weights / total_w
                src = int(src_range[rng.choice(len(src_range), p=probs)])
            else:
                src = int(rng.integers(n))
        else:
            src = int(rng.integers(n))

        # Choose destination: biased towards low-contribution positions
        if rng.random() < 0.4:
            # Far from peak
            dst = int(rng.integers(n))
            attempts = 0
            while dst == src and attempts < 5:
                dst = int(rng.integers(n))
                attempts += 1
        else:
            # From the "quiet" middle region
            mid_start = n // 5
            mid_end = 4 * n // 5
            dst = int(rng.integers(mid_start, mid_end))
            if dst == src:
                dst = (dst + 1) % n

        if src == dst or a[src] < 1e-12:
            continue

        # Random step size
        frac = rng.uniform(0.002, 0.08)
        eps = a[src] * frac

        # Compute trial
        a_trial = a.copy()
        a_trial[src] -= eps
        a_trial[dst] += eps
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        a_trial = a_trial / s_t

        # Evaluate
        m_trial = fast_autocorr(a_trial).max()

        # SA accept/reject
        delta = m_trial - current_m
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            current_m = m_trial
            accepted += 1
            if m_trial < best_m - 1e-16:
                best_m = m_trial
                best_a = a.copy()
                improved += 1
                # Update peak
                peak_k = int(np.argmax(fast_autocorr(a)))

        if verbose and it % 100000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA it={it} ({elapsed:.0f}s): J={J:.6f}, T={T:.2e}, acc={accepted}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123, n_top=30):

def amd_optimize(a0, time_limit, top_k=60, beta=6.0, c0=5.0, restart_every=4000,

        top_idx = np.argsort(ac)[::-1][:k_max]

            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        return grad

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k

            break

        y = x + beta_t * (x - x_prev)

        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_grad(y, ac)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 1.0, 2.0)

    return best_a, best_m
