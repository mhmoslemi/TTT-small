def fast_autocorr(a):
    """Compute full convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    a = np.maximum(a, 0.0)
    s = a.sum()

This is a banded matrix (each row has at most n non-zeros, but typically much fewer since k-i must be in [0, n-1]). For constraint k, the non-zero entries are at positions i where `max(0, k-n+1) <= i <= min(n-1, k)`. The number of non-zeros per row is `min(n, k+1, 2n-1-k)`.

We can use sparse matrices in cvxpy to speed this up.

Let me use `cp.Sparse Git` matrices or just build dense matrices efficiently with numpy.

Actually, for the LP, the constraint matrix has a special structure: it's a product of a Toeplitz-like matrix and a diagonal. We could exploit this, but cvxpy might not let us easily.

Let me just use dense matrices and hope that the LP solver is fast enough. For n=5200 and K=4000, the matrix is about 20.8M floats = ~166MB. That's on the edge of what's practical. Let me use K=2000 to be safe.

Actually, let me reconsider. The LP is:
min M
s.t. sum(a) = 1
     a >= 0
     sum_i a_i * y[k-i] <= M for k in S

This has n+1 variables (a_0, ..., a_{n-1}, M) and |S| + 1 + n constraints (the |S| conv constraints, the equality, and the n non-negativity constraints).

For n=5200 and |S|=2000: 5201 variables, 7201 constraints. The constraint matrix is 7201 x 5201, about 37M elements. That's big but manageable.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Direct cutting-plane LP with warm starts and adaptive constraint management
2. FISTA-accelerated Frank-Wolfe for the non-convex outer problem
3. Multi-length exploration with interpolation
4. Gradient-based refinement with log-sum-exp surrogate
5. Final coordinate descent polish
"""

import numpy as np
import time
import cvxpy as cp
import math

SOLVER = None

def init_solver():
    global SOLVER
    if SOLVER is not None:
        return SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                SOLVER = s
                print(f"  Using solver: {s}")
                return SOLVER
        except:
            continue
    SOLVER = cp.ECOS
    print("  Fallback solver: ECOS")
    return SOLVER


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm1(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def objective(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * len(a) * fast_conv(a).max() / (s * s)


def do_eval(a):
    try:
        lst = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(lst))
    except:
        return float('inf')


def build_coeff_matrix(y, constraint_indices, n):
    """Build the K x n coefficient matrix for the LP.
    Row k, col i: y[k-i] if 0 <= k-i < n, else 0.
    """
    K = len(constraint_indices)
    mat = np.zeros((K, n))
    for jj in range(K):
        k = constraint_indices[jj]
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            mat[jj, i_start:i_end] = y[k - idx]
    return mat


def lp_solve(y, constraint_indices, solver):
    """Solve the LP: min M s.t. sum(a)=1, a>=0, C(a,y)[k] <= M for k in constraint_indices."""
    n = len(y)
    K = len(constraint_indices)
    
    if K == 0:
        return np.ones(n) / n, 1.0 / n
    
    C = build_coeff_matrix(y, constraint_indices, n)
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [
        cp.sum(a_var) == 1,
        C @ a_var <= M_var
    ]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    a_val = np.zeros(n)
    if a_var.value is not None:
        a_val = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_val.sum()
        if s > 1e-15:
            a_val = a_val / s
        else:
            a_val = np.ones(n) / n
    else:
        a_val = np.ones(n) / n
    
    M_val = float(M_var.value) if M_var.value is not None else fast_conv(y).max()
    return a_val, M_val


def fpw_optimize(a0, time_limit, solver, initial_K=300, max_K=3000, verbose=True):
    """FISTA-accelerated Frank-Wolfe with cutting-plane LP oracle."""
    n = len(a0)
    a = norm1(a0.copy())
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_obj = 2.0 * n * best_M
    t0 = time.monotonic()
    
    # Acceleration variables
    y = a.copy()  # current point
    x = a.copy()  # extrapolated point
    z = a.copy()  # previous y
    
    # Initial constraint set: top-K convolution values
    ac = fast_conv(a)
    init_idx = np.argsort(ac)[::-1][:initial_K]
    S = set(int(k) for k in init_idx)
    
    stt = 1.0  # FISTA step parameter
    iter_count = 0
    stalls = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Evaluate at y
        ac_y = fast_conv(y)
        M_y = ac_y.max()
        if M_y < best_M - 1e-15:
            best_M = M_y
            best_a = y.copy()
            best_obj = 2.0 * n * M_y
            stalls = 0
        else:
            stalls += 1
        
        # Extrapolation (FISTA step)
        if t > 1:
            alpha = (1.0 - stt) / (1.0 + stt)
            x = y + alpha * (y - z)
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                x = y.copy()
            else:
                x = x / sx
        else:
            x = y.copy()
        
        # Solve LP using x as the fixed point
        # Constraint set management
        C_list = sorted(S)
        if len(C_list) > max_K:
            margins = np.array([M_y - ac_y[k] for k in C_list])
            keep = np.argsort(margins)[:max_K]
            S = set(int(C_list[i]) for i in keep)
            C_list = sorted(S)
        
        a_lp, M_lp = lp_solve(x, C_list, solver)
        
        if a_lp is None:
            # LP failed, try with smaller constraint set
            C_list = C_list[:min(initial_K, len(C_list))]
            a_lp, M_lp = lp_solve(x, C_list, solver)
            if a_lp is None:
                a_lp = x.copy()
        
        # Update FISTA step
        stt_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * stt * stt))
        stt = stt_new
        
        # Update y (weighted average)
        w = 2.0 / (stt_new + 1.0)
        y_new = (1.0 - w) * y + w * a_lp
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        z = y
        y = y_new
        iter_count += 1
        
        # Add violated constraints
        ac_x = fast_conv(x)
        violations = np.where(ac_x > M_y + 1e-10)[0]
        if len(violations) > 0:
            # Add top violations
            viol_amt = ac_x[violations] - M_y
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                S.add(int(k))
        
        # Also add constraints that are "close" to being active
        close = np.where((ac_x > 0.95 * M_y) & (ac_x <= M_y + 1e-10))[0]
        if len(close) > 0:
            for k in close[:20]:
                S.add(int(k))
        
        # Restart if stalled
        if stalls > 500:
            stalls = 0
            stt = 1.0
            perturb = 0.05 * np.random.default_rng(iter_count).normal(0, 1, n) * best_a
            y = norm1(best_a + perturb)
            x = y.copy()
            z = y.copy()
            S = set(int(k) for k in np.argsort(fast_conv(y))[::-1][:initial_K])
            if verbose:
                print(f"    FPW restart at iter {iter_count}, obj={2*n*best_M:.6f}")
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW iter={iter_count} ({elapsed:.0f}s): obj={2*n*best_M:.6f}, |S|={len(S)}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    FPW done ({elapsed:.0f}s): obj={2*n*best_M:.6f}, iters={iter_count}")
    return best_a, best_M


def grad_refine(a0, time_limit, verbose=True):
    """Gradient refinement using log-sum-exp surrogate."""
    n = len(a0)
    a = norm1(a0.copy())
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    
    tau = best_M * 0.3
    tau_min = best_M * 0.01
    tau_decay = 0.9995
    lr = 0.5
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_conv(a)
        M = ac.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        
        # Log-sum-exp gradient
        ac_s = ac / tau
        ac_s -= ac_s.max()
        p = np.exp(ac_s)
        p /= p.sum()
        
        # grad[i] = 2 * sum_j p[i+j] * a[j] ... 
        # Actually grad of LSE wrt a[i] = 2 * sum_k p[k] * (d(conv(a))[k]/da[i])
        # d(conv(a,a)[k]/da[i]) = a[k-i] + a[i-k] (symmetric)
        # So grad[i] = 2 * sum_k p[k] * (a[k-i] + a[max(0, i-k)]... )
        # Using FFT: grad = 2 * (flip(a) * p) centered
        a_flip = a[::-1]
        size = 2 * n - 1
        fft_sz = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a_flip, fft_sz)
        Pf = np.fft.rfft(p, fft_sz)
        conv_full = np.fft.irfft(Af * Pf, fft_sz)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0)
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mc = m / (1 - b1 ** t)
        vc = v / (1 - b2 ** t)
        
        step = lr / (np.sqrt(vc) + 1e-8)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - step * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a = a_new / a_new.sum()
        
        if stall > 2000:
            stall = 0
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.02, n) * best_a
            a = norm1(best_a + pert)
            m = np.zeros(n)
            v = np.zeros(n)
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    GradRefine done ({elapsed:.0f}s): obj={2*n*best_M:.6f}")
    return best_a, best_M


def coord_polish(a0, time_limit, verbose=False):
    """Coordinate descent polish with targeted transfers."""
    n = len(a0)
    a = norm1(a0.copy())
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_conv(a)
        M = ac.max()
        
        # Find the peak location
        peak_k = int(np.argmax(ac))
        
        # Find contributing pairs for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        if contrib.max() < 1e-20:
            stall += 10
            break
        
        # Find the source (largest contributor)
        src_local = np.argmax(contrib)
        src = idx[src_local]
        src_val = a[src]
        
        if src_val < 1e-14:
            stall += 1
            continue
        
        # Find destination: minimize impact on top peaks
        top_peaks = np.argsort(ac)[::-1][:10]
        impact = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            if s2 <= e2:
                idx2 = np.arange(s2, e2 + 1)
                # Impact of moving from src to j on peak k2:
                # Old contribution to k2 from src: a[src]*a[k2-src] (if valid)
                # New contribution: a[j]*a[k2-j]
                # Change: a[j]*a[k2-j] - 0 (we're removing src's contribution)
                # Actually it's more complex. Let's just compute the "exposure" of each position.
                impact[idx2] += a[k2i - idx2]
                impact[k2i - idx2] += a[idx2]
        
        # Pick the position with lowest impact
        candidates = np.argsort(impact)[:n // 2]  # consider less-exposed positions
        dst = int(candidates[np.argmin(a[candidates])])  # among low-impact, pick smallest value
        
        if dst == src:
            stall += 1
            continue
        
        # Try transfer
        improved = False
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            amt = src_val * frac
            if amt < 1e-16:
                continue
            trial = a.copy()
            trial[src] -= amt
            trial[dst] += amt
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_new = fast_conv(trial).max()
            if M_new < best_M - 1e-16:
                a = trial
                best_M = M_new
                best_a = a.copy()
                stall = 0
                improved = True
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * 0.01
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordPolish ({elapsed:.0f}s): obj={2*n*best_M:.6f}")
    return best_a, best_M


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_n = norm1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_n)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / s


def parametric_starts(n):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    results = []
    
    # Sine-squared
    for f in [1, 2, 3]:
        for ph in [0, 0.25, 0.5]:
            a = np.sin(np.pi * f * x + ph) ** 2
            a = np.maximum(a, 0.01)
            results.append(norm1(a))
    
    # Power laws
    for p in [0.5, 1.0, 2.0]:
        results.append(norm1((i + 1) ** p))
        results.append(norm1((n - i) ** p))
    
    # Bell curves
    for c in [0.3, 0.5, 0.7]:
        for s in [0.1, 0.2]:
            a = np.exp(-((x - c) ** 2) / (2 * s * s))
            results.append(norm1(a))
    
    # Uniform
    results.append(np.ones(n) / n)
    
    # Raised cosine
    results.append(norm1(np.cos(np.pi * (x - 0.5)) ** 2))
    
    return results

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def objective_val(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:

    return 2.0 * len(a) * fast_conv(a).max() / (s * s)

def build_lp_matrix(y, indices, n):
    K = len(indices)
    mat = np.zeros((K, n))
    for jj in range(K):
        k = int(indices[jj])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            mat[jj, i_start:i_end] = y[k - np.arange(i_start, i_end)]
    return mat


def solve_lp(fixed_pt, constraint_idx, n, solver):
    K = len(constraint_idx)
    if K == 0:
        return np.ones(n) / n, 1.0 / n
    C = build_lp_matrix(fixed_pt, constraint_idx, n)
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1.0, C @ a_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    a_val = np.zeros(n)
    if a_var.value is not None:
        a_val = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        sv = a_val.sum()
        if sv > 1e-15:
            a_val /= sv
        else:
            a_val = np.ones(n) / n
    else:
        a_val = np.ones(n) / n
    M_val = float(M_var.value) if M_var.value is not None else fast_conv(fixed_pt).max()
    return a_val, M_val


def cutting_plane_lp(fixed_pt, constraint_set, n, solver, max_inner=15):
    S = sorted(constraint_set)
    M_est = fast_conv(fixed_pt).max()
    for inner in range(max_inner):
        if len(S) > 5000:
            ac_f = fast_conv(fixed_pt)
            margins = np.array([M_est - ac_f[k] for k in S])
            keep = np.argsort(margins)[:4000]
            S = sorted(S[i] for i in keep)
        a_sol, M_sol = solve_lp(fixed_pt, S, n, solver)
        if a_sol is None:
            S = S[:min(300, len(S))]
            a_sol, M_sol = solve_lp(fixed_pt, S, n, solver)
            if a_sol is None:
                return fixed_pt, M_est
        # Verify and add violated constraints
        ac_check = fast_conv(a_sol)
        all_idx = np.arange(2 * n - 1)
        in_S = np.zeros(2 * n - 1, dtype=bool)
        in_S[S] = True
        violations = np.where((~in_S) & (ac_check > M_sol + 1e-9))[0]
        if len(violations) == 0:
            return a_sol, M_sol
        viol_amt = ac_check[violations] - M_sol
        top_v = violations[np.argsort(viol_amt)[::-1][:60]]
        for k in top_v:
            constraint_set.add(int(k))
        S = sorted(constraint_set)
        M_est = M_sol
    a_sol, M_sol = solve_lp(fixed_pt, S, n, solver)
    if a_sol is None:
        return fixed_pt, fast_conv(fixed_pt).max()
    return a_sol, M_sol


def fpw_main(a0, time_limit, solver, init_K=400, max_K=3500, verbose=True):

            stall_count += 1

        if t > 1:
            alpha = (1.0 - stt) / (1.0 + stt)

            x = np.maximum(x, 0.0)

            else:

def grad_lse(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm1(a0.copy())
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()

    ac0 = fast_conv(a)
    tau = max(ac0.max() * 0.4, 1e-8)
    tau_min = max(ac0.max() * 0.005, 1e-12)
    tau_decay = 0.9996
    lr = 0.6
    b1, b2 = 0.9, 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    t_cnt = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t_cnt += 1
        ac = fast_conv(a)
        M = ac.max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        tau = max(tau * tau_decay, tau_min)
        ac_s = ac / tau
        ac_s -= ac_s.max()
        p = np.exp(ac_s)
        p /= p.sum()

        a_flip = a[::-1]
        size = 2 * n - 1
        fft_sz = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a_flip, fft_sz)
        Pf = np.fft.rfft(p, fft_sz)
        conv_full = np.fft.irfft(Af * Pf, fft_sz)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0.0)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mc = m / (1.0 - b1 ** t_cnt)
        vc = v / (1.0 - b2 ** t_cnt)
        step = lr / (np.sqrt(vc) + 1e-8)

        log_a = np.log(np.maximum(a, 1e-30)) - step * grad
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        if stall > 2500:
            stall = 0
            rng = np.random.default_rng(t_cnt)
            pert = rng.normal(0, 0.02, n) * best_a
            a = norm1(best_a + pert)
            m = np.zeros(n)
            v = np.zeros(n)
            tau = max(best_M * 0.3, 1e-8)

        if verbose and t_cnt % 500 == 0:
            print(f"    GradLSE t={t_cnt} ({time.monotonic()-t0:.0f}s): obj={2*n*best_M:.6f}, tau={tau:.2e}")

    if verbose:
        print(f"    GradLSE done ({time.monotonic()-t0:.0f}s): obj={2*n*best_M:.6f}")
    return best_a, best_M


def coord_polish(a0, time_limit, verbose=False):

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_n = norm1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_n)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / s

        a_out, _ = fpw_main(best_seq, t3, solver, init_K=500, max_K=3500, verbose=True)
        update(a_out, f"fpw3_{n_best}")

    # Phase 4: Try a different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    alt_n = n_best + 300
    if remaining > 200 and alt_n <= 6000:
        print(f"\n=== Phase 4: Alt length n={alt_n} ===")
        a_alt = interp_seq(best_seq, alt_n)
        t4a = min(remaining * 0.15, 150.0)
        if t4a > 20:
            a_out, _ = grad_lse(a_alt, t4a, verbose=False)
            update(a_out, f"grad_{alt_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100:
            t4b = min(remaining * 0.25, 200.0)
            if t4b > 30:
                a_out, _ = fpw_main(norm1(best_seq) if len(norm1(best_seq)) == n_best else a_alt, t4b, solver, init_K=400, max_K=2500, verbose=False)
                update(a_out, f"fpw_{alt_n}")

    # Phase 5: Continue FPW on best (convergence phase)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: FPW convergence (n={n_best}) ===")
    if remaining > 60:
        t5 = min(remaining * 0.35, 350.0)
        a_out, _ = fpw_main(best_seq, t5, solver, init_K=600, max_K=4000, verbose=True)
        update(a_out, f"fpw5_{n_best}")

    # Phase 6: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord polish (n={n_best}) ===")
    if remaining > 20:
        t6 = min(remaining * 0.25, 120.0)
        a_out, _ = coord_polish(best_seq, t6, verbose=True)
        update(a_out, f"coord_{n_best}")

    # Phase 7: Final LP push
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    if remaining > 30 and n_best <= 5800:
        t7 = min(remaining * 0.55, 180.0)
        a_out, _ = fpw_main(best_seq, t7, solver, init_K=800, max_K=4500, verbose=False)
        update(a_out, f"fpwfinal_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        el = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={el:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm1(np.sin(np.pi * x) ** 2)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
