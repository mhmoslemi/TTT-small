    return grad, k_star

def subgradient_multi_peak(a, c, top_k=5):
    """Weighted subgradient using top peaks."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx]
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        # Hard weighting: emphasize the worst peak
        w = 1.0 / (top_vals + 1e-15)
        w = w / w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def mirror_descent_adaptive(a0, time_limit, verbose=False):
    """Mirror descent with adaptive step size and multiple peak targeting."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Parameters
    eta = 1.0 / np.sqrt(n)  # initial step size
    momentum = 0.9
    v = np.zeros(n)  # momentum accumulator
    top_k = 3
    
    stall_count = 0
    max_stall = 2000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall_count > max_stall:
            break
        
        # Compute convolution and subgradient
        c = fast_conv(a)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive top_k: start with 1 (hardest focus), increase if stuck
        cur_top_k = 1 if stall_count < 500 else (3 if stall_count < 1500 else 5)
        
        # Compute subgradient
        if cur_top_k == 1:
            grad, _ = subgradient_single_peak(a, c)
        else:
            grad = subgradient_multi_peak(a, c, top_k=cur_top_k)
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        
        # Mirror descent step: a_new ∝ a * exp(-eta * v)
        # Use log-space for numerical stability
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * v
        # Center for stability
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Backtracking: check if the step improved things
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < M * 0.999:  # significant improvement
            a = a_new
            eta = min(eta * 1.1, 10.0)  # increase step size
        else:
            # Partial step or revert
            for alpha in [0.5, 0.25, 0.1]:
                a_trial = norm_simplex((1 - alpha) * a + alpha * a_new)
                c_trial = fast_conv(a_trial)
                if c_trial.max() < M:
                    a = a_trial
                    break
            else:
                eta = max(eta * 0.7, 1e-4)
        
        # Anneal step size slowly
        progress = min(1.0, t / 50000.0)
        eta = eta * (1.0 - 0.001 * progress)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD t={t} ({elapsed:.0f}s): M={best_m:.8f}, eta={eta:.4f}")
    
    return best_a, best_m

def multiplicative_peak_shave(a0, time_limit, verbose=False):
    """Multiplicative update that reduces peaks in the autocorrelation."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    
    eta = 0.3  # step size for multiplicative update
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 500:
            break
        
        c = fast_conv(a)
        M = c.max()
        avg = c.sum() / len(c)
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find all peaks above threshold
        threshold = avg * 1.1  # slightly above average
        peak_idx = np.where(c > threshold)[0]
        
        if len(peak_idx) == 0:
            # No peaks above threshold, use the single max
            peak_idx = np.array([int(np.argmax(c))])
        
        # Cap the number of peaks we handle
        if len(peak_idx) > 20:
            # Take the worst 20
            order = c[peak_idx].argsort()[::-1][:20]
            peak_idx = peak_idx[order]
        
        # Compute the "responsibility" of each a[i] for all peaks
        resp = np.zeros(n)
        for k in peak_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            resp[idx] += a[j_idx]
        
        # Multiplicative update: reduce a[i] proportionally to its responsibility
        # a[i] *= (1 - eta * resp[i] / (M + eps))
        # But this could make some a[i] zero, so use a softer version
        ratio = resp / (M + 1e-15)
        a_new = a * np.exp(-eta * np.minimum(ratio, 2.0))
        
        # Also add mass to positions with low responsibility
        # This is implicit in the normalization step
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check improvement
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < M * 0.9999:
            a = a_new
            eta = min(eta * 1.02, 1.0)
        else:
            # Try smaller step
            for alpha in [0.5, 0.25, 0.1]:
                a_trial = norm_simplex((1 - alpha) * a + alpha * a_new)
                M_trial = fast_conv(a_trial).max()
                if M_trial < M:
                    a = a_trial
                    break
            else:
                eta = max(eta * 0.8, 0.01)
                if eta < 0.01:
                    break
        
        # Periodic random perturbation to escape local minima
        if t % 1000 == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MPS t={t} ({elapsed:.0f}s): M={best_m:.8f}, eta={eta:.4f}")
    
    return best_a, best_m

def freq_domain_flatten(a0, time_limit, verbose=False):
    """Iteratively flatten the autocorrelation using frequency domain operations."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    step = 0.5
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        # The autocorrelation is IFFT(P). We want it flat.
        # Flat autocorrelation means P is concentrated at DC.
        # But we can't make P a delta (that gives uniform a, which gives triangular autocorr).
        # Instead, we iteratively reduce the "spread" of P away from DC.
        
        # Strategy: apply a low-pass filter to P (keep DC, reduce high freq)
        # This makes the autocorrelation more "peaked" at center... wait that's the opposite.
        
        # Actually, let me think about this differently.
        # We want IFFT(P) to be flat. So P should be a delta at 0.
        # But P[0] = (sum a)^2 = 1 (fixed).
        # So we want all other P[k] to be 0.
        # That means A[k] = 0 for k != 0, i.e., a is uniform.
        # But uniform gives triangular autocorr, not flat!
        
        # The issue: with zero-padding, the linear autocorrelation is NOT the same
        # as the circular autocorrelation of the padded sequence.
        
        # So the frequency domain approach doesn't directly apply here.
        # Let me skip this and use a time-domain approach instead.
        
        # Alternative: use the "flat target" in the time domain
        target = np.ones(2 * n - 1) / (2 * n - 1)  # flat target
        c = fast_conv(a)
        error = c - target  # positive where c is above target
        
        # Compute subgradient of sum of positive errors squared
        # d/d a[i] of sum_k max(0, c[k] - target[k])^2
        # = 2 * sum_{k: c[k]>target} (c[k]-target[k]) * d c[k]/d a[i]
        # d c[k]/d a[i] = a[k-i] + a[i-k] (by symmetry of convolution with self)
        # Wait, for c[k] = sum_j a[j]*a[k-j], dc[k]/da[i] = a[k-i] + a[i-k]
        # But since c[k] = c[2n-2-k] (symmetry), we can use this.
        
        # For the peak at k*: dc[k*]/da[i] = a[k*-i] (for valid i)
        # For our purposes, just use the subgradient at the worst peak
        
        # Actually, let me just do a simple "flatten" step:
        # Reduce a[i] where it contributes to peaks, increase where it doesn't
        peak_positions = np.where(error > 0)[0]
        if len(peak_positions) == 0:
            break
        
        # Limit to worst peaks
        n_peaks = min(20, len(peak_positions))
        worst_peaks = peak_positions[np.argsort(error[peak_positions])[::-1][:n_peaks]]
        
        # Compute responsibility
        resp = np.zeros(n)
        for k in worst_peaks:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            resp[idx] += a[j_idx] * (c[k_int] - target[k_int])
        
        # Update: a[i] proportional to exp(-step * resp[i])
        max_resp = resp.max()
        if max_resp < 1e-15:
            break
        a_new = a * np.exp(-step * resp / max_resp)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m * 0.9999:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            step = max(step * 0.8, 0.01)
            if step < 0.01:
                break
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass from peak-contributing positions to others."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 3000:
        c = fast_conv(a)
        M = c.max()
        k_star = int(np.argmax(c))
        
        # Find positions that most contribute to the peak
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_star - idx]
        
        # Get top contributors
        top_contrib = np.argsort(contrib)[::-1][:min(10, len(contrib))]
        src_candidates = idx[top_contrib]
        
        improved = False
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try moving to random positions
            for _ in range(20):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt > src_val:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

def make_init_flat(n):
    return np.ones(n) / n

def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.001)
    return norm_simplex(a)

def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)

def make_init_flattop(n):
    """Flat-top window initialization."""
    x = np.linspace(0, 1, n)
    a = 0.21557895 + 0.41663158 * np.cos(2*np.pi*x) + 0.277263158 * np.cos(4*np.pi*x) \
        + 0.083578947 * np.cos(6*np.pi*x) + 0.006947368 * np.cos(8*np.pi*x)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)

def make_init_blackman(n):
    """Blackman window."""
    x = np.linspace(0, 1, n)
    a = 0.42 - 0.5 * np.cos(2*np.pi*x) + 0.08 * np.cos(4*np.pi*x)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)

def make_init_gaussian(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-(x - 0.5)**2 / (2*sigma**2))
    return norm_simplex(a)

def make_init_dual_gauss(n, pos1=0.3, pos2=0.7, sigma=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-(x-pos1)**2/(2*sigma**2)) + np.exp(-(x-pos2)**2/(2*sigma**2))
    a += 0.05
    return norm_simplex(a)

def make_init_rand_smooth(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15/np.sqrt(f))
        phase = rng.uniform(0, 2*np.pi)
        a += coeff * np.sin(2*np.pi*f*x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with adaptive method selection."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        elif np.isfinite(sc):
            elapsed = time.monotonic() - t_global
            print(f"  [no improvement] {label}: {sc:.6f} (t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4200
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Multi-start mirror descent ===
    print(f"\n=== Phase 1: Mirror Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 480.0)
    
    if t1 > 30:
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(norm_simplex(parent))
            labels.append("parent")
        
        inits.append(make_init_cosine(n0, mode=1))
        labels.append("cos1")
        
        inits.append(make_init_flattop(n0))
        labels.append("flattop")
        
        inits.append(make_init_blackman(n0))
        labels.append("blackman")
        
        inits.append(make_init_bowl(n0, sharpness=2.0))
        labels.append("bowl2")
        
        inits.append(make_init_dual_gauss(n0))
        labels.append("dual")
        
        inits.append(make_init_rand_smooth(n0, seed=42))
        labels.append("rand1")
        
        inits.append(make_init_rand_smooth(n0, seed=123))
        labels.append("rand2")
        
        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0//50)], 5))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels
        
        print(f"  {len(inits)} initializations, {t1:.0f}s total")
        t_per = t1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            t_avail = min(t_per, remaining * 0.9)
            if t_avail < 10:
                break
            print(f"    MD '{lbl}' ({t_avail:.0f}s)...")
            a_out, _ = mirror_descent_adaptive(init, t_avail)
            try_update(a_out, f"md_{lbl}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_mp, _ = multiplicative_peak_shave(a_out, min(20.0, remaining * 0.2))
                try_update(a_mp, f"mps_{lbl}")

            break
        print(f"  n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        try_update(a_interp, f"interp_{new_n}")

        if t5 > 15:

            try_update(a_out, f"md_{new_n}")

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def subgradient_single(a, c):

    i_end = min(n, k_star + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k_star - idx
    grad[idx] = 2.0 * a[j_idx]
    return grad

            stall += 1

            improved = False
            for alpha in [0.5, 0.25, 0.1]:
                a_trial = norm_simplex((1 - alpha) * a + alpha * a_new)
                M_trial = fast_conv(a_trial).max()
                if M_trial < M:
                    a = a_trial
                    improved = True
                    break
            if not improved:
                eta = max(eta * 0.85, 0.01)
                if eta < 0.02:
                    # Perturb
                    if stall > 500:
                        i = int(rng.integers(n))
                        j = int(rng.integers(n))
                        if i != j and a[i] > 1e-8:
                            frac = rng.uniform(0.001, 0.005)
                            amt = a[i] * frac
                            a[i] -= amt
                            a[j] += amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        # Periodic perturbation
        if t % 2000 == 0 and stall > 200:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.0005, 0.002)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MPS t={t} ({elapsed:.0f}s): M={best_m:.8f}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass from peak contributors to others."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M = c.max()
        k_star = int(np.argmax(c))

        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_star - idx]

        top_contrib = np.argsort(contrib)[::-1][:min(8, len(contrib))]
        src_candidates = idx[top_contrib]

        improved = False
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            src_val = a[src]
            if src_val < 1e-10:
                continue

            for _ in range(15):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.003)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def make_init_flat(n):
    return np.ones(n) / n


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_flattop(n):
    x = np.linspace(0, 1, n)
    a = (0.21557895 + 0.41663158 * np.cos(2 * np.pi * x)
         + 0.277263158 * np.cos(4 * np.pi * x)
         + 0.083578947 * np.cos(6 * np.pi * x)
         + 0.006947368 * np.cos(8 * np.pi * x))
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def make_init_blackman(n):
    x = np.linspace(0, 1, n)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def make_init_dual_gauss(n, pos1=0.3, pos2=0.7, sigma=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-(x - pos1) ** 2 / (2 * sigma ** 2)) + np.exp(-(x - pos2) ** 2 / (2 * sigma ** 2))
    a += 0.05
    return norm_simplex(a)


def make_init_rand_smooth(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search: LP direction finding + mirror descent + multiplicative updates."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        elif np.isfinite(sc):
            elapsed = time.monotonic() - t_global
            print(f"  [{label}]: {sc:.6f} (t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP direction finding on parent + best inits ===
    print(f"\n=== Phase 1: LP Direction (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)

    if t1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_simplex(parent))
            labels.append("parent")

        inits.append(make_init_flattop(n0))
        labels.append("flattop")

        inits.append(make_init_cosine(n0, mode=1))
        labels.append("cos1")

        inits.append(make_init_bowl(n0, sharpness=2.5))
        labels.append("bowl2.5")

        inits.append(make_init_dual_gauss(n0))
        labels.append("dual")

        inits.append(make_init_rand_smooth(n0, seed=42))
        labels.append("rand1")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 5))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        print(f"  {len(inits)} inits, {t1:.0f}s budget")
        t_per = t1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_avail = min(t_per, remaining * 0.85)
            if t_avail < 15:
                break
            print(f"    LP '{lbl}' ({t_avail:.0f}s)...")
            a_out, _ = lp_direction_optimize(init, t_avail)
            try_update(a_out, f"lp_{lbl}")

            # Quick MD polish
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_md, _ = mirror_descent_adaptive(a_out, min(25.0, remaining * 0.2))
                try_update(a_md, f"md_{lbl}")

    # === Phase 2: MPS on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MPS (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = multiplicative_peak_shave(a_in, t2)
        try_update(a_out, "mps_best")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out2, _ = lp_direction_optimize(a_out, min(40.0, remaining * 0.3))
            try_update(a_out2, "lp_after_mps")

    # === Phase 3: Long LP run ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Long LP (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = lp_direction_optimize(a_in, t3, verbose=True)
        try_update(a_out, "lp_long")

    # === Phase 4: Alternate MD/MPS ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Alternate methods ===")
    a_cur = norm_simplex(best_seq)
    use_lp = True
    while time.monotonic() - t_global < time_budget - 60:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        chunk = min(remaining * 0.3, 60.0)

        if use_lp:
            a_out, _ = lp_direction_optimize(a_cur, chunk)
        else:
            a_out, _ = multiplicative_peak_shave(a_cur, chunk)

        try_update(a_out, f"alt_{'lp' if use_lp else 'mps'}")
        a_cur = norm_simplex(best_seq)
        use_lp = not use_lp

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (n={n_best}) ===")

    for new_n in [n_best + 400, n_best + 800, n_best - 400, n_best - 800]:
        if new_n < 2000 or new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        try_update(a_interp, f"interp_{new_n}")

        t5 = min(remaining * 0.2, 100.0)
        if t5 > 15:
            a_out, _ = lp_direction_optimize(a_interp, t5)
            try_update(a_out, f"lp_{new_n}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_out2, _ = multiplicative_peak_shave(a_out, min(20.0, remaining * 0.15))
                try_update(a_out2, f"mps_{new_n}")

    # === Phase 6: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Coord (n={n_best}) ===")
    t6 = min(remaining * 0.3, 60.0)
    if t6 > 5:
        a_out, _ = coordinate_polish(norm_simplex(best_seq), t6)
        try_update(a_out, "coord_final")

    # === Phase 7: Final intensive LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 7: Final LP ===")
        a_out, _ = lp_direction_optimize(norm_simplex(best_seq), remaining * 0.7)
        try_update(a_out, "lp_absolute_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        best_seq = make_init_cosine(4000, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
