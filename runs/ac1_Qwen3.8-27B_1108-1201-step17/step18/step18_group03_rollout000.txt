import scipy.optimize as so
import cvxpy as cp

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def score_normalized(a):
    """Compute 2*n*max(ac) assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * ac.max() / (s*s))

                A_ub=A_ub, b_ub=b_ub,
                A_eq=np.concatenate([np.ones(n), [0.0]]).reshape(1, -1),
                b_eq=np.array([1.0]),
                bounds=[(0, None)]*n + [(0, None)],
                method='highs'
            )
        except Exception:
            break
        
        if res.status != 0:
            break
        
        g = np.maximum(res.x[:n], 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        y = y_new
        
        iter_count += 1
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"  LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m
