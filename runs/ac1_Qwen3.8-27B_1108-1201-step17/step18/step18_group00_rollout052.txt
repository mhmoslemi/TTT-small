Let me define q[k] = p[2n-2-k] (reversed p). Then p[m+j] = q[2n-2-m-j].
sum_j a[j] * q[2n-2-m-j] = (a * q)[2n-2-m] (convolution)

So cross_corr[m] = (a * q)[2n-2-m] where q is the reversed p.

I can compute a*q using FFT: ifft(fft(a) * fft(q)).

And the window sum: w_s[m] = sum_{k=m}^{min(2n-2, m+n-1)} p[k] = prefix_sum_p[min(2n-2, m+n-1)+1] - prefix_sum_p[m]

This is O(n) with a prefix sum.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe,
continuation smoothing, and multi-scale exploration.
"""

import numpy as np
import time
import cvxpy as cp

_solver = None

def _get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print("Solver: " + str(s))
                return s
        except:
            continue
    _solver = cp.ECOS
    return _solver

def autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

def eval_score(a):
    """Compute the objective value (assumes a >= 0, sum > 0)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interp_to(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    out = np.interp(new_x, old_x, fn)
    return normalize(out)

def lp_fw(a0, time_limit, solver, K_init=500, K_max=4000, add_per=30, verbose=True):
    """LP-based Frank-Wolfe with multi-alpha line search."""
    n = len(a0)
    y = normalize(a0.astype(np.float64))
    best_a = y.copy()
    best_M = autocorr(y).max()
    t0 = time.monotonic()
    
    ac_y = autocorr(y)
    active = set(int(k) for k in np.argsort(ac_y)[::-1][:K_init])
    
    no_imp = 0
    max_no_imp = 1000
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_imp > max_no_imp:
            break
        
        ac = autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_M - 1e-14:
            best_M = M_curr
            best_a = y.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Manage active set
        if len(active) > K_max:
            margins = np.array([M_curr - ac[int(k)] for k in active])
            keep = np.argsort(margins)[:K_max]
            active = set(int(list(active)[i]) for i in keep)
        
        c_list = sorted(active)
        if not c_list:
            break
        
        # Build constraint matrix
        K = len(c_list)
        C = np.zeros((K, n))
        for jj, k in enumerate(c_list):
            i_s = max(0, k - n + 1)
            i_e = min(n, k + 1)
            if i_s < i_e:
                idx = np.arange(i_s, i_e)
                C[jj, idx] = y[k - idx]
        
        # Solve LP
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [cp.sum(g) == 1, C @ g <= M])
        try:
            prob.solve(solver=solver, verbose=False)
        except:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        
        # Add violated constraints
        # Compute y * g_dir convolution
        fs = 1 << (2*n - 2).bit_length()
        Yf = np.fft.rfft(y, fs)
        Gf = np.fft.rfft(g_dir, fs)
        conv_ys = np.fft.irfft(Yf * Gf, fs)[:2*n-1]
        viol = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(viol) > 0:
            top_v = viol[np.argsort(conv_ys[viol])[::-1][:add_per]]
            for k in top_v:
                active.add(int(k))
        
        # Line search: try multiple alphas
        if t == 1:
            alpha = 1.0
        else:
            default_a = 2.0 / (t + 2.0)
            candidates = [default_a, default_a*0.5, default_a*2.0, 0.1, 0.3, 0.5, 0.7, 1.0]
            candidates = [max(0.01, min(1.0, c)) for c in candidates]
            candidates = list(set(candidates))
            
            best_alpha = default_a
            best_trial = float('inf')
            for a_val in candidates:
                trial = (1-a_val)*y + a_val*g_dir
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                tm = autocorr(trial).max()
                if tm < best_trial:
                    best_trial = tm
                    best_alpha = a_val
            alpha = best_alpha
        
        y = (1-alpha)*y + alpha*g_dir
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = best_a.copy()
        else:
            y /= ys
        
        if verbose and t % 200 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("  LP-FW t=%d (%.0fs): J=%.6f |S|=%d" % (t, el, J, len(active)))
    
    el = time.monotonic() - t0
    J = 2.0 * n * best_M
    if verbose:
        print("  LP-FW done (%.0fs): J=%.6f" % (el, J))
    return best_a, best_M

def smooth_objective_and_grad(a, tau, n):
    """Compute smoothed objective and gradient."""
    ac = autocorr(a)
    S = a.sum()
    
    # Softmax of ac/tau
    ac_t = ac / tau
    ac_t -= ac_t.max()  # for numerical stability
    exp_ac = np.exp(ac_t)
    sum_exp = exp_ac.sum()
    L = np.log(sum_exp) + ac.max() / tau  # logsumexp
    p = exp_ac / sum_exp
    
    # Window sums: w_s[m] = sum_{k=m}^{min(2n-2, m+n-1)} p[k]
    prefix = np.zeros(2*n)
    prefix[1:] = np.cumsum(p)
    w_s = np.zeros(n)
    for m in range(n):
        k_end = min(2*n-2, m+n-1)
        w_s[m] = prefix[k_end+1] - prefix[m]
    
    # Cross-correlation: cc[m] = sum_{j=0}^{n-1} a[j] * p[m+j]
    # = (a * q)[2n-2-m] where q = reversed p
    q = p[::-1]
    fs = 1 << (2*n - 2).bit_length()
    Af = np.fft.rfft(a, fs)
    Qf = np.fft.rfft(q, fs)
    conv_aq = np.fft.irfft(Af * Qf, fs)[:2*n-1]
    cc = conv_aq[2*n-2 - np.arange(n)]
    
    # dL/da[m] = (1/tau) * (cc[m] + a[m] * w_s[m])
    dL = (1.0/tau) * (cc + a * w_s)
    
    # f = 2n * L / S^2
    f = 2.0 * n * L / (S * S)
    
    # df/da[m] = 2n/S^2 * (dL[m] - 2L/S)
    grad = (2.0 * n / (S * S)) * (dL - 2.0 * L / S)
    
    return f, grad

def smooth_optimize(a0, time_limit, verbose=True):
    """Continuation-based smooth optimization using L-BFGS-B."""
    from scipy.optimize import minimize
    
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    ac = autocorr(a)
    M = ac.max()
    
    # Continuation: decrease tau
    tau_start = 0.1 * M
    tau_end = 0.005 * M
    n_stages = 8
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        
        tau = tau_start * (tau_end / tau_start) ** (stage / max(n_stages - 1, 1))
        
        def obj_and_grad(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                return 1e10, np.zeros_like(x)
            f, g = smooth_objective_and_grad(x, tau, n)
            return f, g
        
        res = minimize(obj_and_grad, a, method='L-BFGS-B',
                      bounds=[(0, 1)] * n,
                      options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 2000})
        
        a_new = np.maximum(res.x, 0)
        a_new = normalize(a_new)
        s = eval_score(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
            a = a_new
            if verbose:
                el = time.monotonic() - t0
                print("  Smooth stage %d (tau=%.2e, %.0fs): score=%.6f" % (stage, tau, el, best_score))
        else:
            if verbose:
                el = time.monotonic() - t0
                print("  Smooth stage %d (tau=%.2e, %.0fs): no improvement" % (stage, tau, el))
    
    if verbose:
        el = time.monotonic() - t0
        print("  Smooth done (%.0fs): best=%.6f" % (el, best_score))
    return best_a, best_score

def coord_descent(a0, time_limit, verbose=True):
    """Coordinate descent polish."""
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    best_a = a.copy()
    best_M = autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = autocorr(a)
        top_k = np.argsort(ac)[::-1][:20]
        improved = False
        
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n - 1, ki)
            if i_s > i_e:
                continue
            idx = np.arange(i_s, i_e + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            sv = a[src]
            if sv < 1e-12:
                continue
            
            # Find best destination
            scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            cands = np.where(~in_peak)[0]
            if len(cands) < 10:
                cands = np.arange(n)
            
            dst = int(cands[np.argmin(scores[cands])])
            if dst == src:
                continue
            
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = sv * frac
                if amt < 1e-15:
                    continue
                trial = a.copy()
                trial[src] -= amt
                trial[dst] += amt
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                tm = autocorr(trial).max()
                if tm < best_M - 1e-15:
                    a = trial
                    best_M = tm
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a /= a.sum()
                nm = autocorr(a).max()
                if nm < best_M:
                    best_M = nm
                    best_a = a.copy()
                    stall = 0
        else:
            stall = 0
        
        if verbose and stall % 200 == 0:
            el = time.monotonic() - t0
            print("  Coord (%.0fs): J=%.6f stall=%d" % (el, 2.0*n*best_M, stall))
    
    return best_a, best_M
