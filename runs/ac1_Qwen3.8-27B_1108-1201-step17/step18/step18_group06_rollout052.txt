def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    a = np.maximum(a, 0.0)

    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]
