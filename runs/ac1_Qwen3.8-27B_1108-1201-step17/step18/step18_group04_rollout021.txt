import numpy as np

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_mixed_grad(a, conv, top_k=50, temperature=0.005):
    """Compute gradient using softmax over active peaks, with option to use exact argmax."""
    n = len(a)
    conv_max = conv.max()
    grad = np.zeros(n)
    
    # Find active peaks
    threshold = conv_max - temperature * 10.0
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Use softmax weights
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    
    # Softmax weights
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def amd_optimize_v2(a0, time_limit, top_k_init=30, temp_init=0.005, c0_init=3.0,
                    restart_every=4000, seed=42, use_line_search=True):
    """Improved AMD with adaptive temperature and line search."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    top_k = top_k_init
    temperature = temp_init
    c0 = c0_init
    stall_count = 0
    restart_count = 0
    temp_min = 0.0005
    n_iter = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        n_iter += 1
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        conv_y = fast_conv(y)
        
        # Adaptive temperature: cool down over time
        temperature = max(temp_min, temp_init * np.exp(-0.0001 * t))
        top_k = min(80, top_k_init + t // 2000)
        
        g = compute_mixed_grad(y, conv_y, top_k=top_k, temperature=temperature)
        
        # Step size with optional line search
        eta = c0 / (np.sqrt(t) + 1.0)
        
        if use_line_search and t % 5 == 0:
            # Try a few step sizes
            best_eta = eta
            best_obj = conv_y.max()
            for factor in [0.5, 0.75, 1.0, 1.5, 2.0]:
                eta_try = eta * factor
                log_a = np.log(y) - eta_try * g
                max_log = log_a.max()
                a_try = np.exp(log_a - max_log)
                a_try = np.maximum(a_try, 1e-30)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                M_try = fast_conv(a_try).max()
                if M_try < best_obj:
                    best_obj = M_try
                    best_eta = eta_try
            eta = best_eta
        
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.08, 0.02 + 0.01 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            temperature = max(temp_min, 0.003)
            c0 = 3.0 * (1.0 + 0.05 * min(restart_count, 12))
            if restart_count > 5:
                restart_every = max(2000, restart_every // 2)
    
    return best_x, best_m

def coord_descent_v2(a0, time_limit, seed=42, top_peaks=10, n_candidates=15):
    """Improved coordinate descent with line search and multi-peak awareness."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    n_moves = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        
        # Get top peaks
        top_peak_indices = np.argsort(conv)[::-1][:top_peaks]
        peak_vals = conv[top_peak_indices]
        
        # Compute cost for each position (how much it contributes to peaks)
        costs = np.zeros(n)
        for k2 in top_peak_indices:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]
        
        # Find the main peak
        peak_k = int(top_peak_indices[0])
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Top source positions (contributing most to the main peak)
        n_top_src = max(10, min(30, n // 100))
        top_src = idx[np.argsort(contrib)[::-1][:n_top_src]]
        
        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            
            # Destination candidates: positions with lowest cost
            candidates = np.argsort(costs + 1e-10 * a)[:n_candidates]
            candidates = [int(c) for c in candidates if int(c) != src_i]
            
            for dst_i in candidates:
                found = False
                # Binary search for optimal fraction
                lo_frac = 0.01
                hi_frac = 0.9
                best_frac = 0.3
                best_M = best_m
                
                for _ in range(8):  # 8 iterations of binary search
                    mid_frac = (lo_frac + hi_frac) / 2
                    amt = a[src_i] * mid_frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M:
                        best_M = M_trial
                        best_frac = mid_frac
                        hi_frac = mid_frac
                    else:
                        lo_frac = mid_frac
                
                if best_M < best_m - 1e-14:
                    amt = a[src_i] * best_frac
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_m = fast_conv(a).max()
                    best_a = a.copy()
                    improved = True
                    n_moves += 1
                    found = True
                    break
            
            if found:
                break
        
        if not improved:
            stall += 1
            # Try random perturbations
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a = best_a.copy()
    
    return best_a, best_m

def simulated_annealing(a0, time_limit, seed=42):
    """Simulated annealing to escape local minima."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T_start = 0.01
    T_end = 0.0001
    n_iter_budget = 50000  # max iterations before time limit
    
    for it in range(n_iter_budget):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Temperature schedule
        progress = it / n_iter_budget
        T = T_start * (T_end / T_start) ** progress
        
        # Generate perturbation
        n_pert = rng.integers(1, max(2, n // 1000) + 1)
        indices = rng.integers(0, n, n_pert)
        deltas = rng.normal(0, 0.01, n_pert) * a[indices]
        
        a_trial = a.copy()
        for idx, d in zip(indices, deltas):
            a_trial[idx] += d
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        
        M_trial = fast_conv(a_trial).max()
        M_current = fast_conv(a).max()
        
        delta = M_trial - M_current
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a_trial.copy()
    
    return best_a, best_m

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss
