```python
    """
    Compute the subgradient of max(conv(a,a)) using the top lag peaks.
    grad[i] = sum_k w_k * d(conv[k])/d(a[i])
    d(conv[k])/d(a[i]) = a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n and i-k != k-i
    """
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(top_lags, weights):
        k = int(k)
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_lo = max(0, k - n + 1)
        i_hi = min(n-1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi+1)
            grad[idx] += w * a[k - idx]
        # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
        # And we need i-k != k-i, i.e., 2i != 2k, i.e., i != k (when k is even)
        j_lo = max(0, k)  # i >= k
        j_hi = min(n-1, k + n - 1)  # i <= k+n-1, but i < n so i <= n-1

            contrib = a[idx2 - k]
            # Exclude the case where i-k == k-i, i.e., i == k (only when k is even and 0<=k<n)
            if 0 <= k <= n-1:
                mask = idx2 != k
                contrib = contrib * mask
            grad[idx2] += w * contrib
    return grad
```

```python
import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_subgrad(a, top_lags, weights):
    """
    Compute subgradient of max(conv) using top peaks.
    grad[i] = sum_k w_k * (a[k-i] + a[i-k] for valid indices, with correction)
    
    For linear convolution c = a * a:
    c[k] = sum_j a[j] * a[k-j]
    d c[k] / d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n and i-k != k-i)
    """
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(top_lags, weights):
        k = int(k)
        # Term 1: a[k-i] valid when 0 <= k-i < n => k-n+1 <= i <= k
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += w * a[k - idx]
        # Term 2: a[i-k] valid when 0 <= i-k < n => k <= i <= k+n-1
        # And i-k != k-i => i != k (when k is even... no, i-k = k-i => 2i = 2k => i = k)
        j_lo = max(0, k)
        j_hi = min(n - 1, k + n - 1)
        if j_lo <= j_hi:
            idx2 = np.arange(j_lo, j_hi + 1)
            val = a[idx2 - k]
            # Subtract the double-counted term where i-k == k-i (i.e., i == k)
            if j_lo <= k <= j_hi:
                idx2_masked = idx2[idx2 != k]
                val_masked = val[idx2 != k]
                grad[idx2_masked] += w * val_masked
            else:
                grad[idx2] += w * val
    return grad
```

Wait, I think I'm making an error. Let me reconsider.

c[k] = sum_{j} a[j] * a[k-j] where the sum is over j such that 0 <= j < n and 0 <= k-j < n.

d c[k] / d a[i]:
- a[i] appears in the term where j = i: a[i] * a[k-i] (if 0 <= k-i < n)
- a[i] appears in the term where k-j = i, i.e., j = k-i: a[k-i] * a[i] (if 0 <= k-i < n)

So d c[k] / d a[i] = a[k-i] * [0 <= k-i < n] + a[k-i] * [0 <= k-i < n] = 2 * a[k-i] * [0 <= k-i < n]

Wait, that's only if the two appearances are distinct, i.e., j=i and j=k-i are different terms. They're the same when i = k-i, i.e., i = k/2. In that case, c[k] contains the term a[k/2]^2, and d(a[k/2]^2)/d(a[k/2]) = 2*a[k/2], which is the same as 2*a[k-i].

Actually, let me re-derive. c[k] = sum_j a[j]a[k-j].
If i != k-i (i.e., i != k/2), then a[i] appears in exactly two terms:
- j = i: a[i] * a[k-i]
- j = k-i: a[k-i] * a[i]
So d c[k]/d a[i] = a[k-i] + a[k-i] = 2*a[k-i].

If i = k/2 (only possible when k is even), then a[i] appears in exactly one term:
- j = i = k/2: a[k/2] * a[k/2] = a[k/2]^2
So d c[k]/d a[k/2] = 2*a[k/2].

So in all cases: d c[k]/d a[i] = 2*a[k-i] for valid k-i (i.e., 0 <= k-i < n), and 0 otherwise.

Wait, that can't be right for all cases. Let me check with a small example.

n=3, a = [a0, a1, a2].
c[0] = a0*a0 = a0^2. d c[0]/d a0 = 2*a0. ✓ (k=0, i=0: k-i=0, valid, 2*a[0]=2*a0)
d c[0]/d a1: k=0, i=1: k-i=-1, not valid. ✓
d c[0]/d a2: k=0, i=2: k-i=-2, not valid. ✓

c[1] = a0*a1 + a1*a0 = 2*a0*a1.
d c[1]/d a0: k=1, i=0: k-i=1, valid. 2*a[1] = 2*a1. ✓
d c[1]/d a1: k=1, i=1: k-i=0, valid. 2*a[0] = 2*a0. ✓
d c[1]/d a2: k=1, i=2: k-i=-1, not valid. 0. ✓

c[2] = a0*a2 + a1*a1 + a2*a0 = 2*a0*a2 + a1^2.
d c[2]/d a0: k=2, i=0: k-i=2, valid. 2*a[2] = 2*a2. ✓
d c[2]/d a1: k=2, i=1: k-i=1, valid. 2*a[1] = 2*a1. ✓ (this is the 2*a1 from d(a1^2)/da1)
d c[2]/d a2: k=2, i=2: k-i=0, valid. 2*a[0] = 2*a0. ✓

c[3] = a1*a2 + a2*a1 = 2*a1*a2.
d c[3]/d a1: k=3, i=1: k-i=2, valid. 2*a[2] = 2*a2. ✓
d c[3]/d a2: k=3, i=2: k-i=1, valid. 2*a[1] = 2*a1. ✓
d c[3]/d a0: k=3, i=0: k-i=3, not valid. 0. ✓

c[4] = a2*a2 = a2^2.
d c[4]/d a2: k=4, i=2: k-i=2, valid. 2*a[2] = 2*a2. ✓

So the formula is: d c[k]/d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

This is much simpler than what I had before! The subgradient is just:
grad[i] = sum_{k in peaks} w_k * 2 * a[k-i] * [0 <= k-i < n]

Which can be written as: grad = 2 * (w_conv a) where w is a signal with w[k] = w_k at peak positions, and * denotes cross-correlation with appropriate bounds.

Actually, more precisely: grad[i] = 2 * sum_{k: 0<=k-i<n} w_k * a[k-i] = 2 * sum_{j: 0<=j<n} w_{i+j} * a[j] (substituting j = k-i).

This is a cross-correlation: grad = 2 * (w cross-correlate a) = 2 * (w convolve reverse(a)) evaluated at the right positions.

For efficient computation, I can use FFT:
Let W = FFT(w_padded, 2n) where w_padded has w[k] at positions k=0,...,2n-2.
Let A = FFT(a_padded, 2n) where a_padded has a[0..n-1] at positions 0..n-1.
Then grad_raw = IFFT(W * conj(A)) gives the cross-correlation.

Actually, let me just use the direct formula for efficiency since we only have a few peak lags:

grad[i] = 2 * sum_{k in peaks} w_k * a[k-i] for k-i in [0, n-1]

This can be computed as:
For each peak k: add 2*w_k * a[k-i] to grad[i] for i in [max(0, k-n+1), min(n-1, k)]

This is O(n * num_peaks) per iteration, which is fine.

Let me simplify the code:

```python
def compute_subgrad(a, top_lags, weights):
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(top_lags, weights):
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += 2.0 * w * a[k - idx]
    return grad
```

This is clean and efficient.

Now, for the MGD update:
a_new[i] = a[i] * exp(-eta * grad[i])

Then normalize: a_new = a_new / sum(a_new)

The step size eta needs to be chosen carefully. Too large and we oscillate; too small and we converge slowly.

A good approach is backtracking: start with eta, and if the objective doesn't decrease, halve eta.

But computing the objective (max of autocorrelation) requires an FFT each time, which is O(n log n). For n=5000, this is fast (~100 microseconds).

Let me now write the complete code:

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using MGD with improved initialization
and adaptive step sizes.
"""

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_subgrad(a, top_lags, weights):
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(top_lags, weights):
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += 2.0 * w * a[k - idx]
    return grad


def mgd_optimize(a0, time_limit, top_k=40, eta0=0.03, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = eta0
    patience = 0
    max_patience = 500
    cur_top_k = top_k
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 1.5  # Use power 1.5 instead of 2 for more balanced weights
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = compute_subgrad(a, top_idx, w)
        grad = grad - grad.mean()  # Center the gradient
        
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        # Backtracking line search
        accepted = False
        eta_try = eta
        for _ in range(10):
            a_new = a * np.exp(-eta_try * grad)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            M_new = fast_autocorr(a_new).max()
            if M_new <= M + 1e-12:
                a = a_new
                eta = eta_try * 1.05  # Slightly increase if successful
                accepted = True
                break
            else:
                eta_try *= 0.5
        
        if not accepted:
            eta = eta_try
            if eta < 0.0001:
                # Restart with different parameters
                cur_top_k = max(5, cur_top_k // 2)
                eta = 0.02
                patience = max(0, patience - 100)
        
        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def coord_descent_polish(a0, time_limit, verbose=False):
    """Coordinate descent: try moving mass between pairs of positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        top_k = np.argsort(ac)[::-1][:8]
        improved = False
        
        # Try moving mass from peak-contributing positions to low-impact ones
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Find best destination: position that contributes least to peaks
            peak_pressure = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_pressure[idx2] += a[k2i - idx2]
            
            # Look for low-pressure positions
            candidates = np.argsort(peak_pressure)[:n // 4]
            for dst in candidates[:20]:
                if int(dst) == src:
                    continue
                for frac in [0.01, 0.02, 0.04, 0.06]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def gen_linear_perturb(n, alpha):
    """Generate linear perturbation of uniform."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * (2.0 * i / (n - 1) - 1.0)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def gen_quadratic_perturb(n, alpha):
    """Generate quadratic perturbation (U-shape or V-shape)."""
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / (n - 1) - 1.0
    a = 1.0 + alpha * x * x
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def gen_cos_perturb(n, alpha, freq=1):
    """Generate cosine perturbation of uniform."""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)
