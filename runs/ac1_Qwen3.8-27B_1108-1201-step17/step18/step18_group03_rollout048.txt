def parametric_init(n, best_init=None):
    """Try various parametric shapes and return the best one."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))
    
    # Gaussian
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2 * sigma ** 2))))
    
    # Bimodal
    for sigma in [0.05, 0.1, 0.15]:
        candidates.append(norm_sum1(np.exp(-((x - 0.2) ** 2) / (2 * sigma ** 2)) + 
                                    np.exp(-((x - 0.8) ** 2) / (2 * sigma ** 2))))
    
    # Sine
    for freq in [1, 2, 3]:
        candidates.append(norm_sum1(np.sin(np.pi * freq * x) ** 2))
    
    # Trapped power (U-shaped)
    for p in [1.0, 2.0, 3.0]:
        candidates.append(norm_sum1((x * (1 - x)) ** (-p) if p > 0 else np.ones(n)))
    
    # Blend with best_init if available
    if best_init is not None:
        bi = norm_sum1(best_init)
        for w in [0.3, 0.5, 0.7]:
            for c in candidates[:5]:
                candidates.append(norm_sum1(w * bi + (1 - w) * c))
    
    # Evaluate all
    best_a = None
    best_m = float('inf')
    for a in candidates:
        ac = fast_autocorr(a)
        m = ac.max()
        if m < best_m:
            best_m = m
            best_a = a
    
    return best_a, best_m
