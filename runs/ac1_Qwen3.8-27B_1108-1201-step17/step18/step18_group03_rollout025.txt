def parametric_init(n, t_limit, seed=0):

    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:

        candidates.append((n - i_arr) ** p)

    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta_v in [0.5, 1.0, 1.5, 2.0, 3.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta_v)

    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:

            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    
    # Bimodal
    for c1 in [0.15, 0.2, 0.3]:
        for c2 in [0.7, 0.8, 0.85]:
            for sigma in [0.05, 0.08, 0.12, 0.15, 0.2]:
                candidates.append(np.exp(-((x - c1) ** 2) / (2 * sigma * sigma)) +
                                  np.exp(-((x - c2) ** 2) / (2 * sigma * sigma)))
    
    # Sine-based
    for freq in [1, 2, 3, 4, 5]:
        for ph in [0.0, 0.25, 0.5, 0.75]:
            candidates.append(np.maximum(np.sin(np.pi * freq * x + ph * np.pi), 0.0) ** 2)
            candidates.append(np.maximum(np.sin(np.pi * freq * x + ph * np.pi), 0.0))
    
    # Exponential
    for rate in [1, 2, 3, 5, 10, 20]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1)))
    
    # Hyperbolic secant
    for c in [0.0, 0.1, 0.2, 0.5]:
        for s in [2, 5, 10, 20]:

    for alpha in [0.618, 0.382, 0.754, 0.246, 0.5]:  # golden ratio related
        for gamma in [1, 2, 5, 10]:
            frac = (i_arr * alpha) % 1.0
            candidates.append(np.exp(-gamma * np.minimum(frac, 1.0 - frac)))

def fista_softmax(a0, t_limit, tau_init=0.001, tau_min=1e-7, verbose=True):
    """
    FISTA with adaptive soft-max temperature.
    Minimizes soft_max(ac/tau) as a smooth approximation of max(ac).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    tau = tau_init
    stall = 0
    max_stall = 500
    lr_base = 1.0
    
    # Track the top lags for efficient gradient computation
    top_k_count = min(100, 2 * n - 1)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > t_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation of y
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Check improvement
        if M < best_m - 1e-15:
            best_m = M
            best_a = x.copy()
            stall = 0
        
        # Adaptive temperature: decrease when making progress
        if stall == 0:
            tau = max(tau * 0.99, tau_min)
        else:
            stall += 1
            if stall > max_stall:
                # Restart with perturbation
                rng = np.random.default_rng(t)
                pert = rng.normal(0, 0.05, n) * best_a
                x = norm_sum1(best_a + pert)
                x_prev = x.copy()
                tau = max(tau * 1.5, tau_init * 0.1)
                stall = 0
                continue
        
        # Compute soft-max gradient
        # Soft-max weights
        ac_scaled = ac / tau
        ac_scaled -= ac_scaled.max()  # numerical stability
        w = np.exp(ac_scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(ac)) / len(ac)
        else:
            w = w / w_sum
        
        # Focus on top lags for efficiency
        top_idx = np.argsort(ac)[::-1][:top_k_count]
        w_top = w[top_idx]
        w_top_sum = w_top.sum()
        if w_top_sum < 1e-30:
            w_top = np.ones(top_k_count) / top_k_count
        else:
            w_top = w_top / w_top_sum
        
        # Compute gradient: grad[j] = sum_k w[k] * (y[k-j] + y[j-k]) for valid indices
        grad = np.zeros(n)
        for wi, k in zip(w_top, top_idx):
            if wi < 1e-10:
                continue
            k_int = int(k)
            # conv(a,a)[k] = sum_i a[i]*a[k-i]
            # d/d a[j] = a[k-j] if 0<=k-j<n (from first factor)
            #            + a[j-k] if 0<=j-k<n (from second factor)
            # Combined: 2*a[k-j] when j != k/2
            # For efficiency, just add contributions:
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                # For pair (i, k-i), the gradient w.r.t. a[i] gets w*y[k-i]
                # and w.r.t. a[k-i] gets w*y[i]
                # But since i and k-i are the two indices in the pair:
                # grad[i] += w * y[k-i]
                # grad[k-i] += w * y[i]
                # But a[i]*a[k-i] contributes to both a[i] and a[k-i]:
                # d(a[i]*a[k-i])/da[i] = a[k-i]
                # d(a[i]*a[k-i])/da[k-i] = a[i]
                # So: grad[idx] += w * y[k_int - idx]
                #     grad[k_int - idx] += w * y[idx]
                # But k_int - idx might overlap with idx, in which case we'd double count.
                # Actually, the correct formula is:
                # grad[j] += 2*w*y[k-j] for j where 0<=k-j<n and j!=k/2
                # grad[j] += w*y[j] for j = k/2 (if k is even)
                # Simplification: grad[j] += 2*w*y[k-j] for all valid j
                # (the factor of 2 is correct for j != k/2, and for j = k/2 it should be 1,
                #  but this is a small error that's acceptable for optimization)
                grad[idx] += 2.0 * wi * y[k_int - idx]
        
        # Project gradient to the simplex direction
        # Mirror descent step (exponential family)
        lr = lr_base / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"  FISTA-SoftMax done ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2e}")
    return best_a, best_m

def targeted_subgrad(a0, t_limit, top_k=50, verbose=True):
    """
    Targeted subgradient descent: focus on reducing the worst lags.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > t_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.95
                if lr < 0.001:
                    break
        
        # Get top lags
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_ac = ac[top_idx]
        
        # Weight by how much above average
        ac_mean = ac.mean()
        excess = np.maximum(top_ac - ac_mean, 0)
        w = excess / (excess.sum() + 1e-30)
        
        # Compute gradient from top lags
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-10:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += wi * a[k_int - idx]
        
        # Step: reduce mass at high-gradient positions
        # Move mass from high-grad to low-grad positions
        grad_mean = grad.mean()
        high = grad > grad_mean * 2
        low = grad < grad_mean * 0.5
        
        if high.sum() > 0 and low.sum() > 0:
            # Move a small fraction of mass
            move_frac = lr * 0.1
            total_high = a[high].sum()
            total_low = a[low].sum()
            if total_high > 1e-10:
                amt = total_high * move_frac
                # Distribute to low positions proportional to their current value
                low_masked = a[low].copy()
                low_masked[low] = np.maximum(a[low], 1e-10)
                target = low_masked / low_masked.sum()
                a[high] -= amt * (a[high] / total_high)
                a[low] += amt * target
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"  TargetedSubGrad done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    tau = fast_autocorr(a).max() * 0.1
    tau_min = best_m * 0.001
    stall = 0
    max_stall = 300
    lr_base = 2.0
    top_count = min(80, 2 * n - 1)
    best_tau = tau

    for t in range(1, 1000000):
        if time.monotonic() - t0 > t_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = x.copy()
            stall = 0
            tau = max(tau * 0.95, tau_min)
            best_tau = tau
        else:
            stall += 1
            if stall > max_stall:
                rng = np.random.default_rng(t * 7 + 3)
                noise_scale = 0.1 * np.exp(-stall / max_stall)
                pert = rng.normal(0, noise_scale, n) * best_a
                x = norm_sum1(best_a + pert)
                x_prev = x.copy()
                tau = max(tau * 2.0, tau_min * 10)
                stall = 0
                continue

        # Soft-max weights
        ac_shifted = (ac - ac.max()) / tau
        w = np.exp(ac_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(ac)) / len(ac)
        else:
            w = w / w_sum

        # Focus on top lags
        top_idx = np.argsort(ac)[::-1][:top_count]
        w_top = w[top_idx].copy()
        w_top_sum = w_top.sum()
        if w_top_sum < 1e-30:
            w_top = np.ones(top_count) / top_count
        else:
            w_top = w_top / w_top_sum

        grad = np.zeros(n)
        for i in range(top_count):
            if w_top[i] < 1e-12:
                continue
            k_int = int(top_idx[i])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += w_top[i] * y[k_int - idx]

        lr = lr_base / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={t} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2e}, stall={stall}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"  FISTA done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, t_limit, solver, initial_K=300, max_K=2500, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    failed = 0
    iter_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > t_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        prob = cp.Problem(cp.Minimize(M_var), [cp.sum(g_var) == 1, coeffs @ g_var <= M_var])

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed += 1
            if failed > 5:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yn = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_yn > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_yn[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, |C|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"  LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def targeted_cd(a0, t_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 500

    for t in range(1, 10000000):
        if time.monotonic() - t0 > t_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break

        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:15]:
            if time.monotonic() - t0 > t_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            n_pairs = min(6, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_indices:
                if time.monotonic() - t0 > t_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score destination positions
                low_region = np.zeros(n, dtype=bool)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    low_region[s2:e2 + 1] = True

                cand_mask = ~low_region
                if cand_mask.sum() < 20:
                    cand_mask = np.ones(n, dtype=bool)
                cand_mask[src] = False

                # Find lowest-contributing destination
                dst_scores = np.zeros(n)
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        dst_scores[idx2] += a[k2i - idx2] * ac[k2i]
                        dst_scores[k2i - idx2] += a[idx2] * ac[k2i]

                dst_scores[~cand_mask] = np.inf
                dst = np.argmin(dst_scores)

                if dst == src or a[dst] > 0.8 * src_val:
                    continue

                found = False
                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if not found:
                    break

        if not improved and stall % 50 == 0 and stall < max_stall:
            n_moves = int(rng.integers(2, 10))
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.01, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"  TargetedCD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_gradient(a0, t_limit, verbose=True):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    fft_size = 1 << (2 * n - 2).bit_length()
    lr = 1.0
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > t_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P[P > 1e-30].mean() if (P > 1e-30).any() else 1e-10

        beta_spec = 3.0 + min(t * 0.01, 10.0)
        P_dev = np.abs(P - P_mean) / (P_mean + 1e-30)
        w_spec = np.exp(beta_spec * P_dev)
        w_spec = w_spec / w_spec.sum()

        # Gradient: direction that reduces spectral peaks
        grad_freq = A * (P - beta_spec * P_mean * w_spec) / (P_mean + 1e-30)
        grad_time = np.fft.irfft(grad_freq, fft_size)[:n]

        step_lr = lr / (1.0 + 0.1 * np.sqrt(t))
        a_new = a * np.exp(-step_lr * grad_time)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 1000:
                a = norm_sum1(best_a + rng.normal(0, 0.02, n) * best_a)
                stall = 0
                lr *= 0.8

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"  Spectral done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = eval_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        best_seq, _ = parametric_init(n_base, 30.0)
        best_score = eval_fast(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric initialization ===
    print(f"\n=== Phase 1: Parametric init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 5:
        a_param, _ = parametric_init(n0, t1, seed=1)
        check_update(a_param, "parametric")

    # === Phase 2: FISTA with adaptive soft-max ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: FISTA adaptive (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_adaptive(a_in, t2, verbose=True)
        check_update(a_out, f"fista_n{n_best}")

    # === Phase 3: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, initial_K=400, max_K=3000, verbose=True)
        check_update(a_out, f"lp_n{n_best}")

    # === Phase 4: Targeted coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Targeted CD (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = targeted_cd(a_in, t4, verbose=True)
        check_update(a_out, f"cd_n{n_best}")

    # === Phase 5: Spectral gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Spectral (n={n_best}) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_gradient(a_in, t5, verbose=False)
        check_update(a_out, f"spec_n{n_best}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    length_candidates = []
    for delta in [-1000, -500, 500, 1000, 1500, -1500, 2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000:
            length_candidates.append(new_n)
    for rn in [3000, 4000, 5000, 5500, 6000]:
        if rn != n_best and 2000 <= rn <= 7000:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interp_seq(best_seq, new_n)

        # Quick FISTA on new length
        t6a = min(remaining * 0.3, 100.0)
        if t6a > 10:
            a_out, _ = fista_adaptive(a_long, t6a, verbose=False)
            check_update(a_out, f"fista_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 5500:
            t6b = min(remaining * 0.4, 120.0)
            if t6b > 20:
                a_in2 = norm_sum1(best_seq)
                a_out2, _ = lp_frank_wolfe(a_in2, t6b, solver, initial_K=300,
                                            max_K=2000, verbose=False)
                check_update(a_out2, f"lp_n{new_n}")

    # === Phase 7: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 50 and n_best <= 6000:
        t7 = min(remaining * 0.35, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=500,
                                       max_K=3500, verbose=True)
            check_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final FISTA polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final FISTA (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.40, 150.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = fista_adaptive(a_in, t8, verbose=False)
            check_update(a_out, f"fista_final_n{n_best}")

    # === Phase 9: Final targeted CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final CD (n={n_best}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.50, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = targeted_cd(a_in, t9, verbose=False)
            check_update(a_out, f"cd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
