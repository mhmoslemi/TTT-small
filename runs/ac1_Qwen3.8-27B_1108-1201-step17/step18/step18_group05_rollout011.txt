"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-based Frank-Wolfe with adaptive active set (min-t formulation)
- Peak-equalization coordinate moves exploiting convolution structure
- Multi-temperature smoothed projected gradient
- AMD (accelerated mirror descent) multi-start
- Length exploration and final coordinate polish
"""

import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate as sp_correlate

_solver = None

def detect_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s, verbose=False)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.SCS
    print("  LP solver fallback: SCS")
    return _solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_val(a):
    return 2.0 * len(a) * fast_autocorr(a).max()

def eval_list(a_arr):
    try:
        lst = [float(max(0.0, min(1000.0, x))) for x in np.asarray(a_arr, dtype=np.float64)]
        sc = evaluate_sequence(lst)
        return float(sc) if np.isfinite(sc) else float('inf')
    except Exception:
        return float('inf')

def ns1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones_like(a) / len(a)

def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    rho_arr = np.nonzero(u * np.arange(1, n + 1) > (csv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (csv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def init_u(n, sharp=2.0):
    x = np.linspace(-1, 1, n)
    return ns1(1.0 + sharp * x ** 2)

def init_bimodal(n, w=0.15):
    x = np.linspace(0, 1, n)
    return ns1(np.exp(-((x - 0.15) / w) ** 2) + np.exp(-((x - 0.85) / w) ** 2) + 0.005)

def init_cos(n):
    x = np.linspace(0, 1, n)
    return ns1((0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2)

def init_tapered(n, p=2.0):

    a = np.maximum(np.sin(np.pi * x) ** p, 1e-10)
    return ns1(a)

def init_spectral(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, min(30, n // 4) + 1):
        a += rng.normal(0, 0.12 / np.sqrt(f)) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    return ns1(a)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fi = np.maximum(np.interp(nx, ox, fn), 0.0)
    ss = fi.sum()
    return fi / ss if ss > 1e-15 else np.ones(new_n) / new_n

def soft_subgrad(a, ac, top_k=60, beta=5.0):
    n = len(a)
    kk = min(top_k, len(ac))
    ti = np.argsort(ac)[::-1][:kk]
    tv = np.maximum(ac[ti], 1e-30)
    w = tv ** beta
    w /= max(w.sum(), 1e-30)
    grad = np.zeros(n)
    for wi, k in zip(w, ti):
        ki = int(k)
        lo, hi = max(0, ki - n + 1), min(n, ki + 1)
        idx = np.arange(lo, hi)
        grad[idx] += 2.0 * wi * a[ki - idx]
    return grad

def amd_optimize(a0, t_lim, top_k=60, beta=5.0, c0=5.0, restart_ev=4000, seed=42):
    n = len(a0)
    a = ns1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x, x_prev = a.copy(), a.copy()
    gamma, stall, cur_k, cur_b = c0, 0, top_k, beta
    for t in range(1, 50000000):
        if time.monotonic() - t0 > t_lim:
            break
        bt = (t - 1.0) / (t + 1.5)
        y = np.maximum(x + bt * (x - x_prev), 1e-15)
        sy = y.sum()
        y = y / sy if sy > 1e-15 else x.copy()
        ac = fast_autocorr(y)
        g = soft_subgrad(y, ac, cur_k, cur_b)
        eta = gamma / (np.sqrt(t) + 1.0)
        la = np.log(y) - eta * g
        a_new = np.exp(la - la.max())
        a_new = np.maximum(a_new, 1e-30)
        a_new /= max(a_new.sum(), 1e-15)
        x_prev, x = x, a_new
        M = fast_autocorr(x).max()
        if M < best_m - 1e-15:
            best_m, best_a, stall = M, x.copy(), 0
        else:
            stall += 1
        if stall > restart_ev // 2:
            cur_b = max(cur_b * 0.9, 2.0)
            cur_k = min(cur_k + 5, 150)
        if stall > restart_ev:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            xr = np.maximum(best_a + pert, 1e-10)
            xr /= xr.sum()
            x, x_prev = xr, xr.copy()
            cur_k = min(cur_k + 10, 200)
            cur_b = max(cur_b - 1.0, 2.0)
    return best_a, best_m

def lp_fw(a0, t_lim, solver, verbose=False):
    n = len(a0)
    a = ns1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K_init = min(300, 2 * n - 1)
    K_max = min(2 * n - 1, 2500)
    K = K_init
    it = 0
    last_improve = 0

    while time.monotonic() - t0 < t_lim:
        ac = fast_autocorr(a)
        M = ac.max()

        # Adaptive active set: peaks above fraction of max
        frac = 0.80 + 0.15 * min(1.0, it / 300.0)
        thresh = frac * M
        active = np.where(ac >= thresh)[0]
        if len(active) < 30:
            active = np.argsort(ac)[::-1][:max(30, K)]
        active = active[np.argsort(ac[active])[::-1]][:min(K, K_max)]
        Kc = len(active)

        T = np.zeros((Kc, n))
        for ii, k in enumerate(active):
            ki = int(k)
            lo, hi = max(0, ki - n + 1), min(n, ki + 1)
            T[ii, lo:hi] = a[ki - np.arange(lo, hi)]

        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [
            T @ b_var <= t_var,
            cp.sum(b_var) == 1,
            t_var <= M * 1.001
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, K_max)
            it += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            it += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        bs = b_val.sum()
        if bs < 1e-15:
            it += 1
            continue
        b_dir = b_val / bs

        # Multi-alpha line search
        base = M
        best_alpha = 2.0 / (it + 2.0)
        best_trial = base
        alphas = np.unique(np.clip(np.array([best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                                              0.3, 0.2, 0.1, 0.05]), 0.005, 1.0))
        for alpha in alphas:
            at = np.maximum((1 - alpha) * a + alpha * b_dir, 0.0)
            st = at.sum()
            if st < 1e-15:
                continue
            at /= st
            mt = fast_autocorr(at).max()
            if mt < best_trial:
                best_trial, best_alpha = mt, alpha

        a = np.maximum((1 - best_alpha) * a + best_alpha * b_dir, 0.0)
        a /= a.sum()
        mn = fast_autocorr(a).max()
        if mn < best_m:
            best_m, best_a = mn, a.copy()
            last_improve = it

        it += 1
        if it % 10 == 0:
            K = min(K + 20, K_max)

        if verbose and it % 25 == 0:
            el = time.monotonic() - t0
            print(f"    LP-FW {it} ({el:.0f}s): J={2*n*best_m:.6f} K={Kc} alpha={best_alpha:.4f}")

    return best_a, best_m

def smooth_grad(a0, t_lim, tau0=15.0, tau1=200.0, eta0=0.5, seed=100, verbose=False):
    n = len(a0)
    a = ns1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    eta, stall = eta0, 0

    while time.monotonic() - t0 < t_lim and stall < 500:
        el = time.monotonic() - t0
        prog = min(1.0, el / min(t_lim, 600.0))
        tau = tau0 * (tau1 / tau0) ** prog

        ac = fast_autocorr(a)
        ac_s = tau * (ac - ac.max())
        p = np.exp(ac_s)
        p /= p.sum()

        pc = np.concatenate([[0.0], np.cumsum(p)])
        klo = np.arange(n)
        khi = np.minimum(klo + n, 2 * n - 1)
        S = pc[khi] - pc[klo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g -= g.mean()

        a_new = proj_simplex(a - eta * g)
        mn = fast_autocorr(a_new).max()
        if mn < best_m - 1e-15:
            best_m, best_a, stall = mn, a_new.copy(), 0
            eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.7
            if stall > 200:
                pert = rng.normal(0, 0.005, n) * a
                ap = np.maximum(a + pert, 0.0)
                ap /= ap.sum()
                mp = fast_autocorr(ap).max()
                if mp < best_m:
                    a, best_m, best_a = ap, mp, ap.copy()
                    stall, eta = 0, eta0 * 0.5
                else:
                    a = a_new
            else:
                a = a_new

        if verbose and (it := 0) and True:
            pass

    return best_a, best_m

def peak_eq_step(a0, t_lim, seed=200, verbose=False):

        M = ac.max()

        # Top peaks
        npk = min(30, len(ac))
        peak_idx = np.argsort(ac)[::-1][:npk]

        # Sensitivity: for each position, sum of a[peak - i] over top peaks
        sens = np.zeros(n)
        for k in peak_idx:
            ki = int(k)
            lo, hi = max(0, ki - n + 1), min(n, ki + 1)
            idx = np.arange(lo, hi)
            sens[idx] += a[ki - idx]

        improved = False
        # Try reducing the single largest peak
        k_max = int(peak_idx[0])
        lo, hi = max(0, k_max - n + 1), min(n, k_max + 1)
        idx = np.arange(lo, hi)
        contrib = a[idx] * a[k_max - idx]
        if contrib.max() < 1e-20:
            stall += 1
            continue

        src = idx[np.argmax(contrib)]
        src_val = a[src]
        if src_val < 1e-12:
            stall += 1
            continue

        # Find low-sensitivity destination
        in_peak = np.zeros(n, dtype=bool)
        for k in peak_idx[:15]:
            ki = int(k)
            in_peak[max(0, ki - n + 1):min(n, ki + 1)] = True
        candidates = np.where(~in_peak)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)

        top_cands = candidates[np.argsort(sens[candidates])[:5]]

        found = False
        for dst in top_cands:
            if dst == src or found:
                continue
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                at = a.copy()
                at[src] -= amt
                at[dst] += amt
                at = np.maximum(at, 0.0)
                st = at.sum()
                if st < 1e-15:
                    continue
                at /= st
                mt = fast_autocorr(at).max()
                if mt < best_m - 1e-16:
                    a, best_m, best_a = at, mt, at.copy()
                    improved, found = True, True
                    break

        if not found:
            # Random perturbation
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                at = a.copy()
                at[i] -= amt
                at[j] += amt
                at = np.maximum(at, 0.0)
                at /= at.sum()
                mt = fast_autocorr(at).max()
                if mt < best_m:
                    a, best_m, best_a = at, mt, at.copy()
                    improved = True

        if not improved:
            stall += 1

        if verbose and stall % 50 == 0:
            el = time.monotonic() - t0
            print(f"    PeakEq ({el:.0f}s): J={2*n*best_m:.6f} stall={stall}")

    return best_a, best_m

def coord_polish(a0, t_lim, seed=456, verbose=False):
    n = len(a0)
    a = ns1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall, n_top = 0, 25

    while time.monotonic() - t0 < t_lim and stall < 120:
        ac = fast_autocorr(a)
        M = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:12]:
            ki = int(k)
            lo, hi = max(0, ki - n + 1), min(n, ki + 1)
            idx = np.arange(lo, hi)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > t_lim:
                break
            ki = int(k)
            lo, hi = max(0, ki - n + 1), min(n, ki + 1)
            idx = np.arange(lo, hi)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            sv = a[src]
            if sv < 1e-12:
                continue
            cands = np.where(~in_peak)[0] if np.any(~in_peak) else np.arange(n)
            tc = cands[np.argsort(sens[cands])[:3]]
            for dst in tc:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.1]:
                    amt = sv * frac
                    if amt < 1e-15:
                        continue
                    at = a.copy()
                    at[src] -= amt
                    at[dst] += amt
                    at = np.maximum(at, 0.0)
                    at /= at.sum()
                    mt = fast_autocorr(at).max()
                    if mt < best_m - 1e-16:
                        a, best_m, best_a = at, mt, at.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.002, 0.01)
                at = a.copy()
                at[i] -= amt
                at[j] += amt
                at = np.maximum(at, 0.0)
                at /= at.sum()
                mt = fast_autocorr(at).max()
                if mt < best_m:
                    a, best_m, best_a = at, mt, at.copy()
                    stall = 0
        else:
            stall = 0

        if verbose and stall % 40 == 0:
            el = time.monotonic() - t0
            print(f"    Coord ({el:.0f}s): J={2*n*best_m:.6f} stall={stall}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    tg = time.monotonic()
    budget = min(float(budget_s) - 30.0, 1770.0)

    def remaining():
        return budget - (time.monotonic() - tg)

    solver = detect_solver()
    print(f"Total budget: {budget:.0f}s")

    best_a = None
    best_score = float('inf')

    def try_update(arr, label):
        nonlocal best_a, best_score
        if arr is None:
            return
        ar = np.maximum(np.asarray(arr, dtype=np.float64), 0.0)
        s = ar.sum()
        if s < 0.01:
            return
        ar = ar / s
        sc = eval_list(ar)
        if np.isfinite(sc) and sc < best_score:
            best_score, best_a = sc, ar.copy()
            el = time.monotonic() - tg
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(ar)} t={el:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            ps = eval_list(parent)
            print(f"Parent: n={len(parent)} score={ps:.6f}")
            best_a = ns1(parent)
            best_score = ps
    except Exception:
        pass

    if best_a is None:
        n_default = 5200
        best_a = init_tapered(n_default, p=2.0)
        best_score = eval_list(best_a)
        print(f"Default init: n={len(best_a)} score={best_score:.6f}")

    n0 = len(best_a)

    # === Phase 1: Multi-start AMD (150s) ===
    rem = remaining()
    t1 = min(rem * 0.12, 150.0)
    if t1 > 50:
        print(f"\n=== Phase 1: Multi-start AMD (n={n0}, {t1:.0f}s) ===")
        inits, labels = [], []
        if parent is not None:
            inits.append(ns1(parent)); labels.append("parent")
        inits.append(init_u(n0, 2.0)); labels.append("u2")
        inits.append(init_u(n0, 4.0)); labels.append("u4")
        inits.append(init_bimodal(n0, 0.2)); labels.append("bimod")
        inits.append(init_cos(n0)); labels.append("cos")
        inits.append(init_spectral(n0, seed=7)); labels.append("spec")
        inits.append(init_tapered(n0, 3.0)); labels.append("taper3")

        seen = set()
        uinits, ulabels = [], []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                uinits.append(init); ulabels.append(lbl)
        inits, labels = uinits, ulabels

        t_per = max(10.0, t1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            if remaining() < 50:
                break
            ta = min(t_per, remaining() - 40)
            if ta < 8:
                break
            print(f"  AMD '{lbl}' ({ta:.0f}s)...")
            ao, _ = amd_optimize(init, ta, top_k=60, beta=6.0, c0=5.0,
                                  restart_ev=3000, seed=42 + idx)
            try_update(ao, f"amd_{lbl}")

    # === Phase 2: LP-FW round 1 (600s) ===
    rem = remaining()
    t2 = min(rem * 0.38, 650.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-FW r1 (n={len(best_a)}, {t2:.0f}s) ===")
        ao, _ = lp_fw(ns1(best_a), t2, solver, verbose=True)
        try_update(ao, "lp_r1")

    # === Phase 3: Peak-equalization (200s) ===
    rem = remaining()
    t3 = min(rem * 0.12, 200.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Peak-eq (n={len(best_a)}, {t3:.0f}s) ===")
        ao, _ = peak_eq_step(ns1(best_a), t3, seed=200, verbose=True)
        try_update(ao, "peakeq")

    # === Phase 4: Smoothed gradient (200s) ===
    rem = remaining()
    t4 = min(rem * 0.12, 200.0)
    if t4 > 30:
        print(f"\n=== Phase 4: SmoothGrad (n={len(best_a)}, {t4:.0f}s) ===")
        ao, _ = smooth_grad(ns1(best_a), t4, tau0=15.0, tau1=250.0,
                             eta0=0.6, seed=100, verbose=False)
        try_update(ao, "smgrad")

    # === Phase 5: LP-FW round 2 (400s) ===
    rem = remaining()
    t5 = min(rem * 0.28, 400.0)
    if t5 > 40 and len(best_a) <= 6000:
        print(f"\n=== Phase 5: LP-FW r2 (n={len(best_a)}, {t5:.0f}s) ===")
        ao, _ = lp_fw(ns1(best_a), t5, solver, verbose=True)
        try_update(ao, "lp_r2")

    # === Phase 6: Coord polish + length exploration (remaining) ===
    rem = remaining()
    t6 = min(rem * 0.35, 150.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Coord polish (n={len(best_a)}, {t6:.0f}s) ===")
        ao, _ = coord_polish(ns1(best_a), t6, seed=456, verbose=False)
        try_update(ao, "coord")

    # Quick length exploration
    rem = remaining()
    nb = len(best_a)
    for new_n in [nb + 1000, nb - 1000]:
        if rem < 80 or not (800 <= new_n <= 8000) or new_n == nb:
            continue
        print(f"\n  Trying n={new_n}...")
        al = interpolate_seq(best_a, new_n)
        ts = min(rem * 0.35, 120.0)
        if ts > 20:
            ao, _ = amd_optimize(al, ts, top_k=60, beta=6.0, c0=4.0,
                                  restart_ev=4000, seed=new_n % 1000)
            try_update(ao, f"amd_n{new_n}")
        rem = remaining()
        if rem > 60 and new_n <= 5000:
            tl = min(rem * 0.4, 100.0)
            if tl > 20:
                ao, _ = lp_fw(ns1(interpolate_seq(best_a, new_n)), tl, solver, verbose=False)
                try_update(ao, f"lp_n{new_n}")

    # === Final polish ===
    rem = remaining()
    if rem > 30:
        print(f"\n=== Final polish (n={len(best_a)}, {rem:.0f}s) ===")
        tf1 = min(rem * 0.5, 80.0)
        if tf1 > 15:
            ao, _ = peak_eq_step(ns1(best_a), tf1, seed=789, verbose=False)
            try_update(ao, "final_peakeq")
        rem = remaining()
        if rem > 15 and len(best_a) <= 5000:
            tf2 = min(rem * 0.6, 60.0)
            if tf2 > 10:
                ao, _ = lp_fw(ns1(best_a), tf2, solver, verbose=False)
                try_update(ao, "final_lp")

    # Final validation
    if best_a is not None:
        fs = eval_list(best_a)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f} n={len(best_a)} t={time.monotonic()-tg:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        n_fb = 4000
        best_a = init_tapered(n_fb, 2.0)
        best_score = eval_list(best_a)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_a]
