def fast_autocorr(a):
    """Compute autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_T_matrix(binding_idx, a, n):
    """Build the constraint matrix T where T[i,j] = a[k_i - j] for constraint i."""
    K = len(binding_idx)
    T = np.zeros((K, n))
    for ii, k in enumerate(binding_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            T[ii, j] = a[ki - j]
    return T

    gamma = c0
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * (t - 1)**2)) / 2.0
        beta_t = min(beta_t / t, 2.0)  # Cap for stability
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute subgradient
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = float(fast_autocorr(x).max())
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
    
    return best_a, best_m


def pair_swap_cd(a0, time_limit, verbose=False, rng_seed=123):
    """Pair-swap coordinate descent: move mass from one position to another."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    n_top = 30
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = float(ac.max())
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity: how much each position contributes to peaks
        sens = np.zeros(n)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]
        
        improved = False
        
        # Try to reduce the peak by moving mass from high-sensitivity to low-sensitivity positions
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Find the most contributing source position
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find low-sensitivity destinations
            low_sens = np.where(sens < np.percentile(sens[sens > 0], 25))[0]
            if len(low_sens) == 0:
                low_sens = np.argsort(sens)[:n // 4]
            
            # Try a few destinations
            for dst in low_sens[:5]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    a_trial = a_trial / s_t
                    M_trial = float(fast_autocorr(a_trial).max())
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        # Random perturbations when stuck
        if not improved:
            stall += 1
            n_pert = max(1, n // 200)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = float(fast_autocorr(a).max())
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
    
    J = 2.0 * n * best_m
    if verbose:
        print(f"    PairSwap final: J={J:.6f}, stall={stall}")
    return best_a, best_m


def local_search_fine(a0, time_limit, verbose=False, rng_seed=456):
    """Very fine local search with tiny steps in random directions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    step_sizes = [0.001, 0.0005, 0.002, 0.0001]
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        # Decrease step size over time
        ss_idx = min(len(step_sizes) - 1, int(progress * len(step_sizes)))
        step = step_sizes[ss_idx]
        
        # Generate random direction on the simplex tangent
        d = rng.normal(0, 1, n)
        d -= d.sum() / n  # Make it tangent to the simplex
        d = d / max(np.abs(d).max(), 1e-15)
        
        for sign in [1, -1]:
            a_trial = a + sign * step * d
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = float(fast_autocorr(a_trial).max())
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                break
    
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LocalSearch final: J={J:.6f}")
    return best_a, best_m

    _global_solver = cp.SCS

    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(binding_idx, a, n):
    K = len(binding_idx)
    T = np.zeros((K, n))
    for ii in range(K):
        ki = int(binding_idx[ii])
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        js = np.arange(j_start, j_end)
        T[ii, js] = a[ki - js]
    return T


def lp_frank_wolfe(a0, time_limit, solver, verbose=False,
                   use_all=False, top_K_init=500, top_K_max=None):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iters = 0
    consecutive_fail = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = float(ac.max())
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T = build_constraint_matrix(binding_idx, a, n)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 100, top_K_max)
            consecutive_fail += 1
            if consecutive_fail > 20:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            consecutive_fail += 1
            if consecutive_fail > 20:
                break
            K = min(K + 100, top_K_max)
            continue

        consecutive_fail = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        best_alpha = 2.0 / (iters + 2.0)
        best_trial_m = M_curr

        alphas = list(set([best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                           best_alpha * 2.0, 0.1, 0.05, 0.5, 1.0]))
        alphas = [al for al in alphas if 0 < al <= 1.0]
        alphas.sort(reverse=True)

        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = float(fast_autocorr(a_trial).max())
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = float(fast_autocorr(a).max())
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iters += 1
        if iters % 5 == 0:
            K = min(K + 50, top_K_max)

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iters} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m, iters


def nesterov_mdd(a0, time_limit, top_k=100, beta=6.0, c0=4.0,
                 restart_every=3000, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        mu_t = (1.0 + np.sqrt(1.0 + 4.0 * (t - 1) * (t - 1))) / (2.0 * t)
        y = x + mu_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = float(fast_autocorr(x).max())
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_beta = max(cur_beta - 1.0, 3.0)
            cur_top_k = min(cur_top_k + 20, 300)

    return best_a, best_m


def pair_swap_cd(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 40

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:25]

        sens = np.zeros(n)
        for k in top_k[:12]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            thresh = np.percentile(sens, 30)
            low_sens = np.where(sens <= thresh)[0]
            low_sens = low_sens[low_sens != src]
            if len(low_sens) == 0:
                low_sens = np.argsort(sens)[:n // 5]

            for dst in low_sens[:8]:
                if dst == src:
                    continue
                for frac in [0.003, 0.005, 0.01, 0.02, 0.002]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = float(fast_autocorr(a_trial).max())
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            n_pert = max(2, n // 300)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = float(fast_autocorr(a).max())
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

    if verbose:
        J = 2.0 * n * best_m
        print(f"    PairSwap: J={J:.6f}, stall={stall}")
    return best_a, best_m


def fine_local_search(a0, time_limit, verbose=False, rng_seed=789):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    step_sizes = [0.001, 0.0005, 0.0002, 0.0001]

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        ss_idx = min(len(step_sizes) - 1, int(progress * len(step_sizes)))
        step = step_sizes[ss_idx]

        d = rng.normal(0, 1, n)
        d -= d.sum() / n
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            continue
        d = d / d_norm

        for sign in [1, -1]:
            a_trial = a + sign * step * d
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = float(fast_autocorr(a_trial).max())
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                break

    if verbose:
        J = 2.0 * n * best_m
        print(f"    LocalSearch: J={J:.6f}")
    return best_a, best_m


def try_update(a, label, t_global):
    global _best_state
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < _best_state['score']:
        _best_state['score'] = sc
        _best_state['seq'] = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
