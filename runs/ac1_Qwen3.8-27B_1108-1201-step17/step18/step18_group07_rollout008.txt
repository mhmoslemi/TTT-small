Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-softmax mirror descent (multiplicative weights) with FFT-based gradients
2. Annealed temperature for smooth -> sharp objective
3. Adaptive step size
4. Periodic spectral flattening
5. Multi-start with diverse initializations and lengths
6. Final coordinate descent polish
"""

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax(x, beta=1.0):
    x = np.asarray(x, dtype=np.float64) * beta
    x_max = x.max()
    if x_max > 700:
        x = x - x_max
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:
        return np.ones_like(x) / len(x)
    return ex / s

def compute_gradient(p, a):
    """Compute grad[i] = sum_k p[k] * a[k-i] via FFT.
    p: length 2n-1 (softmax weights over convolution)
    a: length n
    Returns: grad of length n
    """
    n = len(a)
    a_flipped = a[::-1]
    # conv(p, a_flipped) has length (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    AF = np.fft.rfft(a_flipped, N)
    conv = np.fft.irfft(P * AF, N)
    # grad[i] = conv[n-1+i] for i=0,...,n-1
    return conv[n-1:2*n-1].copy()

def mirror_descent(a0, time_limit, init_beta=5.0, final_beta=500.0, 
                   init_eta=0.05, verbose=False):
    """Full-softmax mirror descent with annealing."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    c = fast_conv(a)
    best_M = c.max()
    t0 = time.monotonic()
    
    N_conv = 1 << (2*n - 2).bit_length()
    N_grad = 1 << (3*n - 2).bit_length()
    
    iteration = 0
    total_time = time_limit
    
    while time.monotonic() - t0 < total_time:
        batch_size = 100  # do 100 iterations per time check
        batch_start = time.monotonic()
        
        for _ in range(batch_size):
            if time.monotonic() - t0 > total_time:
                break
            
            progress = min(1.0, (time.monotonic() - t0) / total_time)
            
            # Anneal temperature
            beta = init_beta * (final_beta / init_beta) ** progress
            # Anneal step size
            eta = init_eta * (0.01 / init_eta) ** progress
            
            # Compute convolution
            fft_size = N_conv
            A = np.fft.rfft(a, fft_size)
            c = np.fft.irfft(A * A, fft_size)[:2*n-1]
            
            M = c.max()
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            
            # Compute softmax weights
            p = softmax(c, beta)
            
            # Compute gradient
            a_flipped = a[::-1]
            P = np.fft.rfft(p, N_grad)
            AF = np.fft.rfft(a_flipped, N_grad)
            conv_pa = np.fft.irfft(P * AF, N_grad)
            grad = conv_pa[n-1:2*n-1]
            
            # Multiplicative weights update
            # a_new[i] = a[i] * exp(-eta * grad[i])
            # For numerical stability:
            log_update = -eta * grad
            log_update = np.maximum(log_update, -50.0)
            a_new = a * np.exp(log_update)
            
            # Ensure non-negative (should be automatic)
            a_new = np.maximum(a_new, 1e-30)
            
            # Normalize
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            a = a_new
        
        # Periodic spectral smoothing
        if (time.monotonic() - t0) % 2.0 < (time.monotonic() - batch_start):
            a = spectral_smooth(a, strength=0.1)
            c_check = fast_conv(a)
            if c_check.max() < best_M:

                best_a = a.copy()
        
        if verbose:

            print(f"    MD: J={2*n*best_M:.6f}, beta={beta:.1f}, eta={eta:.4f}, t={elapsed:.0f}s")

def spectral_smooth(a, strength=0.2):
    """Flatten the power spectrum."""
    n = len(a)
    fft_size = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    # Compress: mag_new = mag_mean * ratio^(1-strength)
    mag_new = mag_mean * np.maximum(ratio, 1e-30) ** (1.0 - strength)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    # Blend
    return norm_simplex((1 - strength * 0.5) * a + strength * 0.5 * a_new)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using full-softmax mirror descent
with FFT-accelerated gradients, annealed temperature, adaptive step size,
periodic spectral flattening, and multi-length search.
"""

import numpy as np
import time

# Global time tracking
_t_start = 0.0
_time_budget = 1770.0  # 1800 - 30s reserve

def remaining_time():
    return _time_budget - (time.monotonic() - _t_start)

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_value(a):
    """Compute score = 2n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def evaluate_safe(a):
    """Call the external evaluate_sequence safely."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).tolist()]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def mirror_descent(a0, time_limit, beta_start=5.0, beta_end=500.0,
                   eta_start=0.03, eta_end=0.0005, verbose=False):
    """Full-softmax mirror descent with annealing."""
    n = len(a0)
    a = norm_simplex(a0)
    c = fast_conv(a)
    best_M = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    # Pre-compute FFT sizes
    N_conv = 1 << (2 * n - 2).bit_length()
    N_grad = 1 << (3 * n - 2).bit_length()
    
    a_flipped_buf = np.empty(n)
    
    while time.monotonic() - t0 < time_limit:
        iteration = 0
        batch_end = time.monotonic() + 0.5  # 0.5s batches
        
        while time.monotonic() < batch_end:
            iteration += 1
            progress = min(1.0, (time.monotonic() - t0) / time_limit)
            
            # Anneal
            beta = beta_start * (beta_end / beta_start) ** progress
            eta = eta_start * (eta_end / eta_start) ** progress
            
            # Compute convolution
            A = np.fft.rfft(a, N_conv)
            c = np.fft.irfft(A * A, N_conv)[:2 * n - 1]
            
            M = c.max()
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            
            # Softmax weights
            c_scaled = beta * c
            c_max = c_scaled.max()
            exp_c = np.exp(c_scaled - c_max)
            p = exp_c / exp_c.sum()
            
            # Gradient via FFT: grad[i] = sum_k p[k] * a[k-i]
            a_flipped_buf[:] = a[::-1]
            P = np.fft.rfft(p, N_grad)
            AF = np.fft.rfft(a_flipped_buf, N_grad)
            conv_pa = np.fft.irfft(P * AF, N_grad)
            grad = conv_pa[n - 1:2 * n - 1]
            
            # Clamp gradient for numerical stability
            grad = np.maximum(grad, 0.0)
            grad_max = grad.max()
            if grad_max > 0:
                grad = grad / grad_max  # normalize
            
            # Multiplicative weights update
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-30)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
        
        # Spectral smoothing every ~0.5s
        if time.monotonic() - t0 < time_limit:
            a = spectral_smooth(a, strength=0.05)
            c_check = fast_conv(a)
            if c_check.max() < best_M:
                best_M = c_check.max()
                best_a = a.copy()
        
        if verbose and (time.monotonic() - t0) % 10.0 < 0.5:
            elapsed = time.monotonic() - t0
            print(f"    MD iter~{iteration}: J={2*n*best_M:.6f}, beta={beta:.1f}, eta={eta:.5f}, t={elapsed:.0f}s")
    
    return best_a, best_M

def spectral_smooth(a, strength=0.1):
    """One step of spectral flattening."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.maximum(ratio, 1e-30) ** (1.0 - strength)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    blend = 0.3 * strength
    return norm_simplex((1 - blend) * a + blend * a_new)

def make_init_flat(n):
    return np.ones(n) / n

def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)

def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return norm_simplex(a)

def make_init_dual_peak(n, pos1=0.3, pos2=0.7, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.1
    return norm_simplex(a)

def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)

def make_init_random_smooth(n, rng):
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_polish(a0, time_limit):
    """Local coordinate descent: small mass transfers."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    c = fast_conv(a)
    
    while time.monotonic() - t0 < time_limit:
        # Find top peaks
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try moving mass
            for attempt in range(3):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.02, 0.05]:
                    amt = src_val * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        continue
                    a_t /= s_t
                    M_t = fast_conv(a_t).max()
                    if M_t < best_M - 1e-15:
                        a = a_t
                        best_M = M_t
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    c = fast_conv(a)
                    break
            if improved:
                break
    
    return best_a, best_M
