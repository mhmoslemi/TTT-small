import numpy as np
import time
import math

def fast_conv(a):
    """Full convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def score_of(a):
    """Compute the objective value."""
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * b.max() / (a.sum() ** 2)

# Multiple initialization shapes
def init_ramp(n, p):
    x = np.arange(1, n+1, dtype=np.float64)
    return normalize(x ** p)

def init_sinusoid_mod(n, f, amp, phase=0, trend=0.3):
    x = np.arange(n, dtype=np.float64) / n
    a = (1.0 + trend * x) * (1.0 + amp * np.cos(2*np.pi*f*x + phase))
    return normalize(a)

def init_multi_sinusoid(n, freqs, amps):
    x = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        a += amp * np.cos(2*np.pi*f*x)
    return normalize(a)

def init_random(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    return normalize(a)

# Main optimization: proximal gradient with adaptive step
def optimize_grad(a0, time_limit, n_peaks=20, verbose=False, seed=42):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score_val = 2.0 * n * fast_conv(a).max()  # assuming sum=1
    ferr = time.monotonic()
    rng = np.random.default_rng(seed)
    
    t = 0
    step = 0.1
    sa_temp = 1.0
    sa_temp_min = 0.01
    stall = 0
    
    while time.monotonic() - ferr < time_limit:
        t += 1
        b = fast_conv(a)
        bmax = b.max()
        curr_val = 2.0 * n * bmax  # score for sum=1
        
        # Acceptance (simulated annealing)
        accept = False
        if curr_val < best_score_val - 1e-12:
            best_score_val = curr_val
            best_a = a.copy()
            stall = 0
            accept = True
        elif rng.random() < math.exp(-(curr_val - best_score_val) / sa_temp):
            accept = True  # accept worse (SA)
        
        if not accept:
            # Revert
            a = best_a.copy()
        
        # Compute subgradient: weighted sum of gradients for top peaks
        top_k = np.argsort(b)[::-1][:n_peaks]
        # Weights: exponential of peak values
        peak_vals = b[top_k]
        weights = peak_vals / peak_vals.sum()
        
        grad = np.zeros(n)
        for idx, k in enumerate(top_k):
            ki = int(k)
            wi = weights[idx]
            if wi < 1e-10:
                continue
            # db[ki]/da[j] = a[ki-j] + a[j] (for the two orderings in conv)
            # Actually for full conv: b[ki] = sum_i a[i]*a[ki-i]
            # db[ki]/da[j] = a[ki-j] (if ki-j in [0,n-1]) + a[j-ki+n-1] (if j-ki+n-1 in [0,n-1]... )
            # Wait, let me be more careful.
            # b[ki] = sum_{i: valid} a[i] * a[ki-i]
            # where valid means i in [0,n-1] and ki-i in [0,n-1]
            # = sum_{i=max(0,ki-(n-1))}^{min(n-1,ki)} a[i] * a[ki-i]
            # db[ki]/da[j] = a[ki-j] (if ki-j in [0,n-1]) + a[j] * d(a[ki-j])/da[j]... 
            # No, simply: b[ki] = sum_i a[i]*a[ki-i]
            # If we differentiate w.r.t. a[j]:
            # db[ki]/da[j] = (if j in valid range) a[ki-j] + (if ki-j in valid range) a[j]
            # Wait no: b[ki] = sum_i a[i] * a[ki-i]
            # The term a[j] appears twice: once as a[i] with i=j, and once as a[ki-i] with ki-i=j i.e. i=ki-j.
            # So db[ki]/da[j] = a[ki-j] (from the i=j term) + a[j] (from the i=ki-j term)
            # But only if the indices are valid.
            
            # From i=j term: need ki-j in [0, n-1], i.e., j in [ki-(n-1), ki]
            # From i=ki-j term: need j in [0, n-1] (always) AND ki-j in [0, n-1], i.e., j in [ki-(n-1), ki]
            # Wait, both conditions are the same: j in [max(0, ki-(n-1)), min(n-1, ki)]
            # And in that range: db[ki]/da[j] = a[ki-j] + a[j]
            # But if j = ki-j (i.e., j = ki/2), then it's just a[j] (same term counted once)
            # Actually no: in the sum b[ki] = sum_i a[i]*a[ki-i], the term a[j]*a[ki-j] appears once
            # (for i=j, it contributes a[j]*a[ki-j]; for i=ki-j, it contributes a[ki-j]*a[j] = same thing)
            # Wait, are i=j and i=ki-j the same index? Only if j = ki-j, i.e., j = ki/2.
            # If j != ki/2, then i=j and i=ki-j are different indices, and the sum contains both:
            #   a[j]*a[ki-j] (from i=j) + a[ki-j]*a[j] (from i=ki-j)
            # So db[ki]/da[j] = a[ki-j] + a[j] when j != ki/2
            # And db[ki]/da[j] = 2*a[j] when j = ki/2 (if ki is even)
            
            j_lo = max(0, ki - (n - 1))
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idxs = np.arange(j_lo, j_hi + 1)
            # For each j in idxs: contribution to grad is a[ki-j] + a[j] (or 2*a[j] if j==ki-j)
            # But we can just compute: a[ki - idxs] + a[idxs]
            # This correctly handles the j == ki-j case (gives 2*a[j])
            contrib = a[ki - idxs] + a[idxs]
            grad[idxs] += wi * contrib
        
        # Proximal gradient step in log-space (mirror map for non-negativity)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= step * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        
        # Adaptive step: decrease if oscillating
        if t % 100 == 0:
            step *= 0.99
            sa_temp = max(sa_temp_min, sa_temp * 0.995)
        
        if stall > 2000:
            # Random restart
            a = best_a + rng.normal(0, 0.05, n) * best_a
            a = normalize(a)
            stall = 0
            step = 0.1
        
        if verbose and t % 1000 == 0:
            print(f"  GRAD t={t}: best={best_score_val/(a.sum()**2):.6f} (n={n})")
    
    return best_a, best_score_val / (best_a.sum()**2)  # actual score

def optimize_spectral(a0, time_limit, verbose=False, seed=42):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = 2.0 * n * fast_conv(a).max()
    ferr = time.monotonic()
    rng = np.random.default_rng(seed)
    
    fft_size = 1 << (2*n - 1).bit_length()
    
    for it in range(100000):
        if time.monotonic() - ferr > time_limit:
            break
        
        b = fast_conv(a)
        bmax = b.max()
        curr_val = 2.0 * n * bmax
        
        if curr_val < best_val - 1e-12:
            best_val = curr_val
            best_a = a.copy()
        
        # FFT
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A)**2
        pmean = power.mean()
        
        # Adaptive target: suppress frequencies that contribute to peaks
        # The peak at lag k contributes most at frequency 2*pi*k/fft_size
        top_peaks = np.argsort(b)[::-1][:50]
        
        # Build a suppression mask
        freqs = np.arange(len(power)) / fft_size  # normalized frequency [0, 0.5]
        target = np.ones_like(power)
        for k in top_peaks[:20]:
            ki = int(k)
            # Frequency corresponding to this lag
            f_peak = ki / fft_size
            # Suppress near this frequency (and its mirror 1-f_peak)
            # Actually, the contribution to b[ki] from frequency f is |A(f)|^2 * exp(2*pi*i*f*ki/fft_size)
            # To reduce b[ki], we want to reduce |A(f)|^2 where the phase aligns
            # Simple heuristic: reduce all frequencies a bit, more at the peak frequency
            target *= (1.0 - 0.01 * np.exp(-((freqs - f_peak)**2 + (freqs - (1-f_peak))**2) * fft_size))
        
        target = np.maximum(target, 0.5)  # Don't suppress too much
        
        # Interpolate towards target
        alpha = 0.3  # step size
        power_new = (1 - alpha) * power + alpha * pmean * target
        power_new = np.maximum(power_new, 1e-30)
        
        # Update amplitude, keep phase
        phase = np.angle(A)
        mag_new = np.sqrt(power_new)
        mag_old = np.abs(A)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        # Check improvement
        b_new = fast_conv(a_new)
        if 2.0 * n * b_new.max() < curr_val:
            a = a_new
        else:
            # Blend
            a = 0.7 * a + 0.3 * a_new
            a = normalize(a)
    
    return best_a, best_val / (best_a.sum()**2)

import numpy as np
import time
import math


def fast_conv(a):
    """Full linear convolution via zero-padded FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to unit sum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_value(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Call the external evaluator."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def init_ramp(n, p=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    return normalize(x ** p)


def init_mod_sinusoid(n, f, amp, trend=0.3):
    x = np.arange(n, dtype=np.float64) / n
    a = (1.0 + trend * x) * (1.0 + amp * np.sin(2.0 * np.pi * f * x))
    return normalize(a)


def init_chirp(n, chirp_rate=1.0, trend=0.2):
    x = np.arange(n, dtype=np.float64) / n
    a = (1.0 + trend * x) * (1.0 + 0.3 * np.sin(2.0 * np.pi * (chirp_rate * x * x + x)))
    return normalize(a)


def init_multi_sin(n, freqs, amps, trend=0.2):
    x = np.arange(n, dtype=np.float64) / n
    a = 1.0 + trend * x
    for f, amp in zip(freqs, amps):
        a = a + amp * np.sin(2.0 * np.pi * f * x)
    return normalize(a)


def init_random(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    return normalize(a)


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_n = normalize(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_n)
    return normalize(a_new)


def compute_topk_grad(a, b, n_peaks=30):
    """Compute exact L-infinity top-k subgradient for the convolution."""
    n = len(a)
    top_k = np.argsort(b)[::-1][:min(n_peaks, len(b))]
    peak_vals = b[top_k].astype(np.float64)
    total = peak_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = peak_vals / total

    grad = np.zeros(n, dtype=np.float64)
    for idx in range(len(top_k)):
        ki = int(top_k[idx])
        wi = weights[idx]
        if wi < 1e-12:
            continue
        j_lo = max(0, ki - (n - 1))
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idxs = np.arange(j_lo, j_hi + 1)
        # d(conv)[ki]/da[j] = a[j] + a[ki-j] for j in valid range
        grad[idxs] += wi * (a[idxs] + a[ki - idxs])
    return grad


def prox_grad_optimize(a0, time_limit, n_peaks=30, verbose=False, seed=42,
                       sa_init=0.5, sa_min=0.01, sa_decay=0.999,
                       step_init=0.05, step_decay=0.9995,
                       restart_every=3000):
    """Proximal gradient with top-k L-infinity subgradient and SA acceptance."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    step = step_init
    sa_temp = sa_init
    stall = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_conv(a)
        bmax = b.max()
        curr = 2.0 * n * bmax

        # SA acceptance
        if curr < best_val - 1e-14:
            best_val = curr
            best_a = a.copy()
            stall = 0
        elif rng.random() < math.exp(-(curr - best_val) / max(sa_temp, 1e-10)):
            stall = 0  # accept worse to escape
            a = a.copy()  # keep current (worse) state
            # continue from worse state
        else:
            a = best_a.copy()  # revert

        # Compute gradient
        grad = compute_topk_grad(a, b, n_peinks if False else n_peaks)

        # Proximal step in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= step * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new

        # Adaptive scheduling
        step = max(step * step_decay, 0.001)
        sa_temp = max(sa_temp * sa_decay, sa_min)

        # Restart
        stall += 1
        if stall > restart_every:
            stall = 0
            a = best_a + rng.normal(0, 0.03, n) * best_a
            a = normalize(a)
            step = step_init * 0.8
            sa_temp = max(sa_init * 0.5, 0.05)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PG t={t} ({elapsed:.0f}s): best={best_val/(best_a.sum()**2):.6f}, "
                  f"step={step:.5f}, temp={sa_temp:.4f}, stall={stall}")

    return best_a


def spectral_optimize(a0, time_limit, verbose=False, seed=42, n_suppress=20, alpha=0.25):
    """Spectral optimization with peak-based frequency suppression."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    freqs = np.fft.rfftfreq(fft_size)

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        bmax = b.max()
        curr = 2.0 * n * bmax

        if curr < best_val - 1e-14:
            best_val = curr
            best_a = a.copy()

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        pmean = power.mean()
        if pmean < 1e-30:
            break

        # Build suppression mask from top peaks
        top_peaks = np.argsort(b)[::-1][:n_suppress]
        target = np.ones_like(power)
        for k in top_peaks:
            ki = int(k)
            f_peak = ki / fft_size
            # Gaussian suppression around peak frequency and its mirror
            dist = np.minimum(np.abs(freqs - f_peak), np.abs(freqs - (1.0 - f_peak)))
            sigma = 2.0 / fft_size
            target *= (1.0 - 0.02 * np.exp(-dist ** 2 / (2 * sigma ** 2)))

        target = np.maximum(target, 0.7)

        # Interpolate power towards target
        power_new = (1.0 - alpha) * power + alpha * pmean * target
        power_new = np.maximum(power_new, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(power_new)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        # Check if improved
        b_new = fast_conv(a_new)
        if 2.0 * n * b_new.max() < curr:
            a = a_new
        else:
            # Damped update
            a = 0.6 * a + 0.4 * a_new
            a = normalize(a)

        # Slowly increase alpha
        alpha_eff = min(0.5, alpha + 0.0001 * it)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spec it={it} ({elapsed:.0f}s): best={best_val/(best_a.sum()**2):.6f}")

    return best_a


def peak_surgery(a0, time_limit, verbose=False, seed=42, n_top=15, n_fracs=5):
    """Directly reduce peak contributors by shifting mass to low-cost positions."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        b = fast_conv(a)
        bmax = b.max()
        curr = 2.0 * n * bmax
        stall += 1

        top_peaks = np.argsort(b)[::-1][:n_top]
        improved = False

        for k in top_peaks[:8]:
            ki = int(k)
            j_lo = max(0, ki - (n - 1))
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idxs = np.arange(j_lo, j_hi + 1)
            contrib = a[idxs] * a[ki - idxs]

            if contrib.max() < 1e-20:
                continue

            # Find the position with max contribution
            src_local = np.argmax(contrib)
            src = idxs[src_local]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Cost of each position (how much it contributes to other peaks)
            costs = np.zeros(n)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                j2_lo = max(0, k2i - (n - 1))
                j2_hi = min(n - 1, k2i)
                valid = np.arange(j2_lo, j2_hi + 1)
                if len(valid) > 0:
                    costs[valid] += a[k2i - valid]

            # Destination: low cost, not the source
            costs[src] = 1e10
            dst = int(np.argmin(costs))

            # Try different fractions
            found = False
            for frac in np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.35]):
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M = fast_conv(a_trial).max()
                if 2.0 * n * M < best_val - 1e-15:
                    a = a_trial
                    best_val = 2.0 * n * M
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Random small move
                if a[src] > 1e-8:
                    dst_r = int(rng.integers(n))
                    if dst_r != src:
                        frac_r = rng.uniform(0.005, 0.03)
                        amt_r = a[src] * frac_r
                        a[src] -= amt_r
                        a[dst_r] += amt_r
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_r = fast_conv(a).max()
                        if 2.0 * n * M_r < best_val:
                            best_val = 2.0 * n * M_r
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            # Random perturbation
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-8:
                frac_p = rng.uniform(0.002, 0.02)
                amt_p = a[i1] * frac_p
                a[i1] -= amt_p
                a[i2] += amt_p
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_p = fast_conv(a).max()
                if 2.0 * n * M_p < best_val:
                    best_val = 2.0 * n * M_p
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Surg stall={stall} ({elapsed:.0f}s): best={best_val/(a.sum()**2):.6f}")

    return best_a

    remaining = time_budget - (time.monotonic() - t_start)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Peak Surgery (t={t3:.0f}s) ===")
    if t3 > 20:
        a_out = peak_surgery(normalize(best_seq), t3, verbose=True, seed=42,
                             n_top=20, n_fracs=5)
        try_update(a_out, "surgery1")

    # === Phase 4: Multi-length search ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: Multi-length (remaining={remaining:.0f}s) ===")
    if remaining > 200:
        n_best = len(best_seq)
        lengths = []
        for mult in [0.8, 1.1, 1.3, 0.7]:
            cand_n = int(n_best * mult)
            if 2000 <= cand_n <= 12000 and cand_n not in lengths:
                lengths.append(cand_n)

        for ln in lengths[:2]:
            remaining = time_budget - (time.monotonic() - t_start)
            if remaining < 120:
                break
            print(f"  Trying n={ln}...")
            a_long = interpolate_seq(best_seq, ln)
            try_update(a_long, f"interp_{ln}")

            t4a = min(remaining * 0.35, 150.0)
            if t4a > 20:
                a_out = prox_grad_optimize(a_long, t4a, n_peaks=30,
                                           verbose=False, seed=ln % 500)
                try_update(a_out, f"pg_{ln}")

            remaining = time_budget - (time.monotonic() - t_start)
            if remaining > 80:
                t4b = min(remaining * 0.25, 80.0)
                if t4b > 15:
                    a_out = spectral_optimize(normalize(best_seq), t4b, verbose=False,
                                              seed=1, n_suppress=15, alpha=0.3)
                    try_update(a_out, f"spec_{ln}")

    # === Phase 5: Final polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 5: Final Polish (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        t5 = min(remaining * 0.6, 250.0)
        if t5 > 20:
            a_out = prox_grad_optimize(normalize(best_seq), t5, n_peaks=20,
                                       verbose=True, seed=999,
                                       step_init=0.02, sa_init=0.1,
                                       sa_min=0.005, step_decay=0.9998,
                                       restart_every=5000)
            try_update(a_out, "final_pg")

        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 30:
            t5b = min(remaining * 0.5, 80.0)
            if t5b > 15:
                a_out = peak_surgery(normalize(best_seq), t5b, verbose=False,
                                     seed=777, n_top=10)
                try_update(a_out, "final_surg")

    # Final validation
    if best_seq is not None:
        fs = 2.0 * len(best_seq) * fast_conv(normalize(best_seq)).max()
        if np.isfinite(fs):
            best_score = fs
        # Also verify with actual evaluator
        fs2 = evaluate_safe(best_seq)
        if np.isfinite(fs2) and fs2 < best_score:
            best_score = fs2
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_start:.0f}s")
    else:
        n_fb = 5000
        best_seq = init_ramp(n_fb, 0.4)
        best_score = 2.0 * n_fb * fast_conv(best_seq).max()
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
