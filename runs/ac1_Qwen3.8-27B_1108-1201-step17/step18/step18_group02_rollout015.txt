def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_grad(a, ac, tau):
    n = len(a)
    ac_scaled = ac / tau
    ac_scaled = ac_scaled - ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()
    a_flip = a[::-1].astype(np.float64)
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)


def power_sum_grad(a, ac, p):
    n = len(a)
    ac_shifted = ac - ac.max()
    ac_p = np.exp(p * ac_shifted)
    total = ac_p.sum()
    weight = p * ac_p / total
    a_flip = a[::-1].astype(np.float64)
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Wf = np.fft.rfft(weight.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Wf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)


def smooth_adam(a0, time_limit, tau_decay=0.9997, lr=0.8, beta1=0.9, beta2=0.999,
                restart_every=2500, verbose=False, use_lse=True, p_init=20.0,
                p_decay=0.9999, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    if use_lse:
        tau = max(ac.max() * 0.5, 1e-6)
        tau_min = max(ac.max() * 0.003, 1e-10)
    else:
        p_val = p_init
        p_min = 5.0

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    step_count = 0

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if use_lse:
            tau = max(tau * tau_decay, tau_min)
            g = lse_grad(a, ac, tau)
        else:
            p_val = max(p_val * p_decay, p_min)
            g = power_sum_grad(a, ac, p_val)

        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)

        bc1 = 1.0 - beta1 ** (t + 1)
        bc2 = 1.0 - beta2 ** (t + 1)
        m_c = m_hat / bc1
        v_c = v_hat / bc2

        step = lr / (np.sqrt(v_c) + 1e-8)

        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new
        step_count += 1

        if stall > restart_every:
            stall = 0
            n_pert = max(1, int(0.05 * n))
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            pert = rng.normal(0, 0.05, n_pert) * best_a[idx_pert]
            x = best_a.copy()
            x[idx_pert] += pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Adam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    Adam done ({elapsed:.0f}s): J={2*n*best_m:.6f}, steps={step_count}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=400, max_K=5000,
                   add_per_iter=30, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    ac_init = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac_init.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    lp_fail_count = 0
    no_improve = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac_y = fast_autocorr(y)
        M_y = ac_y.max()

        if M_y < best_m - 1e-14:
            best_m = M_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
            no_improve = 0

        if no_improve > 200:
            break

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_y - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 5:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 15:
                constraint_set = set(int(k) for k in constraint_list[:max(100, initial_K // 2)])
            continue

        lp_fail_count = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_y + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_y
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t <= 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def pairwise_coord_descent(a0, time_limit, rng_seed=7, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            top_candidates = candidates[np.argsort(cand_scores)[:5]]

            found = False
            for frac in [0.003, 0.01, 0.03, 0.06, 0.12, 0.2]:
                if found:
                    break
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for dst in top_candidates:
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break

            if not found and stall > 80:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            if stall % 30 == 0:
                n_pert = max(2, int(0.01 * n))
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx_p in idx_pert:
                    moving_from = idx_p
                    moving_to = int(rng.integers(n))
                    if moving_from != moving_to and a[moving_from] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[moving_from] * frac
                        a_trial[moving_from] -= amt
                        a_trial[moving_to] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    M_new = fast_autocorr(a_trial).max()
                    if M_new < best_m:
                        a = a_trial
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Pairwise ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng_seed=0):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(rng_seed)
    candidates = []

    for freq in [1, 2, 3, 4]:
        for phase in [0.0, 0.2, 0.4, 0.6, 0.8]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.005)
            candidates.append(a / a.sum())

    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())

    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(a / a.sum())

    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.15, 0.25]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())

    a = np.cos(np.pi * (x - 0.5)) ** 2
    candidates.append(a / a.sum())

    candidates.append(np.ones(n) / n)

    for rate in [0.5, 1, 2, 4]:
        a = 1.0 / (i_arr + 1.0) ** (1.0 / rate)
        candidates.append(a / a.sum())

    # Multi-bump
    for n_bumps in [2, 3, 4]:
        centers = np.linspace(0.15, 0.85, n_bumps)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((x - c) ** 2) / (2.0 * 0.1 ** 2))
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())

    # Chirp-like (real part magnitude)
    for freq_start in [1, 2, 5]:
        phase = 2.0 * np.pi * freq_start * i_arr ** 2 / n
        a = np.maximum(np.abs(np.cos(phase)), 0.01)
        candidates.append(a / a.sum())

    # Random smooth
    for _ in range(5):
        n_knots = int(n / 50)
        knot_vals = rng.uniform(0.1, 1.0, n_knots)
        knot_x = np.linspace(0, 1, n_knots)
        a = np.interp(x, knot_x, knot_vals)
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())

    return candidates


def spectral_perturb(a0, time_limit, rng_seed=99, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 1).bit_length()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()

        A = np.fft.rfft(a, fft_size)
        phase = rng.uniform(0, 2 * np.pi, size=len(A))
        amp = rng.uniform(0.95, 1.05, size=len(A))
        A_pert = A * amp * np.exp(1j * phase)
        a_new = np.fft.irfft(A_pert, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < M:
            a = a_new
        else:
            a_new2 = a + rng.normal(0, 0.001, n) * a
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-15:
                a_new2 = a_new2 / s2
                M_new2 = fast_autocorr(a_new2).max()
                if M_new2 < M:
                    a = a_new2

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Multi-length initialization ===
    print(f"\n=== Phase 1: Multi-length init ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 120.0)
    if t1 > 30 and parent is not None:
        n_candidates = [n0]
        for delta in [-500, -300, -100, 100, 300, 500, 800]:
            nn = n0 + delta
            if 2500 <= nn <= 7000:
                n_candidates.append(nn)
        # Deduplicate
        n_candidates = sorted(set(n_candidates))

        per_n_time = t1 / len(n_candidates)
        for nn in n_candidates:
            if time.monotonic() - t_global > time_budget - t1:
                break
            if nn == n0:
                a_init = norm_sum1(best_seq)
            else:
                a_init = interpolate_sequence(best_seq, nn)

            sc_init = evaluate_safe(a_init)
            if not np.isfinite(sc_init):
                continue

            t_short = max(5.0, min(per_n_time, 25.0))
            a_out, _ = smooth_adam(a_init, t_short, tau_decay=0.9999, lr=0.5,
                                   restart_every=500, verbose=False, rng_seed=nn)
            check_and_update(a_out, f"init_n{nn}")

    n_best = len(best_seq)
    print(f"  After Phase 1: best n={n_best}, score={best_score:.6f}")

    # === Phase 2: Main Adam optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam main (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam(a_in, t2, tau_decay=0.99975, lr=0.7,
                               beta1=0.9, beta2=0.999, restart_every=2000,
                               verbose=True, use_lse=True, rng_seed=123)
        check_and_update(a_out, f"adam_main_n{n_best}")

    # === Phase 3: LP-FW ===
    remaining = time_budget - (time.monotonic() - t0 if False else (time.monotonic() - t_global))
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, initial_K=500,
                                   max_K=4000, add_per_iter=40, verbose=True)
        check_and_update(a_out, f"lpfw_n{n_best}")

    # === Phase 4: Spectral perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Spectral perturb (n={n_best}) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_perturb(a_in, t4, rng_seed=77, verbose=False)
        check_and_update(a_out, f"spectral_n{n_best}")

    # === Phase 5: Pairwise coord descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pairwise coord (n={n_best}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_coord_descent(a_in, t5, rng_seed=200, verbose=True)
        check_and_update(a_out, f"pairwise_n{n_best}")

    # === Phase 6: Second LP-FW round ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP-FW round 2 (n={n_best}) ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, initial_K=800,
                                   max_K=6000, add_per_iter=60, verbose=True)
        check_and_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 7: Final coordination ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 60:
        # Quick Adam
        t7a = min(remaining * 0.3, 80.0)
        if t7a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = smooth_adam(a_in, t7a, tau_decay=0.9999, lr=0.3,
                                   restart_every=1000, verbose=False, rng_seed=999)
            check_and_update(a_out, f"adam_final_n{n_best}")

        # Coord descent
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.4, 100.0)
        if t7b > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = pairwise_coord_descent(a_in, t7b, rng_seed=555, verbose=False)
            check_and_update(a_out, f"coord_final_n{n_best}")

        # One more LP-FW if time
        remaining = time_budget - (time.monotonic() - t_global)
        t7c = min(remaining * 0.5, 80.0)
        if t7c > 20 and n_best <= 5500:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7c, solver, initial_K=1000,
                                       max_K=8000, add_per_iter=80, verbose=True)
            check_and_update(a_out, f"lpfw_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
