        if corr > 0.9999:
            # Slightly perturb M to get different direction
            M_margin *= 1.01
            continue

        # Line search for step size
        current_obj = 2.0 * n * M
        best_t = 0.0
        best_t_obj = current_obj
        for t in [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_trial = (1 - t) * a + t * b
            a_trial = norm_simplex(a_trial)
            obj_trial = obj_val(a_trial)
            if obj_trial < best_t_obj:
                best_t_obj = obj_trial
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            a = (1 - best_t) * a + best_t * b
            a = norm_simplex(a)
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
            M_margin = max(1.0, M_margin * 0.999)
        else:
            stall += 1
            if stall >= max_stall:
                # Perturb to escape local optimum
                scale = 0.003 * (1 + stall / max_stall)
                a = norm_simplex(a + rng.normal(0, scale, n))
                stall = 0
                M_margin = min(1.1, M_margin * 1.02)

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iteration} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, M_margin={M_margin:.4f}")

    return best_a, best_obj


def smooth_grad_descent(a0, time_limit, verbose=False):
    """Log-sum-exp smoothed gradient descent with Nesterov acceleration."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()

    T = best_obj / (2.0 * n) * 0.3  # initial temperature
    T_min = T * 0.01
    lr_init = 2.0 / (n * n * T) * 0.5
    momentum = 0.8

    v = np.zeros(n)
    y = a.copy()  # Nesterov position
    t_count = 0

    while time.monotonic() - t0 < time_limit:
        t_count += 1
        progress = min(1.0, t_count / 50000.0)
        T_cur = T * (T_min / T) ** progress
        lr = lr_init / (1.0 + 0.01 * t_count)

        # Nesterov step
        beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y_new = y + momentum * (y - a)
        y_new = np.maximum(y_new, 1e-15)
        sy = y_new.sum()
        y_new = y_new / sy

        # Compute smooth gradient
        c = fast_conv(y_new)
        c_max = c.max()
        if c_max < 1e-30:
            break
        exponents = (c - c_max) / T_cur
        exponents = np.maximum(exponents, -60.0)
        p = np.exp(exponents)
        p = p / p.sum()

        # Gradient: 2n * sum_k p_k * a[k-j]
        grad = np.zeros(n)
        for k in range(len(c)):
            if p[k] < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            grad[idx] += 2.0 * p[k] * y_new[j_idx]

        # Gradient step with projection onto simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Track best (use actual max, not smooth approximation)
        actual_obj = obj_val(a_new)
        if actual_obj < best_obj - 1e-12:
            best_obj = actual_obj
            best_a = a_new.copy()

        v = momentum * v + (1 - momentum) * grad
        a = np.maximum(a - lr * v, 1e-15)
        a = a / a.sum()
        y = y_new

        if verbose and t_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t_count} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, T={T_cur:.6f}")

    return best_a, best_obj


def spectral_smooth(a, strength=0.2):
    """Compress dynamic range of |FFT(a)|."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return norm_simplex((1 - strength * 0.5) * a + strength * 0.5 * a_new)


def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent with structured moves."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            contrib_sorted = np.sort(contrib)[::-1]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:3]]

            for src in top_contrib_idx:
                src_val = a[src]
                if src_val < 1e-10:
                    continue
                for num_dst in [3, 5]:
                    dsts = rng.choice(n, size=num_dst, replace=False)
                    for dst in dsts:
                        if dst == src:
                            continue
                        for frac in [0.005, 0.01, 0.02, 0.05]:
                            amt = src_val * frac
                            a_trial = a.copy()
                            a_trial[src] -= amt
                            a_trial[dst] += amt
                            a_trial = np.maximum(a_trial, 0.0)
                            s_t = a_trial.sum()
                            if s_t < 1e-15:
                                continue
                            a_trial = a_trial / s_t
                            M_trial = fast_conv(a_trial).max()
                            if M_trial < best_obj / (2.0 * n) - 1e-15:
                                a = a_trial
                                best_obj = 2.0 * n * M_trial
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_obj / (2.0 * n):
                        best_obj = 2.0 * n * M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.03
    return norm_simplex(a)


def make_init_uniform(n):
    return np.ones(n) / n


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator for minimizing 2n*max(conv)/(sum^2)."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        n_base = 4200
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    def check_and_update(a_arr, label):
        nonlocal best_seq, best_score
        if a_arr is None:
            return
        a_arr = np.maximum(np.asarray(a_arr, dtype=np.float64), 0.0)
        s = a_arr.sum()
        if s < 0.01:
            return
        a_norm = a_arr / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # === Phase 1: LP-based optimization (main) ===
    print(f"\n=== Phase 1: LP Direction (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 750.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = lp_refine(norm_simplex(best_seq), t1, verbose=True)
        check_and_update(a_out, "lp_main")

        # Spectral polish after LP
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sp = spectral_smooth(norm_simplex(best_seq), strength=0.15)
            check_and_update(a_sp, "spectral_after_lp")
            # LP on spectral-smoothed
            if remaining > 100:
                a_out2, _ = lp_refine(a_sp, min(remaining * 0.3, 200.0), verbose=False)
                check_and_update(a_out2, "lp_after_spectral")

    # === Phase 2: Smooth gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Smooth Grad Descent (n={n_cur}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_out, _ = smooth_grad_descent(norm_simplex(best_seq), t2, verbose=True)
        check_and_update(a_out, "sgd")

    # === Phase 3: LP on different initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Multi-init LP (n={n_cur}) ===")
    t3_total = min(remaining * 0.15, 250.0)
    if t3_total > 30:
        inits = [
            make_init_bowl(n_cur, sharpness=2.0),
            make_init_dual_peak(n_cur, 0.15, 0.85, 0.12),
            make_init_cosine(n_cur, mode=2),
        ]
        t_per = t3_total / len(inits)
        for idx, init in enumerate(inits):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_avail = min(t_per, remaining * 0.4)
            if t_avail < 10:
                break
            print(f"  LP init {idx+1}/{len(inits)} (budget={t_avail:.0f}s)")
            a_out, _ = lp_refine(init, t_avail, verbose=False)
            check_and_update(a_out, f"lp_init{idx+1}")

    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Coord Descent (n={n_cur}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 10:
        a_out, _ = coordinate_descent(norm_simplex(best_seq), t4, verbose=False)
        check_and_update(a_out, "cd")

    # === Phase 5: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Length Variation ===")

    # Try slightly longer
    for delta in [300, 700]:
        new_n = n_cur + delta
        if new_n > 7000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        check_and_update(a_long, f"interp_n{new_n}")
        t5 = min(remaining * 0.2, 150.0)
        if t5 > 20:
            a_out, _ = lp_refine(a_long, t5, verbose=False)
            check_and_update(a_out, f"lp_n{new_n}")

    # Try slightly shorter
    for delta in [300, 700]:
        new_n = n_cur - delta
        if new_n < 2000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_short = interpolate_sequence(best_seq, new_n)
        check_and_update(a_short, f"interp_short_n{new_n}")
        t5s = min(remaining * 0.2, 120.0)
        if t5s > 20:
            a_out, _ = lp_refine(a_short, t5s, verbose=False)
            check_and_update(a_out, f"lp_short_n{new_n}")

    # === Phase 6: Final intensive LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final LP Polish (n={n_cur}) ===")
    t6 = min(remaining * 0.3, 250.0)
    if t6 > 20:
        a_out, _ = lp_refine(norm_simplex(best_seq), t6, verbose=True)
        check_and_update(a_out, "lp_final")

        # One more spectral+LP combo
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_sp = spectral_smooth(norm_simplex(best_seq), strength=0.1)
            check_and_update(a_sp, "spectral_final")
            if remaining > 60:
                a_out2, _ = lp_refine(a_sp, min(remaining * 0.4, 150.0), verbose=False)
                check_and_update(a_out2, "lp_after_spectral_final")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 7: Final CD (n={n_cur}) ===")
        t7 = min(remaining * 0.5, 80.0)
        if t7 > 5:
            a_out, _ = coordinate_descent(norm_simplex(best_seq), t7, verbose=False)
            check_and_update(a_out, "cd_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_uniform(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
