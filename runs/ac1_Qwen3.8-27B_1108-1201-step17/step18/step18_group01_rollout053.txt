"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fixed-point iteration (state-of-the-art approach)
2. Multiplicative gradient descent as refinement
3. Coordinate descent polish
4. Multi-length exploration

Key innovation: Uses the LP from the state-of-the-art to find optimal 
blending directions, which provides a more principled descent direction 
than pure subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def solve_lp_direction(f, M, top_k=200):
    """
    Solve the LP: maximize sum(g) subject to g >= 0 and ||f*g||_inf <= M
    (only for top_k lags).
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    ac = fast_autocorr(f)
    top_lags = np.argsort(ac)[::-1][:top_k]
    
    # Build sparse constraint matrix: A[k, j] = f[k-j] if valid
    A = lil_matrix((len(top_lags), n))
    for row, k in enumerate(top_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        A[row, js] = f[k - js]
    A = A.tocsr()
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b_ub = np.full(len(top_lags), M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0)
        if g.sum() > 1e-10:
            return g
    # Fallback: return f itself (which is feasible)
    return f.copy()


def lp_iteration(f, time_limit, top_k=200, verbose=False):
    """
    Iterative LP-based blending:
    1. Compute M = max(f*f)
    2. Solve LP to get g

    4. Repeat
    """
    n = len(f)
    f = norm_sum1(f)
    best_f = f.copy()
    best_m = fast_autocorr(f).max()
    t0 = time.monotonic()
    iter_count = 0
    blend_t = 0.3  # initial blend factor
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        ac = fast_autocorr(f)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_f = f.copy()
        
        # Solve LP
        g = solve_lp_direction(f, M, top_k=top_k)
        g = norm_sum1(g)
        
        # Check if g is meaningfully different from f
        diff = np.abs(g - f).sum()
        if diff < 1e-6:
            # LP converged, try smaller top_k for refinement
            if top_k > 50:
                top_k = max(50, top_k // 2)
                continue
            else:
                break
        
        # Blend
        t = min(blend_t, 0.5)  # cap blend factor
        f_new = (1 - t) * f + t * g
        f_new = norm_sum1(f_new)
        
        # Check improvement
        ac_new = fast_autocorr(f_new)
        M_new = ac_new.max()
        
        if M_new <= M + 1e-10:
            f = f_new
            blend_t *= 0.99
        else:
            # Try smaller blend
            t = t * 0.3
            f_new = (1 - t) * f + t * g
            f_new = norm_sum1(f_new)
            ac_new = fast_autocorr(f_new)
            M_new = ac_new.max()
            if M_new <= M + 1e-10:
                f = f_new
            blend_t *= 0.8
            if blend_t < 0.01:
                break
        
        iter_count += 1
        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0

        print(f"    LP done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_f


def mgd_optimize(a0, time_limit, top_k, eta0, verbose):
    """Multiplicative Gradient Descent on the simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = eta0
    patience = 0
    max_patience = 500
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 2.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        # Compute subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * a[ki - idx]
        
        grad = grad - grad.mean()
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()
        
        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.5
            if eta < 0.001:
                eta = 0.02
                patience = max(0, patience - 200)
        
        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def coord_polish(a0, time_limit, verbose=False):
    """Coordinate descent polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(8, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination (lowest contribution to peaks)
            peak_contrib = np.zeros(n)
            for k2 in top_k:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
            
            in_peak = peak_contrib > 0
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(peak_contrib[candidates])]
            if dst == src:
                continue
            
            for frac in [0.02, 0.04, 0.06]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < M - 1e-14:
                    a = a_trial
                    M = M_trial
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def parametric_init(n):
    """Try various parametric families and return the best."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    families = []
    # Sine-based
    for m in [1, 2, 3]:
        families.append(np.sin(np.pi * m * x) ** 2)
    # Power law
    for p in [0.5, 1, 1.5, 2]:
        families.append((i_arr + 1) ** p)
    # Gaussian bumps
    for c in [0.25, 0.5, 0.75]:
        for sig in [0.1, 0.15, 0.2]:
            families.append(np.exp(-((x - c) ** 2) / (2 * sig ** 2)))
    # Multiple bumps
    for n_bumps in [2, 3, 4]:
        centers = np.linspace(0.2, 0.8, n_bumps)
        f = np.zeros(n)
        for c in centers:
            f += np.exp(-((x - c) ** 2) / (2 * 0.05 ** 2))
        families.append(f)
    # Cosine window
    families.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # Raised cosine
    families.append(0.5 * (1 + np.cos(np.pi * (2*x - 1))))
    
    best = None
    best_score = np.inf
    for fam in families:
        fam = norm_sum1(fam)
        ac = fast_autocorr(fam)
        score = 2.0 * n * ac.max()
        if score < best_score:
            best_score = score
            best = fam.copy()
    
    return best, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fixed-point iteration (state-of-the-art approach) as primary optimizer
2. Multiplicative gradient descent for warmup and refinement
3. Coordinate descent polish targeting peak-contributing positions
4. Multi-length exploration via interpolation
5. Adaptive step-size and restart strategies

Key innovation: Uses the LP from the state-of-the-art to find optimal blending
directions. The LP maximizes sum(g) subject to g>=0 and top-K cross-correlation
constraints ||f*g||_inf <= M, providing a principled descent direction that
differs from pure subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def solve_lp_direction(f, M, top_k=200):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and cross-corr constraints.
    Only enforces top_k most restrictive lags for efficiency.
    Returns optimal g (unnormalized) or None on failure.
    """
    n = len(f)
    ac = fast_autocorr(f)
    top_lags = np.argsort(ac)[::-1][:top_k]

    A_lil = lil_matrix((len(top_lags), n))
    for row, k in enumerate(top_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A_lil[row, js] = f[k - js]
    A = A_lil.tocsr()

    c = -np.ones(n)
    b_ub = np.full(len(top_lags), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass

    return None


def lp_iteration(f0, time_limit, top_k=200, verbose=False):
    """
    Iterative LP-based blending optimization.
    Each step: solve LP for direction g, blend f toward g, accept if improved.
    """
    n = len(f0)
    f = norm_sum1(f0)
    best_f = f.copy()
    best_m = fast_autocorr(f).max()
    t0 = time.monotonic()
    iter_count = 0
    blend_t = 0.4
    no_improve_count = 0
    max_no_improve = 80
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        ac = fast_autocorr(f)
        M = ac.max()

        if M < best_m - 1e-16:
            best_m = M
            best_f = f.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > max_no_improve:
            # Random restart perturbation
            idx = rng.integers(0, n, size=max(1, n // 50))
            f[idx] += rng.normal(0, 0.01, size=len(idx)) * f.mean()
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            no_improve_count = 0
            continue

        g = solve_lp_direction(f, M, top_k=top_k)
        if g is None:
            # LP failed, reduce top_k and retry
            if top_k > 50:
                top_k = max(50, top_k // 2)
            continue

        g = norm_sum1(g)
        diff = np.abs(g - f).mean()
        if diff < 1e-7:
            if top_k > 50:
                top_k = max(50, top_k // 2)
                continue
            break

        # Line search over blend factors
        best_blend = None
        best_M_blend = M
        for t in [blend_t, blend_t * 0.5, blend_t * 0.25, blend_t * 0.1]:
            f_try = (1.0 - t) * f + t * g
            f_try = norm_sum1(f_try)
            M_try = fast_autocorr(f_try).max()
            if M_try < best_M_blend:
                best_M_blend = M_try
                best_blend = f_try

        if best_blend is not None:
            f = best_blend
            blend_t = min(blend_t * 1.02, 0.6)
        else:
            blend_t *= 0.7
            if blend_t < 0.05:
                blend_t = 0.3
                # Perturb to escape
                idx = rng.integers(0, n, size=max(1, n // 100))
                f[idx] += rng.normal(0, 0.02, size=len(idx)) * f.mean()
                f = np.maximum(f, 0.0)
                f = f / f.sum()

        iter_count += 1
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, "
                  f"curM={M:.8f}, bestM={best_m:.8f}, tk={top_k}, bt={blend_t:.3f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LP done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}, tk={top_k}")
    return best_f


def mgd_optimize(a0, time_limit, top_k, eta0, verbose=False):
    """Multiplicative Gradient Descent on the simplex with multi-lag subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta0
    patience = 0
    max_patience = 400
    iter_count = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            # Random kick to escape
            idx = rng.integers(0, n, size=max(1, n // 80))
            a[idx] += rng.normal(0, 0.03, size=len(idx)) * a.mean()
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            patience = 0
            if time.monotonic() - t0 > time_limit * 0.9:
                break

        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 2.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * a[ki - idx]

        grad = grad - grad.mean()
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()

        if M_new <= M + 1e-12:
            a = a_new
            eta = min(eta * 1.001, 0.05)
        else:
            eta *= 0.6
            if eta < 0.002:
                eta = 0.015
                patience = max(0, patience - 100)

        iter_count += 1
        if verbose and iter_count % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, "
                  f"eta={eta:.5f}, p={patience}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def coord_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            peak_contrib = np.zeros(n)
            for k2 in top_k:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]

            in_peak = peak_contrib > np.percentile(peak_contrib, 70)
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(peak_contrib[candidates])]
            if dst == src:
                continue

            for frac in [0.02, 0.04, 0.06, 0.09]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < M - 1e-14:
                    a = a_trial
                    M = M_trial
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def parametric_init(n):
    """Try various parametric families and return the best one."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    families = []

    for m in [1, 2, 3, 4]:
        families.append(np.sin(np.pi * m * x) ** 2)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        families.append((i_arr + 1.0) ** p)
        families.append((n - i_arr).clip(1.0) ** p)
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.08, 0.12, 0.18, 0.25]:
            families.append(np.exp(-((x - c) ** 2) / (2.0 * sig ** 2)))
    for nb in [2, 3, 4, 5]:
        centers = np.linspace(0.15, 0.85, nb)
        f = np.zeros(n)
        for c in centers:
            f += np.exp(-((x - c) ** 2) / (2.0 * 0.06 ** 2))
        families.append(f)
    families.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    for rate in [0.5, 1, 2, 5]:
        families.append(np.exp(-rate * x))
        families.append(np.exp(rate * (x - 1.0)))

    best = None
    best_score = np.inf
    for fam in families:
        fam = norm_sum1(fam)
        ac = fast_autocorr(fam)
        score = 2.0 * n * ac.max()
        if score < best_score:
            best_score = score
            best = fam.copy()

    return best, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining LP-based iteration, MGD, coordinate
    descent, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = norm_sum1(p)
            p_score = evaluate_safe(p_norm)
            if np.isfinite(p_score):
                best_seq = p_norm
                best_score = p_score
                print(f"Parent: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass

    # Fallback: parametric initialization
    if best_seq is None:
        for n_try in [3500, 4500, 5200]:
            a_p, s_p = parametric_init(n_try)
            sc = evaluate_safe(a_p)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_p.copy()
        print(f"Init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: MGD warmup ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MGD warmup (n={n0}) ===")
    t1 = min(remaining * 0.08, 100.0)
    if t1 > 15:
        a_out = mgd_optimize(norm_sum1(best_seq), t1, top_k=30, eta0=0.03, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  MGD warmup improved to {sc:.6f}")

    # === Phase 2: LP-based iteration (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP iteration (n={n_best}) ===")
    t2 = min(remaining * 0.38, 480.0)
    if t2 > 30:
        a_out = lp_iteration(norm_sum1(best_seq), t2, top_k=200, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** LP iteration: {sc:.6f}")

    # === Phase 3: MGD refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: MGD refinement (n={n_best}) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 15:
        a_out = mgd_optimize(norm_sum1(best_seq), t3, top_k=50, eta0=0.012, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** MGD refinement: {sc:.6f}")

    # === Phase 4: Second LP round with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP round 2 (n={n_best}) ===")
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 20:
        a_out = lp_iteration(norm_sum1(best_seq), t4, top_k=300, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** LP round 2: {sc:.6f}")

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord polish (n={n_best}) ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 10:
        a_out = coord_polish(norm_sum1(best_seq), t5, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Coord: {sc:.6f}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length exploration (base n={n_best}) ===")

    length_candidates = sorted(set([
        n_best - 800, n_best - 400, n_best - 200,
        n_best + 200, n_best + 500, n_best + 800, n_best + 1200
    ]))
    length_candidates = [l for l in length_candidates if 2500 <= l <= 7500]

    for new_n in length_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Exploring n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        # LP on new length
        t6a = min(remaining * 0.25, 80.0)
        if t6a > 15:
            a_out = lp_iteration(a_long, t6a, top_k=150, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** n={new_n} LP: {sc:.6f}")

        # MGD on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            t6b = min(remaining * 0.20, 50.0)
            if t6b > 10:
                a_out = mgd_optimize(norm_sum1(best_seq), t6b, top_k=40, eta0=0.015, verbose=False)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** n={len(best_seq)} MGD: {sc:.6f}")

    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")

    if remaining > 30:
        t7a = min(remaining * 0.35, 100.0)
        if t7a > 15:
            a_out = lp_iteration(norm_sum1(best_seq), t7a, top_k=150, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Final LP: {sc:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t7b = min(remaining * 0.30, 60.0)
        if t7b > 10:
            a_out = mgd_optimize(norm_sum1(best_seq), t7b, top_k=60, eta0=0.008, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Final MGD: {sc:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7c = min(remaining * 0.30, 50.0)
        if t7c > 10:
            a_out = coord_polish(norm_sum1(best_seq), t7c, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Final Coord: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4500
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
