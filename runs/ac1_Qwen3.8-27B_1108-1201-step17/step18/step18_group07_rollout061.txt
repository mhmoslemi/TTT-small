import numpy as np

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def logsumexp_subgradient(a, beta, top_k=None):
    """Compute subgradient of logsumexp(beta * C) / beta where C = conv(a,a).
    
    This approximates the subgradient of max(C) when beta is large.
    Returns (gradient, smooth_obj_value)
    """
    n = len(a)
    c = fast_conv(a)
    
    # Compute softmax weights over C
    max_c = c.max()
    if max_c < 1e-30:
        w = np.ones(len(c)) / len(c)
        val = 0.0
    else:
        scaled = beta * (c - max_c)
        scaled = np.maximum(scaled, -50.0)  # prevent underflow
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        val = max_c + np.log(exp_scaled.sum()) / beta
    
    # Gradient: g[i] = sum_k w[k] * a[k-i] (with bounds)
    grad = np.zeros(n)
    
    if top_k is not None and top_k < len(c):
        # Only use top peaks for efficiency
        top_idx = np.argsort(c)[::-1][:top_k]
        w_top = w[top_idx]
        w_top_sum = w_top.sum()
        if w_top_sum < 1e-15:
            return grad, val
        w_top = w_top / w_top_sum
        top_idx = top_idx.astype(int)
        for wi, k in zip(w_top, top_idx):
            if wi < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    else:
        # Use all peaks (slower but more complete)
        for k in range(len(c)):
            if w[k] < 1e-10:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * w[k] * a[j_idx]
    
    return grad, val


def spectral_equalize_aggressive(a, n_iter=200, verbose=False):

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Range: typically 1.0 (flat) to maybe 3-5 (peaked)
        aggressiveness = min(0.95, 0.3 + 0.1 * (peak_ratio - 1.0))

        mag_mean = mag.mean()
        if mag_mean < 1e-30:
            break

        # Apply compression: new_ratio = ratio^(1-aggressiveness)
        # This reduces the spread while keeping DC component

        # Also apply a "peak notch" - reduce amplitudes that contribute to peaks

        # Reduce amplitudes where contribution is positive
        pos_mask = freq_contrib > 0
        neg_mask = freq_contrib < 0
        # Slightly reduce positive contributors, boost negative ones
        compressed[pos_mask] *= (1.0 - 0.02 * aggressiveness)
        compressed[neg_mask] *= (1.0 + 0.02 * aggressiveness)
        
        mag_new = mag_mean * compressed
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:

def coordinate_descent_fast(a0, time_limit, verbose=False):
    """Fast coordinate descent: move mass between positions to reduce peaks."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_conv(a)
        # Find worst peak
        peak_pos = int(np.argmax(c))
        
        # Find positions that most contribute to this peak
        i_start = max(0, peak_pos - n + 1)
        i_end = min(n, peak_pos + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_pos - idx]
        
        # Move mass from the highest contributor to a random position
        src = int(idx[np.argmax(contrib)])
        if a[src] < 1e-10:
            # Try random pair
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-10:
                continue
            src = i
            dst = j
        else:
            dst = int(rng.integers(n))
            while dst == src:
                dst = int(rng.integers(n))
        
        # Try different fractions
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.2]:
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-10:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved and it % 1000 == 0:
            # Random restart perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
    
    return best_a, best_m


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    a += 0.01
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_simplex(a)


def make_init_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.1
    for f in range(1, 20):
        coeff = rng.normal(0, 0.08 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_peak_targeted(a, n_iter=100, verbose=False):
    """Peak-targeted spectral equalization.
    
    For each major peak in the convolution, identify which frequency components
    contribute most to it and selectively reduce those.
    """
    n = len(a)
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    freqs = np.fft.rfftfreq(fft_size) * fft_size  # frequency indices
    
    for it in range(n_iter):
        if verbose and it % 20 == 0:
            print(f"          Peak-targeted iter {it}: max_C={best_m:.8f}")
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A).copy()
        phase = np.angle(A)
        
        c = fast_conv(a)
        curr_max = c.max()
        
        # Find top few peaks
        n_peaks = min(5, len(c))
        top_peak_positions = np.argsort(c)[::-1][:n_peaks]
        top_peak_vals = c[top_peak_positions]
        
        # For each peak, compute which frequencies contribute positively
        # and reduce those amplitudes
        reduction = np.ones(len(mag))
        
        for p_idx, p_val in zip(top_peak_positions[:3], top_peak_vals[:3]):
            p_int = int(p_idx)
            # Contribution of frequency k to peak at position p:
            # Re(A[k] * A[k] * exp(2*pi*i*k*p/fft_size)) ... 
            # Actually for auto-conv: C[p] = (1/L) sum_k A[k]^2 exp(2*pi*i*k*p/L)
            # The real part contribution from bin k is:
            # Re(A[k]^2) * cos(2*pi*k*p/L) - Im(A[k]^2) * sin(2*pi*k*p/L)
            # Simplified: |A[k]|^2 * cos(2*(phase_k) + 2*pi*k*p/L)
            phase_A = 2.0 * phase  # phase of A[k]^2
            angle = phase_A + 2.0 * np.pi * np.arange(len(mag)) * p_int / fft_size
            contribution = mag ** 2 * np.cos(angle)
            
            # Reduce frequencies that contribute positively to this peak
            positive = contribution > 0
            # Weight by how much they contribute
            weight = np.where(positive, contribution / (np.abs(contribution).max() + 1e-30), 0.0)
            reduction -= 0.05 * weight * (p_val / curr_max)
        
        reduction = np.maximum(reduction, 0.5)  # Don't reduce more than 50%
        
        mag_new = mag * reduction
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        c_new = fast_conv(a_new)
        m_new = c_new.max()
        
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert
            a = norm_simplex(0.8 * a + 0.2 * best_a)
            # If stuck for too long, break
            if it > 30:
                # Try a different approach: uniform compression
                pass
    
    return best_a, best_m

import numpy as np

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):

def subgradient_from_peaks(a, c, beta, top_k=50):
    """Compute subgradient of logsumexp approximation to max(C).
    
    g[i] = 2 * sum_k w[k] * a[k-i] where w = softmax(beta * C)
    """
    n = len(a)
    nk = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:nk]
    top_vals = c[top_idx]
    
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(nk) / nk
    else:
        exponents = beta * (top_vals - max_v) / (max_v + 1e-30)
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(nk) / nk
        else:
            w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def spectral_equalize(a, n_iter, aggressiveness=0.4, peak_aware=True, verbose=False):
    """Spectral equalization with adaptive aggressiveness.
    
    Compresses the dynamic range of the amplitude spectrum, optionally
    with peak-aware modification.
    """
    n = len(a)
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iter):
        if verbose and it % 100 == 0:
            print(f"          Spectral iter {it}: max_C={best_m:.8f}")
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A).copy()
        phase = np.angle(A)
        
        c = fast_conv(a)
        curr_max = c.max()
        mean_c = c.max() / max(1, np.sum(c > c.max() * 0.9))
        
        # Adaptive aggressiveness based on peak-to-mean ratio
        peak_ratio = curr_max / (c.mean() + 1e-30)
        base_agg = min(0.9, 0.2 + 0.15 * (peak_ratio - 1.0))
        curr_agg = base_agg * aggressiveness / 0.4
        
        # Compress dynamic range
        mag_mean = mag[1:].mean() if len(mag) > 1 else mag.mean()
        if mag_mean < 1e-30:
            break
        ratio = mag[1:] / (mag_mean + 1e-30)
        compressed_ratio = ratio ** (1.0 - curr_agg)
        
        if peak_aware:
            # Peak-aware: reduce frequencies that positively contribute to the main peak
            top_peak = int(np.argmax(c))
            freq_indices = np.arange(1, len(mag))
            # Contribution sign: cos(2*phase + 2*pi*k*peak/fft_size)
            angle = 2.0 * phase[1:] + 2.0 * np.pi * freq_indices * top_peak / fft_size
            contrib_sign = np.cos(angle)
            # Reduce positive contributors slightly
            pos_weight = np.where(contrib_sign > 0, 1.0 + 0.03 * curr_agg, 1.0)
            neg_weight = np.where(contrib_sign < 0, 1.0 - 0.01 * curr_agg, 1.0)
            compressed_ratio *= np.where(contrib_sign > 0, 1.0 / pos_weight, neg_weight)
        
        # Reconstruct with DC component preserved
        mag_new = np.empty_like(mag)
        mag_new[0] = mag[0]  # Preserve DC (sum is preserved)
        mag_new[1:] = mag_mean * compressed_ratio
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = norm_simplex(0.5 * a + 0.5 * best_a)
            continue

        c_new = fast_conv(a_new)

            best_m = m_new

            a = a_new

            if it > 200 and best_m - m_new < best_m * 1e-6:
                break
    
    return best_a, best_m


def mirror_descent_lse(a0, time_limit, beta_start=5.0, beta_end=300.0, lr_base=0.15, verbose=False):
    """Mirror descent with log-sum-exp smoothing and annealing."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    t = 0
    lr_scale = 1.0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit * 0.7))
        
        # Anneal beta (square root schedule)
        beta = beta_start + (beta_end - beta_start) * np.sqrt(progress)
        
        # Adaptive top_k: start broad, narrow down
        top_k = max(10, int(100 * (1.0 - progress) + 10))
        
        c = fast_conv(a)
        curr_m = c.max()
        
        # Compute subgradient
        grad = subgradient_from_peaks(a, c, beta=beta, top_k=top_k)
        
        # Multiplicative (mirror) update
        g_max = grad.max()
        if g_max < 1e-15:
            break
        
        lr = lr_base * lr_scale / (1.0 + 0.1 * np.sqrt(t))
        
        log_update = -lr * grad / g_max
        log_update = np.clip(log_update, -8.0, 1.5)
        a_candidate = a * np.exp(log_update)
        a_candidate = np.maximum(a_candidate, 1e-30)
        s = a_candidate.sum()
        if s < 1e-15:
            a = best_a.copy()
            lr_scale *= 0.8
            continue
        a_candidate = a_candidate / s
        
        c_new = fast_conv(a_candidate)
        m_new = c_new.max()
        
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_candidate.copy()
            a = a_candidate
            # Slightly increase lr if improving fast
            if m_new < best_m * 0.999:
                lr_scale = min(2.0, lr_scale * 1.02)
        else:
            # Revert partially
            a = norm_simplex(0.6 * a + 0.4 * best_a)
            lr_scale *= 0.95
            if lr_scale < 0.05:
                break
        
        if verbose and t % 200 == 0:
            print(f"          MD-LSE t={t} ({elapsed:.0f}s): maxC={best_m:.8f}, beta={beta:.1f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            # Then LSE mirror descent
            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now > 15:
                a_des, _ = mirror_descent_lse(a_out, min(40.0, remaining_now * 0.35),
                                              beta_start=10.0, beta_end=200.0, lr_base=0.12)
                try_update(a_des, f"lse_{lbl}")
    
    # === Phase 2: Hybrid alternating on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Hybrid Alternating (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = hybrid_alternating(a_in, t2, verbose=False)
        try_update(a_out, f"hybrid_n{n_best}")
    
    # === Phase 3: Multi-scale optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Multi-scale (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_simplex(best_seq)
        a_out = multi_scale_optimize(a_in, t3, verbose=False)
        try_update(a_out, f"multiscale_n{n_best}")
    
    # === Phase 4: Intensive LSE on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Intensive LSE (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = mirror_descent_lse(a_in, t4 * 0.6, beta_start=5.0, beta_end=400.0, lr_base=0.1)
        try_update(a_out, f"lse_intensive_n{n_best}")
        
        # From perturbed
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            rng = np.random.default_rng(31415)
            a_pert = norm_simplex(best_seq + rng.normal(0, 0.008, n_best))
            a_out, _ = mirror_descent_lse(a_pert, min(60.0, remaining * 0.4),
                                          beta_start=20.0, beta_end=300.0, lr_base=0.08)
            try_update(a_out, f"lse_pert_n{n_best}")
    
    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate Descent (n={n_best}) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 10:
        a_in = norm_simplex(best_seq)
        a_out, _ = coordinate_descent(a_in, t5, verbose=False)
        try_update(a_out, f"cd_n{n_best}")
    
    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length Exploration ===")
    
    for delta in [300, -300, 600]:
        new_n = n_best + delta
        if new_n < 2500 or new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}...")
        a_new = interpolate_sequence(best_seq, new_n)
        try_update(a_new, f"interp_n{new_n}")
        
        t6 = min(80.0, remaining * 0.25)
        if t6 > 15:
            a_out, _ = spectral_equalize(a_new, n_iter=100, aggressiveness=0.5, peak_aware=True)
            try_update(a_out, f"spec_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_des, _ = mirror_descent_lse(a_out, min(30.0, remaining * 0.3),
                                              beta_start=20.0, beta_end=200.0, lr_base=0.1)
                try_update(a_des, f"lse_n{new_n}")
    
    # === Phase 7: Final intensive hybrid ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Intensive Hybrid (n={n_best}) ===")
    t7 = min(remaining * 0.40, 300.0)
    if t7 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = hybrid_alternating(a_in, t7, verbose=True)
        try_update(a_out, f"hybrid_final_n{n_best}")
        
        # Quick coordinate descent after
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_cd, _ = coordinate_descent(a_out, min(30.0, remaining * 0.3), verbose=False)
            try_update(a_cd, f"cd_after_hybrid_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_init_bowl(n_fb, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
