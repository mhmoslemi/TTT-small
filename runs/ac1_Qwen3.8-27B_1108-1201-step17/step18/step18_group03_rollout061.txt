import cvxpy as cp

def fast_autocorr(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based cutting-plane Frank-Wolfe with adaptive constraint management
2. Multi-peak gradient mirror descent for polishing
3. Coordinate descent with lookahead scoring
4. Length interpolation and re-optimization
Key insight: J is shape-determined for large n, so optimize shape at moderate n then polish.
"""

import numpy as np
import time
import cvxpy as cp

_module_solver = None


def get_solver():
    global _module_solver
    if _module_solver is not None:
        return _module_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _module_solver = s
                print(f"  LP solver: {s}")
                return _module_solver
        except Exception:
            continue
    _module_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _module_solver


def fast_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Compute linear convolution of a with b using FFT."""
    n = len(a)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_from_conv(a, conv_val):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * conv_val.max() / (s * s))


def evaluate_safe(a):

def lp_cutting_plane(a0, time_limit, solver, initial_K=300, max_K=2500,
                     add_per_iter=30, verbose=True):
    """LP-based cutting plane Frank-Wolfe for minimizing max linear autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_acc = a.copy()

    ac_y = fast_conv(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        x_ac = fast_conv(x_acc)
        if x_ac.max() < best_m - 1e-15:
            best_m = x_ac.max()
            best_a = x_acc.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 5:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_acc = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_acc = y_new + beta_t * (y_new - y)
            x_acc = np.maximum(x_acc, 0.0)
            sx = x_acc.sum()
            if sx < 1e-15:
                x_acc = y_new.copy()
            else:
                x_acc = x_acc / sx

        y = y_new
        x_acc = x_acc
        iter_count += 1

        if verbose and t % 25 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-CW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def multi_peak_mirror_descent(a0, time_limit, verbose=False):
    """Mirror descent using gradients from multiple peak lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    lr = 2.0
    stall = 0
    n_peaks = 10
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            a = norm_sum1(best_a * (1.0 + 0.005 * rng.standard_normal(n)))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stall = 0
            lr = max(lr * 0.8, 0.1)
            n_peaks = min(n_peaks + 3, 30)
            continue

        # Compute gradient from top peak lags (soft-max weighting)
        top_k = np.argsort(ac)[::-1][:n_peaks]
        top_vals = np.maximum(ac[top_k], 1e-30)
        M_ref = ac.max()
        w = np.exp(5.0 * (top_vals / M_ref - 1.0))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top_k):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[k_int - idx]

        eta = lr / (1.0 + np.sqrt(t))
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        a = a_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MPMD iter={t} ({elapsed:.0f}s): J={J:.6f}, peaks={n_peaks}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent: move mass from peak-contributing positions to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)

    stall = 0
    max_stall = 800
    n_top_lags = 25
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        ac = fast_conv(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        improved = False
        top_lags = np.argsort(ac)[::-1][:n_top_lags]

        for k in top_lags[:min(12, len(top_lags))]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            n_pairs = min(6, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score destinations: low contribution to peak lags
                scores = np.zeros(n)
                for k2 in top_lags[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * ac[k2i]
                        scores[k2i - idx2] += a[idx2] * ac[k2i]

                scores[src] = np.inf
                # Prefer positions that are not in any peak lag support
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_lags[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True
                scores[in_peak] *= 2.0

                dst = int(np.argmin(scores))

                for frac in [0.03, 0.06, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > max_stall:
                break
            if stall % 20 == 0:
                n_moves = int(rng.integers(2, 10))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.005, 0.03))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                if verbose and stall % 200 == 0:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    CoordPolish ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
