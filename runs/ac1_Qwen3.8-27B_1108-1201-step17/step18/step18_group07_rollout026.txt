```python
    """Solve the LP to find direction b that maximizes sum(b) subject to (f*b) <= M, b >= 0."""
    n = len(f)

    # Let me think of a better way.

    # (f*b)[k] = (conv(f, b))[k]
    # We need conv(f, b) <= M * ones(2n-1)
```

```python
import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def compute_grad_smooth(a, beta):
    """Compute gradient of 2n * soft_max_{beta}(conv(a,a)) where sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    
    # Soft-max weights
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.maximum(shifted, -100)  # numerical stability
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # grad[m] = 2n * sum_j a[j] * w[m+j]
    # Compute via FFT cross-correlation
    N = 1 << (3*n - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    grad = 2.0 * n * np.fft.irfft(W_fft * np.conj(A_fft), N)[:n]
    
    return grad, c

def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update: a_new ∝ a * exp(-eta * grad)."""
    log_ratio = -eta * grad
    log_ratio -= log_ratio.max()  # numerical stability
    a_new = a * np.exp(log_ratio)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def optimize_mirror(a0, time_budget, beta_start=10, beta_end=300, verbose=False):
    """Mirror descent with annealed soft-max."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    eta = 1.0  # base step size
    it = 0
    
    while time.monotonic() - t0 < time_budget:
        it += 1
        
        # Anneal beta
        progress = min(1.0, it / max(1, int(time_budget / 0.005)))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Step size decay
        eta_t = eta / (1.0 + 0.01 * it)
        
        # Compute gradient
        grad, c = compute_grad_smooth(a, beta)
        
        # Mirror descent update
        a = mirror_descent_step(a, grad, eta_t)
        
        # Check improvement
        c = fast_conv(a)
        M = c.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={beta:.1f}")
    
    return best_a, best_m
```

1. Compute M = max(f*f)
2. Find b that maximizes sum(b) subject to conv(f,b) <= M, b >= 0
3. Normalize b to have sum 1

The intuition: b puts its mass where it won't create new peaks in conv(f,b) > M. Since f_new = (1-t)*f + t*b, we have:

The cross term f*b is bounded by M (by the LP constraint). The term f*f has max M. The term b*b might have a large max, but for small t, t^2 * b*b is small.

Actually, the key insight is: the LP ensures that conv(f,b) <= M, which bounds the cross term. And since we're maximizing sum(b), b is "spread out" as much as possible, which tends to make b*b also flat.

```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix(f):
    """Build the convolution matrix F where (F @ b)[k] = (conv(f,b))[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        i_start = max(0, k-n+1)
        i_end = min(n, k+1)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_direction(f, M, verbose=False):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    F = build_conv_matrix(f)
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = F
    b_ub = M * np.ones(2*n-1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success and res.x is not None:
        b_val = np.maximum(res.x, 0)
        s = b_val.sum()
        if s > 1e-10:
            b_val = b_val / s
        return b_val
    return None
```

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
