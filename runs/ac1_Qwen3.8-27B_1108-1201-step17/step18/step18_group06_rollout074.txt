def coordinate_polish(a0, time_limit, rng_seed=42, verbose=False):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_J = J_of(a)
    t_start = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    while time.monotonic() - t_start < time_limit and stall < 100:
        conv = fast_convolution(a)
        M = conv.max()
        top_k = np.argsort(conv)[::-1][:15]
        improved = False
        for k in top_k[:8]:
            ki = int(k)
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue
            costs = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                v2 = np.arange(lo2, hi2 + 1)
                if len(v2) > 0:
                    costs[v2] += a[k2i - v2]
            candidates = np.argsort(costs + 5.0 * a)[:n // 4]
            dst = candidates[int(rng.integers(len(candidates)))]
            if dst == src:
                dst = int(rng.integers(n))
                if dst == src:
                    continue
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                J_t = J_of(a_t)
                if J_t < best_J - 1e-14:
                    a = a_t
                    best_J = J_t
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = J_of(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
    return best_a, best_J


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_n)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    global t0_param
    t0_param = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)
    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(12345)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t0_param
            print(f"  *** NEW BEST [{label}]: J={sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_ramp(n_base, 'linear', 1.0)
        best_score = J_of(best_seq)
        print(f"Default init: n={len(best_seq)}, J={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, J={best_score:.6f}")

    # Phase 1: Parametric optimization at base length
    remaining = time_budget - (time.monotonic() - t0_param)
    t1 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 1: Parametric opt (n={n0}, t={t1:.0f}s) ===")
    if t1 > 30:
        a_param, J_param = opt_parametric(n0, t1)
        try_update(a_param, f"param_n{n0}")
        # Also generate diverse ramps
        for shape, param in [('linear', 1.0), ('linear', 2.0), ('quad', 1.0),
                             ('exp', 1.5), ('exp', 3.0), ('sin', 1.5),
                             ('ramp_flat', 0.5)]:
            a_r = make_ramp(n0, shape, param)
            try_update(a_r, f"ramp_{shape}_{param}")

    # Phase 2: Gradient optimization - multiple diverse inits
    remaining = time_budget - (time.monotonic() - t0_param)
    print(f"\n=== Phase 2: Gradient opt (n={n0}, t={remaining*0.45:.0f}s) ===")
    t2_total = remaining * 0.45
    inits = []
    init_labels = []
    if parent is not None:
        inits.append(normalize(parent))
        init_labels.append("parent")
    inits.append(make_ramp(n0, 'linear', 1.5))
    init_labels.append("lin15")
    inits.append(make_ramp(n0, 'exp', 2.0))
    init_labels.append("exp20")
    inits.append(make_ramp(n0, 'quad', 1.0))
    init_labels.append("quad10")
    inits.append(make_ramp(n0, 'sin', 2.0))
    init_labels.append("sin20")
    if best_seq is not None:
        inits.append(normalize(best_seq))
        init_labels.append("best")
    n_inits = min(len(inits), max(3, int(t2_total / 60)))
    t_per_init = max(20.0, t2_total / max(1, n_inits))
    for idx in range(n_inits):
        elapsed_so_far = time.monotonic() - t0_param
        if elapsed_so_far > time_budget - 60:
            break
        t_avail = min(t_per_init, time_budget - 60 - (time.monotonic() - t0_param))
        if t_avail < 15:
            break
        print(f"  GRAD '{init_labels[idx]}' (t={t_avail:.0f}s)...")
        a_out, J_out = grad_optimize(inits[idx], t_avail, T0=5.0, c0=4.0,
                                      restart_every=2000, blend_frac=0.5,
                                      rng_seed=100 + idx * 7, verbose=False)
        try_update(a_out, f"grad_{init_labels[idx]}")

    # Phase 3: Focused gradient with lower temperature on best
    remaining = time_budget - (time.monotonic() - t0_param)
    print(f"\n=== Phase 3: Focused gradient (n={len(best_seq)}, t={remaining*0.25:.0f}s) ===")
    t3 = min(remaining * 0.25, 300.0)
    if t3 > 30:
        a_in = normalize(best_seq)
        a_out, J_out = grad_optimize(a_in, t3, T0=2.0, c0=3.0,
                                      restart_every=3000, blend_frac=0.7,
                                      rng_seed=777, verbose=True)
        try_update(a_out, f"grad_focus_{len(best_seq)}")

    # Phase 4: Coordinate polish
    remaining = time_budget - (time.monotonic() - t0_param)
    print(f"\n=== Phase 4: Coordinate polish (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.12, 120.0)
    if t4 > 15:
        a_in = normalize(best_seq)
        a_out, J_out = coordinate_polish(a_in, t4, rng_seed=999, verbose=False)
        try_update(a_out, f"coord_{len(best_seq)}")

    # Phase 5: Multi-length search
    remaining = time_budget - (time.monotonic() - t0_param)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length (base n={n_best}) ===")
    for delta in [200, -200, 500, -500, 1000]:
        new_n = n_best + delta
        if new_n < 2000 or new_n > 10000:
            continue
        remaining = time_budget - (time.monotonic() - t0_param)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_seq(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")
        t5 = min(remaining * 0.25, 120.0)
        if t5 > 20:
            a_out, _ = grad_optimize(a_long, t5, T0=4.0, c0=4.0,
                                      restart_every=2000, rng_seed=new_n % 1000)
            try_update(a_out, f"grad_n{new_n}")

    # Phase 6: Final combined polish
    remaining = time_budget - (time.monotonic() - t0_param)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_final}, t={remaining:.0f}s) ===")
    if remaining > 30:
        t6a = min(remaining * 0.5, 150.0)
        t6b = remaining - t6a
        if t6a > 20:
            a_in = normalize(best_seq)
            a_out, _ = grad_optimize(a_in, t6a, T0=1.5, c0=2.5,
                                      restart_every=5000, blend_frac=0.8,
                                      rng_seed=31415, verbose=False)
            try_update(a_out, f"grad_final_{n_final}")
        remaining = time_budget - (time.monotonic() - t0_param)
        if remaining > 15:
            t6c = min(remaining * 0.6, 80.0)
            if t6c > 10:
                a_in = normalize(best_seq)
                a_out, _ = coordinate_polish(a_in, t6c, rng_seed=271828)
                try_update(a_out, f"coord_final_{n_final}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: J={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t0_param:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_ramp(n_fb, 'linear', 1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: J={best_score:.6f}")

    return [float(x) for x in best_seq]
