            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))


def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_gradient(a, w, active_idx):
    n = len(a)
    grad = np.zeros(n)
    for wi, k in zip(w, active_idx):
        if wi < 1e-14:
            continue
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad[idx] += 2.0 * wi * a[k - idx]
    return grad


def soft_gradient_step(a, beta, top_k, lr, momentum=None, a_prev=None):
    """One mirror descent step using soft-max gradient."""
    n = len(a)
    ac = fast_autocorr(a)
    
    # Soft-max weights
    ac_max = ac.max()
    exp_ac = np.exp(beta * (ac - ac_max))
    exp_sum = exp_ac.sum()
    w_all = exp_ac / exp_sum
    
    # Select top active lags
    k_sorted = np.argsort(ac)[::-1][:top_k]
    w_active = w_all[k_sorted]
    w_sum = w_active.sum()
    if w_sum < 1e-15:
        w_active = np.ones(top_k) / top_k
    else:
        w_active = w_active / w_sum
    
    # Compute gradient
    grad = compute_gradient(a, w_active, k_sorted)
    
    # Nesterov momentum
    if momentum is not None and a_prev is not None:
        vel = a - a_prev
    else:
        vel = np.zeros(n)
    
    # Multiplicative update
    log_a = np.log(np.maximum(a, 1e-30)) - lr * grad + momentum * vel if momentum else np.log(np.maximum(a, 1e-30)) - lr * grad
    log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-20)
    a_new = a_new / a_new.sum()
    
    return a_new, ac.max()


def soft_mirror_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    beta = 3.0
    top_k = 80
    lr = 2.0
    momentum = 0.0
    stall = 0
    max_stall = 3000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_new, M_new = soft_gradient_step(a, beta, top_k, lr, momentum, None)
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameters
        beta = 3.0 + 0.5 * np.log1p(t / 500.0)
        lr = 2.0 / (1.0 + np.sqrt(t) * 0.1)
        if t > 1000:
            momentum = min(0.3, 0.001 * t)
        
        # Periodic top_k increase
        top_k = min(150, 80 + t // 2000)
        
        if stall > max_stall:
            # Structured restart
            pert_type = int(rng.integers(5))
            if pert_type == 0:
                i_arr = np.arange(n, dtype=np.float64)
                c = float(rng.uniform(0.5, 2.0))
                r = np.exp(-c / n)
                pert = norm1(r ** i_arr)
            elif pert_type == 1:
                i_arr = np.arange(n, dtype=np.float64)
                c = float(rng.uniform(0.5, 2.0))
                r = np.exp(-c / n)
                pert = norm1(r ** (n - 1 - i_arr))
            elif pert_type == 2:
                x_arr = np.linspace(0, 1, n)
                c0 = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.1, 0.4))
                pert = norm1(np.exp(-((x_arr - c0) ** 2) / (2 * sig * sig)))
            elif pert_type == 3:
                noise = rng.normal(0, 0.02, n) * best_a
                pert = norm1(best_a + noise)
            else:
                i_arr = np.arange(n, dtype=np.float64)
                p1 = float(rng.uniform(0.3, 1.0))
                p2 = float(rng.uniform(0.3, 1.0))
                pert = norm1((i_arr + 1) ** p1 * (n - i_arr) ** p2)
            
            mix = float(rng.uniform(0.3, 0.7))
            a = norm1(mix * best_a + (1 - mix) * pert)
            momentum = 0.0
            stall = 0
            if verbose:
                print(f"    SMD restart t={t} ({time.monotonic()-t0:.0f}s): J={2*best_M:.6f}")
        
        a = a_new
        
        if verbose and t % 2000 == 0:
            print(f"    SMD t={t} ({time.monotonic()-t0:.0f}s): J={2*best_M:.6f}, beta={beta:.1f}, lr={lr:.3f}")
    
    return best_a, best_M
