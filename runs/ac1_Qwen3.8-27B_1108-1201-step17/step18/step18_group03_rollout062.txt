Standard Nesterov acceleration for mirror descent:
- Maintain two iterates: y_t (for computing gradient) and x_t (the iterate)
- y_t = (t-1)/(t+2) * (y_{t-1} - x_{t-1}) + x_{t-1} ...

For minimizing f(x) over a convex set with mirror map phi:
- x_0 = y_0 = initial point
- For t = 1, 2, ...:
  - alpha_t = 2/(t+2) (or similar)
  - y_t = (1 - alpha_t) * y_{t-1} + alpha_t * x_{t-1}
  - Compute gradient at y_t: g_t = ∇f(y_t)
  - x_t = argmin_{x in S} { <g_t, x> + (1/eta_t) D_phi(x, y_t) }

For the simplex with negative entropy mirror:
D_phi(x, y) = sum x[i] * log(x[i]/y[i])
The minimizer of <g, x> + (1/eta) * sum x[i]*log(x[i]/y[i]) over the simplex is:
x[i] = y[i] * exp(-eta * g[i]) / sum_j y[j]*exp(-eta*g[j])

So the update is: x_t[i] = y_t[i] * exp(-eta_t * g_t[i]) / Z

Where Z = sum_j y_t[j] * exp(-eta_t * g_t[j]).

This is exactly the multiplicative weight update!

        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 5:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-CW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def batch_coordinate_descent(a0, time_limit, rng_seed, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        improved = False
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            n_pairs = min(6, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * ac[k2i]
                        scores[k2i - idx2] += a[idx2] * ac[k2i]

                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True

                candidate_mask = ~in_peak
                if candidate_mask.sum() < 10:
                    candidate_mask = np.ones(n, dtype=bool)
                candidate_mask[src] = False

                cand_scores = scores.copy()
                cand_scores[~candidate_mask] = np.inf
                dst = np.argmin(cand_scores)

                if dst == src or a[dst] > 0.5 * src_val:
                    continue

                found = False
                for frac in [0.02, 0.04, 0.07, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

                if not found:
                    break

        if not improved:
            stall += 1
            if stall > max_stall:
                break

            if stall % 15 == 0:
                n_moves = int(rng.integers(3, 12))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

            if verbose and stall % 200 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    BatchCoord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        i_arr = np.arange(n_base, dtype=np.float64)
        best_seq = norm_sum1((i_arr + 1.0) ** 0.7)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Parametric search
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 5:
        a_param, _ = parametric_search(n0, t1, best_seq, 2.0 * n0 * fast_autocorr(best_seq).max())
        check_and_update(a_param, "parametric")

    # Phase 2: Smooth mirror descent (main optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Smooth Mirror Descent (n={n_best}) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_mirror_descent(a_in, t2, beta_start=3.0, beta_end=80.0, eta0=1.5, verbose=True)
        check_and_update(a_out, f"smd_n{n_best}")

    # Phase 3: Adaptive smooth mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: SMD Adaptive (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_mirror_descent_adaptive(a_in, t3, verbose=True)
        check_and_update(a_out, f"smd_adapt_n{n_best}")

    # Phase 4: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW cutting plane (n={n_best}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, initial_K=300, max_K=2000,
                                   add_per_iter=50, verbose=True)
        check_and_update(a_out, f"lp_main_n{n_best}")

    # Phase 5: Batch coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Batch coordinate descent (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = batch_coordinate_descent(a_in, t5, rng_seed=77, verbose=True)
        check_and_update(a_out, f"batchcoord_n{n_best}")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    length_candidates = []
    for delta in [500, -500, 1000, -1000, 2000, -2000]:
        new_n = n_best + delta
        if 1500 <= new_n <= 7000:
            length_candidates.append(new_n)
    for rn in [3000, 4000, 5000, 6000]:
        if rn != n_best and 1500 <= rn <= 7000:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.4, 100.0)
        if t6a > 10:
            a_out, _ = smooth_mirror_descent(a_long, t6a, beta_start=3.0, beta_end=60.0, eta0=1.0, verbose=False)
            check_and_update(a_out, f"smd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.4, 80.0)
            if t6b > 10:
                a_in = norm_sum1(best_seq) if len(a_in) == new_n else interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, initial_K=2
