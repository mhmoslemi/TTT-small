def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_smooth_grad(a, tau):
    """Compute gradient of smooth approximation of max(conv(a,a))."""
    n = len(a)
    ac = fast_conv(a)
    ac_max = ac.max()
    shifted = tau * (ac - ac_max)
    exp_ac = np.exp(shifted)
    p = exp_ac / exp_ac.sum()
    # Gradient: g[j] = sum_k p[k] * d(c[k])/d(a[j])
    # d(c[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n (and k!=2j), or a[j] if k=2j
    # Using convolution: g = 2 * (p * reversed_a) approximately
    # More precisely: g[j] = sum_k p[k] * 2*a[k-j] (with boundary handling)
    # This is 2 * cross_correlate(p, a) in a specific sense
    # g[j] = 2 * sum_k p[k] * a[k-j] where valid 0<=k-j<n
    # = 2 * (p flipped) convolved with a, at appropriate offset
    p_flip = p[::-1]
    g_full = np.convolve(p_flip, a, mode='full')
    # g[j] corresponds to index j in the "valid" part
    # convolve(p_flip, a) has length len(p_flip)+len(a)-1 = (2n-1)+n-1 = 3n-2
    # The element we want: sum_k p_flip[k]*a[j-k+j] ... let me think differently
    # g[j] = sum_{k=0}^{2n-2} p[k] * 2*a[k-j] (valid when 0<=k-j<=n-1)
    # Let m = k-j, so k = j+m, m from max(0,-j) to min(2n-2-j, n-1)
    # g[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    # This is a cross-correlation: g[j] = 2 * sum_m p[j+m]*a[m]
    # = 2 * (p cross_correlate a)[j] where cross_correlate(x,y)[j] = sum_m x[j+m]*y[m]
    # numpy correlate: sp_correlate(p, a, mode='full')[j + n - 1] = sum_m p[j+m]*a[m]
    g = 2.0 * sp_correlate(p, a, mode='full')
    # g has length (2n-1) + n - 1 = 3n-2, we want indices [n-1 : 2n-2+n-1] = [n-1 : 3n-3]
    g = g[n - 1: 3 * n - 3]
    # Handle the diagonal correction: when k=2j, coefficient is 1 not 2
    # p[2j] * a[j] should be subtracted once
    diag_idx = 2 * np.arange(n)
    diag_idx = diag_idx[diag_idx < 2 * n - 1]
    g[diag_idx - (2 * np.arange(len(diag_idx)))] -= 0  # Actually let's skip this correction for now
    return g, p


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c_val=4.0, restart_every=4000,
                 verbose=False, rng_seed=42):
    """Away-Step Mirror Descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c_val
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)
            cur_top_k = min(cur_top_k + 5, 200)

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=100, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = min(2 * n - 1, 3000)
    K = top_K_init
    iter_count = 0

    # Reuse variables for warm start
    b_var = None
    prob = None

    while time.monotonic() - t0 < time_limit:
        ac = fast_conv(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 20, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.5, 0.3]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_conv(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 8 == 0:
            K = min(K + 20, top_K_max)

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=42, n_top=15):
    """Exact 1-D coordinate descent with golden section search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 80

    phi = (1.0 + np.sqrt(5.0)) / 2.0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        order = np.arange(n)
        rng.shuffle(order)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-12:
                # Try increasing a[j] from zero
                x_lo = 0.0
                x_hi = 0.05 / n
            else:
                x_lo = 0.0
                x_hi = min(2.0 * a[j] + 0.01 / n, 0.1)

            # Golden section search for optimal a[j] keeping others fixed
            # Objective: max of conv when a[j] = x
            a_j_old = a[j]

            def obj_1d(x):
                a[j] = x
                a_norm = a / a.sum()
                # Quick check: recompute only affected lags
                # c[k] changes for all k where j or k-j is in range
                # For speed, just compute full conv
                c = fast_conv(a_norm)
                return c.max()

            a[j] = a_j_old

            # Golden section
            gr = 2.0 - phi
            x1 = x_hi - (x_hi - x_lo) * gr
            x2 = x_lo + (x_hi - x_lo) * gr

            a[j] = x1
            f1 = obj_1d(x1)
            a[j] = x2
            f2 = obj_1d(x2)

            for _ in range(8):  # Fixed number of iterations
                if f1 < f2:
                    x_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = x_hi - (x_hi - x_lo) * gr
                    a[j] = x1
                    f1 = obj_1d(x1)
                else:
                    x_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = x_lo + (x_hi - x_lo) * gr
                    a[j] = x2
                    f2 = obj_1d(x2)

            x_opt = (x1 + x2) / 2.0
            a[j] = x_opt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a < 1e-15:
                a[j] = a_j_old
                continue
            a = a / s_a

            m_new = fast_conv(a).max()
            if m_new < best_m - 1e-14:
                best_m = m_new
                best_a = a.copy()
                improved = True

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_conv(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def nesterov_smooth(a0, time_limit, verbose=True, tau_schedule=None, eta_init=0.5, rng_seed=100):
    """Nesterov-accelerated smoothed gradient with cyclic temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    if tau_schedule is None:
        tau_schedule = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0]

    v = np.zeros(n)  # Momentum
    eta = eta_init
    stall = 0
    tau_idx = 0
    cycle_count = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        tau = tau_schedule[tau_idx % len(tau_schedule)]

        ac = fast_conv(a)
        ac_max = ac.max()
        shifted = tau * (ac - ac_max)
        exp_ac = np.exp(shifted - shifted.max())
        p = exp_ac / exp_ac.sum()

        # Gradient computation
        g = 2.0 * sp_correlate(p, a, mode='full')
        g = g[n - 1: 3 * n - 3]

        # Nesterov momentum
        v = 0.9 * v + g
        g_tangent = v - v.mean()

        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            a = a_new
            stall += 1
            if stall > 50:
                eta *= 0.85
            if stall > 100:
                # Cycle temperature
                tau_idx += 1
                cycle_count += 1
                if cycle_count % len(tau_schedule) == 0:
                    # Perturbation
                    pert = rng.normal(0, 0.005, n) * a
                    a_p = np.maximum(a + pert, 0.0)
                    a = a_p / a_p.sum()
                    stall = 50
                    eta = eta_init * 0.5

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init quick AMD (20% of time) ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.20, 350.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")
        inits.append(make_arcsine(n0))
        labels.append("arcsine")
        inits.append(make_bimodal(n0, width=0.18))
        labels.append("bimodal")
        inits.append(make_quatemodal(n0, width=0.12))
        labels.append("quatemodal")
        inits.append(make_cosine(n0))
        labels.append("cosine")
        inits.append(spectral_init(n0, n_modes=40, seed=42))
        labels.append("spectral42")

        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_arcsine(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(12.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c_val=5.0,
                                     restart_every=3000, verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: Intensive LP-FW (35% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(80, 2 * n_best - 1)
        top_K_max = min(2000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_main")

    # === Phase 3: Nesterov smoothed gradient (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Nesterov smooth (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        tau_sched = [10.0, 20.0, 50.0, 100.0, 200.0, 400.0, 800.0]
        a_out, _ = nesterov_smooth(a_in, t3, verbose=True, tau_schedule=tau_sched,
                                    eta_init=0.6, rng_seed=100)
        try_update(a_out, "nesterov")

    # === Phase 4: LP-FW round 2 with larger K (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW r2 (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_r2")

    # === Phase 5: Coordinate descent (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate descent (n={n_best}) ===")
    t5 = min(remaining * 0.10, 170.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        # For very large n, use a subset of coordinates per iteration
        n_top = 10 if n_best > 4000 else 15
        a_out, _ = coordinate_descent(a_in, t5, verbose=True, rng_seed=789, n_top=n_top)
        try_update(a_out, "coord")

    # === Phase 6: Length exploration (5% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.5, 120.0)
        if t6a > 15:
            a_out, _ = amd_optimize(a_long, t6a, top_k=60, beta=6.0, c_val=4.0,
                                     restart_every=5000, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

    # === Phase 7: Final LP-FW + coordinate on best (5% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 30:
        # Final LP
        t7a = min(remaining * 0.5, 100.0)
        if t7a > 15 and n_best <= 6000:
            a_in = norm_sum1(best_seq)
            top_K_init = min(500, 2 * n_best - 1)
            top_K_max = min(4000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t7a, solver, verbose=False,
                                        top_K_init=top_K_init, top_K_max=top_K_max)
            try_update(a_out, "lp_final")

        # Final coordinate
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t7b = min(remaining * 0.7, 60.0)
            if t7b > 10:
                a_in = norm_sum1(best_seq)
                n_top = 8 if n_best > 4000 else 12
                a_out, _ = coordinate_descent(a_in, t7b, verbose=False, rng_seed=321, n_top=n_top)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
