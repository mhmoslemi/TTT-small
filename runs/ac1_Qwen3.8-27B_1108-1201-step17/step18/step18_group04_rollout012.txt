    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def subgrad_topk(a, conv, K=40):
    """Exact top-K subgradient of max_k conv_k."""
    n = len(a)
    conv_max = conv.max()
    # Get indices where conv is close to max
    threshold = conv_max - 1e-10
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > K:
        order = np.argsort(conv[active])[::-1][:K]
        active = active[order]
    w = np.ones(len(active)) / len(active)
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad


def subgrad_soft(a, conv, K=50, temp=0.005):
    """Softmax-weighted subgradient for smoother optimization."""
    n = len(a)
    conv_max = conv.max()
    top_idx = np.argsort(conv)[::-1][:K]
    w = np.exp((conv[top_idx] - conv_max) / temp)
    w = w / w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad


def amd_optimize(a0, time_limit, seed=42, K=40, temp=0.004, c0=3.0,
                 restart_every=3000, verbose=False):
    """Nesterov accelerated mirror descent on the simplex."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    restart_count = 0

    for t in range(1, 50_000_000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv_y = fast_conv(y)
        # Alternate between exact and soft subgradient
        if t % 3 == 0:
            g = subgrad_topk(y, conv_y, K)
        else:
            g = subgrad_soft(y, conv_y, K, temp)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.05, 0.01 + 0.005 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            c0 = 3.0 * (1.0 + 0.05 * min(restart_count, 8))

    return best_x, best_m


def peak_shave_cd(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent that specifically targets top convolution peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    n_top_peaks = min(20, 2 * n - 1)

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        top_peaks = np.argsort(conv)[::-1][:n_top_peaks]

        # Compute cost for each position: how much it contributes to top peaks
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        improved = False
        # Find high-cost source positions
        n_src = max(10, n // 200)
        src_positions = np.argsort(costs)[::-1][:n_src]

        for src_i in src_positions:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src_i)
            if a[src_i] < 1e-12:
                continue
            # Find low-cost target positions
            target_candidates = np.argsort(costs)[:n_src * 2]
            for dst_i in target_candidates:
                dst_i = int(dst_i)
                if dst_i == src_i:
                    continue
                for frac in [0.1, 0.2, 0.35, 0.5]:
                    amt = a[src_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def multiplicative_reweight(a0, time_limit, seed=42, alpha_init=0.3, K=30):
    """
    Iterative multiplicative update:
    a_i <- a_i * (T / peak_i)^{alpha}
    where peak_i measures local contribution to max convolution.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    alpha = alpha_init
    stall = 0
    target = best_m

    while time.monotonic() - t0 < time_limit and stall < 100:
        conv = fast_conv(a)
        conv_max = conv.max()
        top_peaks = np.argsort(conv)[::-1][:K]

        # Compute "pressure" on each position from top peaks
        pressure = np.ones(n)
        for k in top_peaks:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            idx = np.arange(lo, hi + 1)
            # Pressure at idx[i] from peak at ki is a[ki - idx[i]]
            local_contrib = a[idx] * a[ki - idx]
            weight = conv[ki] / conv_max
            pressure[lo:hi + 1] += weight * local_contrib

        # Multiplicative update
        target = conv_max * 0.85  # try to reduce max by 15%
        ratio = np.maximum(target / np.maximum(pressure, 1e-15), 0.1)
        a_new = a * np.power(ratio, alpha)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            a = a_new
            best_m = M_new
            best_a = a.copy()
            stall = 0
            # Annotate: decay alpha slowly
            alpha = max(alpha * 0.995, 0.05)
        else:
            a = a_new  # accept anyway to explore
            stall += 1
            if stall % 20 == 0:
                # Periodic reset to best
                a = best_a.copy()
                alpha = max(alpha * 1.1, 0.3)

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_gauss_asym(n, center=0.3, width=0.2):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return norm_sum1(a)


def make_exp_decay_asym(n, rate=2.0, offset=0.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-rate * (x + offset))
    return norm_sum1(a)


def make_ramp_asym(n, power=1.0):
    x = np.linspace(0, 1, n)
    a = np.power(1.0 - x, power)
    return norm_sum1(a)


def make_bell_asym(n, c1=0.25, c2=0.55, w1=0.1, w2=0.15, r1=1.0, r2=0.6):
    x = np.linspace(0, 1, n)
    a = r1 * np.exp(-((x - c1) / w1) ** 2) + r2 * np.exp(-((x - c2) / w2) ** 2)
    return norm_sum1(a)


def make_polynomial_asym(n, p=2, q=1):
    x = np.linspace(0, 1, n)
    a = np.power(x, p) * np.power(1 - x + 1e-6, q)
    return norm_sum1(a)


def make_sin_asym(n, freq=1.0, phase=0.3):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * freq * x + phase * np.pi)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return norm_sum1(np.sin(np.pi * x) ** 2)
    return a / s


def make_random_asym(n, seed=0, n_freq=10):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def try_update(a, label):
    """Track best sequence found."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
