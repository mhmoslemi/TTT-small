def compute_subgradient(a, b, top_k, beta):
    n=len(a)
    l=max(1,min(top_k, 2*n-1))
    # Use argpartition to select unsorted top l
    if l >= len(b):
       idx=np.arange(len(b))
    else:
       idx=np.argpartition(b, -l)[-l:]
       idx=idx[np.argsort(b[idx])[::-1]]
    vals=b[idx]
    vmax=vals.max()
    if vmax <=0:
       return np.zeros(n)
    # weights = (vals/vmax)**beta; if beta maybe 0 -> uniform
    w = np.power(np.maximum(vals, 0.0)/vmax, beta)
    wsum=w.sum()
    if wsum<=0: w=np.ones(len(idx))/len(idx)

def conv_full(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)

def get_fft_size(n):
    if n <= 1:
        return 1
    size = 2 * n - 2
    return 1 << (size - 1).bit_length()


def conv_full(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    if fft_size is None:
        fft_size = get_fft_size(n)
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)
    return out[:2 * n - 1]


def conv_full_max(a, fft_size=None):
    b = conv_full(a, fft_size)
    if b.size == 0:
        return b, 0.0
    return b, float(np.max(b))


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    an = a / s
    _, M = conv_full_max(an)
    return float(2.0 * len(an) * M)

def resample_sequence(a, new_n):
    a = norm_sum1(a)
    old_n = len(a)
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    if old_n == new_n:
        return a.copy()
    f = a * old_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.where(np.isfinite(f_new), f_new, 0.0)
    f_new = np.maximum(f_new, 0.0)
    s = float(f_new.sum())
    if s < 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_new / s

def power_seed(n, alpha, shift=1.0, kind="start"):
    i = np.arange(n, dtype=np.float64)
    if kind == "end":
        a = (n - 1 - i + shift) ** (-alpha)
    else:
        a = (i + shift) ** (-alpha)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-12)
    return norm_sum1(a)


def p_to_sequence(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    pmax = float(p.max())
    if pmax <= 1e-300:
        return None
    p = p / pmax
    if p[0] <= 1e-12:
        return None

    d = np.zeros(n, dtype=np.float64)
    d0 = float(np.sqrt(p[0]))
    d[0] = d0
    denom = 2.0 * d0
    for k in range(1, n):
        if k > 1:
            s = float(np.dot(d[1:k], d[k - 1:0:-1]))
        else:
            s = 0.0
        val = float(p[k] - s)
        if val < -1e-7:
            return None
        if val < 0.0:
            val = 0.0
        dk = val / denom
        if not np.isfinite(dk):
            return None
        d[k] = dk

    ds = float(d.sum())
    if ds <= 1e-300:
        return None
    return d / ds


def recurrence_profile_seeds(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    profiles = []

    profiles.append(("const", np.ones(n, dtype=np.float64)))

    for q in (0.5, 1.0, 2.0, 3.0):
        for depth in (0.1, 0.3, 0.6):
            g = x ** q
            p = 1.0 - depth * g
            p = np.maximum(p, 0.05)
            p = p / p.max()
            profiles.append((f"dec{q}_{depth}", p))

    for f in (1.0, 1.5, 2.0):
        for depth in (0.2, 0.5, 0.8):
            g = np.sin(np.pi * f * x) ** 2
            gm = float(g.max())
            if gm > 1e-12:
                g = g / gm
            p = 1.0 - depth * g
            p = np.maximum(p, 0.05)
            p = p / p.max()
            profiles.append((f"sinfname{f}_{depth}", p))

    for c in (0.2, 0.4, 0.6):
        for sigma in (0.08, 0.15, 0.25):
            for depth in (0.3, 0.6):
                g = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
                gm = float(g.max())
                if gm > 1e-12:
                    g = g / gm
                p = 1.0 - depth * g
                p = np.maximum(p, 0.05)
                p = p / p.max()
                profiles.append((f"gau{c}_{sigma}_{depth}", p))

    for j in range(12):
        f = np.zeros(n, dtype=np.float64)
        for m in range(1, 5):
            amp = float(rng.uniform(0.0, 1.0 / m))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            f += amp * np.cos(2.0 * np.pi * m * x + ph)
        g = np.abs(f) + 0.1
        gm = float(g.max())
        if gm > 1e-12:
            g = g / gm
        depth = float(rng.uniform(0.1, 0.7))
        p = 1.0 - depth * g
        p = np.maximum(p, 0.05)
        p = p / p.max()
        profiles.append((f"rand_{j}", p))

    out = []
    for label, p in profiles:
        a = p_to_sequence(p)
        if a is not None:
            out.append(a)
    return out


def modulated_seeds(n, base, rng, count=8, amp_max=0.35):
    res = []
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    for _ in range(count):
        f = base.copy()
        nmodes = int(rng.integers(1, 4))
        for m in range(1, nmodes + 1):
            amp = float(rng.uniform(0.0, amp_max / m))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            f *= 1.0 + amp * np.cos(2.0 * np.pi * m * x + ph)
        f = np.where(np.isfinite(f), f, 0.0)
        f = np.maximum(f, 1e-12)
        res.append(norm_sum1(f))
    return res


def random_base_seed(n, rng, base=None):
    if base is None:
        base = power_seed(n, 0.5, 1.0, "start")
    x = np.linspace(0.0, 1.0, n)
    f = np.ones(n, dtype=np.float64)
    nmodes = int(rng.integers(2, 5))
    for m in range(1, nmodes + 1):
        amp = float(rng.uniform(0.2, 1.0) / m)
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        f += amp * np.cos(2.0 * np.pi * m * x + ph)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 1e-3)
    f = f / f.sum()
    mix = float(rng.uniform(0.0, 0.7))
    out = (1.0 - mix) * base + mix * f
    return norm_sum1(out)

def compute_subgradient(a, b, top_k, beta):
    n = len(a)
    L = len(b)
    kuse = int(min(max(1, top_k), L))
    if kuse >= L:
        idx = np.arange(L, dtype=np.int64)
    else:
        idx = np.argpartition(b, -kuse)[-kuse:]
        idx = idx.astype(np.int64)
    vals = b[idx]
    vmax = float(vals.max())
    if vmax <= 1e-300:
        return np.zeros(n, dtype=np.float64)

    w = np.power(np.maximum(vals, 0.0) / vmax, beta)
    wsum = float(w.sum())
    if wsum <= 1e-300:
        w = np.ones(len(idx), dtype=np.float64) / len(idx)
    else:
        w = w / wsum

    order = np.argsort(vals)[::-1]
    idx = idx[order]
    w = w[order]

    s = np.zeros(n, dtype=np.float64)
    for ii in range(len(idx)):
        w2 = 2.0 * float(w[ii])
        if w2 == 0.0:
            continue
        k = int(idx[ii])
        start = k - n + 1
        if start < 0:
            start = 0
        end = k + 1
        if end > n:
            end = n
        if start >= end:
            continue
        t_start = k - start
        t_stop = k - end
        if t_stop < 0:
            seg = a[t_start::-1]
        else:
            seg = a[t_start:t_stop:-1]
        s[start:end] += w2 * seg
    return s


def perturb_sequence(a, rng, noise_scale=0.05):
    a = norm_sum1(a)
    n = len(a)
    x = np.linspace(0.0, 1.0, n)
    f = np.ones(n, dtype=np.float64)
    nmodes = int(rng.integers(1, 4))
    for m in range(1, nmodes + 1):
        amp = float(rng.uniform(0.0, 0.5 / m))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        f += amp * np.cos(2.0 * np.pi * m * x + ph)
    f = np.maximum(f, 0.05)
    f = f / f.sum()
    mix = float(rng.uniform(0.05, 0.35))
    p = (1.0 - mix) * a + mix * f
    p = p + float(noise_scale) * rng.normal(0.0, 1.0, size=n) * a
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 1e-12)
    s = float(p.sum())
    if s <= 1e-300:
        return a.copy()
    return p / s


def gradient_mirror_optimize(a0, time_limit, rng_seed=1234, top_k=60, beta=4.0, verbose=True):
    if time_limit <= 0:
        return norm_sum1(a0)

    a = norm_sum1(a0)
    n = len(a)
    fft_size = get_fft_size(n)
    b, M = conv_full_max(a, fft_size)
    best_a = a.copy()
    best_m = M
    t0 = time.monotonic()

    rng = np.random.default_rng(rng_seed)
    temp = 1.0
    max_temp = 4.0
    min_temp = 1e-4
    floor = 1e-14
    stall = 0
    max_stall = 220
    last_best_time = t0
    iter_count = 0
    base_taus = (0.01, 0.03, 0.1, 0.3, 1.0)
    curr_top_k = int(top_k)
    curr_beta = float(beta)

    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        s_grad = compute_subgradient(a, b, curr_top_k, curr_beta)
        s_scale = float(s_grad.max())
        if s_scale <= 1e-30:
            break

        ratio = s_grad / s_scale
        loga_cur = np.log(np.maximum(a, floor))

        good = False
        best_cand = None
        best_cand_b = None
        best_cand_m = M

        for bt in base_taus:
            tau = bt * temp
            if tau <= 0.0:
                continue
            loga = loga_cur - tau * ratio
            loga -= loga.max()
            cand = np.exp(loga)
            cand_sum = float(cand.sum())
            if cand_sum <= 1e-15:
                continue
            cand = cand / cand_sum
            b_c, m_c = conv_full_max(cand, fft_size)
            if m_c < best_cand_m - 1e-14:
                best_cand_m = m_c
                best_cand = cand
                best_cand_b = b_c
            if m_c < M - 1e-14:
                good = True

        if good and best_cand is not None and best_cand_m < M - 1e-14:
            a = best_cand
            b = best_cand_b
            M = best_cand_m
            stall = 0
            temp = min(max(temp * 1.03, 0.05), max_temp)
            if M < best_m - 1e-15:
                best_m = M
                best_a = a.copy()
                last_best_time = time.monotonic()
            iter_count += 1
            if verbose and iter_count % 120 == 0:
                te = time.monotonic() - t0
                print(f"    GM iter={iter_count} t={te:.0f}s J={2.0*n*best_m:.8f} temp={temp:.3f}")
        else:
            stall += 1
            if stall > 5:
                temp = max(temp * 0.5, min_temp)
            if stall > max_stall or time.monotonic() - last_best_time > 35.0:
                a = perturb_sequence(best_a, rng, float(rng.uniform(0.03, 0.15)))
                b, M = conv_full_max(a, fft_size)
                stall = 0
                temp = 1.0
                curr_beta = float(np.clip(curr_beta * float(rng.uniform(0.6, 1.7)), 1.0, 12.0))
                curr_top_k = int(np.clip(int(rng.integers(25, 120)), 15, 200))

    return best_a


def transport_optimize(a0, time_limit, rng_seed=777, verbose=True):

            csum = float(cost.sum())
            if csum <= 1e-300:

            total = frac * ap
            delta[p] -= total
            delta[dst] += (cost / csum) * total

        best_cand = None
        best_cand_b = None
        best_cand_m = M
        for eta in (0.05, 0.1, 0.2, 0.5, 1.0):
            cand = a + eta * delta
            if cand.min() < -1e-10:
                continue
            cand = np.maximum(cand, 0.0)
            cand_sum = float(cand.sum())
            if cand_sum <= 1e-15:

            cand = cand / cand_sum
            b_c, m_c = conv_full_max(cand, fft_size)
            if m_c < best_cand_m - 1e-14:
                best_cand_m = m_c
                best_cand = cand
                best_cand_b = b_c

        if best_cand is not None and best_cand_m < M - 1e-14:
            a = best_cand
            b = best_cand_b
            M = best_cand_m
            stall = 0
            frac = min(0.20, frac * 1.07)
            if M < best_m - 1e-15:
                best_m = M
                best_a = a.copy()
                last_best_time = time.monotonic()
                if verbose:
                    te = time.monotonic() - t0
                    print(f"    TRANS reveal t={te:.0f}s J={2.0*n*best_m:.8f}")
        else:
            stall += 1
            if stall > 20:
                frac = max(0.005, frac * 0.4)
            if stall > max_stall or time.monotonic() - last_best_time > 35.0:
                a = perturb_sequence(best_a, rng, float(rng.uniform(0.03, 0.10)))
                b, M = conv_full_max(a, fft_size)
                frac = 0.05
                stall = 0

    return best_a


def propose_candidate(budget_s=1800.0):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if not np.isfinite(budget):
        budget = 1800.0
    budget = min(max(budget, 60.0), 1800.0)
    total_limit = budget - 45.0
    if total_limit < 30.0:
        total_limit = max(15.0, budget - 5.0)

    t_global = time.monotonic()

    best_a, best_score = load_parent_sequence()
    if best_a is None:
        n_fb = 5200
        best_a = power_seed(n_fb, 0.5, 1.0, "start")
        best_score = local_score(best_a)
        print(f"Default init: n={len(best_a)}, local={best_score:.8f}")

    best_a, best_score = consider_candidate(best_a[::-1], best_a, best_score, "reverse_parent")

    if len(best_a) > 9000:
        bn, sc = consider_candidate(resample_sequence(best_a, 5200), best_a, best_score, "re_to_5200")
    if len(best_a) < 2000:
        best_a, best_score = consider_candidate(resample_sequence(best_a, 5200), best_a, best_score, "re_to_5200")

    n_target = len(best_a)
    n_seed = int(min(1024, max(256, n_target // 5)))

    rem = total_limit - (time.monotonic() - t_global)
    if rem > 30:
        seeds = generate_initial_seeds(n_seed, best_a)
        for it, (sc, a_seed, label) in enumerate(seeds[:18]):
            if rem < 20:
                break
            a_long = resample_sequence(a_seed, n_target)
            best_a, best_score = consider_candidate(a_long, best_a, best_score, "seed_" + label)
            rem = total_limit - (time.monotonic() - t_global)

    rem = total_limit - (time.monotonic() - t_global)
    if rem > 120:
        small_cands = []
        small_cands.append(("parent_small", resample_sequence(best_a, n_seed), 11))
        small_cands.append(("inv_sqrt_small", power_seed(n_seed, 0.5, 1.0, "start"), 12))

        seen = set()
        for it, (sc, a_seed, label) in enumerate(seeds):
            tag = label
            if tag in seen:
                continue
            seen.add(tag)
            small_cands.append((label, a_seed, 100 + it))
            if len(small_cands) >= 5:
                break

        for _ in range(2):
            small_cands.append((f"rand_small_{_}", random_base_seed(n_seed, np.random.default_rng(200 + _)), 300 + _))

        small_cands = small_cands[:6]
        t_each = min(38.0, rem * 0.18 / max(1, len(small_cands)))
        for label, a_seed, sd in small_cands:
            rem = total_limit - (time.monotonic() - t_global)
            if rem < 25 or t_each < 5:
                break
            opt_small = gradient_mirror_optimize(a_seed, t_each, rng_seed=sd, top_k=45, beta=4.0, verbose=False)
            best_a, best_score = consider_candidate(opt_small, best_a, best_score, "small_" + label)
            best_a, best_score = consider_candidate(resample_sequence(opt_small, n_target), best_a, best_score, "small_long_" + label)

    rem = total_limit - (time.monotonic() - t_global)
    long_start = resample_sequence(best_a, n_target)
    if rem > 60:
        t_long = min(520.0, rem * 0.45)
        opt_a = gradient_mirror_optimize(long_start, t_long, rng_seed=2001, top_k=60, beta=4.0, verbose=True)
        best_a, best_score = consider_candidate(opt_a, best_a, best_score, "main_long")

    rem = total_limit - (time.monotonic() - t_global)
    if rem > 60:
        long_start2 = best_a if len(best_a) == n_target else resample_sequence(best_a, n_target)
        t_long2 = min(260.0, rem * 0.30)
        opt_a2 = gradient_mirror_optimize(long_start2, t_long2, rng_seed=3002, top_k=32, beta=2.5, verbose=False)
        best_a, best_score = consider_candidate(opt_a2, best_a, best_score, "long2_lowfocus")

    rem = total_limit - (time.monotonic() - t_global)
    if rem > 70:
        tr_start = best_a if len(best_a) == n_target else resample_sequence(best_a, n_target)
        t_tr = min(180.0, rem * 0.22)
        opt_tr = transport_optimize(tr_start, t_tr, rng_seed=4444, verbose=False)
        best_a, best_score = consider_candidate(opt_tr, best_a, best_score, "transport_long")

    rem = total_limit - (time.monotonic() - t_global)
    if rem > 260:
        if n_target < 8192:
            n_large = 8192
        elif n_target < 12288:
            n_large = 12288
        else:
            n_large = 0
        if n_large > 0:
            large_start = resample_sequence(best_a, n_large)
            best_a, best_score = consider_candidate(large_start, best_a, best_score, "large_resample")
            t_large = min(620.0, rem * 0.55)
            opt_large = gradient_mirror_optimize(large_start, t_large, rng_seed=5555, top_k=90, beta=3.5, verbose=True)
            best_a, best_score = consider_candidate(opt_large, best_a, best_score, "large_opt")

    # Final courtesy candidates: keep both long-target and exact best if different.
    try:
        if len(best_a) != n_target:
            best_a, best_score = consider_candidate(resample_sequence(best_a, n_target), best_a, best_score, "final_resample")
    except Exception:
        pass

    try:
        actual = evaluate_safe(best_a)
        if np.isfinite(actual):
            if actual < best_score:
                best_score = actual
            print(f"FINAL actual evaluator score: {actual:.10f}")
        else:
            print(f"FINAL local score: {best_score:.10f}")
    except Exception:
        print(f"FINAL local score (evaluator unavailable): {best_score:.10f}")

    out = norm_sum1(best_a)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s > 0:
        out = out / s
    if time.monotonic() - t_global > 0:
        pass
    print(f"FINAL returned length: {len(out)}")
    return [float(x) for x in out]
