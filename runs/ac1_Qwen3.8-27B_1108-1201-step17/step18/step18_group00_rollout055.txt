def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def golden_section_minimize(f, a, b, tol=1e-8, max_iter=200):
    """Minimize unimodal f on [a,b] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    x_opt = (a + b) / 2.0
    return x_opt, min(fc, fd)


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

            no_improve += 1

        # Periodic constraint refresh
        if t % refresh_every == 0:
            top = np.argsort(ac_y)[::-1][:max_K // 2]
            active_set = set(int(k) for k in top)

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                top = np.argsort(ac_y)[::-1][:initial_K]
                active_set = set(int(k) for k in top)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Golden section line search from x towards g_dir
        def obj_func(alpha):
            trial = (1.0 - alpha) * x + alpha * g_dir
            trial = np.maximum(trial, 1e-15)
            s_t = trial.sum()
            if s_t < 1e-15:
                return 1e10
            trial = trial / s_t
            return fast_autocorr(trial).max()

        alpha_opt, _ = golden_section_minimize(obj_func, 0.01, 1.0, tol=5e-5, max_iter=30)

        x_prev = x.copy()
        x = (1.0 - alpha_opt) * x + alpha_opt * g_dir
        x = np.maximum(x, 1e-15)
        s_x = x.sum()
        if s_x < 1e-15:
            x = x_prev.copy()
        else:
            x = x / s_x

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW+ iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("  LP-FW+ done (" + str(elapsed) + "s): J=" + repr(J))
    return best_x, best_m


def smoothed_max_gradient_descent(a0, time_limit, T_init=0.1, T_end=0.01,
                                  lr_init=5.0, verbose=True):
    """Mirror descent with log-sum-exp smoothing of the max."""
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    x_prev = x.copy()

    total_iters = 50000

    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = t / total_iters
        T = T_init * (T_end / T_init) ** progress

        # Nesterov lookahead
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)

        # Log-sum-exp smoothing of max
        ac_shifted = ac - ac.max()
        log_sum_exp = ac.max() + np.log(np.sum(np.exp(ac_shifted / T)) * T)

        # Gradient of log-sum-exp WRT y
        weights = np.exp(ac_shifted / T)
        weights = weights / weights.sum()

        # d(log_sum_exp)/dy[j] = sum_k weights[k] * d c_k / dy[j]
        # c_k = sum_i y[i]*y[k-i], so d c_k / dy[j] = y[k-j] (if in range) + y[j]*(if 2j=k)
        # More precisely: d c_k / dy[j] = 2*y[k-j] if j != k/2 and in range, y[k-j] if j = k/2
        # Actually from our earlier analysis: if j in valid range for c_k and j != k-j:
        #   c_k contains 2*y[j]*y[k-j], so d c_k/dy[j] = 2*y[k-j]
        # If j = k-j (k=2j): c_k contains y[j]^2, so d c_k/dy[j] = 2*y[j]

        grad = np.zeros(n)
        for k_idx in range(len(ac)):
            w = weights[k_idx]
            if w < 1e-10:
                continue
            k_int = k_idx
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                partner = k_int - idx
                # d c_k / dy[j] for j in idx: it's y[k-j] = y[partner]
                # But each j appears with weight 2 if j != k-j, or the term is y[j]^2
                # The gradient of c_k = sum_i y[i]*y[k-i] w.r.t. y[j] is:
                # y[k-j] (from term i=j) + y[j]*(1 if k-2j==0 else 0)... 
                # Actually d/dy[j] of sum_i y[i]*y[k-i]:
                # The term with i=j is y[j]*y[k-j], derivative is y[k-j]
                # The term with i=k-j (if different from j) is y[k-j]*y[j], derivative is y[j]
                # So total: y[k-j] + y[j] (if j != k-j), or 2*y[j] (if j = k-j)
                # But y[k-j] = y[partner] where partner = k-j
                grad[idx] += w * (y[partner] + y[partner])  # = 2*w*y[partner] for j != k-j
                # For j where 2j=k, we need to subtract the double count and add 2*y[j]
                # Actually this is getting complicated. Let me use a simpler approach.
        
        # Simpler gradient computation using the fact that:
        # c = a * a (convolution), grad_a of c_k = 2 * (a reversed, shifted)
        # Actually, the gradient of the full convolution c = conv(a,a) w.r.t. a is:
        # grad[k] = sum_j weights[j] * (2 * a.convolve-like contribution)
        # Let me use the FFT trick: if c = ifft(fft(a)^2), then d c_k / da_j involves a cross-correlation.
        # Simplest: just compute it directly for the top-weighted k values.

        # Reset and do it properly
        grad = np.zeros(n)
        top_w_idx = np.argsort(weights)[::-1][:min(200, n + 1)]
        for k_idx in top_w_idx:
            w = weights[k_idx]
            if w < 1e-10:
                continue
            k_int = int(k_idx)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            partner = k_int - idx
            # d c_k / d a[j] for j in idx:
            # From term i=j: contribution is a[k-j] = a[partner]
            # From term i=k-j (if k-j != j and k-j in range): contribution is a[j]
            # But we already counted a[j] in the partner array? No.
            # c_k = sum_{i in range} a[i]*a[k-i]
            # d c_k / d a[j] = (k-j in range) ? a[k-j] : 0  +  (j in range) ? a[j] : 0  ... 
            # Wait: c_k = sum_i a[i]*a[k-i]. 
            # d/d a[j]: the term i=j gives a[k-j]. The term i=k-j gives a[j] (if k-j != j and k-j in range).
            # So d c_k / d a[j] = a[k-j] (if k-j in [0,n-1]) + a[j] (if k-j != j and k-j in [0,n-1])
            # = a[k-j] + (1 if k-j != j else 0) * a[j]  [when k-j in range]
            # But j is in range by construction (j is in idx).
            # k-j = partner, which is in [0, n-1] by construction.
            # So d c_k / d a[j] = a[k-j] + (1 if j != k-j else 0) * a[j]
            # = a[partner] + (1 if 2*j != k else 0) * a[j]
            
            # For j where 2j = k: d c_k/d a[j] = a[j] + a[j] = 2*a[j] (the "1 if j!=k-j" part is 0, but a[k-j]=a[j])
            # Wait: if 2j=k, then partner = k-j = j, so a[partner] = a[j]. And the second term: j != k-j is false, so no addition.
            # So d c_k/d a[j] = a[j] when 2j=k. But the correct derivative of a[j]^2 is 2*a[j]!
            # I think the issue is that when 2j=k, the term in the sum is a[j]*a[j] = a[j]^2, 
            # and its derivative is 2*a[j]. But my formula gives just a[j].
            # The resolution: when i=j and k-j=j, it's the SAME term (not two separate terms).
            # So the sum has only ONE term a[j]*a[j], whose derivative is 2*a[j].
            # My formula: a[k-j] + (1 if j != k-j else 0)*a[j] = a[j] + 0 = a[j]. WRONG!
            # Fix: when j = k-j, the derivative is 2*a[j], not a[j].
            # General formula: d c_k/d a[j] = a[k-j] + (1 if j != k-j else 1)*a[j]
            # = a[k-j] + a[j]  (always, when both j and k-j are in range)
            # When j = k-j: = a[j] + a[j] = 2*a[j]. Correct!
            # When j != k-j: = a[k-j] + a[j]. 
            # But wait, when j != k-j, the sum c_k has TWO terms involving a[j]:
            #   i=j: a[j]*a[k-j], derivative w.r.t. a[j] is a[k-j]
            #   i=k-j: a[k-j]*a[j], derivative w.r.t. a[j] is a[k-j]... no, it's a[j]'s partner.
            # Hmm, I'm confusing myself. Let me just be very explicit.
            # c_k = sum_{i=L}^{R} a[i]*a[k-i] where L=max(0,k-n+1), R=min(k,n-1)
            # d c_k / d a[j] = sum_{i=L}^{R} [delta(i,j)*a[k-i] + a[i]*delta(k-i,j)]
            # = a[k-j] (if j in [L,R]) + a[j] (if k-j in [L,R] and k-j != j... no)
            # = a[k-j] (if j in [L,R]) + a[j] (if k-j in [L,R])
            # The second term: k-j in [L,R] means a[k-j] appears when i=k-j, giving a[k-j]*a[j], 
            # and d/d a[j] of a[k-j]*a[j] = a[k-j].
            # Wait no: d/d a[j] of a[i]*a[k-i] = delta(i,j)*a[k-i] + a[i]*delta(k-i,j)
            # For the second term to be nonzero: k-i = j, i.e., i = k-j.
            # So: d c_k/d a[j] = a[k-j] (from i=j, if j in [L,R]) + a[k-j] (from i=k-j, if k-j in [L,R])
            # Wait, the second term is a[i] where i=k-j, so it's a[k-j].
            # So d c_k/d a[j] = a[k-j] + a[k-j] = 2*a[k-j] (if both j and k-j in [L,R])
            # Or a[k-j] (if only j in [L,R] but k-j not in [L,R])
            # Or a[k-j] (if only k-j in [L,R] but j not in [L,R])... but j is what we're differentiating w.r.t.
            
            # Since j is in [L,R] by construction (j is in idx = [i_start, i_end-1]):
            # d c_k/d a[j] = a[k-j] + (a[k-j] if k-j in [L,R] else 0)
            # = a[k-j] + (a[k-j] if 0 <= k-j <= n-1 else 0)
            # But k-j = partner, and since j in [L,R] and the range is symmetric about k/2, 
            # partner is also in [L,R]. So always: d c_k/d a[j] = 2*a[k-j] = 2*a[partner].
            
            grad[idx] += w * 2.0 * y[partner]

        # Mirror descent update
        lr = lr_init / (np.sqrt(t) + 1.0)
        log_x = np.log(np.maximum(x, 1e-30)) - lr * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        x_prev = x.copy()
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_x = x.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  SMaxGD iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + ", T=" + repr(T))

    return best_x, best_m


def coordinate_descent_polish(a0, time_limit, n_sweeps_max=500, verbose=True):
    """Coordinate descent with 1D optimization per coordinate."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    stall = 0
    max_stall = 80

    for sweep in range(n_sweeps_max):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        # Random coordinate ordering
        order = rng.permutation(n)
        improved_in_sweep = False
        sweep_improvements = 0

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Try moving mass from j to random other positions, and also trying to change a[j]
            # Strategy 1: try a few multiplicative changes to a[j]
            old_val = a[j]
            if old_val < 1e-12:
                # Try adding a small amount from a random source
                src = int(rng.integers(n))
                if src == j or a[src] < 1e-10:
                    continue
                for frac in [0.01, 0.05, 0.1]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_t = fast_autocorr(a_trial).max()
                    if M_t < best_m - 1e-15:
                        a = a_trial
                        best_m = M_t
                        best_a = a.copy()
                        improved_in_sweep = True
                        sweep_improvements += 1
                        break
                continue

            # Try multiplicative changes
            for mult in [0.5, 0.75, 1.25, 1.5, 2.0]:
                a_trial = a.copy()
                a_trial[j] = old_val * mult
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_autocorr(a_trial).max()
                if M_t < best_m - 1e-15:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    improved_in_sweep = True
                    sweep_improvements += 1
                    break

            # Move mass from j to a low-impact position
            if old_val > 1e-8:
                ac = fast_autocorr(a) if sweep_improvements == 0 else None
                if ac is None:
                    ac = fast_autocorr(a)
                # Score each position by its "impact" on the top peaks
                top_peaks = np.argsort(ac)[::-1][:10]
                impact = np.zeros(n)
                for k in top_peaks:
                    k_int = int(k)
                    i_start = max(0, k_int - n + 1)
                    i_end = min(n, k_int + 1)
                    if i_start < i_end:
                        idx = np.arange(i_start, i_end)
                        impact[idx] += a[k_int - idx]
                        impact[k_int - idx] += a[idx]

                # Find low-impact destinations
                low_impact = np.argsort(impact)[:n // 4]
                dst_candidates = [int(x) for x in low_impact if x != j]
                if dst_candidates:
                    dst = dst_candidates[int(rng.integers(len(dst_candidates)))]
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = a[j] * frac
                        a_trial = a.copy()
                        a_trial[j] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_t = fast_autocorr(a_trial).max()
                        if M_t < best_m - 1e-15:
                            a = a_trial
                            best_m = M_t
                            best_a = a.copy()
                            improved_in_sweep = True
                            sweep_improvements += 1
                            break

        if not improved_in_sweep:
            stall += 1
        else:
            stall = 0

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  CoordDescent sweep=" + str(sweep) + " (" + str(elapsed) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def nesterov_fw_lower_bound_check(a, n):
    """Check a few random directions for potential improvement."""
    rng = np.random.default_rng(999)
    best = a.copy()
    best_m = fast_autocorr(a).max()
    for _ in range(20):
        # Random smooth perturbation
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert = 0.1 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        trial = a + pert
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            continue
        trial = trial / s
        m = fast_autocorr(trial).max()
        if m < best_m:
            best = trial
            best_m = m
    return best, best_m


def parametric_init(n, time_limit, verbose=True):
    """Diverse parametric initializations."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = np.ones(n) / n
    best_val = compute_objective(best_a, n)

    candidates = []

    # Power laws (non-symmetric!)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (i_arr + 1.0) ** p
        a = norm_sum1(a)
        candidates.append(a)
        a = (n - i_arr) ** p
        a = norm_sum1(a)
        candidates.append(a)

    # Exponential
    for rate in [0.5, 1, 2, 5, 10]:
        a = np.exp(-rate * x)
        a = norm_sum1(a)
        candidates.append(a)
        a = np.exp(rate * (x - 1.0))
        a = norm_sum1(a)
        candidates.append(a)

    # Bimodal
    for c1 in [0.1, 0.2, 0.3]:
        for c2 in [0.7, 0.8, 0.9]:
            for sigma in [0.03, 0.05, 0.1, 0.15]:
                a = np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2))
                a = norm_sum1(a)
                candidates.append(a)

    # U-shaped
    for p in [1.0, 2.0, 3.0, 5.0]:
        a = x ** p + (1.0 - x) ** p
        a = norm_sum1(a)
        candidates.append(a)

    # Sigmoid-like
    for c in [0.3, 0.5, 0.7]:
        for s in [10, 30, 100]:
            a = 1.0 / (1.0 + np.exp(-s * (x - c)))
            a = norm_sum1(a)
            candidates.append(a)

    # Peak at specific location (non-symmetric)
    for c in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.2]:
            a = np.exp(-((x - c) ** 2) / (2 * sigma ** 2))
            a = norm_sum1(a)
            candidates.append(a)

    # Sawtooth / asymmetric
    for p in [1.0, 2.0]:
        a = (x * (1.0 - x)) ** p * (1.0 + x)
        a = norm_sum1(a)
        candidates.append(a)

    for a in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    elapsed = time.monotonic() - t0
    if verbose:
        print("  Parametric init done (" + str(elapsed) + "s): best J=" + repr(best_val))
    return best_a, best_val

            p = p / p.sum()
            p_score = evaluate_safe(p)

    # Phase 2: Main LP-FW enhanced (35%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 2: Enhanced LP-FW (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60 and n_cur <= 8000:
        a_out, _ = lp_fw_enhanced(norm_sum1(best_seq), t2, solver,
                                  initial_K=400, max_K=3000,
                                  add_per_iter=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(n_cur), t_global)

    # Phase 3: Smoothed max gradient descent (15%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 3: Smoothed max GD (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = smoothed_max_gradient_descent(norm_sum1(best_seq), t3,
                                                 T_init=0.05, T_end=0.005,
                                                 lr_init=8.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smaxgd_n" + str(n_cur), t_global)

    # Phase 4: More LP-FW (25%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 4: LP-FW round 2 (n=" + str(n_cur) + ") ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 60 and n_cur <= 8000:
        a_out, _ = lp_fw_enhanced(norm_sum1(best_seq), t4, solver,
                                  initial_K=500, max_K=4000,
                                  add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(n_cur), t_global)

    # Phase 5: Coordinate descent polish (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 5: Coordinate polish (n=" + str(n_cur) + ") ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t5, n_sweeps_max=300, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(n_cur), t_global)

    # Phase 6: Multi-length exploration (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 6: Multi-length exploration ===")
    longer_ns = [n_cur + 500, n_cur + 1000, n_cur + 2000]
    longer_ns = [l for l in longer_ns if l <= 10000 and remaining > 100]

    for new_n in longer_ns[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  Exploring n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)
        a_long_score = evaluate_safe(a_long)
        if np.isfinite(a_long_score) and a_long_score < best_score:
            best_score = a_long_score
            best_seq = a_long.copy()
            print("  Interpolated to n=" + str(new_n) + ": " + str(best_score))

        t6 = min(remaining * 0.3, 120.0)
        if t6 > 30 and new_n <= 7000:
            a_out, _ = lp_fw_enhanced(a_long, t6, solver,
                                      initial_K=300, max_K=2000,
                                      add_per_iter=20, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_long_n" + str(new_n), t_global)

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 7: Final polish (n=" + str(n_cur) + ") ===")
    if remaining > 40:
        t7a = min(remaining * 0.5, 150.0)
        if t7a > 30 and n_cur <= 8000:
            a_out, _ = lp_fw_enhanced(norm_sum1(best_seq), t7a, solver,
                                      initial_K=500, max_K=3500,
                                      add_per_iter=35, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final_n" + str(n_cur), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.6, 100.0)
            if t7b > 15:
                a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t7b, n_sweeps_max=400, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final_n" + str(n_cur), t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(x ** 2 + 0.1)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
