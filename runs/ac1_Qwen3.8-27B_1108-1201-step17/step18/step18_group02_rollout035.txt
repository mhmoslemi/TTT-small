    size = len(c)
    fft_size = 1 << (size - 1).bit_length()
    Pf = np.fft.rfft(p, fft_size)
    Af = np.fft.rfft(a_flip, fft_size)
    conv_p_flip_a = np.fft.irfft(Pf * Af, fft_size)
    # We need [j] component: sum_k p[k]*a[k-j] for k in [0, 2n-2]
    # This is convolve(p, a)[j + n - 1] ... let me just do it directly
    # grad[j] = sum_k p[k] * a[k-j] where 0<=k<=2n-2 and 0<=k-j<=n-1
    # = sum_{k-j=0}^{n-1} p[k]*a[k-j] with k = j + (k-j), so k ranges from j to j+n-1
    # = sum_{m=0}^{n-1} p[j+m]*a[m] (with p bounded to [0,2n-2])
    # This is like a cross-correlation
    # Let's use: grad = 2 * ifft(fft(p) * conj(fft(a))) shifted ... 
    # Simpler: just compute via the full convolution approach
    size_full = 2 * n - 1
    fft_sz = 1 << (size_full - 1).bit_length()
    Pf2 = np.fft.rfft(p, fft_sz)
    A2f = np.fft.rfft(a, fft_sz)
    # We want sum_k p[k] * a[k-j] for each j
    # This is (flip(p) * a)[j + (2n-2) - (n-1)] ... 
    # Let me just do it more directly with the full conv
    p_flip = p[::-1].copy()
    Pf3 = np.fft.rfft(p_flip, fft_sz)
    conv = np.fft.irfft(Pf3 * A2f, fft_sz)
    # conv[m] = sum_i p_flip[i] * a[m-i] = sum_i p[2n-2-i] * a[m-i]
    # We want sum_k p[k]*a[k-j]. Let k' = 2n-2-k, a[k-j] = a[(2n-2-k')+j-n+1+...]
    # This is getting confusing. Let me just compute it numerically for correctness.
    # Actually the cleanest way: 
    # d/d a[j] of c[k] = a[k-j] (when 0 <= k-j < n)
    # So grad_vec[j] = sum_k p[k] * a[k-j] * 1[0<=k-j<n]
    # = sum_{m=0}^{n-1} p[j+m] * a[m] * 1[0<=j+m<=2n-2]
    # This is a "diagonal" sum that can be computed as:
    # For each j, take p[j:j+n] (bounded) and dot with a[:len]
    # Use Toeplitz or just FFT trick:
    # Define q[j] = sum_{m=0}^{n-1} p[j+m]*a[m] = (a * p)[j] where * is cross-corr
    # Actually this is: convolve(a, p)[j + n - 1] ... no
    # Let me use: itmp = np.zeros(2*n-1); 
    # grad[j] = sum_m a[m]*p[j+m] = the j-th element of (flip(a) * p)
    # (flip(a)*p)[j] = sum_i flip(a)[i]*p[j-i] = sum_i a[n-1-i]*p[j-i]
    # Let i' = n-1-i: = sum_{i'} a[i']*p[j-n+1+i']
    # So (flip(a)*p)[j] = sum_{i'} a[i']*p[j-n+1+i']
    # We want sum_{m} a[m]*p[j+m] = (flip(a)*p)[j] when j-n+1+m = j+m... 
    # No: we want sum_m a[m]*p[j+m]. Let's set k = j+m, m = k-j:
    # = sum_k a[k-j]*p[k] = (a * p)[j] in the "correlation" sense
    # np.correlate(p, a, 'full')[j + n - 1] = sum_k p[k]*a[k - j] ... 
    # I'll just use the direct correlation
    # For efficiency with large n, use FFT:
    # sum_k p[k]*a[k-j] = IDFT[FFT(p) * conjugate(FFT(a))][j] (circular)
    # For linear, we need to be careful. Let's just pad and use the known result.
    
    # Simplest correct approach: use np.correlate equivalent via FFT
    # sum_k p[k] * a[k-j] for j = 0..n-1
    # Pad both to length >= 2n-1
    N = fft_sz
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    # circular correlation: r[j] = sum_k p[k+j]*a[k] = sum_k p[k]*a[k-j] (shifted)
    # np.correlate(p, a, 'full')[m] = sum_k p[k]*a[k-m+n-1]
    # We want sum_k p[k]*a[k-j] = correlate(p, a, 'full')[j + (2n-2) - (n-1)] = correlate(p,a,'full')[n-1+j]
    # Hmm this is getting too complicated. Let me just use the gradient formula directly.
    # 
    # c[k] = sum_i a[i]*a[k-i]
    # dc[k]/da[j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # 
    # Objective: tau * log(sum_k exp(c[k]/tau))
    # d(obj)/da[j] = sum_k p[k] * dc[k]/da[j] = sum_k p[k] * a[k-j] * 1[0<=k-j<n]
    # = sum_{m=0}^{n-1} p[j+m] * a[m]  (where k=j+m, need j+m <= 2n-2)
    
    # This is a Toeplitz-matrix-vector product. Compute via FFT:
    # Define: g[j] = sum_{m=0}^{min(n-1, 2n-2-j)} p[j+m] * a[m]
    # Note this is NOT a standard convolution. It's like taking a sliding dot product.
    # g = T @ a where T[j,m] = p[j+m] for valid j,m
    # T is a (n x n) lower-Hessenberg matrix with T[j,m] = p[j+m]
    # This can be computed as: g[j] = sum_m a[m] * p[j+m]
    # = (a * p)[j] in the "anti-convolution" sense
    # If we reverse a: g[j] = sum_m a[m]*p[j+m] = sum_m a_rev[n-1-m]*p[j+m]
    # Let m' = n-1-m: = sum_{m'} a_rev[m']*p[j+n-1-m']
    # = (a_rev * p)[j+n-1] where * is standard convolution
    a_rev = a[::-1].copy()
    N2 = fft_sz
    ARev_f = np.fft.rfft(a_rev, N2)
    P_f = np.fft.rfft(p, N2)
    conv_ap = np.fft.irfft(ARev_f * P_f, N2)
    # conv_ap[k] = sum_i a_rev[i]*p[k-i]
    # g[j] = conv_ap[j + n - 1]
    g = conv_ap[n - 1: n - 1 + n]
    return 2.0 * g


def accelerated_fw(a0, time_budget, solver, max_constraints=2000, verbose=True):
    """Accelerated Frank-Wolfe for min max_k convolve(y,g)[k] over simplex."""
    n = len(a0)
    y = norm_sum1(a0)
    x = y.copy()
    best_a = y.copy()
    c_init = fast_convolve(y)
    best_M = c_init.max()
    t0 = time.time()
    
    # Initialize constraint set with top lags
    top_k = np.argsort(c_init)[::-1][:200]
    constraint_set = set(int(k) for k in top_k)
    
    iter_count = 0
    no_improve = 0
    
    while time.time() - t0 < time_budget:
        t = iter_count + 1
        c_y = fast_convolve(y)
        M_y = c_y.max()
        if M_y < best_M - 1e-12:
            best_M = M_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Perturb to escape
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.02, n) * best_a
            y_new = best_a + pert
            y = norm_sum1(y_new)
            x = y.copy()
            no_improve = 0
            c_y = fast_convolve(y)
            M_y = c_y.max()
            constraint_set = set(int(k) for k in np.argsort(c_y)[::-1][:200])
            continue
        
        # Build constraint matrix
        clist = sorted(constraint_set)
        if len(clist) > max_constraints:
            # Keep constraints closest to M_y
            margins = np.array([M_y - int(c_y[k]) for k in clist])
            # Actually we want to keep the "tightest" ones (largest c_y[k])
            cvals = c_y[np.array(clist)]
            keep = np.argsort(cvals)[::-1][:max_constraints]
            clist = [clist[i] for i in keep]
            constraint_set = set(clist)
        
        K = len(clist)
        if K == 0:
            break
        
        # Build A matrix: A[k,i] = y[i] if valid for lag clist[k]
        # (y*g)[k] = sum_i y[i]*g[k-i] for valid i
        # For lag k: i ranges from max(0, k-n+1) to min(n-1, k)
        # g index: g[k-i], so if we let j = k-i, then g[j] with j from k-(n-1) to k
        # (y*g)[k] = sum_{j} g[j] * y[k-j] where 0<=j<n and 0<=k-j<n
        #           = sum_{j=max(0,k-n+1)}^{min(n-1,k)} g[j] * y[k-j]
        # So A[k,j] = y[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        
        A_mat = np.zeros((K, n))
        for ii, k in enumerate(clist):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                A_mat[ii, js] = y[k - js]
        
        # Solve LP: min M s.t. sum(g)=1, g>=0, A@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.asarray(g_var.value, dtype=np.float64)
        g_val = np.maximum(g_val, 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        
        # Check for new violations
        c_yg = np.zeros(2 * n - 1)
        # Compute convolve(y, g_dir) via FFT
        fft_sz = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_sz)
        Gf = np.fft.rfft(g_dir, fft_sz)
        conv_yg = np.fft.irfft(Yf * Gf, fft_sz)[:2 * n - 1]
        
        violations = np.where(conv_yg > M_y + 1e-10)[0]
        if len(violations) > 0:
            # Add top violations
            viol_amt = conv_yg[violations] - M_y
            top_v = violations[np.argsort(viol_amt)[::-1][:100]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Accelerated FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        # Also check x
        c_x = fast_convolve(x)
        if c_x.max() < best_M - 1e-12:
            best_M = c_x.max()
            best_a = x.copy()
            no_improve = 0
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.time() - t0
            J = 2.0 * n * best_M
            print(f"    AFW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, LP_M={M_var.value:.6e}")
    
    elapsed = time.time() - t0
    J = 2.0 * n * best_M
    if verbose:
        print(f"    AFW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_M


def smooth_adam_v2(a0, time_budget, tau_init=None, tau_decay=0.9997, lr=0.7,
                   beta1=0.9, beta2=0.999, restart_every=2000, verbose=True):
    """Smoothed multiplicative Adam optimizer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.time()
    rng = np.random.default_rng(77)
    
    if tau_init is None:
        tau_init = max(fast_convolve(a).max() * 0.3, 1e-8)
    tau = tau_init
    tau_min = max(fast_convolve(a).max() * 0.003, 1e-12)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    step = 0
    
    while time.time() - t0 < time_budget:
        step += 1
        c = fast_convolve(a)
        M = c.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        g = lse_gradient(a, tau)
        g = np.maximum(g, 0.0)
        
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        
        bc1 = 1.0 - beta1 ** step
        bc2 = 1.0 - beta2 ** step
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step_size = lr / (np.sqrt(v_c) + 1e-10)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - step_size * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a = a_new / a_new.sum()
        
        if stall > restart_every:
            stall = 0
            # Random restart from best with perturbation
            scale = min(0.1, 0.02 * (stall // restart_every + 1))
            pert = rng.normal(0, scale, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            a = x / x.sum()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.time() - t0
                print(f"    Adam restart ({elapsed:.0f}s): J={2*n*best_M:.6f}, tau={tau:.2e}")
        
        if verbose and step % 200 == 0:
            elapsed = time.time() - t0
            print(f"    Adam step={step} ({elapsed:.0f}s): J={2*n*best_M:.6f}, tau={tau:.2e}")
    
    elapsed = time.time() - t0
    if verbose:
        print(f"    Adam done ({elapsed:.0f}s): J={2*n*best_M:.6f}")
    return best_a, best_M


def coord_descent_v3(a0, time_budget, rng_seed=42, verbose=True):
    """Coordinate descent with optimal transfer and annealed acceptance."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.time()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300
    temperature = 0.01
    
    while time.time() - t0 < time_budget and stall < max_stall:
        c = fast_convolve(a)
        M_curr = c.max()
        
        # Find top peaks
        n_top = min(10, len(c))
        top_k = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:6]:
            if time.time() - t0 > time_budget:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination: minimize impact on top peaks
            scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            cand_scores = scores[candidates]
            candidates = candidates[cand_scores < cand_scores.max() * 0.5]
            if len(candidates) == 0:
                candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            dst = candidates[np.argmin(scores[candidates])]
            if dst == src:
                continue
            
            # Try optimal transfer amount via bisection
            best_frac = None
            best_trial_M = M_curr
            for frac in np.linspace(0.01, 0.3, 8):
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_trial_M:
                    best_trial_M = M_trial
                    best_frac = frac
                    a = a_trial
                    if M_trial < best_M - 1e-16:
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                elif M_trial < best_M + temperature:
                    # Annealed acceptance
                    if rng.random() < math.exp(-(M_trial - best_M) / max(temperature, 1e-10)):
                        a = a_trial
                        temperature *= 0.95
                        stall = 0
                        break
            temperature *= 0.995
        
        if not improved:
            stall += 1
            # Random exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_convolve(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.time() - t0
            print(f"    CD ({elapsed:.0f}s): J={2*n*best_M:.6f}, stall={stall}")
    
    return best_a, best_M


def parametric_init(n):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Inverse sqrt (spectral flatness heuristic)
    a = 1.0 / np.sqrt(i_arr + 1.0)
    candidates.append(a / a.sum())
    a = 1.0 / np.sqrt(n - i_arr)
    candidates.append(a / a.sum())
    a = 1.0 / np.sqrt(np.abs(i_arr - n/2) + 1.0)
    candidates.append(a / a.sum())
    
    # Sin^2 family
    for freq in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.001)
            candidates.append(a / a.sum())
    
    # Power law
    for p in [0.3, 0.5, 0.7, 1.0, 1.5]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())
    
    # Two-mode
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        a = np.zeros(n)
        split = int(n * frac)
        a[:split] = 1.0 / split
        a[split:] = 1.0 / (n - split)
        candidates.append(a / a.sum())
    
    # Gaussian bumps
    for c_pos in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            a = np.exp(-((x - c_pos) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())
    
    # Cosine taper
    a = np.cos(np.pi * (x - 0.5)) ** 2
    candidates.append(a / a.sum())
    a = np.maximum(np.cos(np.pi * (x - 0.5)), 0) ** 0.5
    candidates.append(a / a.sum())
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Chirp-like
    for rate in [1, 3, 5]:
        a = 1.0 / (i_arr + 1.0) ** (1.0 / rate)
        candidates.append(a / a.sum())
    
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4000
        best_seq = parametric_init(n_base)[0]  # inverse sqrt
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.time() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Parametric search (5%) ===
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.time() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        candidates = parametric_init(n0)
        for i, cand in enumerate(candidates):
            if time.time() - t_global > time_budget - t1:
                break
            ac = fast_convolve(cand)
            val = 2.0 * n0 * ac.max()
            if val < 2.0 * n0 * fast_convolve(best_seq).max():
                check_and_update(cand, f"param{i}")
        # Blend
        for w in [0.3, 0.5, 0.7]:
            cand_blend = w * best_seq + (1 - w) * np.ones(n0) / n0
            check_and_update(cand_blend, f"blend_{w}")
    
    n_best = len(best_seq)
    
    # === Phase 2: Accelerated FW (35%) ===
    remaining = time_budget - (time.time() - t_global)
    print(f"\n=== Phase 2: Accel FW (n={n_best}) ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_fw(a_in, t2, solver, max_constraints=1500, verbose=True)
        check_and_update(a_out, f"fw_n{n_best}")
    
    # === Phase 3: Smooth Adam (25%) ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smooth Adam (n={n_best}) ===")
    t3 = min(remaining * 0.25, 380.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam_v2(a_in, t3, tau_decay=0.9996, lr=0.65,
                                  beta1=0.9, beta2=0.999, restart_every=1500, verbose=True)
        check_and_update(a_out, f"adam_n{n_best}")
    
    # === Phase 4: Second FW pass (20%) ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Second FW (n={n_best}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_fw(a_in, t4, solver, max_constraints=2000, verbose=True)
        check_and_update(a_out, f"fw2_n{n_best}")
    
    # === Phase 5: Coordinate descent polish (10%) ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent (n={n_best}) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_v3(a_in, t5, rng_seed=99, verbose=True)
        check_and_update(a_out, f"cd_n{n_best}")
    
    # === Phase 6: Try nearby lengths ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for delta in [300, -300, 500, -500, 200, -200]:
        new_n = n_best + delta
        if 2500 <= new_n <= 6500:
            nearby_lengths.append(new_n)
    
    seen = set()
    unique_lengths = []
    for ln in nearby_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)
    
    for new_n in unique_lengths[:2]:
        remaining = time_budget - (time.time() - t_global)
        if remaining < 200:
            break
        print(f"\n=== Phase 6: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        
        t6a = min(remaining * 0.35, 200.0)
        if t6a > 30:
            a_out, _ = accelerated_fw(a_long, t6a, solver, max_constraints=1000, verbose=False)
            check_and_update(a_out, f"fw_n{new_n}")
            remaining = time_budget - (time.time() - t_global)
        
        if remaining > 100:
            t6b = min(remaining * 0.35, 150.0)
            if t6b > 20:
                a_out, _ = smooth_adam_v2(norm_sum1(best_seq), t6b,
                                          tau_decay=0.9997, lr=0.5,
                                          restart_every=1200, verbose=False)
                check_and_update(a_out, f"adam_n{new_n}")
    
    # === Phase 7: Final polish ===
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 15 and n_best <= 6000:
            a_in = norm_sum1(best_seq)
            a_out, _ = accelerated_fw(a_in, t7, solver, max_constraints=2500, verbose=True)
            check_and_update(a_out, f"fw_final_n{n_best}")
            remaining = time_budget - (time.time() - t_global)
        
        if remaining > 15:
            t7b = min(remaining * 0.5, 80.0)
            if t7b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coord_descent_v3(a_in, t7b, rng_seed=314, verbose=False)
                check_and_update(a_out, f"cd_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.time() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = parametric_init(n_fb)[0]
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
