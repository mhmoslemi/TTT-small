"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_lse_gradient(a, c, beta):
    """Full gradient of logsumexp(beta*c) w.r.t. z (log-param)."""
    n = len(a)
    max_c = c.max()
    w_unnorm = np.exp(beta * (c - max_c))
    S = w_unnorm.sum()
    if S < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w = w_unnorm / S
    
    # h[j] = sum_{i=0}^{n-1} a[i] * w[j+i]
    w_flipped = w[::-1]
    L = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, L)
    WF = np.fft.rfft(w_flipped, L)
    conv_result = np.fft.irfft(A * WF, L)[:3 * n - 2]
    h = conv_result[2 * n - 2 : n - 2 : -1]
    
    g = np.dot(w, c)
    grad = 2.0 * (1.0 - a) * h - 2.0 * a * g
    return grad

def adam_lse_optimize(a0, time_limit, lr=0.05, beta_init=3.0, beta_end=80.0, 
                       beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    n = len(a0)
    a = norm_simplex(a0)
    z = np.log(np.maximum(a, 1e-15))
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    # Annealing: progress over first 30000 iterations
    n_anneal = 30000
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, (t - 1) / n_anneal)
        cur_beta = beta_init * (beta_end / beta_init) ** progress
        cur_lr = lr / (1.0 + 0.1 * np.sqrt(t / 1000.0))
        
        # Compute a from z
        z_shifted = z - z.max()
        a_exp = np.exp(z_shifted)
        s = a_exp.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a_exp / s
        
        # Compute objective and gradient
        c = fast_conv(a)
        M = c.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
        
        # Gradient
        grad = compute_lse_gradient(a, c, cur_beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic perturbation to escape local minima
        if t % 5000 == 0 and t > 10000:
            rng_step = np.random.default_rng(t)
            z += rng_step.normal(0, 0.01, n)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        Adam t={t} ({elapsed:.0f}s): best={2*n*best_m:.6f}, beta={cur_beta:.1f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_conv(a)
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try moving to a random position
            for _ in range(3):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    M_t = fast_conv(a_t).max()
                    if M_t < best_m - 1e-15:
                        a = a_t
                        best_m = M_t
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m

def spectral_refine(a0, time_limit, verbose=False):
    """Iterative spectral flattening."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    step = 0.2
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat mag^4 (i.e., flat mag^2, i.e., flat mag)
        mag_target = np.full_like(mag, mag.mean())
        # Blended: current toward target
        alpha = step
        mag_new = (1 - alpha) * mag + alpha * mag_target
        mag_new = np.maximum(mag_new, 1e-15)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            step = min(0.4, step * 1.02)
        else:
            step = max(0.02, step * 0.8)
            # Revert
            a = norm_simplex(0.7 * a + 0.3 * best_a)
            if step < 0.03:
                break
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
