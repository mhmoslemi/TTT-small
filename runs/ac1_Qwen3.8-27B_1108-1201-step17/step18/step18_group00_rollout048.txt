def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_init(n, n_iters=2000, rng_seed=0):

    fft_size = 1 << (size - 1).bit_length()
    target_amp = np.sqrt(fft_size / size)
    
    # Initialize with a random non-negative sequence
    a = rng.random(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for it in range(n_iters):
        # FFT
        A = np.fft.rfft(a, fft_size)
        # Set magnitude to target, keep phase
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = target_amp * phase
        # Inverse FFT
        a_full = np.fft.irfft(A_new, fft_size)
        a = a_full[:n]
        # Project to non-negative
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:

            s = a.sum()
        a = a / s
        
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    
    return best_a, best_m

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

            if len(active_set) > initial_K:
                constraint_list_tmp = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in constraint_list_tmp])
                keep = np.argsort(margins)[:initial_K]
                active_set = set(int(constraint_list_tmp[i]) for i in keep)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Step size: use line search with multiple candidates
        default_alpha = 2.0 / (t + 2.0)
        alphas_to_try = [default_alpha, default_alpha * 0.5, default_alpha * 1.5,
                         min(1.0, 1.0 / (t + 1.0)), min(1.0, 3.0 / (t + 1.0))]
        alphas_to_try = [max(0.005, min(1.0, a)) for a in alphas_to_try]
        alphas_to_try = list(set(round(a, 6) for a in alphas_to_try))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas_to_try:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(elapsed) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m
