- This is a linear program with O(n) variables and O(n) constraints.

We can approximate this LP with the soft subgradient method. The LP dual is:
- Minimize `sum(w) * M` subject to `w >= 0` and `sum_k w_k * (f*g)_k >= sum(g_j)` for all feasible g...

Actually, the dual of the LP is:
Primal: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
Dual: min M * sum(w) s.t. sum_k w_k * f_contribution_from_j_to_k >= 1 for all j, w >= 0

Where `f_contribution_from_j_to_k = f[k-j] + f[j-k]` (the contribution of g[j] to (f*g)[k]).

So the dual is: min M * sum(w) s.t. (w * f_conv_contribution)[j] >= 1 for all j, w >= 0.

The optimal `w` from the dual gives the weights for the subgradient! And the soft subgradient method is essentially solving this dual approximately.

- `w[k] > 0` only where the constraint is tight (i.e., `(f*g)[k] = M`)
- The complementary slackness condition

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_from_conv(c, n):
    """Compute score from convolution, assuming sum(a)=1."""
    return 2.0 * n * c.max()

def soft_subgradient(a, c, beta):
    """Compute soft subgradient using exp(beta * c) weighting.
    grad[j] = 2 * sum_k w[k] * a[k-j] (symmetric form)
    Actually: grad[j] = sum_k w[k] * (a[k-j] + a[j-k])
    """
    n = len(a)
    w = np.exp(beta * (c - c.max()))
    
    fft_size = 1 << (2 * (2 * n - 1) - 1).bit_length()  # large enough
    # Actually we need fft_size >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n).bit_length()
    
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    # Term B: conv(w, a)[j] = sum_k w[k] * a[j-k] for j=0..n-1
    term_B = np.fft.irfft(W * A, fft_size)[:n]
    
    # Term A: sum_k w[k] * a[k-j] for j=0..n-1
    # This is the cross-correlation
    term_A = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    
    grad = term_A + term_B
    return grad

def spectral_flatten(a, iterations=100, alpha=0.1):
    """Flatten the convolution in frequency domain."""
    n = len(a)
    a = norm_simplex(a)
    best_a = a.copy()

    for _ in range(iterations):

        # We want A[k]^2 to be more concentrated at low k
        # Equivalently, we want |A[k]| to decrease with k (for real, symmetric a)

        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: mag should decrease as a power law

        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_star - idx]

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def soft_subgradient(a, c, beta):
    """Compute soft subgradient: grad[j] = sum_k w[k]*(a[k-j] + a[j-k])
    where w[k] = exp(beta * (c[k] - max(c))).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    w = np.exp(beta * (c - c.max()))
    
    # Need FFT size >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n).bit_length()
    
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    # term_B[j] = sum_k w[k] * a[j-k] = conv(w, a)[j]
    term_B = np.fft.irfft(W * A, fft_size)[:n]
    
    # term_A[j] = sum_k w[k] * a[k-j] = cross-correlation
    term_A = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    
    return term_A + term_B

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    if t2 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = optimize_symmetric(a_in, t2)
        try_update(a_out, "symmetric")
        # Also try symmetric from bowl init
        a_sym_init = 0.5 * (make_init_bowl(n0, 3.0) + make_init_bowl(n0, 3.0)[::-1])
        a_sym_init = norm_simplex(a_sym_init)
        t2b = min(remaining_time() * 0.08, 100.0)
        if t2b > 20:
            a_out2, _ = optimize_symmetric(a_sym_init, t2b)
            try_update(a_out2, "sym_bowl")
    
    # === Phase 3: Spectral flattening ===
    print(f"\n=== Phase 3: Spectral Flattening ===")
    t3 = min(remaining_time() * 0.12, 180.0)
    if t3 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = spectral_flatten(a_in, t3)
        try_update(a_out, "spectral")
        # Phase fix
        t3b = min(remaining_time() * 0.05, 60.0)
        if t3b > 10:
            a_out2, _ = spectral_phase_fix(a_out, t3b)
            try_update(a_out2, "spectral_phase")
    
    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving ===")
    t4 = min(remaining_time() * 0.12, 180.0)
    if t4 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = peak_shave(a_in, t4)
        try_update(a_out, "peak_shave")
    
    # === Phase 5: Differential Evolution ===
    print(f"\n=== Phase 5: Differential Evolution ===")
    t5 = min(remaining_time() * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = differential_evolution(a_in, t5, pop_size=12)
        try_update(a_out, "de")
    
    # === Phase 6: Multi-scale (try different lengths) ===
    print(f"\n=== Phase 6: Multi-Scale ===")
    n_best = len(best_seq)
    for target_n in [n_best + 500, n_best - 500]:
        if target_n < 2000 or target_n > 7000:
            continue
        if remaining_time() < 60:
            break
        print(f"  Trying n={target_n}...")
        a_interp = interpolate_sequence(best_seq, target_n)
        try_update(a_interp, f"interp_n{target_n}")
        
        t6 = min(remaining_time() * 0.15, 120.0)
        if t6 > 20:
            a_out, _ = optimize_soft_mirror(a_interp, t6)
            try_update(a_out, f"sm_n{target_n}")
            if remaining_time() > 20:
                a_out2, _ = peak_shave(a_out, min(30.0, remaining_time() * 0.1))
                try_update(a_out2, f"peak_n{target_n}")
    
    # === Phase 7: Final intensive optimization ===
    print(f"\n=== Phase 7: Final Intensive Optimization ===")
    n_best = len(best_seq)
    t7 = min(remaining_time() * 0.25, 300.0)
    if t7 > 30:
        a_in = norm_simplex(best_seq)
        # Split time between soft mirror and peak shaving
        t7a = t7 * 0.6
        t7b = t7 * 0.4
        
        print(f"  Soft mirror final ({t7a:.0f}s)...")
        a_out, _ = optimize_soft_mirror(a_in, t7a, beta_start=10, beta_end=500, lr_init=0.3)
        try_update(a_out, "final_sm")
        
        if remaining_time() > 15:
            print(f"  Peak shave final ({t7b:.0f}s)...")
            a_out2, _ = peak_shave(a_out, t7b)
            try_update(a_out2, "final_peak")
        
        if remaining_time() > 15:
            print(f"  Spectral final...")
            a_out3, _ = spectral_flatten(a_out2, min(30.0, remaining_time() * 0.3))
            try_update(a_out3, "final_spectral")
    
    # === Phase 8: Final polish with combined pipeline ===
    if remaining_time() > 20:
        print(f"\n=== Phase 8: Final Pipeline ===")
        a_in = norm_simplex(best_seq)
        t8 = remaining_time() * 0.5
        a_out, _ = optimize_soft_mirror(a_in, t8 * 0.5, beta_start=50, beta_end=800, lr_init=0.2)
        try_update(a_out, "pipeline_sm")
        if remaining_time() > 10:
            a_out2, _ = peak_shave(a_out, min(20.0, remaining_time() * 0.3))
            try_update(a_out2, "pipeline_peak")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_init_bowl(n_fb, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
