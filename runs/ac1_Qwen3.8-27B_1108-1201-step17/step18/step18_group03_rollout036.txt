Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax-smoothed mirror descent (multiplicative weights on simplex)
2. FFT-based fast gradient computation  
3. Adaptive softmax parameter and backtracking line search

import numpy as np
import time
import cvxpy as cp

def _next_pow2(n):
    return 1 << (n - 1).bit_length()

def _autocorr_fft(a, fs):
    n = len(a)
    size = 2 * n - 1
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

def _compute_grad(a, w, fs):
    """g[j] = 2 * sum_i a[i] * w[i+j]"""
    n = len(a)
    a_rev = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(a_rev, fs) * np.fft.rfft(w, fs), fs)
    return 2.0 * conv[n-1:2*n-1]

def _norm(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _init_mirror_descent(a_init, time_limit, beta0=50.0, beta1=500.0, 
                         eta0=0.5, verbose=True, symmetric=False, seed=42):
    """
    Mirror descent with softmax-smoothed max objective.
    """
    n = len(a_init)
    fs = _next_pow2(3 * n - 2)
    a = _norm(a_init.copy())
    best_a = a.copy()
    best_score = _autocorr_fft(a, fs).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    eta = eta0
    beta = beta0
    no_improve = 0
    max_no_improve = 2000
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        ac = _autocorr_fft(a, fs)
        M = ac.max()
        
        if M < best_score - 1e-12:
            best_score = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive beta: increase over time
        beta = beta0 + (beta1 - beta0) * min(1.0, it / 5000.0)
        
        # Compute softmax weights
        ac_shift = ac - ac.max()
        exp_ac = np.exp(beta * ac_shift)
        w = exp_ac / exp_ac.sum()
        
        # Compute gradient
        g = _compute_grad(a, w, fs)
        
        # Mirror descent step with backtracking
        improved = False
        for attempt in range(10):

            a_new = np.maximum(a_new, 1e-300)
            s = a_new.sum()
            if s < 1e-30:
                eta *= 0.5
                continue
            a_new = a_new / s
            
            if symmetric:

                a_new = _norm(a_new)
            
            M_new = _autocorr_fft(a_new, fs).max()
            if M_new < M * (1.0 - 1e-10):
                a = a_new
                improved = True
                break
            else:

            eta = eta0 * 0.5

def _lp_polish(a_init, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe polishing."""
    n = len(a_init)
    fs = _next_pow2(2 * n - 1)
    a = _norm(a_init.copy())
    best_a = a.copy()
    best_m = _autocorr_fft(a, fs).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    # Initial constraint set
    ac_y = _autocorr_fft(y, fs)
    M_curr = ac_y.max()
    init_K = min(500, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    max_K = 2000
    add_per_iter = 30
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = _autocorr_fft(y, fs)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = _autocorr_fft(x, fs)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n, k_i + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_i - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        Yf = np.fft.rfft(y, fs)
        Gf = np.fft.rfft(g_dir, fs)
        conv_ys = np.fft.irfft(Yf * Gf, fs)[:2*n-1]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
    
    return best_a, best_m

def _interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    return _norm(a_interp)

def _make_initializations(n, rng):

    for c1 in [0.2, 0.3]:

    # Gaussian center
    for sig in [0.2, 0.3, 0.4]:
        inits.append(_norm(np.exp(-((x-0.5)**2)/(2*sig**2))))
    
    # Sinusoidal
    for freq in [1, 2, 3]:
        inits.append(_norm(np.maximum(np.sin(np.pi * freq * x)**2, 0.001)))
    
    # Exponential
    for rate in [2, 5, 10]:
        inits.append(_norm(np.exp(-rate * x)))
        inits.append(_norm(np.exp(rate * (x - 1))))
    
    # Random
    for _ in range(5):
        inits.append(_norm(rng.exponential(1, n)))
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax-smoothed mirror descent (multiplicative weights) with adaptive parameters
2. FFT-based fast gradient computation
3. Symmetric and asymmetric variants
4. Diverse initialization, multiple lengths, and periodic restarts
5. LP-based Frank-Wolfe final polish
"""

import numpy as np
import time
import cvxpy as cp


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _autocorr_fft(a, fs):
    n = len(a)
    size = 2 * n - 1
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def _compute_grad(a, w, fs):
    n = len(a)
    a_rev = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(a_rev, fs) * np.fft.rfft(w, fs), fs)
    return 2.0 * conv[n - 1:2 * n - 1]


def _norm(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _make_mirror_descent(a_init, time_limit, beta0, beta1, eta0,
                         symmetric, seed, verbose):
    n = len(a_init)
    fs = _next_pow2(3 * n - 2)
    a = _norm(a_init.copy())
    best_a = a.copy()
    best_m = _autocorr_fft(a, fs).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    eta = eta0
    beta = beta0
    no_improve = 0
    max_no_improve = 3000
    uniform = np.ones(n) / n

    for it in range(50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _autocorr_fft(a, fs)
        M = ac.max()

        if M < best_m - 1e-13:
            best_m = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        beta = beta0 + (beta1 - beta0) * min(1.0, it / 10000.0)

        ac_shift = ac - ac.max()
        exp_ac = np.exp(beta * ac_shift)
        w = exp_ac / exp_ac.sum()

        g = _compute_grad(a, w, fs)

        improved = False
        for attempt in range(12):
            a_new = a * np.exp(-eta * g)
            a_new = np.maximum(a_new, 1e-300)
            s = a_new.sum()
            if s < 1e-30:
                eta *= 0.5
                continue
            a_new = a_new / s

            if symmetric:
                a_new = 0.5 * (a_new + a_new[::-1])
                a_new = _norm(a_new)

            M_new = _autocorr_fft(a_new, fs).max()
            if M_new < M * (1.0 - 1e-12):
                a = a_new
                improved = True
                break
            else:
                eta *= 0.5

        if not improved:
            eta = max(eta, 0.0005)

        if no_improve > max_no_improve:
            no_improve = 0
            mix_w = float(rng.uniform(0.1, 0.4))
            a = _norm((1.0 - mix_w) * best_a + mix_w * uniform)
            if symmetric:
                a = 0.5 * (a + a[::-1])
                a = _norm(a)
            eta = eta0
            beta = max(beta0, beta * 0.5)
            if verbose:
                print(f"      [MD] restart it={it} ({time.monotonic()-t0:.0f}s): M={best_m:.8f}")

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      [MD] it={it} ({elapsed:.0f}s): M={best_m:.8f}, beta={beta:.0f}, eta={eta:.5f}")

    return best_a, best_m


def _lp_polish(a_init, time_limit, solver, verbose):
    n = len(a_init)
    fs = _next_pow2(2 * n - 1)
    a = _norm(a_init.copy())
    best_a = a.copy()
    best_m = _autocorr_fft(a, fs).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    ac_y = _autocorr_fft(y, fs)
    M_curr = ac_y.max()
    init_K = min(400, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    max_K = 2500
    add_per_iter = 40

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _autocorr_fft(y, fs)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        ac_x = _autocorr_fft(x, fs)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n, k_i + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_i - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        Yf = np.fft.rfft(y, fs)
        Gf = np.fft.rfft(g_dir, fs)
        conv_ys = np.fft.irfft(Yf * Gf, fs)[:2 * n - 1]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      [LP] t={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    return best_a, best_m


def _get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    print("  Solver: SCS (fallback)")
    return cp.SCS


def _interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    return _norm(a_interp)


def _parametric_inits(n, rng):
    inits = []
    x = np.linspace(0, 1, n)
    i = np.arange(n, dtype=float)

    inits.append(np.ones(n) / n)

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(_norm((i + 1.0) ** p))
        inits.append(_norm((n - i) ** p))

    for a in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(_norm(x ** a + (1.0 - x) ** a))

    for c1 in [0.15, 0.2, 0.25, 0.3]:
        for c2 in [0.7, 0.75, 0.8, 0.85]:
            for sig in [0.03, 0.05, 0.08, 0.12]:
                inits.append(_norm(np.exp(-((x - c1) ** 2) / (2 * sig ** 2)) +
                                   np.exp(-((x - c2) ** 2) / (2 * sig ** 2))))

    for sig in [0.1, 0.15, 0.2, 0.3, 0.4]:
        inits.append(_norm(np.exp(-((x - 0.5) ** 2) / (2 * sig ** 2))))

    for freq in [1, 2, 3, 4]:
        for ph in [0.0, 0.5]:
            inits.append(_norm(np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.001)))

    for rate in [1, 2, 3, 5, 10]:
        inits.append(_norm(np.exp(-rate * x)))
        inits.append(_norm(np.exp(rate * (x - 1.0))))

    # Cosine window variants
    for bw in [0.3, 0.5, 0.7]:
        inits.append(_norm(np.maximum(np.cos(np.pi * (x - 0.5) / bw) ** 2, 0.001)))

    # Random
    for _ in range(8):
        inits.append(_norm(rng.exponential(1.0, n)))
    for _ in range(4):
        inits.append(_norm(rng.normal(0, 1, n) ** 2))

    # Symmetric versions of some
    for a in [0.5, 1.0, 2.0]:
        base = _norm(x ** a + (1.0 - x) ** a)
        inits.append(0.5 * (base + base[::-1]))

    return inits
