```python
    """LP-FW with adaptive constraint addition."""
    n = len(a)
    active_constraints = set()
    # Start with top peaks
    ac = fast_autocorr(a)
    top_idx = np.argsort(ac)[::-1][:min(200, 2*n-1)]
    active_constraints = set(top_idx.tolist())
    
    b = a.copy()  # Initialize with current point
    for iteration in range(100):
        # Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active, b >= 0
        K = len(active_constraints)
        T = build_matrix(a, list(active_constraints), n)
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
        prob.solve()
        b_new = b_var.value
        
        # Check for violated constraints
        ab = np.convolve(a, b_new)
        violations = np.where(ab > M + 1e-8)[0]
        if len(violations) == 0:
            break
        active_constraints.update(violations.tolist())
        if len(active_constraints) > max_constraints:
            # Remove least violated
            all_idx = list(active_constraints)
            vals = [ab[i] for i in all_idx]
            remove_idx = all_idx[np.argmin(vals)]
            active_constraints.discard(remove_idx)
    
    return b_new / b_new.sum()
```
