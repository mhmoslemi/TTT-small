def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_gradient(a, ac, tau):
    n = len(a)
    ac_scaled = ac / tau
    ac_scaled = ac_scaled - ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()
    a_flip = a[::-1].astype(np.float64)
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)


def smooth_adam(a0, time_limit, tau_decay=0.9997, lr=0.7, beta1=0.9, beta2=0.999,
                restart_every=2500, verbose=True):

    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    tau = max(ac.max() * 0.4, 1e-6)
    tau_min = max(ac.max() * 0.003, 1e-10)

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    t_count = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        t_count = t

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        tau = max(tau * tau_decay, tau_min)

        g = lse_gradient(a, ac, tau)

        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)

        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2

        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.04, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SmoothAdam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0

    elapsed = time.monotonic() - t0

        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        M_lp = float(M_var.value) if M_var.value is not None else M_curr

        if t - last_full_check > 10 or t < 5:
            violations = np.where(conv_ys > M_lp + 1e-9)[0]
            if len(violations) > 0:
                viol_amt = conv_ys[violations] - M_lp
                n_add = min(add_per_iter, len(violations))
                top_v = violations[np.argsort(viol_amt)[::-1][:n_add]]
                for k in top_v:
                    constraint_set.add(int(k))
            last_full_check = t

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW3 iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, M_lp={M_lp:.6f}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW3 done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def scipy_polish(a0, time_limit, verbose=True):
    n = len(a0)
    a_init = norm_sum1(a0)
    best_a = a_init.copy()
    best_m = fast_autocorr(a_init).max()
    t0 = time.monotonic()

    x0 = np.log(np.maximum(a_init, 1e-15))
    x0 = x0 - x0.mean()

    def objective_and_grad(x):
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a = exp_x / exp_x.sum()
        ac = fast_autocorr(a)
        tau = 1e-3 * ac.max()
        ac_s = ac / tau
        ac_s = ac_s - ac_s.max()
        exp_ac = np.exp(ac_s)
        p = exp_ac / exp_ac.sum()
        obj = 2.0 * n * tau * (ac_s.max() + np.log(exp_ac.sum())) / 1.0

        a_flip = a[::-1]
        size = len(ac)
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a_flip, fft_size)
        Pf = np.fft.rfft(p.astype(np.float64), fft_size)
        conv_full = np.fft.irfft(Af * Pf, fft_size)
        ac_grad = 2.0 * conv_full[n - 1:n - 1 + n]

        grad_ac = ac_grad / (exp_x.sum()) - ac_grad * np.outer(exp_x / exp_x.sum(), np.ones(n)) / exp_x.sum()
        grad_ac = ac_grad[:n] / exp_x.sum() * (1.0 - exp_x / exp_x.sum())

        g = np.zeros(n)
        g[:] = ac_grad[:n]
        g = g / exp_x.sum() - g * exp_x / (exp_x.sum() ** 2)

        return obj, g

    x_curr = x0.copy()
    tau_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    for tau_frac in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        def obj_smooth(x):
            x_shifted = x - x.max()
            exp_x = np.exp(x_shifted)
            a = exp_x / exp_x.sum()
            ac = fast_autocorr(a)
            tau = tau_frac * ac.max() + 1e-10
            ac_s = ac / tau
            ac_s = ac_s - ac_s.max()
            return 2.0 * n * tau * (ac_s.max() + np.log(np.exp(ac_s).sum()))

        def grad_smooth(x):
            x_shifted = x - x.max()
            exp_x = np.exp(x_shifted)
            a = exp_x / exp_x.sum()
            ac = fast_autocorr(a)
            tau = tau_frac * ac.max() + 1e-10
            ac_s = ac / tau
            ac_s = ac_s - ac_s.max()
            exp_ac = np.exp(ac_s)
            p = exp_ac / exp_ac.sum()

            a_flip = a[::-1]
            size = len(ac)
            fft_size = 1 << (size - 1).bit_length()
            Af = np.fft.rfft(a_flip, fft_size)
            Pf = np.fft.rfft(p.astype(np.float64), fft_size)
            conv_full = np.fft.irfft(Af * Pf, fft_size)
            dc = 2.0 * conv_full[n - 1:n - 1 + n]

            g = dc / exp_x.sum() * (1.0 - exp_x / exp_x.sum())
            return g

        x_curr, f_opt, info = _scipy_single_call(x_curr, obj_smooth, grad_smooth,
                                                  time_limit - (time.monotonic() - t0))
        a_curr = norm_sum1(np.exp(x_curr - x_curr.max()))
        m_curr = fast_autocorr(a_curr).max()
        if m_curr < best_m - 1e-14:
            best_m = m_curr
            best_a = a_curr.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    ScipyPolish tau={tau_frac}: J={2*n*best_m:.6f} ({elapsed:.0f}s)")

    return best_a, best_m


def _scipy_single_call(x0, fun, jac, remaining_time):
    time_start = time.monotonic()

    def callback(xk):
        return (time.monotonic() - time_start) > remaining_time * 0.8

    try:
        res = scipy_minimize(fun, x0, jac=jac, method='L-BFGS-B',
                             options={'maxiter': 500, 'maxfun': 2000,
                                      'callback': callback, 'gtol': 1e-8})
        return res.x, res.fun, res
    except Exception:
        return x0, fun(x0), None


def coord_descent_v2(a0, time_limit, rng_seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 60:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    for freq in [1, 2, 3, 4]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.005)
            candidates.append(a / a.sum())

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())

    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        for beta in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(a / a.sum())

    for c in [0.25, 0.35, 0.5, 0.65, 0.75]:
        for sigma in [0.08, 0.12, 0.18, 0.25, 0.35]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())

    fpower = [1, 2, 3, 4]
    for p in fpower:
        a = (np.cos(np.pi * (x - 0.5)) ** 2) ** p
        candidates.append(a / a.sum())

    candidates.append(np.ones(n) / n)

    for rate in [1, 2, 3, 5]:
        a = 1.0 / (i_arr + 1.0) ** (1.0 / rate)
        candidates.append(a / a.sum())
        a = 1.0 / (n - i_arr) ** (1.0 / rate)
        candidates.append(a / a.sum())

    # Spectral shaping: try to make power spectrum flat
    rng_spec = np.random.default_rng(7)
    for trial in range(5):
        a = rng_spec.dirichlet(np.ones(n) * 0.5)
        candidates.append(a)

    # Bimodal
    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            a = np.exp(-((x - c1) ** 2) / 0.02) + np.exp(-((x - c2) ** 2) / 0.02)
            candidates.append(a / a.sum())

    return candidates


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with improved LP-FW and scipy polish."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric search ===
    print(f"\n=== Phase 1: Parametric search ===")
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        candidates = parametric_init(n0)
        best_param_score = fast_autocorr(best_seq).max()
        for i, cand in enumerate(candidates):
            if time.monotonic() - t_global > time_budget - t1:
                break
            ac = fast_autocorr(cand)
            if ac.max() < best_param_score:
                check_and_update(cand, f"param{i}")
                best_param_score = fast_autocorr(best_seq).max()
        for w in [0.2, 0.4, 0.6, 0.8]:
            cand_blend = w * best_seq + (1 - w) * np.ones(n0) / n0
            cand_blend = norm_sum1(cand_blend)
            check_and_update(cand_blend, f"blend_{w}")

    n_best = len(best_seq)

    # === Phase 2: Smoothed Adam ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Smoothed Adam (n={n_best}) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam(a_in, t2, tau_decay=0.9997, lr=0.7,
                               beta1=0.9, beta2=0.999, restart_every=2000, verbose=True)
        check_and_update(a_out, f"smooth_n{n_best}")

    # === Phase 3: LP-Frank-Wolfe (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW v3 (n={n_best}) ===")
    t3 = min(remaining * 0.42, 700.0)
    if t3 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v3(a_in, t3, solver, initial_K=1000,
                                      max_K=5000, add_per_iter=80, verbose=True)
        check_and_update(a_out, f"lp3_n{n_best}")

    # === Phase 4: Scipy polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Scipy polish (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = scipy_polish(a_in, t4, verbose=True)
        check_and_update(a_out, f"scipy_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length exploration ===")
    nearby_lengths = []
    for delta in [300, -300, 500, -500, 200, -200]:
        new_n = n_best + delta
        if 2500 <= new_n <= 6000:
            nearby_lengths.append(new_n)

    seen = set()
    unique_lengths = []
    for ln in nearby_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    for new_n in unique_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n  Multi-length: n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.3, 100.0)
        if t5a > 20:
            a_out, _ = smooth_adam(a_long, t5a, tau_decay=0.9998, lr=0.5,
                                   beta1=0.9, beta2=0.999, restart_every=1500, verbose=False)
            check_and_update(a_out, f"smooth_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 5500:
            t5b = min(remaining * 0.4, 150.0)
            if t5b > 20:
                a_out, _ = lp_frank_wolfe_v3(norm_sum1(best_seq), t5b, solver,
                                              initial_K=600, max_K=3500,
                                              add_per_iter=60, verbose=False)
                check_and_update(a_out, f"lp3_n{new_n}")

    # === Phase 6: Final LP-FW polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP-FW (n={n_best}) ===")
    if remaining > 60 and n_best <= 5500:
        t6 = min(remaining * 0.4, 200.0)
        if t6 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v3(a_in, t6, solver, initial_K=1200,
                                          max_K=6000, add_per_iter=100, verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord descent (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_v2(a_in, t7, rng_seed=999, verbose=False)
            check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
