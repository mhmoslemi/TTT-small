"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient subgradient descent with Nesterov acceleration and adaptive line search
2. Exploits clean gradient formula: d(conv(a,a)[k])/da[j] = 2*a[k-j] for valid indices
3. Periodic spectral domain smoothing (amplitude compression)
4. Parametric optimization via truncated Fourier representation
5. Active-set peak equalization
6. Multi-length exploration with interpolation
7. Simulated annealing fine-tuning
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_conv(a):
    """FFT-based linear convolution a*a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_max(a):
    """Return only max of conv(a,a) for speed."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return float(c.max())


def norm_simplex(a):
    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * float(c.max())


def evaluate_safe(a):
    """Safely call the external evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def gradient_peak(a, k):
    """
    Gradient of conv(a,a)[k] w.r.t. a.
    Result: grad[j] = 2*a[k-j] for valid indices, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    if j_start >= j_end:
        return grad
    # j ranges over [j_start, j_end), k-j ranges over [k-(j_end-1), k-j_start]
    j_idx = np.arange(j_start, j_end)
    k_minus_j = k - j_idx
    # All k_minus_j values are in [0, n-1] by construction
    grad[j_idx] = 2.0 * a[k_minus_j]
    return grad


def subgradient_top(a, c, top_k=15, beta=30.0):
    """Compute weighted subgradient using top peaks with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()

    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        grad += wi * gradient_peak(a, k_int)
    return grad


def subgradient_descent(a0, time_limit, verbose=False, seed=42):
    """
    Subgradient descent with Nesterov acceleration, adaptive step size,
    and line search. Operates on simplex (sum=1, a>=0).
    """
    n = len(a0)
    a = norm_simplex(a0)
    v = np.zeros(n)  # Nesterov velocity
    best_a = a.copy()
    c = fast_conv(a)
    best_m = float(c.max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Parameters
    eta_init = 0.5 / np.sqrt(n)
    eta_min = 0.01 / np.sqrt(n)
    momentum = 0.9
    top_k_init = 20
    top_k_min = 5
    beta_init = 20.0
    beta_final = 50.0
    n_anneal = 30000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, t / float(n_anneal))
        eta = eta_init * (eta_min / eta_init) ** progress
        cur_top_k = max(top_k_min, int(top_k_init * (1.0 - 0.7 * progress) + top_k_min))
        cur_beta = beta_init + (beta_final - beta_init) * progress
        cur_momentum = 0.8 + 0.15 * progress  # 0.8 -> 0.95

        # Nesterov step
        a_nest = a + cur_momentum * v
        a_nest = np.maximum(a_nest, 0.0)
        s_nest = a_nest.sum()
        if s_nest < 1e-15:
            a_nest = a.copy()
        else:
            a_nest = a_nest / s_nest

        # Compute subgradient at Nesterov point
        c_nest = fast_conv(a_nest)
        g = subgradient_top(a_nest, c_nest, top_k=cur_top_k, beta=cur_beta)

        # Step size normalization
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-30:
            g_norm = 1e-30
        eta_scaled = eta / g_norm

        # Line search: try several step sizes
        best_trial = a
        best_trial_m = best_m
        for mult in [1.0, 0.5, 2.0, 0.25, 4.0]:
            a_trial = a - mult * eta_scaled * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv_max(a_trial)
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_trial = a_trial

        # Update velocity and position
        v = cur_momentum * v + (best_trial - a)
        a = best_trial

        # Track best
        if best_trial_m < best_m - 1e-12:
            best_m = best_trial_m
            best_a = a.copy()

        # Periodic spectral smoothing
        if t % 3000 == 0:
            a_sm = spectral_smooth(a, strength=0.15)
            m_sm = fast_conv_max(a_sm)
            if m_sm < best_m:
                best_m = m_sm
                best_a = a_sm.copy()
                a = a_sm
                v = np.zeros(n)

        # Random perturbation to escape plateaus
        if t % 10000 == 5000:
            pert = rng.normal(0, 0.005, n)
            a_pert = norm_simplex(a + pert)
            m_pert = fast_conv_max(a_pert)
            if m_pert < best_m:
                best_m = m_pert
                best_a = a_pert.copy()
                a = a_pert
                v = np.zeros(n)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SD t={t} ({elapsed:.0f}s): M={best_m:.10f}, J={2*n*best_m:.6f}")

    return best_a, best_m


def spectral_smooth(a, strength=0.2):
    """Compress dynamic range of |A| in frequency domain."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a

    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

    # Blend
    a_out = (1.0 - 0.3 * strength) * a + 0.3 * strength * a_new
    return norm_simplex(a_out)


def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening with adaptive step."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv_max(a)
    t0 = time.monotonic()

    step = 0.25
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        a_new = spectral_smooth(a, strength=step)
        m_new = fast_conv_max(a_new)

        if m_new < best_m - 1e-13:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            step = min(0.4, step * 1.02)
        else:
            a = norm_simplex(0.5 * a + 0.5 * best_a)
            step = max(0.03, step * 0.8)
            if step < 0.04:
                break

    return best_a, best_m


def parametric_optimize(a0, time_limit, n_modes=20, verbose=False):
    """
    Parametric optimization using truncated Fourier series.
    a[i] = 1/n + sum_{m=1}^{M} alpha_m * cos(2*pi*m*i/n + phi_m)
    Ensures non-negativity via clipping after each step.
    """
    n = len(a0)
    a_init = norm_simplex(a0)
    M = n_modes

    def params_to_a(params):
        alpha = params[:M]
        phi = params[M:2*M]
        x = np.arange(n, dtype=np.float64)
        a = np.full(n, 1.0 / n)
        for m in range(1, M + 1):
            a += alpha[m-1] * np.cos(2.0 * np.pi * m * x / n + phi[m-1])
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s

    def objective(params):
        a = params_to_a(params)
        return fast_conv_max(a)

    # Initial parameters: extract from DFT of a_init
    A_init = np.fft.rfft(a_init, n)
    params0 = np.zeros(2 * M)
    for m in range(1, M + 1):
        if m < len(A_init):
            if m == 0:
                continue
            params0[m-1] = 2.0 * np.abs(A_init[m])
            params0[M + m - 1] = np.angle(A_init[m])
        else:
            # Aliased modes - approximate
            m_eff = m % n
            if m_eff < len(A_init) and m_eff > 0:
                params0[m-1] = 2.0 * np.abs(A_init[m_eff]) * 0.5
                params0[M + m - 1] = np.angle(A_init[m_eff])

    t0 = time.monotonic()
    result = scipy_minimize(objective, params0, method='L-BFGS-B',
                            options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-10,
                                    'maxfun': 500000})

    if time.monotonic() - t0 < time_limit - 5:
        # Continue from result
        a_mid = params_to_a(result.x)
        A_mid = np.fft.rfft(a_mid, n)
        params1 = np.zeros(2 * M)
        for m in range(1, M + 1):
            if m < len(A_mid):
                params1[m-1] = 2.0 * np.abs(A_mid[m])
                params1[M + m - 1] = np.angle(A_mid[m])
        result2 = scipy_minimize(objective, params1, method='L-BFGS-B',
                                 options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-10,
                                          'maxfun': 500000})
        if result2.fun < result.fun:
            result = result2

    a_opt = params_to_a(result.x)
    return a_opt, fast_conv_max(a_opt)


def peak_equalize(a0, time_limit, n_peaks=8, verbose=False):
    """
    Active-set peak equalization:
    Identify top K peaks and iteratively push to equalize them.
    """
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv_max(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(a)
        M_curr = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]

        # If top peaks are already close, done
        if top_vals[0] - top_vals[-1] < 1e-8 * top_vals[0]:
            break

        # Compute subgradient weighted by how much each peak exceeds the mean
        peak_mean = top_vals.mean()
        weights = np.maximum(top_vals - peak_mean, 0)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            continue
        weights /= w_sum

        grad = np.zeros(n)
        for wi, k in zip(weights, top_idx):
            if wi > 1e-10:
                grad += wi * gradient_peak(a, int(k))

        # Step with line search
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-30:
            continue

        improved = False
        for lr in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_trial = a - lr * grad / g_norm
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv_max(a_trial)
            if m_trial < best_m - 1e-13:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try adding a small random perturbation
            if it % 100 == 0:
                a_pert = norm_simplex(a + rng.normal(0, 0.003, n))
                m_pert = fast_conv_max(a_pert)
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                else:
                    # Check for stagnation
                    if it % 2000 == 1000:
                        return best_a, best_m

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def annealing_polish(a0, time_limit, verbose=False):
    """
    Simulated annealing in the simplex space.
    Moves mass between random pairs of positions.
    """
    n = len(a0)
    a = norm_simplex(a0).copy()
    best_a = a.copy()
    best_m = fast_conv_max(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    T = 0.05  # initial temperature (fraction of element value)
    T_factor = 0.9995

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        T *= T_factor
        if T < 0.001:
            T = 0.001

        # Pick two random positions
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        # Move fraction f of a[i] to a[j] (and vice versa)
        for direction in [1, -1]:
            frac = rng.uniform(0.01, 0.1) * T * 20
            frac = min(frac, 0.3)
            if direction == 1:
                amt = min(a[i] * frac, a[i])
                a[i] -= amt
                a[j] += amt
            else:
                amt = min(a[j] * frac, a[j])
                a[j] -= amt
                a[i] += amt

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                break
            a = a / s

            m_new = fast_conv_max(a)
            if m_new < best_m - 1e-13:
                best_m = m_new
                best_a = a.copy()
            # Accept or reject (Metropolis)
            if m_new > best_m:
                delta = (m_new - best_m) / (best_m + 1e-30)
                if rng.random() > np.exp(-delta / (T * 0.01 + 1e-15)):
                    # Reject: revert
                    a = best_a.copy()
                    break

        # Multi-move: occasionally try moving to a position far from the peak
        if it % 500 == 250:
            c = fast_conv(a)
            peak_pos = int(np.argmax(c))
            # Find position that contributes most to peak
            j_start = max(0, peak_pos - n + 1)
            j_end = min(n, peak_pos + 1)
            if j_end > j_start:
                idx = np.arange(j_start, j_end)
                contrib = a[idx] * a[peak_pos - idx]
                src = idx[np.argmax(contrib)]
                # Move some mass from src to a random far position
                dst = int(rng.integers(n))
                if dst != src and a[src] > 1e-10:
                    amt = a[src] * rng.uniform(0.02, 0.08)
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        m_new = fast_conv_max(a)
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Anneal it={it} ({elapsed:.0f}s): M={best_m:.10f}, T={T:.4f}")

    return best_a, best_m

        try_update(a_out, "sd_main")

        # Second run with different seed
        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.15, 250.0)
        if t1b > 20 and best_seq is not None:
            # Perturb before second SD run
            rng1 = np.random.default_rng(123)
            perturbed = norm_simplex(best_seq + rng1.normal(0, 0.005, n0))
            a_out2, _ = subgradient_descent(perturbed, t1b, verbose=False, seed=123)
            try_update(a_out2, "sd_perturb")

    # === Phase 2: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 2: Peak Equalization (n={n_best}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_out, _ = peak_equalize(best_seq, t2, n_peaks=10, verbose=False)
        try_update(a_out, "peak_eq")

        # With fewer peaks
        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.08, 120.0)
        if t2b > 15:
            a_out2, _ = peak_equalize(best_seq, t2b, n_peaks=4, verbose=False)
            try_update(a_out2, "peak_eq4")

    # === Phase 3: Spectral smoothing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 3: Spectral Smoothing (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 15:
        a_out, _ = spectral_optimize(best_seq, t3, verbose=False)
        try_update(a_out, "spectral")

        # Then SD after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.08, 130.0)
        if t3b > 15:
            a_out2, _ = subgradient_descent(a_out, t3b, verbose=False, seed=555)
            try_update(a_out2, "sd_after_spectral")

    # === Phase 4: Parametric optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 4: Parametric Optimization (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        try:
            a_out, _ = parametric_optimize(best_seq, t4, n_modes=12, verbose=False)
            try_update(a_out, "parametric")
        except Exception as e:
            print(f"  Parametric failed: {e}")

    # === Phase 5: Annealing polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 5: Annealing Polish (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 15:
        a_out, _ = annealing_polish(best_seq, t5, verbose=True)
        try_update(a_out, "anneal")

    # === Phase 6: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 6: Length Variation ===")

    for delta in [300, 600]:
        new_n = n_best + delta
        if new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t6 = min(remaining * 0.15, 100.0)
        if t6 > 15:
            a_out, _ = subgradient_descent(a_long, t6, verbose=False, seed=777)
            try_update(a_out, f"sd_n{new_n}")

    for delta in [300, 600]:
        new_n = n_best - delta
        if new_n < 2000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}...")
        a_short = interpolate_sequence(best_seq, new_n)
        try_update(a_short, f"interp_short_n{new_n}")

        t6s = min(remaining * 0.12, 100.0)
        if t6s > 15:
            a_out, _ = subgradient_descent(a_short, t6s, verbose=False, seed=888)
            try_update(a_out, f"sd_short_n{new_n}")

    # === Phase 7: Final intensive optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 7: Final Polish (n={n_best}) ===")
    t7 = min(remaining * 0.35, 300.0)
    if t7 > 20:
        # SD with more iterations
        a_out, _ = subgradient_descent(best_seq, t7, verbose=True, seed=2024)
        try_update(a_out, "sd_final")

        # Quick spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_sp, _ = spectral_optimize(a_out, min(30.0, remaining * 0.5), verbose=False)
            try_update(a_sp, "spectral_final")

        # Quick peak equalize
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_pe, _ = peak_equalize(a_out, min(30.0, remaining * 0.5), n_peaks=6)
            try_update(a_pe, "peq_final")

    # === Phase 8: Final annealing on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    t8 = min(remaining * 0.5, 100.0)
    if t8 > 10:
        a_out, _ = annealing_polish(best_seq, t8, verbose=False)
        try_update(a_out, "anneal_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 3000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.maximum(np.cos(np.pi * (x - 0.5)) ** 2, 0.0)
        best_seq = norm_simplex(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
