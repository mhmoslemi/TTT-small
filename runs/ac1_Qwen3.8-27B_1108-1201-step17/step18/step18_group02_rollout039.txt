def smooth_lp_fw(a0, time_limit, solver, init_K=2000, max_K=8000,
                 add_per_iter=100, verbose=True):
    """Frank-Wolfe with LP subproblem and cutting planes."""
    n = len(a0)
    y = a0.astype(np.float64).copy()
    x_avg = y.copy()
    y = y / y.sum()
    x_avg = x_avg / x_avg.sum()
    
    c = fast_conv(y)
    M_curr = c.max()
    best_y = y.copy()
    best_M = M_curr

    top_idx = np.argsort(c)[::-1][:init_K]
    constraints = set(int(k) for k in top_idx)
    
    t0 = time.monotonic()
    iteration = 0
    stall = 0
    best_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        c = fast_conv(y)
        M_curr = c.max()
        
        if M_curr < best_M - 1e-15:
            best_M = M_curr
            best_y = y.copy()

            stall = 0
        else:
            stall += 1
        
        # Check for stall

            # Restart with perturbation
            rng = np.random.default_rng(iteration)
            pert = rng.normal(0, 0.01, n)
            y_new = best_y + pert * best_y
            y_new = np.maximum(y_new, 0)
            s = y_new.sum()
            if s > 1e-10:
                y = y_new / s
            else:
                y = best_y.copy()

            c = fast_conv(y)

            continue
        
        # Manage constraint set
        constraint_list = sorted(constraints)
        if len(constraint_list) > max_K:
            # Remove constraints with largest margin
            margins = np.array([M_curr - c[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraints = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraints)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        A_mat = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):

            i_end = min(n, k_int + 1)

                A_mat[jj, idx] = y[k_int - idx]

            continue

            continue

            continue
        g_dir = g_val / s_g

        violations = np.where(c_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = c_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraints.add(int(k))

            if s_t < 1e-15:

            y_trial = y_trial / s_t
            M_trial = fast_conv(y_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_alpha = trial_alpha
        
        y_new = (1 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
