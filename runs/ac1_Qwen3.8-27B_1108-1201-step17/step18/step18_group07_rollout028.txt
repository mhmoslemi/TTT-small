If `k*` is a unique argmax: `grad = 2 * Q_{k*} * a` (where Q_k is the matrix above)
If multiple argmaxes: `grad = sum w_k * 2 * Q_k * a` for convex combination

`a^(t+1) = proj_simplex( a^(t) - eta * grad^(t) )`

where `proj_simplex` projects onto `{a >= 0, sum(a) = 1}`.

The optimal step size for a quadratic is `eta = 2 / L` where `L` is the Lipschitz constant of the gradient. For our problem, `L` depends on the matrix norm of `Q_k`.

import numpy as np

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    """Project onto probability simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def evaluate_safe(a):

def build_constraint_rows(f, k_values):
    """Build sparse rows of the constraint matrix for given k values."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(k_values):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        cols_k = np.arange(j_start, j_end)
        rows_k = np.full(j_end - j_start, row_idx)
        vals_k = f[k_int - cols_k]
        rows.append(rows_k)
        cols.append(cols_k)
        vals.append(vals_k)
    
    if not rows:
        return None
    
    all_rows = np.concatenate(rows)
    all_cols = np.concatenate(cols)
    all_vals = np.concatenate(vals)
    
    n_cons = len(k_values)
    return csr_matrix((all_vals, (all_rows, all_cols)), shape=(n_cons, n))

        M_new = fast_conv(a_new).max()

            best_a = a_new.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based iterative refinement with cutting-plane (reduced constraints)
2. FISTA with annealed log-sum-exp smoothing
3. Spectral domain amplitude compression

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely evaluate via the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_constraint_matrix(f, k_values):
    """Build sparse constraint matrix rows for selected k values.
    Row k: A[k,j] = f[k-j] for valid j (0 <= k-j < n)."""
    n = len(f)
    all_rows = []
    all_cols = []
    all_vals = []
    for row_idx, k in enumerate(k_values):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        cols_k = np.arange(j_start, j_end)
        rows_k = np.full(j_end - j_start, row_idx)
        vals_k = f[k_int - cols_k]
        all_rows.append(rows_k)
        all_cols.append(cols_k)
        all_vals.append(vals_k)
    if not all_rows:
        return None
    r = np.concatenate(all_rows)
    c = np.concatenate(all_cols)
    v = np.concatenate(all_vals)
    n_cons = len(k_values)
    return csr_matrix((v, (r, c)), shape=(n_cons, n))


def lp_direction(f, M, n_top=300):
    """Solve reduced LP: max sum(g) s.t. f*g <= M (top n_top constraints), g >= 0."""
    n = len(f)
    c = fast_conv(f)
    top_k = np.argsort(c)[::-1][:min(n_top, 2 * n - 1)]
    A_ub = build_constraint_matrix(f, top_k)
    if A_ub is None:
        return None
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_k), M)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.status == 0:
            g = res.x
            if g is not None and np.isfinite(g).all() and g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def fista_optimize(a0, time_limit, verbose=False):
    """FISTA with annealed log-sum-exp smoothing for the max."""
    n = len(a0)
    a = norm_simplex(a0).copy()
    y = a.copy()
    t_k = 1.0
    t0 = time.monotonic()

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / 80000.0)
        beta = 30.0 + 270.0 * progress
        cur_top_k = max(3, int(40 * (1 - 0.85 * progress) + 3))

        # Compute smoothed gradient
        c = fast_conv(y)
        top_idx = np.argsort(c)[::-1][:cur_top_k]
        top_vals = c[top_idx]
        max_v = top_vals[0] if top_vals[0] > 1e-30 else 1.0

        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(cur_top_k) / cur_top_k
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * y[k_int - idx]

        # FISTA step
        L = n * 1.2
        a_new = y - grad / L
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + (t_k - 1.0) / t_new * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a_new.copy()
        else:
            y = y / sy

        a = a_new
        t_k = t_new

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it} ({elapsed:.0f}s): best_M={best_M:.8f}, score={2*n*best_M:.6f}")

    return best_a, best_M

    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for f_mode in range(1, 20):
        coeff = rng.normal(0, 0.15 / np.sqrt(f_mode))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f_mode * x + phase)
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate():
    """Main search orchestrator combining LP, FISTA, spectral, and CD methods."""
    t_global = time.monotonic()
    time_budget = 1800.0 - 40.0

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_init_bowl(n_base, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    print(f"Starting: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 1: Try diverse initializations with quick LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)
    print(f"\n=== Phase 1: Diverse initializations + quick LP (budget={remaining:.0f}s) ===")

    inits = []
    labels = []
    if parent is not None:
        inits.append(norm_simplex(parent))
        labels.append("parent")
    inits.append(make_init_bowl(n0, sharpness=2.5))
    labels.append("bowl2.5")
    inits.append(make_init_bowl(n0, sharpness=3.5))
    labels.append("bowl3.5")
    inits.append(make_init_bowl(n0, sharpness=1.5))
    labels.append("bowl1.5")
    inits.append(make_init_dual_peak(n0, 0.15, 0.85, 0.1))
    labels.append("dual15")
    inits.append(make_init_dual_peak(n0, 0.25, 0.75, 0.15))
    labels.append("dual25")
    inits.append(make_init_sinc(n0, bw=7.0))
    labels.append("sinc7")
    inits.append(make_init_random_smooth(n0, seed=123))
    labels.append("rand123")

    t_per_init = min(remaining * 0.35 / len(inits), 60.0)
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed_so_far = time.monotonic() - t_global
        remaining_now = time_budget - 40 - elapsed_so_far
        if remaining_now < 15:
            break
        t_avail = min(t_per_init, remaining_now)
        if t_avail < 5:
            break
        print(f"  LP '{lbl}' (budget={t_avail:.0f}s)...")
        a_out, _ = lp_optimize(init, t_avail, verbose=False)
        try_update(a_out, f"lp_{lbl}")

    # === Phase 2: Intensive LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Intensive LP (n={n_best}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 500.0)
    if t2 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = lp_optimize(a_in, t2, verbose=True)
        try_update(a_out, f"lp_intensive_n{n_best}")

        # Quick spectral polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_sp, _ = spectral_optimize(a_out, min(30.0, remaining * 0.1), verbose=False)
            try_update(a_sp, f"spectral_n{n_best}")

    # === Phase 3: FISTA refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: FISTA (n={n_best}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 200.0)

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: CD (n={n_best}, budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 100.0)
    if t5 > 10:

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length variation (budget={remaining:.0f}s) ===")
    for delta in [500, 1000, 1500]:
        new_n = n_best + delta
        if new_n > 7000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")
        t6 = min(remaining * 0.3, 100.0)

            a_out, _ = lp_optimize(a_long, t6, verbose=False)
            try_update(a_out, f"lp_n{new_n}")

    # Try shorter
    for delta in [500, 1000]:
        new_n = n_best - delta
        if new_n < 2000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}...")
        a_short = interpolate_sequence(best_seq, new_n)
        try_update(a_short, f"interp_short_n{new_n}")
        t6s = min(remaining * 0.25, 80.0)
        if t6s > 10:

    # === Phase 7: Final intensive LP + polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}, budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.50, 250.0)
    if t7 > 15:
        a_in = norm_simplex(best_seq)
        a_out, _ = lp_optimize(a_in, t7, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_sp, _ = spectral_optimize(a_out, min(20.0, remaining * 0.3), verbose=False)
            try_update(a_sp, f"spectral_final_n{n_best}")
            a_cd, _ = coordinate_descent(a_sp, min(20.0, remaining * 0.3), verbose=False)
            try_update(a_cd, f"cd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_bowl(n_fb, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
