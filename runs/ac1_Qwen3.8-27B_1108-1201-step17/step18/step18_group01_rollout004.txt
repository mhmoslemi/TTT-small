```python
import cvxpy as cp

def lp_update(a, M, n):
    # max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k
    b = cp.Variable(n, nonneg=True)
    
    # Constraint matrix: row k has a[i] at column j where i+j=k
    # (a*b)[k] = sum_{i+j=k} a[i]*b[j] = sum_j a[k-j]*b[j] (where 0<=k-j<n)
    # This is a Toeplitz matrix times b
    
    # Build the Toeplitz matrix
    # Row k: [0,...,0, a[k], a[k-1], ..., a[0], 0, ..., 0, a[1], ..., a[k-n+1], 0, ..., 0]
    # Actually, (a*b)[k] = sum_j b[j] * a[k-j] where 0<=k-j<n
    # So the k-th row of the constraint matrix has: for j=0..n-1, entry = a[k-j] if 0<=k-j<n else 0
    
    # This is a convolution matrix (Toeplitz)
    # We can build it using scipy
    
    from scipy.sparse import csr_matrix
    # For each row k (0 to 2n-2), the non-zero entries are:
    # j from max(0, k-n+1) to min(n-1, k)
    # entry at column j is a[k-j]
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if prob.status == 'optimal':
            return b.value
    except:
        pass
    
    # Try other solvers
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status == 'optimal':
            return b.value
    except:
        pass
    
    return None
```

```python
def fast_conv(a):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_crosconv(a, g):
    """Cross-correlation: result[i] = sum_j g[i+j]*a[j] for valid indices.
    Returns array of length n.
    Computed as conv(a, flip(g))[n-1:2n-1][::-1].
    """
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    n = len(a)
    m = len(g)  # should be 2n-1
    size = n + m - 1

    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g[::-1], fft_size)
    c = np.fft.irfft(A * G, fft_size)[:size]
```
