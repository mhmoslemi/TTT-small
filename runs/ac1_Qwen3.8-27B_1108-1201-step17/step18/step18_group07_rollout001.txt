"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based alternating optimization (constraint generation)
2. Inverse-convolution fast updates (Weiszfeld-like)
3. Spectral amplitude flattening
4. Coordinate descent local search
5. Multi-length and multi-restart strategy
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_conv(a, b):
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_obj_simplex(a):
    """2*n*max(conv(a,a)) for sum(a)=1, a>=0."""
    n = len(a)
    return 2.0 * n * fast_conv(a).max()


def eval_safe(a_arr):
    try:
        a_list = [float(x) for x in np.asarray(a_arr, dtype=np.float64).ravel()]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def solve_lp_direction(f, M, n_sphere=100, max_rounds=3, top_k_init=30):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for selected k.
    Uses constraint generation."""
    n = len(f)
    
    # Initial set of constraints: top peaks
    c_self = fast_conv(f)
    top_idx = np.argsort(c_self)[::-1][:top_k_init]
    
    g = None
    for round_num in range(max_rounds):
        # Build constraint matrix
        A = np.zeros((len(top_idx), n))
        for row, k in enumerate(top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            i_range = np.arange(i_start, i_end)
            j_range = k_int - i_range
            A[row, j_range] = f[i_range]
        
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(len(top_idx), M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        
        if not res.success:
            if g is not None:
                return g
            break
        
        g = res.x
        
        # Check for violated constraints
        fg = cross_conv(f, g)
        violated = np.where(fg > M * 1.001)[0]
        
        if len(violated) == 0:
            return g
        
        # Add most violated constraints
        viol_sorted = violated[np.argsort(fg[violated])[::-1]]
        new_idx = np.unique(np.concatenate([top_idx, viol_sorted[:50]]))
        top_idx = new_idx[:n_sphere]
    
    return g


def fast_inverse_update(f, M, n_peaks=50, beta=20.0):
    """Fast Weiszfeld-like update: g[i] ∝ 1/(sum_k w_k f[k-i])."""
    n = len(f)
    c_self = fast_conv(f)
    
    # Select top peaks with softmax weights
    top_idx = np.argsort(c_self)[::-1][:n_peaks]
    top_vals = c_self[top_idx]
    max_v = top_vals[0]
    exponents = beta * (top_vals - max_v) / max_v
    exponents = np.maximum(exponents, -50.0)
    w = np.exp(exponents)
    w = w / w.sum()
    
    # Compute g[i] = 1 / (sum_k w_k * f[k-i])
    pressure = np.ones(n) * 1e-10
    for wi, k in zip(w, top_idx):
        if wi < 1e-10:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        i_range = np.arange(i_start, i_end)
        j_range = k_int - i_range
        pressure[i_range] += wi * f[j_range]
    
    g = 1.0 / (pressure + 1e-15)
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return g / s


def spectral_smooth(a, alpha=0.3):
    """Compress dynamic range of |FFT(a)|."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.maximum(ratio, 0) ** (1.0 - alpha)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    blend = (1.0 - 0.5 * alpha) * a + 0.5 * alpha * a_new
    return norm_simplex(blend)


def lp_optimize(a0, time_limit, n_sphere=100, max_rounds=3, top_k_init=30,
                blend_rate=0.15, spectral_every=20, verbose=False):
    """Main LP-based optimization loop."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    obj_best = 2.0 * n * best_M
    t0 = time.monotonic()
    
    it = 0
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1
        M_curr = fast_conv(a).max()
        
        # Adaptive blend rate
        t_blend = blend_rate * (1.0 + 0.5 * np.cos(2 * np.pi * it / 200))
        t_blend = max(0.03, min(0.4, t_blend))
        
        # Solve LP for direction
        g = solve_lp_direction(a, M_curr, n_sphere, max_rounds, top_k_init)
        
        if g is None:
            # Fallback to fast inverse update
            g = fast_inverse_update(a, M_curr)
        
        g = norm_simplex(g)
        
        # Update
        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Evaluate
        M_new = fast_conv(a_new).max()
        
        if M_new < best_M * (1.0 - 1e-10):
            best_M = M_new
            best_a = a_new.copy()
            obj_best = 2.0 * n * best_M
            stall = 0
            a = a_new
        else:
            # Accept if slightly worse (to allow exploration)
            if M_new < best_M * 1.01:
                a = a_new
            else:
                a = best_a.copy()
            stall += 1
        
        # Periodic spectral smoothing
        if it % spectral_every == 0:
            a_spec = spectral_smooth(best_a, alpha=0.2 + 0.1 * np.sin(it / 50.0))
            M_spec = fast_conv(a_spec).max()
            if M_spec < best_M:
                best_M = M_spec
                best_a = a_spec.copy()
                obj_best = 2.0 * n * best_M
                stall = 0
                a = a_spec.copy()
        
        if verbose and it % 50 == 0:
            print(f"    LP it={it} ({time.monotonic()-t0:.0f}s): obj={2*n*best_M:.6f}, M={best_M:.8f}")
    
    return best_a, obj_best


def coordinate_descent(a0, time_limit, n_top=10, n_trials=5):
    """Local search: move mass between positions to reduce peak."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_conv(a)
        
        # Find positions contributing most to top peaks
        top_peaks = np.argsort(c)[::-1][:n_top]
        contributors = set()
        for k in top_peaks:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                contrib = a[i] * a[k_int - i]
                if contrib > c.max() * 0.3:
                    contributors.add(i)
        
        contributors = list(contributors)
        if not contributors:
            contributors = list(range(n))
        
        improved = False
        for _ in range(n_trials * len(contributors)):
            if time.monotonic() - t0 > time_limit:
                break
            src = int(rng.choice(contributors))
            dst = int(rng.integers(n))
            if src == dst:
                continue
            frac = rng.choice([0.02, 0.05, 0.1, 0.2])
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a_trial.copy()
                improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.01, 0.03)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, 2.0 * n * best_M


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    return norm_simplex(np.maximum(a, 0.0))


def make_bowl(n, sharp=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharp + 0.1
    return norm_simplex(a)


def make_dual_peak(n, p1=0.25, p2=0.75, w=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - p1)**2) / (2*w**2)) + np.exp(-((x - p2)**2) / (2*w**2)) + 0.05
    return norm_simplex(a)
