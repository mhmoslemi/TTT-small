OK this is getting too complicated in my head. Let me just implement a clean version:

For the gradient of the max (or soft-max) of the autocorrelation, I'll compute it as:
- For each of the top K peak lags k, compute dc_k/da (which is a vector of length n)
- Weight them and sum

The gradient dc_k/da[j] for lag k (0 <= k < n, since c_k = c_{2n-2-k}):
- For j in [0, n-1-k]: dc_k/da[j] = a[j+k]
- For j in [k, n-1]: dc_k/da[j] = a[j-k]
- (Both apply for j in [k, n-1-k])

So: dc_k/da = [a[k], a[k+1], ..., a[n-1], a[n-2-k], ..., a[1], a[0], 0, ..., 0]

Wait, let me be more careful. c_k = sum_{i=0}^{n-1-k} a[i]*a[i+k]

dc_k/da[j]:
- If j < n-k: a[j] appears as the first factor in term i=j, contributing a[j+k]
- If j >= k: a[j] appears as the second factor in term i=j-k, contributing a[j-k]
- If both (k <= j < n-k): dc_k/da[j] = a[j+k] + a[j-k]

So:
- j in [0, k): dc_k/da[j] = a[j+k]
- j in [k, n-k): dc_k/da[j] = a[j+k] + a[j-k]
- j in [n-k, n-1]: dc_k/da[j] = a[j-k]

This can be computed quickly for each k. For K top peaks, it's O(K*n) which for K=50, n=5000 is 250K operations. That's fine.

Let me now write the full algorithm:

```python
import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        alen = len(a_list)
        if alen < 1:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def gradient_of_lag(a, k):
    """Compute dc_k/da for lag k (0 <= k < n)."""
    n = len(a)
    g = np.zeros(n)
    if k == 0:
        g[:] = 2.0 * a
    elif k < n:
        # j in [0, k): g[j] = a[j+k]
        g[0:k] = a[k:n]
        # j in [k, n-k): g[j] = a[j+k] + a[j-k]
        if n - k > k:
            g[k:n-k] = a[2*k:n] + a[0:n-2*k]
        # j in [n-k, n-1]: g[j] = a[j-k]
        g[n-k:] = a[0:k]
    return g

def softmax_mgd_step(a, ac, top_k, alpha):
    """
    Compute the soft-max weighted gradient and perform one MGD step.
    """
    n = len(a)
    # Get top k lags (from the autocorrelation)
    # ac has length 2n-1, symmetric: ac[k] = ac[2n-2-k]
    # Only consider lags 0 to n-1
    ac_half = ac[:n]
    
    # Soft-max weights
    max_v = ac_half.max()
    exp_v = np.exp(alpha * (ac_half - max_v))
    p = exp_v / exp_v.sum()
    
    # Only use top-k significant weights
    threshold = p.max() * 0.01
    mask = p > threshold
    active_lags = np.where(mask)[0]
    active_weights = p[active_lags]
    wsum = active_weights.sum()
    active_weights = active_weights / wsum
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for w, k in zip(active_weights, active_lags):
        grad += w * gradient_of_lag(a, int(k))
    
    # Center the gradient (for simplex constraint)
    grad = grad - grad.mean()
    
    return grad, active_lags, active_weights

def mgd_optimize(a0, time_limit, top_k=50, eta0=0.03, alpha0=50.0, verbose=False):
    """
    Multiplicative Gradient Descent with soft-max weighting.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    eta = eta0
    alpha = alpha0
    patience = 0
    max_patience = 500
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        # Compute gradient
        grad, lags, weights = softmax_mgd_step(a, ac, top_k, alpha)
        
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        
        grad = grad / gmax
        
        # MGD update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()
        
        # Accept or reject
        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.999
            # Gradually increase alpha (sharper max)
            alpha = min(alpha * 1.001, 500.0)
        else:
            eta *= 0.7
            if eta < 0.001:
                eta = 0.01
                alpha = max(alpha * 0.9, 5.0)
                patience = max(0, patience - 100)
        
        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}, alpha={alpha:.1f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a

def peak_shave(a0, time_limit, num_peaks=10, verbose=False):
    """
    Explicit peak shaving: for each peak autocorrelation, find the best
    pair of coordinates to redistribute mass.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n = len(a)
        ac_half = ac[:n]
        
        # Get top peaks
        top_idx = np.argsort(ac_half)[::-1][:num_peaks]
        
        improved = False
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            g = gradient_of_lag(a, k)
            
            # Find position with highest gradient (most contributes to this peak)
            src = int(np.argmax(g))
            
            # Find position with lowest gradient (least impact on this peak)
            dst = int(np.argmin(g))
            
            if src == dst or a[src] < 1e-12:
                continue
            
            # Try moving mass from src to dst
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakShave done ({elapsed:.0f}s): J={J:.6f}")
    return best_a

def block_refine(a0, time_limit, block_size=20, verbose=False):
    """
    Block-level local search: perturb blocks of consecutive coordinates.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        ac_half = ac[:n]
        top_lags = np.argsort(ac_half)[::-1][:10]
        
        improved = False
        for attempt in range(40):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Choose block location
            if attempt < 15:
                # Target near peak-contributing region
                k_pick = int(top_lags[attempt % len(top_lags)])
                center = k_pick // 2
                start = max(0, min(n - block_size, center + int(rng.integers(-10, 10))))
            else:
                start = int(rng.integers(0, max(1, n - block_size + 1)))
            
            end = min(n, start + block_size)
            bs = end - start
            if bs < 2:
                continue
            
            # Perturb the block
            block = a[start:end].copy()
            mean_b = block.mean() + 1e-8
            
            # Different perturbation strategies
            strategy = attempt % 4
            if strategy == 0:
                # Gaussian perturbation
                pert = rng.normal(0, 0.05 * mean_b, bs)
                new_block = np.maximum(block + pert, 0.0)
            elif strategy == 1:
                # Swap within block
                perm = rng.permutation(bs)
                new_block = block[perm]
            elif strategy == 2:
                # Stretch/compress
                new_block = block * rng.uniform(0.7, 1.3)
            else:
                # Shift
                shift = rng.integers(-2, 3)
                new_block = np.roll(block, shift)
                new_block[:1] = 0
                new_block = np.maximum(new_block, 0.0)
            
            a_temp = a.copy()
            a_temp[start:end] = new_block
            a_temp = np.maximum(a_temp, 0.0)
            s_t = a_temp.sum()
            if s_t < 1e-15:
                continue
            a_temp = a_temp / s_t
            M_temp = fast_autocorr(a_temp).max()
            
            if M_temp < M - 1e-14:
                a = a_temp
                M = M_temp
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                improved = True
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Block done ({elapsed:.0f}s): J={J:.6f}")
    return best_a
