    smooth_max_val = conv_max + np.log(sum_exp) / alpha
    
    # Compute gradient via FFT
    # grad[i] = 2 * sum_j a[j] * weights[j+i]
    # = 2 * cross_correlation(a, weights)[i]
    N = 1 << (3 * n - 2 + 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), N)
    W_rev = np.fft.rfft(weights[::-1], N)
    cross = np.fft.irfft(A * W_rev, N)
    
    # grad[i] = 2 * cross[2*(n-1) - i]
    idx = 2*(n-1) - np.arange(n)
    grad = 2.0 * cross[idx]
    
    return smooth_max_val, grad, conv, weights

def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update on the simplex."""
    n = len(a)
    # Center the gradient (subtract mean for numerical stability in mirror descent)
    grad_centered = grad - grad.mean()
    
    # Update
    a_new = a * np.exp(-eta * grad_centered)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new / s
    return a_new

def optimize_smooth(a0, time_limit, alpha_start, alpha_end, verbose=True):
    """Main optimization loop using smooth max gradient + mirror descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_conv_max = fast_conv(a).max()
    t0 = time.monotonic()
    
    alpha = alpha_start
    eta = 0.05
    patience = 0
    max_patience = 200
    last_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_max_val, grad, conv, weights = smooth_max_and_grad(a, alpha)
        true_max = conv.max()
        
        if true_max < best_conv_max - 1e-15:
            best_conv_max = true_max
            best_a = a.copy()
            patience = 0
            last_improve = time.monotonic() - t0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        # Update
        a_new = mirror_descent_step(a, grad, eta)
        
        # Check improvement
        new_conv_max = fast_conv(a_new).max()
        if new_conv_max < true_max + 1e-12:
            a = a_new
            eta *= 0.999
            alpha *= 1.001  # gradually increase smoothness
        else:
            eta *= 0.5
            if eta < 0.001:
                eta = 0.03
                alpha = alpha_start
                patience = max(0, patience - 50)
        
        # Periodically increase alpha (focus on peaks more)
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * * best_conv_max / 1.0  # since sum=1
        print(f"    Smooth opt done ({elapsed:.0f}s): J={J:.6f}, n={n}")
    
    return best_a
