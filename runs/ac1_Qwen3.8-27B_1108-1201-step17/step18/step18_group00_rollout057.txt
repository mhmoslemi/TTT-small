def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_objective(a, n):
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + repr(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def parametric_search(n, time_limit, best_init, best_score):
    """Search over diverse parametric families, emphasizing asymmetric ramps."""
    t0 = time.monotonic()
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    # Power law ramps (most important - these give ~4/3 asymptotic)
    for p in [0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5, 1.7, 2.0, 2.5, 3.0]:
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1((i_arr + 1.0) ** p)
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()
            print("    Ramp p=" + repr(p) + ": score=" + repr(v))

    # Inverted ramps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = norm_sum1((n - i_arr) ** p)
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()
            print("    IRamp p=" + repr(p) + ": score=" + repr(v))

    # Two-sided: ramp on left, small on right (and vice versa)
    for p in [0.5, 1.0, 1.5, 2.0]:
        for w in [0.3, 0.5, 0.7]:
            a_left = (i_arr + 1.0) ** p
            a_right = (n - i_arr) ** p
            a = w * a_left + (1.0 - w) * a_right
            a = norm_sum1(a)
            v = compute_objective(a, n)
            if v < best_val:
                best_val = v
                best_a = a.copy()
                print("    TwoSide p=" + repr(p) + " w=" + repr(w) + ": score=" + repr(v))

    # Exponential ramps
    for rate in [0.5, 1, 2, 3, 5, 10]:
        a = norm_sum1(np.exp(rate * i_arr / n))
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()
            print("    ExpR r=" + repr(rate) + ": score=" + repr(v))

    # Modified ramps: power law with damping
    for p in [0.5, 1.0, 1.5, 2.0]:
        for q in [0.5, 1.0, 2.0]:
            a = (i_arr + 1.0) ** p * ((n - i_arr) / n) ** q
            a = norm_sum1(a)
            v = compute_objective(a, n)
            if v < best_val:
                best_val = v
                best_a = a.copy()
                print("    ModR p=" + repr(p) + " q=" + repr(q) + ": score=" + repr(v))

    # Shifted ramps (mass not starting at 0)
    for start in [0.0, 0.1, 0.2]:
        for p in [1.0, 1.5, 2.0]:
            shifted = np.maximum(i_arr - start * n, 0.0)
            a = (shifted + 1.0) ** p
            a = norm_sum1(a)
            v = compute_objective(a, n)
            if v < best_val:
                best_val = v
                best_a = a.copy()
                print("    ShiftR s=" + repr(start) + " p=" + repr(p) + ": score=" + repr(v))

    # Random initialization
    rng = np.random.default_rng(123)
    for trial in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.3, 4.0))
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()
            print("    Rnd " + str(trial) + ": score=" + repr(v))

    elapsed = time.monotonic() - t0
    print("  Parametric done (" + repr(round(elapsed, 1)) + "s): best score=" + repr(best_val))
    return best_a, best_val


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw_with_linesearch(a0, time_limit, solver, initial_K=300, max_K=2000,
                           add_per_iter=25, verbose=True):
    """LP-based Frank-Wolfe with line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 600

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(constraint_list[i]) for i in range(min(initial_K, len(constraint_list))))
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.5), max(0.01, default_alpha * 1.5), 1.0 / (t + 2)]
        alphas = list(set([max(0.005, min(1.0, a)) for a in alphas]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t <= 3:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW t=" + str(t) + " (" + repr(round(elapsed, 1)) + "s): score=" + repr(J) + " |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + repr(round(elapsed, 1)) + "s): score=" + repr(J))
    return best_a, best_m


def compute_soft_grad(y, ac, top_k, beta, n):
    """Compute weighted subgradient using top-k autocorrelation peaks."""
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * y[j_idx]
    return grad


def mirror_descent(a0, time_limit, top_k=50, beta=3.0, c0=3.0,
                   restart_every=3000, verbose=True, mp=False):
    """Mirror descent with soft subgradient, Nesterov acceleration, and restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)
        grad = compute_soft_grad(y, ac, cur_top_k, cur_beta, n)

        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        eta = gamma / (np.sqrt(t) + 1.0)
        if grad_norm > 1e-15:
            eta = min(eta, 0.5 / grad_norm)

        if mp:
            # Mirror (exponential) step
            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
        else:
            # Euclidean (projected) step
            a_new = y - eta * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a
