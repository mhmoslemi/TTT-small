def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_conv_max(a):
    """Only compute max of convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return c.max()

def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def spectral_compress(a, alpha, n_iter=1):
    """Apply spectral power compression.
    alpha < 1 compresses the dynamic range (flattens the spectrum).
    alpha > 1 expands it.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Power compression
    mag_new = mag ** alpha
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def spectral_anneal(a0, time_limit, alpha_start=0.7, alpha_end=0.95, 
                     n_sweeps=50, verbose=False):
    """Spectral annealing: gradually increase alpha from < 1 to ~1.
    Lower alpha = more aggressive flattening.
    """
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv_max(a)
    t0 = time.monotonic()
    
    total_iters = n_sweeps
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing schedule: alpha goes from alpha_start to alpha_end
        progress = it / max(1, total_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        # Apply compression
        a_new = spectral_compress(a, alpha)
        m_new = fast_conv_max(a_new)
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try a blend
            blend = 0.3
            a_blend = norm_simplex((1-blend)*a + blend*a_new)
            m_blend = fast_conv_max(a_blend)
            if m_blend < best_m - 1e-15:
                best_m = m_blend
                best_a = a_blend.copy()
                a = a_blend
            # else keep current
            # Add small noise to escape

                rng = np.random.default_rng(it)
                noise = rng.normal(0, 0.01, n)

                if m_noisy < best_m:
                    best_m = m_noisy
                    best_a = a_noisy.copy()
                    a = a_noisy
    
    return best_a, 2.0 * n * best_m

def subgrad_descent(a0, time_limit, lr_init=0.5, decay=0.999, 
                    top_k=20, verbose=False):

    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv_max(a)

    max_stall = 200
    
    it = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1
        
        c = fast_conv(a)
        M_curr = c.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Get top peaks
        k_max = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = c[top_idx]
        max_v = top_vals[0]
        
        # Softmax weights
        beta = 50.0
        exponents = beta * (top_vals - max_v) / (max_v + 1e-30)
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w = w / w.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += wi * a[j_idx]
        
        # Projected gradient step (keep on simplex)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        m_new = fast_conv_max(a_new)
        if m_new < best_m:
            a = a_new
        else:
            # Try smaller step
            a_half = norm_simplex(a - 0.5 * lr * grad)
            m_half = fast_conv_max(a_half)
            if m_half < best_m:
                a = a_half
            else:
                # Revert and add noise

                    noise = rng.normal(0, 0.02, n)

        lr *= decay
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): score={2*n*best_m:.6f}")
    
    return best_a, 2.0 * n * best_m

def coordinate_refine(a0, time_limit, verbose=False):

    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv_max(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    stall = 0
    max_stall = 100
    c = fast_conv(a)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        
        # Find the peak
        peak_idx = int(np.argmax(c))
        
        # Try moving mass from positions that contribute to the peak
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n, peak_idx + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_idx - idx]
        
        # Sort by contribution
        order = np.argsort(contrib)[::-1]
        
        for oi in order[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[oi])
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try moving to random positions
            for _ in range(3):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_conv_max(a_trial)
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        c = fast_conv(a)
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Escape: random perturbation
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-8:
                amt = a[i1] * rng.uniform(0.02, 0.08)
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    m = fast_conv_max(a)
                    if m < best_m:
                        best_m = m
                        best_a = a.copy()
                        stall = 0
    
    return best_a, 2.0 * n * best_m

"""

import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_max(a):
    """Only compute max of convolution a*a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return c.max()


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence using the external evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def spectral_compress(a, alpha):
    """Apply spectral power compression. alpha < 1 flattens, alpha > 1 sharpens."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_new = mag ** alpha
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_anneal(a0, time_limit, alpha_start=0.6, alpha_end=0.97,

def compute_subgradient(a, c, top_k=20, beta=60.0):
    """Compute weighted subgradient using top peaks with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()

    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

                        best_m_trial = m_trial

                amt = a[src] * best_frac

            # Spectral step
            a_spec = spectral_compress(a, alpha)
            m_spec = fast_conv_max(a_spec)
            if m_spec < best_m - 1e-15:
                a = a_spec
                best_m = m_spec
                best_a = a.copy()
                stall = 0
                alpha = min(0.99, alpha + 0.003)
            else:
                stall += 1
        else:
            # Subgradient step
            c = fast_conv(a)
            grad = compute_subgradient(a, c, top_k=15, beta=50.0)
            g_norm = np.sqrt(np.sum(grad ** 2))
            if g_norm > 1e-15:
                grad = grad / g_norm
                a_new = norm_simplex(a - lr * grad)
                m_new = fast_conv_max(a_new)
                if m_new < best_m - 1e-15:
                    a = a_new
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                    lr = min(0.8, lr * 1.01)
                else:
                    lr = max(0.05, lr * 0.8)
                    stall += 1

        if stall > 50 and stall % 30 == 0:
            noise = rng.normal(0, 0.03, n)
            a_noise = norm_simplex(a + noise)
            m_noise = fast_conv_max(a_noise)
            if m_noise < best_m:
                best_m = m_noise
                best_a = a_noise.copy()
                a = a_noise
                stall = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Combined it={it} ({elapsed:.0f}s): score={2*n*best_m:.6f}")

    return best_a, best_m


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 1e-6)
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_asymmetric(n, p=3.0):
    x = np.linspace(0, 1, n)
    a = ((1 - x) ** p + 0.01)
    return norm_simplex(a)


def make_init_single_side(n, power=2.0):
    x = np.linspace(0, 1, n)
    a = x ** power + 0.01
    return norm_simplex(a)


def make_init_two_sided(n, gap=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5 * (1 - gap)) / 0.08) ** 2) + np.exp(-((x - 0.5 * (1 + gap)) / 0.08) ** 2)
    a += 0.01
    return norm_simplex(a)


def make_init_random_smooth(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 25):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Helper to try updating best (using global variables to avoid closure)
    # We'll just use local variables and pass around

    # === Phase 1: Spectral annealing from parent (aggressive) ===
    print(f"\n=== Phase 1: Spectral Annealing (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_in = norm_simplex(best_seq)
        a_out, m_out = spectral_anneal(a_in, t1, alpha_start=0.55, alpha_end=0.97)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [spec_anneal]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")
        else:
            print(f"  Spectral anneal: obj={2*len(a_out)*m_out:.6f} (no eval improvement)")

    # === Phase 2: Combined optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Combined Optimize (n={n_best}) ===")
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 20:
        a_in = norm_simplex(best_seq)
        a_out, m_out = combined_optimize(a_in, t2)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [combined]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 3: Subgradient with line search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Subgradient (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 15:
        a_in = norm_simplex(best_seq)
        a_out, m_out = subgrad_optimize(a_in, t3, lr_init=0.4)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [subgrad]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 4: Multi-initialization exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Multi-init Exploration ===")

    inits = []
    labels = []

    inits.append(make_init_bowl(n_best, sharpness=3.0))
    labels.append("bowl3")

    inits.append(make_init_asymmetric(n_best, p=5.0))
    labels.append("asym5")

    inits.append(make_init_asymmetric(n_best, p=2.5))
    labels.append("asym25")

    inits.append(make_init_single_side(n_best, power=4.0))
    labels.append("oneside4")

    inits.append(make_init_two_sided(n_best, gap=0.5))
    labels.append("twosid05")

    inits.append(make_init_random_smooth(n_best, seed=314))
    labels.append("rand314")

    inits.append(make_init_random_smooth(n_best, seed=271))
    labels.append("rand271")

    # Interpolated variants
    for delta in [300, -300, 600]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000:
            inits.append(interpolate_sequence(norm_simplex(best_seq), new_n))
            labels.append(f"interp{new_n}")

    # Deduplicate
    seen = set()
    unique_inits = []
    unique_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, len(init) // 15)], 5))
        if sig not in seen:
            seen.add(sig)
            unique_inits.append(init)
            unique_labels.append(lbl)

    t_per = min(remaining * 0.20 / max(1, len(unique_inits)), 35.0)
    for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 10:
            break
        t_avail = min(t_per, remaining * 0.4)
        if t_avail < 8:
            break
        print(f"    Spectral '{lbl}' (budget={t_avail:.0f}s)...")
        a_out, m_out = spectral_anneal(init, t_avail, alpha_start=0.5, alpha_end=0.93)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [{lbl}]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 5: Pair-swap coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pair-Swap CD (n={n_best}) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 10:
        a_in = norm_simplex(best_seq)
        a_out, m_out = pair_swap_optimize(a_in, t5)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [pairswap]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 6: Final intensive refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Refinement (n={n_best}) ===")

    if remaining > 20:
        # Fine spectral annealing
        t6a = min(remaining * 0.25, 150.0)
        if t6a > 10:
            a_in = norm_simplex(best_seq)
            a_out, _ = spectral_anneal(a_in, t6a, alpha_start=0.82, alpha_end=0.99)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [final_spec]: {sc:.6f}")

        # Combined
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.25, 150.0)
        if t6b > 10:
            a_in = norm_simplex(best_seq)
            a_out, _ = combined_optimize(a_in, t6b)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [final_comb]: {sc:.6f}")

        # Pair swap
        remaining = time_budget - (time.monotonic() - t_global)
        t6c = min(remaining * 0.25, 120.0)
        if t6c > 8:
            a_in = norm_simplex(best_seq)
            a_out, _ = pair_swap_optimize(a_in, t6c)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [final_swap]: {sc:.6f}")

        # Final subgradient
        remaining = time_budget - (time.monotonic() - t_global)
        t6d = min(remaining * 0.2, 80.0)
        if t6d > 8:
            a_in = norm_simplex(best_seq)
            a_out, _ = subgrad_optimize(a_in, t6d, lr_init=0.2)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [final_sg]: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
