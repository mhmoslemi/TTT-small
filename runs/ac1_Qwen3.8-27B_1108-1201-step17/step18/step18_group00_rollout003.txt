    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

            best_a = a.copy()
    
    elapsed = time.monotonic() - t0
    print(f"  Parametric init ({elapsed:.1f}s): J={best_val:.6f}, best p~{p if 'p' in dir() else 'N/A'}")
    return best_a, best_val

def lp_fw(a0, t_limit, solver, init_K=400, max_K=3000):
    """LP-based Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = norm_sum1(a0)
    best_y = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    ac_y = fast_autocorr(y)
    active = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    
    stall = 0
    max_stall = 1500
    iter_count = 0
    
    while time.monotonic() - t0 < t_limit:
        if stall > max_stall:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_y = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Manage active set size
        clist = sorted(active)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active = set(int(clist[i]) for i in keep)
            clist = sorted(active)
        
        K = len(clist)
        if K < 5:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(clist):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob

.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        
        # Check for violated constraints outside active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where((conv_ys > M_curr + 1e-8) & (~np.isin(np.arange(size), np.array(clist))))[0]
        if len(violations) > 0:
            top_v = violations[np.argsort(conv_ys[violations])[::-1][:30]]
            for k in top_v:
                active.add(int(k))
        
        # Line search
        alpha = 2.0 / (iter_count + 2.0)
        best_alpha = alpha
        best_trial_M = M_curr
        for a_try in [alpha, alpha*0.5, alpha*1.5, min(1.0, alpha*2), 0.01, 0.1]:
            a_try = max(0.005, min(1.0, a_try))
            trial = np.maximum((1 - a_try) * y + a_try * g_dir, 0.0)
            ts = trial.sum()
            if ts > 1e-15:
                trial /= ts
                M_t = fast_autocorr(trial).max()
                if M_t < best_trial_M:
                    best_trial_M = M_t
                    best_alpha = a_try
        
        y = np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y /= ys
        
        iter_count += 1
        if iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={2*n*best_m:.6f}, |S|={len(active)}")
    
    J = 2.0 * n * best_m
    print(f"    LP-FW done ({time.monotonic()-t0:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_y, best_m

def interpolate(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss
