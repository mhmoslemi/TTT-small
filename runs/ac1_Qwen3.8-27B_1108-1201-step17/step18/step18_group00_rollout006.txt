def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_objective_fast(a, n):
    ac = fast_autocorr(a)

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, f)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def try_upd(bseq, bscore, a, label):
    if a is None:
        return bseq, bscore
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return bseq, bscore
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < bscore:
        el = time.monotonic() - _T0[0]
        print(f"*** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={el:.0f}s)")
        return a.copy(), sc
    return bseq, bscore

def build_conv_rows(y, ks, n):
    """Build constraint rows for LP. Each row k: A[k,j] = y[k-j] if 0<=k-j<n else 0."""
    K = len(ks)
    A = np.zeros((K, n))
    for i, k in enumerate(ks):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            A[i, idx] = y[k - idx]
    return A

def softmax_md(a0, tlim, t0g, beta0=5.0, beta1=5000.0, eta_c=1.0, verbose=True):
    n = len(a0)
    a = norm1(a0)
    ba = a.copy()
    bm = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    stall = 0
    max_stall = 15000
    
    # Precompute FFT sizes
    fs_ac = 1 << (2 * n - 2).bit_length()
    fs_grad = 1 << (3 * n - 1).bit_length()
    a_rev = a[n-1::-1]  # will update each iteration
    # Actually, a changes each iteration, so I can't precompute. Let me just compute each time.
    
    for it in range(1, 5000000):
        if time.monotonic() - t0 > tlim:
            break
        if stall > max_stall:
            # Restart with noise
            a = ba + rng.normal(0, 0.02, n) * ba
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            stall = 0
            if verbose:
                print(f"    SMD restart at it={it}")
            continue
        
        frac = min(1.0, it / 30000.0)
        beta = beta0 + (beta1 - beta0) * frac * frac
        
        # AC
        ac = fast_autocorr(a)
        
        # Softmax
        acm = ac.max()
        eb = np.exp(beta * (ac - acm))
        p = eb / eb.sum()
        
        # Gradient: g_i = 2 * sum_k p_k * a[k-i]
        # = 2 * (p * a_rev)[n-1+i]
        ar = a[n-1::-1]
        Pf = np.fft.rfft(p, fs_grad)
        Af = np.fft.rfft(ar, fs_grad)
        conv_pa = np.fft.irfft(Pf * Af, fs_grad)
        g = 2.0 * conv_pa[n-1:n-1+n]
        
        # Mirror descent step
        eta = eta_c / (np.sqrt(it) + 1.0)
        g_min = g.min()
        exp_rg = np.exp(-eta * (g - g_min))
        a_new = a * exp_rg
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        Mn = fast_autocorr(a_new).max()
        if Mn < bm - 1e-15:
            bm = Mn
            ba = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        if verbose and it % 10000 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * bm
            print(f"    SMD it={it} ({el:.0f}s): J={J:.6f}, beta={beta:.0f}, stall={stall}")
    
    el = time.monotonic() - t0
    J = 2.0 * n * bm
    if verbose:
        print(f"    SMD done ({el:.0f}s): J={J:.6f}")
    return ba, bm

Wait, I need to double-check the FFT size for the gradient computation.

$p$ has length $2n-1$. $a_{rev}$ has length $n$.
Convolution length: $(2n-1) + n - 1 = 3n - 2$.
FFT size: next power of 2 $\geq 3n-2$.
For n=6700: $3*6700-2 = 20098$. Next power of 2: 32768.

For the autocorrelation:
FFT size: next power of 2 $\geq 2n-1$.
For n=6700: $2*6700-1 = 13399$. Next power of 2: 16384.

OK, these are manageable.
