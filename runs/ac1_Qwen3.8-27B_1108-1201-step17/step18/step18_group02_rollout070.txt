    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, rng_seed=123):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    for freq in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.01)
            candidates.append(a / a.sum())

    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())

    a = np.cos(np.pi * (x - 0.5)) ** 2
    candidates.append(a / a.sum())

    candidates.append(np.ones(n) / n)

    for rate in [1, 2, 3]:
        a = 1.0 / (i_arr + 1.0) ** (1.0 / rate)
        candidates.append(a / a.sum())

    return candidates


def smoothed_gradient_step(a0, time_limit, n_steps=5000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()

    tau_min = max(ac.max() * 0.005, 1e-10)
    tau_decay = 0.9998
    lr = 0.5
    beta1 = 0.9
    beta2 = 0.999

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0

    for t in range(1, n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        tau = max(tau * tau_decay, tau_min)

        ac_scaled = ac / tau
        ac_scaled -= ac_scaled.max()
        exp_vals = np.exp(ac_scaled)
        p = exp_vals / exp_vals.sum()

        a_flip = a[::-1].astype(np.float64)
        size = len(ac)
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a_flip, fft_size)
        Pf = np.fft.rfft(p.astype(np.float64), fft_size)
        conv_full = np.fft.irfft(Af * Pf, fft_size)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0.0)

        m_hat = beta1 * m_hat + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)

        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2

        step = lr / (np.sqrt(v_c) + 1e-8)

        log_a = np.log(np.maximum(a, 1e-30)) - step * m_c
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new

        if stall > 2000:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    SmoothGrad done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

        a_out, _ = fw_with_backtrack(a_in, t3, solver, initial_K=800, max_K=5000,
                                     add_per_iter=60, verbose=True)
        check_and_update(a_out, f"fw_bt_n{n_best}")

    # Phase 4: Simulated annealing
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Simulated annealing (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = simulated_annealing(a_in, t4, verbose=True)
        check_and_update(a_out, f"sa_n{n_best}")

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for delta in [400, 800, -400, -800, 1200, -1200]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000 and new_n != n_best:
            nearby_lengths.append(new_n)

    seen = set()
    unique_lengths = []
    for ln in nearby_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    for new_n in unique_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: n={new_n} ===")
        a_interp = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.20, 120.0)
        if t5a > 15:
            a_out, _ = smoothed_gradient_step(a_interp, t5a, n_steps=2000, verbose=False)
            check_and_update(a_out, f"smooth_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 5500:
            t5b = min(remaining * 0.30, 150.0)
            if t5b > 20:
                a_out, _ = fw_with_backtrack(norm_sum1(best_seq), t5b, solver,
                                             initial_K=600, max_K=4000,
                                             add_per_iter=50, verbose=False)
                check_and_update(a_out, f"fw_n{new_n}")

    # Phase 6: Second FW pass on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Second FW (n={n_best}) ===")
    if remaining > 60 and n_best <= 5500:
        t6 = min(remaining * 0.25, 250.0)
        if t6 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = fw_with_backtrack(a_in, t6, solver, initial_K=1000, max_K=6000,
                                         add_per_iter=80, verbose=True)
            check_and_update(a_out, f"fw2_n{n_best}")

    # Phase 7: Final coord polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord polish (n={n_best}) ===")
    if remaining > 15:
        t7 = min(remaining * 0.30, 100.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_polish(a_in, t7, verbose=False)
            check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
