"""
Minimizes 2*n*max(self_conv(a))/(sum(a))^2 for non-negative a.
Uses: (1) power-law parametric init with alpha sweep,
(2) FFT-based subgradient descent on simplex with adaptive step size,
(3) coordinate descent polish, (4) multi-length exploration.
Key insight: asymmetric sequences (e.g., a[i] ~ (i+1)^(-1/2)) break the
J>=2 barrier for symmetric sequences, achieving J ~ pi/2 ≈ 1.57 asymptotically.
Further optimization refines the shape to approach ~1.50.
"""

import numpy as np
import time

def fast_self_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgrad_fft(a, weights):
    """
    Compute subgradient: grad[j] = 2 * sum_k weights[k] * a[k-j]
    using FFT-based cross-correlation.
    """
    n = len(a)
    m = len(weights)  # 2n-1
    fft_size = 1 << (n + m - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(weights, fft_size)
    # xcorr[j] = sum_k weights[k] * a[k-j] = ifft(conj(W) * A)[j]
    xcorr = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * xcorr

def norm_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_self_conv(a)
    return 2.0 * n * conv.max() / (s * s)

def powerlaw_init(n, alpha):
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-alpha)

def interp_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    return norm_nonneg(a_new)

def mgd_optimize(a0, time_limit, top_k=50, eta0=0.03, power=2.0, verbose=False):
    """Multiplicative gradient descent on simplex with top-K subgradient."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_conv = fast_self_conv(a)
    best_m = best_conv.max()
    t0 = time.monotonic()
    
    eta = eta0
    patience = 0
    max_patience = 500
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_self_conv(a)
        M = conv.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        # Top-K subgradient
        K = min(top_k, 2 * n - 1)
        top_idx = np.argsort(conv)[::-1][:K]
        top_vals = np.maximum(conv[top_idx], 1e-30)
        w = top_vals ** power
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(K) / K
        else:
            w = w / w_sum
        
        # Build full weight vector
        full_w = np.zeros(2 * n - 1)
        full_w[top_idx] = w
        
        # Compute gradient via FFT
        grad = compute_subgrad_fft(a, full_w)
        
        # Normalize gradient
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Line search (accept if improved)
        conv_new = fast_self_conv(a_new)
        M_new = conv_new.max()
        
        if M_new <= M + 1e-12:
            a = a_new
            eta = min(eta * 1.001, 0.1)
        else:
            eta *= 0.5
            if eta < 0.0001:
                eta = eta0 * 0.5
                # Random perturbation to escape
                rng = np.random.default_rng(iters)
                idx = rng.integers(0, n, size=n//10)
                a[idx] *= rng.uniform(0.7, 1.3, n//10)
                a = norm_nonneg(a)
                patience = max(0, patience - 200)
        
        iters += 1
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD it={iters} ({elapsed:.0f}s) J={J:.6f} eta={eta:.5f}")
    
    return best_a

def coord_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass from peak-contributing positions."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    conv = fast_self_conv(a)
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        conv = fast_self_conv(a)
        M = conv.max()
        improved = False
        
        # Focus on top lags
        n_top = min(20, 2 * n - 1)
        top_k = np.argsort(conv)[::-1][:n_top]
        
        for ki in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(ki)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            
            idx = np.arange(j_start, j_end + 1)
            contrib = 2.0 * a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            if a[src] < 1e-12:
                continue
            
            # Find destination: position with minimum total contribution to peaks
            peak_impact = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_impact[idx2] += a[k2i - idx2]
                    peak_impact[k2i - idx2] += a[idx2]
            
            # Choose destination with lowest peak impact
            candidates = np.argsort(peak_impact)[:n//4]
            dst = candidates[rng.integers(len(candidates))]
            if dst == src:
                dst = (dst + 1) % n
            
            for frac in [0.02, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_self_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random escape
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            if i1 != i2 and a[i1] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i1] * frac
                a[i1] -= amt
                a[i2] += amt
                a = norm_nonneg(a)
            else:
                a = norm_nonneg(a + rng.normal(0, 0.01, n))
    
    return best_a

def parametric_search(n, time_budget):
    """Try different parametric families."""
    t0 = time.monotonic()
    best_a = None
    best_J = float('inf')
    
    # Power law: a[i] = (i+1)^(-alpha)
    for alpha in [0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7]:
        if time.monotonic() - t0 > time_budget * 0.5:
            break
        a = powerlaw_init(n, alpha)
        a = norm_nonneg(a)
        J = score(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    # Exponential: a[i] = exp(-beta * i/n)
    for beta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        if time.monotonic() - t0 > time_budget * 0.7:
            break
        x = np.linspace(0, 1, n)
        a = np.exp(-beta * x)
        a = norm_nonneg(a)
        J = score(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    # Mixed: a[i] = (i+1)^(-alpha) * exp(-beta * i/n)
    for alpha in [0.3, 0.5, 0.7]:
        for beta in [0.5, 1.0, 2.0]:
            if time.monotonic() - t0 > time_budget:
                break
            i = np.arange(1, n + 1, dtype=np.float64)
            x = np.linspace(0, 1, n)
            a = (i ** (-alpha)) * np.exp(-beta * x)
            a = norm_nonneg(a)
            J = score(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
    
    return best_a, best_J
