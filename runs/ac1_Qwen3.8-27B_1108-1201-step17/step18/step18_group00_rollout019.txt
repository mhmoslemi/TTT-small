def get_solver():

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute 2*n*max(ac) for normalized a."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def build_lp_matrix(y, lags, n):
    """Build constraint matrix where row k gives (y*g)_k coefficients."""
    K = len(lags)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def parametric_init(n, t_limit):
    """Generate diverse parametric candidates, return best."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)

    candidates.append(0.42 - 0.5 * np.cos(2.0 * np.pi * x) + 0.08 * np.cos(4.0 * np.pi * x))  # Blackman
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i + 1.0) ** p)
        candidates.append((n - i) ** p)
    # Symmetric power mix
    for a_p in [0.5, 1.0, 1.5, 2.0]:
        for b_p in [0.5, 1.0, 1.5, 2.0]:
            candidates.append(np.maximum(i + 1.0, 0.5) ** a_p * np.maximum(n - i, 0.5) ** b_p)
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.1, 0.2, 0.3, 0.4]:

    # Bimodal

    # Exponential

    best_a = y.copy()
    best_M = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac = fast_autocorr(y)
    active = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 500
    iter_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > t_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_M - 1e-14:
            best_M = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set: always keep top lags
        lags_sorted = np.argsort(ac)[::-1]
        top_lags = set(int(k) for k in lags_sorted[:min_K])
        active = active | top_lags

        # Prune if too large
        if len(active) > max_K:
            constraints_sorted = sorted(active)
            margins = np.array([M_curr - ac[k] for k in constraints_sorted])
            keep = np.argsort(margins)[:max_K]
            active = set(int(constraints_sorted[i]) for i in keep)
            active = active | top_lags

        constraints = sorted(active)
        K = len(constraints)
        if K == 0:
            break

        # Solve LP: min M s.t. sum(g)=1, g>=0, (y*g)_k <= M
        coeffs = build_lp_matrix(y, constraints, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        cons = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), cons)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active) > initial_K:
                active = set(int(k) for k in top_lags)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs

        # Add violated constraints
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]
        violations = np.where(conv_yg > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations]
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active.add(int(k))

        # Golden section line search
        if t <= 3:
            alpha = 1.0
        else:
            def obj_alpha(al):
                trial = (1.0 - al) * y + al * g_dir
                trial = np.maximum(trial, 0.0)
                st = trial.sum()
                if st < 1e-15:
                    return float('inf')
                trial = trial / st
                return fast_autocorr(trial).max()

            # Start with a conservative bracket
            al0 = 2.0 / (t + 2.0)
            # Try to find a good bracket
            candidates = [min(1.0, al0 * 0.25), min(1.0, al0 * 0.5), al0,
                          min(1.0, al0 * 1.5), min(1.0, al0 * 2.0), 1.0]
            candidates = sorted(set([max(0.001, c) for c in candidates]))
            vals = [obj_alpha(c) for c in candidates]

            # Pick best from candidates, then refine with golden section around it
            best_idx = int(np.argmin(vals))
            best_alpha = candidates[best_idx]
            best_trial_val = vals[best_idx]

            # Golden section refinement around best
            if best_idx > 0 and best_idx < len(candidates) - 1:
                lo = candidates[best_idx - 1]
                hi = candidates[best_idx + 1]
            elif best_idx == 0:
                lo = 0.001
                hi = candidates[1] if len(candidates) > 1 else 0.5
            else:
                lo = candidates[-2] if len(candidates) > 1 else 0.5
                hi = 1.0

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = obj_alpha(c1)
            f2 = obj_alpha(c2)

            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = obj_alpha(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = obj_alpha(c2)

            mid = (lo + hi) / 2.0
            fm = obj_alpha(mid)
            if fm < best_trial_val:
                best_alpha = mid
                best_trial_val = fm

            alpha = best_alpha

        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new
        iter_count += 1

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    LP-FW iter={} ({}s): J={:.6f}, |S|={}".format(t, round(elapsed, 1), J, len(active)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    print("    LP-FW done ({}s): J={:.6f}, iters={}".format(round(elapsed, 1), J, iter_count))
    return best_a, best_M


def pairwise_polish(a0, t_limit, n_top_lags=50, n_tries_per_iter=200, max_iters=5000,
                    verbose=True):
    """Polish by pairwise mass transfers with exact 1D optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() - t0 > t_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_M - 1e-15:
            best_M = M_curr
            best_a = a.copy()

        # Find top lags
        top_lags = np.argsort(ac)[::-1][:n_top_lags]

        # Identify "hot" indices (those that contribute to top lags)
        hot = np.zeros(n, dtype=np.int32)
        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                prod = a[i] * a[j]
                hot[i] += int(prod > 0.01 * M_curr)

        # Find "cold" indices
        cold_mask = hot == 0
        if not cold_mask.any():
            cold_mask = hot < hot.max()

        cold_indices = np.where(cold_mask)[0]
        if len(cold_indices) == 0:
            cold_indices = np.arange(n)

        improved = False

        for trial in range(n_tries_per_iter):
            if time.monotonic() - t0 > t_limit:
                break

            # Pick source from hot, destination from cold
            hot_indices = np.where(hot > 0)[0]
            if len(hot_indices) == 0:
                break
            src = hot_indices[rng.integers(len(hot_indices))]
            dst = cold_indices[rng.integers(len(cold_indices))]
            if src == dst:
                continue
            if a[src] < 1e-15:
                continue

            # Binary search for optimal transfer amount t (from src to dst)
            # (a*a)_k changes by 2t*(a[k-dst] - a[k-src]) + t^2*(delta_k,2*src + delta_k,2*dst - 2*delta_k,src+dst)
            # but only for valid k; indices outside [0,n-1] give 0

            lo = 0.0
            hi = a[src] * 0.5  # Don't transfer more than half

            if hi < 1e-15:
                continue

            def eval_transfer(tt):
                # Compute max autocorrelation after transferring tt from src to dst
                # Only check top lags and a few extra for efficiency
                # Delta for lag k: 2*tt*(a[max(0,min(n-1,k-dst))] - a[max(0,min(n-1,k-src))]) + tt^2 * quad_term
                deltas = np.zeros(n_top_lags)
                for ii, k in enumerate(top_lags):
                    k_int = int(k)
                    idx_dst = k_int - dst
                    idx_src = k_int - src
                    val_dst = a[idx_dst] if 0 <= idx_dst < n else 0.0
                    val_src = a[idx_src] if 0 <= idx_src < n else 0.0
                    # Quadratic terms
                    quad = 0.0
                    if 0 <= 2 * src < 2 * n - 1 and k_int == 2 * src:
                        quad += 1.0
                    if 0 <= 2 * dst < 2 * n - 1 and k_int == 2 * dst:
                        quad += 1.0
                    if 0 <= src + dst < 2 * n - 1 and k_int == src + dst:
                        quad -= 2.0
                    deltas[ii] = 2.0 * tt * (val_dst - val_src) + tt * tt * quad
                return (ac[top_lags] + deltas).max()

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = eval_transfer(c1)
            f2 = eval_transfer(c2)

            for _ in range(20):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_transfer(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_transfer(c2)

            t_opt = (lo + hi) / 2.0
            f_opt = eval_transfer(t_opt)

            if f_opt < M_curr - 1e-12 and t_opt > 1e-14:
                a[src] -= t_opt
                a[dst] += t_opt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                improved = True

        if not improved and it > 0 and it % 10 == 0:
            # Try random moves to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    if fast_autocorr(a).max() < best_M:
                        best_M = fast_autocorr(a).max()
                        best_a = a.copy()
                        improved = True

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    Polish iter={} ({}s): J={:.6f}".format(it, round(elapsed, 1), J))

        # Convergence check
        if it > 0 and it % 50 == 0 and not improved:
            # Check if we're stuck
            pass

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    print("    Polish done ({}s): J={:.6f}".format(round(elapsed, 1), J))
    return best_a, best_M


def soft_mirror_descent(a0, t_limit, top_k=80, beta=5.0, gamma0=4.0,
                        restart_every=3000, verbose=True):
    """Nesterov-accelerated mirror descent with soft subgradient."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    x_prev = x.copy()
    z = x.copy()
    stall = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > t_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = z + beta_t * (z - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = z.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        # Compute soft gradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Update z (Nesterov momentum)
        z = x

        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            noise = 0.05 / (1.0 + 0.1 * (stall))
            pert = rng.normal(0, noise, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            s = x.sum()
            if s > 1e-15:
                x = x / s
            x_prev = x.copy()
            z = x.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    SSD iter={} ({}s): J={:.6f}".format(t, round(elapsed, 1), J))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    print("    SSD done ({}s): J={:.6f}".format(round(elapsed, 1), J))
    return best_a, best_M


def interpolate_to(a, new_n):
    """Interpolate sequence to new length, maintaining normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [{}]: {:.6f} (n={}, t={:.0f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Solver: {}, budget: {:.0f}s".format(solver, time_budget))

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        parent = None

    # Phase 1: Parametric initialization
    print("\n=== Phase 1: Parametric init ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_base = 6500 if parent is None else len(parent)
    if n_base > 8000:
        n_base = 8000
    if n_base < 3000:
        n_base = 5000

    t1 = min(remaining * 0.03, 40.0)
    a_param, _ = parametric_init(n_base, t1)
    best_seq, best_score = try_update(best_seq, best_score, a_param, "param_init", t_global)

    # If we have a parent, also try it directly
    if parent is not None and best_seq is not None:
        n_use = len(best_seq)
    else:
        n_use = n_base

    # Phase 2: LP-FW (main optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: LP-FW (n={}) ===".format(n_use))
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 30:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver,
                         initial_K=300, max_K=2000, add_per_iter=25,
                         min_K=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n{}".format(n_use), t_global)

    # Phase 3: LP-FW round 2 (from current best)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: LP-FW round 2 ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=400, max_K=2500, add_per_iter=30,
                         min_K=100, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # Phase 4: Pairwise polishing
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Pairwise polish ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 20:
        a_out, _ = pairwise_polish(norm_sum1(best_seq), t4,
                                   n_top_lags=60, n_tries_per_iter=150,
                                   max_iters=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "polish", t_global)

    # Phase 5: Soft mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 5: Mirror descent ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_out, _ = soft_mirror_descent(norm_sum1(best_seq), t5,
                                       top_k=80, beta=5.0, gamma0=4.0,
                                       restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mirror", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length ===")
    longer = [n_best + 500, n_best + 1000]
    longer = [l for l in longer if l <= 8500]

    for new_n in longer:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n  Scaling to n={}".format(new_n))
        a_long = interpolate_to(best_seq, new_n)
        best_seq, best_score = try_update(best_seq, best_score, a_long, "interp_n{}".format(new_n), t_global)

        t6a = min(remaining * 0.3, 200.0)
        if t6a > 30:
            a_out, _ = lp_fw(a_long, t6a, solver,
                             initial_K=300, max_K=2000, add_per_iter=25,
                             min_K=80, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n{}".format(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100:
            t6b = min(remaining * 0.3, 150.0)
            if t6b > 20:
                a_out, _ = pairwise_polish(norm_sum1(best_seq), t6b,
                                           n_top_lags=50, n_tries_per_iter=100,
                                           max_iters=2000, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "polish_n{}".format(new_n), t_global)

    # Phase 7: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n={}) ===".format(n_best))
    if remaining > 60 and n_best <= 7000:
        t7 = min(remaining * 0.35, 250.0)
        if t7 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t7, solver,
                             initial_K=500, max_K=3000, add_per_iter=30,
                             min_K=100, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: Final polish ===")
    if remaining > 30:
        t8 = min(remaining * 0.6, 200.0)
        if t8 > 20:
            a_out, _ = pairwise_polish(norm_sum1(best_seq), t8,
                                       n_top_lags=80, n_tries_per_iter=200,
                                       max_iters=5000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "polish_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.0f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score={:.6f}".format(best_score))

    return [float(x) for x in best_seq]
