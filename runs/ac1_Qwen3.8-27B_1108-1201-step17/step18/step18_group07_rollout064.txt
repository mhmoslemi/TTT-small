import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def score_of(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:

        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_lp_structure(n):
    """Precompute row/col index arrays for the LP constraint matrix."""
    num_rows = 2 * n - 1
    rows_chunks = []
    cols_chunks = []
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        rows_chunks.append(np.full(length, k, dtype=np.int32))
        cols_chunks.append(np.arange(j_start, j_end + 1, dtype=np.int32))
    rows = np.concatenate(rows_chunks)
    cols = np.concatenate(cols_chunks)
    return rows, cols


def solve_lp(f, M, rows, cols):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    num_rows = 2 * n - 1
    idx = rows.astype(np.int64) - cols.astype(np.int64)
    vals = f[idx]
    A = csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    c_obj = -np.ones(n)
    b_ub = np.full(num_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 20, 'presolve': True})
    except Exception:
        return None
    if res.status == 0 and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None

def smoothed_grad(a, T):
    """Compute gradient of 2n*T*log(sum(exp(c/T))) w.r.t. log-space params."""
    n = len(a)
    c = fast_conv(a)
    c_shifted = c / T - c.max() / T
    w = np.exp(c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum

    # corr[j] = sum_k w[k] * a[k-j] via FFT cross-correlation
    fft_size = 1 << (2 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)[:n]

    w_avg = np.dot(w, c)
    grad = 4.0 * n * a * (corr - w_avg)
    return grad, c.max()


def topk_subgrad(a, c, k=5, beta=20.0):
    """Subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    k_max = min(k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx]
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        w = w / w_sum

    grad = np.zeros(n)
    for wi, kk in zip(w, top_idx):
        if wi < 1e-12:
            continue
        ki = int(kk)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def logspace_adam(a_init, time_limit, verbose=False, use_smoothed=True):
    """Adam in log-space with annealed temperature."""
    n = len(a_init)
    a = norm_simplex(a_init)
    best_a = a.copy()

            break

        T = T_init * (T_min / T_init) ** progress
        lr = lr_init * (lr_min / lr_init) ** (progress * 0.8)

        if use_smoothed and progress < switch_frac:
            grad, _ = smoothed_grad(a, T)
        else:
            c = fast_conv(a)
            grad = topk_subgrad(a, c, k=min(10, max(3, int(5 + 10 * (1.0 - progress)))),
                                beta=10.0 + 30.0 * progress)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + ad_eps)
        log_a = log_a - log_a.max()

            print(f"    logspace t={t} ({el:.0f}s): T={T:.8f}, lr={lr:.6f}, score={best_score:.8f}")

    return best_a, best_score


def spectral_step(a, strength=0.25):
    """Compress dynamic range of |FFT(a)|."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a

    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.power(ratio, 1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

    alpha = strength * 0.5
    return (1.0 - alpha) * a + alpha * a_new

def peak_coordinate_descent(a_init, time_limit, verbose=False):
    """Target the highest peaks by redistributing mass."""
    n = len(a_init)
    a = norm_simplex(a_init)
    best_a = a.copy()

    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_conv(a)
        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        improved = False

        for kk in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(kk)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            max_contrib_val = contrib.max()
            if max_contrib_val < 1e-20:
                continue

            for attempt in range(3):
                src_local = np.argmax(contrib)
                src = int(idx[src_local])
                src_val = a[src]
                if src_val < 1e-12:
                    break

                dst = int(rng.integers(n))
                if dst == src:
                    continue

                for frac in [0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if 2.0 * n * M_trial < best_score - 1e-15:
                        a = a_trial
                        best_score = 2.0 * n * M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.02, 0.08)

                    sc = score_of(a)

                        best_a = a.copy()
                        stall = 0

    return best_a, best_score


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 1e-10)
    return norm_simplex(a)


def make_init_bowl(n, sharp=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharp + 0.1
    return norm_simplex(a)


def make_init_dual(n, p1=0.25, p2=0.75, w=0.1):
    x = np.linspace(0, 1, n)
    a = (np.exp(-((x - p1) ** 2) / (2 * w ** 2)) +
         np.exp(-((x - p2) ** 2) / (2 * w ** 2)) + 0.05)
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 1e-10)
    return norm_simplex(a)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = norm_simplex(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return norm_simplex(f_interp)
