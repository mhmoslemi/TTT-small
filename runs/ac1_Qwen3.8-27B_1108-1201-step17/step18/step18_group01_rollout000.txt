def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    return grad

            if js < je:

                grad_g[idx] += a[ki - idx] * violations[k]
        # Update g to reduce violations while maintaining sum
        g = g - lr * grad_g
        g = np.maximum(g, 0)
        s = g.sum()
        if s < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / s
        lr *= 0.98
    
    return g

def fast_autocorr_conv(a, b):
    """Cross-correlation/convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def peak_coordinate_descent(a, time_limit, verbose=False):
    """Coordinate descent focusing on peak-contributing positions."""
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M = ac.max()
        top_k = np.argsort(ac)[::-1][:8]
        improved = False

        for k in top_k:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find destination: position with least contribution to peaks
            peak_influence = np.zeros(n)
            for k2 in top_k:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_influence[idx2] += a[k2i - idx2]
                    peak_influence[k2i - idx2] += a[idx2]

            candidates = np.where(peak_influence < np.median(peak_influence))[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(peak_influence[candidates])]
            if dst == src:
                continue

            for frac in [0.01, 0.02, 0.04, 0.06]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] = max(0, a_trial[src] - amt)
                a_trial[dst] += amt
                a_trial = norm_sum1(a_trial)
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = norm_sum1(a)
                if fast_autocorr(a).max() < best_m:
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordPolish done ({elapsed:.0f}s): J={J:.6f}")
    return best_a
