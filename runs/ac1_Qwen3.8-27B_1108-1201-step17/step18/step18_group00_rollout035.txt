def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J_normalized(a):

def build_constraint_matrix(y, constraint_list, n):
    """Build the constraint matrix for LP-FW."""
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        # (y * g)[k] = sum_i y[k-i] * g[i] where both indices valid
        # g[i] appears in constraint k when 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(k, n-1)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

def smooth_max_descent(a0, time_limit, tau_start=0.01, tau_end=0.001, n_iter=None):
    """Gradient descent on log-sum-exp smoothed max of convolution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if n_iter is None:
        n_iter = 50000
    
    x = a.copy()
    x_prev = a.copy()
    rng = np.random.default_rng(123)
    
    for t in range(1, n_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        mu = (1 + np.sqrt(1 + 4 * (t - 1) ** 2)) / 2
        beta_t = (1 - mu) / (1 + t) if t > 1 else 0
        
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute smoothed max
        tau = tau_start * (tau_end / tau_start) ** (t / n_iter)
        ac = fast_autocorr(y)
        
        # Log-sum-exp for smooth max
        ac_clipped = np.maximum(ac, 1e-30)
        max_ac = ac.max()
        log_sum_exp = max_ac + tau * np.log(np.sum(np.exp((ac - max_ac) / tau)))
        smooth_max = log_sum_exp
        
        # Gradient of smooth max w.r.t. y
        # d/dy_k [log(sum_j exp((ac_j - max)/tau))] 
        # = (1/tau) * sum_j exp((ac_j - max)/tau) * (dac_j/dy_k) / sum_j exp((ac_j - max)/tau)
        # = sum_j w_j * (dac_j/dy_k) where w_j = exp((ac_j-max)/tau) / sum exp(...)
        
        weights = np.exp((ac - max_ac) / tau)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.zeros_like(ac)
            weights[0] = 1.0
        else:
            weights = weights / w_sum
        
        # Gradient of ac[j] w.r.t. y[k]: ac[j] = sum_i y[i]*y[j-i]
        # dac[j]/dy[k] = y[j-k] + y[k-j] (if both indices valid)
        # For efficiency, use the top few weights
        top_k = min(50, len(ac))
        top_idx = np.argsort(weights)[::-1][:top_k]
        top_w = weights[top_idx]
        w_used = top_w.sum()
        if w_used < 1e-30:
            top_w = np.ones(top_k) / top_k
        else:
            top_w = top_w / w_used
        
        grad = np.zeros(n)
        for wi, k in zip(top_w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * y[k_int - idx]
        
        # Mirror descent step (in the simplex)
        eta = 1.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print("    SmoothMax done (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))
    return best_a, best_m


def coordinate_descent(a0, time_limit, rng_seed=77):
    """Coordinate descent: move mass from peak-contributing positions to low-impact ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 2000
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Score destinations by how much they contribute to peak lags
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            # Move to position with lowest impact score
            candidates = np.arange(n)
            candidates = np.delete(candidates, src)
            dst = int(candidates[np.argmin(scores[candidates])])
            
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation when stuck
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print("    CoordDescent (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    rng = np.random.default_rng(42)
    for trial in range(300):
        if time.monotonic() - t0 > time_limit:
            break
        alpha = rng.uniform(0.3, 4.0)
        beta = rng.uniform(0.3, 4.0)
        a = rng.beta(alpha, beta, n)
        v = compute_J_normalized(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    if best_a is None:
        best_a = norm_sum1((i + 1.0) ** 1.5)
        best_val = compute_J_normalized(best_a)

    elapsed = time.monotonic() - t0
    print("  Parametric init (" + str(round(elapsed, 1)) + "s): best J=" + str(round(best_val, 6)))
    return best_a, best_val


def build_constraint_matrix(y, constraint_list, n):
    """Build constraint matrix for LP: coeffs[k, i] = y[k-i] if valid."""
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw_optimized(a0, time_limit, solver, initial_K=300, max_K=2000,
                    add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe with optimized active set management and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 600
    iter_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: no improve for " + str(no_improve) + " iters, stopping")
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                top_k = min(initial_K, len(constraint_list))
                active_set = set(int(constraint_list[i]) for i in range(top_k))
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find and add violating constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search with multiple step sizes
        default_alpha = 2.0 / (t + 2.0)
        alphas_to_try = list(set([
            max(0.01, min(1.0, default_alpha)),
            max(0.01, min(1.0, default_alpha * 0.5)),
            max(0.01, min(1.0, default_alpha * 1.5)),
            max(0.01, min(1.0, 1.0 / max(1, t))),
            max(0.01, min(1.0, 0.1)),
            max(0.01, min(1.0, 0.3)),
        ]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas_to_try:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new
        iter_count += 1

        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + ", iters=" + str(iter_count))
    return best_a, best_m


def soft_subgradient_mind(a0, time_limit, top_k=50, beta=3.0, c0=3.0,
                          restart_every=3000, verbose=False):
    """Mirror descent with soft subgradient and adaptive restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov-like momentum
        beta_t = (1 - (1 + np.sqrt(1 + 4 * (t - 1) ** 2)) / 2) / (1 + t) if t > 1 else 0.0

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)

        # Soft subgradient: weighted combination of top-k peak subgradients
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * y[k_int - idx]

        # Mirror descent step (exponential family / simplex)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive restart with perturbation
        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.02 / (1.0 + 0.03 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 100)
            cur_beta = max(cur_beta - 0.2, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SSD restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    SSD done (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))
    return best_a, best_m


def projection_subgradient(a0, time_limit, verbose=False):
    """Projected subgradient with exact projection onto simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    step_scale = 0.5

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(x)
        argmax_lag = int(np.argmax(ac))

        # Subgradient of (a*a)[K] w.r.t. a: 2*a[K-i] for valid i
        grad = np.zeros(n)
        i_start = max(0, argmax_lag - n + 1)
        i_end = min(n, argmax_lag + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] = 2.0 * x[argmax_lag - idx]

        # Step size
        eta = step_scale / (np.sqrt(t) + 1.0)

        # Take step
        x_new = x - eta * grad

        # Project onto {x >= 0, sum(x) = 1}
        # Find tau such that sum(max(x_new - tau, 0)) = 1
        sorted_x = np.sort(x_new)[::-1]
        cumsum = np.cumsum(sorted_x)
        rho_idx = np.where(sorted_x > (cumsum - 1.0) / (np.arange(1, len(sorted_x) + 1)))[0]
        if len(rho_idx) > 0:
            rho = rho_idx[-1] + 1
            tau = (cumsum[rho - 1] - 1.0) / rho
        else:
            tau = x_new.min() - 1.0 / n

        x_new = np.maximum(x_new - tau, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / s_new

        x = x_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    ProjSubgrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    ProjSubgrad done (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))
    return best_a, best_m


def coordinate_descent(a0, time_limit, rng_seed=77):
    """Adaptive coordinate descent polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score all positions by their total contribution to peak lags
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            candidates = np.arange(n)
            candidates = np.delete(candidates, src)
            dst = int(candidates[np.argmin(scores[candidates])])

            for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall % 30 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print("    CoordDesc (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator: multi-length LP-FW with SSD and coordinate polish.
    Key strategy: emphasize asymmetric sequences, use LP-FW as main optimizer,
    SSD for diversification, and coordinate descent for final polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + str(round(time_budget, 1)) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + str(round(p_score, 6)))
    except Exception:
        parent = None

    # Phase 1: Multi-length parametric search
    print("\n=== Phase 1: Multi-length parametric ===")
    n_candidates = [2000, 3000, 4000, 5000, 6000, 7000, 8000]

    for n in n_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        t_alloc = min(remaining * 0.025, 15.0)
        if t_alloc < 1.0:
            continue
        a_param, v = parametric_init(n, t_alloc)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "param_n" + str(n), t_global)

    if best_seq is None:
        n_fb = 5000
        best_seq, _ = parametric_init(n_fb, 10.0)
        best_score = evaluate_safe(best_seq)
        print("  Fallback init: n=" + str(n_fb) + ", score=" + str(round(best_score, 6)))

    # Phase 2: Main LP-FW on best sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW main (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_out, _ = lp_fw_optimized(norm_sum1(best_seq), t2, solver,
                                   initial_K=400, max_K=2500, add_per_iter=25)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(n_best), t_global)

    # Phase 3: SSD for diversification
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: SSD diversification (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, _ = soft_subgradient_mind(norm_sum1(best_seq), t3, top_k=60, beta=3.5,
                                         c0=3.0, restart_every=2500)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ssd_n" + str(n_best), t_global)

    # Phase 4: LP-FW round 2 (deeper optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW round 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.22, 300.0)
    if t4 > 30:
        a_out, _ = lp_fw_optimized(norm_sum1(best_seq), t4, solver,
                                   initial_K=500, max_K=3000, add_per_iter=30)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(n_best), t_global)

    # Phase 5: Try scaling to other lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    other_ns = []
    for delta in [-1000, -500, 500, 1000, 1500, 2000]:
        cand_n = n_best + delta
        if 1500 <= cand_n <= 9000 and cand_n != n_best:
            other_ns.append(cand_n)

    for new_n in other_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n=== Phase 5: Scale to n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t5 = min(remaining * 0.18, 120.0)
        if t5 > 25:
            a_out, _ = lp_fw_optimized(a_long, t5, solver,
                                       initial_K=300, max_K=1800, add_per_iter=20)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "scale_n" + str(new_n), t_global)

    # Phase 6: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Coordinate polish (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.20, 180.0)
    if t6 > 15:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t6, rng_seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1_n" + str(n_best), t_global)

    # Phase 7: Final LP-FW deep optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t7 = min(remaining * 0.45, 350.0)
        if t7 > 30:
            a_out, _ = lp_fw_optimized(norm_sum1(best_seq), t7, solver,
                                       initial_K=600
