def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def basis_evaluate(coeffs, n, n_fourier, n_poly):
    """Evaluate the objective given basis coefficients."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    # Fourier modes
    for i in range(n_fourier):
        freq = i + 1
        amp = coeffs[2 * i]
        phase = coeffs[2 * i + 1]
        a += amp * np.sin(2 * np.pi * freq * x + phase)
    # Polynomial modes (shifted Legendre-like)
    poly_start = 2 * n_fourier
    for i in range(n_poly):
        a += coeffs[poly_start + i] * x ** (i + 1)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    ac = fast_autocorr(a)
    score = 2.0 * n * ac.max()
    return score

        elif start == 2:
            # Stronger U-shape
            coeffs = np.zeros(n_coeffs)
            coeffs[0] = -0.5
            coeffs[n_fourier * 2] = 0.2  # polynomial term
        else:
            coeffs = rng.normal(0, 0.1, n_coeffs)
        
        # Use L-BFGS-B
        try:
            result = minimize(
                lambda c: basis_evaluate(c, n, n_fourier, n_poly),
                coeffs,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8}
            )
            if result.fun < best_score:
                best_score = result.fun
                best_coeffs = result.x.copy()
        except Exception:
            pass
    
    if best_coeffs is None:
        return None
    
    # Reconstruct
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for i in range(n_fourier):
        freq = i + 1
        a += best_coeffs[2 * i] * np.sin(2 * np.pi * freq * x + best_coeffs[2 * i + 1])
    for i in range(n_poly):
        a += best_coeffs[2 * n_fourier + i] * x ** (i + 1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def nesterov_mw_update(a0, time_limit, top_k=80, beta_init=8.0, c0=5.0,
                        momentum_init=0.5, rng_seed=42, verbose=False):
    """Nesterov-accelerated multiplicative weight update."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    v = np.zeros(n)  # velocity for Nesterov
    mu = momentum_init
    gamma = c0
    stall = 0
    cur_k = top_k
    cur_beta = beta_init
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        beta_t = (t - 1.0) / (t + 5.0)
        y = x + mu * (x - x_prev) + beta_t * (mu - 1) * (x_prev - x_prev)
        # Simpler: use standard Nesterov
        y = (1 + mu) * x - mu * x_prev
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        k_max = min(cur_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        # Compute subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        # Multiplicative weight update
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Update momentum (polyak averaging style)
        mu = 0.999 * mu + 0.001 * 0.5  # slowly decay to 0.5
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameters
        if stall > 5000:
            cur_beta = max(cur_beta * 0.95, 3.0)
            cur_k = min(cur_k + 10, 200)
        
        if stall > 10000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            mu = 0.8  # boost momentum after restart
            cur_k = min(cur_k + 20, 300)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Nesterov-MW t={t} ({elapsed:.0f}s): J={J:.6f}, k={cur_k}, beta={cur_beta:.1f}, mu={mu:.3f}")
    
    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=False, top_K_init=100):
    """Improved LP-based Frank-Wolfe with warm starts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    K = top_K_init
    K_max = min(3000, 2 * n - 1)
    iter_count = 0
    b_prev = None  # warm start
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, K_max)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 30, K_max)
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Backtracking line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 2.0), min(1.0, best_alpha * 3.0), 0.3, 0.15]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()
        
        iter_count += 1
        
        # Adapt K more aggressively
        if iter_count % 5 == 0:
            K = min(K + 40, K_max)
        
        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW2 iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")
    
    return best_a, best_m


def smooth_grad_optimize(a0, time_limit, verbose=False, tau_schedule=None, rng_seed=100):
    """Smoothed projected gradient with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = 0.5
    stall = 0
    total_iters = 0
    
    if tau_schedule is None:
        tau_schedule = [5, 10, 20, 50, 100, 200, 400]
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        stall = 0
        phase_start = time.monotonic()
        phase_limit = max(20, time_limit / len(tau_schedule))
        
        while time.monotonic() - phase_start < phase_limit and stall < 200:
            ac = fast_autocorr(a)
            ac_max = ac.max()
            ac_shifted = tau * (ac - ac_max)
            exp_ac = np.exp(ac_shifted)
            p = exp_ac / exp_ac.sum()
            
            p_cum = np.concatenate([[0.0], np.cumsum(p)])
            k_lo = np.arange(n)
            k_hi = np.minimum(k_lo + n, 2 * n - 1)
            S = p_cum[k_hi] - p_cum[k_lo]
            cross = sp_correlate(p, a, mode='valid')
            g = a * S + cross
            g_tangent = g - g.mean()
            
            a_new = project_simplex(a - eta * g_tangent)
            m_new = fast_autocorr(a_new).max()
            
            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                eta = min(eta * 1.01, 2.0)
            else:
                a = a_new
                stall += 1
                if stall > 50:
                    eta *= 0.85
                if stall > 100:
                    pert = rng.normal(0, 0.003, n) * a
                    a_p = np.maximum(a + pert, 0.0)
                    a = a_p / a_p.sum()
                    stall = 60
            
            total_iters += 1
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad tau={tau} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def pair_coordinate_descent(a0, time_limit, verbose=False, rng_seed=456):
    """Coordinate descent with pair-swap moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    n_top = 30
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k_idx = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k_idx[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]
        
        improved = False
        
        # Try moves from high-sensitivity to low-sensitivity positions
        high_sens_idx = np.where(in_peak)[0]
        low_sens_idx = np.where(~in_peak)[0]
        
        if len(high_sens_idx) == 0 or len(low_sens_idx) == 0:
            stall += 1
            break
        
        # Sort by sensitivity
        high_order = high_sens_idx[np.argsort(sens[high_sens_idx])[::-1]]
        low_order = low_sens_idx[np.argsort(sens[low_sens_idx])]
        
        n_try = min(5000, time_limit * 100)  # limit operations
        for trial in range(n_try):
            if time.monotonic() - t0 > time_limit:
                break
            
            src = int(high_order[trial % len(high_order)])
            dst = int(low_order[trial % len(low_order)])
            
            if a[src] < 1e-12:
                continue
            
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                a = a / s
                M_new = fast_autocorr(a).max()
                if M_new < best_m - 1e-16:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[src] += amt  # undo
                    a[dst] -= amt
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairCD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PairCD final: J={J:.6f}")
    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_asymmetric_u(n, sharpness=2.0, asymmetry=0.3):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2 + asymmetry * x
    a = np.maximum(a, 0.01)
    return norm_sum1(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Basis-parameterized optimization (Fourier + polynomial) via L-BFGS-B
2. Nesterov-accelerated multiplicative weight updates with adaptive params

import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize

_solver = None


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_asym_u(n, sharpness=2.0, asym=0.3):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2 + asym * x
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def basis_evaluate(coeffs, n, n_fourier, n_poly):
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for i in range(n_fourier):
        freq = i + 1
        amp = coeffs[2 * i]
        phase = coeffs[2 * i + 1]
        a += amp * np.sin(2 * np.pi * freq * x + phase)
    poly_start = 2 * n_fourier
    for i in range(n_poly):
        a += coeffs[poly_start + i] * x ** (i + 1)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def _basis_obj_wrapper(coeffs, n, n_fourier, n_poly):
    return basis_evaluate(coeffs, n, n_fourier, n_poly)


def basis_optimize(n, time_limit, n_fourier=12, n_poly=4, rng_seed=0):
    t0 = time.monotonic()

    best_coeffs = None
    n_starts = 25

    for start in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        if start == 0:
            coeffs = rng.normal(0, 0.03, n_coeffs)
        elif start == 1:
            coeffs = np.zeros(n_coeffs)
            coeffs[0] = -0.3
        elif start == 2:
            coeffs = np.zeros(n_coeffs)
            coeffs[0] = -0.5
            coeffs[2 * n_fourier] = 0.15
        elif start == 3:
            coeffs = np.zeros(n_coeffs)
            coeffs[0] = -0.4
            coeffs[2] = 0.15
            coeffs[3] = np.pi
        elif start == 4:
            coeffs = np.zeros(n_coeffs)
            coeffs[0] = -0.6
            coeffs[2 * n_fourier] = -0.1
            coeffs[2 * n_fourier + 1] = 0.1
        else:
            coeffs = rng.normal(0, 0.08, n_coeffs)
            if start % 2 == 0:
                coeffs[0] = -abs(coeffs[0]) - 0.2

        try:
            result = minimize(
                _basis_obj_wrapper, coeffs,

        a += best_coeffs[2 * n_fourier + i] * x ** (i + 1)

def nesterov_mw_update(a0, time_limit, top_k=80, beta_init=8.0, c0=5.0,
                        momentum_init=0.6, rng_seed=42, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    mu = momentum_init
    gamma = c0
    stall = 0
    cur_k = top_k
    cur_beta = beta_init

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        y = (1.0 + mu) * x - mu * x_prev
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Decay momentum
        mu = 0.9995 * mu + 0.0005 * 0.4

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000:
            cur_beta = max(cur_beta * 0.95, 3.0)
            cur_k = min(cur_k + 10, 250)

        if stall > 12000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            mu = 0.8
            cur_k = min(cur_k + 20, 350)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Nesterov-MW t={t} ({elapsed:.0f}s): J={J:.6f}, k={cur_k}, beta={cur_beta:.1f}")

    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=False, top_K_init=100):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    K = top_K_init
    K_max = min(4000, 2 * n - 1)
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 30, K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 2.0), min(1.0, best_alpha * 3.0), 0.3, 0.15, 0.08]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 5 == 0:
            K = min(K + 50, K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def smooth_grad_optimize(a0, time_limit, verbose=False, rng_seed=100):

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        stall = 0
        phase_start = time.monotonic()
        phase_limit = max(20.0, (time_limit - (phase_start - t0)) / len(tau_schedule))

        while time.monotonic() - phase_start < phase_limit and stall < 200:
            ac = fast_autocorr(a)
            ac_max = ac.max()
            ac_shifted = tau * (ac - ac_max)
            exp_ac = np.exp(ac_shifted)

def pair_coordinate_descent(a0, time_limit, verbose=False, rng_seed=456):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        top_k_idx = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k_idx[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False
        high_sens_idx = np.where(in_peak)[0]
        low_sens_idx = np.where(~in_peak)[0]

        if len(high_sens_idx) == 0 or len(low_sens_idx) == 0:
            stall += 1
            break

        high_order = high_sens_idx[np.argsort(sens[high_sens_idx])[::-1]]
        low_order = low_sens_idx[np.argsort(sens[low_sens_idx])]

        max_ops = 3000
        for trial in range(max_ops):
            if time.monotonic() - t0 > time_limit:
                break

            src = int(high_order[trial % len(high_order)])
            dst = int(low_order[trial % len(low_order)])

            if a[src] < 1e-12:
                continue

            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                s_val = a.sum()
                a = a / s_val
                M_new = fast_autocorr(a).max()
                if M_new < best_m - 1e-16:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[src] += amt
                    a[dst] -= amt

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairCD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PairCD final: J={J:.6f}")
    return best_a, best_m

                    best_score = sc2
                    best_seq = norm_sum1(a_b2).copy()
                    print(f"  *** NEW BEST [basis_v2_n{n0}]: {sc2:.6f} (t={time.monotonic()-t_global:.0f}s)")

        # Try different lengths with basis
        remaining = time_budget - (time.monotonic() - t_global)
        for new_n in [n0 + 500, n0 - 500, n0 + 1000]:
            if remaining < 30:
                break
            if 2000 <= new_n <= 8000:
                a_bn = basis_optimize(new_n, min(50.0, remaining * 0.3), n_fourier=10, n_poly=3, rng_seed=new_n % 500)
                if a_bn is not None:
                    a_bn = np.maximum(a_bn, 0.0)
                    sc_n = evaluate_safe(a_bn / a_bn.sum())
                    if np.isfinite(sc_n) and sc_n < best_score:
                        best_score = sc_n
                        best_seq = norm_sum1(a_bn).copy()
                        print(f"  *** NEW BEST [basis_n{new_n}]: {sc_n:.6f} (t={time.monotonic()-t_global:.0f}s)")
                remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 2: Nesterov MW ===
    print(f"\n=== Phase 2: Nesterov MW (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 350.0)

    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mw_update(a_in, t2, top_k=80, beta_init=8.0,
                                       c0=5.0, momentum_init=0.6,
                                       rng_seed=42, verbose=True)
        a_out = np.maximum(a_out, 0.0)
        sc = evaluate_safe(a_out / a_out.sum())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = norm_sum1(a_out).copy()
            print(f"  *** NEW BEST [nesterov_mw]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

        # Second run with different params
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t2b = min(remaining * 0.12, 200.0)
            if t2b > 30:
                a_out2, _ = nesterov_mw_update(norm_sum1(best_seq), t2b, top_k=50,
                                                beta_init=15.0, c0=3.0,
                                                momentum_init=0.7, rng_seed=123,
                                                verbose=False)
                a_out2 = np.maximum(a_out2, 0.0)
                sc2 = evaluate_safe(a_out2 / a_out2.sum())
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = norm_sum1(a_out2).copy()
                    print(f"  *** NEW BEST [nesterov_mw2]: {sc2:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 3: LP-FW ===
    print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.18, 320.0)

    if t3 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t3, solver, verbose=True, top_K_init=100)
        a_out = np.maximum(a_out, 0.0)
        sc = evaluate_safe(a_out / a_out.sum())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = norm_sum1(a_out).copy()
            print(f"  *** NEW BEST [lpfw]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 4: Smooth gradient ===
    print(f"\n=== Phase 4: Smooth gradient (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 200.0)

    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_grad_optimize(a_in, t4, verbose=True, rng_seed=200)
        a_out = np.maximum(a_out, 0.0)
        sc = evaluate_safe(a_out / a_out.sum())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = norm_sum1(a_out).copy()
            print(f"  *** NEW BEST [smooth]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.08, 120.0)

    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_coordinate_descent(a_in, t5, verbose=True, rng_seed=456)
        a_out = np.maximum(a_out, 0.0)
        sc = evaluate_safe(a_out / a_out.sum())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = norm_sum1(a_out).copy()
            print(f"  *** NEW BEST [coord]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    trial_lengths = []
    for delta in [500, 1000, -500, -1000, 1500, 2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.25, 100.0)
        if t6a > 20:
            a_out, _ = nesterov_mw_update(a_long, t6a, top_k=60, beta_init=8.0,
                                           c0=4.0, momentum_init=0.6,
                                           rng_seed=new_n % 1000, verbose=False)
            a_out = np.maximum(a_out, 0.0)
            sc = evaluate_safe(a_out / a_out.sum())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = norm_sum1(a_out).copy()
                print(f"  *** NEW BEST [mw_n{new_n}]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 5000:
            t6b = min(remaining * 0.25, 80.0)
            if t6b > 20:
                a_out, _ = lp_frank_wolfe_v2(norm_sum1(interpolate_sequence(best_seq, new_n)),
                                              t6b, solver, verbose=False, top_K_init=100)
                a_out = np.maximum(a_out, 0.0)
                sc = evaluate_safe(a_out / a_out.sum())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = norm_sum1(a_out).copy()
                    print(f"  *** NEW BEST [lp_n{new_n}]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 7: Final combined optimization ===
    print(f"\n=== Phase 7: Final optimization (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    if remaining > 30:
        t7a = min(remaining * 0.30, 130.0)
        if t7a > 20 and n_best <= 6000:
            a_out, _ = lp_frank_wolfe_v2(norm_sum1(best_seq), t7a, solver,
                                           verbose=False, top_K_init=200)
            a_out = np.maximum(a_out, 0.0)
            sc = evaluate_safe(a_out / a_out.sum())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = norm_sum1(a_out).copy()
                print(f"  *** NEW BEST [lp_final]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.25, 80.0)
            if t7b > 15:
                a_out, _ = pair_coordinate_descent(norm_sum1(best_seq), t7b,
                                                    verbose=False, rng_seed=789)
                a_out = np.maximum(a_out, 0.0)
                sc = evaluate_safe(a_out / a_out.sum())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = norm_sum1(a_out).copy()
                    print(f"  *** NEW BEST [coord_final]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7c = min(remaining * 0.35, 100.0)
            if t7c > 15:
                a_out, _ = nesterov_mw_update(norm_sum1(best_seq), t7c, top_k=100,
                                               beta_init=12.0, c0=4.0,
                                               momentum_init=0.5, rng_seed=999,
                                               verbose=False)
                a_out = np.maximum(a_out, 0.0)
                sc = evaluate_safe(a_out / a_out.sum())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = norm_sum1(a_out).copy()
                    print(f"  *** NEW BEST [mw_final]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
