where g_t is a subgradient of f at a_t.

With Nesterov acceleration:
- y_t = a_t + (t-1)/(t+2) * (a_t - a_{t-1})
- g_t = subgrad f(y_t)
- a_{t+1} = (2/(t+2)) * y_t + (t/(t+2)) * (a_t+1 from mirror descent step on y_t)

- t=1: y_1 = a_0, compute g_1, a_1 = mirror_step(a_0, g_1)

where alpha_t = 2/(t+2), beta_t = (t-1)/(t+2).

Actually, the standard Nesterov accelerated gradient for mirror descent is:
- y_t = (1 - alpha_t) * a_t + alpha_t * a_star  (a_star is the "heavy ball" position)
- Update: a_{t+1} = mirror_step(y_t, g_t) + momentum

- y = a + momentum * (a - a_prev)  (extrapolation)

This gives a smoother gradient and can be more effective.

import numpy as np
import time
import cvxpy as cp
import sys

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def grad_autocorr(a, k, n):
    """Gradient of conv[k] with respect to a."""
    g = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        idx = np.arange(j_start, j_end + 1)
        g[idx] = 2.0 * a[k - idx]
    return g

def compute_soft_grad(a, ac, n, beta):
    """Compute soft gradient using log-sum-exp."""
    # Use the top peaks for efficiency
    n_ac = len(ac)
    max_ac = ac.max()
    # Shift for numerical stability
    shifted = beta * (ac - max_ac)
    # For large n, only use top elements
    n_top = min(200, n_ac)
    top_idx = np.argsort(ac)[::-1][:n_top]
    top_vals = shifted[top_idx]
    w = np.exp(top_vals)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(n_top) / n_top
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-10:
            continue
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += wi * 2.0 * a[k_int - idx]
    return g

def mirror_descent_step(a, g, eta):
    """Mirror descent step: a_new ∝ a * exp(-eta * g), normalized to sum=1."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_a = log_a - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def normalize_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_score(a):
    """Compute J = 2*n*max_conv/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def subgradient_optimize(a0, time_limit, rng, verbose=True):
    """Nesterov-accelerated mirror descent with soft-max subgradient."""
    n = len(a0)
    a = normalize_sum1(a0)
    a_prev = a.copy()
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_max = ac.max()
    t0 = time.monotonic()
    
    # Adaptive parameters
    eta = 0.1  # learning rate
    beta = 10.0  # soft-max temperature
    iter_count = 0
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        y = a + beta_t * (a - a_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y
        
        # Compute autocorrelation and gradient at y
        ac_y = fast_autocorr(y)
        g = compute_soft_grad(y, ac_y, n, beta)
        
        # Mirror descent step
        a_new = mirror_descent_step(y, g, eta)
        
        # Nesterov combination
        alpha = 2.0 / (t + 2.0)
        a_combined = (1.0 - alpha) * a + alpha * a_new
        a_combined = np.maximum(a_combined, 1e-15)
        sa = a_combined.sum()
        if sa < 1e-15:
            a_combined = a_new
        else:
            a_combined = a_combined / sa
        
        a_prev = a
        a = a_combined
        
        # Check improvement
        ac_a = fast_autocorr(a)
        if ac_a.max() < best_max - 1e-12:
            best_max = ac_a.max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameter adjustment
        if stall > 500:
            eta *= 0.7
            beta *= 1.3
            stall = 0
            if verbose and eta < 0.001:
                print(f"    SG: eta too small, restarting")
                # Perturb and restart
                perturb = rng.normal(0, 0.01, n) * best_a
                a = normalize_sum1(best_a + perturb)
                a_prev = a.copy()
                eta = 0.05
                beta = 5.0
        
        iter_count += 1
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print(f"    SG iter={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, beta={beta:.1f}")
    
    return best_a, best_max

def lp_polish(a0, time_limit, solver, max_constraints=2000, verbose=True):
    """LP-based cutting plane optimization."""
    n = len(a0)
    a = normalize_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    # Initialize constraints with top lags
    init_K = min(300, 2 * n - 1)
    ac_y = fast_autocorr(y)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()
        
        # Manage constraint set
        if len(constraint_set) > max_constraints:
            margins = np.array([M_curr - ac_y[k] for k in constraint_set])
            keep = np.argsort(margins)[:max_constraints]
            constraint_list_tmp = sorted(constraint_set)
            constraint_set = set(int(constraint_list_tmp[i]) for i in keep)
        
        constraint_list = sorted(constraint_set)
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP: minimize M s.t. coeffs @ g <= M, sum(g) = 1, g >= 0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce constraints and retry
            constraint_set = set(constraint_list[:init_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        sg = g_val.sum()
        if sg < 1e-15:
            continue
        g_dir = g_val / sg
        
        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            add_n = min(30, len(violations))
            top_v = violations[np.argsort(conv_ys[violations])[::-1][:add_n]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 1e-15)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 1e-15)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new

    J = 2.0 * n * best_m
    if verbose:

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    sn = a_new.sum()
    if sn < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / sn

def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0, 1, n)
    i = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5]:
        candidates.append(normalize_sum1((i + 1) ** p))
        candidates.append(normalize_sum1((n - i) ** p))
    
    # Beta-like
    for alpha in [0.5, 1.0, 2.0]:
        for beta_p in [0.5, 1.0, 2.0]:
            candidates.append(normalize_sum1(
                np.maximum(i + 1, 0.5) ** alpha * np.maximum(n - i, 0.5) ** beta_p))
    
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.15, 0.25, 0.35]:
            candidates.append(normalize_sum1(np.exp(-((x - c)**2) / (2*sig*sig))))
    
    # Sinusoidal
    for freq in [1, 2, 3]:
        candidates.append(normalize_sum1(np.maximum(np.sin(np.pi*freq*x)**2, 0.01)))
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Random smooth
    for _ in range(5):
        rough = rng.uniform(0, 1, n // 10 + 1)
        smooth = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(rough)), rough)
        candidates.append(normalize_sum1(smooth))
    
    return candidates
