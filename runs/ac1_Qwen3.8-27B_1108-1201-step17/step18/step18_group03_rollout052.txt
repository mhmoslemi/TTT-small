def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a / s)
    return float(2.0 * n * ac.max())

def project_simplex(z):
    """Project z onto {a >= 0, sum(a) = 1}."""
    n = len(z)
    z = np.asarray(z, dtype=np.float64)
    # Sort in decreasing order
    z_sorted = np.sort(z)[::-1]
    # Find tau
    cumsum = np.cumsum(z_sorted)
    # tau_k = (cumsum[k] - 1) / (k+1)
    k_vals = np.arange(1, n + 1, dtype=np.float64)
    tau_candidates = (cumsum - 1.0) / k_vals
    # Find the largest k such that z_sorted[k-1] > tau_candidates[k-1]
    # (using 0-indexed: z_sorted[k] > tau_candidates[k] for k=0,...,n-1)
    valid = z_sorted > tau_candidates
    if not valid.any():
        tau = tau_candidates[-1]
    else:
        k_max = np.max(np.where(valid)[0])
        tau = tau_candidates[k_max]
    result = np.maximum(z - tau, 0.0)
    # Ensure exact sum
    s = result.sum()
    if s > 0:
        result = result / s
    else:
        result = np.ones(n) / n
    return result

    ac = fast_autocorr(a)

    shifted = ac - max_ac

    mask = w > w_thresh
    if mask.sum() == 0:
        mask[np.argmax(ac)] = True
    w = w * mask

def project_simplex(z):
    n = len(z)
    z = np.asarray(z, dtype=np.float64).copy()
    z_sorted = np.sort(z)[::-1]
    cumsum = np.cumsum(z_sorted)
    k_vals = np.arange(1, n + 1, dtype=np.float64)
    tau_candidates = (cumsum - 1.0) / k_vals
    valid = z_sorted > tau_candidates
    if not valid.any():
        tau = tau_candidates[-1]
    else:
        k_max = np.max(np.where(valid)[0])
        tau = tau_candidates[k_max]
    result = np.maximum(z - tau, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
        improved = False
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            n_pairs = min(5, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]
            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * ac[k2i]
                candidate_mask = np.ones(n, dtype=bool)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    candidate_mask[s2:e2 + 1] = False
                candidate_mask[src] = False
                if candidate_mask.sum() < 5:
                    candidate_mask = np.ones(n, dtype=bool)
                    candidate_mask[src] = False
                cand_scores = scores.copy()
                cand_scores[~candidate_mask] = np.inf
                dst = int(np.argmin(cand_scores))
                if dst == src:
                    continue
                found = False
                for frac in [0.02, 0.04, 0.06, 0.1, 0.15]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
        if not improved:
            stall += 1
            if stall > max_stall:
                break
            if stall % 5 == 0:
                n_moves = int(rng.integers(2, 8))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
        if verbose and stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDescent ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    return best_a, best_m

def parametric_search(n, time_limit, best_init, best_score_init):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = norm_sum1(best_init)
    best_val = best_score_init

    def try_update(a, label):
        nonlocal best_a, best_val
        a_n = norm_sum1(a)
        ac = fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = a_n.copy()
            if time.monotonic() - t0 < 5:
                print(f"    Param [{label}]: J={val:.6f}")

    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        try_update((i_arr + 1.0) ** p, f"p{p}")
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        try_update((n - i_arr) ** p, f"ip{p}")
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta_p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta_p
            try_update(a, f"ba{alpha}b{beta_p}")
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            try_update(a, f"g{c}_{sigma}")
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for sigma in [0.05, 0.08, 0.12, 0.15, 0.2]:
                a = np.exp(-((x - c1) ** 2) / (2 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2 * sigma * sigma))
                try_update(a, f"bi{sigma}")
    for freq in [1, 2, 3, 4]:
        for ph in [0.0, 0.125, 0.25, 0.375, 0.5]:
            try_update(np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01), f"sn{freq}_{ph}")
    for rate in [1, 2, 3, 5, 8, 12, 20]:
        try_update(np.exp(-rate * x), f"er{rate}")
        try_update(np.exp(rate * (1 - x)), f"enr{rate}")
    # Double exponential (sharp at both ends)
    for r1 in [2, 5, 10]:
        for r2 in [2, 5, 10]:
            a = np.exp(-r1 * x) + np.exp(-r2 * (1 - x))
            try_update(a, f"de{r1}_{r2}")
    # Triangle / tent
    for width in [0.5, 0.7, 0.9]:
        a = np.maximum(0, width - np.abs(x - 0.5) / (1.0 - width))
        try_update(a, f"tent{width}")
    # Sinc-like
    for bw in [5, 10, 20, 50]:
        t = (x - 0.5) * bw
        a = np.sinc(t)
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-10:
            try_update(a, f"sinc{bw}")
    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.0f}s): best J={best_val:.6f}")
    return best_a, best_val

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = eval_score_normalized(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.exp(-8 * x * (1 - x) * 4))
        best_score = eval_score_normalized(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_score_normalized(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Parametric
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 5:
        print(f"\n=== Phase 1: Parametric (n={n0}) ===")
        a_param, _ = parametric_search(n0, t1, best_seq, 2.0 * n0 * fast_autocorr(best_seq).max())
        check_update(a_param, "parametric")

    # Phase 2: Nesterov Mirror Descent (fast, many iterations)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: NSGD-Mirror (n={n_best}) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_optimize(a_in, t2, c0=3.0, beta_start=8.0, beta_end=35.0, verbose=True)
        check_update(a_out, f"nsmd_n{n_best}")

    # Phase 3: LP-FW (precision)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, initial_K=400, max_K=2500, add_per_iter=40, verbose=True)
        check_update(a_out, f"lp_n{n_best}")

    # Phase 4: Nesterov Euclidean (different dynamics)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: NSGD-Euclid (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_euclidean_optimize(a_in, t4, c0=2.5, beta_start=12.0, beta_end=50.0, verbose=True)
        check_update(a_out, f"neuc_n{n_best}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: CoordDescent (n={n_best}) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_polish(a_in, t5, verbose=True)
        check_update(a_out, f"coord_n{n_best}")

    # Phase 6: Try longer sequences
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    length_candidates = []
    for new_n in [n_best + 1000, n_best + 2000, n_best - 500, 6000, 7000, 8000]:
        if 3000 <= new_n <= 9000 and new_n != n_best:
            length_candidates.append(new_n)
    length_candidates = sorted(set(length_candidates))[:3]

    for new_n in length_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        # Quick NSGD
        t6a = min(remaining * 0.5, 150.0)
        if t6a > 15:
            a_out, _ = nesterov_mirror_optimize(a_long, t6a, c0=3.0, beta_start=8.0, beta_end=30.0, verbose=False)
            check_update(a_out, f"nsmd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 7000:
            t6b = min(remaining * 0.5, 120.0)
            if t6b > 20:
                a_in = norm_sum1(best_seq if len(best_seq) == new_n else a_long)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, initial_K=300, max_K=2000, add_per_iter=30, verbose=False)
                check_update(a_out, f"lp_n{new_n}")

    # Phase 7: Final LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 7000:
        t7 = min(remaining * 0.4, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=500, max_K=3000, add_per_iter=50, verbose=True)
            check_update(a_out, f"lp_final_n{n_best}")

    # Phase 8: Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final CoordDescent ===")
    if remaining > 15:
        t8 = min(remaining * 0.4, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_polish(a_in, t8, verbose=True)
            check_update(a_out, f"coord_final_n{n_best}")

    # Final
    if best_seq is not None:
        fs = eval_score_normalized(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = eval_score_normalized(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
