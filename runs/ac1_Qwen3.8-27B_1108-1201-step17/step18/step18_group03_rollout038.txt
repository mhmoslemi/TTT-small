def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_gradient_soft(a, ac, beta, top_k=None):
    """Compute soft gradient of max(ac) using top-k lags with softmax weighting."""
    n = len(a)
    size = len(ac)
    if top_k is not None:
        top_k = min(top_k, size)
    top_idx = np.argsort(ac)[::-1][:top_k] if top_k else np.arange(size)
    top_vals = ac[top_idx]
    # Softmax weighting
    max_v = top_vals.max()
    w = np.exp(beta * (top_vals - max_v))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum
    # Compute gradient
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # grad[j] += 2 * wi * a[k-j] for valid 0 <= k-j <= n-1
        # a[k-j] is valid when 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k
        j_start = max(0, k_int - (n - 1))
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            j_idx = np.arange(j_start, j_end + 1)
            a_vals = a[k_int - j_idx]
            # Special case: when k_int = 2*j, the factor should be 1 not 2
            # But for large n this is negligible, so we use 2 for all
            grad[j_idx] += 2.0 * wi * a_vals
    return grad

def mirror_descent_step(a, grad, eta):
    """Mirror descent step on the simplex using exponential weighting."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def parametric_init(n, best_init=None, best_score=None):
    """Try various parametric shapes and return the best."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_val = float('inf')
    
    if best_init is not None:
        best_a = norm_sum1(best_init)
        best_val = 2.0 * n * fast_autocorr(best_a).max()
    
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    
    # Beta-like
    for alpha in [0.5, 1.0, 1.5]:
        for beta in [0.5, 1.0, 1.5]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
    
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.15, 0.25, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    
    # Bimodal
    for sigma in [0.08, 0.15]:
        candidates.append(np.exp(-((x - 0.15) ** 2) / (2 * sigma * sigma)) + 
                         np.exp(-((x - 0.85) ** 2) / (2 * sigma * sigma)))
    
    # Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [10, 30]:
            candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))
    
    # Sinusoidal
    for freq in [1, 2, 3]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))
    
    # U-shape (high at ends)
    candidates.append(1.0 / (0.1 + x * (1 - x)))
    candidates.append((x * (1 - x)) ** 0.3 + 0.1)
    
    for cand in candidates:
        a = norm_sum1(cand)
        val = 2.0 * n * fast_autocorr(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
    
    return best_a, best_val

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_soft_grad(a, ac, beta, top_k):
    """Compute soft subgradient of max(ac) at point a using top-k lags."""
    n = len(a)
    size = len(ac)
    top_k = min(top_k, size)
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[top_idx]
    max_v = top_vals.max()
    w = np.exp(beta * (top_vals - max_v))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - (n - 1))
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            j_idx = np.arange(j_start, j_end + 1)
            a_vals = a[k_int - j_idx]
            grad[j_idx] += 2.0 * wi * a_vals
    return grad


def mirror_step(a, grad, eta):
    """Mirror descent step: multiplicative update on simplex."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def amd_optimize(a0, time_limit, beta_start, beta_end, top_k_start, top_k_end,
                 c0, max_iters=50000000, verbose=False):
    """Accelerated mirror descent with adaptive softmax parameter."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 3000
    rng = np.random.default_rng(777)

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 5000.0)
        beta_t = beta_start + (beta_end - beta_start) * frac
        top_k_t = int(top_k_start + (top_k_end - top_k_start) * frac)

        # Nesterov extrapolation
        beta_nest = (t - 1.0) / (t + 1.5)
        y = x + beta_nest * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_grad(y, ac, beta_t, top_k_t)

        eta = gamma / (np.sqrt(t) + 1.0)
        x_new = mirror_step(y, g, eta)

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            noise = rng.normal(0, 0.015, n) * best_a
            x_r = norm_sum1(best_a + noise)
            x = x_r
            x_prev = x_r.copy()
            stall = 0
            gamma *= 0.85
            max_stall = int(max_stall * 0.75)
            if max_stall < 100:
                break

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, b={beta_t:.1f}, tk={top_k_t}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    AMD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=2500,
                   add_per_iter=30, verbose=False):
    """LP-based Frank-Wolfe with cutting planes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed += 1
            if failed > 10:
                constraint_list = sorted(constraint_set)[:init_K]
                constraint_set = set(constraint_list)
                failed = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LP done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_polish(a0, time_limit, rng_seed=42, verbose=False):
    """Coordinate descent: move mass from high-contributing to low-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0

        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            n_pairs = min(4, len(idx))
            pair_idx = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_idx:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Find best destination
                n_cand = min(500, n)
                if n <= 500:
                    cand_rng = np.arange(n)
                else:
                    cand_rng = rng.integers(0, n, size=n_cand)

                scores = np.zeros(len(cand_rng))
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    dc = k2i - cand_rng
                    valid = (dc >= 0) & (dc < n)
                    if valid.any():
                        scores[valid] += a[dc[valid]] * ac[k2i]
                scores[cand_rng == src] = 1e10
                scores += a[cand_rng] * 0.05

                best_cand = cand_rng[np.argmin(scores)]
                if best_cand == src:
                    continue

                for frac in [0.02, 0.05, 0.1, 0.15]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_cand] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

                if improved:
                    break

        if not improved:
            stall += 1
            if stall > max_stall:
                n_pert = int(rng.integers(5, 15))
                esc = False
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.005, 0.04))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                esc = True
                                break
                if not esc and stall > max_stall * 3:
                    break
            elif verbose and stall % 100 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def parametric_init(n, best_init=None, best_val=None):
    """Try asymmetric parametric shapes; return best."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    if best_init is not None:
        best_a = norm_sum1(best_init)
        best_v = 2.0 * n * fast_autocorr(best_a).max()
    else:
        best_a = np.ones(n) / n
        best_v = 2.0

    candidates = []

    # Asymmetric power laws (key: these are the most promising)
    for p in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 2.5, 3.0]:
        candidates.append((1.0 - x + 1e-6) ** p)
        candidates.append((1.0 - i_arr / (n - 1) + 1e-6) ** p)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5]:
        candidates.append((x + 1e-6) ** p)

    # Asymmetric exponential
    for rate in [1, 2, 3, 5, 7, 10]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * x))

    # Beta-like asymmetric
    for a_param in [0.5, 1.0, 1.5, 2.0]:
        for b_param in [0.5, 1.0, 2.0, 3.0]:
            candidates.append(np.maximum(x, 1e-6) ** (a_param - 1) * np.maximum(1 - x, 1e-6) ** (b_param - 1))

    # Gaussian (symmetric, for comparison)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.15, 0.25, 0.35]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sigma * sigma)))

    # Bimodal asymmetric
    for sigma in [0.08, 0.12]:
        candidates.append(2.0 * np.exp(-((x - 0.1) ** 2) / (2 * sigma * sigma)) +
                         np.exp(-((x - 0.8) ** 2) / (2 * sigma * sigma)))

    # Sigmoid (asymmetric step)
    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20]:
            candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))
            candidates.append(1.0 / (1.0 + np.exp(s * (x - c))))

    for cand in candidates:
        a = norm_sum1(cand)
        ac = fast_autocorr(a)
        val = 2.0 * n * ac.max()
        if val < best_v:
            best_v = val
            best_a = a.copy()

    return best_a, best_v


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator for minimizing 2*n*max(conv(a,a))/(sum(a))^2."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        best_seq, _ = parametric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric init ===
    print(f"\n=== Phase 1: Parametric init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        a_p, _ = parametric_init(n0, best_seq, 2.0 * n0 * fast_autocorr(best_seq).max())
        try_update(a_p, "param")

    # === Phase 2: AMD on current length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AMD (n={n_best}) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 20:
        a_out, _ = amd_optimize(norm_sum1(best_seq), t2,
                                 beta_start=5.0, beta_end=50.0,
                                 top_k_start=20, top_k_end=120,
                                 c0=4.0, verbose=True)
        try_update(a_out, f"amd_n{n_best}")

    # === Phase 3: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 15 and n_best <= 6000:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t3, solver,
                                   initial_K=300, max_K=2500,
                                   add_per_iter=40, verbose=True)
        try_update(a_out, f"lp_n{n_best}")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord polish (n={n_best}) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        a_out, _ = coord_polish(norm_sum1(best_seq), t4, rng_seed=99, verbose=True)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ===")

    length_cands = []
    for delta in [-500, -1000, 500, 1000, 1500, 2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000:
            length_cands.append(new_n)
    for rn in [3000, 4000, 5000, 6000, 7000]:
        if rn != n_best and 2000 <= rn <= 7000:
            length_cands.append(rn)
    length_cands = sorted(set(length_cands))

    for new_n in length_cands[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  n={new_n} (rem={remaining:.0f}s)")
        a_interp = interpolate_seq(best_seq, new_n)

        t_amd = min(remaining * 0.3, 100.0)
        if t_amd > 10:
            a_out, _ = amd_optimize(a_interp, t_amd,
                                     beta_start=5.0, beta_end=35.0,
                                     top_k_start=20, top_k_end=80,
                                     c0=3.0, verbose=False)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 5500:
            t_lp = min(remaining * 0.3, 100.0)
            if t_lp > 10:
                a_out, _ = lp_frank_wolfe(a_interp, t_lp, solver,
                                           initial_K=200, max_K=1500,
                                           add_per_iter=25, verbose=False)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final AMD (n={n_best}) ===")
    t6 = min(remaining * 0.25, 150.0)
    if t6 > 10:
        a_out, _ = amd_optimize(norm_sum1(best_seq), t6,
                                 beta_start=30.0, beta_end=80.0,
                                 top_k_start=60, top_k_end=200,
                                 c0=5.0, verbose=False)
        try_update(a_out, f"amd_final_n{n_best}")

    # === Phase 7: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    t7 = min(remaining * 0.35, 200.0)
    if t7 > 10 and n_best <= 6000:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t7, solver,
                                   initial_K=400, max_K=3000,
                                   add_per_iter=50, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.3, 100.0)
    if t8 > 5:
        a_out, _ = coord_polish(norm_sum1(best_seq), t8, rng_seed=314, verbose=False)
        try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x + 1e-6) ** 1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
