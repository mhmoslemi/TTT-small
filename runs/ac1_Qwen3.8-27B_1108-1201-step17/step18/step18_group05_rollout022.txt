def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    # grad[j] = sum_m weights[j+m] * a[m] for valid m
    
    # This is essentially (weights * a) cross-correlation
    # Actually it's: grad = correlate(weights, a, mode='valid') ... no
    # Let me think again.
    
    # grad[j] = sum_{k} weights[k] * (d ac[k]/d a[j])
    # d ac[k]/d a[j] = a[k-j] if 0 <= k-j < n, else 0
    # So grad[j] = sum_{k: 0<=k-j<n} weights[k] * a[k-j]
    # Let i = k-j: grad[j] = sum_{i: 0<=i<n, 0<=j+i<2n-1} weights[j+i] * a[i]
    
    # This is a sliding dot product: for each j, dot product of a with weights[j:j+n] (padded)
    
    # Pad weights to handle boundaries
    # weights has length 2n-1
    # For j=0: grad[0] = sum_{i=0}^{n-1} weights[i] * a[i]
    # For j=1: grad[1] = sum_{i=0}^{n-1} weights[1+i] * a[i]
    # For j=n-1: grad[n-1] = sum_{i=0}^{n-1} weights[n-1+i] * a[i]
    
    # This is like: for each j, compute dot(a, weights[j:j+n])
    # We can do this efficiently with FFT or just a loop for small n
    # Actually, this is the cross-correlation of a with weights:
    # corr[j] = sum_i a[i] * weights[j+i]
    
    # Using scipy's correlate: correlate(weights, a, mode='valid') gives
    # result[j] = sum_i weights[j+i] * a[i] for j = 0, ..., len(weights)-len(a)
    # len(weights) = 2n-1, len(a) = n, so result has length n-1+1 = n. 
    
    # Wait, correlate(v, w, mode='valid') where len(v) >= len(w) gives:
    # result[k] = sum_i v[k+i] * w[i] for k = 0, ..., len(v)-len(w)
    
    # So correlate(weights, a, mode='valid') gives n values, which is what we want!
    
    grad = sp_correlate(weights, a, mode='valid')
    # But we need to verify: the gradient of the smooth max is:
    # d/d a[j] sum_k w_k * ac[k] where w_k = exp(tau*ac_k) / sum exp(tau*ac_k)
    # = tau * sum_k w_k * ac_k * d ac[k]/d a[j] ... no
    # Actually the smooth max is S(a) = (1/tau) * log(sum_k exp(tau * ac[k]))
    # dS/da[j] = sum_k (tau * exp(tau*ac[k]) / sum exp(tau*ac[k])) * d ac[k]/da[j]
    # = tau * sum_k w_k * d ac[k]/da[j]
    # = tau * grad[j] (as computed above)
    
    # So the actual gradient is tau * grad. But since we're doing gradient descent,
    # we can absorb tau into the step size. Let's return grad without the tau factor
    # and let the caller handle the scaling.
    
    return grad, weights, ac


def nesterov_smooth_optimize(a0, time_limit, tau_init=30.0, tau_final=500.0,
                              eta_init=0.5, momentum=0.9, verbose=True, rng_seed=0):
    """Nesterov-accelerated smooth projected gradient descent with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = eta_init
    stall = 0
    total_iters = 0
    t_momentum = momentum
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_init * (tau_final / tau_init) ** progress
        
        # Nesterov lookahead
        y = a + t_momentum * v
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Compute smooth gradient at y
        grad, weights, ac = smooth_max_grad(y, tau=tau)
        
        # Project gradient onto simplex tangent space
        g_tangent = grad - grad.mean()
        
        # Backtracking line search
        best_alpha = eta
        best_m_trial = best_m
        base_m = fast_autocorr(a).max()
        
        for alpha in [eta, eta*0.5, eta*0.25, eta*0.1, eta*2.0]:
            if alpha <= 0:
                continue
            a_trial = a - alpha * g_tangent
            a_trial = project_simplex(a_trial)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_alpha = alpha
        
        a_new = a - best_alpha * g_tangent
        a_new = project_simplex(a_new)
        
        # Update momentum
        v = t_momentum * v + (a_new - a)
        # Decay momentum slightly
        t_momentum = 0.999 * t_momentum + 0.001 * momentum
        a = a_new
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                eta *= 0.8
            if stall > 500:
                # Perturbation
                pert = rng.normal(0, 0.01, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a_p = a_p / a_p.sum()
                m_p = fast_autocorr(a_p).max()
                if m_p < best_m:
                    best_m = m_p
                    best_a = a_p.copy()
                    a = a_p
                    v = np.zeros(n)
                    stall = 0
                    eta = eta_init * 0.5
                else:
                    a = a_p
                    v = np.zeros(n)
                    stall = 300
        
        total_iters += 1
        
        if verbose and total_iters % 500 == 0:
            J = 2.0 * n * best_m
            print(f"    Nesterov it={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K_init=300, top_K_max=None, use_all=False):
    """Improved LP-based Frank-Wolfe with dual-guided constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            # Select binding constraints: top K by value, but also include
            # some that are close to M_curr for diversity
            sorted_idx = np.argsort(ac)[::-1]
            binding_idx = sorted_idx[:K_sel]
        
        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 100, top_K_max)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Backtracking with more aggressive exploration
        best_alpha = 1.0 / (iter_count + 1.0)
        best_trial_m = M_curr
        alphas = [best_alpha, min(1.0, best_alpha * 2), min(1.0, best_alpha * 3),
                  0.3, 0.2, 0.1, min(1.0, best_alpha * 0.5)]
        for alpha in alphas:
            if alpha <= 0 or alpha > 1:
                continue
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        iter_count += 1
        
        # Adapt K
        if stall_count > 20:
            K = min(K + 50, top_K_max)
            stall_count = 0
        
        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW2 iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")
    
    return best_a, best_m


def peak_shaving_cd(a0, time_limit, n_peaks=5, verbose=True, rng_seed=0):
    """Coordinate descent that specifically targets the top autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        top_k = np.argsort(ac)[::-1][:n_peaks]
        
        improved = False
        
        for k_idx in top_k:
            ki = int(k_idx)
            # The peak at lag ki is: sum_j a[j] * a[ki-j] for valid j
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[ki - idx]
            
            # Find the pair (i, ki-i) with maximum contribution
            max_pi = np.argmax(contrib)
            src = idx[max_pi]
            dst = ki - src
            
            if src == dst:
                continue
            
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try moving mass from src to a random low-contribution position
            # Find positions that don't contribute to any top peak
            in_peak = np.zeros(n, dtype=bool)
            for kk in top_k:
                ks = int(kk)
                s = max(0, ks - n + 1)
                e = min(n - 1, ks)
                in_peak[s:e+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Try a few candidates
            for attempt in range(5):
                if len(candidates) > 10:
                    dst_choice = candidates[rng.integers(len(candidates))]
                else:
                    dst_choice = candidates[rng.integers(len(candidates))]
                
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst_choice] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if time.monotonic() - t0 > time_limit:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def dual_guided_lp(a0, time_limit, solver, verbose=True):
    """LP-FW using dual variables to guide constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K = 200
    max_K = min(3000, 2 * n - 1)
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, max_K)
        # Select constraints: top K values
        binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, max_K)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stall += 1
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            stall += 1
            continue
        
        g_dir = b_val / b_sum
        
        # Line search
        best_alpha = 1.0 / (iter_count + 2.0)
        best_m_trial = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha*2), 0.3, 0.5, 0.1, 0.05]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                K = min(K + 100, max_K)
        
        iter_count += 1
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    DualLP iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")
    
    return best_a, best_m


def multi_temperature_smooth(a0, time_limit, verbose=True, rng_seed=0):
    """Smooth gradient with multiple temperature scales (coarse to fine)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Phase 1: Coarse (low tau, large steps)
    # Phase 2: Medium
    # Phase 3: Fine (high tau, small steps)
    
    phases = [
        (10.0, 30.0, 1.0, 30),   # tau_range, eta, max_stall
        (30.0, 100.0, 0.5, 50),
        (100.0, 300.0, 0.2, 80),
        (300.0, 500.0, 0.1, 100),
    ]
    
    for tau_lo, tau_hi, eta_init, max_stall in phases:
        if time.monotonic() - t0 > time_limit:
            break
        
        phase_time = min(time_limit - (time.monotonic() - t0), time_limit / len(phases))
        phase_start = time.monotonic()
        eta = eta_init
        stall = 0
        v = np.zeros(n)  # momentum
        momentum = 0.9
        
        while time.monotonic() - phase_start < phase_time and stall < max_stall:
            elapsed_phase = time.monotonic() - phase_start
            progress = min(1.0, elapsed_phase / max(phase_time, 1.0))
            tau = tau_lo * (tau_hi / tau_lo) ** progress
            
            # Nesterov
            y = a + momentum * v
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
            
            grad, weights, ac = smooth_max_grad(y, tau=tau)
            g_tangent = grad - grad.mean()
            
            a_new = project_simplex(a - eta * g_tangent)
            v = momentum * v + (a_new - a)
            a = a_new
            stall += 1
        
            m_new = fast_autocorr(a).max()
            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a.copy()
                stall = 0  # Reset stall on improvement
            # Actually, let's be more lenient with stall
            # Only count as stall if we haven't improved in the last 100 iterations
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultiTemp phase tau=[{tau_lo},{tau_hi}] ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
