   A=np.fft.rfft(a, fft_size)

   cmax=c.max()
   if cmax <= 1e-300:
       g=np.zeros(n); return (c,cmax,g) if need_c else (cmax,g)
   if active_tol > 0 or tau_frac <= 0:
       thresh=cmax - max(1e-300, active_tol*cmax)
       idx=np.where(c >= thresh)[0]

          idx=idx[np.argsort(c[idx])[::-1][:top_k]]
       K=len(idx)
       w=np.full(K,1.0/K)

def compute_gradient(a, fft_size, top_k=64, tau_frac=0.05, mix=0.1, active_tol=0.0):
    n=len(a); L=2*n-1
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A*A, fft_size)[:L]
    c = np.maximum(c, 0.0)
    cmax = float(np.max(c)) if L else 0.0
    if cmax <= 1e-300:
        return 0.0, np.zeros(n, dtype=np.float64)
    if top_k > L: top_k = L
    if top_k <= 0: top_k = 1
    if active_tol > 0.0:
        thresh = cmax - max(1e-300, active_tol*cmax)
        idx = np.where(c >= thresh)[0]
        if len(idx) > top_k:
            idx = idx[np.argsort(c[idx])[::-1][:top_k]]
        K = len(idx)
        w = np.full(K, 1.0/K, dtype=np.float64)
    else:
        if top_k >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(c, L-top_k)[L-top_k:]
        K = len(idx)
        vals = c[idx]
        mu = float(vals.max())
        tau = max(1e-300, float(tau_frac)*cmax)
        logw = (vals - mu) / tau
        logw = np.clip(logw, -700.0, 0.0)
        ew = np.exp(logw)
        s = float(ew.sum())
        if s <= 1e-300:
            w = np.zeros(K, dtype=np.float64)
            w[int(np.argmax(vals))] = 1.0
        else:
            w = ew / s
        if mix > 0.0 and K > 0:
            w = (1.0 - mix) * w + mix * (1.0/K)
            s = float(w.sum())
            if s > 1e-300:
                w /= s
            else:
                w = np.full(K,1.0/K)
    wfull = np.zeros(L, dtype=np.float64)
    wfull[idx] = w
    W = np.fft.rfft(wfull, fft_size)
    corr = np.fft.irfft(np.conj(A)*W, fft_size)[:n]
    g = 2.0 * corr
    g = np.maximum(np.where(np.isfinite(g), g, 0.0), 0.0)
    return cmax, g
```

"""
Search for low-peak non-negative step coefficients minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Approach:
- Work on the probability simplex (sum(a)=1, a>=0), where the objective is 2*n*max(conv(a,a)).
- Use exact FFT-based convolution and a fast FFT cross-correlation to compute weighted
  subgradients / smoothed gradients of max convolution.
- Alternate several optimizers:
  1) Nesterov-style mirror (KL) descent on the simplex.
  2) Adam optimization in log/softmax parameterization.
  3) Gradient-guided coordinate/block swaps and local polish.
- Explore diverse low-dimensional initializations and nearby sequence lengths.
- Optional small SciPy LP escape direction if SciPy is available.
- Always return the best finite sequence found before the time budget expires.
"""

import numpy as np
import time

try:
    import scipy.optimize as _scipy_opt
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False

_best = {"seq": None, "score": float("inf")}


def next_pow2(n):
    n = int(max(1, n))
    return 1 << (n - 1).bit_length()


def normalize_a(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones_like(arr, dtype=np.float64) / float(arr.size)
    return arr / s


def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    m = float(np.max(z)) if z.size else 0.0
    e = np.exp(np.clip(z - m, -745.0, 0.0))
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    return e / s


def fast_conv_max(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return 0.0
    L = 2 * n - 1
    if fft_size is None:
        fft_size = next_pow2(L)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:L]
    c = np.maximum(c, 0.0)
    return float(np.max(c)) if L else 0.0


def compute_gradient(a, fft_size, top_k=64, tau_frac=0.05, mix=0.10, active_tol=0.0):
    """
    Return (cmax, g) where g is a non-negative subgradient/smoothed gradient of
    max conv(a,a) with respect to a, using weighted active peaks.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return 0.0, np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:L]
    c = np.maximum(c, 0.0)
    cmax = float(np.max(c)) if L else 0.0
    if cmax <= 1e-300:
        return cmax, np.zeros(n, dtype=np.float64)

    if top_k > L:
        top_k = L
    if top_k <= 0:
        top_k = 1

    if active_tol > 0.0:
        thresh = cmax - max(1e-300, float(active_tol) * cmax)
        idx = np.where(c >= thresh)[0]
        if idx.size > top_k:
            idx = idx[np.argsort(c[idx])[::-1][:top_k]]
        K = int(idx.size)
        if K == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
            K = 1
        w = np.full(K, 1.0 / K, dtype=np.float64)
    else:
        if top_k >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(c, L - top_k)[L - top_k:]
        K = int(idx.size)
        if K == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
            K = 1
        vals = c[idx]
        mu = float(vals.max())
        tau = max(1e-300, float(tau_frac) * cmax)
        logw = (vals - mu) / tau
        logw = np.clip(logw, -700.0, 0.0)
        ew = np.exp(logw)
        s = float(ew.sum())
        if s <= 1e-300:
            w = np.zeros(K, dtype=np.float64)
            w[int(np.argmax(vals))] = 1.0
        else:
            w = ew / s
        if mix > 0.0:
            w = (1.0 - mix) * w + mix * (1.0 / K)
            s = float(w.sum())
            if s > 1e-300:
                w = w / s
            else:
                w = np.full(K, 1.0 / K, dtype=np.float64)

    wfull = np.zeros(L, dtype=np.float64)
    wfull[idx] = w
    W = np.fft.rfft(wfull, fft_size)
    corr = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
    g = 2.0 * corr
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    return cmax, g

def make_uniform(n):
    n = int(max(1, n))
    return np.ones(n, dtype=np.float64) / float(n)


def make_gaussian(n, center=0.5, width=0.15):
    n = int(max(2, n))
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.exp(-(((x - center) / width) ** 2))
    return normalize_a(a)


def make_two_gaussian(n, c1=0.28, c2=0.72, w1=0.08, w2=0.12, ratio=1.0):
    n = int(max(2, n))
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.exp(-(((x - c1) / w1) ** 2)) + ratio * np.exp(-(((x - c2) / w2) ** 2))
    return normalize_a(a)


def make_sinc(n, bw=5.0, center=0.5):
    n = int(max(2, n))
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.sinc((x - center) * bw) ** 2
    a = 0.02 + np.maximum(a, 0.0)
    return normalize_a(a)


def make_cosine(n, mode=1, amp=0.6):
    n = int(max(2, n))
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = (1.0 - amp) + amp * (np.cos(2.0 * np.pi * mode * x) ** 2)
    return normalize_a(a)


def make_osc(n, freq=3, phase=0.0, power=1.0):
    n = int(max(2, n))
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    s = np.sin(2.0 * np.pi * freq * x + phase)
    a = 0.2 + 0.8 * np.maximum(s, 0.0) ** power
    return normalize_a(a)


def make_ramp(n, increasing=True, power=0.7):
    n = int(max(2, n))
    x = np.arange(1, n + 1, dtype=np.float64)
    if increasing:
        a = x ** power
    else:
        a = (float(n) + 1.0 - x) ** power
    return normalize_a(np.maximum(a, 0.0))


def make_block(n, frac=0.5, start=0.0):
    n = int(max(2, n))
    a = np.zeros(n, dtype=np.float64)
    s = int(max(0, min(n - 1, int(start * n))))
    length = int(max(1, min(n - s, frac * n)))
    a[s:s + length] = 1.0
    return normalize_a(a)


def make_random_smooth(n, seed=0, n_freq=6):
    n = int(max(2, n))
    rng = np.random.default_rng(seed)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.full(n, 0.55, dtype=np.float64)
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0.0, 0.16 / np.sqrt(float(f)))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        a = a + coeff * np.sin(2.0 * np.pi * f * x + phase)
    a = 0.05 + np.maximum(a, 0.0)
    return normalize_a(a)


def make_zadoff_like(n, m=1):
    n = int(max(2, n))
    if m <= 1:
        m = 1
    x = np.arange(n, dtype=np.float64)
    ex = np.pi * float(m) * (x * (x + 1.0)) / float(n)
    z = np.exp(-1j * ex)
    a = np.abs(z.real) + np.abs(z.imag) + 0.1
    return normalize_a(a)


def interpolate_sequence(a, new_n):
    a = normalize_a(a)
    old_n = int(a.size)
    new_n = int(max(1, new_n))
    if old_n == new_n:
        return a.copy()
    old_x = np.arange(old_n, dtype=np.float64) + 0.5
    new_x = np.arange(new_n, dtype=np.float64) + 0.5
    y = np.interp(new_x, old_x, a)
    y = np.maximum(y, 0.0)
    s = float(y.sum())
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return y / s


def zero_small_values(a, rel=1e-15):
    arr = normalize_a(a)
    if arr.size == 0:
        return arr
    thr = float(rel) / float(arr.size)
    arr[arr < thr] = 0.0
    s = float(arr.sum())
    if s < 1e-15:
        return normalize_a(a)
    return arr / s


def mirror_optimize(
    a0,
    time_limit,
    seed=0,
    tau_frac_init=0.35,
    tau_frac_min=0.03,
    top_k=56,
    mix=0.12,
    eta_base=0.55,
    max_step=1.6,
    restart_every=3000
):
    if time_limit <= 0:
        return normalize_a(a0)
    a = normalize_a(a0)
    n = int(a.size)
    if n <= 1:
        return a
    fft_size = next_pow2(2 * n - 1)
    eps = 1e-300

    log_a = np.log(np.maximum(a, eps))
    log_prev = log_a.copy()
    best_a = a.copy()
    best_J = 2.0 * n * fast_conv_max(a, fft_size)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    t = 0
    stall = 0
    restart_count = 0
    eta_scale = float(eta_base)
    curr_top_k = int(top_k)
    tau_boost = 1.0

    while time.monotonic() - start < time_limit:
        t += 1
        elapsed = time.monotonic() - start
        frac = max(0.0, 1.0 - elapsed / max(1e-9, time_limit))
        base_tau = tau_frac_min + (tau_frac_init - tau_frac_min) * frac
        tau_frac = min(1.25, base_tau * tau_boost)
        tau_boost = max(1.0, tau_boost * 0.9985)

        beta = (t - 1.0) / (t + 1.5)
        log_y = log_a + beta * (log_a - log_prev)
        if not np.all(np.isfinite(log_y)):
            log_y = log_a.copy()
            log_prev = log_a.copy()

        y = softmax(log_y)
        cmax_y, g = compute_gradient(y, fft_size, curr_top_k, tau_frac, mix, 0.0)
        g_inf = float(np.max(g)) if g.size else 1.0
        if g_inf <= 1e-300:
            break

        eta = eta_scale / (np.sqrt(t + 1.0) * g_inf)
        step = eta * g
        step = np.clip(step, -max_step, max_step)

        log_next = log_y - step
        m = float(np.max(log_next)) if log_next.size else 0.0
        log_next = log_next - m
        e = np.exp(np.clip(log_next, -745.0, 0.0))
        s = float(e.sum())
        if not np.isfinite(s) or s <= 1e-300:
            log_next = log_a.copy()
        else:
            log_next = log_next - np.log(s)

        old_log_a = log_a.copy()
        log_a = log_next
        log_prev = old_log_a

        a = softmax(log_a)
        cmax = fast_conv_max(a, fft_size)
        J = 2.0 * n * cmax

        if np.isfinite(J) and J < best_J - 1e-15:
            best_J = float(J)
            best_a = a.copy()
            stall = 0
            eta_scale = min(1.35, eta_scale * 1.02)
        else:
            stall += 1
            if not np.isfinite(J) or J > 3.0 * best_J:
                eta_scale = max(0.03, eta_scale * 0.7)

        if stall > restart_every:
            restart_count += 1
            strength = min(0.35, 0.03 + 0.025 * min(restart_count, 12))
            log_best = np.log(np.maximum(best_a, eps))
            log_next = log_best + rng.normal(0.0, strength, n)
            m = float(np.max(log_next)) if log_next.size else 0.0
            log_next = log_next - m
            e = np.exp(np.clip(log_next, -745.0, 0.0))
            s = float(e.sum())
            if s > 1e-300:
                log_a = log_next - np.log(s)
                log_prev = log_a.copy()
            else:
                log_a = log_best.copy()
                log_prev = log_a.copy()
            eta_scale = max(0.05, min(0.9, eta_scale * 0.85))
            tau_boost = min(3.0, tau_boost * 1.45)
            curr_top_k = min(128, curr_top_k + 6)
            restart_every = max(2500, restart_every + 500)
            stall = 0
        elif t % 5000 == 0:
            log_a = log_a + rng.normal(0.0, 0.004, n)
            log_prev = log_a.copy()
            stall = min(stall, restart_every // 2)

    return normalize_a(best_a)


def adam_optimize(
    a0,
    time_limit,
    seed=0,
    tau_frac_init=0.25,
    tau_frac_min=0.015,
    top_k=64,
    mix=0.10,
    lr=0.045,
    beta1=0.90,
    beta2=0.999,
    restart_every=6000
):
    if time_limit <= 0:
        return normalize_a(a0)
    a = normalize_a(a0)
    n = int(a.size)
    if n <= 1:
        return a
    fft_size = next_pow2(2 * n - 1)
    epsz = 1e-300
    z = np.log(np.maximum(a, epsz))
    z = z - float(np.mean(z))
    z = np.clip(np.where(np.isfinite(z), z, 0.0), -60.0, 60.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    best_a = a.copy()
    best_J = 2.0 * n * fast_conv_max(a, fft_size)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    t = 0
    stall = 0
    restart_count = 0
    lr_cur = float(lr)
    curr_top_k = int(top_k)
    tau_boost = 1.0

    while time.monotonic() - start < time_limit:
        t += 1
        elapsed = time.monotonic() - start
        frac = max(0.0, 1.0 - elapsed / max(1e-9, time_limit))
        base_tau = tau_frac_min + (tau_frac_init - tau_frac_min) * frac
        tau_frac = min(1.25, base_tau * tau_boost)
        tau_boost = max(1.0, tau_boost * 0.998)

        a = softmax(z)
        cmax, g = compute_gradient(a, fft_size, curr_top_k, tau_frac, mix, 0.0)
        J = 2.0 * n * cmax
        if np.isfinite(J) and J < best_J - 1e-15:
            best_J = float(J)
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if cmax <= 1e-300:
            break

        alpha = float(np.dot(a, g))
        grad_z = a * (g - alpha)
        grad_z = grad_z - float(np.mean(grad_z))
        grad_z = np.clip(np.where(np.isfinite(grad_z), grad_z, 0.0), -10.0, 10.0)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z * grad_z
        mhat = m / (1.0 - (beta1 ** t))
        vhat = v / (1.0 - (beta2 ** t))
        step_scale = lr_cur / np.sqrt(1.0 + 0.00001 * t)
        z = z - step_scale * mhat / (np.sqrt(vhat) + 1e-8)

        if t % 1000 == 0:
            z = z - float(np.mean(z))
            z = np.where(np.isfinite(z), z, 0.0)
            z = np.clip(z, -60.0, 60.0)

        if stall > restart_every:
            restart_count += 1
            strength = min(0.28, 0.02 + 0.015 * min(restart_count, 12))
            log_best = np.log(np.maximum(best_a, 1e-300))
            z = log_best + rng.normal(0.0, strength, n)
            z = z - float(np.mean(z))
            z = np.where(np.isfinite(z), z, 0.0)
            z = np.clip(z, -60.0, 60.0)
            m.fill(0.0)
            v.fill(0.0)
            lr_cur = max(0.01, min(0.16, lr_cur * 0.9))
            curr_top_k = min(128, curr_top_k + 6)
            tau_boost = min(3.0, tau_boost * 1.5)
            restart_every = max(4000, restart_every + 1000)
            stall = 0
        elif t % 8000 == 0:
            z = z + rng.normal(0.0, 0.003, n)
            z = z - float(np.mean(z))
            stall = min(stall, restart_every // 2)

    return normalize_a(best_a)


def gradient_swap_polish(
    a0,
    time_limit,
    seed=0,
    top_k=96,
    tau_frac=0.015,
    mix=0.04,
    active_tol=0.0,
    max_trials=35,
    max_stall=450
):
    if time_limit <= 0:
        return normalize_a(a0)
    a = normalize_a(a0)
    n = int(a.size)
    if n <= 1:
        return a
    fft_size = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_J = 2.0 * n * fast_conv_max(a, fft_size)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    outer = 0
    stall = 0

    while time.monotonic() - start < time_limit and stall < max_stall:
        outer += 1
        if active_tol > 0.0:
            cmax, g = compute_gradient(a, fft_size, top_k, 0.0, 0.0, active_tol)
        else:
            cmax, g = compute_gradient(a, fft_size, top_k, tau_frac, mix, 0.0)

        J = 2.0 * n * cmax
        if np.isfinite(J) and J < best_J - 1e-15:
            best_a = a.copy()
            best_J = float(J)
            stall = 0

        if cmax <= 1e-300:
            break

        improved = False

        # Block move: transfer some mass from high gradient/mass sources to low-gradient sites.
        src_score = g * np.maximum(a, 1e-300)
        n_src = min(12, max(2, n // 200))
        src_idx = np.argsort(src_score)[::-1][:n_src]
        n_dst = min(24, max(4, n // 80))
        dst_idx = np.argsort(g)[:n_dst]

        src_set = set(int(x) for x in src_idx)
        dst_list = [int(x) for x in dst_idx if int(x) not in src_set]
        if not dst_list:
            dst_list = [int(dst_idx[0])]
        src_arr = np.array([int(x) for x in src_idx if int(x) not in set(dst_list)], dtype=int)

        if src_arr.size > 0 and len(dst_list) > 0:
            src_mass = float(a[src_arr].sum())
            if src_mass > 1e-8:
                for frac in (0.2, 0.5, 0.8, 0.35, 0.65):
                    delta = frac * src_mass
                    a2 = a.copy()
                    sub = a[src_arr] * (delta / src_mass)
                    a2[src_arr] = a[src_arr] - sub
                    a2[dst_list] = a2[dst_list] + (delta / float(len(dst_list)))
                    a2 = np.where(a2 < 0.0, 0.0, a2)
                    ssum = float(a2.sum())
                    if ssum > 1e-15:
                        a2 = a2 / ssum
                        J2 = 2.0 * n * fast_conv_max(a2, fft_size)
                        if np.isfinite(J2) and J2 < best_J - 1e-15:
                            a = a2
                            best_a = a2.copy()
                            best_J = float(J2)
                            improved = True
                            stall = 0
                            break

        # Pair moves guided by gradient difference.
        if not improved:
            n_src2 = min(8, max(2, n // 250))
            n_dst2 = min(20, max(3, n // 100))
            src_idx2 = np.argsort(g)[::-1][:n_src2]
            dst_idx2 = np.argsort(g)[:n_dst2]
            trials = 0
            for si in src_idx2:
                si = int(si)
                if a[si] <= 1e-14:
                    continue
                for di in dst_idx2:
                    di = int(di)
                    if di == si or g[di] >= g[si]:
                        continue
                    for frac in (0.25, 0.5, 0.75, 0.1, 0.4):
                        if trials >= max_trials:
                            break
                        trials += 1
                        amt = a[si] * frac
                        a2 = a.copy()
                        a2[si] -= amt
                        a2[di] += amt
                        a2 = np.where(a2 < 0.0, 0.0, a2)
                        ssum = float(a2.sum())
                        if ssum < 1e-15:
                            continue
                        a2 = a2 / ssum
                        J2 = 2.0 * n * fast_conv_max(a2, fft_size)
                        if np.isfinite(J2) and J2 < best_J - 1e-15:
                            a = a2
                            best_a = a2.copy()
                            best_J = float(J2)
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            accepted = False
            for _ in range(8):
                si = int(rng.integers(n))
                di = int(rng.integers(n))
                if si == di or a[si] <= 1e-14 or g[si] <= 0.0:
                    continue
                frac = float(rng.uniform(0.05, 0.35))
                amt = a[si] * frac
                a2 = a.copy()
                a2[si] -= amt
                a2[di] += amt
                a2 = np.where(a2 < 0.0, 0.0, a2)
                ssum = float(a2.sum())
                if ssum < 1e-15:
                    continue
                a2 = a2 / ssum
                J2 = 2.0 * n * fast_conv_max(a2, fft_size)
                if np.isfinite(J2) and J2 < best_J - 1e-15:
                    a = a2
                    best_a = a2.copy()
                    best_J = float(J2)
                    stall = 0
                    accepted = True
                    break
            if not accepted and stall > 20:
                strength = min(0.06, 0.004 + 0.0008 * min(stall, 40))
                pert = best_a * rng.normal(0.0, strength, n)
                a = best_a + pert
                a = np.where(a < 0.0, 0.0, a)
                ssum = float(a.sum())
                if ssum > 1e-15:
                    a = a / ssum

        if outer % 50 == 0:
            print(f"    swap polish t={time.monotonic() - start:.0f}s J={best_J:.7f} stall={stall}")

    ac = best_a.copy()
    ac[ac < 1e-18] = 0.0
    s = float(ac.sum())
    if s > 1e-15:
        ac = ac / s
    else:
        ac = best_a
    Jc = 2.0 * n * fast_conv_max(ac, fft_size)
    if np.isfinite(Jc) and Jc < best_J:
        best_a = ac
        best_J = float(Jc)

    return normalize_a(best_a)


def run_hybrid(a0, time_limit, seed, label, mirror_frac=0.35, adam_frac=0.35, swap_frac=0.30):
    if time_limit <= 0:
        return
    a0 = normalize_a(a0)
    total = float(mirror_frac) + float(adam_frac) + float(swap_frac)
    if total <= 0:
        return
    time_limit = float(time_limit)

    if time_limit < 50.0:
        a = gradient_swap_polish(a0, time_limit, seed=seed)
        update_best(a, label + ":swap_only")
        return

    tm = time_limit * float(mirror_frac) / total
    ta = time_limit * float(adam_frac) / total
    ts = max(0.0, time_limit - tm - ta)

    if tm > 25.0:
        am = mirror_optimize(
            a0,
            tm,
            seed=seed,
            tau_frac_init=0.45,
            tau_frac_min=0.025,
            top_k=56,
            mix=0.12,
            eta_base=0.55,
            max_step=1.6,
            restart_every=2500
        )
        update_best(am, label + ":mirror")

    base = _best["seq"] if _best["seq"] is not None else a0
    if ta > 25.0:
        aa = adam_optimize(
            base,
            ta,
            seed=seed + 101,
            tau_frac_init=0.30,
            tau_frac_min=0.012,
            top_k=64,
            mix=0.10,
            lr=0.05,
            restart_every=5000
        )
        update_best(aa, label + ":adam")

    base = _best["seq"] if _best["seq"] is not None else a0
    if ts > 18.0:
        ag = gradient_swap_polish(
            base,
            ts,
            seed=seed + 202,
            top_k=88,
            tau_frac=0.012,
            mix=0.04,
            active_tol=0.0,
            max_trials=35,
            max_stall=350
        )
        update_best(ag, label + ":swap")


def choose_initializations(n, parent, max_count=6):
    items = []
    if parent is not None:
        items.append(("parent", interpolate_sequence(parent, n)))
    items.append(("sinc5", make_sinc(n, 5.0)))
    items.append(("sinc8", make_sinc(n, 8.0)))
    items.append(("two_gauss", make_two_gaussian(n, 0.27, 0.73, 0.08, 0.12, 0.9)))
    items.append(("cos2", make_cosine(n, 2, 0.6)))
    items.append(("gauss", make_gaussian(n, 0.5, 0.18)))
    items.append(("osc3", make_osc(n, 3, 0.7, 1.0)))
    items.append(("rnd", make_random_smooth(n, 123, 8)))
    items.append(("zadoff", make_zadoff_like(n, 3)))

    seen = set()
    out = []
    for label, arr in items:
        arr = normalize_a(arr)
        step = max(1, int(arr.size // 30))
        sig = tuple(np.round(arr[::step], 4).tolist())
        if sig not in seen:
            seen.add(sig)
            out.append((label, arr))
        if len(out) >= int(max_count):
            break
    return out


def candidate_lengths(parent_n, max_count=5):
    base = set()
    if parent_n is not None:
        pn = int(parent_n)
        for delta in (-1000, -700, -500, -300, -100, 100, 300, 500, 700, 1000, 1400, 1800):
            nn = pn + delta
            if 2500 <= nn <= 8000:
                base.add(int(nn))
    for nn in (3600, 4000, 4096, 4400, 4600, 4700, 4900, 5000, 5200, 5500, 5800, 6000, 6500, 7000):
        if 2500 <= nn <= 8000:
            base.add(int(nn))

    out = []
    for nn in base:
        if parent_n is None or int(nn) != int(parent_n):
            out.append(int(nn))
    if not out:
        return out

    arr = np.array(out, dtype=int)
    if parent_n is not None:
        order = np.argsort(np.abs(arr - int(parent_n)))
    else:
        order = np.argsort(np.abs(arr - 4700))
    out = [int(out[i]) for i in order]
    return out[:int(max_count)]


def lp_direction(f, n_lp=500):
    if not _HAVE_SCIPY:
        return None
    f = norm = normalize_a(f)
    n_lp = int(max(120, min(800, n_lp)))
    f = interpolate_sequence(f, n_lp)
    f = f + 1e-8
    f = f / f.sum()
    n = int(f.size)
    L = 2 * n - 1

    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(L, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = _scipy_opt.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:
        try:
            res = _scipy_opt.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds)
        except Exception:
            return None

    if res is None:
        return None
    if not bool(getattr(res, "success", False)):
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, None)
    s = float(g.sum())
    if s > 1e-12:
        return g / s
    return None


def lp_escape_attempt(best_a, t_available, seed=0):
    if not _HAVE_SCIPY or t_available < 160.0:
        return
    if best_a is None:
        return
    n_best = int(best_a.size)
    if n_best > 3500:
        n_lp = 520
    elif n_best > 2000:
        n_lp = 420
    else:
        n_lp = max(220, n_best // 2)
    print(f"  LP escape attempt: n_best={n_best}, n_lp={n_lp}, t={t_available:.0f}s")
    g = lp_direction(best_a, n_lp)
    if g is None:
        print("  LP escape skipped/failed")
        return
    g_full = interpolate_sequence(g, n_best)
    base = normalize_a(best_a)
    for alpha in (0.1, 0.2, 0.35, 0.5, 0.7, 0.9):
        a = (1.0 - alpha) * base + alpha * g_full
        update_best(a, f"lp_a{alpha}")
    print("  LP escape mixes evaluated")


def remaining_time(deadline):
    return max(0.0, float(deadline) - time.monotonic())


def finalize_best(deadline):
    if _best["seq"] is None:
        a = make_sinc(4096, 5.0)
        _best["seq"] = a
        _best["score"] = float("inf")

    a = normalize_a(_best["seq"])
    candidates = [a]
    for rel in (1e-17, 1e-16, 1e-15, 1e-14, 1e-12):
        candidates.append(zero_small_values(a, rel))

    best_seq = a
    best_sc = float("inf")
    for cand in candidates:
        cand = normalize_a(cand)
        sc = exact_score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_seq = cand

    if not np.isfinite(best_sc):
        best_seq = make_uniform(4096)
        best_sc = exact_score(best_seq)

    _best["seq"] = best_seq
    _best["score"] = float(best_sc)
    print(f"  FINAL exact: J={_best['score']:.7f}, n={int(best_seq.size)}")
    return [float(x) for x in best_seq]


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    try:
        budget = max(60.0, min(float(budget_s) - 45.0, 1755.0))
    except Exception:
        budget = 1755.0
    deadline = t0 + budget

    _best["seq"] = None
    _best["score"] = float("inf")

    parent = load_parent()
    if _best["seq"] is None:
        a0 = make_sinc(4096, 5.0)
        update_best(a0, "default_sinc")

    n_best = int(_best["seq"].size) if _best["seq"] is not None else 0
    print(f"  starting best: n={n_best}, J={_best['score']:.7f}, budget={budget:.0f}s")

    # Phase 1: focus on the best known sequence.
    rem = remaining_time(deadline)
    if rem > 150.0:
        t1 = min(rem * 0.38, 720.0)
        if t1 > 60.0:
            print(f"  Phase 1: parent hybrid t={t1:.0f}s")
            run_hybrid(_best["seq"], t1, seed=11, label="p1", mirror_frac=0.34, adam_frac=0.36, swap_frac=0.30)

    # Phase 2: diverse initializations at the current best length.
    rem = remaining_time(deadline)
    if rem > 320.0:
        n_cur = int(_best["seq"].size) if _best["seq"] is not None else 4700
        inits = choose_initializations(n_cur, parent, max_count=7)
        # parent is already handled; skip if present to save time.
        inits = [(lbl, arr) for (lbl, arr) in inits if lbl != "parent"]
        if inits:
            t_each = min(105.0, (rem * 0.42) / float(len(inits)))
            count = 0
            for lbl, arr in inits:
                if remaining_time(deadline) < 180.0:
                    break
                tt = min(t_each, remaining_time(deadline) - 120.0)
                if tt < 45.0:
                    break
                print(f"  Phase 2 init {lbl}: t={tt:.0f}s")
                run_hybrid(arr, tt, seed=1000 + count, label=f"p2_{lbl}", mirror_frac=0.38, adam_frac=0.38, swap_frac=0.24)
                count += 1

    # Phase 3: cheap nearest-interpolation over several lengths.
    rem = remaining_time(deadline)
    if rem > 120.0:
        lens = candidate_lengths(_best["seq"].size, max_count=14)
        for nn in lens:
            if remaining_time(deadline) < 90.0:
                break
            a_int = interpolate_sequence(_best["seq"], nn)
            update_best(a_int, f"quick_len_{nn}")

    # Phase 4: optimize a few promising lengths.
    rem = remaining_time(deadline)
    if rem > 350.0:
        lens = candidate_lengths(_best["seq"].size, max_count=4)
        seen_len = set()
        done_counts = 0
        for nn in lens:
            if remaining_time(deadline) < 220.0:
                break
            if nn in seen_len:
                continue
            seen_len.add(nn)
            a_int = interpolate_sequence(_best["seq"], nn)
            update_best(a_int, f"len_pre_{nn}")
            tt = min(230.0, remaining_time(deadline) - 150.0)
            if tt > 80.0:
                print(f"  Phase 4 optimize length {nn}: t={tt:.0f}s")
                run_hybrid(a_int, tt, seed=3000 + (nn % 997), label=f"len_{nn}", mirror_frac=0.35, adam_frac=0.35, swap_frac=0.30)
            done_counts += 1
            if done_counts >= 3:
                break

    # Phase 5: optional LP escape.
    rem = remaining_time(deadline)
    if rem > 300.0:
        lp_escape_attempt(_best["seq"], rem, seed=17)
        rem = remaining_time(deadline)
        if rem > 110.0:
            tt = min(170.0, rem - 70.0)
            if tt > 45.0:
                print(f"  post-LP hybrid t={tt:.0f}s")
                run_hybrid(_best["seq"], tt, seed=501, label="postlp", mirror_frac=0.30, adam_frac=0.40, swap_frac=0.30)

    # Phase 6: final exact-peak polish.
    rem = remaining_time(deadline)
    if rem > 80.0:
        tt = min(210.0, rem - 55.0)
        if tt > 30.0:
            print(f"  Phase 6 final swap polish t={tt:.0f}s")
            a = gradient_swap_polish(
                _best["seq"],
                tt,
                seed=4242,
                top_k=128,
                tau_frac=0.004,
                mix=0.02,
                active_tol=1e-8,
                max_trials=45,
                max_stall=600
            )
            update_best(a, "final_swap_active")

    rem = remaining_time(deadline)
    if rem > 70.0:
        tt = min(130.0, rem - 45.0)
        if tt > 25.0:
            print(f"  Phase 6 final Adam t={tt:.0f}s")
            a = adam_optimize(
                _best["seq"],
                tt,
                seed=7777,
                tau_frac_init=0.09,
                tau_frac_min=0.004,
                top_k=112,
                mix=0.05,
                lr=0.025,
                restart_every=4000
            )
            update_best(a, "final_adam")

    rem = remaining_time(deadline)
    if rem > 50.0:
        tt = min(90.0, rem - 40.0)
        if tt > 20.0:
            print(f"  final light swap t={tt:.0f}s")
            a = gradient_swap_polish(
                _best["seq"],
                tt,
                seed=9999,
                top_k=128,
                tau_frac=0.002,
                mix=0.01,
                active_tol=1e-9,
                max_trials=40,
                max_stall=500
            )
            update_best(a, "final_swap2")

    return finalize_best(deadline)
