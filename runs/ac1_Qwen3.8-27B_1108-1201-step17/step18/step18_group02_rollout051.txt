        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        
        # Add violated constraints
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_yg > M_y + 1e-8)[0]
        if len(violations) > 0:
            # Add top violators
            viol_amt = conv_yg[violations] - M_y
            top_v = violations[np.argsort(viol_amt)[::-1][:100]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new / sx
        
        y, x = y_new, x_new
        iter_count += 1
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"  LP-FW done ({elapsed:.0f}s): J={2.0*n*best_m:.6f}, iters={iter_count}")
    return best_a, best_m

def smooth_adam_v2(a0, time_limit, tau_init=None, tau_decay=0.9995, lr=1.0,
                   restart_every=2000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    if tau_init is None:
        tau_init = ac.max() * 0.3
    tau = max(tau_init, 1e-8)
    tau_min = tau_init * 0.001
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        
        # Compute gradient of LSE
        ac_scaled = ac / tau
        ac_scaled -= ac_scaled.max()
        exp_vals = np.exp(ac_scaled)
        p = exp_vals / exp_vals.sum()
        
        # grad[i] = 2 * sum_k p[k] * (contributions of a[i] to conv[k])
        # conv[k] = sum_j a[j]*a[k-j], so d(conv[k])/d(a[i]) = a[k-i] + (if valid) a[i-k+2n-2]
        # Actually: d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-(k-(2n-2))]... 
        # More simply: conv = ifft(fft(a)*fft(a)), so d(conv)/d(a[i]) is the contribution of a[i]
        # For linear conv: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/d(a[i]) = a[k-i] + a[i-k+2n-2]... no.
        # d(conv[k])/d(a[i]) = a[k-i] (when 0<=k-i<n) + a[(2n-2-k)-i+... 
        # Actually: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/d(a[i]): appears when j=i (giving a[k-i]) or k-j=i i.e. j=k-i (giving a[i])
        # So d(conv[k])/d(a[i]) = a[k-i] + a[i] (counted twice if k-i = i, i.e., k=2i)
        # Wait no: if j=i, term is a[i]*a[k-i]. If j=k-i, term is a[k-i]*a[i]. These are the same term!
        # So d(conv[k])/d(a[i]) = a[k-i] if 0<=k-i<n, else 0.
        # (The term a[i]*a[k-i] appears only once in the sum, not twice, unless k-i = i.)
        # Actually wait: conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
        # If we differentiate w.r.t. a[i]:
        #   - j = i: gives a[k-i] (valid if 0<=k-i<n)
        #   - k-j = i, i.e., j = k-i: gives a[i] (valid if 0<=k-i<n)
        #   - If i = k-i, i.e., k = 2i: the same term appears once, so derivative is a[i] (not 2*a[i])
        #   - If i != k-i: both terms appear, derivative is a[k-i] + a[i]... 
        #   Wait no. Let me be very careful.
        #   conv[k] = sum_j a[j] * a[k-j]
        #   = ... + a[i] * a[k-i] + ... + a[k-i] * a[i] + ...
        #   These are TWO separate terms (unless i = k-i).
        #   So d(conv[k])/d(a[i]) = a[k-i] + a[i] if i != k-i, else a[i] (just one term).
        #   Hmm wait, that's not right either. Let me think again.
        #   conv[k] = a[0]*a[k] + a[1]*a[k-1] + ... + a[k]*a[0] + ... (for k < n)
        #   Each a[i] appears in at most 2 places: as a[j] when j=i, and as a[k-j] when k-j=i i.e. j=k-i.
        #   So d(conv[k])/d(a[i]) = a[k-i] (from j=i) + a[i] (from j=k-i) if both are valid and i != k-i
        #                        = a[i] (from j=i=k-i) if i = k-i
        #                        = a[k-i] (only one valid) otherwise
        #   
        #   For the gradient of the LSE: grad[i] = sum_k p[k] * d(conv[k])/d(a[i])
        #   = sum_k p[k] * (a[k-i] * I[0<=k-i<n] + a[i] * I[0<=k-i<n, k!=2i])
        #   = a[i] * sum_k p[k] * I[valid k for term a[i]] + sum_k p[k] * a[k-i] * I[valid]
        #   
        #   The first sum: sum_k p[k] for k such that 0<=k-i<n AND k!=2i
        #   ≈ sum_k p[k] (approximately, since k=2i is just one value)
        #   
        #   The second sum: sum_k p[k] * a[k-i] for 0<=k-i<n
        #   = sum_j p[i+j] * a[j] for j = 0 to n-1 (where j = k-i, so k = i+j)
        #   = (flip(p_shifted) * a)[...] 
        #   
        #   More precisely: grad[i] ≈ a[i] * (sum p - p[2i]) + sum_j p[i+j] * a[j]
        #   ≈ 2 * sum_j p[i+j] * a[j] (since a[i]*sum(p) ≈ a[i] and sum_j p[i+j]*a[j] includes the j=i term)
        #   
        #   Actually the simplest correct formula is:
        #   grad = 2 * ifft_contribution... 
        #   
        #   Let me just use the FFT approach: 
        #   conv = ifft(fft(a) * fft(a))
        #   d(conv)/d(a) = 2 * conv_matrix(a) (but this is an n x (2n-1) matrix...)
        #   
        #   Actually, the cleanest way: 
        #   Let F = fft(a, fft_size). Then conv_full = ifft(F*F, fft_size).
        #   d(conv_full[k])/d(a[i]) = 2 * F[i] * (improper)... 
        #   
        #   No, let me just compute it directly using the formula:
        #   grad[i] = sum_k p[k] * (a[k-i] + a[i]) for valid k, with the correction for k=2i.
        #   
        #   For simplicity and correctness, I'll compute:
        #   grad[i] = sum_{k: 0<=k-i<n} p[k] * a[k-i] + sum_{k: 0<=k-i<n, k!=2i} p[k] * a[i]
        #   
        #   The second sum = a[i] * (sum_k p[k] - p[2i] if valid)
        #   ≈ a[i] * (1 - p[2i]) ≈ a[i]
        #   
        #   The first sum = sum_j p[i+j] * a[j] (where j = k-i, valid when 0<=j<n, i.e., 0<=k-i<n)
        #   = correlation of p with a, shifted by i
        #   
        #   So grad[i] ≈ a[i] + (p_shifted * a)[i] where p_shifted is p shifted appropriately.
        #   
        #   Actually, let me just use: grad ≈ 2 * ifft(fft(p) * fft(a))[i] (approximately)
        #   This is the cross-correlation of p and a, which gives sum_j p[j-i+a] * a[j] ... 
        #   
        #   Let me just be practical. The previous program used:
        #   a_flip = a[::-1]
        #   conv_full = ifft(fft(a_flip) * fft(p))
        #   grad = 2 * conv_full[n-1:n-1+n]
        #   
        #   This computes 2 * (flip(a) * p)[n-1 + i] = 2 * sum_j a[n-1-j] * p[n-1+i-j]
        #   = 2 * sum_j a[n-1-j] * p[i-j]
        #   Let l = n-1-j: = 2 * sum_l a[l] * p[i - (n-1-l)]
        #   = 2 * sum_l a[l] * p[i - n + 1 + l]
        #   
        #   Hmm, this is getting confusing. Let me just use the formula from the previous program since it was working.
        
        a_flip = a[::-1].astype(np.float64)
        Af = np.fft.rfft(a_flip, fft_size)
        Pf = np.fft.rfft(p.astype(np.float64), fft_size)
        conv_full = np.fft.irfft(Af * Pf, fft_size)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0.0)  # Ensure non-negative gradient
        
        # Adam update
        m_hat = 0.9 * m_hat + 0.1 * grad
        v_hat = 0.999 * v_hat + 0.001 * (grad * grad)
        bc1 = 1.0 - 0.9 ** t
        bc2 = 1.0 - 0.999 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        log_a = np.log(np.maximum(a, 1e-30)) - step * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a = a_new / a_new.sum()
        
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.05, n) * best_a
            a_new2 = best_a + pert
            a_new2 = np.maximum(a_new2, 1e-10)
            a = a_new2 / a_new2.sum()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                print(f"    SmoothAdam restart t={t}: J={2*n*best_m:.6f}")
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothAdam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")
    
    return best_a, best_m

def coord_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Focus on top peaks
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination
            best_dst = -1
            best_new_m = best_m
            
            # Try moving to random positions (faster than checking all)
            for _ in range(30):
                dst = int(rng.integers(n))
                if dst == src or a[dst] > 0.9 * src_val:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = src_val * frac
