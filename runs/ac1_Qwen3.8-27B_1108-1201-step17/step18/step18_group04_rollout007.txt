                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            # Random perturb to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    break
        if not improved and pass_count % 50 == 0:
            # Larger random jump
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    return best_a, best_m


def peak_equalize(a0, time_limit, seed=42):
    """Targeted moves to equalize top peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        n_peaks = min(10, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_peaks]
        max_v = conv[top_peaks[0]]
        min_top_v = conv[top_peaks[-1]]

        if max_v - min_top_v < 1e-8 * max_v:
            break

        # Focus on the largest peak
        k = int(top_peaks[0])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k - idx]
        total_contrib = contrib.sum()

        # Move mass from top contributor to reduce this peak
        top_contrib_idx = int(idx[np.argmax(contrib)])
        amt = a[top_contrib_idx] * 0.05
        # Move to a "quiet" position
        quiet_positions = np.argsort(a)[:min(50, n // 10)]
        for dst in quiet_positions:
            a_trial = a.copy()
            a_trial[top_contrib_idx] -= amt
            a_trial[int(dst)] += amt
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                break

        if time.monotonic() - t0 > time_limit - 2:
            break

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    _g_best['seq'] = None
    _g_best['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _g_best['seq'] = parent.copy()
            _g_best['score'] = p_score
    except Exception:
        pass

    n0 = 4700
    if parent is not None:
        n0 = len(parent)

    if _g_best['seq'] is None:
        _g_best['seq'] = make_sine(n0)
        _g_best['score'] = evaluate_sequence([float(v) for v in _g_best['seq']])
        print(f"Default init: n={n0}, score={_g_best['score']:.6f}")

    print(f"Starting: n={n0}, score={_g_best['score']:.6f}")

    # === Phase 1: Multi-start MW optimization ===
    print(f"\n=== Phase 1: MW Multi-Start (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_p1 = min(remaining * 0.35, 580.0)

    if t_p1 > 30:
        inits = []
        labels = []
        if parent is not None:
            inits.append(normalize(parent))
            labels.append("parent")
        inits.append(make_sine(n0)); labels.append("sine")
        inits.append(make_gauss(n0, 0.3)); labels.append("gauss30")
        inits.append(make_gauss(n0, 0.4)); labels.append("gauss40")
        inits.append(make_sin_power(n0, 1.5)); labels.append("sinp15")
        inits.append(make_sin_power(n0, 2.0)); labels.append("sinp20")
        inits.append(make_two_hump(n0, 0.25, 0.75, 0.18)); labels.append("2hump")
        inits.append(make_cos_bell(n0, 2)); labels.append("cosbell2")
        inits.append(make_random_smooth(n0, seed=7, n_freq=8)); labels.append("rnd1")
        inits.append(make_random_smooth(n0, seed=99, n_freq=5)); labels.append("rnd2")

        n_inits = len(inits)
        t_per = max(5.0, t_p1 / n_inits)
        for idx in range(n_inits):
            rem = time_budget - 30 - (time.monotonic() - t_global)
            if rem < 10:
                break
            t_avail = min(t_per, rem - 5)
            if t_avail < 5:
                break
            lbl = labels[idx]
            print(f"  MW '{lbl}' t={t_avail:.0f}s")
            a_out, _ = mw_optimize(inits[idx], t_avail, seed=100 + idx, verbose=False)
            try_update(a_out, f"mw_{lbl}")

    # === Phase 2: Best refinement (longer run) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Refine Best ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_in = normalize(_g_best['seq'])
        print(f"  Refining n={len(a_in)}, current={_g_best['score']:.6f}")
        a_out, _ = mw_optimize(a_in, t2, seed=7777, verbose=True)
        try_update(a_out, "mw_refine")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord Descent ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 15:
        a_in = normalize(_g_best['seq'])
        a_out, _ = coord_descent_polish(a_in, t3, seed=314, verbose=True)
        try_update(a_out, "cd1")

    # === Phase 4: Multi-n search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_g_best['seq'])
    print(f"\n=== Phase 4: Multi-n (base n={n_best}) ===")
    candidate_ns = []
    for delta in [-1000, -500, 500, 1000, 1500, 2000, 2500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            candidate_ns.append(new_n)
    # Sort by closeness to base
    candidate_ns.sort(key=lambda x: abs(x - n_best))

    for new_n in candidate_ns[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}")
        a_interp = interpolate_seq(_g_best['seq'], new_n)
        try_update(a_interp, f"intp{new_n}")
        t4a = min(remaining * 0.25, 120.0)
        if t4a > 10:
            a_out, _ = mw_optimize(a_interp, t4a, seed=555 + new_n, verbose=False)
            try_update(a_out, f"mw_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            t4b = min(remaining * 0.15, 80.0)
            if t4b > 8:
                a_out, _ = coord_descent_polish(a_interp, t4b, seed=222, verbose=False)
                try_update(a_out, f"cd_n{new_n}")

    # === Phase 5: Peak equalization + final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_g_best['seq'])
    print(f"\n=== Phase 5: Peak Equalize + Final (n={n_best}) ===")
    if remaining > 20:
        t5a = min(remaining * 0.30, 150.0)
        if t5a > 10:
            a_in = normalize(_g_best['seq'])
            a_out, _ = peak_equalize(a_in, t5a, seed=999)
            try_update(a_out, "peak_eq")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t5b = min(remaining * 0.35, 200.0)
        if t5b > 10:
            a_in = normalize(_g_best['seq'])
            a_out, _ = coord_descent_polish(a_in, t5b, seed=2718, verbose=True)
            try_update(a_out, "cd_final")

    # === Phase 6: Final MW with very high beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t6 = min(remaining * 0.60, 250.0)
        if t6 > 15:
            a_in = normalize(_g_best['seq'])
            a_out, _ = mw_optimize(a_in, t6, seed=424242, verbose=True)
            try_update(a_out, "mw_final")

    # Final
    if _g_best['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _g_best['seq']])
        if np.isfinite(fs):
            _g_best['score'] = fs
        print(f"\nFINAL: score={_g_best['score']:.6f}, n={len(_g_best['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        _g_best['seq'] = make_sine(n_fb)
        _g_best['score'] = evaluate_sequence([float(v) for v in _g_best['seq']])
        print(f"Fallback: score={_g_best['score']:.6f}")

    return [float(v) for v in _g_best['seq']]
