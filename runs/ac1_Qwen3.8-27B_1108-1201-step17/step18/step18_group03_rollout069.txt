                cand_scores[~cand_mask] = 1e18
                dst = np.argmin(cand_scores)

                if dst == src or a[dst] > 0.6 * src_val:
                    continue

                for frac in [0.05, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

                if not found_check(found=False):
                    break

        if not improved:
            stall += 1
            if stall > 400:
                break
            if stall % 20 == 0:
                nmoves = int(rng.integers(3, 10))
                for _ in range(nmoves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.01, 0.05))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

    return best_a


def found_check(found):
    return found


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, base_seq):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    # Power laws
    for p in [0.2, 0.4, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
    for p in [0.5, 1.0, 2.0]:
        candidates.append(norm_sum1((n - i_arr) ** p))

    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.15, 0.25, 0.35, 0.5]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sig * sig))))

    # V-shape (high at edges)
    for p in [1.0, 1.5, 2.0, 3.0]:
        d = np.abs(x - 0.5)
        candidates.append(norm_sum1(d ** p))

    # U-shape
    for c in [0.5]:
        for w in [0.1, 0.2, 0.3]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * w * w)) * 0.1 +
                                        np.abs(x - 0.5) ** 2))

    # Bimodal
    for c1, c2 in [(0.2, 0.8), (0.25, 0.75), (0.15, 0.85)]:
        for sig in [0.08, 0.12, 0.2]:
            a = np.exp(-((x - c1) ** 2) / (2.0 * sig * sig)) + np.exp(-((x - c2) ** 2) / (2.0 * sig * sig))
            candidates.append(norm_sum1(a))

    # Exponential
    for rate in [2, 4, 8]:
        candidates.append(norm_sum1(np.exp(-rate * x)))
        candidates.append(norm_sum1(np.exp(rate * (x - 1))))

    # Blend with base
    base_norm = norm_sum1(base_seq)
    for w in [0.3, 0.5, 0.7]:
        for p in [0.5, 1.0, 2.0]:
            a_lin = norm_sum1((i_arr + 1.0) ** p)
            candidates.append(norm_sum1(w * base_norm + (1.0 - w) * a_lin))

    # Evaluate all
    best = base_seq.copy()
    best_score = eval_score(base_seq)
    for c in candidates:
        s = eval_score(c)
        if s < best_score:
            best_score = s
            best = c.copy()

    return best, best_score


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        i_arr = np.arange(n_base, dtype=np.float64)
        best_seq = norm_sum1((i_arr + 1.0) ** 0.7)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Parametric init (n={n0}, remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 5:
        t1_start = time.monotonic()
        a_param, val_param = parametric_init(n0, best_seq)
        check_and_update(a_param, "parametric")
        print(f"  Parametric done ({time.monotonic()-t1_start:.0f}s): J={val_param:.6f}")

    # === Phase 2: Main LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Main LP-FW (n={n_best}, remaining={remaining:.0f}s) ===")
    K_full = 2 * n_best - 1
    K_main = min(4000, K_full)
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out = lp_frank_wolfe(a_in, t2, K_max=K_main, verbose=True, label="main")
        check_and_update(a_out, f"lp_main_n{n_best}")

    # === Phase 3: Multi-restart with perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Multi-restart (n={n_best}, remaining={remaining:.0f}s) ===")
    rng = np.random.default_rng(2024)
    n_restarts = 5
    t_per_restart = min(remaining / (n_restarts + 1), 120.0)

    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break

        # Create perturbed starting point
        pert_type = r % 3
        if pert_type == 0:
            a_start = norm_sum1(best_seq + rng.normal(0, 0.01, n_best) * best_seq)
        elif pert_type == 1:
            i_arr = np.arange(n_best, dtype=np.float64)
            p = float(rng.uniform(0.3, 2.0))
            a_blend = norm_sum1(0.7 * best_seq + 0.3 * (((i_arr + 1.0) ** p)))
            a_start = a_blend
        else:
            x_arr = np.linspace(0, 1, n_best)
            c = float(rng.uniform(0.3, 0.7))
            sig = float(rng.uniform(0.1, 0.4))
            a_gauss = norm_sum1(np.exp(-((x_arr - c) ** 2) / (2.0 * sig * sig)))
            a_start = norm_sum1(0.8 * best_seq + 0.2 * a_gauss)

        K_r = min(K_main, 3000)
        t_r = min(t_per_restart, remaining * 0.15)
        if t_r < 20:
            continue

        print(f"  Restart {r+1}/{n_restarts} (type={pert_type}, t={t_r:.0f}s)")
        a_out = lp_frank_wolfe(a_start, t_r, K_max=K_r, verbose=False, label=f"r{r}")
        check_and_update(a_out, f"restart_{r}")

    # === Phase 4: Mirror descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Mirror descent (n={n_best}, remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out = mirror_descent(a_in, t4, top_k=80, beta=3.5, verbose=False, label="md")
        check_and_update(a_out, f"md_n{n_best}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent (n={n_best}, remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out = coordinate_descent(a_in, t5, verbose=False, label="cd")
        check_and_update(a_out, f"cd_n{n_best}")

    # === Phase 6: Final LP-FW with more constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP-FW (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 50:
        K_final = min(6000, 2 * n_best - 1)
        t6 = min(remaining * 0.35, 300.0)
        if t6 > 30:
            a_in = norm_sum1(best_seq)
            a_out = lp_frank_wolfe(a_in, t6, K_max=K_final, verbose=True, label="final")
            check_and_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (remaining={remaining:.0f}s) ===")

    length_candidates = []
    for delta in [500, -500, 1000, -1000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            length_candidates.append(new_n)

    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick LP-FW
        t7 = min(remaining * 0.3, 150.0)
        if t7 > 30:
            K_l = min(3000, 2 * new_n - 1)
            a_out = lp_frank_wolfe(a_long, t7, K_max=K_l, verbose=False, label=f"len{new_n}")
            check_and_update(a_out, f"len_{new_n}")

    # === Phase 8: Final quick polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            K8 = min(5000, 2 * n_best - 1)
            a_out = lp_frank_wolfe(a_in, t8, K_max=K8, verbose=False, label="polish")
            check_and_update(a_out, f"polish_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
