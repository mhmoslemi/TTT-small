import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient: g[i] = 2 * sum_k weights[k] * a[k-i] via FFT cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(weights, fft_size)
    # Cross-correlation: ifft(W * conj(A)) gives sum_k W[k]*A[k-m]
    g = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * g


def softmax_weights(ac, temperature=1.0, n_peaks=100):
    """Compute softmax weights over autocorrelation peaked near max."""
    top_k = min(n_peaks, len(ac))
    sorted_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[sorted_idx]
    log_w = np.log(np.maximum(top_vals, 1e-300)) / temperature
    log_w -= log_w.max()
    exp_w = np.exp(log_w)
    w = exp_w / exp_w.sum()
    return w, sorted_idx


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def multiplicative_descent(a0, time_limit, n_peaks=100, temperature0=2.0,
                           eta0=1.0, nesterov=True, restart_every=8000,
                           verbose=False, rng_seed=42):
    """Multiplicative (mirror) gradient descent with backtracking and Nesterov."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x_prev = a.copy()
    eta = eta0
    stall = 0
    cur_temp = temperature0
    restart_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if nesterov and t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            y = y / sy
        else:
            y = x.copy()

        # Compute autocorrelation
        A = np.fft.rfft(y, fft_size)
        ac = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        M_y = ac.max()

        # Compute softmax weights
        w, top_idx = softmax_weights(ac, cur_temp, n_peaks)
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w

        # Compute gradient via FFT cross-correlation
        W = np.fft.rfft(weights, fft_size)
        g = np.fft.irfft(W * np.conj(A), fft_size)[:n] * 2.0

        # Backtracking line search
        best_trial_m = M_y
        best_trial_a = y.copy()
        for _bt in range(8):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_trial = np.exp(log_a - max_log)
            a_trial = np.maximum(a_trial, 1e-30)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                a_trial = y.copy()
            else:
                a_trial = a_trial / s_t

            At = np.fft.rfft(a_trial, fft_size)
            ac_t = np.fft.irfft(At * At, fft_size)[:2 * n - 1]
            M_t = ac_t.max()

            if M_t < best_trial_m - 1e-15:
                best_trial_m = M_t
                best_trial_a = a_trial.copy()
                break
            eta *= 0.7

        x_prev = x
        x = best_trial_a
        eta = min(eta * 1.05, 5.0)  # Slightly increase if backtracking worked

        if best_trial_m < best_m - 1e-15:
            best_m = best_trial_m
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adapt temperature
        cur_temp = max(0.5, temperature0 * np.exp(-0.001 * stall))

        # Restart
        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.1, 0.02 + 0.005 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            eta = eta0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart {restart_count} ({elapsed:.0f}s): J={J:.6f}, temp={cur_temp:.3f}")

        if verbose and t % 10000 == 0:

            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, eta={eta:.3f}")

    return best_a, best_m


def smooth_objective_grad(a, n_peaks=50, temperature=0.1):
    """Compute smooth approximation of max autocorr and its gradient."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]

    # LogSumExp approximation
    top_k = min(n_peaks, size)
    sorted_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[sorted_idx]
    max_val = top_vals[0]
    log_sum = max_val + np.log(np.sum(np.exp((top_vals - max_val) / temperature)) * temperature)
    obj = log_sum  # Smooth approximation of max

    # Gradient: (1/T) * sum_k softmax_k * 2*a[k-i]
    log_w = (top_vals - max_val) / temperature
    log_w -= log_w.max()
    w = np.exp(log_w) / np.exp(log_w).sum()
    weights = np.zeros(size)
    weights[sorted_idx] = w

    W = np.fft.rfft(weights, fft_size)
    g = np.fft.irfft(W * np.conj(A), fft_size)[:n] * 2.0 / temperature

    return obj, g, ac


def scipy_smooth_optimize(a0, time_limit, n_peaks=50, temperature=0.1,
                          max_iter=5000, verbose=False, rng_seed=42):
    """Optimize smooth logsumexp approximation using custom L-BFGS-like method."""
    from scipy.optimize import minimize

    n = len(a0)
    a_init = norm_sum1(a0)
    best_a = a_init.copy()
    best_m = fast_autocorr(a_init).max()
    t0 = time.monotonic()

    # Parameterize: work on log(a) for positivity
    # Use simplex constraint via softmax reparameterization
    # t_i = log(a_i), constraint: sum(exp(t)) = 1

    def objective_and_grad(t_vec):
        if time.monotonic() - t0 > time_limit:
            return 1e10, np.zeros_like(t_vec)
        # Convert to simplex
        t_max = t_vec.max()
        a_exp = np.exp(t_vec - t_max)
        a_s = a_exp / a_exp.sum()
        obj, g, ac = smooth_objective_grad(a_s, n_peaks, temperature)
        # Chain rule: da/dt = a * (I - a*a^T) / sum(a)
        # Grad in t-space: g_t = a * (g - sum(a*g))
        g_t = a_s * (g - np.sum(a_s * g))
        return obj, g_t

    # Initial t
    t_init = np.log(np.maximum(a_init, 1e-30))
    t_init -= t_init.max()  # Center for numerical stability

    # Use L-BFGS-B on the reparameterized problem (no constraints needed)
    # But we need to ensure the mapping is correct
    result = minimize(
        fun=lambda t: objective_and_grad(t)[0],
        x0=t_init,
        jac=lambda t: objective_and_grad(t)[1],
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'maxfun': max_iter * 5, 'gtol': 1e-8}
    )

    # Convert back
    t_opt = result.x
    t_max = t_opt.max()
    a_opt = np.exp(t_opt - t_max)
    a_opt = a_opt / a_opt.sum()
    a_opt = np.maximum(a_opt, 0.0)
    a_opt = a_opt / a_opt.sum()

    M_opt = fast_autocorr(a_opt).max()
    if M_opt < best_m:
        best_m = M_opt
        best_a = a_opt.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        J_best = 2.0 * n * best_m
        J_opt = 2.0 * n * M_opt
        print(f"    Scipy smooth ({elapsed:.0f}s): best J={J_best:.6f}, opt J={J_opt:.6f}")

    return best_a, best_m


def cost_map(a, ac, n_top=15):
    """Compute cost for each position based on involvement in top peaks."""
    n = len(a)
    costs = np.zeros(n)
    top_idx = np.argsort(ac)[::-1][:n_top]
    for k in top_idx:
        ki = int(k)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair = ki - idx
        costs[idx] += a[pair]
    return costs

                a_trial = a_trial / s_t
                At = np.fft.rfft(a_trial, fft_size)
                ac_t = np.fft.irfft(At * At, fft_size)[:2 * n - 1]
                M_t = ac_t.max()
                if M_t < best_m - 1e-15:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Try random destination
                if a[src] > 1e-10:
                    for _ in range(3):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        frac = rng.uniform(0.01, 0.06)
                        amt = a[src] * frac
                        a[src] -= amt
                        a[dst] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        At = np.fft.rfft(a, fft_size)
                        ac_t = np.fft.irfft(At * At, fft_size)[:2 * n - 1]
                        M_t = ac_t.max()
                        if M_t < best_m:
                            best_m = M_t
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

        if not improved:
            stall += 1
            # Larger perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                At = np.fft.rfft(a, fft_size)
                ac_t = np.fft.irfft(At * At, fft_size)[:2 * n - 1]
                M_t = ac_t.max()
                if M_t < best_m:
                    best_m = M_t
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def simulated_annealing(a0, time_limit, verbose=False, rng_seed=42):
    """Simulated annealing with structured mass-transfer moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 1).bit_length()

    T_start = 0.05 / n
    T_end = 0.001 / n
    n_steps = 200000

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Temperature schedule
        progress = step / n_steps
        T = T_start * (T_end / T_start) ** progress

        # Choose move type
        move_type = rng.integers(3)
        a_trial = a.copy()

        if move_type == 0:
            # Transfer mass between two positions
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.001, 0.05)
            amt = a[i] * frac
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 1:
            # Transfer between adjacent-ish positions
            i = int(rng.integers(n))
            offset = int(rng.integers(1, max(2, n // 10)))
            j = (i + offset) % n
            if i == j:
                continue
            frac = rng.uniform(0.001, 0.03)
            amt = min(a[i], a[j]) * frac
            if rng.random() < 0.5:
                a_trial[i] -= amt
                a_trial[j] += amt
            else:
                a_trial[j] -= amt
                a_trial[i] += amt
        else:
            # Perturb a small window
            i = int(rng.integers(n))
            win = int(rng.integers(2, min(10, n // 4)))
            positions = [(i + d) % n for d in range(win)]
            pert = rng.normal(0, 0.01 / n, win)
            for p, pp in zip(positions, pert):
                a_trial[p] += pp

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        At = np.fft.rfft(a_trial, fft_size)
        ac_t = np.fft.irfft(At * At, fft_size)[:2 * n - 1]
        M_t = ac_t.max()

        # Metropolis acceptance
        delta = M_t - best_m
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_trial
            if M_t < best_m:
                best_m = M_t
                best_a = a_trial.copy()

        if verbose and step % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA step={step} ({elapsed:.0f}s): J={J:.6f}, T={T:.2e}")

    return best_a, best_m


def make_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_linear_ramp(n, slope=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 + slope * x
    return norm_sum1(a)


def make_optimal_shape(n, alpha=1.0):
    """Try to approximate the theoretically optimal shape."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * x + 0.1 * x * x
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_ramp(n_base, power=0.4)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Multiplicative descent with multiple initializations ===
    print(f"\n=== Phase 1: Multiplicative descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 600.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        # Various initializations
        inits.append(make_ramp(n0, power=0.3))
        labels.append("ramp_03")
        inits.append(make_ramp(n0, power=0.4))
        labels.append("ramp_04")
        inits.append(make_ramp(n0, power=0.5))
        labels.append("ramp_05")
        inits.append(make_ramp(n0, power=0.6))
        labels.append("ramp_06")
        inits.append(make_linear_ramp(n0, slope=0.2))
        labels.append("lin_02")
        inits.append(make_linear_ramp(n0, slope=0.4))
        labels.append("lin_04")
        inits.append(make_optimal_shape(n0, alpha=0.8))
        labels.append("quad_08")
        inits.append(make_optimal_shape(n0, alpha=1.2))
        labels.append("quad_12")

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  MD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = multiplicative_descent(init, t_avail, n_peaks=100,
                                               temperature0=2.0, eta0=1.0,
                                               nesterov=True, restart_every=6000,
                                               verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"md_{lbl}")

    # === Phase 2: Fine-tune best with smaller steps ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Fine-tune (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multiplicative_descent(a_in, t2, n_peaks=150,
                                           temperature0=1.0, eta0=0.5,
                                           nesterov=True, restart_every=10000,
                                           verbose=False, rng_seed=777)
        try_update(a_out, "md_fine")

    # === Phase 3: Scipy smooth optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Scipy smooth (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)

    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)

    lengths_to_try = []

        if 2000 <= new_n <= 12000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))

    for new_n in lengths_to_try[:6]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:

        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

    if remaining > 60:
        t7a = min(remaining * 0.30, 150.0)
        t7b = min(remaining * 0.30, 150.0)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:

            a_out, _ = peak_pushing(a_in, min(remaining * 0.4, 80.0),
                                     verbose=False, rng_seed=271)
            try_update(a_out, "push_final")

        remaining = time_budget - (time.monotonic() - t_global)

                                            verbose=False, rng_seed=181)

    # === Phase 8: One more aggressive round on the absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)

                                           temperature0=0.5, eta0=0.2,
                                           nesterov=True, restart_every=15000,
                                           verbose=True, rng_seed=999)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):

    else:
