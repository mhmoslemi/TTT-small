def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def coordinate_refine(a0, time_limit, rng_seed=42, verbose=True):

    t0 = time.monotonic()

        ac = fast_autocorr(a)
        # Find the worst lag
        k_star = np.argmax(ac)

        s = max(0, k_star - n + 1)
        e = min(n - 1, k_star)
        idx = np.arange(s, e + 1)
        contrib = a[idx] * a[k_star - idx]
        max_contrib_idx = idx[np.argmax(contrib)]
        
        # Find a good destination (low autocorrelation contribution)
        # Try positions in the "trough" of the sequence
        mean_val = a.mean()
        low_positions = np.where(a < 0.5 * mean_val)[0]
        low_positions = low_positions[low_positions != max_contrib_idx]
        
        if len(low_positions) == 0:
            low_positions = np.arange(n)
            low_positions = low_positions[low_positions != max_contrib_idx]

                a_trial[max_contrib_idx] -= amt
                a_trial[dst] += amt

                a_trial = a_trial / a_trial.sum()
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:

                    best_m = m_trial
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation

def interpolate_length(a, new_n):

    a_new = np.interp(new_x, old_x, a_norm)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smoothed mirror descent with adaptive temperature annealing
2. Alternating soft/hard subgradient modes
3. Nesterov-style momentum with adaptive step size
4. Targeted coordinate descent for peak shaving
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.signal import correlate as sp_correlate


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    t = 2.0 * x - 1.0
    a = 1.0 + sharpness * t ** 2
    return norm_sum1(a)


def make_u_shape4(n, a2=1.0, a4=0.3):
    x = np.linspace(0, 1, n)
    t = 2.0 * x - 1.0
    a = 1.0 + a2 * t ** 2 + a4 * t ** 4
    return norm_sum1(a)


def make_w_shape(n, depth=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 - depth * np.cos(2 * np.pi * x) ** 2
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def smoothed_mirror_descent(a0, time_limit, T_init=15.0, T_final=400.0,
                             eta_init=0.3, use_momentum=True,
                             momentum_decay=0.95, rng_seed=42, verbose=True,
                             hard_frac=0.0, hard_top_k=1):
    """
    FFT-based smoothed mirror descent (multiplicative weight) for minimizing max autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    a_prev = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    T = T_init
    eta = eta_init
    stall = 0
    iter_count = 0
    velocity = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (time_limit * 0.85))

        # Momentum extrapolation
        if use_momentum and iter_count > 0:
            beta = momentum_decay * (1.0 - 0.3 * progress)
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            else:
                y = a.copy()
        else:
            y = a.copy()

        # Compute autocorrelation
        ac = fast_autocorr(y)
        ac_max = ac.max()

        # Determine weights (soft or hard)
        use_hard = (hard_frac > 0.0 and (iter_count % 100 < int(hard_frac * 100)))
        if use_hard and hard_top_k <= 20:
            sorted_idx = np.argsort(ac)[::-1]
            top_indices = sorted_idx[:hard_top_k]
            p = np.zeros_like(ac)
            top_vals = ac[top_indices]
            w = np.exp(T * (top_vals - top_vals.max()))
            p[top_indices] = w
            p_sum = p.sum()
            if p_sum > 1e-15:
                p = p / p_sum
            else:
                p[top_indices] = 1.0 / hard_top_k
        else:
            shifted = T * (ac - ac_max)
            p = np.exp(shifted)
            p_sum = p.sum()
            if p_sum > 1e-15:
                p = p / p_sum
            else:
                p = np.ones_like(ac) / len(ac)

        # Compute gradient: g[i] = 2 * sum_m p[i+m] * y[m]
        g = 2.0 * sp_correlate(p, y, mode='valid')

        # Multiplicative (mirror) update
        log_step = -eta * g
        log_step = np.clip(log_step, -40.0, 40.0)
        a_new = y * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        m_new = fast_autocorr(a_new).max()

        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.02, 3.0)
        else:
            stall += 1
            if stall > 20:
                eta *= 0.85
            if stall > 100:
                # Perturbation restart
                noise_scale = max(0.003, 0.015 * np.exp(-stall / 200.0))
                pert = rng.normal(0, noise_scale, n) * best_a
                a_reject = best_a + pert
                a_reject = np.maximum(a_reject, 1e-10)
                a_reject = a_reject / a_reject.sum()
                m_check = fast_autocorr(a_reject).max()
                if m_check < best_m:
                    best_m = m_check
                    best_a = a_reject.copy()
                a = a_reject
                a_prev = a.copy()
                velocity = np.zeros(n)
                eta = max(eta_init * 0.3, 0.05)
                stall = 50
                iter_count += 1
                continue

        a_prev = a
        a = a_new

        # Animate temperature (exponential schedule)
        T = T_init * (T_final / T_init) ** (progress ** 0.7)

        iter_count += 1
        if verbose and iter_count % 3000 == 0:
            J = 2.0 * n * best_m
            print(f"    SMD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, T={T:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def coordinate_refine(a0, time_limit, rng_seed=42, verbose=True, top_n=30):
    """Targeted coordinate descent: move mass from peak-contributing to trough positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Focus on the top lags
        top_lags = np.argsort(ac)[::-1][:min(5, len(ac))]

        improved_this_round = False

        for k in top_lags:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Find low-value positions as destinations
            mean_val = a.mean()
            low_mask = a < 0.4 * mean_val
            low_mask[src] = False
            low_positions = np.where(low_mask)[0]
            if len(low_positions) == 0:
                low_positions = np.arange(n)
                low_positions = low_positions[low_positions != src]

            # Try moving small amounts to the best destinations
            tried = False
            for frac in [0.003, 0.005, 0.01, 0.002, 0.015]:
                if tried:
                    break
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                # Sort destinations by value (lowest first)
                dest_sort = low_positions[np.argsort(a[low_positions])][:8]
                for dst in dest_sort:
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True
                        tried = True
                        break

        if not improved_this_round:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.002, 0.008)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    return best_a, best_m

def interpolate_length(a, new_n):

    a_new = np.interp(new_x, old_x, a_norm)
    return norm_sum1(a_new)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Fine-tune (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_mirror_descent(a_in, t3, T_init=80.0, T_final=800.0,
                                            eta_init=0.08, use_momentum=False,
                                            rng_seed=300, verbose=True,
                                            hard_frac=0.5, hard_top_k=1)
        try_update(a_out, "fine_hard")

    # === Phase 4: Coordinate refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate Refine (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_refine(a_in, t4, rng_seed=500, verbose=True)
        try_update(a_out, "coord1")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length Exploration ===")

    trial_lengths = [n_best + 1200, n_best - 800, n_best + 2000]
    trial_lengths = [l for l in trial_lengths if 3000 <= l <= 8000 and l != n_best]
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_length(best_seq, new_n)
        t5 = min(remaining * 0.35, 250.0)
        if t5 > 20:
            a_out, _ = two_phase_optimize(a_interp, t5, rng_seed=new_n % 500, verbose=False)
            try_update(a_out, f"two{n}")

    # === Phase 6: Final SMD + coord on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final (n={n_best}) ===")
    if remaining > 40:
        t6a = min(remaining * 0.4, 120.0)
        if t6a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = smoothed_mirror_descent(a_in, t6a, T_init=50.0, T_final=500.0,
                                                eta_init=0.1, use_momentum=True,
                                                momentum_decay=0.95, rng_seed=800,
                                                verbose=False, hard_frac=0.4, hard_top_k=2)
            try_update(a_out, "final_smd")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.6, 100.0)
            if t6b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_refine(a_in, t6b, rng_seed=900, verbose=False)
                try_update(a_out, "final_coord")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_u_shape(n_fb, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
