def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def gs_optimize(a0, time_limit, verbose=False, rng_seed=42):
    """Gerchberg-Saxton with random phase perturbations and adaptive alpha."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Adaptive alpha: more aggressive early, gentler later
        progress = min(it / 2000.0, 1.0)
        alpha = 0.2 + 0.5 * progress
        
        # Target: partially flattened spectrum
        P_target = P_mean * (P / P_mean) ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)
        
        phase = np.angle(A)
        mag_new = np.sqrt(P_target / np.maximum(P, 1e-30))
        mag_new = np.clip(mag_new, 0.5, 2.0)
        A_new = A * mag_new
        
        # Random phase perturbation every 200 iterations
        if it % 200 == 0:
            noise_scale = 0.1 * (1.0 - progress)
            A_new *= np.exp(1j * rng.normal(0, noise_scale, len(A_new)))
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped step
            blend = 0.2 + 0.3 * progress
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()
    
    return best_a, best_m


def mirror_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Mirror descent with adaptive temperature and backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    temperature = 5.0
    eta = 0.1
    stall = 0
    restarts = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        g = compute_gradient_fft(a, ac, temperature=temperature, n_peaks=50)
        
        # Mirror descent step (multiplicative weights)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase eta on success
            eta = min(eta * 1.02, 0.5)
        else:
            stall += 1
            # Backtrack
            eta = max(eta * 0.85, 0.005)
            # Damped update
            blend = min(0.3, 0.1 + stall * 0.01)
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()
        
        # Adaptive temperature: decrease over time
        temperature = max(0.5, 5.0 * np.exp(-stall / 2000.0))
        
        # Restart if stuck
        if stall > 3000:
            restarts += 1
            strength = min(0.1, 0.02 + 0.01 * restarts)
            pert = rng.normal(0, strength, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a = a_r / a_r.sum()
            stall = 0
            eta = 0.1
            temperature = 3.0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart {restarts} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
