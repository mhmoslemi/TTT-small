def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def smooth_grad_optimize(a0, time_limit, verbose=True, tau_init=10.0, tau_final=500.0,
                          eta_init=1.0, use_away=True, rng_seed=100):
    """Smoothed gradient with away-steps and adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = eta_init
    stall = 0
    max_stall = 800
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 600.0))
        tau = tau_init * (tau_final / tau_init) ** progress
        
        ac = fast_autocorr(a)
        ac_max = ac.max()
        
        # Softmax weights for smoothing
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        # Handle numerical issues
        if not np.isfinite(exp_ac).all():
            exp_ac = np.exp(np.minimum(ac_shifted, 50))
        p = exp_ac / exp_ac.sum()
        
        # Compute gradient
        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        
        # Tangent projection
        g_tangent = g - g.mean()
        
        # Mirror descent step (multiplicative)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * g_tangent / a
        max_log = max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Away step: if we're at a vertex, move away
        if use_away and total_iters % 5 == 0:
            # Find the component closest to 0 (most "vertex-like")
            min_idx = np.argmin(a)
            if a[min_idx] < 1e-6:
                # Move away by increasing this component
                away = np.zeros(n)
                away[min_idx] = 1.0
                a_new = 0.95 * a_new + 0.05 * away
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 3.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.8
            if stall > 300:
                # Perturbation
                pert = rng.normal(0, 0.003, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-15)
                a = a_p / a_p.sum()
                stall = 150
            else:
                a = a_new
        
        total_iters += 1
        
        if verbose and total_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}")
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Enhanced LP Frank-Wolfe with adaptive K, M-reduction, and backtracking line search
2. Smoothed mirror descent with away-steps and adaptive temperature
3. Multi-start with diverse initializations
4. Length exploration with interpolation
5. Targeted peak-reduction coordinate descent
"""

import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate as sp_correlate

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    return norm_sum1(a)


def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_w_shape(n, peaks=3, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.01
    for i in range(peaks):
        center = (i + 1) / (peaks + 1)
        a += np.exp(-((x - center) / width) ** 2)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe_enhanced(a0, time_limit, solver, verbose=True,
                             top_K_init=200, top_K_max=None,
                             m_reduction=0.0, k_growth=40,
                             stall_limit=40):
    """
    Enhanced LP Frank-Wolfe with adaptive K, M-reduction, and backtracking.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    top_K_max = min(top_K_max, 2 * n - 1)
    K = top_K_init
    iter_count = 0
    stall_count = 0
    lp_solve_times = []

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Adaptive K: grow over time with square-root schedule
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        K_target = int(top_K_init + (top_K_max - top_K_init) * (progress ** 0.4))
        K = min(K_target, top_K_max, K + k_growth)

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix T_sel
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        # Try with M-reduction for tighter direction
        M_use = M_curr * (1.0 - m_reduction)
        lp_t0 = time.monotonic()

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_use]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 10, top_K_max)
            continue

        lp_time = time.monotonic() - lp_t0
        lp_solve_times.append(lp_time)
        # Keep rolling average of last 10 LP times
        avg_lp_time = np.mean(lp_solve_times[-10:]) if len(lp_solve_times) >= 3 else lp_time

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # If infeasible with reduced M, retry with full M
            if m_reduction > 0:
                constraints = [T_sel @ b_var <= M_curr]
                prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
                try:
                    prob.solve(solver=solver, verbose=False)
                except Exception:
                    continue
                if prob.status not in ('optimal', 'optimal_inaccurate'):
                    continue
            else:
                continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search with multiple candidates
        base_obj = M_curr
        candidates_alpha = [2.0 / (iter_count + 2.0)]
        # Add diverse alpha values
        for ca in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
            if ca not in candidates_alpha:
                candidates_alpha.append(ca)

        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        best_trial_a = None

        for alpha in sorted(candidates_alpha, reverse=True):
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m - 1e-15:
                best_trial_m = m_trial
                best_alpha = alpha
                best_trial_a = a_trial.copy()

        if best_trial_a is not None:
            a = best_trial_a
        else:
            a = (1.0 - best_alpha) * a + best_alpha * g_dir
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a < 1e-15:
                break
            a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= stall_limit:
                # Smart restart: perturbation guided by autocorrelation structure
                rng = np.random.default_rng(iter_count * 31 + 7)
                # Scale perturbation by local structure
                pert_scale = 0.01 + 0.02 * (stall_count / stall_limit)
                pert = rng.normal(0, pert_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-15)
                a_r = a_r / a_r.sum()
                a = a_r
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                stall_count = 0
                # Reset K to allow regrowth
                K = max(int(K * 0.7), top_K_init)

        # If LP is taking too long, reduce K
        if avg_lp_time > 3.0 and K > top_K_init:
            K = max(int(K * 0.8), top_K_init)

        iter_count += 1

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW+ iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, "
                  f"alpha={best_alpha:.3f}, lp_t={avg_lp_time:.2f}s")

    return best_a, best_m


def smooth_mirror_descent(a0, time_limit, verbose=True, tau_init=10.0, tau_final=500.0,
                           eta_init=1.0, use_away=True, rng_seed=100):
    """Smoothed multiplicative mirror descent with away-steps."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    max_stall = 600
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break
        progress = min(1.0, elapsed / min(time_limit, 600.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()

        # Softmax weights
        ac_shifted = tau * (ac - ac_max)
        ac_shifted = np.minimum(ac_shifted, 50.0)  # Numerical safety
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        # Compute gradient of smoothed objective
        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross

        # Tangent projection (for simplex constraint)
        g_tangent = g - g.mean()

        # Multiplicative (mirror) update
        ratio = np.exp(-eta * g_tangent / np.maximum(a, 1e-20))
        ratio = np.clip(ratio, 1e-10, 1e10)
        a_new = a * ratio
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Away step every few iterations
        if use_away and total_iters % 7 == 3:
            min_idx = np.argmin(a)
            if a[min_idx] < 1e-5:
                # Push mass to the minimum component
                amt = a.max() * 0.01
                a_new[min_idx] += amt
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 5.0)
        else:
            stall += 1
            if stall > 80:
                eta *= 0.85
            if stall > 250:
                # Perturbation restart
                pert = rng.normal(0, 0.005, n) * best_a
                a_p = np.maximum(best_a + pert, 1e-15)
                a = a_p / a_p.sum()
                m_check = fast_autocorr(a).max()
                if m_check < best_m:
                    best_m = m_check
                    best_a = a.copy()
                stall = 100
                eta = eta_init * 0.5
            else:
                a = a_new

        total_iters += 1

        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MirrorDesc iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.3f}")

    return best_a, best_m


def peak_reduction_cd(a0, time_limit, verbose=True, rng_seed=456):
    """Targeted peak-reduction coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 50
    n_top_peaks = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top_peaks]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit - 5:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            pair_idx = ki - idx
            contrib = a[idx] * a[pair_idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue

            # Find the pair that contributes most
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-contribution destinations
            in_peak = np.zeros(n, dtype=bool)
            for kk in top_k[:8]:
                kki = int(kk)
                ps = max(0, kki - n + 1)
                pe = min(n - 1, kki)
                in_peak[ps:pe+1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Sort candidates by their contribution to peak lags
            cand_contrib = np.zeros(len(candidates))
            for ci, c_idx in enumerate(candidates):
                for kk in top_k[:5]:
                    kki = int(kk)
                    if abs(kki - c_idx) < n:
                        partner = c_idx - kki if kki <= c_idx else c_idx + kki
                        if 0 <= c_idx < n and 0 <= (kki - c_idx) < n:
                            cand_contrib[ci] += a[c_idx] * a.get((ggi - c_idx), 0) if False else 0
                # Simpler: just use a[c_idx] as proxy
            cand_contrib = a[candidates]

            # Try moving to low-value destinations
            top_cands = candidates[np.argsort(cand_contrib)[:5]]

            for dst in top_cands:
                if dst == src or improved:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining enhanced LP Frank-Wolfe,
    smoothed mirror descent, peak-reduction CD, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            a_list = [float(x) for x in a_norm]
            sc = evaluate_sequence(a_list)
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            try:
                p_list = [float(x) for x in parent]
                p_score = evaluate_sequence(p_list)
            except Exception:
                p_score = float('inf')
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = make_u_shape(n_base, 2.0)
        try:
            best_score = evaluate_sequence([float(x) for x in best_seq])
        except Exception:
            best_score = float('inf')
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick diverse initialization exploration ===
    print(f"\n=== Phase 1: Multi-init exploration (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.06, 100.0)

    if t_phase1 > 15:
        inits = [norm_sum1(best_seq)]
        labels = ["best"]
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")
        inits.append(make_bimodal(n0, 0.2))
        labels.append("bimodal")
        inits.append(make_w_shape(n0, 4, 0.08))
        labels.append("wshape")
        inits.append(spectral_init(n0, 40, 42))
        labels.append("spectral")

        t_per = max(5.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 5:
                break
            a_out, _ = smooth_mirror_descent(init, t_avail, verbose=False,
                                              tau_init=10.0, tau_final=100.0,
                                              eta_init=1.0, rng_seed=42 + idx)
            try_update(a_out, f"init_{lbl}")

    # === Phase 2: Main LP-FW with growing K ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    t2 = min(remaining * 0.40, 700.0)

                                             m_reduction=0.0, k_growth=50, stall_limit=35)
        try_update(a_out, "lp_main")

    # === Phase 3: LP-FW with M-reduction (tighter constraints) ===

        try_update(a_out, "peakcd")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    trial_lengths = []
    for delta in [500, 1000, -500, 1500, -1000, 2000, -1500]:
        new_n = n_best + delta
        if 1500 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:

        print(f"  Trying n={new_n}...")

                                              tau_init=10.0, tau_final=200.0,
                                              eta_init=1.0, rng_seed=new_n % 1000)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 6000:

    if remaining > 30 and n_best <= 6000:

        if t7 > 20:
            a_in = norm_sum1(best_seq)
            top_K_init = min(800, 2 * n_best - 1)
            top_K_max = min(2 * n_best - 1, 6000)
            a_out, _ = lp_frank_wolfe_enhanced(a_in, t7, solver, verbose=True,
                                                 top_K_init=top_K_init, top_K_max=top_K_max,
                                                 m_reduction=0.01, k_growth=30, stall_limit=25)
            try_update(a_out, "final_lp")

    # === Phase 8: Final mirror descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Final mirror polish (n={n_best}) ===")
        t8 = min(remaining * 0.5, 100.0)
