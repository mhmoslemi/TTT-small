```python
from scipy.sparse import csr_matrix, diags, banded_matrix
from scipy.optimize import linprog
import numpy as np

def lp_step(f, M, max_iter=1, verbose=False):
    """One LP step: find g that maximizes sum(g) subject to g>=0 and (f*g)<=M."""
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A (num_constraints x n)
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: g_i >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=np.full(num_constraints, M), bounds=bounds, method='highs')
    
    if res.status != 0:
        return f  # Return original if LP fails
    
    return res.x
```

```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]
```

```python
def lp_step(f, M, rows, cols, verbose=False):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    num_rows = 2 * n - 1
    
    # Build constraint values: A[k,j] = f[k-j]
    vals = f[rows.astype(np.int64) - cols.astype(np.int64)]
    
    # Check for any zero or negative values (shouldn't happen if f >= 0)
    if np.any(vals < -1e-15):
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    
    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(num_rows, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0:
        if verbose:
            print(f"  LP status: {res.status}, message: {res.message}")
        return None
    
    return res.x

def smoothed_gradient(a, T):
    """Compute gradient of 2n*T*log(sum(exp(c/T))) w.r.t. log-space params theta."""
    n = len(a)
    c = fast_conv(a)
    
    # Softmax weights
    c_norm = c / T
    c_norm = c_norm - c_norm.max()
    w = np.exp(c_norm)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum
    
    # Compute corr[j] = sum_k w[k] * a[k-j] using FFT
    fft_size = 1 << (2 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    
    # Weighted average of autocorrelation
    w_avg = np.dot(w, c)
    
    # Gradient in log-space: dJ/dtheta_j = 2n * 2 * a_j * (corr[j] - w_avg)
    # = 4n * a_j * (corr[j] - w_avg)

    T_min = best_score / (2 * n) * 0.1  # final temperature (sharper)

    lr_min = 0.001

        if time.monotonic() - t0 > time_limit:

        T = T_init * (T_min / T_init) ** progress
        lr = lr_init * (lr_min / lr_init) ** progress

        grad, M = smoothed_gradient(a, T)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Update in log-space
        log_a = np.log(np.maximum(a, 1e-30))

def spectral_step(a, strength=0.3):
    """Compress dynamic range of |FFT(a)|."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.power(ratio, 1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
```

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""

    if s < 1e-15:
