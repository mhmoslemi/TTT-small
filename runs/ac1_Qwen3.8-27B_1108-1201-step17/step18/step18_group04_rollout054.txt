def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad_v2(a, conv, top_k=40, temperature=0.002, sharp=2.0):
    """Improved subgradient with sharper peak focus."""
    n = len(a)
    conv_max = conv.max()
    # Use sharper threshold
    threshold = conv_max - temperature * 6.0
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    # Sharper softmax weighting
    w = np.exp((conv[active] - conv_max) / (temperature / sharp))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad


def peak_directed_perturb(a, conv, rng, strength=0.03):

def alternating_step(a, M, n_iters=50):
    """
    Approximate the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0
    Using iterative water-filling heuristic.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    max_a = a.max()
    if max_a < 1e-15:
        return np.ones(n) / n
    # Initialize b
    b = np.minimum(M / (max_a * 2.0), 1.0)
    for _ in range(n_iters):
        conv_ab = fast_conv(a, b) if hasattr(np, 'convolve') else None
        # Compute conv(a, b)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        B = np.fft.rfft(b, fft_size)
        conv_ab = np.fft.irfft(A * B, fft_size)[:size]
        # Find violations
        violations = np.maximum(0, conv_ab - M)
        if violations.max() < 1e-12:
            break
        # For each position j, find the "tightest" constraint
        # b[j] contributes a[k-j]*b[j] to conv[k] for k in [j, j+n-1]
        # Reduce b where it causes most violation
        # Simple approach: proportional reduction
        for j in range(n):
            if b[j] < 1e-15:
                continue
            k_min = max(0, j)
            k_max = min(2 * n - 2, j + n - 1)
            ks = np.arange(k_min, k_max + 1)
            contribs = a[ks - j] * b[j]
            local_viol = np.maximum(0, conv_ab[ks] - M)
            if local_viol.max() > 0:
                # Scale down b[j] proportionally
                worst = local_viol.max()
                max_contrib = contribs.max()
                if max_contrib > 1e-15:
                    scale = max(0, 1.0 - worst / (max_contrib + 1e-15))
                    b[j] = max(0, b[j] * scale)
    return norm_sum1(b)

def interpolate_fourier(a, new_n):
    """Fourier-based interpolation (preserves spectral content better)."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    # Use FFT-based interpolation
    fft_size = 1 << (2 * max(old_n, new_n) - 1).bit_length()
    A = np.fft.rfft(a_norm, fft_size)
    # Zero out high frequencies for smoothness
    n_keep = min(len(A), fft_size // 2)
    A_interp = np.fft.irfft(A, fft_size)
    A_interp = np.maximum(A_interp, 0)
    s = A_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    # Resample
    old_x = np.linspace(0, 1, fft_size)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, A_interp)
    result = np.maximum(result, 0)
    s2 = result.sum()
    if s2 < 1e-15:
        return np.ones(new_n) / new_n
    return result / s2


def interpolate_linear(a, new_n):
    """Simple linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_increasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((x + 1) ** power)


def make_mixed(n, pos1=0.2, pos2=0.65, width=0.1, ratio=1.0):
    x = np.arange(n) / n
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / (width * 1.5)) ** 2)
    return norm_sum1(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_centered_power(n, power=0.5):
    x_c = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm_sum1(x_c ** (-power))


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_exponential_decay(n, rate=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm_sum1(np.exp(-rate * x))


def make_bell(n, width=0.3, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return norm_sum1(a)


def make_power_bell(n, power=2.0, width=0.4, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.maximum(0, 1 - ((x - center) / width) ** 2) ** power
    return norm_sum1(a)


def try_update(a, label):
    """Track best sequence."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def fine_grained_local_search(a0, time_limit, seed=42):
    """Very fine-grained local search for the final polish phase."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    conv = fast_conv(a)
    peak_k = int(np.argmax(conv))

    while time.monotonic() - t0 < time_limit and stall < 500:
        conv = fast_conv(a)
        M = conv.max()
        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()
            peak_k = int(np.argmax(conv))
            stall = 0
        else:
            stall += 1

        # Focus on top 3 peaks
        top_peaks = np.argsort(conv)[::-1][:3]
        # Compute cost vector
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        # Find the peak's top contributors
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]
        top_src = idx[np.argsort(contrib)[::-1][:10]]

        improved = False
        for src in top_src:
            src_i = int(src)
            if a[src_i] < 1e-13:
                continue
            # Try moving small amounts to low-cost positions
            low_cost = np.argsort(costs)[:20]
            for dst_i in low_cost:
                dst_i = int(dst_i)
                if dst_i == src_i:
                    continue
                for delta_frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = a[src_i] * delta_frac
                    if a[src_i] - amt < 0:
                        continue
                    a_try = a.copy()
                    a_try[src_i] -= amt
                    a_try[dst_i] += amt
                    M_try = fast_conv(a_try).max()
                    if M_try < best_m - 1e-16:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and stall > 50:
            # Micro-perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.02)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 4500
        x = np.linspace(0, 1, n_base)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Default init: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    n0 = len(_best_tracker['seq'])
    print(f"Starting: n={n0}, score={_best_tracker['score']:.6f}")

    # Phase 1: AMD on parent (focused, long)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD on parent (n={n0}) ===")
    t_phase1 = min(remaining * 0.35, 550.0)
    if t_phase1 > 30:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = amd_optimize_v2(a_in, t_phase1,
                                   top_k_init=30, temp_init=0.002, c0_init=2.5,
                                   restart_every=4000, seed=42)
        try_update(a_out, "amd_parent")

    # Phase 2: Diverse initializations with short AMD
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Diverse inits ===")
    t_phase2 = min(remaining * 0.20, 300.0)
    if t_phase2 > 30:
        inits = []
        labels = []
        inits.append(make_decreasing(n0, 0.3)); labels.append("dec03")
        inits.append(make_decreasing(n0, 0.6)); labels.append("dec06")
        inits.append(make_increasing(n0, 0.3)); labels.append("inc03")
        inits.append(make_mixed(n0, 0.15, 0.6, 0.07)); labels.append("mixA")
        inits.append(make_mixed(n0, 0.1, 0.7, 0.1, 0.9)); labels.append("mixB")
        inits.append(make_sinc(n0, 5.0)); labels.append("sinc5")
        inits.append(make_sinc(n0, 8.0)); labels.append("sinc8")
        inits.append(make_cosine(n0, 1)); labels.append("cos1")
        inits.append(make_centered_power(n0, 0.7)); labels.append("cctr07")
        inits.append(make_exponential_decay(n0, 1.0)); labels.append("exp10")
        inits.append(make_bell(n0, 0.25, 0.4)); labels.append("bell25")
        inits.append(make_bell(n0, 0.35, 0.45)); labels.append("bell35")
        inits.append(make_power_bell(n0, 3.0, 0.35, 0.45)); labels.append("pbell")
        inits.append(make_random_smooth(n0, seed=7, n_freq=5)); labels.append("rnd7")
        inits.append(make_random_smooth(n0, seed=13, n_freq=8)); labels.append("rnd13")

        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(5.0, t_phase2 / n_inits)
        for idx in range(n_inits):
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 4:
                break
            lbl = unique_labels[idx]
            a_out, _ = amd_optimize_v2(unique_inits[idx], t_avail,
                                       top_k_init=25, temp_init=0.003, c0_init=3.0,
                                       restart_every=3000, seed=200 + idx)
            try_update(a_out, f"amd_{lbl}")

    # Phase 3: Coordinate descent on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord Descent ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 10:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = coord_descent_v2(a_in, t3, seed=42)
        try_update(a_out, "cd1")

    # Phase 4: AMD refinement on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: AMD Refine ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = amd_optimize_v2(a_in, t4, top_k_init=40, temp_init=0.0015,
                                   c0_init=2.0, restart_every=6000, seed=999)
        try_update(a_out, "amd_refine")

    # Phase 5: Multi-n search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 5: Multi-n (base n={n_best}) ===")
    candidate_ns = sorted(set([n_best + d for d in [-800, -400, -200, 200, 400, 800, 1200, -1200]
                               if 2000 <= n_best + d <= 8000]))

    for new_n in candidate_ns[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}")
        a_interp = interpolate_linear(_best_tracker['seq'], new_n)
        try_update(a_interp, f"intp_lin{new_n}")
        a_interp_f = interpolate_fourier(_best_tracker['seq'], new_n)
        try_update(a_interp_f, f"intp_fft{new_n}")

        # Optimize the best interpolation
        a_start = a_interp if fast_conv(a_interp).max() <= fast_conv(a_interp_f).max() else a_interp_f
        t5a = min(remaining * 0.25, 120.0)
        if t5a > 10:
            a_out, _ = amd_optimize_v2(a_start, t5a, top_k_init=30,
                                       temp_init=0.0025, c0_init=2.5,
                                       restart_every=5000, seed=777)
            try_update(a_out, f"amd_n{new_n}")

    # Phase 6: Fine-grained local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 6: Fine Local Search (n={n_best}) ===")
    if remaining > 40:
        t6 = min(remaining * 0.35, 300.0)
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = fine_grained_local_search(a_in, t6, seed=2024)
        try_update(a_out, "fine_local")

    # Phase 7: Final AMD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 7: Final Polish (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.60, 250.0)
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = amd_optimize_v2(a_in, t7, top_k_init=50,
                                   temp_init=0.001, c0_init=1.8,
                                   restart_every=8000, seed=31415)
        try_update(a_out, "final_amd")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7b = min(remaining * 0.70, 150.0)
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = fine_grained_local_search(a_in, t7b, seed=99999)
        try_update(a_out, "final_fine")

    # Final evaluation
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, "
              f"t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
