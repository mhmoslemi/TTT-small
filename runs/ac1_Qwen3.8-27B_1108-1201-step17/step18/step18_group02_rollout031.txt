```python
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver: ECOS (fallback)")
    return cp.ECOS


def build_coeff_matrix(constraint_list, n, y):
    """Build the K x n coefficient matrix for the LP constraints."""
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_frank_wolfe_pairwise(a0, time_limit, solver, initial_K=300, max_K=2500,
                             verbose=True):
    """
    Pairwise LP Frank-Wolfe with away-steps and adaptive step size.
    """
    n = len(a0)
    y = norm_sum1(a0).copy()
    x = y.copy()  # Best iterate (averaged)
    ac = fast_autocorr(y)
    best_a = y.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    
    # Initialize constraint set with top lags
    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            # Keep tightest constraints + some margin
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build LP
        coeffs_matrix = build_coeff_matrix(constraint_list, n, y)
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # On failure, reduce constraints
            constraint_set = set(int(constraint_list[i]) for i in range(min(initial_K, len(constraint_list))))
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if len(constraint_list) > initial_K:
                constraint_set = set(int(constraint_list[i]) for i in range(initial_K))
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Also periodically add top lags from current iteration
        if iter_count % 10 == 0:
            top_lags = np.argsort(ac_y)[::-1][:50]
            for k in top_lags:
                constraint_set.add(int(k))
        
        # ---- Pairwise FW step ----
        # Standard direction
        d_std = g_dir - y
        
        # Away direction: move away from the vertex with max component
        s_idx = int(np.argmax(y))
        e_s = np.zeros(n)
        e_s[s_idx] = 1.0
        d_away = g_dir - e_s
        
        # Check if away step is feasible (y[s_idx] > 0) and reduces gap
        gap_std = np.dot(y, d_std)
        gap_away = np.dot(y, d_away)
        
        use_away = (y[s_idx] > 1e-6 and gap_away < gap_std - 1e-12)
        
        if use_away:
            direction = d_away
        else:
            direction = d_std
        
        # Adaptive step size: try a few values
        best_alpha = 2.0 / (iter_count + 2.0)
        best_obj = M_curr
        
        for alpha in [1.0, 0.5, 0.3, 0.2, best_alpha, 0.1, 0.05]:
            if use_away and y[s_idx] < alpha * (1.0 + 1e-9):
                continue  # Would make y[s_idx] negative
            y_trial = y + alpha * direction
            if y_trial.min() < -1e-12:
                continue
            y_trial = np.maximum(y_trial, 0.0)
            st = y_trial.sum()
            if st < 1e-15:
                continue
            y_trial = y_trial / st
            ac_trial = fast_autocorr(y_trial)
            M_trial = ac_trial.max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_alpha = alpha
        
        y_new = y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Update best iterate (running average for FW convergence)
        beta_t = (iter_count + 1.0) / (iter_count + 3.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        # Check x as well
        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPFW-pairwise iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LPFW-pairwise done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m
```

Now let me think about the Newton polish. The idea is:

Given a near-optimal solution a*, identify the active set S = {k : conv_k(a*) > M* - eps}. Then solve a local optimization problem that only enforces the constraints for k in S and the normalization.

The subproblem: min M s.t. a^T A_k a <= M for k in S, sum(a) = 1, a >= 0

This is still a QP, but with only |S| quadratic constraints (typically |S| is small, like 5-20). We can solve this with a penalty method or direct QP.

Actually, a simpler approach: use the KKT conditions directly.

At the optimum, for the active constraints k in S:
  2 * (A_k a) + nu * 1 = lambda (componentwise, where lambda[i] >= 0 for i in support, = 0 otherwise)

And nu is the multiplier for sum(a) = 1.

If we know S (say m constraints), we have:
  2 * sum_{k in S} lambda_k * (A_k a)[i] + nu = 0 for i in support
  a[i] = 0 for i not in support

This gives us a system of equations. But it's nonlinear (a appears both inside A_k a and as the multiplier depends on a).

A practical Newton approach:
1. Start from a* (LP-FW result)
2. Identify active set S
3. Solve the system:
   - conv_k(a) = M for k in S (m equations)
   - sum(a) = 1 (1 equation)
   - KKT: grad_M(a) is in the span of {1} ∪ {grad_{conv_k}(a) : k in S} (n - |support| equations)
   
   Total: m + 1 + (n - |support|) equations in n + m + 1 unknowns (a, M, lambda_k, nu)
   
   This is typically solvable with Newton's method.

But this is complex to implement robustly. Let me use a simpler approach:

**Penalty method on the active constraints:**
min_a M + mu * sum_{k not in S} max(0, conv_k(a) - M)^2
s.t. conv_k(a) <= M for k in S (soft constraint)
sum(a) = 1, a >= 0

Or even simpler: just do gradient descent on the smoothed objective with a very small temperature, using the exact gradient.

The gradient of max_k conv_k(a) is:
  grad = (1/|S|) * sum_{k in S} grad conv_k(a)  (subgradient, averaged over active set)

where grad conv_k(a)[j] = a[k-j] + a[j-k] (with appropriate indexing)

Let me implement a "subgradient Newton" method:

```python
def subgrad_newton_polish(a0, time_limit, verbose=True):
    """
    Polish using subgradient of max-conv with adaptive step size.
    Uses the active set to compute the subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    tau = best_m * 0.01  # Smoothing parameter
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
        
        # Compute subgradient: average over active lags
        threshold = M - tau
        active = ac > threshold
        if not active.any():
            active[np.argmax(ac)] = True
        
        # Gradient of conv_k w.r.t. a[j]:
        # conv_k = sum_i a[i] * a[k-i]
        # d(conv_k)/da[j] = a[k-j] + a[j-k] (if indices valid and j != k-j)
        #                    2*a[k-j] (if 2j = k, i.e., j = k/2)
        
        # For efficiency, compute using FFT
        # grad[j] = sum_{k in active} d(conv_k)/da[j]
        # = sum_{k in active} (a[k-j] + a[j-k]) [with validity checks]
        
        # This is: sum_{k in active} a[k-j] + sum_{k in active} a[j-k]
        # = (active_flip * a)[j] + (active * a)[j]
        # where active_flip[k] = active[n-1+k - j]... hmm, this is getting complicated.
        
        # Let me think differently. conv_k(a) = (a * a)[k] where * is convolution.
        # d(conv_k)/da[j] = a * e^[k-j] + e^[j] * a... 
        
        # Actually, let's use the property that:
        # conv(a, a)[k] = sum_i a[i] a[k-i]
        # d/d a[j] of this = a[k-j] + a[j-k] (with the convention that a[x] = 0 for x outside [0,n-1])
        # Wait no: d/d a[j] sum_i a[i] a[k-i]
        # = sum_i [delta(i,j) a[k-i] + a[i] delta(k-i, j)]
        # = a[k-j] + a[k-j] (if k-j != j, i.e., the two terms are the same index)
        # Hmm no. Let me be very careful.
        # sum_i a[i] a[k-i]. The term a[j] appears when:
        #   i = j: gives a[k-j]
        #   k-i = j, i.e., i = k-j: gives a[k-j] (if k-j is a valid index for i, i.e., 0 <= k-j <= n-1)
        # So d/d a[j] = a[k-j] * [0<=k-j<n] + a[k-j] * [0<=k-j<n] 
        # = 2*a[k-j] * [0<=k-j<n]  if j != k-j (i.e., 2j != k)
        # = a[k-j] * [0<=k-j<n] + a[k-j] * [0<=k-j<n] = 2*a[k-j] * [0<=k-j<n] if j = k-j
        # Wait, that's the same! Let me redo.
        # If 2j = k: the term i=j gives a[j], and the term i=k-j=j also gives a[j]. So it's a[j]*a[j] = a[j]^2, and derivative is 2*a[j].
        # If 2j != k: the term i=j gives a[k-j], and the term i=k-j gives a[j]. So d/d a[j] = a[k-j] + a[j].
        # Wait no! d/d a[j] of a[i]*a[k-i] for fixed i:
        #   If i = j: d/d a[j] (a[j] * a[k-j]) = a[k-j] (treating a[k-j] as constant w.r.t. a[j] if k-j != j)
        #     or = 2*a[j] if k-j = j
        #   If k-i = j, i.e., i = k-j: d/d a[j] (a[k-j] * a[j]) = a[k-j] (if k-j != j)
        #     or this is the same term as above if k-j = j
        # So: d/d a[j] of conv_k = 
        #   If 2j != k and both j and k-j are in [0,n-1]: a[k-j] + a[j]
        #   If 2j = k and j in [0,n-1]: 2*a[j]
        #   If only j in [0,n-1] but k-j not: a[k-j]... no, a is 0 outside [0,n-1]
        
        # OK so in general (with a[x] = 0 for x outside [0,n-1]):
        # d(conv_k)/da[j] = a[k-j] + a[j-k]  (with the understanding that a[x]=0 for x outside [0,n-1])
        # But if k-j = j (i.e., 2j=k), this gives 2*a[j], which is correct.
        # If k-j != j, this gives a[k-j] + a[j-k]. But a[j-k] = 0 unless j-k >= 0, i.e., j >= k.
        # And a[k-j] = 0 unless 0 <= k-j <= n-1.
        
        # So: d(conv_k)/da[j] = a[k-j] + a[j-k] (with a[x] = 0 outside [0,n-1])
        
        # Now, grad_obj[j] = sum_{k in active} d(conv_k)/da[j]
        # = sum_{k in active} a[k-j] + sum_{k in active} a[j-k]
        # = (active_reversed_conv_with_a)[j] + (active_conv_with_a)[j]
        # Hmm, let me define:
        # (sum_k active[k] * a[k-j]) = (active * a)[j] where active is treated as a sequence and * is cross-correlation
        # Actually: sum_k active[k] * a[k-j] is the cross-correlation of active and a at lag j.
        
        # Let me just compute it directly using FFT.
        # grad[j] = sum_{k in active} (a[k-j] + a[j-k])
        # Let b = active (as a sequence of length 2n-1)
        # sum_k b[k] * a[k-j]: this is the correlation of b and a
        # = (b * flip(a))[j + n - 1] in convolution notation
        # Or: sum_k b[k] * a[k-j] = sum_k b[k+j] * a[k] (substituting k' = k-j)
        # = (flip(b) * a)[j + offset]... 
        
        # Let me just do it numerically for clarity, then optimize if needed.
        # For n up to 5000 and |active| up to maybe 20, direct computation is fine.
        
        grad = np.zeros(n)
        active_lags = np.where(active)[0]
        for k in active_lags:
            k_i = int(k)
            # For each j, d(conv_k)/da[j] = a[k-j] + a[j-k] (with bounds)
            for j in range(n):
                kj = k_i - j
                jk = j - k_i
                val = 0.0
                if 0 <= kj < n:
                    val += a[kj]
                if 0 <= jk < n:
                    val += a[jk]
                grad[j] += val
        
        # This is O(n * |active|) per iteration, which for n=5000, |active|=10 is 50000 ops. Fine.
        
        # Actually, let me vectorize this.
        # grad[j] = sum_{k in active} (a[k-j] + a[j-k])
        # Let's define the contribution from the first term:
        # f1[j] = sum_{k in active} a[k-j] (with a[x]=0 outside [0,n-1])
        # f1[j] = (active_conv_a)[j] where active_conv_a is the "valid" cross-correlation
        # Using FFT: f1 = ifft(fft(active) * conj(fft(a_padded)))... 
        
        # Actually, let me use a different approach:
        # sum_k active[k] * a[k-j] for k in [0, 2n-2], j in [0, n-1]
        # = sum_k active[k] * a[k-j]
        # Let me define active_padded = active (length 2n-1)
        # and a_padded = a (length n)
        # The cross-correlation: sum_k active[k] * a[k-j] is the "anti-correlation"
        # = (flip(active) * a)[j + (2n-2)] in full convolution... 
        
        # I'll just use the direct computation for now since |active| is small.
        # But let me vectorize the inner loop:
        
        # Actually, let me redo the gradient computation more carefully using FFT.
        # 
        # conv(a,a)[k] = sum_i a[i] a[k-i] for k = 0, ..., 2n-2
        # 
        # The gradient of max_k conv(a,a)[k] w.r.t. a[j] (for j in the active set of argmax):
        # For a specific k: d conv(a,a)[k] / da[j] = a[k-j] + a[j-k] (with zero extension)
        # 
        # Sum over active k:
        # grad[j] = sum_{k in active} a[k-j] + sum_{k in active} a[j-k]
        # 
        # First term: sum_{k in active} a[k-j]
        # Let c = active (length 2n-1). Then:
        # sum_k c[k] a[k-j] = (c cross-correlate a)[j] in some sense
        # = sum_k c[k] a[k-j] 
        # If we define r[k] = c[k] (length 2n-1) and s[k] = a[k] (length n), then:
        # sum_k r[k] s[k-j] = (r * flip(s))[j + n - 1]... 
        # 
        # Let me just use: Fourier transform approach
        # DFT{a}(f) A(f), DFT{c}(f) C(f)
        # sum_k c[k] a[k-j] = IDFT{C(f) * conj(A(f))}[j]... no that's not right either.
        # 
        # OK I'll just use the direct computation. For n=5000 and |active| <= 20, it's fast enough.
        
        # Let me redo with vectorization:
        grad = np.zeros(n)
        for k in active_lags:
            k_i = int(k)
            # Contribution to grad from this k:
            # da[k-j] for j = 0..n-1: this is a[k], a[k-1], ..., a[k-(n-1)] (with zero extension)
            # = a[max(0,k_i+1-n) : min(n, k_i+1)][::-1] zero-padded
            # da[j-k] for j = 0..n-1: this is a[-k], a[1-k], ..., a[n-1-k] (with zero extension)
            # = nonzero for j >= k_i: a[j-k_i] for j = k_i..n-1
            
            # First term: a[k_i - j] for j in 0..n-1
            # Valid when 0 <= k_i - j < n, i.e., k_i - n + 1 <= j <= k_i
            j_start = max(0, k_i - n + 1)
            j_end = min(n - 1, k_i)
            if j_start <= j_end:
                grad[j_start:j_end+1] += a[k_i - np.arange(j_start, j_end+1)]
            
            # Second term: a[j - k_i] for j in 0..n-1
            # Valid when 0 <= j - k_i < n, i.e., k_i <= j <= k_i + n - 1
            j_start2 = max(0, k_i)
            j_end2 = min(n - 1, k_i + n - 1)
            if j_start2 <= j_end2:
                grad[j_start2:j_end2+1] += a[np.arange(j_start2, j_end2+1) - k_i]
        
        # Project gradient onto tangent space of simplex
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean
        
        # Multiplicative update in log space
        step = 0.1
        log_a = np.log(np.maximum(a, 1e-30)) - step * grad_tangent
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        ac_new = fast_autocorr(a_new)
        if ac_new.max() < M:
            a = a_new
            # Decay step size
            step *= 0.999
        else:
            step *= 0.5
            if step < 1e-8:
                break
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubgradNewton it={it} ({elapsed:.0f}s): J={J:.6f}, |active|={len(active_lags)}")
    
    return best_a, best_m
```

For K=2000 and n=5000, this creates a 2000x5000 = 10M element matrix. Each row has at most n elements but typically fewer (around n/2 on average for random k). So the matrix is actually sparse (about 50% dense). 

The time to build this matrix is O(K * n_avg) where n_avg is the average number of non-zero entries per row, which is about n/2. So O(K*n/2) = O(2000*2500) = 5M operations. This should be fast in numpy.

Actually wait, I'm doing this in a Python loop over K constraints. That's K Python iterations, each with a numpy operation. For K=2000, this is 2000 Python iterations which might be slow. Let me vectorize.

Alternative: I can use the fact that the (k,j) entry is y[k-j] if 0 <= k-j < n, else 0. This is a Toeplitz-like structure. I can build it using scipy's toeplitz or by constructing the full matrix via indexing.

Actually, the simplest vectorized approach:
```python
def build_coeff_matrix_fast(constraint_list, n, y):
    K = len(constraint_list)
    k_arr = np.array(constraint_list, dtype=np.int64)  # shape (K,)
    j_arr = np.arange(n)  # shape (n,)
    # diff[k, j] = k_arr[k] - j_arr[j]
    diff = k_arr[:, None] - j_arr[None, :]  # shape (K, n)
    # Valid where 0 <= diff < n
    mask = (diff >= 0) & (diff < n)
    # coeffs[k, j] = y[diff[k,j]] where mask[k,j] is True
    coeffs = np.where(mask, y[diff.clip(0, n-1)], 0.0)
    return coeffs
```

This creates a (K, n) array for diff, which for K=2000, n=5000 is 10M elements = 80MB. That's a lot but manageable. And it's fully vectorized (no Python loop).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Frank-Wolfe with away-steps, line search, and vectorized constraint construction
2. Subgradient-based polish with adaptive step
3. Diverse initialization (spectral, comb, interpolated)
4. Multi-length exploration

import numpy as np
import time
import cvxpy as cp
import math


def fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_solver():
    """Find working LP solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver: ECOS (fallback)")
    return cp.ECOS


def build_lp_matrix(constraint_list, n, y):
    """Build K x n matrix where row k is y[k-j] for valid j."""
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_frank_wolfe(a0, time_limit, solver, K_init=200, K_max=2000, verbose=True):
    """LP Frank-Wolfe with away-steps and line search."""
    n = len(a0)
    y = norm_sum1(a0).copy()
    x_avg = y.copy()
    ac = fast_autocorr(y)
    best_a = y.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    
    top_idx = np.argsort(ac)[::-1][:K_init]
    constraint_set = set(int(k) for k in top_idx)
    it = 0
    consecutive_fails = 0
    
    while time.monotonic() - t0 < time_limit:
        ac_y = fast_autocorr(y)
        M = ac_y.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = y.copy()
        ac_x = fast_autocorr(x_avg)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x_avg.copy()
        
        # Constraint management
        cl = sorted(constraint_set)
        if len(cl) > K_max:
            margins = np.array([M - ac_y[k] for k in cl])
            keep = np.argsort(margins)[:K_max]
            constraint_set = set(int(cl[i]) for i in keep)
            cl = sorted(constraint_set)
        
        K = len(cl)
        if K < 10:
            break
        
        # Build LP
        C = build_lp_matrix(cl, n, y)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        con = [cp.sum(g_var) == 1, C @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), con)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            consecutive_fails += 1
            if consecutive_fails > 3:
                constraint_set = set(int(k) for k in np.argsort(ac_y)[::-1][:K_init])
                consecutive_fails = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            consecutive_fails += 1
            if consecutive_fails > 5 and K > K_init:
                constraint_set = set(int(k) for k in cl[:K_init])
                consecutive_fails = 0
            continue
        
        consecutive_fails = 0
        g = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g.sum()
        if gs < 1e-15:
            continue
        g = g / gs
        
        # Add violated constraints
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:2*n-1]
        viol = np.where(conv_yg > M + 1e-8)[0]
        if len(viol) > 0:
            va = conv_yg[viol] - M
            top_v = viol[np.argsort(va)[::-1][:20]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Periodically refresh with top lags
        if it % 15 == 0:
            for k in np.argsort(ac_y)[::-1][:30]:
                constraint_set.add(int(k))
        
        # Away-step decision
        s_idx = int(np.argmax(y))
        e_s = np.zeros(n); e_s[s_idx] = 1.0
        
        d_std = g - y
        d_away = g - e_s
        gap_std = y @ d_std
        gap_away = y @ d_away
        
        use_away = (y[s_idx] > 1e-5 and gap_away < gap_std - 1e-10)
        direction = d_away if use_away else d_std
        
        # Line search
        best_alpha = 2.0 / (it + 2.0)
        best_M = M
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, best_alpha, 0.05]:
            if use_away and y[s_idx] < alpha * (1.0 + 1e-10):
                continue
            yt = y + alpha * direction
            if yt.min() < -1e-12:
                continue
            yt = np.maximum(yt, 0.0)
            st = yt.sum()
            if st < 1e-15:
                continue
            yt /= st
            Mt = fast_autocorr(yt).max()
            if Mt < best_M:
                best_M = Mt
                best_alpha = alpha
        
        y_new = y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new /= sy
        
        # Update average
        beta = (it + 1.0) / (it + 3.0)
        x_avg = y_new + beta * (y_new - y)
        x_avg = np.maximum(x_avg, 0.0)
        sx = x_avg.sum()
        x_avg = x_avg / sx if sx > 1e-15 else y_new.copy()
        
        y = y_new
        it += 1
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFW iter={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    LPFW done ({elapsed:.0f}s): J={2*n*best_m:.6f}, it={it}")
    return best_a, best_m


def gradient_polish(a0, time_limit, verbose=True):
    """Subgradient-based polish with adaptive step."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.3
    stall = 0
    
    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break
        
        # Active set
        thresh = M - max(M * 0.005, 1e-10)
        active_lags = np.where(ac >= thresh)[0]
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(ac))])
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active_lags:
            ki = int(k)
            # d(conv_k)/da[j] = a[ki-j] + a[j-ki] (zero-padded)
            js1 = max(0, ki - n + 1)
            je1 = min(n - 1, ki)
            if js1 <= je1:
                grad[js1:je1+1] += a[ki - np.arange(js1, je1+1)]
            js2 = max(0, ki)
            je2 = min(n - 1, ki + n - 1)
            if js2 <= je2:
                grad[js2:je2+1] += a[np.arange(js2, je2+1) - ki]
        
        # Project to tangent
        grad -= grad.mean()
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - step * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        ac_new = fast_autocorr(a_new)
        if ac_new.max() < M - 1e-15:
            a = a_new
            step = min(step * 1.02, 1.0)
        else:
            step *= 0.7
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradPolish it={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m


def make_initializations(n):
    """Generate diverse initial sequences of length n."""
    x = np.linspace(0, 1, n)
    i = np.arange(n, dtype=np.float64)
    inits = []
    
    # Sine squared
    for freq in [1, 2, 3, 4]:
        a = np.sin(np.pi * freq * x) ** 2
        a = np.maximum(a, 1e-6)
        inits.append(a / a.sum())
    
    # Cosine window
    a = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(a / a.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Two-sided decay
    for sigma in [0.1, 0.2, 0.3]:
        c = n // 2
        a = np.exp(-((i - c) ** 2) / (2 * (sigma * n) ** 2))
        inits.append(a / a.sum())
    
    # Edge concentration
    for alpha in [0.5, 1.0, 2.0]:
        a = np.minimum(i + 1, n - i) ** alpha
        a = np.maximum(a, 1e-6)
        inits.append(a / a.sum())
    
    # Alternating (comb-like)
    for period in [2, 3, 4, 5]:
        a = np.where(np.arange(n) % period < period // 2, 1.0, 0.1)
        inits.append(a / a.sum())
    
    # Chirp
    for rate in [2, 5, 10]:
        phase = rate * x ** 2
        a = (1 + np.cos(2 * np.pi * phase)) / 2
        a = np.maximum(a, 0.05)
        inits.append(a / a.sum())
    
    return inits


def interpolate_seq(f, new_n):
    """Linearly interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    si = fi.sum()
    return fi / si if si > 1e-15 else np.ones(new_n) / new_n
