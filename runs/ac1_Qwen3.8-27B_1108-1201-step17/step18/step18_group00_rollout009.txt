            stall = 0
            gamma = max(2.0, c0 * 0.85)
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta_exp = max(cur_beta_exp - 0.5, 2.0)
            if verbose:
                print(f"    NSG restart t={t} ({time.monotonic()-t0:.1f}s): J={2*n*best_m:.6f}")
            continue

        # Nesterov extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = x + (beta - 1.0) * (x - x_prev)
        beta = beta_new
        y = project_simplex(y)

        # Compute soft subgradient
        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        # Soft weights (temperature-based)
        scaled = top_vals / max(top_vals.max(), 1e-15)
        w = np.exp(scaled * cur_beta_exp - cur_beta_exp)
        w = np.maximum(w, 1e-10)
        w /= w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        z = y - eta * grad
        x_new = project_simplex(z)

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            if adaptive and t % 500 == 0:
                gamma = min(gamma * 1.05, 200.0)
        else:
            stall += 1

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG t={t} ({elapsed:.1f}s): J={2*n*best_m:.6f}, eta={eta:.4f}")

    return best_a, best_m

def lp_fw_optimize(a0, time_limit, solver, initial_K=500, max_K=8000,
                   reeval_every=50, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    ac = fast_autocorr(a)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    K = initial_K
    no_improve = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 800:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Grow active set
        if t % reeval_every == 0:
            K = min(K + 200, max_K)
            # Re-evaluate top K
            top_idx = np.argsort(ac)[::-1][:K]
            active_set = set(int(k) for k in top_idx)

        # Truncate if needed
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:max_K]
            active_set = set(int(k) for k in list(active_set)[keep_idx])

        constraint_list = sorted(active_set)
        Kc = len(constraint_list)
        if Kc < 10:
            break

        # Build and solve LP
        coeffs = np.zeros((Kc, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        M_opt = float(M_var.value)

        # Add violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ag = np.fft.irfft(Af * Gf, fft_size)[:size]
        violations = np.where(conv_ag > M_opt + 1e-5)[0]
        if len(violations) > 0:
            viol_amt = conv_ag[violations] - M_opt
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                if len(active_set) < max_K:
                    active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        best_alpha = default_alpha
        best_trial_m = float('inf')
        for al in [default_alpha, 0.5 * default_alpha, 1.5 * default_alpha,
                   1.0 / (t + 1.0), 2.0 * default_alpha]:
            al = max(0.005, min(1.0, al))
            trial = (1.0 - al) * a + al * g_dir
            trial = norm_sum1(trial)
            m_try = fast_autocorr(trial).max()
            if m_try < best_trial_m:
                best_trial_m = m_try
                best_alpha = al

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = norm_sum1(a)

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-FW t={t} ({elapsed:.1f}s): J={2*n*best_m:.6f}, |S|={len(active_set)}, M_opt={M_opt:.6f}")

    return best_a, best_m

def coordinate_polish(a0, time_limit, rng_seed=99, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])
            if dst == src:
                continue

            found = False
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 200:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    amt = a[src] * rng.uniform(0.001, 0.005)
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.001, 0.01)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Coord ({elapsed:.1f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_score = evaluate_safe(p)
            if np.isfinite(p_score):
                best_seq = norm_sum1(p)
                best_score = p_score
                print(f"Parent: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 5000
        x = np.linspace(0, 1, n_init)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(seq, label):
        nonlocal best_seq, best_score
        if seq is None:
            return
        seq = np.maximum(np.asarray(seq, dtype=np.float64), 0.0)
        s = seq.sum()
        if s < 0.01:
            return
        seq_n = seq / s
        sc = evaluate_safe(seq_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"*** NEW BEST [{label}]: {sc:.6f} (n={len(seq)}, t={elapsed:.0f}s)")

    # Phase 1: Diverse parametric search across n values
    print("\n=== Phase 1: Parametric search ===")
    n_values = [3000, 4000, 5000, 6000, 7000]
    if len(best_seq) in n_values:
        n_values.remove(len(best_seq))
    n_values.append(len(best_seq))

    for n_try in n_values:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_pool = min(60.0, remaining * 0.1)
        if t_pool < 5:
            break
        cands = parametric_pool(n_try)
        for c in cands:
            v = obj_val(c)
            if v < 2.0:  # Only consider promising ones
                sc = evaluate_safe(c)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = c.copy()
                    print(f"  Param n={n_try}: J={sc:.6f}")

    # Phase 2: Nesterov subgradient on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Nesterov subgradient (n={n_best}) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, _ = nesterov_subgrad(best_seq, t2, top_k=80, beta_exp=6.0,
                                     c0=20.0, verbose=True)
        try_update(a_out, f"NSG_n{n_best}")

    # Phase 3: LP-FW main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW main (n={n_best}) ===")
    t3 = min(remaining * 0.30, 400.0)
    if t3 > 30 and n_best <= 8000:
        a_out, _ = lp_fw_optimize(best_seq, t3, solver, initial_K=500,
                                   max_K=6000, reeval_every=50, verbose=True)
        try_update(a_out, f"LPFW_n{n_best}")

    # Phase 4: More Nesterov with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Nesterov subgradient v2 (n={n_best}) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        a_out, _ = nesterov_subgrad(best_seq, t4, top_k=120, beta_exp=4.0,
                                     c0=40.0, verbose=False)
        try_update(a_out, f"NSG2_n{n_best}")

    # Phase 5: LP-FW with larger active set
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW large K (n={n_best}) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 30 and n_best <= 8000:
        a_out, _ = lp_fw_optimize(best_seq, t5, solver, initial_K=1000,
                                   max_K=10000, reeval_every=30, verbose=True)
        try_update(a_out, f"LPFW_largeK_n{n_best}")

    # Phase 6: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate polish (n={n_best}) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 15:
        a_out, _ = coordinate_polish(best_seq, t6, rng_seed=55, verbose=True)
        try_update(a_out, f"Coord_n{n_best}")

    # Phase 7: Try slightly different n
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    for delta in [500, -500, 1000]:
        new_n = n_best + delta
        if new_n < 2000 or new_n > 9000:
            continue
        if remaining < 120:
            break
        print(f"\n=== Phase 7: Scale n={new_n} ===")
        # Interpolate
        old_n = n_best
        s = best_seq.sum()
        f_norm = best_seq / s
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, new_n)
        a_long = np.interp(new_x, old_x, f_norm)
        a_long = norm_sum1(a_long)

        t7a = min(100.0, remaining * 0.2)
        if t7a > 20:
            a_out, _ = nesterov_subgrad(a_long, t7a, top_k=60, beta_exp=5.0,
                                         c0=15.0, verbose=False)
            try_update(a_out, f"NSG_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 7000:
            t7b = min(80.0, remaining * 0.3)
            if t7b > 20:
                a_out, _ = lp_fw_optimize(a_long, t7b, solver, initial_K=500,
                                           max_K=4000, reeval_every=40, verbose=False)
                try_update(a_out, f"LPFW_n{new_n}")

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP-FW + Coord (n={n_best}) ===")
    if remaining > 60 and n_best <= 8000:
        t8a = min(remaining * 0.5, 200.0)
        if t8a > 20:
            a_out, _ = lp_fw_optimize(best_seq, t8a, solver, initial_K=800,
                                       max_K=8000, reeval_every=25, verbose=True)
            try_update(a_out, f"LPFW_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8b = min(remaining * 0.6, 100.0)
        if t8b > 10:
            a_out, _ = coordinate_polish(best_seq, t8b, rng_seed=314, verbose=True)
            try_update(a_out, f"Coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        x = np.linspace(0, 1, 5000)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
