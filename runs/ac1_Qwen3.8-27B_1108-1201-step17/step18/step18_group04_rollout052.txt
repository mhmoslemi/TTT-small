    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]

def compute_smooth_subgrad(a, conv, temperature=0.01):

    n = len(a)
    conv_max = conv.max()
    # Use top-k peaks with softmax weights

2. Improved coordinate descent with FFT-based cost computation

import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

_best_tracker = {'seq': None, 'score': float('inf')}

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _lbfgs_obj(z):
    n = _opt_state['n']
    beta = _opt_state['beta']
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = fast_conv(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Numerically stable log-sum-exp
    max_shifted = shifted.max()
    if max_shifted > 500:
        log_sum_exp = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    else:
        log_sum_exp = np.log(np.sum(np.exp(shifted)))
    smooth_max = c_max + log_sum_exp / beta
    return 2.0 * n * smooth_max

def _lbfgs_grad(z):
    n = _opt_state['n']
    beta = _opt_state['beta']
    fft_size = _opt_state['fft_size']
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = fast_conv(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    max_shifted = shifted.max()
    if max_shifted > 500:
        w = np.exp(shifted - max_shifted)
    else:
        w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    # Gradient of smooth_max w.r.t. a: g_a[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * cross_correlation(w, a)_j
    # FFT: cross_correlation(f, g) = IFFT(conj(FFT(f)) * FFT(g))
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    g_a = np.fft.ifft(np.conj(W) * A).real[:n] * 2.0
    # Chain rule: d(obj)/d(z_j) = a_j * (g_a[j] - sum_i g_a[i]*a_i)
    dot_ga_a = np.dot(g_a, a)
    grad_z = a * (g_a - dot_ga_a)
    return 2.0 * n * grad_z

def lbfgs_optimize(a0, time_limit, beta_start=10.0, beta_end=200.0, 
                   n_stages=4, max_iter_per_stage=300, seed=42):

    a = norm_sum1(a0)
    fft_size = 1 << (2*n - 1).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()

        # Anneal beta

        beta = beta_start * (beta_end / beta_start) ** frac
        _opt_state['n'] = n
        _opt_state['beta'] = beta
        _opt_state['fft_size'] = fft_size
        
        z = np.log(a + 1e-30)
        remaining = time_limit - (time.monotonic() - t0)
        
        # Limit max evaluations based on time
        maxfun = max(100, int(remaining / 0.05))  # ~50ms per eval+grad
        
        try:
            res = minimize(_lbfgs_obj, z, jac=_lbfgs_grad, 
                          method='L-BFGS-B',
                          options={'maxiter': max_iter_per_stage, 
                                   'maxfun': maxfun,
                                   'ftol': 1e-12, 'gtol': 1e-8})
            z_new = res.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0.0)
            obj_new = 2.0 * n * fast_conv(a_new).max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        except Exception as e:
            print(f"  L-BFGS stage {stage} error: {e}")
            break
    
    return best_a, best_obj
```

Now let me think about the coordinate descent improvement. The key insight is that after L-BFGS converges, the sequence is close to a local minimum of the smooth approximation. The coordinate descent should then work on the true (non-smooth) objective.

For coordinate descent, the key operation is: "move mass from coordinate j to coordinate k, and check if the max convolution decreases."

The efficient way to check this:
- Current max is M at position k*.
- If we decrease a[j] by δ and increase a[k] by δ, the change in c is:
  - `c_new[m] = c[m] + δ*(a[m-k] - a[m-j]) + δ^2*(delta terms)`
  - For small δ, the dominant change is `δ*(a[m-k] - a[m-j])`
- The new max is approximately `max(M, max over m of (c[m] + δ*(a[m-k] - a[m-j])))`
- For the max to decrease, we need `a[m-k] < a[m-j]` for all m near the peak.

This is a complex condition. In practice, the coordinate descent just tries the move and checks.

Let me make the coordinate descent more efficient by:
1. Pre-computing the "cost" vector (how much each coordinate contributes to the peak)
2. Only trying moves that are likely to help
3. Using binary search on the transfer amount

Here's my improved coordinate descent:

```python
def coord_descent_v2(a0, time_limit, seed=42):
    """Improved coordinate descent with FFT-based cost computation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    fft_size = 1 << (2*n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))
        
        # Compute cost of each coordinate (contribution to peak)
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        cost = a[idx] * a[peak_k - idx]
        cost_full = np.zeros(n)
        cost_full[idx] = cost
        
        # Find top peaks
        n_peaks = min(20, 2*n - 1)
        peak_positions = np.argsort(conv)[::-1][:n_peaks]
        
        # Compute total peak cost
        total_cost = np.zeros(n)
        for pk in peak_positions:
            pk_i = int(pk)
            lo2 = max(0, pk_i - n + 1)
            hi2 = min(n - 1, pk_i)
            if lo2 > hi2:
                continue
            idx2 = np.arange(lo2, hi2 + 1)
            total_cost[idx2] += a[idx2] * a[pk_i - idx2]
        
        # Source: high cost coordinates
        src_candidates = np.argsort(total_cost)[::-1][:min(50, n//20)]
        # Destination: low cost coordinates
        dst_candidates = np.argsort(total_cost)[:min(50, n//20)]
        
        improved = False
        for src_i in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src_i)
            if a[src_i] < 1e-12:
                continue
            for dst_i in dst_candidates:
                dst_i = int(dst_i)
                if dst_i == src_i:
                    continue
                # Try binary search on transfer amount
                lo_amt = 0.0
                hi_amt = a[src_i] * 0.5
                best_frac = 0.0
                best_trial_m = best_m
                for _ in range(8):
                    mid_amt = (lo_amt + hi_amt) / 2.0
                    a_trial = a.copy()
                    a_trial[src_i] -= mid_amt
                    a_trial[dst_i] += mid_amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_frac = mid_amt
                        lo_amt = mid_amt
                    else:
                        hi_amt = mid_amt
                
                if best_frac > 0 and best_trial_m < best_m - 1e-14:
                    a[src_i] -= best_frac
                    a[dst_i] += best_frac
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_m = best_trial_m
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = min(10, n // 100)
            sources = rng.choice(n, n_pert, replace=False)
            targets = rng.choice(n, n_pert, replace=False)
            for s, d in zip(sources, targets):
                if a[s] > 1e-10 and s != d:
                    amt = a[s] * rng.uniform(0.001, 0.01)
                    a[s] -= amt
                    a[d] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    
    return best_a, best_m
```

Now let me think about the LP-based approach. The LP from the SOTA is:
Maximize `sum(b)` subject to `b >= 0` and `conv(f, b)[k] <= M` for all k.

For n=4000, this is a large LP. Let me see if I can solve it efficiently.

The constraint matrix A has shape (2n-1) × n, with A[k,j] = f[j] if 0 <= k-j <= n-1.

Total non-zeros: sum_{k=0}^{2n-2} min(k+1, n, 2n-1-k) = n^2 (approximately).

For n=4000, that's 16 million non-zeros. Building this matrix might take a second or two. Solving the LP with HiGHS might take 10-30 seconds.

```python
def build_conv_matrix(f):
    """Build the convolution matrix A where A[k,j] = f[j] if 0<=k-j<=n-1."""
    n = len(f)
    # Use scipy's convolution to build each row
    # Row k: A[k,j] = f[j] for j in [max(0,k-n+1), min(n-1,k)]
    # This is like a band matrix
    from scipy.sparse import diags
    
    # For row k, the non-zero entries are f[k-n+1 : k+1] (clipped to [0,n-1])
    # The matrix is a "convolution matrix" (also called a Toeplitz-like matrix)
    
    # Alternative: use the fact that A = T(f) where T is the Toeplitz operator
    # For full convolution (not circular), this is a (2n-1) x n matrix.
    
    # Let me just build it row by row but in numpy (faster than pure Python)
    A_rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        row = np.zeros(n)
        row[lo:hi+1] = f[lo:hi+1]
        A_rows.append(row)
    return np.array(A_rows)

**Alternative to LP: "Iterative water-filling"**

Instead of solving the LP exactly, I can use an iterative approach:
1. Start with b = f (normalized)
2. Find the constraint that's most violated: `k* = argmax_k conv(f, b)[k]`
3. Project b onto the half-space `conv(f, b)[k*] <= M`
4. Repeat

The projection onto `conv(f, b)[k*] <= M` with `b >= 0` is:
`minimize ||b - b_old||^2 subject to sum_j f[j]*b[k*-j] <= M, b >= 0`

This is a quadratic program with one linear constraint and box constraints. It can be solved in O(n) using a water-filling algorithm.

Actually, this is similar to the "projected subgradient method" or "POCS" (Projection Onto Convex Sets).

Let me implement this:

```python
def pocs_direction(f, M, n_iter=100):
    """Find b >= 0, sum(b)=sum(f) such that conv(f,b) <= M, using POCS."""
    n = len(f)
    b = norm_sum1(f).copy()
    
    for _ in range(n_iter):
        # Project onto {b: conv(f,b) <= M}
        conv_fb = fast_conv(f, b)  # Need to handle this
        # Actually, conv(f, b) where f and b have different roles
        # conv(f,b)[k] = sum_j f[j] * b[k-j]
        
        # The constraint is: for all k, sum_j f[j] * b[k-j] <= M
        # The most violated constraint is at k* = argmax_k conv(f,b)[k]
        
        # Project b onto {b: sum_j f[j] * b[k*-j] <= M} ∩ {b >= 0}
        # This is: minimize sum_j (b[j] - b_old[j])^2 s.t. sum_j f[j]*b[k*-j] <= M, b >= 0
        
        # The constraint sum_j f[j]*b[k*-j] <= M involves b at positions k*-j for valid j
        # Let S = {j: 0 <= k*-j <= n-1} = {j: k*-n+1 <= j <= k*} ∩ [0, n-1]
        # For j in S, the coefficient is f[j]
        
        # This is a QP: min sum (b-j - b_old-j)^2 s.t. sum_{j in S} f[j]*b[k*-j] <= M, b >= 0
        # In the active set (j in S), the constraint is linear in b.
        # Outside S, b is free (just b >= 0).
        
        # By KKT, the solution has b[j] = max(0, b_old[j] - λ*f[j]*(some factor)) for j in S
        # and b[j] = max(0, b_old[j]) for j not in S
        
        # Actually, the constraint is: sum_{j in S} f[j] * b[k*-j] <= M
        # Let me reindex: let i = k*-j, so j = k*-i. The constraint becomes:
        # sum_{i: 0<=i<n, 0<=k*-i<n} f[k*-i] * b[i] <= M
        # The coefficient of b[i] in the constraint is f[k*-i] (if k*-i is valid)
        
        # So the constraint is: sum_i c[i]*b[i] <= M where c[i] = f[k*-i] if 0<=k*-i<n, else 0
        
        # QP: min sum_i (b[i] - b_old[i])^2 s.t. sum_i c[i]*b[i] <= M, b >= 0
        # Solution: b[i] = max(0, b_old[i] - λ*c[i]) where λ is chosen so that
        # sum_i c[i]*max(0, b_old[i] - λ*c[i]) = M (if the constraint is active)
        
        k_star = int(np.argmax(conv_fb))
        c = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        c[lo:hi+1] = f[lo:hi+1]
        
        # Check if constraint is violated
        current = np.dot(c, b)
        if current <= M + 1e-10:
            # Constraint satisfied, just project onto b >= 0
            b = np.maximum(b, 0)
            continue
        
        # Solve for λ: sum_i c[i]*max(0, b[i] - λ*c[i]) = M
        # For i where c[i] > 0: max(0, b[i] - λ*c[i]) = b[i] - λ*c[i] if λ < b[i]/c[i], else 0
        # Binary search on λ
        lo_lambda = 0.0
        hi_lambda = np.max(b[np
