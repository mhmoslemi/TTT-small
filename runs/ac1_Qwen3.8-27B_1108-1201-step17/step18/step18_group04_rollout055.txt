import numpy as np

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_power_grad(a, conv, p):
    """
    Compute the full gradient of (sum conv^p)^(1/p) w.r.t. a.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    conv_pos = np.maximum(conv, 0.0)
    power_sum = np.sum(conv_pos ** p)
    if power_sum < 1e-30:
        return np.zeros(n)
    
    # w[k] = 2*p * (power_sum)^(1/p - 1) * conv[k]^(p-1)
    coeff = 2.0 * p * power_sum ** (1.0 / p - 1.0)
    w = coeff * np.power(conv_pos, p - 1.0)
    
    # grad[j] = sum_i a[i] * w[j+i] for valid j+i
    # = (conv(a, w_reversed))[2n-2-j]
    v = w[::-1]  # reverse w
    fft_size = 1 << (n + len(v) - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    V = np.fft.rfft(v, fft_size)
    full_conv = np.fft.irfft(A * V, fft_size)
    
    # grad[j] = full_conv[2*n-2-j] for j = 0, ..., n-1
    grad = full_conv[2*n-2 : n-2 : -1]
    return grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with full FFT-based gradient of power-sum surrogate
2. Adaptive power parameter (30→150) for smooth-to-peak transition
3. Diverse window-function initializations (Hann, Hamming, Blackman, U-shape, Gaussian)
4. Multi-length interpolation search
5. Coordinate descent final polish
Key innovation: Full O(n log n) gradient via FFT cross-correlation, enabling faster
convergence than the previous top-k subgradient approach.
"""

import numpy as np
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    if size <= 0:
        return np.zeros(1)
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_power_grad(a, conv, p):
    n = len(a)
    conv_mean = conv.mean()
    if conv_mean < 1e-30:
        return np.zeros(n)
    conv_norm = conv / conv_mean
    conv_norm = np.maximum(conv_norm, 1e-30)
    log_conv = np.log(conv_conv if False else conv_norm)
    log_w = (p - 1.0) * log_conv
    max_log_w = log_w.max()
    if max_log_w < -700:
        return np.zeros(n)
    w = np.exp(log_w - max_log_w)
    v = w[::-1]
    total_len = n + len(v) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    V = np.fft.rfft(v, fft_size)
    full_conv = np.fft.irfft(A * V, fft_size)
    grad = full_conv[2 * n - 2: n - 2: -1]
    gmax = np.abs(grad).max()
    if gmax > 1e-30:
        grad = grad / gmax
    return grad


def nesterov_mirror_opt(a0, time_limit, p_start=30, p_end=120, c_init=3.0,
                        decay_power=0.6, seed=42, verbose=False):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 4000
    rng = np.random.default_rng(seed)
    restart_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, t / 8000.0)
        p = p_start + (p_end - p_start) * frac
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        conv_y = fast_conv(y)
        grad = compute_power_grad(y, conv_y, p)
        eta = c_init / (t + 1.0) ** decay_power
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        x_prev = x
        x = a_new
        M = fast_conv(x).max()
        if M < best_m - 1e-14:
            best_m = M
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            stall = 0
            restart_count += 1
            max_stall = max(1500, max_stall // 2)
            strength = min(0.05, 0.01 + 0.004 * restart_count)
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose and restart_count % 2 == 0:
                J = 2.0 * n * best_m
                print(f"      restart {restart_count} t={t}: J={J:.6f}")

    return best_x, best_m


def coord_polish(a0, time_limit, seed=42, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]
        n_top = max(8, min(30, n // 100))
        top_src = idx[np.argsort(contrib)[::-1][:n_top]]
        n_top_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_top_peaks]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]
        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            candidates = np.argsort(costs + 1e-10 * a)[:max(15, n // 25)]
            candidates = [int(c) for c in candidates if int(c) != src_i]
            for dst_i in candidates[:12]:
                found = False
                for frac in [0.03, 0.07, 0.15, 0.25, 0.4, 0.6]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_hann(n):
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_hamming(n):
    x = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / max(n - 1, 1))
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_blackman(n):
    x = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / max(n - 1, 1)) + 0.08 * np.cos(4.0 * np.pi * x / max(n - 1, 1))
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_u_shape(n, alpha=1.0):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    t = np.abs(x - center) / center
    a = 1.0 + alpha * t ** 2
    return norm_sum1(a)


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return norm_sum1(a)


def make_decreasing(n, power=0.5):

def make_increasing(n, power=0.5):

def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2

    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2

def make_bimodal(n, pos1=0.15, pos2=0.85, width=0.08):

def make_random_smooth(n, seed=0, n_freq=8):

                                           p_start=50, p_end=150, c_init=3.0,
                                           decay_power=0.6, seed=555, verbose=False)
            try_update(a_out, f"gd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t4b = min(remaining * 0.25, 100.0)
            if t4b > 10:
                a_out2, _ = coord_polish(a_out, t4b, seed=314, verbose=False)
                try_update(a_out2, f"cd_n{new_n}")

    # Phase 5: Final gradient polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 5: Final (n={n_best}, t={remaining:.0f}s) ===")
    if remaining > 40 and _best_tracker['seq'] is not None:
        a_in = norm_sum1(_best_tracker['seq'])
        t5 = min(remaining * 0.6, 300.0)
        a_out, _ = nesterov_mirror_opt(a_in, t5,
                                       p_start=100, p_end=250, c_init=1.5,
                                       decay_power=0.7, seed=2024, verbose=True)
        try_update(a_out, "gd_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25 and _best_tracker['seq'] is not None:
        a_in = norm_sum1(_best_tracker['seq'])
        t5b = min(remaining * 0.7, 150.0)
        a_out2, _ = coord_polish(a_in, t5b, seed=9999, verbose=False)
        try_update(a_out2, "cd_final")

    # Final
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        _best_tracker['seq'] = make_hann(n_fb)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
