def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    """Normalize to non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def gs_spectral_shaping(a0, n_iter=5000, verbose=False):
    """
    Gerchberg-Saxton style spectral shaping.
    Iteratively: FFT -> flatten magnitude (partially) -> IFFT -> clip negative -> normalize.
    This encourages flat power spectrum which reduces peak autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    
    # Start with strong flattening, decrease over time
    alpha_start = 0.5
    alpha_end = 0.05
    
    for it in range(n_iter):
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(n_iter - 1, 1))
        
        A = np.fft.rfft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: blend current magnitude with flat
        target_mag = alpha * mag + (1 - alpha) * mag.max()
        # Preserve DC
        target_mag[0] = mag[0]
        
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        
        # Enforce non-negativity and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        ac = fast_autocorr(a_new)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 1000 == 0:
            print(f"    GS iter {it}: J={val:.6f}")
    
    return best_a, best_val


def lse_gradient(a, ac, tau):
    """Compute gradient of tau*log(sum exp(ac_k/tau)) w.r.t. a using FFT."""
    n = len(a)
    ac_scaled = ac / tau
    ac_scaled -= ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()
    
    # grad[i] = 2 * sum_j p[i+j] * a[j] for valid indices
    a_flip = a[::-1].astype(np.float64)
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return grad


def smooth_adam_v3(a0, time_limit, tau_decay=0.99985, lr=1.0, beta1=0.9, beta2=0.999,
                   restart_every=3000, verbose=True, rng_seed=42):
    """
    Improved multiplicative Adam with log-sum-exp smoothing.
    Key improvements: adaptive tau based on current max, better restart strategy.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    tau = max(ac.max() * 0.3, 1e-8)
    tau_min = max(ac.max() * 0.001, 1e-12)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    best_grad = None
    
    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive tau: make it proportional to current max
        target_tau = best_m * 0.1
        tau = max(tau * tau_decay, min(target_tau, tau))
        tau = max(tau, tau_min)
        
        g = lse_gradient(a, ac, tau)
        
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new
        
        if stall > restart_every:
            stall = 0
            # More diverse restart: blend with random
            w = rng.uniform(0.3, 0.7)
            pert = np.ones(n) * rng.uniform(0.5, 1.5)
            x = w * best_a + (1 - w) * pert / pert.sum()
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            tau = max(fast_autocorr(a).max() * 0.3, 1e-8)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SmoothAdam v3 restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothAdam v3 t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    SmoothAdam v3 done ({elapsed:.0f}s): J={2*n*best_m:.6f}")

            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW3 iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW3 done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_descent_v3(a0, time_limit, rng_seed=42, verbose=True):
    """
    Coordinate descent with optimal transfer search.
    Tries to move mass from peak-contributing positions to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
        
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination using score
            scores = np.zeros(n)
            for k2 in top_k[:4]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:4]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]
            
            if dst == src:
                continue
            
            # Try fractions
            found = False
            for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and stall > 50:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
            # Random exploration
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord3 ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, seed=0):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(seed)
    candidates = []
    
    # Sin^2 family
    for freq in [1, 2, 3]:
        for phase in [0.0, 0.2, 0.5, 0.8]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.005)
            candidates.append(a / a.sum())
    
    # Power law
    for p in [0.2, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())
    
    # Beta-like
    for alpha in [0.3, 0.5, 1.0, 2.0, 4.0]:
        for beta in [0.3, 0.5, 1.0, 2.0, 4.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(a / a.sum())
    
    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.05, 0.1, 0.15, 0.25, 0.4]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())
    
    # Dual Gaussian
    for c1, c2 in [(0.3, 0.7), (0.2, 0.8), (0.4, 0.6)]:
        for sigma in [0.1, 0.15, 0.2]:
            a = np.exp(-((x - c1) ** 2) / (2 * sigma**2)) + np.exp(-((x - c2) ** 2) / (2 * sigma**2))
            candidates.append(a / a.sum())
    
    # Cosine taper variants
    for p in [1, 2, 3, 4]:
        a = np.cos(np.pi * (x - 0.5)) ** (2 * p)
        a = np.maximum(a, 0.001)
        candidates.append(a / a.sum())
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Random smooth
    for _ in range(5):
        rough = rng.uniform(0.1, 1.0, n)
        # Smooth with moving average
        k = max(3, n // 100)
        kernel = np.ones(k) / k
        smooth = np.convolve(rough, kernel, mode='same')
        candidates.append(norm_sum1(smooth))
    
    return candidates


def subgradient_descent(a0, time_limit, lr_start=0.1, lr_decay=0.999, verbose=True):
    """
    Subgradient descent on the max autocorrelation.
    The subgradient of max_k (a*a)[k] w.r.t. a is determined by the argmax.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = lr_start
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        k_star = int(np.argmax(ac))
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
        
        # Subgradient: grad_i = 2 * (a*a)[k_star - i] type term
        # Actually: d/dc max_k conv(a,a)[k] at k=k* is 2*a convolved with delta at k*
        # The subgradient w.r.t. a_i is 2 * a_{k*-i} (for valid i)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        grad = np.zeros(n)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad[idx] = 2.0 * a[k_star - idx]
        
        # Projected subgradient step
        lr_t = lr / (1.0 + 0.01 * t)
        a_new = a - lr_t * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        lr *= lr_decay
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    Subgrad done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-resolution approach."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Multi-resolution optimization ===
    print(f"\n=== Phase 1: Multi-resolution ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    # Start small and work up
    n_start = min(len(best_seq), 200)
    n_sequence = [n_start]
    # Build sequence of sizes up to target
    target_n = len(best_seq)
    cur = n_start
    while cur < target_n and remaining > 100:
        cur = min(int(cur * 1.6), target_n)
        if cur > n_sequence[-1]:
            n_sequence.append(cur)
    
    print(f"  Lengths: {n_sequence}")
    
    # Initialize at smallest
    a_small = interpolate_sequence(best_seq, n_start)
    a_small = norm_sum1(a_small)
    
    for si, n_i in enumerate(n_sequence):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        
        if si == 0:
            a_cur = a_small
        else:
            a_cur = interpolate_sequence(a_cur, n_i)
            a_cur = norm_sum1(a_cur)
        
        # Quick GS at this size
        t_gs = min(remaining * 0.15, 30.0)
        if t_gs > 5:
            a_gs, _ = gs_spectral_shaping(a_cur, n_iter=2000)
            ac_gs = fast_autocorr(a_gs)
            ac_cur = fast_autocorr(a_cur)
            if ac_gs.max() < ac_cur.max():
                a_cur = a_gs
        
        # Quick Adam at this size
        t_adam = min(remaining * 0.25, 40.0)
        if t_adam > 10:
            a_adam, _ = smooth_adam_v3(a_cur, t_adam, verbose=False, rng_seed=si)
            ac_adam = fast_autocorr(a_adam)
            ac_cur = fast_autocorr(a_cur)
            if ac_adam.max() < ac_cur.max():
                a_cur = a_adam
        
        a_small = a_cur
        elapsed = time.monotonic() - t_global
        ac_cur = fast_autocorr(a_cur)
        print(f"  Multi-res n={n_i} ({elapsed:.0f}s): max_ac={ac_cur.max():.6e}, J={2*n_i*ac_cur.max():.6f}")
    
    check_and_update(a_small, "multi_res")
    
    # === Phase 2: GS spectral shaping on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: GS ({n_best}) ===")
    t_gs_main = min(remaining * 0.10, 100.0)
    if t_gs_main > 20:
        a_gs, _ = gs_spectral_shaping(norm_sum1(best_seq), n_iter=10000, verbose=True)
        check_and_update(a_gs, "gs_main")
        # Also try starting from multi-res result
        a_gs2, _ = gs_spectral_shaping(a_small, n_iter=5000, verbose=False)
        check_and_update(a_gs2, "gs_res")
    
    # === Phase 3: Smoothed Adam (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: SmoothAdam ({n_best}) ===")
    t_adam_main = min(remaining * 0.25, 400.0)
    if t_adam_main > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam_v3(a_in, t_adam_main, tau_decay=0.9999, lr=0.8,
                                   beta1=0.9, beta2=0.999, restart_every=2500,
                                   verbose=True, rng_seed=77)
        check_and_update(a_out, f"adam_main_n{n_best}")
        
        # Also run from multi-res result
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100:
            a_out2, _ = smooth_adam_v3(a_small, min(remaining * 0.2, 200.0),
                                         tau_decay=0.9999, lr=0.7, verbose=False, rng_seed=123)
            check_and_update(a_out2, "adam_res")
    
    # === Phase 4: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW ({n_best}) ===")
    t_lp = min(remaining * 0.30, 500.0)
    if t_lp > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v3(a_in, t_lp, solver, initial_K=400,
                                      max_K=3500, add_per_iter=40, verbose=True)
        check_and_update(a_out, f"lp_n{n_best}")
    
    # === Phase 5: Subgradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 5: Subgradient ({n_best}) ===")
        t_sg = min(remaining * 0.1, 80.0)
        if t_sg > 20:
            a_out, _ = subgradient_descent(norm_sum1(best_seq), t_sg, lr_start=0.05, verbose=True)
            check_and_update(a_out, "subgrad")
    
    # === Phase 6: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for delta in [300, -300, 500, -500, 1000]:
        new_n = n_best + delta
        if 1500 <= new_n <= 6000:
            nearby_lengths.append(new_n)
    
    seen = set()
    unique_lengths = []
    for ln in nearby_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)
    
    for new_n in unique_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 6: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        
        t6a = min(remaining * 0.3, 120.0)
        if t6a > 20:
            a_out, _ = smooth_adam_v3(a_long, t6a, verbose=False, rng_seed=200+new_n)
            check_and_update(a_out, f"adam_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 5500:
            t6b = min(remaining * 0.4, 150.0)
            if t6b > 20:
                a_out, _ = lp_frank_wolfe_v3(a_long, t6b, solver, initial_K=300,
                                              max_K=2500, add_per_iter=30, verbose=False)
                check_and_update(a_out, f"lp_n{new_n}")
    
    # === Phase 7: Final LP-FW push ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP ({n_best}) ===")
    if remaining > 60 and n_best <= 5500:
        t7 = min(remaining * 0.4, 250.0)
        if t7 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v3(a_in, t7, solver, initial_K=500,
                                          max_K=4000, add_per_iter=50, verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")
    
    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Coord ({n_best}) ===")
    if remaining > 15:
        t8 = min(remaining * 0.6, 120.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_v3(a_in, t8, rng_seed=314, verbose=False)
            check_and_update(a_out, f"coord_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
