def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgrad_full(a, conv, threshold_frac=0.01):
    """Compute subgradient using all peaks within threshold of max."""
    n = len(a)
    conv_max = conv.max()
    threshold = conv_max * (1 - threshold_frac)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Weight by how close to max
    w = np.exp((conv[active] - conv_max) / (conv_max * threshold_frac + 1e-15))
    w_sum = w.sum()
    w = w / w_sum

        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

The constraint conv(a, g)[k] = sum_j a[j] * g[k-j] <= M.

This is a linear program. The optimal solution g* will have sum(g*) >= 1 (since a itself is a feasible solution with sum(a) = 1 and conv(a,a)[k] <= M).

If we scale g = g* / sum(g*), then sum(g) = 1 and conv(a,g)[k] <= M / sum(g*).

If sum(g*) > 1, then the new "effective M" is smaller, which is good.

The update a_new = (1-t)*a + t*g:
conv(a_new, a_new) = (1-t)^2 conv(a,a) + 2t(1-t) conv(a,g) + t^2 conv(g,g)

For small t, the dominant change is 2t(1-t)(conv(a,g) - conv(a,a)).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start AMD (Accelerated Mirror Descent) with adaptive active-set subgradient
2. LP-based polish steps (solve max sum(g) s.t. conv(a,g)<=M, g>=0)
3. Coordinate descent final polish
4. Multi-length search with interpolation
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad(a, conv, threshold_frac=0.005, max_peaks=50):
    """Subgradient of max_k conv[k] using top peaks within threshold."""
    n = len(a)
    conv_max = conv.max()
    if conv_max < 1e-15:
        return np.zeros(n)
    threshold = conv_max * (1.0 - threshold_frac)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    if len(active) > max_peaks:
        # Keep only the top max_peaks
        order = np.argsort(conv[active])[::-1][:max_peaks]
        active = active[order]
    
    # Temperature for softmax weighting
    temp = conv_max * threshold_frac * 0.5
    w = np.exp((conv[active] - conv_max) / temp)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad


def amd_optimize(a0, time_limit, threshold_frac=0.005, c0=3.0,
                 restart_every=3000, seed=42, max_peaks=40):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall_count = 0
    restart_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv_y = fast_conv(y)
        g = compute_subgrad(y, conv_y, threshold_frac=threshold_frac, max_peaks=max_peaks)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.1, 0.02 + 0.01 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Tighten the active set over time
            threshold_frac = max(threshold_frac * 0.85, 0.001)

    return best_x, best_m


def lp_polish(a0, time_limit, max_n_lp=2500, seed=42):
    """LP-based polish: repeatedly solve max sum(g) s.t. conv(a,g)<=M, g>=0."""
    n = len(a0)
    a_full = norm_sum1(a0)
    best_a = a_full.copy()
    best_m = fast_conv(a_full).max()
    t0 = time.monotonic()
    
    # Work with reduced n if needed
    n_work = min(n, max_n_lp)
    if n_work < n:
        old_x = np.linspace(0, 1, n)
        new_x = np.linspace(0, 1, n_work)
        a_work = np.interp(new_x, old_x, a_full)
        a_work = norm_sum1(a_work)
    else:
        a_work = a_full.copy()
    
    step_size = 0.4
    no_improve = 0
    
    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_work = fast_conv(a_work)
        M_work = conv_work.max()
        
        # Build LP
        n_w = len(a_work)
        rows = np.zeros(n_w * n_w, dtype=np.int32)
        cols = np.zeros(n_w * n_w, dtype=np.int32)
        vals = np.zeros(n_w * n_w, dtype=np.float64)
        ptr = 0
        for k in range(2 * n_w - 1):
            lo = max(0, k - n_w + 1)
            hi = min(n_w - 1, k)
            for j in range(lo, hi + 1):
                rows[ptr] = k
                cols[ptr] = j
                vals[ptr] = a_work[j]
                ptr += 1
        
        A = csr_matrix((vals[:ptr], (rows[:ptr], cols[:ptr])), shape=(2*n_w-1, n_w))
        c = -np.ones(n_w)
        b_ub = np.full(2*n_w-1, M_work)
        
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n_w, method='highs',
                            options={'presolve': True, 'time_limit': max(1, time_limit - (time.monotonic()-t0))})
        except Exception:
            break
        
        if not result.success:
            break
        
        g = result.x
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        # Blend
        a_new = (1 - step_size) * a_work + step_size * g
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        M_new = fast_conv(a_new).max()
        if M_new < M_work * (1 - 1e-8):
            a_work = a_new
            no_improve = 0
            # Increase step size slightly
            step_size = min(0.7, step_size * 1.05)
        else:
            step_size *= 0.6
            no_improve += 1
            if no_improve > 5:
                break
        
        # Update best on full sequence
        if n_work < n:
            a_trial = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_work), a_work)
            a_trial = norm_sum1(a_trial)
        else:
            a_trial = a_work
        
        M_full = fast_conv(a_trial).max()
        if M_full < best_m - 1e-12:
            best_m = M_full
            best_a = a_trial.copy()
    
    return best_a, best_m


def coord_descent_polish(a0, time_limit, seed=42):
    """Move mass from high-contribution positions to low-contribution ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        n_top = max(10, min(30, n // 150))
        top_src = idx[np.argsort(contrib)[::-1][:n_top]]
        
        # Compute cost for each position (contribution to top peaks)
        top_peaks = np.argsort(conv)[::-1][:min(20, 2*n-1)]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]
        
        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            n_cand = max(15, n // 25)
            candidates = np.array([int(c) for c in np.argsort(costs + 1e-10 * a)[:n_cand] if int(c) != src_i])
            for dst_i in candidates[:12]:
                found = False
                for frac in [0.03, 0.07, 0.15, 0.3, 0.5]:
                    amt = a[src_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_sin2(n):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.sin(np.pi * x) ** 2)


def make_cos2(n):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.cos(np.pi * (x - 0.5)) ** 2)


def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_increasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((x + 1) ** power)


def make_gaussian(n, center=0.5, width=0.15):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.exp(-((x - center) / width) ** 2))


def make_two_gauss(n, c1=0.25, c2=0.75, w=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / (w * 1.3)) ** 2)
    return norm_sum1(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_exponential(n, rate=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm_sum1(np.exp(-rate * x))


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_b_spline(n, order=3):
    """Approximate a B-spline shape."""
    x = np.linspace(0, 1, n)
    # Cubic B-spline on [0,1] is approximately a raised cosine
    a = 1.0 - (2 * x - 1) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

        else:
            a_trial = a_work

        M_full = fast_conv(a_trial).max()
        if M_full < best_m - 1e-12:
            best_m = M_full
            best_a = a_trial.copy()

    return best_a, best_m


def coord_descent_polish(a0, time_limit, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]

        n_top = max(10, min(30, n // 150))
        top_src = idx[np.argsort(contrib)[::-1][:n_top]]

        top_peaks = np.argsort(conv)[::-1][:min(20, 2 * n - 1)]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            n_cand = max(15, n // 25)
            cand_arr = np.array([int(c) for c in np.argsort(costs + 1e-10 * a)[:n_cand] if int(c) != src_i])
            for dst_i in cand_arr[:12]:
                found = False
                for frac in [0.03, 0.07, 0.15, 0.3, 0.5]:
                    amt = a[src_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_sin2(n):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.sin(np.pi * x) ** 2)


def make_cos2(n):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.cos(np.pi * (x - 0.5)) ** 2)


def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_increasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((x + 1) ** power)


def make_gaussian(n, center=0.5, width=0.15):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.exp(-((x - center) / width) ** 2))


def make_two_gauss(n, c1=0.25, c2=0.75, w=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / (w * 1.3)) ** 2)
    return norm_sum1(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_exponential(n, rate=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm_sum1(np.exp(-rate * x))


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_b_spline(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_asym_gauss(n, lc=0.3, rc=0.7, lw=0.12, rw=0.2):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - lc) / lw) ** 2) + 0.7 * np.exp(-((x - rc) / rw) ** 2)
    return norm_sum1(a)


def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()

    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 4000

        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])

    n0 = len(_best_tracker['seq'])

    # Phase 1: Multi-start AMD
    print(f"\n=== Phase 1: Multi-start AMD ===")
    t1 = min((time_budget - (time.monotonic() - t_global)) * 0.42, 700.0)

        inits = []

        inits.append(make_sin2(n0)); labels.append("sin2")
        inits.append(make_cos2(n0)); labels.append("cos2")

        inits.append(make_decreasing(n0, 0.7)); labels.append("dec07")
        inits.append(make_increasing(n0, 0.3)); labels.append("inc03")
        inits.append(make_gaussian(n0, 0.5, 0.2)); labels.append("gauss")
        inits.append(make_two_gauss(n0)); labels.append("2gauss")

        inits.append(make_random_smooth(n0, seed=42, n_freq=10)); labels.append("rnd2")
        inits.append(make_b_spline(n0)); labels.append("bspl")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(15.0, t1 / n_inits)
        for idx in range(n_inits):
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - elapsed)
            if t_avail < 10:
                break
            lbl = unique_labels[idx]
            print(f"  AMD '{lbl}' t={t_avail:.0f}s")
            a_out, _ = amd_optimize(unique_inits[idx], t_avail,
                                    threshold_frac=0.005, c0=3.5,
                                    restart_every=2500, seed=100 + idx)
            try_update(a_out, f"amd_{lbl}")

    # Phase 2: LP + AMD refinement on best

    # Phase 3: Coordinate descent

        a_interp = interpolate_seq(_best_tracker['seq'], new_n)
