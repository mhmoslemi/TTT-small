    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _score_from_autocorr(a, ac):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * ac.max() / (s * s))


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _parametric_search(n, time_limit, best_init, best_score_init):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    best_a = _norm_sum1(best_init)
    best_val = best_score_init
    i_arr = np.arange(n, dtype=np.float64)

    def _try_update(a):
        nonlocal best_a, best_val
        a_n = _norm_sum1(a)
        ac = _fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = a_n.copy()

    for p in [0.3, 0.5, 0.7, 0.9, 1.0, 1.1, 1.3, 1.5, 2.0, 3.0]:
        _try_update((i_arr + 1.0) ** p)
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        _try_update((n - i_arr) ** p)

    for alpha in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            _try_update(a)

    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            _try_update(a)

    for c1 in [0.15, 0.2, 0.3]:
        for c2 in [0.7, 0.8, 0.85]:
            for sigma in [0.05, 0.08, 0.12, 0.15, 0.2]:
                a = np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma))
                _try_update(a)

    for c in [0.3, 0.5, 0.7]:
        for s in [3, 5, 10, 20, 50, 100]:
            _try_update(1.0 / (1.0 + np.exp(-s * (x - c))))

    for freq in [1, 2, 3, 4, 5]:
        for ph in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01)
            _try_update(a)

    for rate in [0.5, 1, 2, 3, 5, 8, 15]:
        _try_update(np.exp(-rate * x))
        _try_update(np.exp(rate * (x - 1.0)))

    for c in [0.0, 0.1, 0.2, 0.3]:
        for s in [1, 2, 3, 5, 10]:
            _try_update(np.cosh(s * (x - c)))

    # Hann window variations
    for w in [0.3, 0.5, 0.7, 1.0]:
        a = w * np.sin(np.pi * x) ** 2 + (1 - w) * 0.5 * (1 + np.cos(2 * np.pi * x))
        _try_update(a)

    # Triangle
    _try_update(1.0 - 2.0 * np.abs(x - 0.5))
    # Parabolic
    _try_update(1.0 - 4.0 * (x - 0.5) ** 2)

    if time.monotonic() - t0 < time_limit * 0.7:
        a_base = best_init / best_init.sum()
        for w in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_lin = (i_arr + 1.0) / (i_arr + 1.0).sum()
            _try_update(w * a_base + (1.0 - w) * a_lin)
        x_arr = np.linspace(0, 1, n)
        for w in [0.1, 0.2, 0.3, 0.5]:
            a_gauss = np.exp(-20.0 * (x_arr - 0.5) ** 2)
            a_gauss = a_gauss / a_gauss.sum()
            _try_update(w * a_base + (1.0 - w) * a_gauss)

    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.0f}s): best J={best_val:.6f}")
    return best_a, best_val


def _lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=2500, add_per_iter=80):
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_ = a.copy()

    ac_y = _fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0
    last_print = t0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - last_print > 30:
            last_print = time.monotonic()
            ac_chk = _fast_autocorr(y)
            M_chk = ac_chk.max()
            if M_chk < best_m - 1e-15:
                best_m = M_chk
                best_a = y.copy()
            ac_x_chk = _fast_autocorr(x_)
            if ac_x_chk.max() < best_m - 1e-15:
                best_m = ac_x_chk.max()
                best_a = x_.copy()
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

        ac = _fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        ac_x = _fast_autocorr(x_)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x_.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 3:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x_ = x_new
        iter_count += 1

    ac_final = _fast_autocorr(y)
    if ac_final.max() < best_m - 1e-15:
        best_m = ac_final.max()
        best_a = y.copy()
    ac_x_final = _fast_autocorr(x_)
    if ac_x_final.max() < best_m - 1e-15:
        best_m = ac_x_final.max()
        best_a = x_.copy()

    J = 2.0 * n * best_m
    print(f"    LP-FW done ({time.monotonic()-t0:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def _smooth_mirror_descent(a0, time_limit, top_k=100, beta_start=3.0, c0=5.0, restart_every=3000):
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta_start

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = _fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = _fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert_type = int(rng.integers(4))
            if pert_type == 0:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.5, 2.5))
                pert_seq = _norm_sum1((i_arr + 1.0) ** p)
            elif pert_type == 1:
                x_arr = np.linspace(0, 1, n)
                c = float(rng.uniform(0.3, 0.7))
                sig = float(rng.uniform(0.1, 0.4))
                pert_seq = _norm_sum1(np.exp(-((x_arr - c) ** 2) / (2 * sig * sig)))
            elif pert_type == 2:
                x_arr = np.linspace(0, 1, n)
                freq = float(rng.uniform(1, 5))
                pert_seq = _norm_sum1(np.maximum(np.sin(np.pi * freq * x_arr) ** 2, 0.01))
            else:
                pert = rng.normal(0, 0.02, n) * best_a
                pert_seq = _norm_sum1(best_a + pert)

            mix_w = float(rng.uniform(0.1, 0.4))
            x_r = _norm_sum1((1 - mix_w) * best_a + mix_w * pert_seq)
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 20, 200)
            cur_beta = max(cur_beta - 0.2, 2.0)
            cur_beta = min(cur_beta, 5.0)

    return best_a, best_m


def _coordinate_descent(a0, time_limit, rng_seed=42, verbose=True):
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 2000
    fft_size = 1 << (2 * n - 2).bit_length()
    n_top = min(50, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0

        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        n_tries = min(30, n)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            if a[i] < 1e-10:
                continue

            # Score candidate destination positions
            scores = np.zeros(n)
            for k2 in top_k[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2] * ac[k2i]
                    scores[k2i - idx2] += a[idx2] * ac[k2i]

            scores[i] = np.inf
            # Prefer low-score positions
            cand_range = sorted(range(n), key=lambda j: scores[j])[:20]
            j_choices = [j for j in cand_range if j != i and a[j] < 0.5 * a[i]]
            if not j_choices:
                j_choices = [j for j in cand_range if j != i][:5]
            if not j_choices:
                continue
            j = j_choices[0]

            found = False
            for frac in [0.02, 0.04, 0.06, 0.1, 0.15, 0.25]:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = _fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if found:
                continue

        if not improved:
            stall += 1
            if stall > max_stall:
                break

            if stall % 50 == 0:
                n_moves = int(rng.integers(3, 10))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = _fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

            if verbose and stall % 500 == 0:
                J = 2.0 * n * best_m
                print(f"    CoordDesc ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def _spectral_flattening(a0, time_limit):
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    fft_size = 1 << (2 * n - 2).bit_length()
    learning_rate = 0.3
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean() + 1e-30

        beta_spec = 3.0 + t * 0.0005
        P_dev = (P - P_mean) / P_mean
        w_spec = np.exp(beta_spec * np.abs(P_dev))
        w_spec = w_spec / w_spec.sum()

        grad_freq = 2.0 * A * w_spec / max(P.max(), 1e-30)
        grad_time = np.fft.irfft(grad_freq, fft_size)[:n]

        lr = learning_rate / (1.0 + np.sqrt(t))
        a_new = a * np.exp(-lr * grad_time)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        M_new = _fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                a = _norm_sum1(best_a + rng.normal(0, 0.01, n) * best_a)
                stall = 0
                learning_rate *= 0.6
                if learning_rate < 0.01:
                    break

    return best_a, best_m


def _pairwise_optimize(a0, time_limit, rng_seed=99):
    """Try pairwise mass transfers with analytical scoring."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 3000
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0

        improved = False
        n_top = min(30, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:n_top]

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            n_pairs = min(5, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Find best destination
                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * ac[k2i]
                        scores[k2i - idx2] += a[idx2] * ac[k2i]

                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True

                cand_mask = ~in_peak
                if cand_mask.sum() < 20:
                    cand_mask = np.ones(n, dtype=bool)
                cand_mask[src] = False

                cand_scores = scores.copy()
                cand_scores[~cand_mask] = np.inf
                dst = int(np.argmin(cand_scores))

                if a[dst] > 0.7 * src_val:
                    continue

                for frac in [0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = _fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            if stall > max_stall:
                break
            if stall % 20 == 0:
                n_moves = int(rng.integers(5, 15))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.005, 0.03))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = _fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

    return best_a, best_m


def _annealing_polish(a0, time_limit, rng_seed=555, T_start=0.001, T_end=0.00001):
    """Simulated annealing with small perturbations."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    n_total = 50000
    for step in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break

        progress = step / n_total
        T = T_start * (T_end / T_start) ** progress

        # Random perturbation
        n_pert = int(rng.integers(1, 5))
        a_trial = a.copy()
        for _ in range(n_pert):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                frac = float(rng.uniform(0.001, 0.02))
                amt = a_trial[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        M_trial = _fast_autocorr(a_trial).max()
        M_curr = _fast_autocorr(a).max()
        delta = M_trial - M_curr

        if delta < 0 or np.random.random() < np.exp(-delta / (T * M_curr + 1e-15)):
            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a_trial.copy()

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = _get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = _norm_sum1(p)
            p_score = _evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        i_arr = np.arange(n_base, dtype=np.float64)
        best_seq = _norm_sum1((i_arr + 1.0) ** 0.7)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def _check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Parametric search
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        a_param, val_param = _parametric_search(n0, t1, best_seq, 2.0 * n0 * _fast_autocorr(best_seq).max())
        _check_and_update(a_param, "parametric")

    # Phase 2: LP-FW main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30 and n_best <= 7000:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_frank_wolfe(a_in, t2, solver, initial_K=400, max_K=2500, add_per_iter=80)
        _check_and_update(a_out, f"lp_n{n_best}")

    # Phase 3: Smooth mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smooth MD (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _smooth_mirror_descent(a_in, t3, top_k=100, beta_start=4.0, c0=5.0, restart_every=2500)
        _check_and_update(a_out, f"smd_n{n_best}")

    # Phase 4: LP-FW second pass with more constraints
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW pass 2 (n={n_best}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30 and n_best <= 7000:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_frank_wolfe(a_in, t4, solver, initial_K=600, max_K=3000, add_per_iter=100)
        _check_and_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Pairwise coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
