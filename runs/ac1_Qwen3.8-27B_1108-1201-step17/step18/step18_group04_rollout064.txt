"""

import numpy as np
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clamp to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def smooth_max_and_grad(a, T):
    """
    Compute smoothed max of conv(a,a) and its gradient.
    Returns (smoothed_max, gradient) where gradient[i] = sum_k w[k]*(a[k-i]+a[i-k]).
    """
    n = len(a)
    conv = fast_conv(a)
    conv_max = conv.max()
    shifted = np.clip((conv - conv_max) / T, -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / w_sum

    obj = conv_max + T * np.log(w_sum)

    # Compute gradient via FFT
    # Need fft_size >= (2n-1) + n - 1 = 3n-2 to avoid circular aliasing
    need = 3 * n - 2
    fft_size = 1 << (need - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)

    # Part 1: P1[i] = sum_k w[k]*a[i-k] (standard convolution)
    P1 = np.fft.irfft(W * A, fft_size)[:n]

    # Part 2: P2[i] = sum_k w[k]*a[k-i] (cross-correlation)
    P2 = np.fft.irfft(np.conj(W) * A, fft_size)[:n]

    grad = P1 + P2
    return obj, grad


def mirror_descent_optimize(a0, time_limit, eta_scale=0.5, seed=42):
    """
    Mirror descent with multiplicative updates and temperature annealing.
    a_new[i] = a[i] * exp(-eta * grad[i]) / normalization
    """

    a = norm_sum1(a0).copy()

    T_init = max_conv * 0.3
    T_final = max_conv * 0.003

    beta_m = 0.7  # Momentum for gradient
    m = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = evaluate_sequence([float(x) for x in a])
    stall = 0
    rng = np.random.default_rng(seed)
    eval_interval = max(200, n // 10)

    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** progress

        _, grad = smooth_max_and_grad(a, T)

        # Momentum smoothing
        m = beta_m * m + (1.0 - beta_m) * grad

        # Multiplicative update
        log_update = -eta * m
        log_update = log_update - log_update.max()
        a_new = a * np.exp(log_update)

            a_new = a.copy()

        # Track best

                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            mix = 0.75 * best_a + 0.25 * a
            noise = rng.normal(0, 0.008, n) * best_a
            a = np.maximum(mix + noise, 1e-10)
            a = a / a.sum()
            m = np.zeros(n)
            # Reduce eta slightly on restart
            eta *= 0.9
            eta = max(eta, 0.1 * n)

    return best_a, best_score


def adam_fft_optimize(a0, time_limit, lr_scale=1.0, seed=42):
    """
    Adam optimizer using FFT-based smoothed gradient.
    Learning rate scaled as lr_scale / n for appropriate step size.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    t0 = time.monotonic()

    conv = fast_conv(a)
    max_conv = conv.max()

    T_init = max_conv * 0.3
    T_final = max_conv * 0.003

    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr_base = lr_scale / n

    best_a = a.copy()
    best_score = evaluate_sequence([float(x) for x in a])
    stall = 0
    rng = np.random.default_rng(seed)
    eval_interval = max(200, n // 10)

    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** progress

        _, grad = smooth_max_and_grad(a, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_base / (1.0 + 0.005 * t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        # Track best
        if t % eval_interval == 0:
            score = evaluate_sequence([float(x) for x in a])
            if score < best_score - 1e-10:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 25:
            stall = 0
            a = 0.8 * best_a + 0.2 * a
            a += rng.normal(0, 0.005, n) * best_a
            a = np.maximum(a, 0)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 1

    return best_a, best_score


def coord_descent_polish(a0, time_limit, seed=42):
    """
    Coordinate descent polish: move mass from high-impact to low-impact positions.
    Uses top convolution peaks to identify critical coordinates.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()

    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))

        # Find top contributors to the peak
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]

        n_top = max(10, min(60, n // 80))
        top_src = idx[np.argsort(contrib)[::-1][:n_top]]

        # Compute cost vector across all top peaks
        n_peaks = min(30, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_peaks]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue

            # Find low-cost destinations
            n_cand = max(20, min(100, n // 40))
            candidates = np.argsort(costs + 1e-14 * a)[:n_cand]

                found = False

                    amt = a[src_i] * frac
                    a_trial = a.copy()

                    a_trial[dst_i] += amt

                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()

                        stall = 0
                        found = True

            # Random perturbation to escape local minima

                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    return norm_sum1(a_interp)


def make_ramp(n, power=0.8):
    """Decreasing ramp: a[i] ~ (1 - i/(n-1))^power."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = (1.0 - x) ** power
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_cosine_bump(n):
    """Raised cosine bump centered."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_gaussian(n, sigma=0.3, center=0.4):
    """Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / sigma) ** 2)
    return norm_sum1(a)


def make_asymmetric(n, left_frac=0.6, left_power=0.7, right_power=1.5):
    """Asymmetric shape: strong left side, decay right."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.zeros(n)
    mask_left = x < left_frac
    a[mask_left] = (1.0 - x[mask_left] / left_frac) ** left_power
    mask_right = ~mask_left
    a[mask_right] = ((x[mask_right] - left_frac) / (1.0 - left_frac)) ** right_power * 0.3
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_sinc_sq(n, bw=3.0, center=0.4):
    """Squared sinc function."""
    x = np.linspace(0, 1, n)
    a = np.sinc((x - center) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_power_law(n, power=0.3):
    """Power law decay."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** (-power)
    return norm_sum1(a)


def make_two_peak(n, pos1=0.2, pos2=0.7, w1=0.1, w2=0.12, ratio=0.7):
    """Two Gaussian peaks."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / w1) ** 2) + ratio * np.exp(-((x - pos2) / w2) ** 2)
    return norm_sum1(a)


def make_flat_edge(n, edge_frac=0.1):
    """Flat in the middle, decay at edges."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    decay = edge_frac * n
    for i in range(int(decay)):
        f = 1.0 - i / decay
        a[i] = f
        a[n - 1 - i] = f
    return norm_sum1(a)
