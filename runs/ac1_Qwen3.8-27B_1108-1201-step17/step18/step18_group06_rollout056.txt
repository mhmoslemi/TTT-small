- Update: a_new[i] ∝ a[i] * exp(-eta * grad[i])
- Step size: eta_t = c / sqrt(t) for some constant c

The constant c needs to be tuned. Too large and we oscillate; too small and we converge slowly.

An alternative is to use projected gradient descent:
a_new = proj_simplex(a - eta * grad)

where proj_simplex is the Euclidean projection onto the simplex {x >= 0, sum(x) = 1}.

The Euclidean projection can be computed in O(n log n) using a bisection on a threshold.

I'll use both approaches and see which works better.

Let me also consider: since the problem is scale-invariant, I can work with any normalization. I'll use sum(a) = 1 for simplicity.

One more consideration: the evaluator's convolution is np.convolve(a, a) which gives the full linear convolution of length 2n-1. I'll use FFT to compute this efficiently.

Let me write the final code:

import numpy as np
import time

def fast_convolve_square(a):
    """Compute np.convolve(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return normalize(a)

def peak_gradient(a, n_peaks=50, temperature=1.0):
    """Compute weighted gradient targeting top peaks of self-convolution."""
    n = len(a)
    conv = fast_convolve_square(a)
    M = conv.max()
    
    # Get top peaks
    k_max = min(n_peaks, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_max]
    
    # Softmax weights
    vals = conv[top_idx]
    log_vals = np.log(np.maximum(vals, 1e-30))
    exp_vals = np.exp((log_vals - log_vals.max()) / temperature)
    w = exp_vals / exp_vals.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        ki = int(k)
        wi = w[i]
        if wi < 1e-10:
            continue
        # d(conv[ki])/d(a[j]) = a[ki-j] for valid j
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair = ki - idx
        grad[idx] += wi * a[pair]
    
    return grad, M

def mirror_descent_step(a, grad, eta):
    """Mirror descent step on the simplex."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def proj_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def gradient_optimize(a0, time_limit, n_peaks=50, temperature=1.0, c0=3.0,
                      use_mirror=True, restart_every=5000, verbose=False, rng_seed=42):
    """Accelerated gradient descent for peak reduction."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_convolve_square(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    restart_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient
        grad, M_y = peak_gradient(y, n_peaks=n_peaks, temperature=temperature)
        
        # Step size
        eta = gamma / (np.sqrt(t) + 1.0)
        
        if use_mirror:
            a_new = mirror_descent_step(y, grad, eta)
        else:
            a_new = proj_simplex(y - eta * grad)
        
        x_prev = x
        x = a_new
        
        # Check improvement
        M_x = fast_convolve_square(x).max()
        if M_x < best_M - 1e-15:
            best_M = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.1, 0.02 + 0.01 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            if verbose:
                print(f"      Restart {restart_count} at t={t}, M={best_M:.8f}")
        
        if verbose and t % 10000 == 0:
            J = 2.0 * n * best_M
            print(f"      t={t}, J={J:.6f}, stall={stall}")
    
    return best_a, best_M
