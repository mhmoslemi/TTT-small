def spectral_smooth(a, alpha=0.25):
    """Compress dynamic range of |FFT(a)|."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.maximum(ratio, 0) ** (1.0 - alpha)
    A_new = mag_new * np.exp(1j * phase)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    out = (1.0 - 0.4 * alpha) * a + 0.4 * alpha * a_new
    return norm_simplex(out)

def lp_solve_direction(f, M, top_k=50, lp_time_limit=15.0):

def spectral_smooth(a, alpha=0.25):
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.maximum(ratio, 0.0) ** (1.0 - alpha)
    A_new = mag_new * np.exp(1j * phase)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    out = (1.0 - 0.4 * alpha) * a + 0.4 * alpha * a_new
    return norm_simplex(out)

            w = np.ones(k_max) / k_max

            exponents = np.maximum(exponents, -50.0)

            if ws < 1e-30:
                w = np.ones(k_max) / k_max

            if wi < 1e-12:
                continue

            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)

        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            best_obj = 2.0 * n * best_m

        if t % spectral_every == 0:
            a_spec = spectral_smooth(a, alpha=0.2)
            M_spec = fast_conv(a_spec).max()
            if M_spec < best_m:
                best_m = M_spec
                best_a = a_spec.copy()
                best_obj = 2.0 * n * best_m
                a = a_spec.copy()
                x = a.copy()

        if verbose and t % 10000 == 0:
            print("    MD t=%d (%.0fs): obj=%.6f" % (t, time.monotonic() - t0, best_obj))

    return best_a, best_obj


def local_search(a0, time_limit, n_top_peaks=8, max_stall=150, verbose=False):
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_obj = 2.0 * n * best_M
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        top_peaks = np.argsort(c)[::-1][:n_top_peaks]

        contributors = set()
        for k in top_peaks:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                contrib = a[i] * a[k_int - i]
                if contrib > c.max() * 0.25:
                    contributors.add(i)

        contrib_list = list(contributors)
        if not contrib_list:
            contrib_list = list(range(n))

        improved = False
        n_trials = max(20, min(200, len(contrib_list) * 3))
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            src = int(rng.choice(contrib_list))
            dst = int(rng.integers(n))
            if src == dst:
                continue
            frac = rng.choice([0.02, 0.05, 0.08, 0.12, 0.2])
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1.0 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a_trial.copy()
                best_obj = 2.0 * n * best_M
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.01, 0.04)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        best_obj = 2.0 * n * best_M
                        stall = 0

        if verbose and stall % 20 == 0:
            print("    LS stall=%d (%.0fs): obj=%.6f" % (stall, time.monotonic() - t0, best_obj))

    return best_a, best_obj


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    return norm_simplex(np.maximum(a, 0.0))


def make_bowl(n, sharp=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharp + 0.1
    return norm_simplex(a)


def make_dual_peak(n, p1=0.25, p2=0.75, w=0.1):
    x = np.linspace(0, 1, n)
    a = (np.exp(-((x - p1) ** 2) / (2 * w ** 2)) +
         np.exp(-((x - p2) ** 2) / (2 * w ** 2)) + 0.05)
    return norm_simplex(a)


def make_random_smooth(n, seed=1):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.1 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def _try(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = eval_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.0fs)" % (
                label, sc, len(a), time.monotonic() - t_global))

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            ps = eval_safe(parent)
            print("Parent: n=%d, score=%.6f" % (len(parent), ps))
            best_seq = norm_simplex(parent)
            best_score = ps
    except Exception:
        pass

    if best_seq is None:
        n_default = 4000
        best_seq = make_cosine(n_default)
        best_score = eval_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))

    n0 = len(best_seq)
    print("Start: n=%d, score=%.6f" % (n0, best_score))

    # Phase 1: LP optimization on parent
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    print("\n=== Phase 1: LP optimization (n=%d, %.0fs) ===" % (n0, t1))
    if t1 > 60:
        a_out, _ = lp_optimize(norm_simplex(best_seq), t1,
                               top_k=50, blend_base=0.15, lp_time=10.0,
                               spectral_every=40, verbose=True)
        _try(a_out, "lp1_n%d" % n0)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sp, _ = spectral_optimize(a_out, min(30.0, remaining * 0.2))
            _try(a_sp, "spec1_n%d" % n0)

    # Phase 2: LP with more constraints and different blend
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    print("\n=== Phase 2: LP variant (n=%d, %.0fs) ===" % (n_best, t2))
    if t2 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = lp_optimize(a_in, t2,
                               top_k=100, blend_base=0.08, lp_time=15.0,
                               spectral_every=30, verbose=False)
        _try(a_out, "lp2_n%d" % n_best)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_md, _ = mirror_descent(a_out, min(60.0, remaining * 0.4))
            _try(a_md, "md2_n%d" % n_best)

    # Phase 3: Local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    print("\n=== Phase 3: Local search (n=%d, %.0fs) ===" % (n_best, t3))
    if t3 > 15:
        a_out, _ = local_search(norm_simplex(best_seq), t3,
                                n_top_peaks=8, max_stall=200, verbose=False)
        _try(a_out, "ls_n%d" % n_best)

    # Phase 4: Alternative initializations + LP
    remaining = time_budget - (time.monotonic() - t_global)
    t4_total = min(remaining * 0.15, 200.0)
    print("\n=== Phase 4: Alt inits (n=%d, %.0fs) ===" % (n_best, t4_total))
    if t4_total > 40:
        inits = [
            (make_cosine(n_best, 1), "cos1"),
            (make_bowl(n_best, 2.0), "bowl2"),
            (make_bowl(n_best, 3.0), "bowl3"),
            (make_dual_peak(n_best, 0.2, 0.8, 0.12), "dp"),
            (make_random_smooth(n_best, seed=11), "rs1"),
            (make_random_smooth(n_best, seed=22), "rs2"),
            (make_sinc(n_best, bw=6.0), "sinc"),
        ]
        t_per = t4_total / len(inits)
        for init, lbl in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            t_avail = min(t_per, remaining)
            if t_avail < 10:
                break
            a_out, _ = lp_optimize(init, t_avail,
                                   top_k=50, blend_base=0.18, lp_time=8.0,
                                   spectral_every=25, verbose=False)
            _try(a_out, "alt_%s_n%d" % (lbl, n_best))

    # Phase 5: Length variation
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Length variation ===")
    for new_n in [n_best - 400, n_best + 400, n_best - 800, n_best + 800,
                  n_best + 1200]:
        if new_n < 1500 or new_n > 9000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print("  Trying n=%d..." % new_n)
        a_long = interpolate_seq(best_seq, new_n)
        _try(a_long, "interp_n%d" % new_n)

        t5 = min(remaining * 0.22, 130.0)
        if t5 > 20:
            a_out, _ = lp_optimize(a_long, t5,
                                   top_k=50, blend_base=0.15, lp_time=8.0,
                                   spectral_every=30, verbose=False)
            _try(a_out, "lp_n%d" % new_n)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_sp, _ = spectral_optimize(a_out, min(20.0, remaining * 0.2))
                _try(a_sp, "spec_n%d" % new_n)

    # Phase 6: Final intensive optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final intensive (n=%d) ===" % n_best)

    t6a = min(remaining * 0.35, 350.0)
    if t6a > 30:
        print("  LP final (%.0fs)..." % t6a)
        a_in = norm_simplex(best_seq)
        a_out, _ = lp_optimize(a_in, t6a,
                               top_k=80, blend_base=0.10, lp_time=12.0,
                               spectral_every=30, verbose=True)
        _try(a_out, "lp_final_n%d" % n_best)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6b = min(remaining * 0.20, 150.0)
    if t6b > 20:
        print("  MD final (%.0fs)..." % t6b)
        a_in = norm_simplex(best_seq)
        a_out, _ = mirror_descent(a_in, t6b, top_k=40,
                                  beta_init=15.0, beta_end=60.0,
                                  c0=1.5, spectral_every=25, verbose=False)
        _try(a_out, "md_final_n%d" % n_best)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6c = min(remaining * 0.20, 120.0)
    if t6c > 15:
        print("  LS final (%.0fs)..." % t6c)
        a_in = norm_simplex(best_seq)
        a_out, _ = local_search(a_in, t6c, n_top_peaks=10, max_stall=200)
        _try(a_out, "ls_final_n%d" % n_best)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6d = min(remaining * 0.25, 100.0)
    if t6d > 15:
        print("  Spectral final (%.0fs)..." % t6d)
        a_in = norm_simplex(best_seq)
        a_out, _ = spectral_optimize(a_in, t6d)
        _try(a_out, "spec_final_n%d" % n_best)

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("\nFINAL: score=%.6f, n=%d, time=%.0fs" % (
            best_score, len(best_seq), time.monotonic() - t_global))
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_cosine(n_fb)
        best_score = eval_safe(best_seq)
        print("Fallback: score=%.6f" % best_score)

    return [float(x) for x in best_seq]
