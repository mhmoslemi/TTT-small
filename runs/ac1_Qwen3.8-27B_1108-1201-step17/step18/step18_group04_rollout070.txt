"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based active-set steepest descent with quadratic line search
2. Nesterov mirror descent with refined softmax subgradient (fallback)

import numpy as np
import time
from scipy.optimize import linprog

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def grad_conv_k(a, k):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    paired = k - idx
    g[idx] += a[paired]
    mask = paired != idx
    if mask.any():
        g[paired[mask]] += a[idx[mask]]
    return g


def compute_subgrad(a, conv, top_k=30, temperature=0.003):
    n = len(a)
    conv_max = conv.max()
    threshold = conv_max - temperature * 8.0
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        idx = np.arange(lo, hi + 1)
        paired = k_int - idx
        g_local = np.zeros(n)
        g_local[idx] += a[paired]
        mask = paired != idx
        if mask.any():
            g_local[paired[mask]] += a[idx[mask]]
        grad += wi * g_local
    return grad


def lp_direction(a, active_peaks, conv, max_bound=0.1):
    n = len(a)
    m = len(active_peaks)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in active_peaks:
        v_k = grad_conv_k(a, int(k))
        row = np.zeros(n + 1)
        row[:n] = v_k
        row[n] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(0.0)
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(a[i])
        row2 = np.zeros(n + 1)
        row2[i] = 1.0
        A_ub_list.append(row2)
        b_ub_list.append(max_bound)
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'presolve': True})
    if res.success:
        d = res.x[:n]
        theta = res.x[n]
        return d, theta
    return None, None


def quadratic_line_search(a, d, active_peaks, conv, t_max=2.0):
    n = len(a)
    candidates = [0.0]
    for k in active_peaks:
        k_i = int(k)
        conv_k = conv[k_i]
        v_k = grad_conv_k(a, k_i)
        b = v_k @ d
        q = d @ grad_conv_k(np.where(d + a > 0, d, 0) + a, k_i)
        # q = d^T C_k d, computed as sum of d[i]*d[k-i] type terms
        # Actually q is the second-order term: d^T C_k d
        # C_k has entries (i,j) = 1 if j = k-i
        # So d^T C_k d = sum_i d[i]*d[k-i] (for valid indices)
        lo = max(0, k_i - n + 1)
        hi = min(n - 1, k_i)
        idx = np.arange(lo, hi + 1)
        paired = k_i - idx
        q_val = np.sum(d[idx] * d[paired])
        # Vertex of this quadratic: t = -b / (2*q_val) if q_val > 0
        if q_val > 1e-15:
            t_vertex = -b / (2.0 * q_val)
            if 0 < t_vertex < t_max:
                candidates.append(t_vertex)
    for i in range(len(active_peaks)):
        for j in range(i + 1, len(active_peaks)):
            k_i = int(active_peaks[i])
            k_j = int(active_peaks[j])
            c1 = conv[k_i]
            c2 = conv[k_j]
            b1 = grad_conv_k(a, k_i) @ d
            b2 = grad_conv_k(a, k_j) @ d
            lo1 = max(0, k_i - n + 1)
            hi1 = min(n - 1, k_i)
            idx1 = np.arange(lo1, hi1 + 1)
            q1 = np.sum(d[idx1] * d[k_i - idx1])
            lo2 = max(0, k_j - n + 1)
            hi2 = min(n - 1, k_j)
            idx2 = np.arange(lo2, hi2 + 1)
            q2 = np.sum(d[idx2] * d[k_j - idx2])
            dq = q1 - q2
            db = b1 - b2
            dc = c1 - c2
            if abs(dq) < 1e-20:
                if abs(db) > 1e-20:
                    t_cross = -dc / db
                    if 0 < t_cross < t_max:
                        candidates.append(t_cross)
            else:
                disc = db * db - 4.0 * dq * dc
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-db + sqrt_disc) / (2.0 * dq)
                    t2 = (-db - sqrt_disc) / (2.0 * dq)
                    for tc in [t1, t2]:
                        if 0 < tc < t_max:
                            candidates.append(tc)
    candidates.append(t_max)
    best_t = 0.0
    best_val = fast_conv(a).max()
    for t in candidates:
        if t <= 0:
            continue
        a_trial = a + t * d
        if np.any(a_trial < -1e-10):
            continue
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        M = fast_conv(a_trial).max()
        if M < best_val:
            best_val = M
            best_t = t
    return best_t, best_val


def lp_optimize(a0, time_limit, n_active_max=4, max_bound=0.05, seed=42):

    a = norm_sum1(a0)
    best_a = a.copy()
    conv = fast_conv(a)
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 200:
            break
        active = np.argsort(conv)[::-1][:n_active_max]
        max_val = convmax = conv[active[0]]
        # Check if only one peak is truly at max
        if len(active) > 1 and conv[active[1]] > max_val * 0.999:
            active_peaks = active
        else:
            # Include peaks within 1e-6 relative
            threshold = max_val - 1e-6

                active_all = active[:1]
            elif len(active_all) > n_active_max:
                active_all = active_all[np.argsort(conv[active_all])[::-1]][:n_active_max]
            active_peaks = active_all

        d, theta = lp_direction(a, active_peaks, conv, max_bound=max_bound)
        if d is None:
            stall += 1
            # Perturb
            pert = rng.normal(0, 0.01, n) * a
            a = np.maximum(a + pert, 0)
            a = a / a.sum()
            conv = fast_conv(a)
            continue

        t_step, new_m = quadratic_line_search(a, d, active_peaks, conv, t_max=2.0)
        if t_step > 1e-6:
            a_new = np.maximum(a + t_step * d, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                if new_m < best_m - 1e-14:
                    best_m = new_m

                    conv = fast_conv(a)
                    iteration += 1
                else:
                    # Try smaller step
                    for frac in [0.5, 0.3, 0.1]:
                        a_try = np.maximum(a + frac * t_step * d, 0.0)
                        a_try = a_try / a_try.sum()
                        M_try = fast_conv(a_try).max()
                        if M_try < best_m - 1e-14:
                            best_m = M_try
                            best_a = a_try.copy()
                            a = a_try
                            conv = fast_conv(a)
                            stall = 0
                            iteration += 1
                            break
                    else:
                        stall += 1
            else:
                stall += 1
        else:
            stall += 1
            # Add more active peaks or perturb
            n_active_max = min(n_active_max + 1, 8)
            if stall > 10:
                pert = rng.normal(0, 0.005, n) * a
                a = np.maximum(a + pert, 0)
                a = a / a.sum()
                conv = fast_conv(a)
                n_active_max = 4
        if iteration % 500 == 0 and iteration > 0:
            print(f"    LP iter {iteration}: best_m={best_m:.10f}, stall={stall}")
    return best_a, best_m


def amd_optimize(a0, time_limit, top_k_init=25, temp_init=0.003, c0_init=3.0,
                 restart_every=5000, seed=42, verbose=False):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    conv = fast_conv(x)
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    top_k = top_k_init
    temperature = temp_init
    c0 = c0_init
    stall_count = 0
    restart_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv_y = fast_conv(y)
        g = compute_subgrad(y, conv_y, top_k=top_k, temperature=temperature)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.06, 0.015 + 0.005 * min(restart_count, 8))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            top_k = min(top_k + 5, 80)
            temperature = max(temperature - 0.0004, 0.0008)
            c0 = 3.0 * (1.0 + 0.04 * min(restart_count, 10))
            restart_every = max(3000, restart_every + 500)

    return best_x, best_m


def coord_descent_polish(a0, time_limit, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    conv = fast_conv(a)
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]

        n_top_src = max(5, min(25, n // 200))
        top_src = idx[np.argsort(contrib)[::-1][:n_top_src]]

        n_top_peaks = min(15, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_top_peaks]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            candidates = np.argsort(costs + 1e-10 * a)[:max(10, n // 30)]
            candidates = [int(c) for c in candidates if int(c) != src_i]
            for dst_i in candidates[:10]:
                found = False
                for frac in [0.03, 0.07, 0.15, 0.3, 0.5, 0.75]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_increasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((x + 1) ** power)


def make_mixed(n, pos1=0.2, pos2=0.65, width=0.1, ratio=1.0):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / (width * 1.5)) ** 2)
    return norm_sum1(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_centered_power(n, power=0.5):
    x_c = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm_sum1(x_c ** (-power))


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_exponential_decay(n, rate=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm_sum1(np.exp(-rate * x))


def make_two_block(n, frac=0.5):
    a = np.zeros(n)
    a[:int(frac * n)] = 1.0
    return norm_sum1(a)

            t_avail = min(t_per, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 5:
                break
            lbl = unique_labels[idx]
            a_out, _ = amd_optimize(unique_inits[idx], t_avail,
                                    top_k_init=25, temp_init=0.003, c0_init=3.0,
                                    restart_every=4000, seed=100 + idx)
            try_update(a_out, f"amd_{lbl}")

    # Phase 3: LP refinement on best so far
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Refine (n={len(_best_tracker['seq'])}) ===")
    t_phase3 = min(remaining * 0.25, 400.0)
    if t_phase3 > 30:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = lp_optimize(a_in, t_phase3, n_active_max=4, max_bound=0.03, seed=777)
        try_update(a_out, "lp_refine")

    # Phase 4: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord Descent ===")
    t_phase4 = min(remaining * 0.12, 150.0)
    if t_phase4 > 15:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = coord_descent_polish(a_in, t_phase4, seed=42)
        try_update(a_out, "cd_polish")

    # Phase 5: Multi-n search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 5: Multi-n (base={n_best}) ===")
    candidate_ns = []
    for delta in [-300, -500, 300, 500, 800, -800, 1000, -1200, 1200]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            candidate_ns.append(new_n)

    for new_n in candidate_ns[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}")
        a_interp = interpolate_seq(_best_tracker['seq'], new_n)
        try_update(a_interp, f"intp{new_n}")
        t5a = min(remaining * 0.25, 100.0)
        if t5a > 15:
            a_out, _ = lp_optimize(a_interp, t5a, n_active_max=3, max_bound=0.05, seed=500 + new_n % 100)
            try_update(a_out, f"lp_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.10, 60.0)
            if t5b > 10:
                a_out, _ = coord_descent_polish(a_interp, t5b, seed=900 + new_n % 100)
                try_update(a_out, f"cd_n{new_n}")

    # Phase 6: Final aggressive LP + AMD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 6: Final (n={n_best}) ===")
    if remaining > 60:
        t6a = min(remaining * 0.50, 200.0)
        if t6a > 20:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = lp_optimize(a_in, t6a, n_active_max=5, max_bound=0.02, seed=2024)
            try_update(a_out, "lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.35, 100.0)
            if t6b > 10:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = amd_optimize(a_in, t6b, top_k_init=50, temp_init=0.0015,
                                        c0_init=2.0, restart_every=8000, seed=31415)
                try_update(a_out, "amd_final")

    # Final evaluation
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
