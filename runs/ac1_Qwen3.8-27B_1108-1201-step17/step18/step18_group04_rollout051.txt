"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B optimization of log-sum-exp smoothed objective in log-space (fast local optimizer)
2. Nesterov-accelerated mirror descent (AMD) for robust refinement
3. Direct peak-reduction via targeted mass redistribution
4. Multi-initialization with structural diversity (spectral, difference-set inspired, random)
5. Adaptive temperature (T) scheduling for the smoothed objective
6. Multi-n search with interpolation
7. Coordinate descent final polish
"""

import numpy as np
import time
from scipy.optimize import minimize

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1 with non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def safe_softmax(u):
    """Numerically stable softmax."""
    u_sh = u - u.max()
    e = np.exp(u_sh)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(u)) / len(u)
    return e / s


def smooth_obj_grad(u, T):
    """
    Compute (1/T)*log(sum exp(T*conv_k(u))) and gradient w.r.t. u.
    where u parameterizes a = softmax(u).
    """
    a = safe_softmax(u)
    n = len(a)
    conv = fast_conv(a)

    # Log-sum-exp for numerical stability
    conv_max = conv.max()
    T_shifted = T * (conv - conv_max)
    T_shifted = np.clip(T_shifted, -500, 500)
    ms = T_shifted.max()
    exp_vals = np.exp(T_shifted - ms)
    log_sum_exp = (conv_max + ms + np.log(exp_vals.sum()))  # this is T * (1/T * log(sum exp(T*conv)))
    obj_val = log_sum_exp / T

    # Weights p_k
    p = exp_vals / exp_vals.sum()

    # Gradient: sum_k p_k * d(conv_k)/du
    # d(conv_k)/du[m] = 2*a[m]*(a[k-m] - conv_k) where a[k-m]=0 if out of range
    grad = np.zeros(n)

    # Only process significant components
    thresh = 1e-8
    active = np.where(p > thresh)[0]

    for k in active:
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        # a[k-m] for m in idx
        a_km = a[k_int - idx]
        # Contribution to grad for m in idx: p[k] * 2 * a[m] * (a[k-m] - conv[k])
        # Contribution to grad for m NOT in idx: p[k] * 2 * a[m] * (0 - conv[k]) = -p[k]*2*a[m]*conv[k]

        # Efficient computation:
        # grad[m] += p[k] * 2 * a[m] * (a[k-m] - conv[k]) for all m
        # = p[k] * 2 * a[m] * a[k-m] (for m in idx)  -  p[k] * 2 * a[m] * conv[k] (for ALL m)

        # The second term is a constant (per k) times a[m], so:
        grad -= p[k] * 2.0 * conv[k_int] * a
        grad[idx] += p[k] * 2.0 * a[idx] * a_km

    return obj_val, grad


def lbfgs_run(a_init, T, max_iter=300, maxfun=500):
    """Run L-BFGS-B on the smoothed objective. Returns optimized a."""
    u0 = np.log(a_init + 1e-30)
    u0 = u0 - u0.mean()  # center for numerical stability

    def fun_and_grad(u):
        val, grad = smooth_obj_grad(u, T)
        if not np.isfinite(val):
            return 1e10, np.zeros_like(u)
        return val, grad

    try:
        result = minimize(
            fun_and_grad, u0, jac=True,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': maxfun, 'iprint': 0,
                     'gtol': 1e-8}
        )
        a_opt = safe_softmax(result.x)
        return a_opt
    except Exception:
        return a_init


def amd_optimize(a0, time_limit, top_k_init=30, temp_init=0.003, c0_init=2.5,
                 restart_every=3000, seed=42):
    """Nesterov-accelerated mirror descent with multiplicative updates."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    top_k = top_k_init
    temperature = temp_init
    c0 = c0_init
    stall_count = 0
    restart_count = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv_y = fast_conv(y)
        conv_max = conv_y.max()
        threshold = conv_max - temperature * 8.0
        active = np.where(conv_y >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_y)])
        if len(active) > top_k:
            order = np.argsort(conv_y[active])[::-1][:top_k]
            active = active[order]
        w = np.exp((conv_y[active] - conv_max) / temperature)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(active)) / len(active)
        else:
            w = w / w_sum

        g = np.zeros(n)
        for wi, k in zip(w, active):
            k_int = int(k)
            lo = max(0, k_int - n + 1)
            hi = min(n - 1, k_int)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * wi * y[k_int - idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        a_new = y * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.08, 0.02 + 0.008 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            top_k = min(top_k + 3, 80)
            temperature = max(temperature * 0.93, 0.0008)
            c0 *= 1.03

    return best_x, best_m
