def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def lp_frank_wolfe_twotier(a0, time_limit, solver, verbose=True,

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    K2 = top_K2 + top_K1
    iter_count = 0
    stall = 0

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K2_sel = min(K2, 2 * n - 1)

        top_idx = np.argsort(ac)[::-1][:K2_sel]
        tier1_idx = top_idx[:K1_sel]
        tier2_idx = top_idx[K1_sel:K2_sel]

        T_sel = np.zeros((K2_sel, n))
        for ii, k in enumerate(top_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)

        b_var = cp.Variable(n, nonneg=True)

            constraints.append(T_sel[K1_sel:] @ b_var <= M_curr * (1.0 + eps))

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K1 = min(K1 + K1_step, 2 * n - 1)
            K2 = min(K2 + K2_step, 2 * n - 1)

            K2 = min(K2 + K2_step, 2 * n - 1)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:

        g_dir = b_val / b_sum

        # Golden section line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = fast_autocorr((1 - best_alpha) * a + best_alpha * g_dir).max()

        lo, hi = 0.0, 1.0

        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = fast_autocorr((1 - c1) * a + c1 * g_dir).max()

                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = fast_autocorr((1 - c1) * a + c1 * g_dir).max()
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = fast_autocorr((1 - c2) * a + c2 * g_dir).max()
        best_alpha = (lo + hi) / 2
        best_trial_m = min(f1, f2)

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()
            stall = 0
            if stall == 0 and iter_count > 0:
                eps = max(eps * 0.95, 0.005)
        else:
            stall += 1
            if stall > 5:
                K1 = min(K1 + K1_step, 2 * n - 1)
                K2 = min(K2 + K2_step, 2 * n - 1)
                eps = min(eps * 1.2, 0.15)
            if stall > 30:
                K1 = top_K1
                K2 = K1 + top_K2
                eps = 0.02
                pert = np.random.default_rng(iter_count).normal(0, 0.01, n) * best_a
                a = np.maximum(best_a + pert, 0.0)
                a = a / a.sum()
                stall = 0

        iter_count += 1

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP2T iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K1={K1_sel}, K2={K2_sel}, eps={eps:.4f}, alpha={best_alpha:.4f}, stall={stall}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=True, tau_init=20.0, tau_final=500.0,
                              eta_init=0.5, rng_seed=100):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross

        g_tangent = g - g.mean()

        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            a = a_new
            stall += 1
            if stall > 50:
                eta *= 0.8
            if stall > 150:
                pert = rng.normal(0, 0.003, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a = a_p / a_p.sum()
                stall = 80

        total_iters += 1

        if verbose and total_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    n_top = 30

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:15]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved_this_round = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.01, 0.005, 0.02, 0.03, 0.05, 0.08, 0.12]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved_this_round = True
                        found = True
                        break

            if stall > 50 and a[src] > 1e-10:
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        if not improved_this_round:
            stall += 1
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def active_set_reduction(a0, time_limit, solver, verbose=True, n_rounds=20):
    """Iteratively tighten constraints on active lags to reduce the peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Find initial active set
    ac = fast_autocorr(a)
    M = ac.max()
    active = np.where(ac >= M * 0.95)[0]

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Try to reduce M by solving LP with slightly tighter constraints
        for tight_factor in [0.99, 0.98, 0.97, 0.96, 0.95]:
            M_target = M * tight_factor
            K_sel = min(len(active), 500)
            active_sel = active[np.argsort(ac[active])[::-1][:K_sel]]

            T_sel = np.zeros((K_sel, n))
            for ii, k in enumerate(active_sel):
                ki = int(k)
                j_start = max(0, ki - n + 1)
                j_end = min(n, ki + 1)
                for j in range(j_start, j_end):
                    T_sel[ii, j] = a[ki - j]

            b_var = cp.Variable(n, nonneg=True)
            constraints = [T_sel @ b_var <= M_target]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue

            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            b_sum = b_val.sum()
            if b_sum < 1e-15:
                continue
            g_dir = b_val / b_sum

            # Line search
            best_alpha = 0.1
            best_trial_m = M
            for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                a_trial = (1 - alpha) * a + alpha * g_dir
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_alpha = alpha

            a = (1 - best_alpha) * a + best_alpha * g_dir
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M = fast_autocorr(a).max()

            if M < best_m:
                best_m = M
                best_a = a.copy()

            # Update active set
            ac = fast_autocorr(a)
            active = np.where(ac >= M * 0.9)[0]
            break

        if verbose and rnd % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ASR round={rnd} ({elapsed:.0f}s): J={J:.6f}, M={M:.2e}, n_active={len(active)}")

    return best_a, best_m
