    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_geometric(n, ratio=0.95):
    """Exponentially decaying sequence."""
    x = np.arange(n, dtype=np.float64)
    return norm_sum1(ratio ** x)


def make_geometric_asc(n, ratio=0.95):
    """Exponentially increasing sequence."""
    x = np.arange(n, dtype=np.float64)
    return norm_sum1(ratio ** (n - 1 - x))


def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_increasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((x + 1) ** power)


def make_mixed(n, pos1=0.2, pos2=0.7, width=0.1, ratio=1.0):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / (width * 1.3)) ** 2)
    return norm_sum1(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_centered_power(n, power=0.5):
    x_c = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm_sum1(x_c ** (-power))


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.1 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_two_block(n, frac=0.5):
    a = np.zeros(n)
    a[:int(frac * n)] = 1.0
    return norm_sum1(a)


def make_asymmetric_bump(n, peak_pos=0.3, width=0.15):
    """Asymmetric bump - high at one end, decays."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - peak_pos) / width) ** 2)
    return norm_sum1(a)


def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 5000
        _best_tracker['seq'] = make_decreasing(n_base, 0.4)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Default init: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    n0 = len(_best_tracker['seq'])
    print(f"Starting: n={n0}, score={_best_tracker['score']:.6f}")

    # Phase 1: AMD on diverse inits
    print(f"\n=== Phase 1: AMD diverse inits (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 600.0)

    if t_phase1 > 30:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_decreasing(n0, 0.3)); labels.append("dec03")
        inits.append(make_decreasing(n0, 0.6)); labels.append("dec06")

        inits.append(make_mixed(n0, 0.1, 0.8, 0.12, 0.8)); labels.append("mixB")
        inits.append(make_sinc(n0, 5.0)); labels.append("sinc5")
        inits.append(make_sinc(n0, 8.0)); labels.append("sinc8")
        inits.append(make_cosine(n0, 1)); labels.append("cos1")
        inits.append(make_cosine(n0, 3)); labels.append("cos3")
        inits.append(make_centered_power(n0, 0.4)); labels.append("cctr04")
        inits.append(make_centered_power(n0, 0.8)); labels.append("cctr08")

        inits.append(make_random_smooth(n0, seed=99, n_freq=12)); labels.append("rnd2")
        inits.append(make_two_block(n0, 0.4)); labels.append("blk40")
        inits.append(make_two_block(n0, 0.7)); labels.append("blk70")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 20)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(6.0, t_phase1 / n_inits)
        for idx in range(n_inits):
            if time.monotonic() - t_global > time_budget - 20:

            t_avail = min(t_per, time_budget - 20 - (time.monotonic() - t_global))

            print(f"  AMD '{lbl}' t={t_avail:.0f}s")
            a_out, _ = amd_optimize(unique_inits[idx], t_avail,
                                    top_k_init=15, temp_init=0.002, c0_init=3.0,
                                    restart_every=2500, seed=200 + idx, verbose=False)

    candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_best))

    for new_n in candidate_ns[:4]:
        remaining = time_budget - (time.monotonic() - t_global)

        print(f"  Trying n={new_n}")
        a_interp = interpolate_seq(_best_tracker['seq'], new_n)
