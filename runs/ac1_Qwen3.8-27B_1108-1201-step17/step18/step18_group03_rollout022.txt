def interp_resample(a, new_n):
    """Resample sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        cons = [cp.sum(g_var) == 1, C @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), cons)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed += 1
            if failed > 5:
                # Reduce constraints
                clist = sorted(constraint_set)[:min(200, len(constraint_set))]
                constraint_set = set(clist)
                failed = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Add violated constraints
        fft_size = 1 << (2*n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2*n-1]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW2 it={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    LP-FW2 done ({elapsed:.0f}s): J={2*n*best_m:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_descent_v2(a0, time_limit, rng_seed=42, verbose=True):
    """Greedy coordinate descent with 1D ternary search for optimal value."""
    n = len(a0)
    a = norm1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Random perturbation
                n_pert = int(rng.integers(5, 20))
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.01, 0.1))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_check = fast_conv(a).max()
                        if M_check < best_m:
                            best_m = M_check
                            best_a = a.copy()
                            stall = 0
                continue
        
        # Find the peak lag
        peak_k = int(c.argmax())
        
        # For coordinates that contribute to the peak, try to reduce
        i_s = max(0, peak_k - n + 1)
        i_e = min(n, peak_k + 1)
        if i_s >= i_e:
            stall += 1
            continue
        
        # Get contributing indices sorted by contribution
        idx = np.arange(i_s, i_e)
        contrib = a[idx] * a[peak_k - idx]
        
        # Try to optimize the top contributors
        n_try = min(10, len(idx))
        top_contrib = np.argsort(contrib)[::-1][:n_try]
        
        improved = False
        for pi in top_contrib:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[pi])
            
            # Ternary search for optimal a[src]
            # We redistribute the removed mass to random non-peak positions
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try reducing a[src] and adding to a "good" position
            # Find a position that doesn't contribute much to top lags
            top_lags = np.argsort(c)[::-1][:15]
            
            # Score positions
            scores = np.zeros(n)
            for k2 in top_lags:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += c[k2i] * a[k2i - idx2]
            
            scores[src] = np.inf
            dst = int(np.argmin(scores))
            
            if a[dst] > 0.8 * src_val:
                continue
            
            # Try different transfer amounts
            for frac in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_conv(a_trial).max()
                if M_trial < M - 1e-16:
                    a = a_trial
                    M = M_trial
                    if M < best_m - 1e-15:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break
            if improved:
                stall = 0
                break
        
        if not improved:
            stall += 1
            if verbose and stall % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    CoordD2 ({elapsed:.0f}s): J={2*n*best_m:.6f}, stall={stall}")
    
    return best_a, best_m
