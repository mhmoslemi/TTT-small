The objective in log-space: Let `a[i] = exp(x[i]) / sum(exp(x))` (softmax parameterization). Then:
- `c[k] = (a*a)[k]`

- We minimize `f(x)` over `x ∈ R^n` (unconstrained in x, since the softmax handles the simplex constraint)

The gradient: `df/dx[i] = S * df/da[i] - a[i] * sum_j a[j] * df/da[j]` (where S = 1 for normalized a, so it simplifies). Actually for the softmax parameterization:
`da[i]/dx[j] = a[i] * (delta_ij - a[j])`
So `df/dx[j] = sum_i (df/da[i]) * a[i] * (delta_ij - a[j]) = a[j] * (df/da[j] - sum_i df/da[i] * a[i])`

For `f = max_k c[k]` with active set `k*`:
`df/da[i] = a[k*-i]` (for the hard max)
So `df/dx[j] = a[j] * (a[k*-j] - sum_i a[i] * a[k*-i])`

Note: `sum_i a[i] * a[k*-i] = c[k*] = M` (the max value)

So `df/dx[j] = a[j] * (a[k*-j] - M)`

This is a very nice form! The gradient in log-space is `a[j] * (a[k*-j] - M)`.

For the LSE smoothing: `df/da[i] = sum_k w_k * a[k-i]` where `w_k = softmax_k(beta * c)`
And `df/dx[j] = a[j] * (sum_k w_k * a[k-j] - sum_i a[i] * sum_k w_k * a[k-i])`
= `a[j] * (sum_k w_k * a[k-j] - sum_k w_k * c[k])`
= `a[j] * (sum_k w_k * a[k-j] - weighted_avg_c)`

This is still computable efficiently.

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def obj_from_conv(a, c):
    """Compute 2n * max(c) assuming a is on the simplex (sum=1)."""
    n = len(a)
    return 2.0 * n * c.max()


def lse_gradient(a, c, beta, top_k=None):
    """Compute gradient of LSE(beta*c)/beta in a-space.
    
    Returns grad[i] = sum_k w_k * a[k-i] where w_k is softmax(beta*c)/sum(softmax(beta*c))
    If top_k is given, only consider top-k peaks.
    """
    n = len(a)
    if top_k is not None and top_k < len(c):
        # Zero out non-top-k entries
        c_work = np.zeros_like(c)
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        c_work[top_idx] = c[top_idx]
    else:
        c_work = c.copy()
    
    # Compute softmax weights
    max_c = c_work.max()
    if max_c < 1e-30:
        w = np.ones(len(c_work)) / len(c_work)
    else:
        scaled = beta * (c_work - max_c)
        scaled = np.clip(scaled, -50, 50)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(c_work)) / len(c_work)
        else:
            w = w / w_sum
    
    # Compute weighted convolution: grad = w * a (but w is over k, a is over i)
    # grad[i] = sum_k w[k] * a[k-i] for valid (i,k)
    # This is the correlation of w with a, which is w reversed convolved with a
    # Actually: grad[i] = sum_k w[k] * a[k-i] = (reversed_w * a)[i + n - 1]... 
    # Let me think more carefully.
    # (a*a)[k] = sum_i a[i] * a[k-i]
    # d/d a[i] (a*a)[k] = a[k-i] + a[i-k]... wait no.
    # Actually d/d a[i] sum_j a[j] a[k-j] = a[k-i] (for valid k-i)
    # So d/d a[i] LSE = sum_k w[k] * a[k-i]
    
    # grad[i] = sum_k w[k] * a[k-i]
    # This is the cross-correlation of w and a, evaluated at specific lags.
    # For each i: grad[i] = sum_k w[k] * a[k-i] where k-i must be in [0, n-1]
    # So k must be in [i, i+n-1] ∩ [0, 2n-2]
    
    # This can be computed as: grad = correlate(w, a) but with the right definition
    # Let's use FFT: 
    # grad[i] = sum_k w[k] * a[k-i] = sum_j w[i+j] * a[j] (letting j = k-i)
    # = (w_windowed * a)[i] where w_windowed is w shifted
    
    # Simpler: use the FFT-based correlation
    # grad = irfft(FFT(w, N) * conj(FFT(a, N))) -- this gives circular correlation
    # But we need linear correlation with specific bounds.
    
    # Actually, let's just use the direct formula via FFT:
    # grad[i] = sum_{k} w[k] * a[k-i] for k-i in [0,n-1], i in [0,n-1]
    # = sum_{j=0}^{n-1} w[i+j] * a[j]  (substituting j = k-i, k = i+j)
    # This is the "cross-correlation" of w (length 2n-1) with a (length n)
    
    fft_size = 1 << (2*n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # correlation[i] = sum_j w[i+j] * a[j] = ifft(conj(W) * A)[i + n - 1] ... 
    # Hmm, let me use a different approach.
    # 
    # grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is the convolution of w with reversed a, evaluated at specific positions.
    # Let ra = a reversed: ra[j] = a[n-1-j]
    # Then sum_j w[i+j] * a[j] = sum_j w[i+j] * ra[n-1-j] = (w * ra)[i + n - 1]
    # where * is linear convolution.
    
    ra = a[::-1]
    fft_size2 = 1 << (3*n - 2).bit_length()  # w has length 2n-1, ra has length n, conv has length 3n-2
    W = np.fft.rfft(w, fft_size2)
    RA = np.fft.rfft(ra, fft_size2)
    conv_wr = np.fft.irfft(W * RA, fft_size2)
    grad = conv_wr[n-1 : 2*n-1]  # extract positions i=0,...,n-1 → index i+n-1 in conv
    
    # Wait, let me verify: (w * ra)[m] = sum_j w[j] * ra[m-j]
    # We want grad[i] = sum_j w[i+j] * a[j] = sum_j w[i+j] * ra[n-1-j]
    # Let l = n-1-j, so j = n-1-l:
    # grad[i] = sum_l w[i+n-1-l] * ra[l] = (w * ra)[i+n-1]
    # where (w * ra)[m] = sum_l w[m-l] * ra[l]
    # Yes! So grad[i] = (w * ra)[i+n-1]
    
    return grad


def nesterov_mirror_descent(a0, time_limit, verbose=False):

    best_c = fast_conv(a)

    beta_end = 5.0    # final temperature

    # Adaptive step size
    lr = 0.5  # initial learning rate
    lr_decay = 0.9999  # geometric decay per step
    lr_min = 0.01
    
    # Nesterov momentum
    mu = 0.9  # momentum parameter
    
    iter_count = 0
    last_report = 0
    
    while time.monotonic() - t0 < time_limit:

        progress = min(1.0, iter_count / max(1, int(time_limit / (time.monotonic() - t0 + 1e-10) * 10)))

        a_y = np.exp(y_shifted)
        s_y = a_y.sum()
        if s_y < 1e-15:
            a_y = np.ones(n) / n
        else:
            a_y = a_y / s_y
        a_y = np.maximum(a_y, 0.0)
        a_y = a_y / a_y.sum()

        grad_x = a_y * (g - mean_g)

        if iter_count > 10:

        lr = max(lr_min, lr * lr_decay)

        x_new = y - lr * grad_x
        x_prev = x
        x = x_new
        
        # Track best
        c_new = fast_conv(normalize_simplex(np.exp(x - x.max())))
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_obj = 2.0 * n * M_new

def spectral_project(a, strength=0.3):
    """Project towards flat power spectrum while maintaining non-negativity."""
    n = len(a)
    fft_size = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Target: flat magnitude
    mag_target = np.ones_like(mag) * mag.mean()
    
    # Blended magnitude
    mag_new = (1 - strength) * mag + strength * mag_target
    
    # Reconstruct
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    
    # Blend with original for stability
    return (1 - strength * 0.5) * a + strength * 0.5 * a_new


def spectral_project_aggressive(a, strength=0.5):
    """More aggressive spectral flattening with amplitude compression."""
    n = len(a)
    fft_size = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()
    
    # Compress dynamic range aggressively
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    
    return norm_simplex(a_new)


def gradient_descent_fixedpoint(a0, time_limit, verbose=False):
    """Fixed-point iteration for the KKT conditions.
    
    At optimality: a[i] ∝ exp(-eta * sum_k w_k * a[k-i])
    where w_k = softmax(beta * c)
    Iterate: a_new[i] ∝ a[i] * exp(-eta * grad[i])
    This is multiplicative weight updates.
    """
    n = len(a0)
    t0 = time.monotonic()
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    eta = 1.0
    beta = 30.0
    top_k = 10
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        # Compute gradient (weighted)
        g = lse_gradient(a, c, beta, top_k=top_k)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - eta * g
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Damping
        a_blend = 0.7 * a + 0.3 * a_new
        a_blend = normalize_simplex(a_blend)
        
        c_new = fast_conv(a_blend)
        M_new = c_new.max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_blend.copy()
            a = a_blend
            eta = min(eta * 1.01, 2.0)
        else:
            a = normalize_simplex(0.5 * a + 0.5 * best_a)
            eta = max(eta * 0.95, 0.1)
            
            if eta < 0.1:
                break
        
        # Anneal beta
        if it % 1000 == 0:
            beta = max(5.0, beta * 0.98)
            top_k = max(3, int(top_k * 0.95))
    
    return best_a, 2.0 * n * best_m


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent: try moving mass between positions to reduce peak."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        
        # Find top peaks
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            # Find position with highest contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try moving to random positions
            for attempt in range(10):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] = max(0, a_trial[src] - amt)
                    a_trial[dst] += amt
                    a_trial = normalize_simplex(a_trial)
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = normalize_simplex(a)
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, 2.0 * n * best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def norm_simplex(a):
    """Alias for normalize_simplex."""
    return normalize_simplex(a)

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def log_softmax(x):
    """Compute log(softmax(x)) stably."""
    xm = x - x.max()
    log_sum = np.log(np.sum(np.exp(xm)))
    return xm - log_sum


def lse_gradient_a_space(a, c, beta, top_k=None):
    """Compute gradient of LSE(beta*c)/beta with respect to a-space.

    n = len(a)
    c_work = c.copy()
    if top_k is not None and top_k < len(c_work):
        top_idx = np.argpartition(c_work, -top_k)[-top_k:]
        mask = np.zeros(len(c_work), dtype=bool)
        mask[top_idx] = True
        c_work[~mask] = 0.0
    
    max_c = c_work.max()
    if max_c < 1e-30:
        w = np.ones(len(c_work)) / len(c_work)
    else:
        scaled = beta * (c_work - max_c)
        scaled = np.clip(scaled, -60.0, 60.0)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(c_work)) / len(c_work)
        else:
            w = w / w_sum

            a_sp = spectral_project(a_curr, strength=0.25)
            c_sp = fast_conv(a_sp)
            if c_sp.max() < best_m:
                best_m = c_sp.max()
                best_obj = 2.0 * n * best_m
                best_a = a_sp.copy()
                x = np.log(np.maximum(a_sp, 1e-30))
                x_prev = x.copy()
                stall_count = 0
        
        # Restart mechanism if stalled
        if stall_count > stall_threshold:
            a_restart = normalize_simplex(best_a + np.random.default_rng(iter_count).normal(0, 0.005, n))
            x = np.log(np.maximum(a_restart, 1e-30))
            x_prev = x.copy()
            stall_count = 0
            eta = max(eta_min, eta * 0.8)
        
        if verbose and iter_count - last_report >= 5000:
            last_report = iter_count
            print(f"      MND iter={iter_count} ({elapsed:.0f}s): obj={best_obj:.6f}, beta={cur_beta:.1f}, eta={cur_eta:.4f}")
    
    return best_a, best_obj


def spectral_project(a, strength=0.3):
    """Project towards flat power spectrum while maintaining non-negativity."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()
    
    # Blend towards flat
    mag_new = (1.0 - strength) * mag + strength * mag_mean
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    
    return normalize_simplex(a_new)


def spectral_compress(a, strength=0.5):
    """Compress dynamic range of amplitude spectrum."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def fixed_point_optimize(a0, time_limit, verbose=False):
    """Multiplicative weight fixed-point iteration for KKT conditions."""
    n = len(a0)
    t0 = time.monotonic()
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    eta = 0.8
    beta = 25.0
    top_k = 12
    damping = 0.5
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        g_a = lse_gradient_a_space(a, c, beta, top_k=top_k)
        
        # Multiplicative update: a_new ∝ a * exp(-eta * g_a)
        log_ratio = -eta * g_a
        log_ratio -= log_ratio.max()
        a_new = a * np.exp(log_ratio)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Damped update
        a_blend = (1.0 - damping) * a + damping * a_new
        a_blend = normalize_simplex(a_blend)
        
        c_new = fast_conv(a_blend)
        M_new = c_new.max()
        
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_blend.copy()
            a = a_blend
            eta = min(eta * 1.005, 2.0)
        else:
            a = normalize_simplex(0.5 * a + 0.5 * best_a)
            eta = max(eta * 0.98, 0.1)
            
            if eta < 0.15:
                break
        
        if it % 500 == 0:
            beta = max(5.0, beta * 0.99)
            top_k = max(3, int(top_k * 0.97))
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FP iter={it} ({elapsed:.0f}s): obj={2.0*n*best_m:.6f}, eta={eta:.3f}")
    
    return best_a, 2.0 * n * best_m


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent: move mass between positions to reduce peak."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            for attempt in range(8):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.003, 0.008, 0.02, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] = max(0.0, a_trial[src] - amt)
                    a_trial[dst] += amt
                    a_trial = normalize_simplex(a_trial)
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = normalize_simplex(a)
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, 2.0 * n * best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return normalize_simplex(a)


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return normalize_simplex(a)


def make_init_hann(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 0.0)
    return normalize_simplex(a)

        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            parent = p.copy()

            best_seq = normalize_simplex(parent)
            best_score = p_score
    except Exception:

    if best_seq is None:
