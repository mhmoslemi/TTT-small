    return np.fft.irfft(A * A, fft_size)[:size]

        if time.monotonic() - t0 > 300:  # Safety timeout per call
            break

        M = c.max()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based linearized peak reduction with active set (top-P peaks)
2. Nesterov mirror descent with adaptive softmax subgradient
3. Coordinate descent on peak-contributing elements
4. Multi-length search with interpolation
Key innovation: at each step, solve a sparse LP to find the optimal
linearized descent direction for the convolution peak, then take a
damped step with line search. This converges to a flat-peak solution.
"""

import numpy as np
from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog
import time


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_power_law_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return a / a.sum()


def make_mild_ramp(n, center_frac=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 + center_frac * x
    return a / a.sum()


def make_quadratic_ramp(n, alpha=0.2, beta=0.1):
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * x + beta * x * x
    return a / a.sum()


def make_asym_gaussian(n, center=0.65, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) ** 2) / (2 * width * width))

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_descent_step(a, c, P=100):
    n = len(a)
    top_k = np.argsort(c)[::-1][:min(P, len(c))]
    P_actual = len(top_k)

    n_rows = P_actual + n
    A_ineq = lil_matrix((n_rows, n + 1))
    b_ineq = np.zeros(n_rows)

    for idx, k in enumerate(top_k):
        ki = int(k)
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        for i in range(i_lo, i_hi + 1):
            A_ineq[idx, i] = a[ki - i]
        A_ineq[idx, n] = -1.0
        b_ineq[idx] = -c[ki]

    for i in range(n):
        A_ineq[P_actual + i, i] = -1.0
        b_ineq[P_actual + i] = a[i]

    A_ineq = A_ineq.tocsr()

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    try:
        result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                         A_eq=A_eq, b_eq=b_eq,
                         bounds=[(None, None)] * n + [(0, None)],
                         method='highs')
    except Exception:
        return None

    if not result.success:
        return None
    return result.x[:n]


def lp_peak_reduction(a0, time_limit, P=100, verbose=False, rng_seed=42):

    for it in range(2000):

            if s_new < 1e-15:

def compute_gradient(a, ac, n_peaks=80, temperature=1.0):
    n = len(a)
    k_max = min(n_peaks, len(ac))
    ac_clipped = np.minimum(ac, ac.max())
    log_ac = np.log(np.maximum(ac_clipped, 1e-30))
    exp_vals = np.exp((log_ac - log_ac.max()) / temperature)
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    top_idx = np.argsort(ac)[::-1][:k_max]
    for k in top_idx:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-10:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    return grad


def grad_optimize(a0, time_limit, n_peaks=80, temperature0=3.0, c0=5.0,
                  restart_every=5000, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_temp = temperature0
    restart_count = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_gradient(y, ac, n_peaks=n_peaks, temperature=cur_temp)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = ac.max()
        M_x = fast_autocorr(x).max()
        M_eff = min(M_new, M_x)
        if M_eff < best_m - 1e-15:
            best_m = M_eff
            if M_x <= M_new:
                best_a = x.copy()
            else:
                best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        cur_temp = max(0.8, temperature0 * (1.0 - 0.1 * min(stall_count / 2000.0, 1.0)))

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.01 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if restart_count % 3 == 0:
                n_peaks = min(n_peaks + 10, 150)
            cur_temp = max(0.8, temperature0 - 0.2 * restart_count)
            gamma = c0 * (1.0 + 0.05 * min(restart_count, 5))

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            costs = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                j2_lo = max(0, k2i - n + 1)
                j2_hi = min(n - 1, k2i)
                valid = np.arange(j2_lo, j2_hi + 1)
                if len(valid) > 0:
                    costs[valid] += a[k2i - valid]

            candidates = np.argsort(costs + 10 * a)[:n // 3]
            best_dst = candidates[np.argmin(a[candidates])]
            if best_dst == src:
                continue

            found = False
            for frac in [0.03, 0.06, 0.1, 0.15, 0.25]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and a[src] > 1e-10:
                dst = int(rng.integers(n))
                if dst != src:
                    frac = rng.uniform(0.01, 0.06)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def parametric_optimize(n, time_limit):
    """Optimize a 3-parameter family: a[i] = (1 + p1*x + p2*x^2) * exp(p3*x)"""
    from scipy.optimize import minimize

    def objective(params):
        p1, p2, p3 = params
        x = np.linspace(0, 1, n)
        a = (1.0 + p1 * x + p2 * x * x) * np.exp(p3 * x)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return 1e10
        a = a / s
        c = fast_autocorr(a)
        return 2.0 * n * c.max()

    best_result = None
    best_val = 1e10

    init_points = [
        [0.3, 0, 0],
        [0.5, 0, 0],
        [0.2, 0.1, 0],
        [0.4, -0.1, 0],
        [0, 0, 0.5],
        [0.2, 0, 0.3],
        [0.1, 0.2, -0.2],
        [0.5, 0.1, 0.2],
    ]

    for init in init_points:
        if time.monotonic() - time.monotonic() > 0:
            pass
        try:
            result = minimize(objective, init, method='Nelder-Mead',
                              options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
            if result.fun < best_val:
                best_val = result.fun
                best_result = result.x
        except Exception:
            pass

    if best_result is None:
        return make_power_law_ramp(n, 0.5)

    p1, p2, p3 = best_result
    x = np.linspace(0, 1, n)
    a = (1.0 + p1 * x + p2 * x * x) * np.exp(p3 * x)
    a = np.maximum(a, 0.0)
    return a / a.sum()

        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Multi-length search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length search ===")
    lengths_to_try = []
    for delta in [-1000, -500, 500, 1000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))

    for new_n in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t5a = min(remaining * 0.35, 120.0)
        if t5a > 15:
            a_out, _ = lp_peak_reduction(a_long, t5a, P=100, verbose=False, rng_seed=42)
            try_update(a_out, f"lp_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.25, 80.0)
            if t5b > 15:
                a_out, _ = grad_optimize(a_long, t5b, n_peaks=80, temperature0=2.5,
                                          c0=5.0, restart_every=4000, verbose=False,
                                          rng_seed=42)
                try_update(a_out, f"grad_n{new_n}")

    # === Phase 6: Final LP + coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}) ===")
    if remaining > 40:
        t6a = min(remaining * 0.45, 150.0)
        if t6a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_peak_reduction(a_in, t6a, P=150, verbose=True, rng_seed=777)
            try_update(a_out, f"lp_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.50, 100.0)
            if t6b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_descent(a_in, t6b, verbose=False, rng_seed=555)
                try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 7: One more gradient pass with aggressive params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t7 = min(remaining * 0.60, 120.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = grad_optimize(a_in, t7, n_peaks=150, temperature0=1.0,
                                      c0=3.0, restart_every=8000, verbose=True,
                                      rng_seed=123)
            try_update(a_out, f"grad_very_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_power_law_ramp(n_fb, power=0.4)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
