def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Fast local evaluation (assumes a is normalized, non-negative)."""
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * b.max() / (S * S))


def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_smoothed_obj(a, b, tau):
    """Compute log-sum-exp smoothed version of max(b)."""
    b_scaled = b / tau
    b_scaled -= b_scaled.max()
    exp_b = np.exp(b_scaled)
    return tau * (np.log(exp_b.sum()) + b.max() / tau)


def lse_grad(a, b, tau):
    """Gradient of log-sum-exp smoothed max(b) w.r.t. a, via FFT."""
    n = len(a)
    b_scaled = b / tau
    b_scaled -= b_scaled.max()
    p = np.exp(b_scaled)
    p = p / p.sum()
    
    # grad_a[i] = 2 * sum_k p[k] * (db[k]/da[i])
    # db[k]/da[i] = a[k-i] if 0 <= k-i < n, else 0
    # So grad_a[i] = 2 * sum_{k: valid} p[k] * a[k-i]
    # = 2 * (convolve(p, a)[i + n - 1]) using appropriate indexing
    # Actually: sum_k p[k] * a[k-i] for k-i in [0,n-1] means k in [i, i+n-1]
    # This is the cross-correlation of p and a at lag i
    # = (flip(a) * p)[i + n - 1] in full convolution
    size = len(b)  # 2n-1
    fft_size = 1 << (size - 1).bit_length()
    a_flip = a[::-1].astype(np.float64)
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p, fft_size)
    full_conv = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * full_conv[n - 1:n - 1 + n]
    return grad


def multiplicative_descent(a0, time_limit, tau_init=None, tau_decay=0.9997,
                           tau_min_ratio=0.003, lr=0.8, beta1=0.9, beta2=0.999,
                           verbose=True, seed=42):
    """Multiplicative gradient descent with log-sum-exp smoothing and Adam."""
    n = len(a0)
    a = norm1(a0.copy())
    b = fast_conv(a)
    best_a = a.copy()
    best_obj = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    if tau_init is None:
        tau = max(b.max() * 0.4, 1e-6)
    else:
        tau = tau_init
    tau_min = max(b.max() * tau_min_ratio, 1e-12)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    iter_count = 0
    
    store_a = [a.copy()]
    store_interval = 50
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit * 0.85:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        
        g = lse_grad(a, b, tau)
        
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        # Adaptive: reduce step if tau is small (approaching true max)
        effective_step = step * min(1.0, tau / max(b.max() * 0.1, 1e-10))
        
        log_a = np.log(np.maximum(a, 1e-30)) - effective_step * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        a = a_new
        
        if t % store_interval == 0:
            store_a.append(a.copy())
        
        if stall > 2000:
            stall = 0
            # Local perturbation
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    MD restart t={t} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, tau={tau:.2e}")
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, tau={tau:.2e}")
        
        iter_count += 1
    
    # Polyak average over last stored iterates
    if len(store_a) > 10:
        avg = np.mean(store_a[-20:], axis=0)
        avg = norm1(avg)
        avg_obj = fast_conv(avg).max()
        if avg_obj < best_obj:
            best_a = avg.copy()
            best_obj = avg_obj
            if verbose:
                print(f"    MD: Polyak avg improved to {2*n*best_obj:.6f}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    MD done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, iters={iter_count}")
    return best_a, best_obj


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=3000,
                   add_per_iter=30, warm_start_alpha=0.9, verbose=True):
    """LP Frank-Wolfe with warm starts and adaptive constraint management."""
    n = len(a0)
    a = norm1(a0.copy())
    b_init = fast_conv(a)
    best_a = a.copy()
    best_obj = b_init.max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    # Initialize constraint set with top peaks
    init_idx = np.argsort(b_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    lp_fail = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_y = fast_conv(y)
        M_y = b_y.max()
        
        if M_y < best_obj - 1e-14:
            best_obj = M_y
            best_a = y.copy()
        
        b_x = fast_conv(x)
        M_x = b_x.max()
        if M_x < best_obj - 1e-14:
            best_obj = M_x
            best_a = x.copy()
        
        # Adaptive constraint management
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_y - b_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[ki - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail += 1
            if lp_fail > 10:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail += 1
            if lp_fail > 15:
                constraint_set = set(int(k) for k in constraint_list[:max(100, initial_K//2)])
            continue
        
        lp_fail = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_yg > M_y + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_y
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # FW update with FISTA acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / sy if sy > 1e-15 else y.copy()
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            x_new = x_new / sx if sx > 1e-15 else y_new.copy()
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, iters={iter_count}")
    return best_a, best_obj


def coord_descent(a0, time_limit, rng_seed=42, verbose=True):
    """Targeted coordinate descent with binary search for optimal transfer amount."""
    n = len(a0)
    a = norm1(a0.copy())
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        b = fast_conv(a)
        M_curr = b.max()
        if M_curr < best_obj - 1e-15:
            best_obj = M_curr
            best_a = a.copy()
        
        # Focus on top peaks
        n_top = min(20, len(b))
        top_k = np.argsort(b)[::-1][:n_top]
        
        improved = False
        
        for k_idx in range(min(10, len(top_k))):
            if time.monotonic() - t0 > time_limit * 0.95:
                break
            k = int(top_k[k_idx])
            if b[k] < best_obj * 0.95:
                continue
            
            # Find which (i, k-i) pairs contribute most to b[k]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            pairs = idx * (k - idx)
            if len(pairs) == 0:
                continue
            pair_max = pairs.max()
            if pair_max < 1e-20:
                continue
            
            # Source: the pair with maximum contribution
            src_pos = idx[np.argmax(pairs)]
            src_val = a[src_pos]
            if src_val < 1e-12:
                continue
            
            # Find destination: position that minimally affects top peaks
            # Score each position by how much it appears in top peaks
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2 = int(k2)
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2 - idx2]
                    scores[k2 - idx2] += a[idx2]
            
            # Prefer positions with low score (less impact on peaks)
            # and not the source
            scores[src_pos] = np.inf
            dst = int(np.argmin(scores))
            
            if dst == src_pos:
                continue
            
            # Binary search for optimal transfer fraction
            lo = 0.001
            hi = 0.3
            best_frac = 0.0
            best_trial_obj = best_obj
            
            for _ in range(8):
                mid = (lo + hi) / 2.0
                amt = src_val * mid
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[src_pos] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                trial_obj = fast_conv(a_trial).max()
                if trial_obj < best_trial_obj - 1e-16:
                    best_trial_obj = trial_obj
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid
            
            if best_frac > 0:
                amt = src_val * best_frac
                a[src_pos] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                improved = True
                stall = 0
        
        if not improved:
            stall += 1
            # Random small perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_obj:
                    best_obj = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall > 0 and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CD done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}")
    return best_a, best_obj

    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Polyak step size parameters
    obj_bound = 2.0 / n  # Upper bound: uniform gives 2/n for max_b
    t = 0
    
    for t in range(n_iters_max):
        if time.monotonic() - t0 > time_limit * 0.95:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
        
        # Subgradient of max_k b[k]: average of gradients of top peaks
        top_count = min(20, len(b))
        top_idx = np.argsort(b)[::-1][:top_count]
        
        # Weight by how close to max
        weights = b[top_idx] - b[top_idx].min() + 1e-10
        weights = np.exp(weights / max(1e-10, (b[top_idx].max() - b[top_idx].min())))
        weights = weights / weights.sum()
        
        # Compute gradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            g[idx] += w * 2.0 * a[ki - idx]
        
        # Step size
        gap = M - obj_bound
        if gap < 1e-6:
            step = 0.01 / (t + 1)
        else:
            step = 2.0 * gap / (np.dot(g, g) + 1e-10)
            step = min(step, 0.1)
        
        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        a = a_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    SubGrad done ({elapsed:.0f}s): obj={2*n*best_obj:.6f}")
    return best_a, best_obj


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, seed=0):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(seed)
    candidates = []
    
    # Sin^2 family
    for freq in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.001)
            candidates.append(a / a.sum())
    
    # Cosine taper (Hann window)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(a / a.sum())
    
    # Blackman-like
    a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())
    
    # Power law
    for p in [0.5, 1.0, 2.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())
    
    # Multi-peak
    for n_peaks in [3, 5, 7]:
        peaks = np.linspace(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for pk in peaks:
            a += np.exp(-((x - pk) ** 2) / 0.01)
        a = np.maximum(a, 0.001)
        candidates.append(a / a.sum())
    
    # Random smooth
    for _ in range(5):
        freqs = rng.integers(1, 5, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        amps = rng.uniform(0.5, 1.0, size=5)
        a = np.ones(n) * 0.5
        for f, ph, am in zip(freqs, phases, amps):
            a += am * np.sin(2 * np.pi * f * x + ph) ** 2
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    return candidates


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver() if HAS_CVXPY else None
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p = norm1(p)
            p_score = eval_local(p)
            if np.isfinite(p_score):
                best_seq = p.copy()
                best_score = p_score
                print(f"Parent: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * x))
        best_seq = norm1(best_seq)
        best_score = eval_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_local(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Quick parametric search ===
    print(f"\n=== Phase 1: Parametric search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        n0 = len(best_seq)
        candidates = parametric_init(n0, seed=1)
        for i, cand in enumerate(candidates):
            if time.monotonic() - t_global > time_budget - t1:
                break
            sc = eval_local(cand)
            if np.isfinite(sc) and sc < best_score:
                check_update(cand, f"param{i}")
    
    n_best = len(best_seq)
    
    # === Phase 2: Multiplicative descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multiplicative descent (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = norm1(best_seq)
        a_out, _ = multiplicative_descent(a_in, t2, tau_decay=0.9998, lr=0.7,
                                          beta1=0.9, beta2=0.999, verbose=True, seed=7)
        check_update(a_out, f"MD_n{n_best}")
    
    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP Frank-Wolfe (n={n_best}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30 and solver is not None and n_best <= 6000:
        a_in = norm1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, initial_K=400,
                                   max_K=3000, add_per_iter=30, verbose=True)
        check_update(a_out, f"LP_n{n_best}")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        a_in = norm1(best_seq)
        a_out, _ = coord_descent(a_in, t4, rng_seed=99, verbose=True)
        check_update(a_out, f"CD_n{n_best}")
    
    # === Phase 5: Direct subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Direct subgradient (n={n_best}) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        a_in = norm1(best_seq)
        a_out, _ = direct_subgradient(a_in, t5, verbose=False)
        check_update(a_out, f"SG_n{n_best}")
    
    # === Phase 6: Restart and refine ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Restart & refine (n={n_best}) ===")
    if remaining > 120:
        rng = np.random.default_rng(123)
        t6 = min(remaining * 0.25, 250.0)
        
        # Try a few perturbed restarts
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            noise_scale = 0.015 * (1 + trial * 0.5)
            pert = rng.normal(0, noise_scale, n_best) * best_seq
            a_restart = best_seq + pert
            a_restart = np.maximum(a_restart, 1e-10)
            a_restart = a_restart / a_restart.sum()
            
            t_re = min(remaining * 0.35, 120.0)
            if t_re < 20:
                break
            
            a_out, _ = multiplicative_descent(a_restart, t_re, tau_decay=0.9998,
                                              lr=0.5, beta1=0.9, beta2=0.999,
                                              verbose=False, seed=100+trial)
            check_update(a_out, f"restart{trial}_n{n_best}")
            
            # Quick CD polish
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t_cd = min(remaining * 0.15, 60.0)
                if t_cd > 10:
                    a_out2, _ = coord_descent(norm1(best_seq), t_cd, rng_seed=200+trial, verbose=False)
                    check_update(a_out2, f"restart{trial}_CD_n{n_best}")
    
    # === Phase 7: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    nearby = []
    for delta in [200, -200, 400, -400, 500, -500]:
        new_n = n_best + delta
        if 3000 <= new_n <= 6500:
            nearby.append(new_n)
    
    seen = set()
    unique_n = []
    for ln in nearby:
        if ln not in seen:
            seen.add(ln)
            unique_n.append(ln)
    
    for new_n in unique_n[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_seq(best_seq, new_n)
        sc = eval_local(a_long)
        check_update(a_long, f"interp_n{new_n}")
        
        if remaining > 60:
            t7 = min(remaining * 0.3, 100.0)
            a_out, _ = multiplicative_descent(a_long, t7, tau_decay=0.9999, lr=0.4,
                                              beta1=0.9, beta2=0.999, verbose=False, seed=300)
            check_update(a_out, f"MD_n{new_n}")
    
    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 30:
        # Very short MD with small learning rate
        t8a = min(remaining * 0.3, 60.0)
        if t8a > 10:
            a_out, _ = multiplicative_descent(norm1(best_seq), t8a, tau_decay=0.9999,
                                              lr=0.3, beta1=0.95, beta2=0.999,
                                              verbose=False, seed=500)
            check_update(a_out, f"polish_MD_n{n_best}")
        
        # Short CD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.4, 50.0)
            if t8b > 10:
                a_out, _ = coord_descent(norm1(best_seq), t8b, rng_seed=600, verbose=False)
                check_update(a_out, f"polish_CD_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = eval_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * x))
        best_seq = norm1(best_seq)
        best_score = eval_local(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
