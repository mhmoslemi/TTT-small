def lse_obj_and_grad(a, beta, top_k=200):
    """Compute LSE-smoothed max autocorrelation and its gradient."""
    ac = fast_autocorr(a)
    n = len(a)
    
    # For numerical stability, work with top-k values
    if top_k < len(ac):
        top_idx = np.argpartition(ac, -top_k)[-top_k:]
        top_vals = ac[top_idx]
        # Softmax over top_k values (with temperature beta)
        shifted = beta * (top_vals - top_vals.max())
        exp_shifted = np.exp(shifted)
        p_top = exp_shifted / exp_shifted.sum()
    else:
        top_idx = np.arange(len(ac))
        top_vals = ac
        shifted = beta * (top_vals - top_vals.max())
        exp_shifted = np.exp(shifted)
        p_top = exp_shifted / exp_shifted.sum()
    
    # LSE value (approximate, over top_k only)
    lse_val = top_vals.max() + np.log(exp_shifted.sum()) / beta

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent with exact gradient and beta-annealing
2. LP-based Frank-Wolfe with SOTA formulation (maximize sum(g) s.t. f*g <= M)
3. Multi-scale parametric initialization
4. Adaptive coordinate descent polish
5. Multi-length exploration with spectral interpolation
"""

import numpy as np
import time
import cvxpy as cp

_GLOBAL_SOLVER = None


def _get_solver():
    global _GLOBAL_SOLVER
    if _GLOBAL_SOLVER is not None:
        return _GLOBAL_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _GLOBAL_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _GLOBAL_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _GLOBAL_SOLVER


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _objective(a):
    n = len(a)
    return 2.0 * n * _fast_autocorr(a).max()


def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _lse_gradient(a, beta, top_k=300):
    """Compute LSE-smoothed max autocorrelation value and exact gradient."""
    n = len(a)
    ac = _fast_autocorr(a)
    ac_len = len(ac)

    if top_k < ac_len:
        top_idx = np.argpartition(ac, -top_k)[-top_k:]
    else:
        top_idx = np.arange(ac_len)

    top_vals = ac[top_idx]
    max_val = top_vals.max()
    shifted = beta * (top_vals - max_val)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        p = np.ones(len(top_idx)) / len(top_idx)
    else:
        p = exp_vals / exp_sum

    lse_val = max_val + np.log(exp_sum) / beta

    grad = np.zeros(n)
    for idx_i in range(len(top_idx)):
        k = int(top_idx[idx_i])
        w = 2.0 * p[idx_i]
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += w * a[k - idx]

    return lse_val, grad, ac


def _mirror_descent_lse(a_init, time_limit, beta_start=20.0, beta_end=800.0,
                        top_k=300, c0=2.0, momentum=0.9, verbose=True):
    """Mirror descent on LSE-smoothed objective with momentum and annealing."""
    n = len(a_init)
    a = _normalize(a_init)
    best_a = a.copy()
    best_obj = _objective(a)
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum

    total_planned = 80000
    total_expand = 0.0

    for t in range(1, total_planned + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(t / total_planned, 1.0)
        # Exponential annealing for smoother transition
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)

        lse_val, grad, ac = _lse_gradient(a, beta, top_k)

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-10:
            v = v / v_norm

        # Adaptive step size
        eta = c0 / (np.sqrt(t) + 1.0)

        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-30)) - eta * v
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        # Check true objective periodically
        if t % 50 == 0 or t < 5:
            obj = _objective(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose and t % 500 == 0:
                    elapsed = time.monotonic() - t0
                    print("    MD-LSE t=" + str(t) + " beta=" + str(round(beta)) +
                          " J=" + repr(best_obj) + " (" + str(round(elapsed)) + "s)")

        a = a_new

    # Final check
    obj = _objective(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print("    MD-LSE done (" + str(round(elapsed)) + "s): J=" + repr(best_obj))
    return best_a, best_obj


def _lp_fw_sota(a_init, time_limit, solver, initial_K=200, max_K=1200,
                add_per_iter=20, verbose=True):
    """LP-FW using SOTA formulation: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(a_init)
    y = _normalize(a_init)
    best_a = y.copy()
    best_m = _fast_autocorr(y).max()
    t0 = time.monotonic()

    ac_y = _fast_autocorr(y)
    M_curr = ac_y.max()
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 1000

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stall, stopping at iter " + str(t))
            break

        ac_y = _fast_autocorr(y)
        M_curr = ac_y.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix: (y * g)[k] <= M for k in active_set
        # (y*g)[k] = sum_j y[j] * g[k-j]
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[ki - idx]

        # LP: maximize sum(g) subject to g >= 0, coeffs @ g <= M_curr
        g_var = cp.Variable(n, nonneg=True)
        constraints = [coeffs @ g_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search: try several step sizes
        default_alpha = 2.0 / (t + 2.0)
        alphas = list(set([
            max(0.01, min(1.0, default_alpha)),
            max(0.01, min(1.0, default_alpha * 0.5)),
            max(0.01, min(1.0, default_alpha * 2.0)),
            max(0.01, min(1.0, 1.0 / (t + 1))),
        ]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = _fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = y.copy()
            else:
                y = y / s_y

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed)) +
                  "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed)) + "s): J=" + repr(J))
    return best_a, best_m


def _proj_grad_descent(a_init, time_limit, lr_init=0.01, decay=0.999, verbose=True):
    """Projected gradient descent on the true (unsmoothed) objective via subgradient."""
    n = len(a_init)
    a = _normalize(a_init)
    best_a = a.copy()
    best_obj = _objective(a)
    t0 = time.monotonic()
    lr = lr_init
    rng = np.random.default_rng(99)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(a)
        M = ac.max()
        peak_k = int(np.argmax(ac))

        # Subgradient of max at the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[peak_k - idx]

        # Add small noise to escape symmetry
        if t % 1000 == 0:
            grad += 0.01 * rng.normal(0, 1, n) * a

        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new)

        obj = _objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new
        lr *= decay

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    PGrad t=" + str(t) + " (" + str(round(elapsed)) + "s): J=" + repr(best_obj))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    PGrad done (" + str(round(elapsed)) + "s): J=" + repr(best_obj))
    return best_a, best_obj


def _coord_descent(a_init, time_limit, rng_seed=42, verbose=True):
    """Coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a_init)
    a = _normalize(a_init)
    best_a = a.copy()
    best_obj = _objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    step_fracs = [0.003, 0.007, 0.01, 0.02, 0.04, 0.08, 0.12]

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination: minimize impact on top peaks
            dst_score = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    dst_score[idx2] += a[k2i - idx2]
                    dst_score[k2i - idx2] += a[idx2]

            # Avoid source
            dst_score[src] = np.inf
            dst = int(np.argmin(dst_score))

            for frac in step_fracs:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                obj_t = _objective(a_t)
                if obj_t < best_obj - 1e-15:
                    a = a_t
                    best_obj = obj_t
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 300:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        obj_t = _objective(a)
                        if obj_t < best_obj:
                            best_obj = obj_t
                            best_a = a.copy()
                            stall = 0
            if stall > 2000:
                break
        else:
            stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (" + str(round(elapsed)) + "s): J=" + repr(best_obj) +
                  " stall=" + str(stall))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Coord done (" + str(round(elapsed)) + "s): J=" + repr(best_obj))
    return best_a, best_obj


def _parametric_init(n, time_limit, verbose=True):
    """Try diverse parametric families to find good initial points."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = np.ones(n) / n
    best_val = _objective(best_a)

    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(_normalize((i_arr + 1.0) ** p))
        candidates.append(_normalize((n - i_arr) ** p))

    # Bivariate
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            candidates.append(_normalize(
                np.maximum(i_arr + 1.0, 0.5) ** alpha *
                np.maximum(n - i_arr, 0.5) ** beta))

    # Gaussians
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3]:
            candidates.append(_normalize(np.exp(-((x - c) ** 2) / (2.0 * sigma ** 2))))

    # Double peak
    for c1 in [0.25, 0.3, 0.35]:
        for c2 in [0.65, 0.7, 0.75]:
            for sigma in [0.05, 0.1, 0.15]:
                candidates.append(_normalize(
                    np.exp(-((x - c1) ** 2) / (2.0 * sigma ** 2)) +
                    np.exp(-((x - c2) ** 2) / (2.0 * sigma ** 2))))

    # U-shaped
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(_normalize(x ** p + (1.0 - x) ** p))

    # Cosine
    for freq in [1, 2, 3]:
        for ph in [0.0, np.pi / 2]:
            candidates.append(_normalize(
                np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.01)))

    # Sinc-like
    for w in [3.0, 5.0, 8.0, 12.0]:
        t = (x - 0.5) * w
        sin_t = np.sin(np.pi * t)
        sinc = np.where(np.abs(t) < 1e-10, 1.0, sin_t / (np.pi * t))
        candidates.append(_normalize(sinc ** 2))

    for a_cand in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = _objective(a_cand)
        if v < best_val:
            best_val = v
            best_a = a_cand.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print("  Parametric init (" + str(round(elapsed)) + "s): best J=" + repr(best_val))
    return best_a, best_val


def _try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = _normalize(a)
    sc = _evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(round(sc, 6)) +
              " (n=" + str(len(a)) + ", t=" + str(round(elapsed)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: LSE mirror descent + LP-FW + coordinate polish + multi-length."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = _get_solver()
    print("Time budget: " + str(round(time_budget)) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = _normalize(p)
            p_score = _evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = _normalize(np.sin(np.pi * x) ** 2)
        best_score = _evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric search for better init
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Parametric search (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 5:
        a_param, _ = _parametric_init(n0, t1, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_param, "param", t_global)

    # Phase 2: LSE Mirror Descent (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 2: LSE Mirror Descent (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.30, 420.0)
    if t2 > 20:
        a_out, _ = _mirror_descent_lse(_normalize(best_seq), t2,
                                       beta_start=20.0, beta_end=600.0,
                                       top_k=300, c0=2.0, momentum=0.9, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "mdlse_n" + str(n_cur), t_global)

    # Phase 3: Second round MD-LSE with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 3: MD-LSE round 2 (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_out, _ = _mirror_descent_lse(_normalize(best_seq), t3,
                                       beta_start=50.0, beta_end=1000.0,
                                       top_k=200, c0=1.5, momentum=0.85, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "mdlse2_n" + str(n_cur), t_global)

    # Phase 4: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=" + str(n_cur) + ") ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30 and n_cur <= 7000:
        a_out, _ = _lp_fw_sota(_normalize(best_seq), t4, solver,
                               initial_K=200, max_K=1000,
                               add_per_iter=20, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw_n" + str(n_cur), t_global)

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 5: Coord descent (n=" + str(n_cur) + ") ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_out, _ = _coord_descent(_normalize(best_seq), t5, rng_seed=42, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "coord_n" + str(n_cur), t_global)

    # Phase 6: Try slightly longer sequences
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    longer_lengths = [n_cur + 500, n_cur + 1000]
    longer_lengths = [l for l in longer_lengths if l <= 8000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n=== Phase 6: Extend to n=" + str(new_n) + " ===")
        a_long = _interpolate_seq(best_seq, new_n)

        t6 = min(remaining * 0.20, 150.0)
        if t6 > 20:
            a_out, _ = _mirror_descent_lse(a_long, t6,
                                           beta_start=30.0, beta_end=500.0,
                                           top_k=200, c0=2.0, momentum=0.9, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "mdlse_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 6500:
            t6b = min(remaining * 0.25, 150.0)
            if t6b > 20:
                a_out, _ = _lp_fw_sota(_normalize(best_seq), t6b, solver,
                                       initial_K=150, max_K=800,
                                       add_per_iter=15, verbose=True)
                best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # Phase 7: Final LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_cur) + ") ===")
    if remaining > 60 and n_cur <= 7000:
        t7 = min(remaining * 0.30, 200.0)
        if t7 > 20:
            a_out, _ = _lp_fw_sota(_normalize(best_seq), t7, solver,
                                   initial_K=300, max_K=1500,
                                   add_per_iter=30, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw_final_n" + str(n_cur), t_global)

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 8: Final coord (n=" + str(n_cur) + ") ===")
    if remaining > 20:
        t8 = min(remaining * 0.50, 100.0)
        if t8 > 10:
            a_out, _ = _coord_descent(_normalize(best_seq), t8, rng_seed=314, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "coord_final_n" + str(n_cur), t_global)

    # Phase 9: One more MD-LSE with high beta for final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print("\n=== Phase 9: Final MD-LSE high-beta (n=" + str(n_cur) + ") ===")
        t9 = min(remaining * 0.60, 120.0)
        if t9 > 10:
            a_out, _ = _mirror_descent_lse(_normalize(best_seq), t9,
                                           beta_start=200.0, beta_end=2000.0,
                                           top_k=150, c0=1.0, momentum=0.8, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "mdlse_final_n" + str(n_cur), t_global)

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) +
              ", time=" + str(round(elapsed)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = _normalize(np.sin(np.pi * x) ** 2)
        best_score = _evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
