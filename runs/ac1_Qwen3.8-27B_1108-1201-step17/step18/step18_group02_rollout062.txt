"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent with adaptive log-sum-exp tau
2. Spectral shaping (Gerchberg-Saxton style iterative Fourier amplitude modulation)
3. Enhanced LP Frank-Wolfe with adaptive constraint tightening
4. Exact coordinate descent with bisection for optimal transfer
5. Multi-length exploration with U-shaped and parametric inits
"""

import numpy as np
import time
import cvxpy as cp


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_proj(v):
    """Project v onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, C, tau):
    """Gradient of tau*log(sum exp(C/tau)) w.r.t. a."""
    n = len(a)
    C_scaled = C / tau
    C_scaled -= C_scaled.max()
    p = np.exp(C_scaled)
    p_sum = p.sum()
    if p_sum < 1e-30:
        return np.zeros(n)
    p = p / p_sum

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    p_rev = p[::-1].copy()
    p_pad = np.zeros(fft_size)
    p_pad[:len(p_rev)] = p_rev
    A = np.fft.rfft(a_pad)
    P = np.fft.rfft(p_pad)
    conv = np.fft.irfft(A * P)[:size]
    s = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * s


def spectral_shape(a, n_iter=3, flatness=0.7):
    """GS-style spectral flattening. Returns modified a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    a_work = a.copy()
    for _ in range(n_iter):
        A = np.fft.rfft(a_work, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: flatten magnitude spectrum
        # Keep DC component (sum), reduce others
        mag_flat = np.zeros_like(mag)
        mag_flat[0] = mag[0]
        others = mag[1:]
        target_level = flatness * others.mean()
        mag_flat[1:] = np.minimum(others, target_level)
        # Blend: don't change too abruptly
        mag_new = 0.5 * mag + 0.5 * mag_flat
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_work = a_new / s
    return a_work


def coord_descent_exact(a, C, n_top=10, max_iters=200, time_limit=30.0):
    """Coordinate descent with exact transfer via bisection."""
    n = len(a)
    t0 = time.monotonic()
    best_m = C.max()
    best_a = a.copy()

    for _ in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        k_top = np.argsort(C)[::-1][:n_top]
        improved = False

        for k in k_top[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Find best destination: minimize max increase in C
            # Cost of adding delta to position j: 2*delta*a[|k-j|] for each top peak
            scores = np.zeros(n)
            for k2 in k_top[:5]:
                k2i = int(k2)
                for j in range(n):
                    d = abs(k2i - j)
                    if d < n:
                        scores[j] += a[d]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in k_top[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            candidates = candidates[candidates != src]
            if len(candidates) == 0:
                continue
            dst = candidates[np.argmin(scores[candidates])]

            # Bisection on transfer fraction
            lo, hi = 0.0, 0.5
            best_frac = 0.0
            best_trial_m = best_m
            for _ in range(12):
                mid = 0.5 * (lo + hi)
                amt = src_val * mid
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    lo = mid
                    continue
                a_trial = a_trial / st
                C_trial = fast_conv(a_trial)
                m_trial = C_trial.max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid

            if best_frac > 0.01:
                amt = src_val * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                C = fast_conv(a)
                if C.max() < best_m:
                    best_m = C.max()
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a


def lp_frank_wolfe(a, time_limit, solver, M_quantile=0.95, max_K=3000, verbose=True):
    """LP-FW with adaptive constraint tightening."""
    n = len(a)
    a_curr = a.copy()
    C = fast_conv(a_curr)
    best_m = C.max()
    best_a = a_curr.copy()
    t0 = time.monotonic()

    # Initial constraint set: top peaks
    n_init = min(300, 2 * n - 1)
    init_idx = np.argsort(C)[::-1][:n_init]
    constraint_set = sorted(set(int(k) for k in init_idx))

    y = a_curr.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        C_y = fast_conv(y)
        M_curr = C_y.max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()

        # Adaptive M: use quantile to force flatter solutions
        M_target = np.quantile(C_y, M_quantile)
        M_use = max(M_target, best_m * 1.001)

        # Manage constraint set
        if len(constraint_set) > max_K:
            margins = np.array([M_curr - C_y[k] for k in constraint_set])
            keep = np.argsort(margins)[:max_K]
            constraint_set = sorted(set(int(constraint_set[i]) for i in keep))

        K = len(constraint_set)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_set):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[ki - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_use]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check if g_dir is actually better
        C_g = fast_conv(g_dir)
        M_g = C_g.max()

        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        C_new = fast_conv(y_new)
        M_new = C_new.max()
        if M_new < best_m:
            best_m = M_new
            best_a = y_new.copy()

        # Add violated constraints
        C_ys = fast_conv(g_dir)
        violations = np.where(C_ys > M_use + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = C_ys[violations] - M_use
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                constraint_set.append(int(k))
            constraint_set = sorted(set(constraint_set))

        # Also add current top peaks periodically
        if t % 50 == 0:
            top_now = np.argsort(C_new)[::-1][:50]
            for k in top_now:
                constraint_set.append(int(k))
            constraint_set = sorted(set(constraint_set))

        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): best_J={2*n*best_m:.6f}, |S|={len(constraint_set)}")

    return best_a


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    print("  Solver: ECOS (fallback)")
    return cp.ECOS


def make_u_shaped(n, sharpness=2.0):
    """U-shaped sequence: more mass at edges."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 + sharpness * np.abs(x - 0.5) * 2.0
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_parametric(n, freq=1.0, mode='cos'):
    """Various parametric initial sequences."""
    x = np.linspace(0, 1, n)
    if mode == 'cos2':
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif mode == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif mode == 'flat':
        a = np.ones(n)
    elif mode == 'u':
        a = 1.0 + 3.0 * (x - 0.5) ** 2
    elif mode == 'w':
        a = 0.5 + np.abs(np.sin(2 * np.pi * x))
    elif mode == 'chirp':
        a = 1.0 / (1.0 + 5.0 * np.abs(x - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def smooth_gradient_descent(a, time_limit, tau_init_factor=0.3, tau_min_factor=0.01,
                            lr_init=0.5, decay=0.9998, verbose=True):
    """Smoothed gradient descent with adaptive tau and restarts."""
    n = len(a)
    a_curr = a.copy()
    C = fast_conv(a_curr)
    best_m = C.max()
    best_a = a_curr.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    tau_max = C.max()
    tau = tau_init_factor * tau_max
    tau_min = tau_min_factor * tau_max

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    t_step = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t_step += 1
        C = fast_conv(a_curr)
        M = C.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = a_curr.copy()
            stall = 0
        else:
            stall += 1

        tau = max(tau * decay, tau_min)
        g = smooth_grad(a_curr, C, tau)

        m_hat = beta1 * m_hat + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * (g * g)
        bc1 = 1.0 - beta1 ** t_step
        bc2 = 1.0 - beta2 ** t_step
        m_c = m_hat / bc1
        v_c = v_hat / bc2

        lr = lr_init / (1.0 + 0.1 * t_step)
        step = lr / (np.sqrt(v_c) + 1e-8)

        # Multiplicative update
        log_a = np.log(np.maximum(a_curr, 1e-30)) - step * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        a_curr = a_new

        if stall > 2000:
            # Restart with perturbation
            pert = rng.normal(0, 0.05, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a_curr = x
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            t_step = 0
            stall = 0
            tau = tau_init_factor * fast_conv(a_curr).max()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    GradDescent restart ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")

        if verbose and t_step % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradDescent t={t_step} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")

    return best_a


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multiple strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            a_list = [float(x) for x in a_norm]
            sc = evaluate_sequence(a_list)
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(x) for x in (parent / parent.sum())])
            if np.isfinite(p_score):
                best_seq = parent / parent.sum()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_u_shaped(n_base, sharpness=1.5)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Try diverse initial conditions ===
    print(f"\n=== Phase 1: Diverse inits ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 10:
        inits = [
            ('u_sharp', make_u_shaped(n0, 3.0)),
            ('u_med', make_u_shaped(n0, 1.5)),
            ('cos2', make_parametric(n0, mode='cos2')),
            ('w', make_parametric(n0, mode='w')),
            ('chirp', make_parametric(n0, mode='chirp')),
        ]
        for label, cand in inits:
            if time.monotonic() - t_global > time_budget - t1:
                break
            check_update(cand, f"init_{label}")
        # Blend parent with U-shaped
        for w in [0.3, 0.5, 0.7]:
            blend = w * best_seq + (1 - w) * make_u_shaped(n0, 2.0)
            check_update(blend, f"blend_{w}")

    n_best = len(best_seq)

    # === Phase 2: Smoothed gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Smoothed GradDescent (n={n_best}) ===")
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 30:
        a_out = smooth_gradient_descent(best_seq, t2, tau_init_factor=0.2,
                                        tau_min_factor=0.005, lr_init=0.6,
                                        decay=0.9997, verbose=True)
        check_update(a_out, "grad_n{:d}".format(n_best))

    # === Phase 3: Spectral shaping iterations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral Shaping ===")
    if remaining > 30:
        a_spec = best_seq.copy()
        for it in range(5):
            if time.monotonic() - t_global > time_budget - 60:
                break
            for flatness in [0.5, 0.7, 0.85]:
                a_try = spectral_shape(a_spec, n_iter=2, flatness=flatness)
                check_update(a_try, "spec_{:d}_f{:.2f}".format(it, flatness))
                if best_seq is not None and (a_spec is best_seq or not np.allclose(a_spec, best_seq)):
                    a_spec = best_seq.copy()
            # Also try gradient step after spectral
            a_grad = smooth_gradient_descent(a_spec, min(remaining * 0.05, 30.0),
                                             tau_init_factor=0.1, tau_min_factor=0.01,
                                             lr_init=0.8, decay=0.999, verbose=False)
            check_update(a_grad, "spec_grad_{:d}".format(it))

    # === Phase 4: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_best}) ===")
    if remaining > 100 and n_best <= 5500:
        t4 = min(remaining * 0.25, 400.0)
        a_out = lp_frank_wolfe(best_seq, t4, solver, M_quantile=0.90,
                                max_K=3000, verbose=True)
        check_update(a_out, "lpfw_n{:d}".format(n_best))

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord Descent (n={n_best}) ===")
    if remaining > 30:
        t5 = min(remaining * 0.12, 120.0)
        C_best = fast_conv(best_seq)
        a_out = coord_descent_exact(best_seq, C_best, n_top=12,
                                     max_iters=500, time_limit=t5)
        check_update(a_out, "coord_n{:d}".format(n_best))

    # === Phase 6: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    if remaining > 150:
        for new_n in [n_best - 500, n_best + 500, n_best - 1000]:
            if time.monotonic() - t_global > time_budget - 100:
                break
            if new_n < 2000 or new_n > 6000:
                continue
            a_interp = interpolate_seq(best_seq, new_n)
            check_update(a_interp, "interp_n{:d}".format(new_n))
            # Quick gradient on new length
            t6 = min(remaining * 0.15, 100.0)
            if t6 > 20:
                a_out = smooth_gradient_descent(a_interp, t6, tau_init_factor=0.15,
                                                 tau_min_factor=0.005, lr_init=0.5,
                                                 decay=0.999, verbose=False)
                check_update(a_out, "grad_n{:d}".format(new_n))
            remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Final intensive LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 80 and n_best <= 5500:
        t7 = min(remaining * 0.35, 250.0)
        a_out = lp_frank_wolfe(best_seq, t7, solver, M_quantile=0.85,
                                max_K=4000, verbose=True)
        check_update(a_out, "lpfw_final_n{:d}".format(n_best))

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Coord Polish ===")
    if remaining > 20:
        C_best = fast_conv(best_seq)
        a_out = coord_descent_exact(best_seq, C_best, n_top=15,
                                     max_iters=300, time_limit=min(remaining * 0.6, 60.0))
        check_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        final_score = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(final_score):
            best_score = final_score
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        best_seq = make_u_shaped(n_fb, 2.0)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Fallback: score={best_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
