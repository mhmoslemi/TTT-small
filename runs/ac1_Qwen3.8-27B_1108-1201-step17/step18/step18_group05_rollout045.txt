def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_init_v2(n, n_modes=50, seed=0):
    """Spectral initialization with more modes and structured frequencies."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3

        a += coeff_s * np.sin(2 * np.pi * freq * x + phase)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_grad_smooth(a, ac, top_k=100, beta=8.0):
    """Compute smoothed gradient of max autocorrelation."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    # Geometric weighting: heavier on the top peak
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # ∂(a*a)[k] / ∂a[i] = 2*a[k-i] if 0 <= k-i < n
        i_vals = np.arange(n)
        j_vals = k_int - i_vals
        valid = (j_vals >= 0) & (j_vals < n)
        if valid.any():
            grad[valid] += 2.0 * wi * a[j_vals[valid]]
    return grad

def compute_grad_equalize(a, ac, top_k=50):
    """Gradient that tries to equalize the top peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    mean_val = top_vals.mean()
    # Weight proportional to (peak - mean) to push down the highest
    w = np.maximum(top_vals - mean_val, 0)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_vals = np.arange(n)
        j_vals = k_int - i_vals
        valid = (j_vals >= 0) & (j_vals < n)
        if valid.any():
            grad[valid] += 2.0 * wi * a[j_vals[valid]]
    return grad

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + momentum * (a - norm_sum1(np.exp(log_theta - np.max(log_theta)) / np.sum(np.exp(log_theta - np.max(log_theta)))))
        
        # Actually, let me use a simpler formulation:
        # Work directly in a-space with projection
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Compute gradient
        g = compute_grad_smooth(a, ac, top_k=best_top_k, beta=best_beta)
        
        # Project gradient onto tangent space (sum to zero)
        g_tangent = g - g.mean()
        
        # Line search
        best_step = lr
        best_obj = M_curr
        for alpha_mult in [1.0, 0.5, 0.25, 0.1, 0.5]:
            alpha = lr * alpha_mult
            a_trial = a - alpha * g_tangent
            a_trial = project_simplex(a_trial)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_obj:
                best_obj = m_trial
                best_step = alpha
        
        a_new = a - best_step * g_tangent
        a_new = project_simplex(a_new)
        
        # Update momentum
        v = momentum * v + (a - a_new)
        a_momentum = a + v
        a_momentum = project_simplex(a_momentum)
        
        # Accept if better
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Adaptive parameters
        if stall > 500:
            lr *= 0.8
            best_beta = max(best_beta * 0.9, 3.0)
            best_top_k = min(best_top_k + 20, 300)
            if stall > 2000:
                # Restart with perturbation
                pert = rng.normal(0, 0.02, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-10)
                a_r = a_r / a_r.sum()
                a = a_r
                v = np.zeros(n)
                lr = lr_init
                stall = 0
                best_beta = 8.0
                best_top_k = 100
        
        total_iters += 1
        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={total_iters} ({elapsed:.0f}s): J={J:.6f}, lr={lr:.4f}, beta={best_beta:.1f}, K={best_top_k}, stall={stall}")
    
    return best_a, best_m

def peak_equalization(a0, time_limit, verbose=True, rng_seed=100):
    """
    Peak equalization: try to make the top peaks more uniform.
    Uses a different gradient that specifically targets the gap.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    lr = 0.05
    stall = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < 5000:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Try both smooth and equalize gradients
        g_smooth = compute_grad_smooth(a, ac, top_k=80, beta=10.0)
        g_equal = compute_grad_equalize(a, ac, top_k=50)
        
        # Use a blend
        g = 0.5 * g_smooth + 0.5 * g_equal
        g_tangent = g - g.mean()
        
        a_trial = a - lr * g_tangent
        a_trial = project_simplex(a_trial)
        m_trial = fast_autocorr(a_trial).max()
        
        if m_trial < best_m - 1e-15:
            best_m = m_trial
            best_a = a_trial.copy()
            stall = 0
            lr = min(lr * 1.01, 0.3)
        else:
            stall += 1
            if stall % 100 == 0:
                lr *= 0.9
            if stall > 2000:
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = np.maximum(best_a + pert, 0.0)
                a = a_p / a_p.sum()
                lr = 0.05
                stall = 1500
        
        a = a_trial
        total_iters += 1
        
        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakEq t={total_iters} ({elapsed:.0f}s): J={J:.6f}, lr={lr:.4f}, stall={stall}")
    
    return best_a, best_m

def coordinate_descent_v2(a0, time_limit, verbose=True, rng_seed=456):
    """
    Improved coordinate descent: for each top peak, try to reduce it
    by moving mass from contributing positions to non-contributing ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    total_rounds = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:30]
        
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            # Contributing indices: i where both i and ki-i are in [0, n)
            i_vals = np.arange(n)
            j_vals = ki - i_vals
            valid = (j_vals >= 0) & (j_vals < n)
            if not valid.any():
                continue
            
            valid_idx = i_vals[valid]
            valid_j = j_vals[valid]
            contrib = a[valid_idx] * a[valid_j]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the position with maximum contribution
            src_pos = valid_idx[np.argmax(contrib)]
            src_val = a[src_pos]
            if src_val < 1e-12:
                continue
            
            # Find positions NOT contributing to any top peak
            in_any_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                i2 = np.arange(n)
                j2 = k2i - i2
                v2 = (j2 >= 0) & (j2 < n)
                in_any_peak[i2[v2]] = True
                in_any_peak[j2[v2]] = True
            
            candidates = np.where(~in_any_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Try moving a small fraction to the least impactful position
            for frac in [0.02, 0.01, 0.05, 0.03, 0.005]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                # Try a few destinations (random + least loaded)
                dst_candidates = rng.choice(candidates, size=min(5, len(candidates)), replace=False)
                for dst in dst_candidates:
                    a_trial = a.copy()
                    a_trial[src_pos] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                break  # Restart the loop with new autocorrelation
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if a[src_pos if 'src_pos' in dir() else 0] > 1e-10:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
        
        total_rounds += 1
        if verbose and total_rounds % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord2 round={total_rounds} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def fista_optimize(a0, time_limit, verbose=True, top_k=100, beta=8.0, rng_seed=200):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) for the smoothed problem.
    Uses a carefully chosen smoothing parameter and backtracking.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # FISTA variables
    x = a.copy()
    z = a.copy()
    t_k = 1.0
    lr = 0.1  # Step size (1/L where L is Lipschitz constant)
    
    # Estimate L (Lipschitz constant of the gradient)
    # The gradient is 2*sum w_m * S_m(a), so L ~ 2 * sum w_m * max |a| ~ 2
    # Be conservative
    L = 4.0
    
    stall = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < 3000:
        # Compute gradient at z
        ac_z = fast_autocorr(z)
        g_z = compute_grad_smooth(z, ac_z, top_k=top_k, beta=beta)
        g_tangent = g_z - g_z.mean()
        
        # Gradient step
        y = z - (1.0 / L) * g_tangent
        y = project_simplex(y)
        
        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        x_new = y + ((t_k - 1.0) / t_new) * (y - z)
        x_new = project_simplex(x_new)
        
        # Check improvement
        m_new = fast_autocorr(x_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        z = x_new
        t_k = t_new
        x = x_new
        
        # Adaptive step size
        if stall > 200:
            L *= 1.2  # Decrease step size
        elif stall > 1000:
            # Restart
            z = best_a.copy()
            x = best_a.copy()
            t_k = 1.0
            L = 4.0
            stall = 0
        
        total_iters += 1
        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={total_iters} ({elapsed:.0f}s): J={J:.6f}, L={L:.1f}, stall={stall}")
    
    return best_a, best_m

def multi_peak_reduction(a0, time_limit, verbose=True, rng_seed=300):
    """
    Directly target the top peaks one by one.
    For each peak, compute the optimal 1D move that reduces it the most.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top 5 peaks
        top_idx = np.argsort(ac)[::-1][:5]
        
        improved = False
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            
            # For this peak, find the pair (i, ki-i) with maximum product
            i_vals = np.arange(n)
            j_vals = ki - i_vals
            valid = (j_vals >= 0) & (j_vals < n)
            if not valid.any():
                continue
            
            valid_i = i_vals[valid]
            valid_j = j_vals[valid]
            products = a[valid_i] * a[valid_j]
            
            if products.max() < 1e-20:
                continue
            
            # Try reducing the top contributing pair
            top_contrib_idx = np.argsort(products)[::-1][:5]
            
            for ci in top_contrib_idx:
                src_i = valid_i[ci]
                src_j = valid_j[ci]
                
                # Try moving mass from src_i to a random position far from any peak
                for frac in [0.03, 0.015, 0.05, 0.01]:
                    amt = a[src_i] * frac
                    if amt < 1e-15:
                        continue
                    # Choose destination: random position not contributing to top peaks
                    in_peak = np.zeros(n, dtype=bool)
                    for k2 in top_idx[:3]:
                        k2i = int(k2)
                        i2 = np.arange(n)
                        j2 = k2i - i2
                        v2 = (j2 >= 0) & (j2 < n)
                        in_peak[i2[v2]] = True
                        in_peak[j2[v2]] = True
                    
                    free_pos = np.where(~in_peak)[0]
                    if len(free_pos) == 0:
                        free_pos = np.arange(n)
                    dst = free_pos[rng.integers(len(free_pos))]
                    
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 500 == 0:
                # Bigger perturbation
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = np.maximum(best_a + pert, 0.0)
                a = a_p / a_p.sum()
                stall = 0
        
        total_iters += 1
        if verbose and total_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultiPeak t={total_iters} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def log_space_optimize(a0, time_limit, verbose=True, lr_init=0.05, rng_seed=500):
    """
    Optimization in log-space: a_i = exp(theta_i) / sum(exp(theta_j)).
    This naturally maintains positivity and can escape zero-valued positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Initialize theta
    theta = np.log(a + 1e-30)
    theta_v = np.zeros(n)  # Velocity for momentum
    
    lr = lr_init
    momentum = 0.9
    stall = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < 5000:
        # Current a from theta
        log_max = theta.max()
        exp_theta = np.exp(theta - log_max)
        a_curr = exp_theta / exp_theta.sum()
        
        # Compute gradient in a-space
        ac = fast_autocorr(a_curr)
        g_a = compute_grad_smooth(a_curr, ac, top_k=80, beta=8.0)
        g_a_tangent = g_a - g_a.mean()
        
        # Transform gradient to theta-space
        # da/dtheta_i = a_i (1 - a_i) for the softmax parameterization
        # So dtheta = g_a / (a * (1-a)) approximately, but we use a simpler approach
        # Just use the gradient directly as a direction in theta-space
        # The effect of increasing theta_i by delta is to increase a_i proportionally
        # So the "natural" gradient in theta-space is g_a / a (Fisher metric)
        g_theta = g_a_tangent / (a_curr + 1e-10)
        # Clip to prevent explosion
        g_theta = np.clip(g_theta, -10, 10)
        
        # Momentum update
        theta_v = momentum * theta_v + lr * g_theta
        theta_new = theta - theta_v
        
        # Convert back to a-space
        log_max_new = theta_new.max()
        exp_new = np.exp(theta_new - log_max_new)
        a_new = exp_new / exp_new.sum()
        
        # Check
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            theta = theta_new
            stall = 0
        else:
            stall += 1
            if stall > 500:
                lr *= 0.85
                momentum = min(momentum + 0.01, 0.95)
            if stall > 2000:
                # Restart
                a_r = best_a + rng.normal(0, 0.01, n) * best_a
                a_r = np.maximum(a_r, 1e-10)
                a_r = a_r / a_r.sum()
                theta = np.log(a_r)
                theta_v = np.zeros(n)
                lr = lr_init
                momentum = 0.9
                stall = 0
            else:
                theta = theta_new  # Still update to avoid cycling
        
        a = a_new
        total_iters += 1
        
        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LogSpace t={total_iters} ({elapsed:.0f}s): J={J:.6f}, lr={lr:.4f}, stall={stall}")
    
    return best_a, best_m
