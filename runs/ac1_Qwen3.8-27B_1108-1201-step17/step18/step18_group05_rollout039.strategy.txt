def fast_autocorr(a):
    """Evaluates autocorrelation using FFT."""
    n = len(a)
    # Precompute via linear convolution
    a = a / np.linalg.norm(a)  # Normalize for improved peak-ratio calculation
    autocorr = np.fft.irfft(np.fft.rfft(a) * np.fft.rfft(a))
    return autocorr

def norm_sum1(a):
    """Normalizes to unit simplex."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Projects onto the simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1] if any(u * np.arange(1, n+1) > (cssv - 1)) else n
    theta = (cssv[rho] - 1) / (rho + 1.0) if rho < n else 0.0
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(seq, n):
    """Interpolates a sequence to new length, normalizing appropriately."""
    old_n = len(seq)
    new_t = np.linspace(0, 1, n)
    old_t = np.linspace(0, 1, old_n)
    # Normalize via interpolation (ensuring positivity and not exceeding max norm)
    new_vals = np.interp(new_t, old_t, seq)
    new_vals = [max(0, x) for x in new_vals]  # Element-wise
    s = np.sum(new_vals)
    if s < 0.01:
        # Use uniform sequence as fallback
        new_vals = np.ones(n) / n * 0.99  # Maintain feasiility
        s = np.sum(new_vals)
    return new_vals / s if s > 0 else new_vals / (n * 0.1)  # Ensure creation
