import scipy.optimize as opt

_best_tracker = {'seq': None, 'score': float('inf')}

def fast_conv(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a)/len(a)
    return a/s

def lp_direction(a, top_k=30):
    """Solve LP to find best direction g."""
    n = len(a)
    conv_aa = fast_conv(a)
    M = conv_aa.max()
    
    # Get top k peaks
    top_peaks = np.argsort(conv_aa)[::-1][:top_k]
    
    # Build constraint matrix A_c (top_k x n)
    # (a * g)_k = sum_j a_{k-j} * g_j
    A_c = np.zeros((top_k, n))
    for i, k in enumerate(top_peaks):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_c[i, j] = a[k - j]
    
    # LP: max sum(g) s.t. A_c @ g <= M, g >= 0
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    res = opt.linprog(c, A_ub=A_c, b_ub=np.full(top_k, M), bounds=(0, None), method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s  # normalize to sum 1

def subgrad_optimize(a0, time_limit, seed=42, top_k_init=20, temp_init=0.004, c0_init=4.0):

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

        a = np.maximum(a, 0)
        inits.append(norm_sum1(a)); labels.append(f"sinc{bw}")

    return inits[:n_variants], labels[:n_variants]


def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 4500
        x = np.linspace(0, 1, n_base)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Default init: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    n0 = len(_best_tracker['seq'])
    print(f"Starting: n={n0}, score={_best_tracker['score']:.6f}")

    # === Phase 1: LP-based optimization on parent (primary method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Optimize (n={n0}) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = lp_optimize(a_in, t1, seed=100, init_top_k=15, max_top_k=80, t_init=0.35)
        try_update(a_out, "lp_parent")

    # === Phase 2: Subgradient on diverse inits ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Subgradient diverse inits (n={n0}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 20:
        inits, labels = make_variadic_inits(n0, n_variants=8)
        if parent is not None:
            inits.insert(0, norm_sum1(parent))
            labels.insert(0, "parent")

        # Deduplicate
        seen = set()
        uniq_inits = []
        uniq_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 20)], 3))
            if sig not in seen:
                seen.add(sig)
                uniq_inits.append(init)
                uniq_labels.append(lbl)

        n_inits = len(uniq_inits)
        t_per = max(10.0, t2 / n_inits)
        for idx in range(n_inits):
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            lbl = uniq_labels[idx]
            print(f"  Subgrad '{lbl}' t={t_avail:.0f}s")
            a_out, _ = subgrad_optimize(uniq_inits[idx], t_avail, seed=200 + idx,
                                        top_k_init=20, temp_init=0.004, c0_init=4.0)
            try_update(a_out, f"sg_{lbl}")

    # === Phase 3: Coordinate descent polish on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord Descent (n={len(_best_tracker['seq'])}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 10:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = coord_descent(a_in, t3, seed=42)
        try_update(a_out, "cd1")

    # === Phase 4: LP refine on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 4: LP Refine (n={n_best}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = lp_optimize(a_in, t4, seed=300, init_top_k=25, max_top_k=120, t_init=0.25)
        try_update(a_out, "lp_refine")

    # === Phase 5: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 5: Multi-n (base n={n_best}) ===")
    candidate_ns = []
    for delta in [-1000, -500, 500, 1000]:
        new_n = n_best + delta
        if 2500 <= new_n <= 8000:
            candidate_ns.append(new_n)

    for new_n in candidate_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  n={new_n}")
        a_interp = interpolate_seq(_best_tracker['seq'], new_n)
        try_update(a_interp, f"intp{new_n}")

        # LP on interpolated
        t5a = min(remaining * 0.35, 180.0)
        if t5a > 15:
            a_out, _ = lp_optimize(a_interp, t5a, seed=500 + new_n,
                                   init_top_k=15, max_top_k=60, t_init=0.3)
            try_update(a_out, f"lp_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.25, 100.0)
            if t5b > 10:
                a_out, _ = subgrad_optimize(a_interp, t5b, seed=700 + new_n,
                                            top_k_init=25, temp_init=0.003, c0_init=3.5)
                try_update(a_out, f"sg_n{new_n}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(_best_tracker['seq'])
    print(f"\n=== Phase 6: Final Polish (n={n_final}) ===")
    if remaining > 40:
        t6a = min(remaining * 0.45, 200.0)
        if t6a > 15:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = lp_optimize(a_in, t6a, seed=999, init_top_k=30, max_top_k=150, t_init=0.2)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.45, 150.0)
            if t6b > 10:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = coord_descent(a_in, t6b, seed=31415)
                try_update(a_out, "cd_final")

    # Final check
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
