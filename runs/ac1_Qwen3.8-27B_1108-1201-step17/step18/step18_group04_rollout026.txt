import numpy as np
import time

def fast_conv(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives to 0."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def logsumexp_max(a, t):
    """Smooth approximation to max(a): t * log(sum(exp(a/t)))."""
    a_max = a.max()
    if a_max < -1e30:
        return 0.0
    e = np.exp((a - a_max) / t)
    return a_max + t * np.log(e.sum())


def subgrad_max_conv(a, conv, temperature=0.0, top_k=50):
    """
    Compute subgradient of max_k conv[k] w.r.t. a.
    If temperature > 0, use softmax-weighted combination near the max.
    If temperature = 0, use the exact subgradient of the argmax.
    """
    n = len(a)
    conv_max = conv.max()
    
    if temperature > 0:
        # Softmax-weighted subgradient
        threshold = conv_max - max(temperature * 10.0, 1e-10)
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        if len(active) > top_k:
            order = np.argsort(conv[active])[::-1][:top_k]
            active = active[order]
        w = np.exp((conv[active] - conv_max) / temperature)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(active)) / len(active)
        else:
            w = w / w_sum
    else:
        active = np.array([np.argmax(conv)])
        w = np.array([1.0])
    
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad


def amd_optimize_v2(a0, time_limit, seed=42, verbose=False):

    Improved AMD with:
    - Log-sum-exp annealing (smooth to sharp)
    - Adaptive temperature
    - Better restart strategy

    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Annealing schedule: temperature for log-sum-exp
    # Start smooth (large t), end sharp (small t)
    t_lse_start = 0.05
    t_lse_end = 0.001
    # Subgradient temperature
    sg_temp_start = 0.005
    sg_temp_end = 0.0005
    # Step size parameter
    c0 = 2.5
    top_k = 30
    
    stall_count = 0
    restart_count = 0
    restart_every = 3000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing progress (0 to 1)
        progress = min(1.0, t / (time_limit * 200))  # adjust based on expected iterations
        
        # Interpolate temperatures
        t_lse = t_lse_start * (t_lse_end / t_lse_start) ** progress
        sg_temp = sg_temp_start * (sg_temp_end / sg_temp_start) ** progress
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)

        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Step size

        # Mirror descent step (exponential update)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()

            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Check improvement
        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart on stall

            top_k = min(top_k + 3, 60)

def coord_descent_v2(a0, time_limit, seed=42, verbose=False):

    - Use ternary search for the optimal transfer amount
    - Consider top peaks in the cost computation

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        
        # Compute cost for each coordinate (how much it contributes to peaks)
        n_top_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_top_peaks]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]
        
        # Find the coordinate with highest cost (most "expensive")
        # and try to move some mass to a low-cost coordinate
        src_candidates = np.argsort(costs)[::-1][:max(10, n // 200)]

        improved = False
        for src_i in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src_i)
            if a[src_i] < 1e-12:
                continue
            
            for dst_i in dst_candidates:
                dst_i = int(dst_i)
                if dst_i == src_i:
                    continue

                    if M_trial < best_trial_m:

                    a[src_i] -= amt

                    best_m = best_trial_m

                    improved = True

                M_new = fast_conv(a).max()

def perturbation_search(a0, time_limit, seed=42, verbose=False):
    """
    Random perturbation local search.
    Try many small random transfers and keep improvements.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    n_tries = 0
    while time.monotonic() - t0 < time_limit:
        # Batch of random perturbations
        batch_size = min(1000, max(100, int((time_limit - (time.monotonic() - t0)) * 50)))
        
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            n_tries += 1
            
            # Pick a random source with mass
            # Weight by current value to prefer moving from larger entries
            weights = np.sqrt(a + 1e-15)
            weights /= weights.sum()
            src_i = int(rng.choice(n, p=weights))
            dst_i = int(rng.integers(n))
            if src_i == dst_i:
                continue
            
            frac = rng.uniform(0.001, 0.05)
            amt = a[src_i] * frac
            a_trial = a.copy()
            a_trial[src_i] -= amt
            a_trial[dst_i] += amt
            a_trial = np.maximum(a_trial, 0.0)

            if s_t < 1e-15:

            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
    
    if verbose:
        print(f"    Perturbation: {n_tries} tries, final M={best_m:.10f}")
    
    return best_a, best_m


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def make_raised_cosine(n, power=1.0):

def make_gaussian(n, sigma=0.3):

    return norm_sum1(a)


def make_power_decay(n, power=0.5):

    a = np.maximum(0, 1 - (x / frac) ** power)

        a = make_gaussian(n, s)
        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    
    # Try power decay
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        if time.monotonic() - t0 > time_limit * 0.7:
            break
        a = make_power_decay(n, p)
        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    
    # Try mixed Gaussians
    for p1, p2, w, ratio in [(0.25, 0.75, 0.12, 1.0), (0.3, 0.7, 0.15, 0.8),
                              (0.2, 0.8, 0.1, 1.2), (0.35, 0.65, 0.12, 0.7)]:
        if time.monotonic() - t0 > time_limit:
            break
        a = make_mixed_gaussians(n, p1, p2, w, w, ratio)
        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    
    # Try flat top
    for frac, power in [(0.4, 2), (0.5, 2), (0.6, 3), (0.5, 4), (0.7, 2)]:
        if time.monotonic() - t0 > time_limit:
            break
        a = make_flat_top(n, frac, power)
        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    
    return best_a, best_m


def try_update(a, label):
    """Track best sequence."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
