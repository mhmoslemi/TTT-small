            print("Minimal initialization: score", J0)
            n_fb = n_base

    n_best = len(a0)
    best_score = J0

    # --- 2. Primary Optimization (Asymmetric Step Function) ---
    time_remaining = time_limit - 30  # Reserve 30 seconds for backup steps

    if time_remaining <= 0:

    # Track best candidate during gradient-based refinement
    print("\n+++ Asymmetric Optimization Phase +++")
    alpha_range = [0.3, 0.5, 0.7, 0.9]

def a_optimize_asymmetric(n_base, n_new, alpha_vals):
    """Generate a candidate asymmetric sequence and evaluate adaptively"""
    # Create staircase-like structure with optimized step boundaries
    asym_a = np.ones(n_base)  # Base shape
    
    # Apply asymmetric pulse mapping to ensure minimal autocorrelation
    shift_amount = (n_base - n_new) // 4
    asym_a[:shift_amount] *= 0.65  # Dampen leading edge
    asym_a[shift_amount:] = symmetry_transform(shift_amount, n_base) + 0.05
    
    # Normalize and evaluate
    asym_a_norm = norm_sum1(asym_a)
    asym_a_norm = asym_a_norm / asym_a_norm.sum() * np.sqrt(2 * n_new)
    
    J_candidate = evaluate_safe(asym_a_norm)
    return asym_a_norm

def symmetry_transform(i, n):
    """Smoothly adjust symmetry properties"""
    if i < n//2:
        return np.exp(-2 * np.log(0.5) * (i - n//3))
    else:
        return np.exp(-3 * np.log(0.6) * (i - n//2))

def coordinate_descent_centroidal(a_init, time_limit):
    """Coordinate descent method targeting powerful sequence families"""
    # Generate time-frequency features for multiple sequence families
    n = len(a_init)
    best_sequence = a_init.copy()
    best_score = float('inf')
    
    print("Centroidal coordinate descent attempting families...")
    families_to_try = [
        (np.sin, 1.0, 0.4),
        (np.cos, 0.5, 0.2),
        (np.exp, -4, 0.3),
        (np.log, 0.7, 0.1),
        (np.tanh, 2.0, 0.2),
        (lambda x: np.gammaln(x+1, scale=0.5), 0.3, 0.4)
    ]
    
    for tfun, scale, offset in families_to_try:
        try:
            x = np.linspace(0.0, 1.0, n)
            family_a = tfun(np.ones(n)) + offset
            # Apply transformation and normalization
            transformed_a = family_a * np.exp(-scale * np.abs(x - 0.5))
            
            # Ensure segment continuity
            transformed_a = np.maximum(transformed_a, 0.01)
            transformed_a = norm_sum1(transformed_a)
            s_trans = transformed_a.sum()
            if s_trans >= 1e-10:
                plt_success = evaluate_safe(transformed_a) < best_score
                if plt_success:
                    # Target more refined centroidal transformation
                    print("Family candidate found improvement:", plt_success)
                    best_score = plt_success
                    best_sequence = transformed_a.copy()
                
                # Check surrounding neighborhoods for supporting transformations
                x_neighbor = np.linspace(0.0, 1.0, n+100)
                transformed_a_large = tfun(np.ones(len(x_neighbor))) * np.exp(-scale * np.abs(x_neighbor - 0.5))
                transformed_a_large = np.maximum(transformed_a_large, 0.01)
                transformed_a_large = transform_lengths(
                    transformed_a_large, 2.0, n//2
                )
                
                best_score_large = evaluate_safe(transformed_a_large)
                if best_score_large <= best_score:

                                             np.linspace(0, 1, len(best_sequence)),
                                             transformed_a_large)
                    best_score = best_score_large
                
        except Exception as e:
            print("Symmetry transform failed:", str(e))
            
    return best_sequence, best_score
