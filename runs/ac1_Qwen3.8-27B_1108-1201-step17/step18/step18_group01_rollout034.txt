import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def autocorr_max(a):
    """Compute max autocorrelation for normalized sequence."""
    ac = fast_autocorr(a)
    return ac.max(), ac

def mirror_descend_step(a, grad, eta):
    """One mirror descent step: a' = a * exp(-eta * grad) / sum(...)"""
    a_new = a * np.exp(-eta * grad)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

def logsumexp_gradient(a, ac, T, top_k=50):
    """
    Compute gradient of the logsumexp-smoothed max autocorrelation.
    Returns the gradient vector and the smoothed objective value.
    """
    n = len(a)
    k_use = min(top_k, len(ac))
    
    # Take top k values for computational efficiency
    top_idx = np.argsort(ac)[::-1][:k_use]
    top_vals = ac[top_idx]
    
    # Logsumexp of top values
    max_v = top_vals.max()
    exp_vals = np.exp((top_vals - max_v) / T)
    w = exp_vals / exp_vals.sum()
    
    # Weighted gradient
    grad = np.zeros(n)
    for wi, k_idx in zip(w, top_idx):
        ki = int(k_idx)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    
    # Smoothed objective (for tracking)
    smooth_obj = max_v + T * np.log(np.exp((top_vals - max_v) / T).sum() / k_use)
    
    return grad, smooth_obj

    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            patience = 0

            patience += 1

        # Compute smoothed gradient
        grad, _ = logsumexp_gradient(a, ac, T, top_k=top_k)
        
        # Center the gradient (remove mean to preserve sum constraint direction)
        grad = grad - grad.mean()
        
        gmax = np.abs(grad).max()
        if gmax < 1e-15:

        grad = grad / gmax
        
        # Backtracking line search
        accepted = False
        eta_try = eta

    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(m, n))
    b = np.full(m, M)
    
    # Maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    
    if verbose:
        print(f"      LP: n={n}, constraints={m}, nnz={A.nnz}")
        t_lp = time.monotonic()

            t_lp_elapsed = time.monotonic() - t_lp
            print(f"      LP solved in {t_lp_elapsed:.1f}s, status={res.status}")
        
        if res.status == 0:
            g = res.x
            if g is not None and g.max() > 1e-10:
                return g
        else:
            if verbose:
                print(f"      LP failed: {res.message}")
    except Exception as e:
        if verbose:
            print(f"      LP error: {e}")
    
    return None

def lp_blend_step(f, time_limit, blend_alpha=0.1, verbose=True):
    """
    One LP-based blending step: compute LP direction, blend with current f.
    """
    n = len(f)
    g = lp_direction(f, verbose=verbose)
    
    if g is None:
        return f.copy(), fast_autocorr(f).max()
    
    g = norm_sum1(g)
    
    # Try different blend factors
    best_f = f.copy()
    best_M = fast_autocorr(f).max()
    
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3]:
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        M_new = fast_autocorr(f_new).max()
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
    
    return best_f, best_M

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coord_polish(a0, time_limit, verbose=True):
    """
    Coordinate descent polish: move mass from peak-contributing positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False
        
        # Top contributing lags
        n_top = min(8, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        for k in top_k[:4]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination: position that contributes least to peaks
            peak_contrib = np.zeros(n)
            for k2 in top_k[:4]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]
            
            candidates = np.where(peak_contrib < peak_contrib.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(peak_contrib[candidates])]
            if dst == src:
                continue
            
            for frac in [0.01, 0.02, 0.04, 0.06]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-15:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"      Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a

def parametric_init(n, best_a, best_score):
    """Try various parametric families for initialization."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    results = []
    
    families = [
        # Sine variants
        np.sin(np.pi * x) ** 2,
        np.sin(np.pi * x) ** 4,
        np.sin(0.5 * np.pi * x) ** 2,
        # Raised cosine
        0.5 * (1 + np.cos(2 * np.pi * x)),
        # Uniform
        np.ones(n),
        # Triangular
        1 - np.abs(2 * x - 1),
        # Power laws
        (i_arr + 1) ** 0.5,
        (i_arr + 1) ** 1.5,
        (n - i_arr).clip(1) ** 0.5,
        # Exponential decay
        np.exp(-2 * x),
        np.exp(-5 * x),
        np.exp(2 * (x - 1)),
        # Gaussian bumps
        np.exp(-((x - 0.3) ** 2) / 0.02),
        np.exp(-((x - 0.7) ** 2) / 0.02),
        np.exp(-((x - 0.5) ** 2) / 0.05),
        # Double bump
        np.exp(-((x - 0.25) ** 2) / 0.03) + np.exp(-((x - 0.75) ** 2) / 0.03),
        # Window functions
        0.4 + 0.6 * np.cos(2 * np.pi * (x - 0.5)),  # Hanning-like
        # Prolate-like (approximation)
        np.sinc(2 * x - 1) ** 2,
    ]
    
    best_local = best_a.copy()
    best_local_score = best_score
    
    for fam in families:
        fam = np.maximum(fam, 0.0)
        fam_norm = norm_sum1(fam)
        ac = fast_autocorr(fam_norm)
        val = 2.0 * n * ac.max()
        if val < best_local_score:
            best_local_score = val
            best_local = fam_norm.copy()
    
    return best_local, best_local_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton (flat-spectrum) optimization in the frequency domain
2. Mirror descent (MGD) with logsumexp-smoothed subgradient
3. LP-based direction finding (maximize sum(g) s.t. ||f*g||_inf <= M)
4. Multi-length exploration via interpolation
5. Coordinate descent polish
The GS approach is fundamentally different from the previous MGD-only approach,
exploiting the Wiener-Khinchin theorem: flat |DFT|^2 => flat autocorrelation.

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def gs_optimize(a0, time_limit, verbose=True):
    """
    Gerchberg-Saxton / error reduction for flat spectrum.
    Minimizes max autocorrelation by driving |DFT|^2 toward flatness.
    """
    n = len(a0)
    a = norm_sum1(a0)
    N = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    max_stall = 200

        # 1 + 2*(N/2-1)*C^2 + C^2 = N*a2_sum
        # 1 + (N-1)*C^2 = N*a2_sum
        C2 = (N * a2_sum - 1.0) / (N - 1.0)
        if C2 < 1e-15:
            C2 = a2_sum * 0.5
        C = math.sqrt(C2)

        target_mag = np.full(N // 2 + 1, C)

        phase = np.exp(1j * np.angle(A))

        print(f"      GS done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def gs_relaxed(a0, time_limit, relaxation=0.7, verbose=True):
    """
    Relaxed GS: blend target magnitude with current magnitude.
    Often converges better than hard GS.
    """
    n = len(a0)
    a = norm_sum1(a0)
    N = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        A = np.fft.rfft(a, N)
        cur_mag = np.abs(A)
        phase = np.exp(1j * np.angle(A))

        a2_sum = float(np.sum(a * a))
        C2 = (N * a2_sum - 1.0) / (N - 1.0)
        if C2 < 1e-15:
            C2 = a2_sum * 0.5
        C = math.sqrt(C2)

        target_mag = np.full(N // 2 + 1, C)
        target_mag[0] = a.sum()

        # Relaxed update: blend current and target magnitudes
        new_mag = relaxation * target_mag + (1.0 - relaxation) * cur_mag
        new_A = new_mag * phase

        a_new = np.fft.irfft(new_A, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Adapt relaxation when stalling
            if stall % 50 == 0:
                relaxation = max(0.3, relaxation * 0.9)

        a = a_new
        iter_count += 1

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"      GS_relax iter={iter_count} ({elapsed:.0f}s"): J={J:.6f}, r={relaxation:.2f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"      GS_relax done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def logsumexp_gradient(a, ac, T, top_k=50):
    """
    Compute gradient of logsumexp-smoothed max autocorrelation.
    """
    n = len(a)
    k_use = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_use]
    top_vals = ac[top_idx]

    max_v = top_vals.max()
    exp_vals = np.exp((top_vals - max_v) / T)
    w = exp_vals / exp_vals.sum()

    grad = np.zeros(n)
    for wi, k_idx in zip(w, top_idx):
        ki = int(k_idx)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[ki - idx]

    return grad


def mgd_optimize(a0, time_limit, T_init=0.003, T_min=0.0003, top_k=40, eta0=0.05, verbose=True):
    """
    Mirror Gradient Descent with logsumexp-smoothed gradient and adaptive step size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = eta0
    T = T_init
    patience = 0
    max_patience = 500
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        grad = logsumexp_gradient(a, ac, T, top_k=top_k)
        grad = grad - grad.mean()

        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        # Backtracking line search
        accepted = False
        eta_try = eta
        for _ in range(8):
            a_new = a * np.exp(-eta_try * grad)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            M_new = fast_autocorr(a_new).max()
            if M_new <= M + 1e-13:
                a = a_new
                accepted = True
                break
            eta_try *= 0.5

        if not accepted:
            eta *= 0.5
            if eta < 0.0005:
                break
        else:
            T = max(T_min, T * 0.997)
            if M < best_M:
                eta = min(eta * 1.005, 0.15)

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"      MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, T={T:.5f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"      MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def lp_direction(f, verbose=False):
    """
    LP-based direction: maximize sum(g) s.t. (f*g)[k] <= M for active k, g >= 0.
    """
    n = len(f)
    ac = fast_autocorr(f)
    M = ac.max()

    if M < 1e-15:
        return None

    # Use constraints where ac[k] >= 0.75*M
    threshold = 0.75 * M
    active_k = np.where(ac >= threshold)[0]
    m = len(active_k)

    if m == 0:
        active_k = np.array([np.argmax(ac)])
        m = 1

    # Build sparse constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []

    for j in range(m):
        k = int(active_k[j])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_end > i_start:
            rows_list.extend([j] * (i_end - i_start))
            cols_list.extend(range(i_start, i_end))
            vals_list.extend(f[i_start:i_end].tolist())

    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(m, n))
    b = np.full(m, M)
    c = -np.ones(n)

    if verbose:
        print(f"      LP: n={n}, constraints={m}, nnz={len(vals_list)}")
        t_lp = time.monotonic()

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if verbose:
            print(f"      LP done in {time.monotonic()-t_lp:.1f}s, status={res.status}")
        if res.status == 0 and res.x is not None:
            g = res.x
            if g.sum() > 1e-10:
                return g
    except Exception as e:
        if verbose:
            print(f"      LP error: {e}")

    return None


def lp_blend_step(f, verbose=True):
    """One LP-based blending step."""
    n = len(f)
    g_raw = lp_direction(f, verbose=verbose)

    if g_raw is None:
        return f.copy(), fast_autocorr(f).max()

    g = norm_sum1(g_raw)
    best_f = f.copy()
    best_M = fast_autocorr(f).max()

    for alpha in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3]:
        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        M_new = fast_autocorr(f_new).max()
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()

    return best_f, best_M


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coord_polish(a0, time_limit, verbose=True):
    """Coordinate descent: move mass from peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False
        n_top = min(6, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        for k in top_k[:3]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-impact destination
            peak_contrib = np.zeros(n)
            for k2 in top_k[:3]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]

            candidates = np.where(peak_contrib < peak_contrib.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(peak_contrib[candidates])]
            if dst == src:
                continue

            for frac in [0.01, 0.02, 0.04, 0.06, 0.08]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-15:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"      Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def parametric_init(n, best_a, best_score):
    """Try parametric families for initialization."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    families = [
        np.sin(np.pi * x) ** 2,
        np.sin(np.pi * x) ** 4,
        0.5 * (1 + np.cos(2 * np.pi * x)),
        np.ones(n),
        1 - np.abs(2 * x - 1),
        (i_arr + 1) ** 0.3,
        (n - i_arr).clip(1) ** 0.3,
        np.exp(-3 * x),
        np.exp(-3 * (x - 1)),
        np.exp(-((x - 0.3) ** 2) / 0.03) + np.exp(-((x - 0.7) ** 2) / 0.03),
        np.exp(-((x - 0.5) ** 2) / 0.08),
        0.4 + 0.6 * np.cos(2 * np.pi * (x - 0.5)),
        np.sinc(1.5 * x - 0.75) ** 2,
        np.maximum(np.sin(np.pi * x) * np.sin(2 * np.pi * x) + 0.5, 0.0) ** 2,
    ]

    best_local = best_a.copy()
    best_local_score = best_score

    for fam in families:
        fam = np.maximum(fam, 0.0)
        s = fam.sum()
        if s < 1e-10:
            continue
        fam_norm = fam / s
        ac = fast_autocorr(fam_norm)
        val = 2.0 * n * ac.max()
        if val < best_local_score:
            best_local_score = val
            best_local = fam_norm.copy()

    return best_local, best_local_score


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining GS, MGD, LP, multi-length, and coordinate polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = norm_sum1(p)
            p_score = evaluate_safe(p_norm)
            if np.isfinite(p_score):
                best_seq = p_norm
                best_score = p_score
                print(f"Parent: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 4500
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Parametric search ===
    print(f"\n=== Phase 1: Parametric (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 30.0)
    if t1 > 8:
        a_p, s_p = parametric_init(n0, best_seq, 2.0 * n0 * fast_autocorr(best_seq).max())
        sc_p = evaluate_safe(a_p)
        if np.isfinite(sc_p) and sc_p < best_score:
            best_score = sc_p
            best_seq = a_p.copy()
            print(f"  Parametric improved to {sc_p:.6f}")

    # === Phase 2: GS optimization (fast, many iterations) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: GS optimization (n={n_best}) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 20:
        a_gs = gs_optimize(norm_sum1(best_seq), t2, verbose=True)
        sc = evaluate_safe(a_gs)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_gs.copy()
            print(f"  *** NEW BEST GS: {sc:.6f}")

    # === Phase 3: Relaxed GS ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Relaxed GS (n={n_best}) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 20:
        a_gs2 = gs_relaxed(norm_sum1(best_seq), t3, relaxation=0.6, verbose=True)
        sc = evaluate_safe(a_gs2)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_gs2.copy()
            print(f"  *** NEW BEST GS-relax: {sc:.6f}")

    # === Phase 4: MGD polish after GS ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: MGD after GS (n={n_best}) ===")
    t4 = min(remaining * 0.25, 300.0)
    if t4 > 20:
        a_mgd = mgd_optimize(norm_sum1(best_seq), t4, T_init=0.003, T_min=0.0003,
                            top_k=30, eta0=0.04, verbose=True)
        sc = evaluate_safe(a_mgd)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_mgd.copy()
            print(f"  *** NEW BEST MGD: {sc:.6f}")

    # === Phase 5: LP-based direction (if n is manageable) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP direction (n={n_best}) ===")
    if remaining > 150 and n_best <= 5500:
        t5 = min(remaining * 0.15, 180.0)
        t5_start = time.monotonic()
        a_lp = norm_sum1(best_seq).copy()

        for lp_iter in range(3):
            if time.monotonic() - t5_start > t5:
                break
            print(f"    LP iteration {lp_iter + 1}...")
            a_lp, M_lp = lp_blend_step(a_lp, verbose=True)

    # === Phase 6: Multi-length exploration ===

    for new_n in length_candidates[:3]:

        print(f"    Trying n={new_n}...")

            sc = evaluate_safe(a_gs_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_gs_new.copy()
                print(f"  *** NEW BEST GS n={new_n}: {sc:.6f}")

        # MGD on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.25, 80.0)
            if t6b > 15:
                a_mgd_new = mgd_optimize(norm_sum1(best_seq), t6b, T_init=0.002, T_min=0.0002,
                                        top_k=25, eta0=0.03, verbose=False)
                sc = evaluate_safe(a_mgd_new)
                if np.isfinite(sc) and sc < best_score:

                                top_k=15, eta0=0.01, verbose=False)
            sc = evaluate_safe(a_out)

                best_score = sc

                print(f"  *** NEW BEST final MGD: {sc:.6f}")

    n_best = len(best_seq)
