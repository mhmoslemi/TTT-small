def fast_conv(a):
    """Fast convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad(a, t, top_k=None):

def smoothed_objective(a, t):
    """Compute 2n * logsumexp(t*conv(a,a)) / t."""
    n = len(a)
    conv = fast_conv(a)
    tc = t * conv
    tc_max = tc.max()
    return 2.0 * n * (tc_max + np.log(np.exp(tc - tc_max).sum())) / t

def nesterov_smoothed_grad(a0, time_limit, t_start=10.0, t_end=500.0, 
                            eta=0.5, n_phases=8, verbose=True):
    """Multi-scale Nesterov accelerated projected gradient on smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Log-scale temperature
    log_t_start = np.log(t_start)
    log_t_end = np.log(t_end)
    t_values = np.exp(np.linspace(log_t_start, log_t_end, n_phases))
    
    for phase, t in enumerate(t_values):
        phase_t0 = time.monotonic()
        phase_limit = time_limit / n_phases
        iters = 0
        
        x = a.copy()
        y = a.copy()
        stall = 0
        
        while time.monotonic() - phase_t0 < phase_limit:
            # Nesterov momentum
            beta = 0.5 / (iters + 1.5)  # or (1 - sqrt(1+4*iters^2))/(2*iters)
            z = y + beta * (y - x) if iters > 0 else y
            z = np.maximum(z, 1e-15)
            z = z / z.sum()
            
            # Compute smoothed gradient
            g, conv = compute_smoothed_grad(z, t)
            
            # Projected gradient step
            x_new = project_simplex(z - eta * g)
            
            # Check improvement
            m_new = fast_conv(x_new).max()
            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = x_new.copy()
                stall = 0
            else:
                stall += 1
            
            x = y
            y = x_new
            iters += 1
            
            if stall > 200:
                break
        
        a = y
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  NSG phase={phase}, t={t:.1f}, iters={iters}: J={J:.6f} ({elapsed:.0f}s)")
    
    return best_a, best_m

def make_geometric(n, r=0.9):
    """Create a geometric sequence."""
    a = r ** np.arange(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_logarithmic(n):
    """Create a logarithmic sequence."""
    a = 1.0 / (np.arange(n) + 1.0)
    s = a.sum()
    return a / s

def make_sparse(n, k=5, rng=None):
    """Create a sparse sequence with k non-zero elements."""
    if rng is None:
        rng = np.random.default_rng(0)
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n)
    weights = rng.dirichlet(np.ones(k))
    a[idx] = weights
    return a

def make_sinusoidal_asym(n, freq=1.0, asym=0.3):
    """Create an asymmetric sinusoidal sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5 + asym * (x - 0.5)) ** 2) / (2 * 0.3**2))
    a += 0.01 * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def coordinate_polish_fast(a0, time_limit, n_top=10, verbose=False):
    """Fast coordinate polish: move mass from high-contribution positions to low ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 50:
        conv = fast_conv(a)
        M = conv.max()
        top_k = np.argsort(conv)[::-1][:n_top]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            # Find the pair (i, j) with i+j=k that contributes most to conv[ki]
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[ki - idx]
            max_i = idx[np.argmax(contrib)]
            
            if a[max_i] < 1e-10:
                continue
            
            # Try moving mass from max_i to a random position
            for _ in range(3):
                dst = int(rng.integers(n))
                if dst == max_i:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    amt = a[max_i] * frac
                    a_trial = a.copy()
                    a_trial[max_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

        phase_limit = (time_limit - (phase_t0 - t0)) / (n_phases - phase)
        
        # Parameterize: a = softmax(x) for x in R^n
        # Start from x = log(a)
        x0 = np.log(np.maximum(a, 1e-30))
        
        def obj_and_grad(x):
            a_x = np.exp(x - x.max())
            a_x = a_x / a_x.sum()
            conv = fast_conv(a_x)
            obj_val = 2.0 * n * (t * conv).max()  # simplified
            # Actually use logsumexp for smoothness
            tc = t * conv
            tc_max = tc.max()
            obj_val = 2.0 * n * (tc_max + np.log(np.exp(tc - tc_max).sum())) / t
            
            # Gradient through the softmax parameterization
            g_a, _ = compute_smoothed_grad(a_x, t)
            # dx = ga / a (chain rule for softmax)
            grad_x = g_a / np.maximum(a_x, 1e-30)
            return obj_val, grad_x
        
        try:
            result = minimize(obj_and_grad, x0, method='L-BFGS-B',
                             options={'maxiter': 100, 'maxfun': 200,
                                      'ftol': 1e-10, 'gtol': 1e-8})
            x_opt = result.x
            a_new = np.exp(x_opt - x_opt.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            
            m_new = fast_conv(a_new).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new
        except Exception as e:
            pass
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  LBF phase={phase}, t={t:.1f}: J={J:.6f} ({elapsed:.0f}s)")
    
    return best_a, best_m
