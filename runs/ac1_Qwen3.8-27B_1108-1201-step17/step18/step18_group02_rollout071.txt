def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_value(ac, tau):
    max_ac = ac.max()
    if tau < 1e-12:
        return max_ac
    shifted = (ac - max_ac) / tau
    return max_ac + tau * math.log(np.sum(np.exp(shifted)))


def lse_grad_a(a, ac, tau, n):
    """Gradient of LSE(ac, tau) w.r.t. a. Returns array of length n."""
    max_ac = ac.max()
    p = np.exp((ac - max_ac) / tau)
    p = p / p.sum()
    # grad[i] = 2 * sum_j a[j] * p[i+j]
    L = 1 << (3 * n - 2 - 1).bit_length()
    P_f = np.fft.rfft(p, L)
    A_f = np.fft.rfft(a, L)
    corr_full = np.fft.irfft(P_f * np.conj(A_f), L)
    grad = 2.0 * corr_full[n - 1:2 * n - 1]
    return grad


def lse_grad_x(a, grad_a):
    """Convert gradient w.r.t. a to gradient w.r.t. log-space x."""
    S = grad_a.sum()
    return a * (grad_a - a * S)

    a = np.cos(np.pi * (x - 0.5)) ** 2

    a_current = norm_sum1(a0)
    best_a = a_current.copy()
    best_J = 2.0 * n * fast_autocorr(a_current).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x0 = np.log(np.maximum(a_current, 1e-30))

    for restart in range(3):
        if time.monotonic() - t0 > time_limit:
            break
        tau = fast_autocorr(a_current).max() * (0.3 ** restart)
        tau = max(tau, 1e-6)
        call_count = [0]

        def obj_grad(x):
            call_count[0] += 1
            a = np.exp(x - x.max())
            a = a / a.sum()
            ac = fast_autocorr(a)
            val = 2.0 * n * lse_value(ac, tau)
            grad_a = lse_grad_a(a, ac, tau, n)
            grad_x = lse_grad_x(a, grad_a)
            return val, grad_x

        try:
            res = scipy_minimize(obj_grad, x0, method='L-BFGS-B', jac=True,
                                  options={'maxiter': 300, 'gtol': 1e-7, 'maxfun': 600})
        except Exception:
            continue

        x_opt = res.x
        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = np.maximum(a_opt, 0.0)
        a_opt = a_opt / a_opt.sum()
        m_opt = fast_autocorr(a_opt).max()
        J_opt = 2.0 * n * m_opt

        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()
            x0 = x_opt.copy()
            if verbose:
                print("    L-BFGS restart " + str(restart) + ": J=" + format(J_opt, '.6f'))
        else:
            x0 = best_a_log = np.log(np.maximum(best_a, 1e-30))
            x0 = x0 + rng.normal(0, 0.05, n)

    if verbose:
        print("    L-BFGS done (" + format(time.monotonic() - t0, '.0f') + "s): J=" + format(best_J, '.6f'))
    return best_a, best_J / (2.0 * n)


def coord_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Aggressive coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 300:
            break
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(20, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destinations: positions that minimally affect top peaks
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.10, 0.15]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

        if verbose and stall > 0 and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + format(elapsed, '.0f') + "s): J=" + format(J, '.6f') + " stall=" + str(stall))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + format(time_budget, '.0f') + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + format(p_score, '.6f'))
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + format(best_score, '.6f'))

    n0 = len(best_seq)

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print("  *** NEW BEST [" + label + "]: " + format(sc, '.6f') + " (n=" + str(len(a)) + ", t=" + format(elapsed, '.0f') + "s)")

    # === Phase 1: Parametric search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Parametric search (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 10:
        candidates = parametric_init(n0)
        best_param_m = fast_autocorr(best_seq).max()
        for i, cand in enumerate(candidates):
            if time.monotonic() - t_global > time_budget - t1:
                break
            ac = fast_autocorr(cand)
            val = 2.0 * n0 * ac.max()
            if val < 2.0 * n0 * best_param_m:
                check_update(cand, "param" + str(i))

    # Also try different n values for init
    for trial_n in [3000, 4000, 5000, 6000]:
        if trial_n == n0:
            continue
        x = np.linspace(0, 1, trial_n)
        a_try = np.sin(np.pi * x) ** 2
        a_try = norm_sum1(a_try)
        sc_try = evaluate_safe(a_try)
        if np.isfinite(sc_try) and sc_try < best_score:
            check_update(a_try, "init_n" + str(trial_n))

    n_best = len(best_seq)

    # === Phase 2: AFW main run ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: AFW (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = afw_optimize(a_in, t2, solver, K_init=400, K_max=3000, verbose=True)
        check_update(a_out, "afw_n" + str(n_best))

    # === Phase 3: L-BFGS ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: L-BFGS (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfgs_optimize(a_in, t3, verbose=True, rng_seed=77)
        check_update(a_out, "lbfgs_n" + str(n_best))

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Coord descent (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent(a_in, t4, verbose=False, rng_seed=123)
        check_update(a_out, "coord_n" + str(n_best))

    # === Phase 5: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-n exploration ===")
    nearby_lengths = []
    for delta in [200, 400, 600, -200, -400, -600, 800, -800]:
        new_n = n_best + delta
        if 2500 <= new_n <= 6500:
            nearby_lengths.append(new_n)
    seen = set()
    unique_lengths = []
    for ln in nearby_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    for new_n in unique_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick AFW
        t5a = min(remaining * 0.30, 200.0)
        if t5a > 30 and new_n <= 6000:
            a_out, _ = afw_optimize(a_long, t5a, solver, K_init=300, K_max=2500, verbose=False)
            check_update(a_out, "afw_n" + str(new_n))

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.20, 100.0)
            if t5b > 15:
                a_out, _ = coord_descent(norm_sum1(best_seq) if len(best_seq) == new_n else interpolate_sequence(best_seq, new_n), t5b, verbose=False, rng_seed=555)
                check_update(a_out, "coord_n" + str(new_n))

    # === Phase 6: Final AFW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final AFW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 5500:
        t6 = min(remaining * 0.35, 350.0)
        if t6 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = afw_optimize(a_in, t6, solver, K_init=600, K_max=4000, verbose=True)
            check_update(a_out, "afw_final_n" + str(n_best))

    # === Phase 7: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final coord (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t7 = min(remaining * 0.50, 150.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent(a_in, t7, verbose=False, rng_seed=999)
            check_update(a_out, "coord_final_n" + str(n_best))

    # Final L-BFGS polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.40, 100.0)
        if t8 > 15:
            print("\n=== Phase 8: Final L-BFGS (n=" + str(n_best) + ") ===")
            a_in = norm_sum1(best_seq)
            a_out, _ = lbfgs_optimize(a_in, t8, verbose=True, rng_seed=313)
            check_update(a_out, "lbfgs_final_n" + str(n_best))

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + format(best_score, '.6f') + ", n=" + str(len(best_seq)) + ", time=" + format(elapsed, '.0f') + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + format(best_score, '.6f'))

    return [float(x) for x in best_seq]
