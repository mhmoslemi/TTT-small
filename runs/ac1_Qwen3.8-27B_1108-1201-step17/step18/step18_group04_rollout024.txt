def fourier_projection(a0, time_limit, n_iter_max=500, verbose=False):
    """POCS-style: alternate between flat-spectrum in freq domain and non-negativity in time domain."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    N = 1 << (2 * n - 1).bit_length()

    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        # Forward FFT (zero-padded)
        A = np.fft.rfft(a, N)
        # Target: flat magnitude = 1/sqrt(N) (so that |A|^2 = 1/N, giving flat circular conv = 1/N)
        # But we want to boost high freq relative to DC
        target_mag = 1.0 / np.sqrt(N)
        # Blend: keep current phase, set magnitude to target
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = target_mag * phase
        # Inverse FFT
        a_time = np.fft.irfft(A_new, N)[:n]
        # Project to non-negative, sum=1
        a_time = np.maximum(a_time, 0.0)
        s = a_time.sum()
        if s < 1e-15:
            break
        a_time = a_time / s
        a = a_time
        M = fast_conv(a).max()
        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()

    return best_a, best_m


def lp_direction(a, conv, M, n_constraints=50, solver='GLOP'):
    """Solve reduced LP to find best improvement direction.
    max sum(b) s.t. b>=0, (a*b)_k <= M for top-k constraints."""
    import cvxpy as cp
    n = len(a)
    b = cp.Variable(n)
    # Select the most binding constraints
    conv_sorted = np.argsort(conv)[::-1][:n_constraints]
    constraints = [b >= 0]
    for k in conv_sorted:
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        coeff = a[k_int - idx]
        constraints.append(coeff @ b <= M)
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return None
    if b.value is None:
        return None
    b_val = np.maximum(b.value, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

        improved = False
        for frac in [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = a * (1.0 - frac) + b_dir * frac
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < M - 1e-15:
                a = a_trial
                improved = True
                break

        if not improved:
            # Try smaller perturbations
            break

        if verbose and it % 10 == 0:
            J = 2.0 * n * best_m
            print(f"    LP iter {it}: J={J:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_m


def coord_descent_polish(a0, time_limit, seed=42, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]

        n_top = max(5, min(30, n // 150))
        top_src = idx[np.argsort(contrib)[::-1][:n_top]]

        # Compute cost vector: how much each position contributes to top peaks
        n_top_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_top_peaks]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            weights = np.exp(-(conv[k2i] - conv[top_peaks[0]]) * 20.0)
            w2 = conv[k2i] / max(conv[top_peaks[0]], 1e-30)
            costs[idx2] += w2 * a[idx2] * a[k2i - idx2]

        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            # Find best destination: low cost, not the source
            candidates = np.argsort(costs + 1e-12 * a)[:max(15, n // 25)]
            candidates = [int(c) for c in candidates if int(c) != src_i][:10]
            for dst_i in candidates:
                found = False
                for frac in [0.03, 0.08, 0.15, 0.25, 0.4, 0.6]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
            if verbose and stall % 40 == 0:
                J = 2.0 * n * best_m
                print(f"    CD ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_mixed(n, pos1=0.2, pos2=0.65, width=0.1, ratio=1.0):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / (width * 1.5)) ** 2)
    return norm_sum1(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 4700
        x = np.linspace(0, 1, n_base)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Default init: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    n0 = len(_best_tracker['seq'])
    print(f"Starting: n={n0}, score={_best_tracker['score']:.6f}")

    # Phase 1: AMD on parent and diverse inits
    print(f"\n=== Phase 1: AMD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 550.0)

    if t_phase1 > 30:
        a_in = norm_sum1(_best_tracker['seq'])
        # Run AMD on parent with careful settings
        t1a = min(t_phase1 * 0.5, 250.0)
        if t1a > 20:
            print(f"  AMD on parent, t={t1a:.0f}s")
            a_out, _ = amd_optimize(a_in, t1a, top_k_init=40, temp_init=0.0015,
                                    c0_init=2.5, restart_every=5000, seed=42, verbose=False)
            try_update(a_out, "amd_parent")

        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.5, 200.0)
        if t1b > 20:
            # Try a few diverse inits
            inits = []
            labels = []
            inits.append(make_decreasing(n0, 0.4)); labels.append("dec04")
            inits.append(make_mixed(n0, 0.15, 0.7, 0.06)); labels.append("mixA")
            inits.append(make_sinc(n0, 6.0)); labels.append("sinc6")
            inits.append(make_random_smooth(n0, seed=77, n_freq=12)); labels.append("rnd77")

            t_per = t1b / len(inits)
            for idx in range(len(inits)):
                if time.monotonic() - t_global > time_budget - 60:
                    break
                t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
                if t_avail < 10:
                    break
                print(f"  AMD '{labels[idx]}' t={t_avail:.0f}s")
                a_out, _ = amd_optimize(inits[idx], t_avail, top_k_init=30,
                                        temp_init=0.002, c0_init=3.0,
                                        restart_every=4000, seed=200 + idx, verbose=False)
                try_update(a_out, f"amd_{labels[idx]}")

    # Phase 2: Fourier projection
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Fourier Projection ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 15:
        a_in = norm_sum1(_best_tracker['seq'])
        print(f"  Fourier projection, t={t2:.0f}s")
        a_out, _ = fourier_projection(a_in, t2, n_iter_max=1000, verbose=False)
        try_update(a_out, "fourier")

    # Phase 3: LP-based refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Refinement ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_in = norm_sum1(_best_tracker['seq'])
        print(f"  LP optimization, t={t3:.0f}s")
        try:
            a_out, _ = lp_optimize(a_in, t3, verbose=True)
            try_update(a_out, "lp")
        except Exception as e:
            print(f"  LP failed: {e}")
            # Fallback: try one LP step manually
            a_in = norm_sum1(_best_tracker['seq'])
            conv = fast_conv(a_in)
            M = conv.max()
            b_dir = lp_direction(a_in, conv, M, n_constraints=30)
            if b_dir is not None:
                for frac in [0.1, 0.05, 0.02]:
                    a_trial = a_in * (1 - frac) + b_dir * frac
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    try_update(a_trial, f"lp1step_{frac}")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord Descent ===")
    t4 = min(remaining * 0.18, 270.0)
    if t4 > 15:
        a_in = norm_sum1(_best_tracker['seq'])
        print(f"  CD polish, t={t4:.0f}s")
        a_out, _ = coord_descent_polish(a_in, t4, seed=42, verbose=False)
        try_update(a_out, "cd1")

    # Phase 5: AMD refinement on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: AMD Refine ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 15:
        a_in = norm_sum1(_best_tracker['seq'])
        print(f"  AMD refine, t={t5:.0f}s")
        a_out, _ = amd_optimize(a_in, t5, top_k_init=60, temp_init=0.001,
                                c0_init=2.0, restart_every=7000, seed=999, verbose=False)
        try_update(a_out, "amd_refine")

    # Phase 6: Multi-length search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 6: Multi-n (base n={n_best}) ===")
    if remaining > 100:
        candidate_ns = []
        for delta in [-300, -150, 150, 300, 500, -500, 1000]:
            new_n = n_best + delta
            if 3000 <= new_n <= 7000:
                candidate_ns.append(new_n)

        for new_n in candidate_ns[:4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"  Trying n={new_n}")
            a_interp = interpolate_seq(_best_tracker['seq'], new_n)
            try_update(a_interp, f"intp{new_n}")
            t6a = min(remaining * 0.3, 120.0)
            if t6a > 15:
                a_out, _ = amd_optimize(a_interp, t6a, top_k_init=40,
                                        temp_init=0.002, c0_init=3.0,
                                        restart_every=4000, seed=555, verbose=False)
                try_update(a_out, f"amd_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t6b = min(remaining * 0.3, 80.0)
                if t6b > 10:
                    a_out, _ = coord_descent_polish(a_interp, t6b, seed=777, verbose=False)
                    try_update(a_out, f"cd_n{new_n}")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 7: Final Polish (n={n_best}) ===")
    if remaining > 40:
        t7a = min(remaining * 0.5, 150.0)
        if t7a > 15:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = amd_optimize(a_in, t7a, top_k_init=80, temp_init=0.0008,
                                    c0_init=1.5, restart_every=8000, seed=2024, verbose=False)
            try_update(a_out, "amd_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        t7b = min(remaining * 0.6, 120.0)
        if t7b > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = coord_descent_polish(a_in, t7b, seed=31415, verbose=False)
            try_update(a_out, "cd_final")

    # Phase 8: One more LP attempt if time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8 = min(remaining * 0.7, 100.0)
        if t8 > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            try:
                a_out, _ = lp_optimize(a_in, t8, verbose=False)
                try_update(a_out, "lp_final")
            except Exception:
                pass

    # Final
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:

        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
