        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new

        if stall > restart_every:
            stall = 0
            noise_scale = 0.02 + 0.03 * min(stall / restart_every, 2.0)
            pert = rng.normal(0, noise_scale, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            step_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SmoothAdam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothAdam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    SmoothAdam done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def build_coeff_matrix(y, k_list, n):
    K = len(k_list)
    mat = np.zeros((K, n))
    for jj, k in enumerate(k_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            mat[jj, idx] = y[k_int - idx]
    return mat


def lp_frank_wolfe(a0, time_limit, solver, initial_K=800, max_K=6000,
                   add_per_iter=80, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    ac_init = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac_init.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    away = np.zeros(n)

    init_idx = np.argsort(ac_init)[::-1][:min(initial_K, 2 * n - 1)]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    lp_fail_count = 0
    prev_obj = best_m
    no_improve = 0
    rng = np.random.default_rng(seed)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
            no_improve = 0

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = build_coeff_matrix(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 5 and len(constraint_list) > initial_K:
                top_k = np.argsort(ac)[::-1][:initial_K]
                constraint_set = set(int(k) for k in top_k)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 8:
                top_k = np.argsort(ac)[::-1][:max(300, initial_K // 2)]
                constraint_set = set(int(k) for k in top_k)
            continue

        lp_fail_count = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Away-step FW
        away_frac = np.dot(away, y)
        if away_frac > 0.05:
            away_dir_norm = np.linalg.norm(away)
            if away_dir_norm > 1e-10:
                H_away = np.zeros(n)
                for k in constraint_list[:200]:
                    k_int = int(k)
                    i_start = max(0, k_int - n + 1)
                    i_end = min(n, k_int + 1)
                    if i_start < i_end:
                        idx = np.arange(i_start, i_end)
                        H_away[idx] += away * y[k_int - idx]
                if np.dot(H_away, y) < 0:
                    alpha_away = min(0.1, 0.5 * (M_curr - fast_autocorr(y - 0.1 * away).max()) / max(np.dot(H_away, y), 1e-10))
                    alpha_away = max(0, alpha_away)
                    y_away = y - alpha_away * away
                    y_away = np.maximum(y_away, 0)
                    sa = y_away.sum()
                    if sa > 1e-10:
                        y_away = y_away / sa
                        M_away = fast_autocorr(y_away).max()
                        if M_away < M_curr:
                            y = y_away
                            M_curr = M_away
                            if M_curr < best_m - 1e-14:
                                best_m = M_curr
                                best_a = y.copy()

        # Standard FW step
        alpha = 2.0 / (t + 2.0)
        if no_improve > 50:
            alpha = min(alpha, 1.0 / (1.0 + 0.01 * no_improve))
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        # Update away direction
        if away_frac > 0.01:
            away = y - g_dir
        else:
            away = (0.9 * away + 0.1 * (y - g_dir))

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, steps={iter_count}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_descent(a0, time_limit, rng_seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination using impact scoring
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            # Pick a few random candidates too for exploration
            n_rand = min(5, len(candidates))
            rand_idx = rng.choice(len(candidates), n_rand, replace=False)
            for ri in rand_idx:
                cand_scores[ri] *= 0.5

            dst = candidates[np.argmin(cand_scores)]

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 80:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_reweight(a, time_limit, n_iterations=200, verbose=False):
    """Iteratively reweight in frequency domain to flatten autocorrelation peaks."""
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        # Compute periodogram-based weights
        fft_size = 1 << (2 * n - 2).bit_length()
        Af = np.fft.rfft(a, fft_size)
        power = np.abs(Af) ** 2

        # Target: make autocorrelation as flat as possible
        # a_new[i] ~ a[i] / sqrt(sum of power contributions to peak lags)
        # Simpler: dampen positions that contribute most to peaks
        peak_lags = np.argsort(ac)[::-1][:min(5, n)]
        contrib = np.zeros(n)
        for k in peak_lags:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js <= je:
                idx = np.arange(js, je + 1)
                contrib[idx] += a[idx] * a[k_int - idx] * ac[k_int] / max(ac.max(), 1e-20)

        # Reweight: reduce high-contribution positions
        weight = 1.0 / (1.0 + contrib / max(a.max(), 1e-20))
        a_new = a * weight
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Blend to avoid oscillation
        blend = 0.3
        a = (1 - blend) * a + blend * a_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def parametric_init(n, seed=0):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(seed)
    candidates = []

    for freq in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.001)
            candidates.append(a / a.sum())

    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())

    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(a / a.sum())

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.05, 0.1, 0.15, 0.25]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())

    # Multi-peak structures
    for n_peaks in [2, 3, 4, 5]:
        centers = rng.uniform(0.1, 0.9, n_peaks)
        sigma = 0.08
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((x - c) ** 2) / (2.0 * sigma ** 2))
        if a.sum() > 0:
            candidates.append(a / a.sum())

    # Alternating with envelope
    for env_p in [0.5, 1.0, 2.0]:
        env = (x * (1 - x)) ** env_p
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
        a = env * alt
        candidates.append(a / a.sum())

    candidates.append(np.ones(n) / n)
    return candidates


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with improved LP-FW, spectral reweighting, and multi-start."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick parametric scan + blend ===
    print(f"\n=== Phase 1: Parametric scan (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 10:
        candidates = parametric_init(n0, seed=7)
        for i, cand in enumerate(candidates):
            if time.monotonic() - t_global > time_budget - t1:
                break
            sc = evaluate_safe(cand)
            if np.isfinite(sc) and sc < best_score:
                check_and_update(cand, f"param{i}")
        # Blend parent with uniform
        for w in [0.2, 0.4, 0.6, 0.8]:
            cand_blend = w * best_seq + (1 - w) * np.ones(n0) / n0
            check_and_update(cand_blend, f"blend_w{w}")

    n_best = len(best_seq)

    # === Phase 2: Smoothed Adam (initial convergence) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Smoothed Adam (n={n_best}) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam(a_in, t2, tau_decay=0.9998, lr=0.6,
                               restart_every=1500, verbose=True, seed=42)
        check_and_update(a_out, f"adam_n{n_best}")

    # === Phase 3: LP-FW (primary optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW primary (n={n_best}) ===")
    t3 = min(remaining * 0.35, 550.0)
    if t3 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, initial_K=800,
                                   max_K=6000, add_per_iter=80, verbose=True, seed=13)
        check_and_update(a_out, f"lpfw_n{n_best}")

    # === Phase 4: Spectral reweighting ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Spectral reweight (n={n_best}) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_reweight(a_in, t4, n_iterations=300, verbose=False)
        check_and_update(a_out, f"spectral_n{n_best}")

    # === Phase 5: LP-FW round 2 on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW round 2 (n={n_best}) ===")
    t5 = min(remaining * 0.25, 350.0)
    if t5 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, initial_K=1200,
                                   max_K=8000, add_per_iter=100, verbose=True, seed=99)
        check_and_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 6: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord descent (n={n_best}) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent(a_in, t6, rng_seed=271, verbose=False)
        check_and_update(a_out, f"coord_n{n_best}")

    # === Phase 7: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    target_n = int(n_best * 1.05)
    print(f"\n=== Phase 7: Length change to {target_n} ===")
    if remaining > 200 and 2000 <= target_n <= 6500:
        a_interp = interpolate_sequence(best_seq, target_n)
        check_and_update(a_interp, f"interp_n{target_n}")

        # Quick smooth on new length
        t7a = min(remaining * 0.15, 120.0)
        if t7a > 15:
            a_out, _ = smooth_adam(a_interp, t7a, tau_decay=0.9998, lr=0.5,
                                   restart_every=1000, verbose=False, seed=77)
            check_and_update(a_out, f"adam_n{target_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 150 and target_n <= 6000:
            t7b = min(remaining * 0.30, 250.0)
            if t7b > 20:
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t7b, solver,
                                           initial_K=600, max_K=5000,
                                           add_per_iter=60, verbose=False, seed=55)
                check_and_update(a_out, f"lpfw_n{target_n}")

    # === Phase 8: Final LP-FW on global best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP-FW (n={n_best}) ===")
    if remaining > 50 and n_best <= 6000:
        t8 = min(remaining * 0.40, 250.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, initial_K=1500,
                                       max_K=8000, add_per_iter=120, verbose=True, seed=256)
            check_and_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 9: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord (n={n_best}) ===")
    if remaining > 15:
        t9 = min(remaining * 0.50, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent(a_in, t9, rng_seed=512, verbose=False)
            check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
