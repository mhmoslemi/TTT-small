def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

def smooth_obj_grad_z(z, tau):
    """Compute F_tau(softmax(z)) and gradient w.r.t. z."""
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones_like(z) / len(z)
    else:
        a = a / a_sum

    n = len(a)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    shifted = tau * (ac - max_ac)
    exp_shifted = np.exp(shifted)
    obj = max_ac + np.log(exp_shifted.sum()) / tau

    w = exp_shifted / exp_shifted.sum()

    # Gradient w.r.t. a via FFT: g_a[j] = 2 * sum_m a[m] * w[m+j]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A_rev = np.fft.rfft(a[::-1], fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(A_rev * W, fft_size)
    g_a = 2.0 * conv_result[n - 1:2 * n - 1]

    dot_ag = np.dot(a, g_a)
    g_z = a * (g_a - dot_ag)

    return obj, g_z


def smooth_lbfgs(a0, time_limit, verbose=True):
    """L-BFGS-B on smoothed objective with annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    z = np.log(a + 1e-30)

    tau_schedule = [1.0, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0, 35.0, 50.0, 80.0]
    iters_per_tau = 150

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        def obj_grad(z_in):
            return smooth_obj_grad_z(z_in, tau)

        res = scipy_minimize(obj_grad, z, method='L-BFGS-B',
                             options={'maxiter': iters_per_tau, 'gtol': 1e-8, 'maxls': 50})

        z_new = res.x
        z_max = z_new.max()
        a_new = np.exp(z_new - z_max)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()

        z = z_new
        elapsed = time.monotonic() - t0
        if verbose:
            print("    Smooth tau={}: J={}, time={:.1f}s".format(
                tau, repr(2.0 * n * best_m), elapsed))

    return best_a, best_m

def try_update(best_seq, best_score, a, label, t_global):
    """Return (best_seq, best_score) after checking candidate a."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [{}]: {} (n={}, t={:.1f}s)".format(label, repr(sc), len(a), elapsed))
    return best_seq, best_score

                                  initial_K=300, max_K=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n{}".format(n_best), t_global)

    # Phase 3: Smooth L-BFGS annealing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Smooth L-BFGS (n={}) ===".format(n_best))
    t3 = min(remaining * 0.15, 220.0)
    if t3 > 30 and n_best <= 7000:
        a_out, _ = smooth_lbfgs(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_n{}".format(n_best), t_global)

    # Phase 4: LP-FW fine (larger active set)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW fine (n={}) ===".format(n_best))
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30 and n_best <= 7000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t4, solver,
                                  initial_K=400, max_K=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_fine_n{}".format(n_best), t_global)

    # Phase 5: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Coord polish (n={}) ===".format(n_best))
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 15:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t5, rng_seed=77, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n{}".format(n_best), t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 8000]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n=== Phase 6: Scale to n={} ===".format(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.20, 140.0)
        if t6a > 30:
            a_out, _ = lp_fw_improved(a_long, t6a, solver,
                                      initial_K=250, max_K=2000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n{}".format(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.15, 100.0)
            if t6b > 20:
                a_out, _ = smooth_lbfgs(norm_sum1(best_seq), t6b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_n{}".format(new_n), t_global)

    # Phase 7: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n={}) ===".format(n_best))
    if remaining > 60 and n_best <= 7000:
        t7 = min(remaining * 0.35, 250.0)
        if t7 > 30:
            a_out, _ = lp_fw_improved(norm_sum1(best_seq), t7, solver,
                                      initial_K=500, max_K=3000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final_n{}".format(n_best), t_global)

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final coord (n={}) ===".format(n_best))
    if remaining > 20:
        t8 = min(remaining * 0.50, 120.0)
        if t8 > 10:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, rng_seed=314, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final_n{}".format(n_best), t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={}, n={}, time={:.1f}s".format(repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score={}".format(repr(best_score)))

    return [float(x) for x in best_seq]
