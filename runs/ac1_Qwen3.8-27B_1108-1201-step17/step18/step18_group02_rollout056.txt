    ac_init = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac_init.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    lp_fail_count = 0
    stall_count = 0
    best_m_history = [best_m]

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
            stall_count = 0

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = build_constraint_matrix(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 5 and len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 10:
                half = max(200, initial_K // 2)
                constraint_set = set(int(k) for k in constraint_list[:half])
            continue

        lp_fail_count = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Multi-alpha line search
        alphas = [2.0 / (t + 2.0)]
        if t > 10:
            alphas.extend([0.1, 0.2, 0.3, 0.5, 0.7, 1.0])
        # Remove duplicates and sort
        alphas = sorted(set(min(1.0, max(0.01, a)) for a in alphas))

        best_alpha = 2.0 / (t + 2.0)
        best_y_new = None
        best_obj = M_curr + 1.0

        for alpha in alphas:
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            s_y = y_trial.sum()
            if s_y < 1e-15:
                continue
            y_trial = y_trial / s_y
            ac_trial = fast_autocorr(y_trial)
            M_trial = ac_trial.max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_alpha = alpha
                best_y_new = y_trial

        if best_y_new is None:
            best_alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new / s_y
            else:
                y_new = y.copy()
        else:
            y_new = best_y_new

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x > 1e-15:
                x_new = x_new / s_x
            else:
                x_new = y_new.copy()

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, stall={stall_count}")

        if stall_count > 200 and iter_count > 100:
            # Restart with perturbation
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.02, n) * best_a
            y_restart = best_a + pert
            y_restart = np.maximum(y_restart, 1e-10)
            y_restart = y_restart / y_restart.sum()
            y = y_restart
            x = y_restart.copy()
            constraint_set = set(int(k) for k in np.argsort(fast_autocorr(y_restart))[::-1][:initial_K])
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP-FW RESTART at iter={t} ({elapsed:.0f}s)")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def smooth_adam(a0, time_limit, tau_decay=0.9997, lr=0.6, beta1=0.9, beta2=0.999,
                restart_every=2000, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    tau = max(ac.max() * 0.5, 1e-6)
    tau_min = max(ac.max() * 0.003, 1e-10)

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        tau = max(tau * tau_decay, tau_min)

        g = lse_gradient(a, ac, tau)

        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)

        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2

        step = lr / (np.sqrt(v_c) + 1e-8)

        log_a = np.log(np.maximum(a, 1e-30)) - step * m_c
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a = a_new / s

        if stall > restart_every:
            stall = 0
            # Structured restart: add frequency-domain perturbation
            n_pert = int(rng.integers(1, 5))
            pert = np.zeros(n)
            for _ in range(n_pert):
                freq = int(rng.integers(1, n // 2))
                phase = rng.uniform(0, 2 * np.pi)
                pert += 0.03 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
            x = best_a + pert * best_a
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SmoothAdam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothAdam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")

    elapsed = time.monotonic() - t0
    print(f"    SmoothAdam done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def coord_descent(a0, time_limit, rng_seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destinations by their impact on top peaks
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            else:
                candidates = candidates[(candidates != src)]

            if len(candidates) == 0:
                continue

            cand_scores = scores[candidates]
            # Try top-5 least impactful destinations
            n_try = min(5, len(candidates))
            dst_candidates = candidates[np.argsort(cand_scores)[:n_try]]

            for dst in dst_candidates:
                if time.monotonic() - t0 > time_limit:
                    break

                # Exact 1D minimization: transfer amt from src to dst
                # Try golden section over amt in [0, src_val*0.5]
                lo = 0.0
                hi = src_val * 0.4
                best_amt = 0.0
                best_M = np.inf

                for _ in range(8):
                    mid1 = lo + 0.382 * (hi - lo)
                    mid2 = lo + 0.618 * (hi - lo)

                    a1 = a.copy()
                    a1[src] -= mid1
                    a1[dst] += mid1
                    a1 = np.maximum(a1, 0.0)
                    s1 = a1.sum()
                    if s1 > 1e-15:
                        a1 = a1 / s1
                        M1 = fast_autocorr(a1).max()
                    else:
                        M1 = np.inf

                    a2 = a.copy()
                    a2[src] -= mid2
                    a2[dst] += mid2
                    a2 = np.maximum(a2, 0.0)
                    s2 = a2.sum()
                    if s2 > 1e-15:
                        a2 = a2 / s2
                        M2 = fast_autocorr(a2).max()
                    else:
                        M2 = np.inf

                    if M1 < M2:
                        hi = mid2
                    else:
                        lo = mid1

                    if min(M1, M2) < best_M:
                        test_a = a1 if M1 < M2 else a2
                        best_M = min(M1, M2)
                        best_amt = mid1 if M1 < M2 else mid2

                if best_M < best_m - 1e-16:
                    a_trial = a.copy()
                    a_trial[src] -= best_amt
                    a_trial[dst] += best_amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        a = a_trial
                        best_m = best_M
                        best_a = a.copy()
                        improved = True
                        stall = 0

        if not improved:
            stall += 1
            # Random pairwise swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10 and a[j] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = min(a[i] * frac, a[j] * frac)
                if amt > 1e-15:
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

            if stall > 50:
                # Larger perturbation
                n_pert = int(rng.integers(max(2, n // 100), max(5, n // 20)))
                indices = rng.integers(0, n, size=n_pert * 2)
                for pi in range(0, n_pert * 2, 2):
                    if a[indices[pi]] > 1e-8:
                        amt = a[indices[pi]] * rng.uniform(0.01, 0.05)
                        a[indices[pi]] -= amt
                        a[indices[pi + 1]] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    # Spectral interpolation for smoother results
    N = len(f)
    F = np.fft.rfft(f_norm)
    # Zero-pad in frequency domain
    new_N = new_n
    F_pad = np.zeros(new_N // 2 + 1, dtype=np.complex128)
    copy_len = min(len(F), new_N // 2 + 1)
    F_pad[:copy_len] = F[:copy_len]
    f_interp = np.fft.irfft(F_pad, new_N)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        # Fallback to linear interpolation
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, new_n)
        f_interp = np.interp(new_x, old_x, f_norm)
        f_interp = np.maximum(f_interp, 0.0)
        ss = f_interp.sum()
        if ss < 1e-15:
            return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    for freq in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.01)
            candidates.append(a / a.sum())

    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())

    for alpha in [0.5, 1.0, 2.0, 3.0]:
        for beta in [0.5, 1.0, 2.0, 3.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(a / a.sum())

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.05, 0.1, 0.2, 0.3]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())

    a = np.cos(np.pi * (x - 0.5)) ** 2
    candidates.append(a / a.sum())
    a = np.cos(np.pi * (x - 0.5)) ** 4
    candidates.append(a / a.sum())

    candidates.append(np.ones(n) / n)

    # Two-peak structures
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            a = np.exp(-((x - c1) ** 2) / 0.02) + np.exp(-((x - c2) ** 2) / 0.02)
            candidates.append(a / a.sum())

    return candidates


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        candidates = parametric_init(n0)
        for i, cand in enumerate(candidates):
            if time.monotonic() - t_global > time_budget - t1:
                break
            ac = fast_autocorr(cand)
            val = 2.0 * n0 * ac.max()
            if val < 2.0 * n0 * fast_autocorr(best_seq).max():
                check_and_update(cand, f"param{i}")

    n_best = len(best_seq)

    # === Phase 2: Smoothed Adam (multiple restarts) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Smoothed Adam (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        # Run 1: standard
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam(a_in, t2, tau_decay=0.9997, lr=0.6,
                               beta1=0.9, beta2=0.999, restart_every=2000, verbose=True, rng_seed=123)
        check_and_update(a_out, f"smooth1_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            # Run 2: different hyperparams
            a_in = norm_sum1(best_seq)
            t2b = min(remaining * 0.3, 120.0)
            if t2b > 20:
                a_out, _ = smooth_adam(a_in, t2b, tau_decay=0.9995, lr=0.8,
                                       beta1=0.85, beta2=0.99, restart_every=1000, verbose=False, rng_seed=456)
                check_and_update(a_out, f"smooth2_n{n_best}")

    # === Phase 3: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.30, 450.0)
    if t3 > 30 and n_best <= 5500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, initial_K=800,
                                   max_K=6000, add_per_iter=80, verbose=True)
        check_and_update(a_out, f"lp_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent (n={n_best}) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent(a_in, t4, rng_seed=111, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Second LP-FW pass (warmed up) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW pass 2 (n={n_best}) ===")
    if remaining > 60 and n_best <= 5500:
        t5 = min(remaining * 0.25, 300.0)
        if t5 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t5, solver, initial_K=1000,
                                       max_K=8000, add_per_iter=100, verbose=True)
            check_and_update(a_out, f"lp2_n{n_best}")

    # === Phase 6: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for delta in [200, 400, 600, -200, -400]:
        new_n = n_best + delta
        if 2000 <= new_n <= 5500:
            nearby_lengths.append(new_n)

    seen = set()
    unique_lengths = []
    for ln in nearby_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    for new_n in unique_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 200:
            break
        print(f"\n=== Phase 6: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.3, 120.0)
        if t6a > 20:
            a_out, _ = smooth_adam(a_long, t6a, tau_decay=0.9997, lr=0.5,
                                    beta1=0.9, beta2=0.999, restart_every=1500, verbose=False, rng_seed=789)
            check_and_update(a_out, f"smooth_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 5000:
            t6b = min(remaining * 0.4, 150.0)
            if t6b > 20:
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t6b, solver,
                                           initial_K=600, max_K=4000,
                                           add_per_iter=60, verbose=False)
                check_and_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final LP-FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 5500:
        t7 = min(remaining * 0.40, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=1200,
                                       max_K=8000, add_per_iter=100, verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord (n={n_best}) ===")
    if remaining > 15:
        t8 = min(remaining * 0.50, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent(a_in, t8, rng_seed=271, verbose=False)
            check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
