"""

import numpy as np

import scipy.optimize

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def binomial_init(n, power=0.5, offset=0.0):
    """Binomial-like initialization: a_j ∝ (j+1)^(-power) + offset."""
    j = np.arange(n, dtype=np.float64)
    a = (j + 1.0) ** (-power) + offset
    return norm_sum1(a)

def binomial_exact(n):
    """Exact binomial coefficients / 4^j."""
    j = np.arange(n, dtype=np.float64)
    # Use log to avoid overflow: log(binom(2j,j)/4^j) = log(binom(2j,j)) - j*log(4)
    # For large j, use Stirling: binom(2j,j) ≈ 4^j / sqrt(pi*j)
    # So binom(2j,j)/4^j ≈ 1/sqrt(pi*j) for large j
    # For small j, compute exactly
    a = np.zeros(n)
    for i in range(min(n, 100)):
        from math import comb, log
        a[i] = comb(2*i, i) / (4.0 ** i)
    if n > 100:
        # For j >= 100, use the approximation 1/sqrt(pi*j)
        j_large = np.arange(100, n, dtype=np.float64)
        a[100:] = 1.0 / np.sqrt(np.pi * j_large)
        # Smooth the transition
        a[99] = 0.5 * (a[99] + a[100])
    return norm_sum1(a)

def get_active_set(conv, tol=1e-10):
    """Get the set of k values where conv_k is within tol of the max."""
    max_val = conv.max()
    return np.where(conv >= max_val - tol)[0]

def compute_subgrad_exact(a, conv, active_set):
    """Exact subgradient: average of gradients over the active set."""
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        k_i = int(k)
        lo = max(0, k_i - n + 1)
        hi = min(n - 1, k_i)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        g[idx] += a[k_i - idx]
    g /= len(active_set)
    return g

def projected_gradient_step(a, g, eta):
    """L2 projected gradient step: project onto {a>=0, sum(a)=1}."""
    # a_new = proj(a - eta*g) onto the simplex
    v = a - eta * g
    # Project onto simplex: find nu such that sum(max(v - nu, 0)) = 1
    # Bisection on nu
    v_sorted = np.sort(v)[::-1]
    n = len(v)
    # Find the right threshold
    cumsum = np.cumsum(v_sorted)
    rho = np.where(v_sorted > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    # Find largest rho+1 such that v_sorted[rho] > (cumsum[rho] - 1) / (rho+1)
    tmp = (cumsum - 1.0) / np.arange(1, n + 1)
    valid = v_sorted > tmp
    if not valid.any():
        return np.ones(n) / n
    rho = np.max(np.where(valid)[0])
    nu = cumsum[rho] / (rho + 1.0)
    a_new = np.maximum(v - nu, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_new / s

    from scipy.sparse import csr_matrix
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])  # A[k,j] = a_{k-j} but we need a[j] * ... 
            # Wait, (a*b)_k = sum_j a_j * b_{k-j}
            # So the constraint is sum_j a_j * b_{k-j} <= M
            # This is: sum_j a_j * (b shifted)_j <= M
            # In matrix form: A[k,j] * b_j where A[k,j] = a_{k-j} if k-j is valid
    
    # Actually, let me reconsider. (a*b)_k = sum_j a_j * b_{k-j}
    # For fixed k, as j ranges, k-j ranges from k to k-(n-1)
    # We need 0 <= k-j <= n-1, so k-n+1 <= j <= k, and 0 <= j <= n-1
    # So j ranges from max(0, k-n+1) to min(n-1, k)
    # And the coefficient of b_j in constraint k is a_{k-j}
    
    # But wait, b is the variable, and its index is j. The constraint is:
    # sum_j a_{k-j} * b_j <= M for each k
    # So A[k, j] = a_{k-j} (if 0 <= k-j <= n-1, else 0)
    
    # Let me rebuild:
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

            cols.append(j)

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

            else:

def mirror_descent_optimize(a0, time_limit, tol=1e-12, lp_every=1000, verbose=False):
    """Mirror descent with exact subgradient and periodic LP."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_conv = fast_conv(a)
    best_max = best_conv.max()
    t0 = time.monotonic()
    
    # Adaptive parameters
    eta = 0.1  # initial step size
    eta_decay = 0.999
    stall_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv(a)
        max_val = conv.max()
        
        # Find active set (k values within tol of max)
        active = np.where(conv >= max_val - max(tol, 1e-12 * max_val))[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Compute exact subgradient
        g = compute_subgrad_exact(a, conv, active)
        
        # Projected gradient step
        a_new = projected_gradient_step(a, g, eta)
        
        # Check improvement
        conv_new = fast_conv(a_new)
        max_new = conv_new.max()
        
        if max_new < best_max - 1e-14:
            best_max = max_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            # Try smaller step
            a_half = projected_gradient_step(a, g, eta * 0.5)
            conv_half = fast_conv(a_half)
            max_half = conv_half.max()
            if max_half < max_val - 1e-14:
                a = a_half
                if max_half < best_max - 1e-14:
                    best_max = max_half
                    best_a = a.copy()
                stall_count = 0
            else:
                a = a_new  # accept anyway (trust region)
                stall_count += 1
        
        # Decay step size
        eta *= eta_decay
        
        # Periodic LP
        if lp_every > 0 and t % lp_every == 0 and time.monotonic() - t0 < time_limit - 10:
            b_star = solve_lp_direction(a, best_max, max_time=5.0)
            if b_star is not None:
                # Blend with LP direction
                b_norm = b_star / b_star.sum()
                blend = 0.3
                a_mix = (1 - blend) * a + blend * b_norm
                a = norm_sum1(a_mix)
                conv_mix = fast_conv(a)
                if conv_mix.max() < best_max - 1e-14:
                    best_max = conv_mix.max()
                    best_a = a.copy()
                    stall_count = 0
                elif verbose:
                    pass  # LP direction didn't help immediately
        
        # Restart if stalled
        if stall_count > 500:
            # Add noise
            noise = np.random.default_rng(t).normal(0, 0.01, n) * a
            a_r = a + noise
            a = norm_sum1(a_r)
            eta = 0.05
            stall_count = 0
    
    return best_a, best_max
