    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_solver():
    # Try solvers in order of preference
    for s in [cp.CBC, cp.GLPK, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    tmp = u * np.arange(1, n + 1)
    rho = np.nonzero(tmp > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(y, p):
    """Compute 2 * sum_l y[l] * p[j+l] for j=0..n-1 via scipy correlate."""
    n = len(y)
    full = sp_correlate(p, y, mode='full')
    f = full[n - 1: 2 * n - 1]
    return 2.0 * f


def make_u_shape(n, sharpness=2.0):

    a = 1.0 + sharpness * x ** 2

def make_sig_u(n, alpha=0.7):

def spectral_diversify(a, rng, strength=0.3):
    """Mix with a random spectral component to escape local minima."""
    n = len(a)
    A = np.fft.rfft(a, 2 * n)
    freqs = len(A)
    mask = np.ones(freqs, dtype=float)
    mask[0] = 0.0
    noise = rng.normal(0, 1, freqs) + 1j * rng.normal(0, 1, freqs)
    A_new = A + strength * noise * mask
    A_new = A_new * np.conj(A_new)
    A_new = np.real(A_new)
    A_new = np.maximum(A_new, 0)
    a_new = np.fft.irfft(A_new, 2 * n)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    blend = 0.7 * a + 0.3 * a_new
    return norm_sum1(blend)


def nesterov_smoothed(a0, time_limit, tau_init=8.0, tau_end=0.05,
                       eta_init=1.0, verbose=True, rng_seed=42):
    """Nesterov-accelerated log-sum-exp smoothed subgradient with annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x_prev = a.copy()
    y = a.copy()
    beta = 1.0
    eta = eta_init
    stall = 0
    max_stall = 3000
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_init * (tau_end / tau_init) ** progress

        ac = fast_autocorr(y)
        ac_max = ac.max()
        shifted = (ac - ac_max) / tau
        shifted -= shifted.max()
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()

        grad = compute_smoothed_grad(y, p)

        x_curr = project_simplex(y - eta * grad)
        beta_next = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = (beta - 1.0) / beta_next
        y_new = x_curr + momentum * (x_curr - x_prev)
        y_new = np.maximum(y_new, 0.0)
        s_val = y_new.sum()
        if s_val < 1e-15:
            y_new = x_curr.copy()
        else:
            y_new = y_new / s_val

        x_prev = x_curr
        y = y_new
        beta = beta_next
        total_iters += 1

        m_new = fast_autocorr(y).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = y.copy()
            stall = 0
            eta = min(eta * 1.01, 5.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.8
            if stall > 600:
                pert = rng.normal(0, 0.004, n) * y
                y_p = np.maximum(y + pert, 0.0)
                y_p = y_p / y_p.sum()
                m_p = fast_autocorr(y_p).max()
                if m_p < best_m:
                    y = y_p
                    x_prev = y_p.copy()
                    beta = 1.0
                    eta = eta_init
                    stall = 0
                else:
                    y_spectral = spectral_diversify(y, rng, strength=0.2)
                    m_s = fast_autocorr(y_spectral).max()
                    if m_s < best_m:
                        y = y_spectral
                        x_prev = y_spectral.copy()
                        beta = 1.0
                        eta = eta_init
                        stall = 0
                    else:
                        y = x_curr
                        x_prev = x_curr.copy()
                        beta = 1.0

        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Nest iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.3f}, eta={eta:.3f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None):
    """LP-based Frank-Wolfe with progressive K and backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = min(2 * n - 1, 5000)
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                idx = np.arange(j_start, j_end)
                T_sel[ii, idx] = a[ki - idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        base_m = fast_autocorr(a).max()
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_m
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                  0.5, 0.3, 0.2, 0.1]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 8 == 0:
            K = min(K + 40, top_K_max)

        if verbose and iter_count % 4 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Fast coordinate descent with sensitivity-based move selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 80
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:4]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.01, 0.02, 0.005, 0.03, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            while j == i:
                j = int(rng.integers(n))
            if a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if verbose and stall % 40 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (prev={old_score:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_sig_u(n_base, alpha=0.6)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: Nesterov smoothed (main optimizer)
    print(f"\n=== Phase 1: Nesterov smoothed (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smoothed(a_in, t1, tau_init=10.0, tau_end=0.05,
                                      eta_init=1.5, verbose=True, rng_seed=42)
        try_update(a_out, "nest1")

    # Phase 2: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2000)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1) if not use_all else (2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp1")

    # Phase 3: Nesterov round 2 (lower tau)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Nesterov r2 (n={n_best}) ===")
    t3 = min(remaining * 0.2, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smoothed(a_in, t3, tau_init=3.0, tau_end=0.02,
                                      eta_init=0.8, verbose=True, rng_seed=200)
        try_update(a_out, "nest2")

    # Phase 4: LP-FW round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW r2 (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1) if not use_all else (2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp2")

    # Phase 5: Multi-start Nesterov from different inits
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-start (n={n_best}) ===")
    if remaining > 80:
        inits = []
        labels = []
        inits.append(make_sig_u(n_best, alpha=0.5))
        labels.append("sig_u05")
        inits.append(make_u_shape(n_best, sharpness=1.5))
        labels.append("u15")
        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_sig_u(n_best, 0.8)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        t_per = min(60.0, remaining * 0.15)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_avail = min(t_per, remaining - 40)
            if t_avail < 20:
                break
            print(f"  Multi-start '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_smoothed(init, t_avail, tau_init=5.0, tau_end=0.1,
                                          eta_init=1.0, verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"ms_{lbl}")

    # Phase 6: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate polish (n={n_best}) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord1")

    # Phase 7: Brief length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    trial_lengths = [n_best + 1000, n_best - 1000, n_best + 2000]
    trial_lengths = [l for l in trial_lengths if 2000 <= l <= 8000 and l != n_best]
    trial_lengths = list(dict.fromkeys(trial_lengths))[:2]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        t7 = min(remaining * 0.35, 150.0)
        if t7 > 30:
            a_out, _ = nesterov_smoothed(a_interp, t7, tau_init=8.0, tau_end=0.1,
                                          eta_init=1.0, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"nest_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40 and new_n <= 5000:
                t7b = min(remaining * 0.5, 80.0)
                if t7b > 20:
                    a_in = norm_sum1(a_out)
                    a_out2, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                                top_K_init=200, top_K_max=min(2000, 2*new_n-1))
                    try_update(a_out2, f"lp_n{new_n}")

    # Phase 8: Final Nesterov + LP on absolute best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final optimization (n={n_best}) ===")
    if remaining > 40:
        t8a = min(remaining * 0.5, 120.0)
        if t8a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_smoothed(a_in, t8a, tau_init=2.0, tau_end=0.01,
                                          eta_init=0.5, verbose=True, rng_seed=999)
            try_update(a_out, "nest_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 6000:
            t8b = min(remaining * 0.55, 80.0)
            if t8b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False,
                                           top_K_init=400, top_K_max=min(4000, 2*n_best-1))
                try_update(a_out, "lp_final")

    # Phase 9: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
        t9 = min(remaining * 0.7, 60.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 4000
        best_seq = make_sig_u(n_fb, alpha=0.6)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
