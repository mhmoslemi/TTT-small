def gradient_descent(a0, time_limit, n_peaks=50, temperature=2.0, c0=3.0,

def spectral_flatten(a0, time_limit, alpha0=0.3, verbose=False):

    best_a = a.copy()

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        alpha = alpha0 + (1.0 - alpha0) * min(1.0, it / 1000.0)
        P_target = P_mean * (P / np.maximum(P_mean, 1e-30)) ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)
        
        mag_ratio = np.sqrt(P_target / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.7, 1.4)
        A_new = A * mag_ratio
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.2
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:

                best_a = a.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with multi-peak subgradient
2. LP-based direction finding (solving small LPs periodically for better moves)
3. Spectral flattening (modified Gerchberg-Saxton)
4. Coordinate descent polish on peak-contributing positions
5. Multi-length search with interpolation
6. Adaptive step size and smart restarts

Key improvements over previous:
- LP-based periodic updates that solve a small LP to find optimal direction
- Nesterov momentum for faster convergence
- Adaptive temperature that cools down to focus on hardest peaks
- Larger n (6000-10000) for better continuous approximation
- More diverse initializations including the analytical power-law family
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_ramp(n, power):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_mild_ramp(n, center_frac):
    x = np.linspace(0, 1, n)
    a = 1.0 + center_frac * x
    return norm_sum1(a)


def make_two_phase(n, frac, alpha):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    mask = x > frac
    a[mask] = 1.0 + alpha * (x[mask] - frac) / (1.0 - frac)
    return norm_sum1(a)


def compute_multi_peak_gradient(a, ac, n_peaks, temperature):
    """Compute softmax-weighted multi-peak subgradient."""
    n = len(a)
    k_max = min(n_peaks, len(ac))
    ac_clipped = np.maximum(ac, 1e-30)
    log_ac = np.log(ac_clipped)
    exp_vals = np.exp((log_ac - log_ac.max()) / temperature)
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    top_idx = np.argsort(ac)[::-1][:k_max]
    for k in top_idx:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    return grad


def lp_direction(a, ac, n_peaks):
    """Solve LP to find good direction g minimizing max conv(g,a) at top peaks."""
    n = len(a)
    top_k = np.argsort(ac)[::-1][:n_peaks]
    M = ac[top_k[0]]

    # LP: minimize t s.t. conv(g,a)[k] <= t for k in top_k, sum(g)=1, g>=0
    # conv(g,a)[k] = sum_i g[i]*a[k-i]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((n_peaks, n + 1))
    b_ub = np.zeros(n_peaks)
    for i, k in enumerate(top_k):
        ki = int(k)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            pair = ki - idx
            A_ub[i, idx] = a[pair]
        A_ub[i, -1] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 5})
        if result.success:
            g = result.x[:n]
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
                return g
    except Exception:
        pass

    return a.copy()


def gradient_descent(a0, time_limit, n_peaks, temperature, c0,
                     lp_every, lp_peaks, verbose, rng_seed):
    """Nesterov mirror descent with multi-peak subgradient and periodic LP updates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    it = 0
    cur_temp = temperature
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = (it - 1.0) / (it + 2.0)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M_y = ac.max()

        if M_y < best_m - 1e-15:
            best_m = M_y
            best_a = y.copy()
            stall = 0
        else:
            M_x = fast_autocorr(x).max()
            if M_x < best_m - 1e-15:
                best_m = M_x
                best_a = x.copy()
                stall = 0
            else:
                stall += 1

        g = compute_multi_peak_gradient(y, ac, n_peaks=n_peaks, temperature=cur_temp)
        eta = c0 / (np.sqrt(it) + 1.0)

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Adaptive temperature: cool down when stagnating
        cur_temp = max(0.8, temperature * (1.0 - 0.15 * min(stall / 3000.0, 1.0)))

        # Periodic LP update
        if it % lp_every == 0 and time.monotonic() - t0 < time_limit - 10:
            ac_x = fast_autocorr(x)
            g_lp = lp_direction(x, ac_x, n_peaks=lp_peaks)
            alpha_blend = 0.2 + 0.1 * min(stall / 5000.0, 1.0)
            x_lp = (1 - alpha_blend) * x + alpha_blend * g_lp
            x_lp = norm_sum1(x_lp)
            M_lp = fast_autocorr(x_lp).max()
            if M_lp < best_m:
                best_m = M_lp
                best_a = x_lp.copy()
                x = x_lp
                x_prev = x_lp.copy()
                stall = 0
            elif M_lp < M_x + 1e-15:
                # Accept even if not new best, to diversify
                x = x_lp
                x_prev = x_lp.copy()

        # Restart if stuck
        if stall > 4000:
            stall = 0
            restart_count += 1
            strength = min(0.2, 0.05 + 0.02 * restart_count)
            pert = rng.normal(0, strength, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            cur_temp = temperature * 0.9
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    GRAD restart {restart_count} at it={it} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GRAD it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, temp={cur_temp:.2f}")

    return best_a, best_m


def spectral_flatten(a0, time_limit, alpha0):
    """Modified Gerchberg-Saxton spectral flattening."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        alpha = alpha0 + (1.0 - alpha0) * min(1.0, it / 500.0)
        P_target = P_mean * np.power(P / np.maximum(P_mean, 1e-30), (1.0 - alpha))
        P_target = np.maximum(P_target, 1e-30)

        mag_ratio = np.sqrt(P_target / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.7, 1.4)
        A_new = A * mag_ratio

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.2
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}")

    # === Phase 1: Multi-init gradient at base length ===
    print(f"\n=== Phase 1: Multi-init gradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)

    if t1 > 60:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_ramp(n0, 0.3))
        labels.append("ramp_03")
        inits.append(make_ramp(n0, 0.4))
        labels.append("ramp_04")
        inits.append(make_ramp(n0, 0.5))
        labels.append("ramp_05")
        inits.append(make_ramp(n0, 0.6))
        labels.append("ramp_06")
        inits.append(make_mild_ramp(n0, 0.3))
        labels.append("mild_03")
        inits.append(make_two_phase(n0, 0.5, 0.4))
        labels.append("twophase")

        t_per = max(15.0, t1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            if time.monotonic() - t_global > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  GRAD '{lbl}' ({t_avail:.0f}s)...")
            a_out, _ = gradient_descent(init, t_avail, n_peaks=50, temperature=2.5,
                                         c0=3.0, lp_every=150, lp_peaks=10,
                                         verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"grad_{lbl}")

    # === Phase 2: Spectral on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Spectral (n={n_best}) ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 20:
        a_out, _ = spectral_flatten(norm_sum1(best_seq), t2, alpha0=0.3)
        try_update(a_out, f"spectral_{n_best}")

    # === Phase 3: Aggressive gradient+LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Aggressive gradient+LP (n={n_best}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 20:
        a_out, _ = gradient_descent(norm_sum1(best_seq), t3, n_peaks=80, temperature=1.5,
                                     c0=4.0, lp_every=50, lp_peaks=15,
                                     verbose=True, rng_seed=7)
        try_update(a_out, f"gradLP_{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_out, _ = coordinate_descend(norm_sum1(best_seq), t4, rng_seed=42)
        try_update(a_out, f"coord_{n_best}")

    # === Phase 5: Multi-length search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length ===")
    lengths_to_try = []
    for delta in [1000, 2000, -1000, 3000, -2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 12000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))

    for new_n in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_{new_n}")

        t5a = min(remaining * 0.30, 120.0)
        if t5a > 15:
            a_out, _ = gradient_descent(a_long, t5a, n_peaks=50, temperature=2.0,
                                         c0=3.5, lp_every=100, lp_peaks=10,
                                         verbose=False, rng_seed=99)
            try_update(a_out, f"grad_{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.20, 80.0)
            if t5b > 10:
                a_out, _ = spectral_flatten(norm_sum1(best_seq), t5b, alpha0=0.4)
                try_update(a_out, f"spectral_n{new_n}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}) ===")
    if remaining > 30:
        t6 = min(remaining * 0.45, 250.0)
        if t6 > 15:
            a_out, _ = gradient_descent(norm_sum1(best_seq), t6, n_peaks=120, temperature=1.0,
                                         c0=2.5, lp_every=30, lp_peaks=20,
                                         verbose=True, rng_seed=123)
            try_update(a_out, f"final_{n_best}")

    # === Phase 7: Final coordinate + spectral combo ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t7a = min(remaining * 0.4, 80.0)
        t7b = (remaining - t7a) * 0.7
        if t7a > 10:
            a_out, _ = coordinate_descend(norm_sum1(best_seq), t7a, rng_seed=777)
            try_update(a_out, f"coord_final_{n_best}")
        if t7b > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = gradient_descent(a_in, t7b, n_peaks=80, temperature=1.2,
                                         c0=3.0, lp_every=40, lp_peaks=15,
                                         verbose=False, rng_seed=555)
            try_update(a_out, f"grad_veryfinal_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6000
        best_seq = make_ramp(n_fb, 0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
