"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (maximize sum(b) s.t. conv(a,b)[k] <= M, b >= 0)
2. Adaptive mixing with line search
3. MGD (multiplicative gradient descent) polish
4. Coordinate descent for peak reduction
5. Multi-length exploration via interpolation
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute linear autocorrelation (full convolution) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lp_direction(a, M, K=150):
    """
    Solve LP: maximize sum(b) subject to conv(a,b)[k] <= M for top-K lags, b >= 0.
    Returns b or None if LP fails.
    """
    n = len(a)
    ac = fast_autocorr(a)
    K = min(K, len(ac))
    top_lags = np.argsort(ac)[::-1][:K]

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        count = i_end - i_start + 1
        if count == 0:
            continue
        rows.extend([idx] * count)
        cols.extend(range(i_start, i_end + 1))
        vals.extend(a[i_start:i_end + 1].tolist())

    if len(vals) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(rows) // max(1, len(vals) // max(1, count)), n))
    # Rebuild properly
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))

    c = -np.ones(n)
    b_ub = np.full(K, M)

    options = {'presolve': True, 'dual_feasibility_tolerance': 1e-9,
               'primal_feasibility_tolerance': 1e-9, 'time_limit': 15}
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs', options=options)

    if res.success:
        return res.x
    return None

def mgd_optimize(a0, time_limit, top_k=40, eta0=0.04, verbose=False):
    """
    Multiplicative Gradient Descent on the simplex with multi-lag subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = eta0
    patience = 0
    max_patience = 200
    cur_top_k = top_k
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 2.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        # Compute subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * a[ki - idx]

        grad = grad - grad.mean()
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()

        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.998
        else:
            eta *= 0.5
            if eta < 0.0005:
                cur_top_k = max(5, cur_top_k // 2)
                eta = 0.02
                patience = max(0, patience - 100)

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def coord_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: move mass from peak-contributing positions to low-impact ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score candidates by their contribution to peaks
            peak_contrib = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]

            # Find position with lowest peak contribution
            dst = np.argmin(peak_contrib)
            if dst == src:
                candidates = np.argsort(peak_contrib)
                dst = candidates[1] if len(candidates) > 1 else src
            if dst == src:
                continue

            for frac in [0.01, 0.02, 0.04, 0.06, 0.08]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a

    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        families.append((i_arr + 1.0) ** p)
        families.append((n - i_arr).clip(0.5) ** p)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            families.append(np.exp(-((x - c) ** 2) / (2.0 * sigma ** 2)))
    for freq in [1, 2, 3, 5]:
        families.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))
    for rate in [1, 3, 10]:
        families.append(np.exp(-rate * x))
        families.append(np.exp(rate * (x - 1.0)))
    # Periodic-like patterns
    for period in [4, 6, 8, 12]:
        pattern = np.ones(n)
        for i in range(n):
            pattern[i] = 1.0 + 0.5 * np.cos(2 * np.pi * i / period)
        families.append(np.maximum(pattern, 0.1))

    best_local = best_a.copy()
    best_local_score = best_score

    for fam in families:
        fam_norm = norm_sum1(fam)
        ac = fast_autocorr(fam_norm)
        val = 2.0 * n * ac.max()
        if val < best_local_score:
            best_local_score = val
            best_local = fam_norm.copy()

    return best_local, best_local_score


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining LP-based optimization with MGD and coordinate descent.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = norm_sum1(p)
            p_score = evaluate_safe(p_norm)
            if np.isfinite(p_score):
                best_seq = p_norm
                best_score = p_score
                print(f"Parent: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Parametric search ===
    print(f"\n=== Phase 1: Parametric (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        a_p, s_p = parametric_init(n0, best_seq, 2.0 * n0 * fast_autocorr(best_seq).max())
        sc_p = evaluate_safe(a_p)
        if np.isfinite(sc_p) and sc_p < best_score:
            best_score = sc_p
            best_seq = a_p.copy()
            print(f"  Parametric improved to {sc_p:.6f}")

    # === Phase 2: LP-based optimization (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP main (n={n_best}) ===")
    t2 = min(remaining * 0.30, 420.0)
    if t2 > 30:
        a_out = lp_optimize(norm_sum1(best_seq), t2, K=120, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST LP: {sc:.6f}")

    # === Phase 3: LP round 2 (different K) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP r2 (n={n_best}) ===")
    t3 = min(remaining * 0.20, 280.0)
    if t3 > 30:
        a_out = lp_optimize(norm_sum1(best_seq), t3, K=60, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST LP r2: {sc:.6f}")

    # === Phase 4: MGD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: MGD polish (n={n_best}) ===")
    t4 = min(remaining * 0.12, 160.0)
    if t4 > 20:
        a_out = mgd_optimize(norm_sum1(best_seq), t4, top_k=40, eta0=0.02, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST MGD: {sc:.6f}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord polish (n={n_best}) ===")
    t5 = min(remaining * 0.10, 130.0)
    if t5 > 15:
        a_out = coord_polish(norm_sum1(best_seq), t5, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Coord: {sc:.6f}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 8000]

    for new_n in longer_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 6: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST n={new_n} LP: {sc:.6f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.20, 60.0)
            if t6b > 10:
                a_out = mgd_optimize(norm_sum1(best_seq), t6b, top_k=30, eta0=0.015, verbose=False)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST n={new_n} MGD: {sc:.6f}")

    # === Phase 7: Final MGD + Coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:

            a_out = mgd_optimize(norm_sum1(best_seq), t7, top_k=50, eta0=0.008, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final MGD: {sc:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.40, 80.0)
        if t8 > 10:
            a_out = coord_polish(norm_sum1(best_seq), t8, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final coord: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
