    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(a, tau, num_lags=None):
    """Compute gradient of smoothed max using log-sum-exp approximation.
    Returns (gradient, smoothed_max_value)."""
    n = len(a)
    ac = fast_autocorr(a)
    if num_lags is not None:
        num_lags = min(num_lags, len(ac))
        ac_work = ac[:num_lags]
    else:
        ac_work = ac
    ac_max = ac_work.max()
    # Shift to avoid overflow
    shifted = tau * (ac_work - ac_max)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    # Gradient of 0.5*sum_k a^T T_k a * p_k w.r.t. a
    # = sum_k p_k * (T_k a) where (T_k a)[j] = a[k-j] (valid range)
    # Fast computation using cumsum
    p_cum = np.concatenate([[0.0], np.cumsum(p)])
    k_lo = np.arange(n)
    k_hi = np.minimum(k_lo + n, len(ac_work))
    S = p_cum[k_hi] - p_cum[k_lo]
    # Cross term: for each j, sum_k p[k] * a[k-j] for valid k
    # This is a correlation between p and a
    cross = sp_correlate(p, a, mode='full')[n-1:2*n-1]
    g = a * S + cross
    smoothed_val = ac_max + np.log(exp_vals.sum()) / tau
    return g, smoothed_val, ac


def smoothed_gradient_bb(a0, time_limit, verbose=True, tau_min=5.0, tau_max=500.0,
                          initial_tau=None, rng_seed=42):
    """Smoothed projected gradient with Barzilai-Borwein step sizes and momentum."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_obj = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    if initial_tau is None:
        current_tau = tau_min
    else:
        current_tau = initial_tau

    # Track BB step sizes
    a_prev = a.copy()
    g_prev = None
    last_step = None
    last_grad_diff = None

    stall = 0
    max_stall = 2000
    iter_count = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        # Adaptive temperature: start low, increase to focus on top peaks
        progress = min(1.0, elapsed / min(time_limit, time_limit * 0.9))
        current_tau = tau_min * (tau_max / tau_min) ** (progress ** 0.7)

        g, sm_val, ac = compute_smoothed_grad(a, current_tau)
        g_tangent = g - g.mean()
        
        # Compute BB step size
        if last_step is not None and last_grad_diff is not None:
            s_norm_sq = np.dot(last_step, last_step)
            y_norm_sq = np.dot(last_grad_diff, last_grad_diff)
            sy = np.dot(last_step, last_grad_diff)
            if s_norm_sq > 1e-20 and y_norm_sq > 1e-20:
                eta_plus = s_norm_sq / sy if sy > 1e-15 else 0.02
                eta_minus = sy / y_norm_sq
                eta = 0.5 * (eta_plus + eta_minus)
                eta = np.clip(eta, 0.001, 0.5)
            else:
                eta = 0.02
        else:
            eta = 0.02

        # Momentum
        if g_prev is not None:
            momentum = 0.9
            g_mom = g_tangent + momentum * (g_tangent - (g_prev - g_prev.mean()))
        else:
            g_mom = g_tangent

        # Scale momentum gradient
        g_scale = np.linalg.norm(g_tangent) / (np.linalg.norm(g_mom) + 1e-15)
        g_mom = g_mom * g_scale

        a_trial = a - eta * g_mom
        a_new = project_simplex(a_trial)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        obj_new = 2.0 * n * m_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                eta *= 0.85
            if stall > 300:
                # Perturbation
                pert = rng.normal(0, 0.002 * stall / 300, n) * a
                a_p = a + pert
                a_p = np.maximum(a_p, 1e-15)
                a_p = a_p / a_p.sum()
                m_p = fast_autocorr(a_p).max()
                if 2.0 * n * m_p < best_obj:
                    a = a_p
                    best_obj = 2.0 * n * m_p
                    best_m = m_p
                    best_a = a.copy()
                    stall = 0
                else:
                    a = a_new
            else:
                a = a_new

        # Update BB tracking
        last_step = a_new - a_prev
        last_grad_diff = g_tangent - (g_prev - g_prev.mean()) if g_prev is not None else None
        g_prev = g_tangent.copy()
        a_prev = a_new.copy()

        total_iters += 1
        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothGradBB iter={total_iters} ({elapsed:.0f}s): obj={best_obj:.6f}, tau={current_tau:.1f}, eta={eta:.5f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe_advanced(a0, time_limit, solver, verbose=True, top_K_init=300,
                             top_K_max=None, momentum=0.0, momentum_decay=0.01):
    """LP-based Frank-Wolfe with momentum and adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    v = np.zeros(n)  # momentum buffer

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix T_sel where T_sel[i,j] = a[k_i - j] for valid j
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 30, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Momentum update
        v = (1 - momentum_decay) * v + momentum * (g_dir - a)
        # Line search with momentum-adjusted direction
        direction = g_dir + v / (np.linalg.norm(v) + 1e-15) if momentum > 0 else g_dir
        
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                  0.3, 0.2, 0.15, 0.1, 0.05]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 5 == 0:
            K = min(K + 20, top_K_max)

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True, rng_seed=77):
    """Direct peak shaving: iteratively reduce the highest autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    top_k = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:top_k]

        improved = False
        for k in top_peaks[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            idx = np.arange(j_start, j_end)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the pair contributing most to this peak
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Find destination: low-sensitivity location
            # Compute sensitivity for each position
            sens = np.zeros(n)
            for kp in top_peaks[:5]:
                kvp = int(kp)
                s2 = max(0, kvp - n + 1)
                e2 = min(n - 1, kvp)
                idx2 = np.arange(s2, e2 + 1)
                sens[idx2] += a[kvp - idx2]
            
            # Try moving mass to low-sensitivity positions
            low_sens = np.argsort(sens)[:20]
            found = False
            for dst in low_sens:
                if dst == src or found:
                    continue
                if sens[dst] > 0.5 * sens[src]:
                    continue
                # Try different fractions
                for frac in [0.005, 0.01, 0.02, 0.015, 0.008, 0.003]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
            if found:
                stall = 0
            else:
                stall += 1

        if not improved and stall > 50:
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 10

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def nesterov_peak_subgrad(a0, time_limit, verbose=True, top_k=30, beta_sub=8.0,
                           c0=3.0, rng_seed=55):
    """Nesterov-accelerated soft-subgradient method for the max of autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta_sub
    beta_nesterov = 0.5

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        beta_t = (1 + beta_nesterov) / (2 + t)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi_val, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi_val * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        grad_tangent = grad - grad.mean()
        
        # Multiplicative update (exponentiated gradient)
        log_a = np.log(y) - eta * grad_tangent
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive parameters
        if stall_count > 3000:
            cur_beta = max(cur_beta * 0.95, 3.0)
            cur_top_k = min(cur_top_k + 10, 100)

        if stall_count > 8000:
            # Restart with perturbation
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 20, 150)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Nesterov t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, k={cur_top_k}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except (NameError, Exception):
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        # Use an asymmetric exponential as initial guess
        a_init = np.exp(-1.5 * x)
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting optimization with n={n0}, score={best_score:.6f}")

    # === Phase 1: Nesterov peak subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov peak subgrad (n={n0}) ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_peak_subgrad(a_in, t1, verbose=True, top_k=40,
                                          beta_sub=7.0, c0=4.0, rng_seed=11)
        try_update(a_out, "nesterov1")

    # === Phase 2: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW advanced (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(400, 2 * n_best - 1)
        top_K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe_advanced(a_in, t2, solver, verbose=True,
                                            top_K_init=top_K_init,
                                            top_K_max=top_K_max,
                                            momentum=0.0)
        try_update(a_out, "lp_adv1")

    # === Phase 3: Smoothed gradient BB ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: SmoothGradBB (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_gradient_bb(a_in, t3, verbose=True,
                                         tau_min=8.0, tau_max=400.0,
                                         rng_seed=42)
        try_update(a_out, "smoothbb1")

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak shaving (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t4, verbose=True, rng_seed=77)
        try_update(a_out, "peakshave1")

    # === Phase 5: LP-FW round 2 with larger K ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(800, 2 * n_best - 1)
        top_K_max = min(6000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe_advanced(a_in, t5, solver, verbose=False,
                                            top_K_init=top_K_init,
                                            top_K_max=top_K_max,
                                            momentum=0.3)
        try_update(a_out, "lp_adv2")

    # === Phase 6: Final smooth gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final smooth (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_gradient_bb(a_in, t6, verbose=False,
                                         tau_min=50.0, tau_max=600.0,
                                         initial_tau=100.0, rng_seed=99)
        try_update(a_out, "smooth_final")

    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final peak shave (n={n_best}) ===")
    t7 = min(remaining * 0.06, 100.0)
    if t7 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t7, verbose=False, rng_seed=123)
        try_update(a_out, "peakshave_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.exp(-1.5 * x)
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
