    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def _interp(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = (np.arange(old_n) + 0.5) / old_n
    nx = (np.arange(new_n) + 0.5) / new_n
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:

    return fi / ss

        try_update(w * a_base + (1 - w) * a_lin, f"Bl_l{w:.1f}")
        try_update(w * a_base + (1 - w) * a_gauss, f"Bl_g{w:.1f}")
    
    print(f"  Parametric done ({time.monotonic()-t0:.1f}s): best J={best_val:.6f}")
    return best_a, best_val

def _md_optimize(a0, t_limit, T_start, T_end, eta0, nesterov=True, verbose=True):
    """Mirror descent with soft-max subgradient."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_M = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    iter_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > t_limit:
            break
        
        # Nesterov extrapolation
        if nesterov and t > 1:
            beta = (t - 1.0) / (t + 1.5)
            y = x + beta * (x - x_prev)
            y = _norm_sum1(y)
        else:
            y = x.copy()
        
        ac = _fast_autocorr(y)
        
        # Adaptive temperature
        frac = min(1.0, (t - t0) / t_limit)  # 0 to 1
        T = T_start * (T_end / T_start) ** frac  # geometric interpolation
        
        grad = _soft_subgrad(y, ac, T)
        
        eta = eta0 / (1.0 + 0.1 * np.sqrt(t))
        log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = _fast_autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            iter_count += 1
            if verbose and (iter_count <= 5 or iter_count % 50 == 0):
                el = time.monotonic() - t0
                print(f"    MD t={t} ({el:.1f}s): J={2*n*best_M:.6f}, T={T:.4f}")
    
    el = time.monotonic() - t0
    if verbose:
        print(f"    MD done ({el:.1f}s): J={2*n*best_M:.6f}")
    return best_a, best_M
