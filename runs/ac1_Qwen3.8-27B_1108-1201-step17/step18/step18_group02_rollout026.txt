"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Ramp-based and multi-structure initialization (exploiting asymmetry)
2. Frank-Wolfe with backtracking line search and adaptive constraint management
3. Smoothed multiplicative Adam with log-sum-exp (spectral perspective)

Key insight: optimal sequences are asymmetric; ramps give ~4/3 in continuous limit.

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_subgradient(a, ac, tau):
    n = len(a)
    ac_scaled = ac / tau
    ac_scaled -= ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()
    a_flip = a[::-1].astype(np.float64)
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)


def smooth_adam(a0, time_limit, tau_decay=0.9998, lr=0.7, beta1=0.9, beta2=0.999,
                restart_every=2000, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    tau = max(ac.max() * 0.5, 1e-6)
    tau_min = max(ac.max() * 0.005, 1e-10)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        tau = max(tau * tau_decay, tau_min)
        g = lse_subgradient(a, ac, tau)
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        step = lr / (np.sqrt(v_c) + 1e-8)
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.04, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Adam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")

    elapsed = time.monotonic() - t0
    print(f"    Adam done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def frank_wolfe_linesearch(a0, time_limit, solver, initial_K=400, max_K=5000,
                           add_per_iter=60, verbose=True, seed=7):
    n = len(a0)
    a = norm_sum1(a0)
    ac_init = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac_init.max()
    t0 = time.monotonic()
    y = a.copy()
    rng = np.random.default_rng(seed)

    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    iter_count = 0
    lp_fail_count = 0
    stall = 0
    best_stall_m = best_m

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 200:
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            y = x.copy()
            stall = 0
            constraint_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] + 1e-12 for k in constraint_list])
            margins = np.maximum(margins, 0)
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 8 and len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
                constraint_list = sorted(constraint_set)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 12:
                constraint_set = set(int(k) for k in constraint_list[:max(200, initial_K // 2)])
                constraint_list = sorted(constraint_set)
            continue

        lp_fail_count = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Line search: try several alpha values
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        ac_g = fast_autocorr(g_dir)

        # Add violated constraints
        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Remove least active constraints (keep some headroom)
        if len(constraint_set) > max_K + 500:
            margins2 = np.array([M_curr - ac_y[k] + 1e-12 for k in constraint_set])
            margins2 = np.maximum(margins2, 0)
            keep_idx = np.argsort(margins2)[:max_K]
            constraint_list_sorted = sorted(constraint_set)
            constraint_set = set(int(constraint_list_sorted[i]) for i in keep_idx)

        # Backtracking line search
        alphas = [1.0, 0.8, 0.6, 0.4, 0.25]
        best_alpha = 2.0 / (t + 2.0)  # Default
        best_f = 2.0 * n * M_curr
        for alpha in alphas:
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            s_t = y_trial.sum()
            if s_t < 1e-15:
                continue
            y_trial = y_trial / s_t
            ac_trial = fast_autocorr(y_trial)
            M_trial = ac_trial.max()
            f_trial = 2.0 * n * M_trial
            if f_trial < best_f - 1e-12:
                best_f = f_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new
        iter_count += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW-LN iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, alpha={best_alpha:.2f}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    FW-LN done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_descent(a0, time_limit, rng_seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 60:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_inits(n):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    # Linear ramp (key: gives ~4/3 in continuous limit)
    for shift in [0, 0.1, 0.2, 0.3]:
        a = (i_arr + 1.0 + shift * n)
        a = np.maximum(a, 0.1)
        candidates.append(a / a.sum())

    # Quadratic ramp
    for shift in [0, 0.2]:
        a = (i_arr + 1.0) ** 2
        candidates.append(a / a.sum())
        a = (n - i_arr + shift * n) ** 2
        a = np.maximum(a, 0.1)
        candidates.append(a / a.sum())

    # Cubic
    a = (i_arr + 1.0) ** 3
    candidates.append(a / a.sum())

    # Mixed: ramp * envelope
    for env_c in [0.3, 0.5, 0.7]:
        env = np.exp(-((x - env_c) ** 2) / 0.1)
        a = (i_arr + 1.0) * env
        candidates.append(a / a.sum())

    # Power law
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())

    # Asymmetric Gaussian pair
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            a = np.exp(-((x - c1) ** 2) / 0.04) + 2.0 * np.exp(-((x - c2) ** 2) / 0.04)
            candidates.append(a / a.sum())

    # Sine-based asymmetric
    for freq in [1, 2]:
        for phase in [0.0, 0.5, 1.0]:
            a = np.maximum(np.sin(np.pi * freq * x + phase), 0.05)
            candidates.append(a / a.sum())

    # Uniform (baseline)
    candidates.append(np.ones(n) / n)

    # Log ramp
    a = np.log(i_arr + 2.0)
    candidates.append(a / a.sum())

    # 1/(x+c) type
    for c in [0.1, 0.3, 0.5]:
        a = 1.0 / (x + c)
        candidates.append(a / a.sum())

    # Sawtooth-ish
    a = ((i_arr + 1.0) % 3.0) + 0.5
    candidates.append(a / a.sum())

    return candidates

                                          verbose=True, seed=7)
        check_and_update(a_out, f"fw1_n{n_best}")

    # === Phase 3: Smoothed Adam ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smoothed Adam (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam(a_in, t3, tau_decay=0.9998, lr=0.6,
                               beta1=0.9, beta2=0.999, restart_every=1500,
                               verbose=True, seed=123)
        check_and_update(a_out, f"adam_n{n_best}")

    # === Phase 4: Second FW pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: FW-LineSearch pass 2 (n={n_best}) ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = frank_wolfe_linesearch(a_in, t4, solver, initial_K=800,
                                          max_K=6000, add_per_iter=100,
                                          verbose=True, seed=99)
        check_and_update(a_out, f"fw2_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length ===")
    target_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000]:
        new_n = n_best + delta
        if 2500 <= new_n <= 8000 and new_n != n_best:
            target_lengths.append(new_n)

    seen = set()
    unique_lengths = []
    for ln in target_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    for new_n in unique_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n  --- n={new_n} ---")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick FW at new length
        t5 = min(remaining * 0.2, 150.0)
        if t5 > 20:
            a_out, _ = frank_wolfe_linesearch(a_long, t5, solver, initial_K=400,
                                              max_K=3000, add_per_iter=50,
                                              verbose=False, seed=new_n % 100)
            check_and_update(a_out, f"fw_n{new_n}")

    # === Phase 6: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord descent (n={n_best}) ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent(a_in, t6, rng_seed=271, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")

    # === Phase 7: Final FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final FW (n={n_best}) ===")
    t7 = min(remaining * 0.35, 300.0)
    if t7 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = frank_wolfe_linesearch(a_in, t7, solver, initial_K=1000,
                                          max_K=7000, add_per_iter=120,
                                          verbose=True, seed=456)
        check_and_update(a_out, f"fw_final_n{n_best}")

    # === Phase 8: Final coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord (n={n_best}) ===")
    t8 = min(remaining * 0.4, 100.0)
    if t8 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent(a_in, t8, rng_seed=789, verbose=False)
        check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        i_arr = np.arange(n_fb, dtype=np.float64)
        best_seq = (i_arr + 1.0)
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
