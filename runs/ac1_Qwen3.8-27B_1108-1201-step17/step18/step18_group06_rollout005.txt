    if old_n == new_n:

    s = f.sum()
    if s < 1e-15:

    old_x = np.linspace(0.0, 1.0, old_n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation via cross-correlation (O(n log n) per step)
2. Nesterov-accelerated mirror descent on simplex with adaptive temperature
3. Softmax-smoothed subgradient over all lags
4. Periodic peak-shaving coordinate moves targeting worst pairs
5. Spectral domain refinement (Gerchberg-Saxton variant)
6. Multi-length search with interpolation and re-optimization
Key insight: gradient of weighted autocorrelation w.r.t. a is a cross-correlation,
computable as a single FFT, enabling millions of fast iterations.
"""

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_power_law_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_mild_ramp(n, center_frac=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 + center_frac * x
    return norm_sum1(a)


def make_exponential_ramp(n, rate=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return norm_sum1(a)


def compute_fft_gradient(a, w, N):
    """Compute grad[i] = sum_j a[j]*w[i+j] via FFT cross-correlation."""
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    R = np.fft.ifft(W * np.conj(A))
    return R[:len(a)].real


def mirror_descent_fft(a0, time_limit, n_peaks_temp=0.005, c0=3.0,
                       restart_every=20000, verbose=False, rng_seed=42,
                       max_iters=50000000):
    """Nesterov mirror descent with FFT-based gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    C = fast_autocorr(a)
    best_m = C.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    N = 1 << (3 * n - 2).bit_length()
    x = a.copy()
    x_prev = a.copy()
    v = np.zeros(n)  # momentum
    gamma = c0
    stall_count = 0
    cur_temp = n_peaks_temp
    restart_count = 0
    tfinal = max_iters
    lam = 2.0 / (tfinal + 2.0)  # Nesterov weight

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov: y = x + alpha_t * (x - x_prev)
        alpha_t = (1.0 + np.sqrt(1.0 + 4.0 * lam * lam * (t - 1))) / (2.0 * lam * (t + 1))
        y = x + alpha_t * (x - x_prev)
        y = np.maximum(y, 1e-18)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation
        C = fast_autocorr(y)
        M = C.max()
        C_mean = C.mean()

        # Softmax weighting over all lags
        # w[k] proportional to exp((C[k] - M) / (T * M))
        scale = cur_temp * M
        if scale < 1e-30:
            peaks = np.where(C >= M - 1e-10 * M)[0]
            w = np.zeros(len(C))
            w[peaks] = 1.0 / len(peaks)
        else:
            log_w = (C - M) / scale
            log_w -= log_w.max()
            w = np.exp(log_w)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(len(C)) / len(C)
            else:
                w = w / w_sum

        # Compute gradient via FFT
        grad = compute_fft_gradient(y, w, N)
        grad = np.maximum(grad, 0.0)  # gradient of C w.r.t. a is always >= 0 for a >= 0

        # Momentum (Nesterov)
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * lam * lam * (t - 1))) / (2.0 * lam * (t + 1))
        v = beta_t * v + (1.0 - beta_t) * grad

        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-18)) - eta * v
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Evaluate
        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive temperature: decrease to focus on hardest peaks
        cur_temp = max(0.001, n_peaks_temp * np.exp(-0.02 * stall_count / 1000.0))

        # Restart
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.1, 0.02 + 0.005 * min(restart_count, 8))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            v = np.zeros(n)
            gamma = c0 * (1.0 + 0.03 * min(restart_count, 10))
            cur_temp = max(0.001, n_peaks_temp * 0.7)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, temp={cur_temp:.5f}")

        if verbose and t % 100000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}, M={best_m:.2e}")

    return best_a, best_m


def peak_shave(a0, time_limit, verbose=False, rng_seed=42):
    """Targeted peak shaving: reduce worst-contributing pairs."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    C = fast_autocorr(a)
    best_m = C.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    N = 1 << (3 * n - 2).bit_length()
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        C = fast_autocorr(a)
        M = C.max()
        # Top peaks
        n_top = min(30, 2 * n - 1)
        top_k = np.argsort(C)[::-1][:n_top]

        improved = False
        for ki in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(ki)
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue

            # Find worst pair
            src_local = np.argmax(contrib)
            src = idx[src_local]
            partner = ki - src

            if a[src] < 1e-12 or a[partner] < 1e-12:
                continue

            # Find destination: minimize impact on other peaks
            # Cost = sum over top peaks of contribution if we add here
            cost = a.copy()
            for k2 in top_k[:10]:
                k2i = int(k2)
                j2_lo = max(0, k2i - n + 1)
                j2_hi = min(n - 1, k2i)
                if j2_lo <= j2_hi:
                    valid = np.arange(j2_lo, j2_hi + 1)
                    cost[valid] += 0.1 * a[k2i - valid]

            # Pick lowest cost destination among low-value positions
            candidates = np.argsort(cost + 5.0 * a)[:max(1, n // 5)]
            dst = candidates[np.argmin(a[candidates])]
            if dst == src or dst == partner:
                dst = int(rng.integers(n))
                if dst == src or dst == partner:
                    continue

            # Try moving mass
            found = False
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.35]:
                amt = min(a[src], a[partner]) * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[partner] -= amt
                a_trial[dst] += amt * 2
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                C_t = fast_autocorr(a_trial)
                if C_t.max() < best_m - 1e-16:
                    a = a_trial
                    best_m = C_t.max()
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Single-sided move
                if a[src] > 1e-10:
                    amt = a[src] * rng.uniform(0.01, 0.05)
                    a[src] -= amt
                    dst2 = int(rng.integers(n))
                    if dst2 != src:
                        a[dst2] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        C_new = fast_autocorr(a)
                        if C_new.max() < best_m:
                            best_m = C_new.max()
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.005, 0.03)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                C_new = fast_autocorr(a)
                if C_new.max() < best_m:
                    best_m = C_new.max()
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def spectral_refine(a0, time_limit, verbose=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    C = fast_autocorr(a)
    best_m = C.max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    n_iters = 0

    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        n_iters += 1

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Progressive flattening
        alpha = 0.3 + 0.6 * min(1.0, it / 500.0)
        P_clip = np.maximum(P, 1e-30)
        ratio = P_clip / P_mean
        P_target = P_mean * ratio ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target / P_clip)
        mag_new = np.clip(mag_new, 0.8, 1.25)
        A_new = A * mag_new

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        # Damped update
        blend = min(0.5, 0.1 + 0.001 * it)
        a = (1 - blend) * a + blend * a_new
        a = norm_sum1(a)

        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = peak_shave(norm_sum1(best_seq), t3, verbose=False, rng_seed=77)
        try_update(a_out, "peakshave")
        # Another pass
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.08, 100.0)
        if t3b > 15:
            a_out, _ = peak_shave(norm_sum1(best_seq), t3b, verbose=False, rng_seed=177)
            try_update(a_out, "peakshave2")

    # === Phase 4: Spectral refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Spectral refinement (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_out, _ = spectral_refine(norm_sum1(best_seq), t4, verbose=False)
        try_update(a_out, "spectral")

    # === Phase 5: Multi-length fine search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length fine search ===")

    lengths_to_try = []
    for delta in [-200, -400, -800, 200, 400, 800, 1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))

    for new_n in lengths_to_try[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        # Quick MD on interpolated
        t5a = min(remaining * 0.25, 120.0)
        if t5a > 20:
            a_out, _ = mirror_descent_fft(a_long, t5a, n_peaks_temp=0.005, c0=3.0,
                                           restart_every=15000, verbose=False, rng_seed=300)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.15, 60.0)
            if t5b > 10:
                a_out, _ = peak_shave(norm_sum1(best_seq), t5b, verbose=False, rng_seed=400)
                try_update(a_out, f"peakred_n{new_n}")

    # === Phase 6: Final intensive polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, score={best_score:.6f}) ===")
    if remaining > 50:
        t6 = min(remaining * 0.45, 300.0)
        if t6 > 20:
            print(f"  Final MD for {t6:.0f}s...")
            a_out, _ = mirror_descent_fft(norm_sum1(best_seq), t6, n_peaks_temp=0.002, c0=2.0,
                                           restart_every=30000, verbose=True, rng_seed=500)
            try_update(a_out, "final_md")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.30, 100.0)
            if t6b > 15:
                a_out, _ = peak_shave(norm_sum1(best_seq), t6b, verbose=False, rng_seed=600)
                try_update(a_out, "final_peakred")

    # === Phase 7: Ultra-fine MD (very low temp, small steps) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t7 = min(remaining * 0.50, 200.0)
        if t7 > 20:
            print(f"\n=== Phase 7: Ultra-fine MD ({t7:.0f}s) ===")
            a_out, _ = mirror_descent_fft(norm_sum1(best_seq), t7, n_peaks_temp=0.001, c0=1.5,
                                           restart_every=40000, verbose=True, rng_seed=700)
            try_update(a_out, "ultra_md")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_power_law_ramp(n_fb, power=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
