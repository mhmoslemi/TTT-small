def fft_conv(a):
    """Linear self-convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), N)
    return np.fft.irfft(A * A, N)[:size]


def norm1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score(a):
    """Compute J = 2n * max(a*a) / (sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    an = a / s
    c = fft_conv(an)
    return 2.0 * n * c.max()


def lse_grad(a, c, tau):
    """Gradient of LSE-smoothed max w.r.t. a."""
    n = len(a)
    cs = c / tau
    cs -= cs.max()
    e = np.exp(cs)
    p = e / e.sum()
    # grad[i] = 2 * sum_k p[k] * a_padded[k-i]
    # = 2 * (p * flip(a))[n-1+i]
    a_f = a[::-1]
    N = 1 << (len(p) + n - 2).bit_length()
    Pf = np.fft.rfft(p.astype(np.float64), N)
    Af = np.fft.rfft(a_f, N)
    conv = np.fft.irfft(Pf * Af, N)
    return 2.0 * conv[n - 1: n - 1 + n]

def _conv_deriv(a, k):
    """Gradient of (a*a)[k] w.r.t. a: result[j] = 2*a_padded[k-j]."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        # g[j] = 2*a[k-j] for j in [lo, hi]
        # k-j ranges from k-lo down to k-hi
        g[lo:hi+1] = 2.0 * a[k-hi:k-lo+1]  # a[k-hi], a[k-hi+1], ..., a[k-lo]
        # Wait, let me be more careful
        # For j = lo: g[lo] = 2*a[k-lo]
        # For j = hi: g[hi] = 2*a[k-hi]
        # So g[lo:hi+1] = 2*a[k-hi : k-lo+1]  (reversed)
        g[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
    return g


def _quad_form(a, d, k):
    """Compute (d*d)[k] = sum_i d[i]*d[k-i]."""
    n = len(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo > hi:
        return 0.0
    # sum_{i=lo}^{hi} d[i]*d[k-i]
    return np.dot(d[lo:hi+1], d[k-hi:k-lo+1][::-1])

def parametric_inits(n):

    inits.append(a / a.sum())
    
    # Linear ramp (decreasing)

        a = (n - i) ** p
        inits.append(a / a.sum())
    
    # Product of two ramps (hump)
    for a1 in [0.3, 0.5, 0.7]:

            if a1 + a2 > 1:

            a = (i + 1) ** a1 * (n - i) ** a2
            a = np.maximum(a, 1e-10)
            inits.append(a / a.sum())
    
    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for s in [0.05, 0.1, 0.2, 0.3]:
            a = np.exp(-((x - c)**2) / (2*s*s))
            inits.append(a / a.sum())

    for p in [0.1, 0.3, 0.5, 1.0, 2.0]:
        a = 1.0 / (i + 1) ** p
        inits.append(a / a.sum())
    
    return inits


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    fn = norm1(f)
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    s = fi.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return fi / s

        m_hat = b1 * m_hat + (1.0 - b1) * g
        v_hat = b2 * v_hat + (1.0 - b2) * g * g
        bc1 = 1.0 - b1 ** t
        bc2 = 1.0 - b2 ** t
        step = lr / (np.sqrt(v_hat / bc2) + 1e-8)
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        if stall > 4000:
            stall = 0
            noise = rng.normal(0, 0.05, n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                print(f"    Adam restart ({time.monotonic()-t0:.0f}s): J={best_j:.6f}")

        if verbose and t % 2000 == 0:
            print(f"    Adam t={t} ({time.monotonic()-t0:.0f}s): J={best_j:.6f}, tau={tau:.2e}")

    return best_a, best_j


def conv_grad(a, k):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k - hi: k - lo + 1][::-1]
    return g


def quad_form(d, k):
    n = len(d)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo > hi:
        return 0.0
    return np.dot(d[lo:hi + 1], d[k - hi: k - lo + 1][::-1])


def peak_shave(a0, tlimit, n_top=3, seed=77, verbose=False):
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_j = 2.0 * n * fft_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)

    for it in range(5000000):
        if time.monotonic() - t0 > tlimit:
            break
        c = fft_conv(a)
        M = c.max()
        J = 2.0 * n * M
        if J < best_j - 1e-13:
            best_j = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 800:
            for _ in range(10):
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-12:
                    amt = a[i1] * rng.uniform(0.01, 0.05)
                    a[i1] -= amt
                    a[i2] += amt
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall = 0
            continue

        top_k = np.argsort(c)[::-1][:n_top].tolist()

        g = np.zeros(n)
        for ki in top_k:
            g += conv_grad(a, int(ki))

        d = g.mean() - g
        gd = np.dot(g, d)

        if abs(gd) < 1e-20:
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-12:
                amt = a[i1] * 0.02
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            continue

        B = 0.0
        for ki in top_k:
            B += quad_form(d, int(ki))

        if B < 1e-20:
            break

        t_uncon = -gd / (2.0 * B)

        neg = d < -1e-15
        if neg.any():
            t_feas = float(np.min(-a[neg] / d[neg]))
        else:
            t_feas = 1.0

        t_opt = min(max(t_uncon, 0.0), t_feas * 0.999, 1.0)

        if t_opt < 1e-12:
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-12:
                amt = a[i1] * 0.01
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            continue

        a = a + t_opt * d
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if verbose and it % 5000 == 0:
            print(f"    PeakShave it={it} ({time.monotonic()-t0:.0f}s): J={best_j:.6f}")

    return best_a, best_j


def coord_desc(a0, tlimit, seed=99, verbose=False):
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_j = 2.0 * n * fft_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(5000000):
        if time.monotonic() - t0 > tlimit:
            break
        c = fft_conv(a)
        M = c.max()
        J = 2.0 * n * M
        if J < best_j - 1e-13:
            best_j = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 400:
            for _ in range(8):
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-12:
                    amt = a[i1] * rng.uniform(0.005, 0.04)
                    a[i1] -= amt
                    a[i2] += amt
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall = 0
            continue

        k_star = int(c.argmax())
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo > hi:
            continue

        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k_star - idx]
        src_local = int(np.argmax(contrib))
        src = int(idx[src_local])

        if a[src] < 1e-14:
            continue

        top_peaks = np.argsort(c)[::-1][:5]
        damage = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for kp in top_peaks:
            kpi = int(kp)
            lp = max(0, kpi - (n - 1))
            hp = min(n - 1, kpi)
            if lp <= hp:
                damage[lp:hp + 1] += a[kpi - hp: kpi - lp + 1][::-1]
                in_peak[lp:hp + 1] = True

        mask = ~in_peak
        if not mask.any():
            mask = np.ones(n, dtype=bool)
        mask[src] = False
        candidates = np.where(mask)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
            candidates = candidates[candidates != src]

        dst = int(candidates[np.argmin(damage[candidates])])

        lo_t = 0.0
        hi_t = a[src] * 0.4
        for _ in range(18):
            mid_t = (lo_t + hi_t) / 2.0
            a_trial = a.copy()
            a_trial[src] -= mid_t
            a_trial[dst] += mid_t
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                hi_t = mid_t
                break
            a_trial = a_trial / s_t
            J_trial = 2.0 * n * fft_conv(a_trial).max()
            if J_trial < best_j:
                lo_t = mid_t
            else:
                hi_t = mid_t

        if lo_t > 1e-10:
            a[src] -= lo_t
            a[dst] += lo_t
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        if verbose and it % 5000 == 0:
            print(f"    Coord it={it} ({time.monotonic()-t0:.0f}s): J={best_j:.6f}")

    return best_a, best_j


def parametric_inits(n):
    x = np.linspace(0, 1, n)
    i = np.arange(n, dtype=np.float64)
    inits = []

    for f in [1, 2, 3]:
        for ph in [0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * f * x + ph) ** 2, 0.001)
            inits.append(a / a.sum())

    a = i + 1.0
    inits.append(a / a.sum())
    a = (n - i).astype(float) + 0.1
    inits.append(a / a.sum())

    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (i + 1) ** p
        inits.append(a / a.sum())
        a = (n - i) ** p + 0.1
        a = np.maximum(a, 0.1)
        inits.append(a / a.sum())

    for a1 in [0.3, 0.5, 1.0]:
        for a2 in [0.3, 0.5, 1.0]:
            a = (i + 1) ** a1 * (n - i) ** a2
            a = np.maximum(a, 1e-10)
            inits.append(a / a.sum())

    for c in [0.3, 0.5, 0.7]:
        for s in [0.05, 0.1, 0.15, 0.2, 0.3]:
            a = np.exp(-((x - c) ** 2) / (2 * s * s))
            inits.append(a / a.sum())

    a = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    inits.append(a / a.sum())
    inits.append(np.ones(n) / n)

    for p in [0.1, 0.3, 0.5, 1.0, 2.0, 3.0]:
        a = 1.0 / (i + 1) ** p
        inits.append(a / a.sum())

    return inits


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    fn = norm1(f)
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    s = fi.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return fi / s
