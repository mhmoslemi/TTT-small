def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    """Normalize to probability simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lse_smooth(a, beta):
    """Compute log-sum-exp smoothed objective and probability weights."""
    n = len(a)
    c = fast_conv(a)
    max_c = c.max()
    if max_c < 1e-30:
        return 0.0, np.ones(len(c)) / len(c)
    shifted = beta * (c - max_c)
    shifted = np.clip(shifted, -50, 0)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    if Z < 1e-30:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_shifted / Z
    obj = max_c + np.log(Z) / beta
    return obj, p, c

def compute_gradient(a, p):
    """Compute gradient of LSE objective w.r.t. a.
    grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    """
    n = len(a)
    a_rev = a[::-1]
    # conv_result[k] = sum_j p[j] * a_rev[k-j]
    # We want: grad[i] = 2 * sum_j p[j] * a[i+j] = 2 * conv_result[i + n - 1]
    fft_size = 1 << (3 * n - 1).bit_length()  # need size >= (2n-1) + n - 1 = 3n-2
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_full = np.fft.irfft(P * AR, fft_size)
    grad = 2.0 * conv_full[n - 1 : 2 * n - 1]
    return grad

def mirror_descend_step(a, grad, eta):
    """Mirror descent (exponentiated gradient) step on the simplex."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - eta * grad
    max_log = log_a_new.max()
    a_new = np.exp(log_a_new - max_log)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

        rows_sub = []

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based active-set optimization (from SOTD description)

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_conv(p, a):
    """Compute sum_j p[j] * a[i+j] for i=0..n-1 via FFT."""
    n = len(a)
    a_rev = a[::-1]
    total_len = len(p) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    result = np.fft.irfft(P * AR, fft_size)
    return result[n - 1 : n - 1 + n]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_lse_grad(a, beta):
    """Compute gradient of the log-sum-exp smoothed objective.
    Returns (smoothed_obj, gradient, raw_conv).
    """
    n = len(a)
    c = fast_conv(a)
    max_c = c.max()
    if max_c < 1e-30:
        p = np.ones(len(c)) / len(c)
        return 0.0, np.zeros(n), c
    shifted = beta * (c - max_c)
    shifted = np.clip(shifted, -60.0, 0.0)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    if Z < 1e-30:
        p = np.ones(len(c)) / len(c)
        obj = max_c
    else:
        p = exp_s / Z
        obj = max_c + np.log(Z) / beta
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # = 2 * (cross_correlation of p and a)[i]
    grad = 2.0 * fast_cross_conv(p, a)
    return obj, grad, c


def mirror_step(a, grad, eta):
    """Exponentiated gradient (mirror descent) step on simplex."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_new = log_a - eta * grad
    max_log = log_new.max()
    new = np.exp(log_new - max_log)
    s = new.sum()
    if s < 1e-15:
        return a.copy()
    return new / s


def lse_mirror_descent(a0, time_limit, verbose=False, beta_start=5.0, beta_end=2000.0,
                       n_iters=None, momentum_decay=0.95):
    """Nesterov-accelerated mirror descent with annealed log-sum-exp."""
    n = len(a0)
    a = norm_simplex(a0).copy()
    best_a = a.copy()
    c0 = fast_conv(a)
    best_M = c0.max()
    t0 = time.monotonic()

    if n_iters is None:
        n_iters = 100000

    # Nesterov momentum
    v = np.zeros(n)
    momentum = 0.0

    # Step size
    eta = 1.0 / np.sqrt(n) * 2.0

    stall_count = 0
    max_stall = 200

    prev_a = a.copy()
    iter_count = 0

    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        iter_count = t

        # Annealing schedule (exponential)
        progress = min(1.0, t / (n_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress

        # Nesterov extrapolation
        momentum = momentum_decay ** t
        y = a + momentum * (a - prev_a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Compute gradient at y
        obj_y, grad_y, c_y = compute_lse_grad(y, beta)

        # Step size decay
        eta_t = eta / (1.0 + 0.01 * t)

        # Mirror descent step
        a_new = mirror_step(y, grad_y, eta_t)

        # Line search: try a few alpha values
        best_alpha = 0.0
        best_obj_val = fast_conv(a).max()
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            h = (1.0 - alpha) * a + alpha * a_new
            h = np.maximum(h, 0.0)
            sh = h.sum()
            if sh < 1e-15:
                continue
            h = h / sh
            obj_h = fast_conv(h).max()
            if obj_h < best_obj_val:
                best_obj_val = obj_h
                best_alpha = alpha

        if best_alpha < 0.01:
            # Try smaller steps
            for alpha in [0.01, 0.03, 0.05]:
                h = (1.0 - alpha) * a + alpha * a_new
                h = np.maximum(h, 0.0)
                sh = h.sum()
                if sh < 1e-15:
                    continue
                h = h / sh
                obj_h = fast_conv(h).max()
                if obj_h < best_obj_val:
                    best_obj_val = obj_h
                    best_alpha = alpha

        if best_alpha < 1e-6:
            stall_count += 1
            if stall_count > max_stall:
                # Restart with perturbation
                rng_local = np.random.default_rng(t)
                pert = rng_local.normal(0, 0.01, n)
                a = norm_simplex(a + pert)
                stall_count = 0
                prev_a = a.copy()
                continue
        else:
            stall_count = 0

        prev_a = a.copy()
        a = (1.0 - best_alpha) * a + best_alpha * a_new
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa < 1e-15:
            a = best_a.copy()
        else:
            a = a / sa

        M_new = fast_conv(a).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-MD t={t} ({elapsed:.0f}s): J={2*n*best_M:.6f}, beta={beta:.1f}")

    return best_a, best_M


def lp_optimize(a0, time_limit, max_iters=200, verbose=False):

    no_improve_count = 0
    max_no_improve = 30

        if no_improve_count >= max_no_improve:
            break
        it += 1

        c_curr = fast_conv(a_curr)
        M_curr = c_curr.max()

        # Active set: top K peaks
        if it <= 10:
            K = 30
        elif it <= 30:
            K = 60
        elif it <= 60:
            K = 100
        else:
            K = 150
        K = min(K, n_constraints)

        top_idx = np.argsort(c_curr)[::-1][:K]

        # Build sparse constraint matrix
        rows = []
        cols = []
        vals = []
        for k in top_idx:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            for i in range(i_start, i_end):
                rows.append(ki)
                cols.append(i)
                vals.append(a_curr[ki - i])

        if len(rows) == 0:
            break

        np_rows = np.array(rows)
        np_cols = np.array(cols)
        np_vals = np.array(vals)

        A_mat = csr_matrix((np_vals, (np_rows, np_cols)), shape=(n_constraints, n))

        # Small cuboid bound
        b_ub = np.full(n_constraints, M_curr + 1e-10)
        bounds = [(0.0, 5.0)] * n

        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': remaining_lp})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b_star = res.x
        sum_b = b_star.sum()
        if sum_b < 1e-10:
            break

        g = b_star / sum_b

        # Line search (coarse to fine)
        best_t = 0.0
        best_obj_val = M_curr

            h = (1.0 - t_candidate) * a_curr + t_candidate * g

            if sh < 1e-15:

            h = h / sh

                best_t = t_candidate

            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.05)
            for t_candidate in np.linspace(lo, hi, 15):
                h = (1.0 - t_candidate) * a_curr + t_candidate * g
                h = np.maximum(h, 0.0)
                sh = h.sum()
                if sh < 1e-15:

                h = h / sh

                if obj_h < best_obj_val:
                    best_obj_val = obj_h
                    best_t = t_candidate

        if best_t < 1e-8:
            no_improve_count += 1
            # Try perturbing the active set
            top_idx_alt = np.argsort(c_curr)[::-1][:K]
            np.random.shuffle(top_idx_alt)
            continue

        a_curr = (1.0 - best_t) * a_curr + best_t * g
        a_curr = np.maximum(a_curr, 0.0)
        s = a_curr.sum()
        if s < 1e-15:
            a_curr = best_a.copy()
        else:
            a_curr = a_curr / s

        if M_new < best_M - 1e-15:

            best_a = a_curr.copy()
            no_improve_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP it={it} ({elapsed:.0f}s): J={2*n*best_M:.6f}, t={best_t:.4f}")

    return best_a, best_M


def spectral_smooth(a, strength=0.3):
    """One step of spectral flattening."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a

    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.power(ratio, 1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

    blend = min(1.0, 0.5 * strength + 0.5)
    a_out = (1.0 - blend) * a + blend * a_new
    return norm_simplex(a_out)


def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening with adaptive step."""
    n = len(a0)
    a = norm_simplex(a0).copy()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()

    step = 0.2
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 100:
            break

        a_new = spectral_smooth(a, step)
        M_new = fast_conv(a_new).max()

        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            step = min(0.5, step * 1.03)
            stall = 0
        else:
            step = max(0.03, step * 0.85)
            a = norm_simplex(0.5 * a + 0.5 * best_a)
            stall += 1

    return best_a, best_M


def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions targeting top peaks."""
    n = len(a0)
    a = norm_simplex(a0).copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]

        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-12:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Try moving to positions that reduce this peak
            for attempt in range(8):
                # Prefer moving to positions far from the peak contribution
                candidates = np.setdiff1d(np.arange(n), [src])
                if len(candidates) == 0:
                    break
                dst = candidates[int(rng.integers(len(candidates)))]

                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-8:
                frac = rng.uniform(0.02, 0.08)
                amt = a[i1] * frac
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_M


def make_init_bowl(n, sharpness=2.0):
    """Bowl shape: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = np.power(4.0 * (x - 0.5) ** 2, sharpness) + 0.05
    return norm_simplex(a)


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.2, pos2=0.8, width=0.15):

def make_init_w_shape(n):

    a = np.abs(4 * (x - 0.25) - 1) ** 2 * 0.5 + np.abs(4 * (x - 0.75) - 1) ** 2 * 0.5 + 0.1
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    print(f"\n=== Phase 4: Coordinate Descent (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 10:
        a_in = norm_simplex(best_seq)
        a_out, _ = coordinate_descent(a_in, t4)
        try_update(a_out, "cd")

    # === Phase 5: Multi-start exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-start (base n={n_best}) ===")
    t5_total = min(remaining * 0.25, 350.0)

    if t5_total > 60:
        inits = []
        labels = []

        inits.append(make_init_bowl(n_best, 2.0))
        labels.append("bowl2")
        inits.append(make_init_bowl(n_best, 3.0))
        labels.append("bowl3")
        inits.append(make_init_bowl(n_best, 4.0))
        labels.append("bowl4")
        inits.append(make_init_cosine(n_best, mode=1))
        labels.append("cos1")
        inits.append(make_init_cosine(n_best, mode=2))
        labels.append("cos2")
        inits.append(make_init_dual_peak(n_best, 0.15, 0.85, 0.12))
        labels.append("dualpeak")
        inits.append(make_init_w_shape(n_best))
        labels.append("wshape")

        # Perturb best sequence
        rng_ms = np.random.default_rng(123)
        for s_idx in [1, 2, 3]:
            pert = rng_ms.normal(0, 0.03, n_best)
            inits.append(norm_simplex(best_seq + pert))
            labels.append(f"pert{s_idx}")

        t_per = t5_total / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            remaining_now = time_budget - 30 - elapsed_so_far
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 10:
                break
            print(f"    Multi-start '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = lse_mirror_descent(init, t_avail, beta_start=10, beta_end=2000)
            try_update(a_out, f"ms_{lbl}")

    # === Phase 6: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length variation ===")
    t6_total = min(remaining * 0.15, 200.0)

    if t6_total > 60:
        for new_n in [n_best + 500, n_best - 500]:
            if new_n < 2000 or new_n > 8000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t6 = min(t6_total / 2, remaining - 10)
            if t6 < 15:
                continue
            print(f"  Trying n={new_n} (budget={t6:.0f}s)...")
            a_long = interpolate_sequence(best_seq, new_n)
            try_update(a_long, f"interp_n{new_n}")
            a_out, _ = lse_mirror_descent(a_long, t6, beta_start=10, beta_end=2000)
            try_update(a_out, f"lse_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_cd, _ = coordinate_descent(a_out, min(30.0, remaining - 5))
                try_update(a_cd, f"cd_n{new_n}")

    # === Phase 7: Final intensive polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Polish (n={n_best}) ===")
    t7 = min(remaining * 0.35, 250.0)

    if t7 > 20:
        # LP again on best
        t7a = min(t7 * 0.3, 80.0)
        if t7a > 10:
            a_in = norm_simplex(best_seq)
            a_out, _ = lp_optimize(a_in, t7a, max_iters=50)
            try_update(a_out, "lp_final")

        # LSE with very high temperature range
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.4, 120.0)
        if t7b > 10:
            a_in = norm_simplex(best_seq)
            a_out, _ = lse_mirror_descent(a_in, t7b, beta_start=100, beta_end=10000)
            try_update(a_out, "lse_final_high_beta")

        # CD
        remaining = time_budget - (time.monotonic() - t_global)
        t7c = min(remaining * 0.3, 60.0)
        if t7c > 5:
            a_in = norm_simplex(best_seq)
            a_out, _ = coordinate_descent(a_in, t7c)
            try_update(a_out, "cd_final")

        # Spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_in = norm_simplex(best_seq)
            a_out, _ = spectral_optimize(a_in, min(30.0, remaining - 5))
            try_update(a_out, "spec_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_init_bowl(n_fb, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
