def fast_conv(a, b):
    """Fast linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def smooth_adam_optimize(a0, time_limit, tau_decay=0.9998, lr=0.7, beta1=0.9, beta2=0.999,
                         restart_every=2000, verbose=True):
    """Optimize using log-sum-exp smoothed gradient with Adam."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    tau = max(ac.max() * 0.3, 1e-8)
    tau_min = max(ac.max() * 0.002, 1e-12)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        
        _, p = lse_value_and_probs(ac, tau)
        g = lse_gradient(a, p)
        
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        a = a_new
        
        if stall > restart_every:
            stall = 0
            # Restart with perturbation
            noise_scale = 0.05
            pert = rng.normal(0, noise_scale, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                print(f"      SmoothAdam restart ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")
        
        if verbose and t % 1000 == 0:
            print(f"      SmoothAdam t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, tau={tau:.1e}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=2000, verbose=True):

    best_m = ac.max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    # Initial constraint set: top K autocorrelation values
    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    lp_fail = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_autocorr(y)
        M_y = ac_y.max()

        if len(constraint_list) > max_K:
            # Prune: keep constraints that are most "active" (a* close to M_y)
            margins = np.array([M_y - ac_y[k] for k in constraint_list])
            keep_n = max_K
            keep = np.argsort(margins)[:keep_n]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        A_mat = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                A_mat[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail += 1
            if lp_fail > 10:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail += 1
            if lp_fail > 15:
                constraint_set = set(int(k) for k in constraint_list[:max(200, initial_K//2)])
            continue
        
        lp_fail = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check violations using FFT
        conv_ys = fast_conv(y, g_dir)
        violations = np.where(conv_ys > M_y + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_y
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        y = y_new
        
        if verbose and t % 20 == 0:
            print(f"      LP-FW t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, |S|={len(constraint_set)}")
    
    return best_a, best_m

def peak_shaving(a0, time_limit, verbose=True):
    """Coordinate descent that specifically targets the peak autocorrelation value."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()

        ac = fast_autocorr(a)

        # Find the best single move to reduce the peak
        # For the peak lag k*, moving mass from i to j changes:
        # (a*a)[k*] by: -a[i]*a[k*-i] - a[j]*a[k*-j] + cross terms
        # Simplified: if we move small δ from i to j:
        # Δ(a*a)[k*] ≈ δ * (a[k*-j] - a[k*-i]) + δ * (a[k*-j] + a[k*-j] - a[k*-i] - a[k*-i])
        # Hmm, let me just try random moves and keep improvements.

        k_int = peak_k

        if js <= je:

            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            
            if total_contrib > 0:
                # Source: position with highest contribution
                src = idx[np.argmax(contrib)]
                
                # Destination: position far from peak lag
                # Score: minimize impact on top few lags
                top_lags = np.argsort(ac)[::-1][:5]
                scores = np.zeros(n)
