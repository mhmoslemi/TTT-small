def make_recursive_flat_init(n, target_sum_sq=1.0):
    """
    Generate initial sequence based on the flat convolution condition.
    We want conv(a,a)[k] ≈ C for all k.
    This gives a recursive relation:
    a_0 = sqrt(C)
    a_k = (C - sum_{i=1}^{k-1} a_i * a_{k-i}) / (2*a_0) for k >= 1
    (approximately, for k < n)
    
    We'll compute this recursively and normalize.
    """
    C = target_sum_sq / (2*n - 1)  # target convolution value
    a = np.zeros(n)
    a[0] = np.sqrt(C)
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            s += a[i] * a[k-i]
        # c_k = 2*a_0*a_k + 2*sum_{i=1}^{k-1} a_i*a_{k-i} = C
        # (the factor of 2 accounts for the symmetry)
        # Wait, c_k = sum_{i=0}^{k} a_i * a_{k-i}
        # = a_0*a_k + a_1*a_{k-1} + ... + a_k*a_0
        # = 2*a_0*a_k + 2*sum_{i=1}^{k-1} a_i*a_{k-i} (if k >= 2)
        # For k=1: c_1 = 2*a_0*a_1
        # For k=0: c_0 = a_0^2
        if k == 1:
            a[k] = C / (2*a[0])
        else:
            # c_k = 2*a[0]*a[k] + sum_{i=1}^{k-1} 2*a[i]*a[k-i] = C
            # a[k] = (C - 2*sum_{i=1}^{k-1} a[i]*a[k-i]) / (2*a[0])
            s = 0.0
            for i in range(1, k):
                s += 2.0 * a[i] * a[k-i]
            a[k] = (C - s) / (2*a[0])
            if a[k] < 0:
                a[k] = 0.0  # clip negative values
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Recursive flat-convolution initialization (from the flat spectrum condition)
2. FISTA-style accelerated mirror descent with adaptive-temperature soft-max subgradient
3. FFT-based full gradient computation for efficiency at large n
4. Multi-length search (4000, 8000, 12000, 16000)
5. Spectral (Gerchberg-Saxton) refinement with non-negativity projection
6. Simulated-annealing restarts to escape local minima
7. Final direct peak-reduction polishing
Key insight: larger n helps; optimal shape is a mild ramp (not uniform);
adaptive temperature annealing outperforms fixed temperature.

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_flat = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_flat]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_power_law_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_mild_ramp(n, center_frac=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 + center_frac * x
    return norm_sum1(a)


def make_exponential_ramp(n, rate=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return norm_sum1(a)


def make_recursive_flat_init(n):
    """
    Generate initial sequence from the flat-convolution recursive condition.
    For conv(a,a)[k] = C for all k:
      a[0] = sqrt(C)
      a[k] = (C - 2*sum_{i=1}^{k-1} a[i]*a[k-i]) / (2*a[0])
    """
    C = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(C)
    for k in range(1, n):
        if k == 1:
            a[k] = C / (2.0 * a[0])
        else:
            # Sum of cross terms
            cross = np.dot(a[1:k], a[k-1:0:-1]) * 2.0
            a[k] = (C - cross) / (2.0 * a[0])
            if a[k] < 0:
                a[k] = 0.0
    # Reverse so it's increasing (by symmetry, gives same max convolution)
    a = a[::-1]
    return norm_sum1(a)


def make_curved_ramp(n, curvature=0.3):
    """Ramp with slight concavity: a[i] = (i+1)^p * (1 - c*(i/n)^2)"""
    x = np.arange(1, n + 1, dtype=np.float64)
    t = np.linspace(0, 1, n)
    a = x ** 0.5 * (1.0 - curvature * t ** 2)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def compute_grad_direct(a, w, n_top=80):
    """Compute gradient using top-peak direct summation. O(n * n_top)."""
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(n_top, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    grad = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    return grad, ac


def compute_grad_fft(a, w):
    """
    Compute full gradient g[j] = 2 * sum_k w[k] * a[k-j] using FFT.
    O(n log n).
    """
    n = len(a)
    nw = len(w)
    size = 1 << (n + nw - 1).bit_length()
    A = np.fft.rfft(a, size)
    Wf = np.fft.rfft(w[::-1], size)  # flip w for cross-correlation
    C = np.fft.irfft(A * Wf, size)
    # g[j] = sum_m a[m] * w[m+j]
    # C[k] = sum_m a[m] * w_flipped[k-m] = sum_m a[m] * w[nw-1-(k-m)] = sum_m a[m] * w[nw-1-k+m]
    # We want g[j] = sum_m a[m]*w[m+j], so nw-1-k+m = m+j → k = nw-1-j
    # g[j] = C[nw-1-j]
    g = np.zeros(n)
    for j in range(n):
        k = nw - 1 - j
        if 0 <= k < size:
            g[j] = C[k]
        else:
            # Handle wrap-around (shouldn't happen with proper size)
            g[j] = 0.0
    return 2.0 * g


def optimize_mirror(a0, time_limit, n_peaks=80, T0=8.0, T_min=0.8,
                    c0=4.0, decay_rate=0.9995, restart_every=4000,
                    use_fft_grad=False, verbose=False, rng_seed=42):
    """
    Accelerated mirror descent on the simplex with adaptive-temperature soft-max.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_temp = T0
    restart_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if (t & 1023) == 0:  # Check time less frequently
            if time.monotonic() - t0 > time_limit:
                break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Soft-max weights with current temperature
        ac_clipped = np.maximum(ac, 1e-30)
        log_ac = np.log(ac_clipped)
        log_max = log_ac.max()
        exp_vals = np.exp((log_ac - log_max) / cur_temp)
        w = exp_vals / exp_vals.sum()

        # Compute gradient
        if use_fft_grad and n >= 4000:
            g = compute_grad_fft(y, w)
        else:
            g, _ = compute_grad_direct(y, w, n_top=n_peaks)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Evaluate
        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Anneal temperature
        cur_temp = max(T_min, T0 * (decay_rate ** t))

        # Restart
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.2, 0.03 + 0.02 * min(restart_count, 8))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Vary parameters on restart
            if restart_count % 4 == 0:
                n_peaks = min(n_peaks + 20, 200)
            cur_temp = max(T_min, cur_temp * 1.5)  # Re-hang
            gamma = c0 * (1.0 + 0.1 * min(restart_count, 5))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      restart t={t} ({elapsed:.0f}s): J={J:.6f}, T={cur_temp:.2f}, peaks={n_peaks}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      t={t} ({elapsed:.0f}s): J={J:.6f}, T={cur_temp:.2f}, stall={stall_count}")

def spectral_flatten(a0, time_limit, alpha0=0.3, verbose=False, rng_seed=42):
    """Gerchberg-Saxton with non-negativity projection and occasional randomized restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        it += 1
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        alpha = alpha0 + (1.0 - alpha0) * min(it / 5000.0, 1.0)
        P_target = P_mean * (P / np.maximum(P_mean, 1e-30)) ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target / np.maximum(P, 1e-30))
        mag_new = np.clip(mag_new, 0.6, 1.5)
        A_new = np.abs(A) * mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Blend
            blend = 0.3 + 0.4 * (stall / 50.0)
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    # Random perturbation
                    pert = rng.normal(0, 0.05, n) * best_a
                    a = best_a + pert
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()
                    stall = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Spectral it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

            if best_dst == src:

                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and a[src] > 1e-10:
                dst = int(rng.integers(n))
                if dst != src:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_power_law_ramp(n_base, power=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}")

    # === Phase 1: Optimize at parent length with multiple inits ===
    print(f"\n=== Phase 1: Multi-init opt at n={n0} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 450.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_power_law_ramp(n0, power=0.35))
        labels.append("ramp035")
        inits.append(make_power_law_ramp(n0, power=0.45))
        labels.append("ramp045")
        inits.append(make_power_law_ramp(n0, power=0.55))
        labels.append("ramp055")
        inits.append(make_recursive_flat_init(n0))
        labels.append("recursive")
        inits.append(make_curved_ramp(n0, curvature=0.2))
        labels.append("curved")
        inits.append(make_exponential_ramp(n0, rate=2.0))
        labels.append("exp2")

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            use_fft = n0 >= 4000
            a_out, _ = optimize_mirror(init, t_avail, n_peaks=80, T0=8.0,
                                       T_min=0.5, c0=5.0, decay_rate=0.9998,
                                       restart_every=3000, use_fft_grad=use_fft,
                                       verbose=False, rng_seed=42 + idx * 7)
            try_update(a_out, f"grad_{lbl}")

    # === Phase 2: Try larger n ===
    print(f"\n=== Phase 2: Larger n search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    # Try 2x and 4x the current length
    larger_ns = []
    for target_n in [8000, 12000, 16000]:
        if target_n > n_best and target_n <= 20000:
            larger_ns.append(target_n)

    for new_n in larger_ns:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")

        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        # Also try recursive init at this length
        a_rec = make_recursive_flat_init(new_n)
        try_update(a_rec, f"recursive_n{new_n}")

        # Pick the better initialization
        m_interp = fast_autocorr(a_long).max()
        m_rec = fast_autocorr(a_rec).max()
        if m_rec < m_interp:
            a_start = a_rec
            lbl = "rec"
        else:
            a_start = a_long
            lbl = "interp"

        t_opt = min(remaining * 0.40, 300.0)
        if t_opt > 30:
            a_out, _ = optimize_mirror(a_start, t_opt, n_peaks=100, T0=10.0,
                                       T_min=0.5, c0=6.0, decay_rate=0.9997,
                                       restart_every=4000, use_fft_grad=True,
                                       verbose=False, rng_seed=100 + new_n)
            try_update(a_out, f"grad_n{new_n}_{lbl}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t_spec = min(remaining * 0.25, 120.0)
            if t_spec > 20:
                a_out, _ = spectral_flatten(norm_sum1(best_seq), t_spec, alpha0=0.3,
                                            verbose=False, rng_seed=555)
                try_update(a_out, f"spectral_n{len(best_seq)}")

    # === Phase 3: Spectral refinement on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Spectral refinement (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_flatten(a_in, t3, alpha0=0.3, verbose=False, rng_seed=777)
        try_update(a_out, f"spectral_best")

        # Gradient after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.10, 100.0)
        if t3b > 15:
            a_out, _ = optimize_mirror(norm_sum1(best_seq), t3b, n_peaks=120,
                                       T0=5.0, T_min=0.3, c0=5.0,
                                       decay_rate=0.9999, restart_every=5000,
                                       use_fft_grad=True, verbose=False, rng_seed=333)
            try_update(a_out, f"grad_post_spec")

    # === Phase 4: Peak polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak polishing (n={n_best}) ===")
    t4 = min(remaining * 0.12, 120.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_polish(a_in, t4, verbose=False, rng_seed=888)
        try_update(a_out, f"peakpolish")

    # === Phase 5: Aggressive final optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final aggressive opt (n={n_best}) ===")
    t5 = min(remaining * 0.35, 300.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = optimize_mirror(a_in, t5, n_peaks=150, T0=3.0,
                                   T_min=0.3, c0=4.0, decay_rate=0.99995,
                                   restart_every=6000, use_fft_grad=True,
                                   verbose=True, rng_seed=999)
        try_update(a_out, f"final_grad")

    # === Phase 6: Another round of spectral + gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 6: Final spectral+grad combo ===")
        t6a = min(remaining * 0.40, 100.0)
        t6b = min(remaining * 0.40, 100.0)
        if t6a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = spectral_flatten(a_in, t6a, alpha0=0.5, verbose=False, rng_seed=1111)
            try_update(a_out, f"spec_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = optimize_mirror(a_in, min(remaining * 0.5, 100.0),
                                       n_peaks=100, T0=2.0, T_min=0.2,
                                       c0=3.0, decay_rate=0.99995,
                                       restart_every=8000, use_fft_grad=True,
                                       verbose=False, rng_seed=2222)
            try_update(a_out, f"grad_very_final")

    # === Phase 7: Try slightly different lengths around best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 7: Length sweep around n={n_best} ===")
        for delta in [n_best // 4, -n_best // 4]:
            new_n = n_best + delta
            if 2000 <= new_n <= 20000:
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining < 40:
                    break
                a_cand = interpolate_sequence(best_seq, new_n)
                try_update(a_cand, f"interp_n{new_n}_7")
                t7 = min(remaining * 0.40, 80.0)
                if t7 > 15:
                    a_out, _ = optimize_mirror(a_cand, t7, n_peaks=100, T0=6.0,
                                               T_min=0.5, c0=5.0,
                                               decay_rate=0.9998, restart_every=3000,
                                               use_fft_grad=True, verbose=False,
                                               rng_seed=3333 + new_n)
                    try_update(a_out, f"grad_n{new_n}_7")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_power_law_ramp(n_fb, power=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
