3. Move a small amount of mass from high-contribution positions to low-contribution positions.
4. Repeat.

This is a form of "greedy" optimization that directly targets the objective.

In the frequency domain, if we let `A = FFT(a, L)` where L >= 2n-1, then `FFT(a*a, L) = A * conj(A) = |A|^2` (element-wise).

The time-domain signal `a*a` is `IFFT(|A|^2)`.

We want to minimize `max(a*a)`. This is equivalent to minimizing the peak of `IFFT(|A|^2)`.

Now, `|A|^2` is determined by `a` (which has support on [0, n-1]). The constraint is that `a >= 0` and `sum(a) = 1`.

In the frequency domain, the constraint `sum(a) = 1` becomes `A[0] = 1` (DC component).

The constraint `a >= 0` is hard to express in the frequency domain.

But we can work in the frequency domain:
- Start with some `|A|^2` spectrum.
- Compute `a = IFFT(A)` (choosing phases to make `a` real and non-negative... this is the hard part).
- The key insight is that if we keep the phases of A fixed and only adjust the magnitudes, we can iteratively improve.

This is essentially the **Gerchberg-Saxton** or **error reduction** algorithm, adapted for our problem:

1. Start with some initial `a` (e.g., the parent sequence).
2. Compute `A = FFT(a, L)`.
3. Modify `|A|` to reduce the peak of `IFFT(|A|^2)`:
   - Compute `r = IFFT(|A|^2)` (the current convolution).
   - Find the peak position k*.
   - We want to reduce `r[k*]`. Since `r[k*] = (1/L) sum_m |A[m]|^2 * exp(2πimk*/L)`, we can reduce `|A[m]|^2` for those m that constructively interfere at k*.
   - This is equivalent to reducing `|A[m]|` for m where `exp(2πimk*/L)` aligns with the peak direction.
4. Go back to time domain: `a_new = IFFT(A_new)`, where `A_new` has modified magnitudes but same phases.
5. Enforce `a_new >= 0` (set negative values to 0) and renormalize.
6. Repeat.

This is a form of **iterative spectral shaping**. It might be more effective than the previous "compress dynamic range" approach because it specifically targets the peaks.

    """Compute weighted subgradient of max(conv) using softmax-weighted top peaks."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()

    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        scale = beta * (top_vals - max_v) / max_v
        scale = np.clip(scale, -100.0, 0.0)
        w = np.exp(scale)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w /= w_sum

    grad = np.zeros(n)
    for t in range(k_max):
        wi = w[t]
        if wi < 1e-15:
            continue
        k = int(top_idx[t])
        # d/d a[j] (a*a)[k] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        j1s = max(0, k - n + 1)
        j1e = min(n - 1, k)
        if j1s <= j1e:
            idx1 = np.arange(j1s, j1e + 1)
            grad[idx1] += wi * a[k - idx1]
        j2s = max(0, k)
        j2e = min(n - 1, k + n - 1)
        if j2s <= j2e:
            idx2 = np.arange(j2s, j2e + 1)
            grad[idx2] += wi * a[idx2 - k]
    return grad


def ftrl_optimize(a0, time_limit, verbose=False):
    """FTRL mirror descent with adaptive temperature and step size."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    c_best = fast_conv(a)
    best_m = c_best.max()
    t0 = time.monotonic()

    c0 = 2.0
    T_init = max(best_m * 0.3, 1e-6)
    T_min = max(best_m * 0.005, 1e-8)
    top_k_init = 30
    top_k_min = 4

    x = a.copy()
    no_improve = 0
    max_no_improve = 5000

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        progress = min(1.0, t / 40000.0)
        T = T_init * (T_min / T_init) ** (progress ** 0.7)
        cur_top_k = max(top_k_min, int(top_k_init * (1 - 0.7 * progress) + top_k_min))
        cur_beta = 15.0 + 60.0 * (progress ** 1.5)

        # Nesterov
        beta_t = (t - 1.0) / max(t, 1)
        y = x + 0.6 * beta_t * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        c = fast_conv(y)
        g = compute_subgradient(y, c, top_k=cur_top_k, beta=cur_beta)

        # Adaptive step size with line search
        eta_base = c0 / (np.sqrt(t) + 1.0)

        # Try a few step sizes
        best_eta = eta_base
        best_trial_m = c.max()
        for mult in [0.5, 0.75, 1.0, 1.5]:
            eta_trial = eta_base * mult
            log_a = np.log(np.maximum(y, 1e-15)) - eta_trial * g
            max_log = log_a.max()
            a_trial = np.exp(log_a - max_log)
            a_trial = np.maximum(a_trial, 1e-30)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial /= st
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_eta = eta_trial

        a_new = np.exp(np.log(np.maximum(y, 1e-15)) - best_eta * g - np.log(np.maximum(y, 1e-15)).max())
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new /= sn

        x = a_new
        a = a_new

        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic spectral smoothing
        if t % 3000 == 0:
            a = spectral_smooth(a, strength=0.25)
            c_chk = fast_conv(a)
            if c_chk.max() < best_m:
                best_m = c_chk.max()
                best_a = a.copy()
            no_improve = min(no_improve, 1000)

        if verbose and t % 5000 == 0:
            print(f"      FTRL t={t}: obj={2*n*best_m:.6f}, T={T:.8f}, beta={cur_beta:.1f}")

    return best_a, best_m


def spectral_smooth(a, strength=0.3):
    """Spectral flattening: compress dynamic range of |A|."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a

    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.power(ratio, 1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new /= s

    return normalize_simplex((1 - strength * 0.5) * a + strength * 0.5 * a_new)


def spectral_peak_reduction(a, time_limit, verbose=False):
    """Target specific peaks in frequency domain for reduction."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    a = normalize_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    alpha = 0.1

    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        c = fast_conv(a)
        n_peaks = min(5, len(c))
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[peak_idx]
        peak_max = peak_vals[0]
        if peak_max < 1e-15:
            break

        # Compute alignment of each frequency bin with each peak
        # r[k] = (1/L) sum_m |A[m]|^2 * exp(2*pi*i*m*k/L)
        # To reduce r[k*], reduce |A[m]| where exp(2*pi*i*m*k*/L) has positive real part
        # Weighted by peak value
        freqs = np.arange(len(mag), dtype=np.float64)
        target = np.zeros(len(mag))
        for pi in range(n_peaks):
            k = int(peak_idx[pi])
            w = peak_vals[pi] / peak_max
            target += w * np.cos(2.0 * np.pi * freqs * k / fft_size)

        # Normalize target
        t_max = np.abs(target).max()
        if t_max < 1e-15:
            break
        target_norm = target / t_max

        # Reduce magnitude where target is positive (constructive interference at peaks)
        reduction = alpha * np.maximum(target_norm, 0)
        mag_new = mag * (1.0 - reduction)
        # Slightly boost where target is negative (destructive at peaks)
        mag_new += alpha * 0.3 * np.maximum(-target_norm, 0) * mag_mean_safe
        mag_mean_safe = mag.mean()
        mag_new = np.maximum(mag_new, 1e-30)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = normalize_simplex(0.5 * a + 0.5 * best_a)
            alpha *= 0.7
            continue
        a_new /= s

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(0.2, alpha * 1.02)
        else:
            alpha *= 0.85
            if alpha < 0.005:
                break
            # Blend back toward best
            a = normalize_simplex(0.5 * a + 0.5 * best_a)

        if verbose and it % 200 == 0:
            print(f"      SPR it={it}: obj={2*n*best_m:.6f}, alpha={alpha:.4f}")

    return best_a, best_m


def coord_descent_subgrad(a0, time_limit, verbose=False):
    """Coordinate descent guided by subgradient information."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]

        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)

            # Find positions contributing most to this peak
            j1s = max(0, ki - n + 1)
            j1e = min(n - 1, ki)
            j2s = max(0, ki)
            j2e = min(n - 1, ki + n - 1)

            # Contribution to peak from position j is a[j]*a[ki-j] + a[j]*a[j-ki]
            contrib = np.zeros(n)
            if j1s <= j1e:
                idx1 = np.arange(j1s, j1e + 1)
                contrib[idx1] += a[idx1] * a[ki - idx1]
            if j2s <= j2e:
                idx2 = np.arange(j2s, j2e + 1)
                contrib[idx2] += a[idx2] * a[idx2 - ki]

            # Find high-contribution source positions
            src_candidates = np.argsort(contrib)[::-1][:5]
            src_candidates = [s for s in src_candidates if a[s] > 1e-8]

            # Find low-contribution destination positions (across ALL peaks)
            total_contrib = np.zeros(n)
            for kp in top_peaks:
                kpi = int(kp)
                if j1s <= j1e:
                    idx1 = np.arange(max(0, kpi - n + 1), min(n - 1, kpi) + 1)
                    total_contrib[idx1] += a[idx1] * a[kpi - idx1]
                idx2 = np.arange(max(0, kpi), min(n - 1, kpi + n - 1) + 1)
                total_contrib[idx2] += a[idx2] * a[idx2 - kpi]

            dst_candidates = np.argsort(total_contrib)[:10]
            dst_candidates = [d for d in dst_candidates if d not in src_candidates]

            for src in src_candidates[:3]:
                if improved:
                    break
                for dst in dst_candidates[:5]:
                    if src == dst:
                        continue
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-10:
                            break
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = normalize_simplex(a_trial)
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random escape
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-8:
                amt = a[i1] * rng.uniform(0.02, 0.08)
                a[i1] -= amt
                a[i2] += amt
                a = normalize_simplex(a)
                M_new = fast_conv(a).max()
                if M_new < best_m + 0.01 * best_m:
                    pass  # Accept for exploration
                else:
                    a = best_a.copy()

    return best_a, best_m


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return normalize_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return normalize_simplex(a)


def make_init_asymmetric(n, phase=0.5):
    """Asymmetric initialization - key for beating the palindrome bound."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x + phase) ** 2
    a = np.maximum(a, 0.0)
    return normalize_simplex(a)


def make_init_chirp(n, rate=3.0):
    """Chirp-like initialization for asymmetric structure."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2 * (0.5 + 0.5 * np.cos(2 * np.pi * rate * x))
    a = np.maximum(a, 0.0)
    return normalize_simplex(a)


def make_init_two_gaussians(n, p1=0.3, p2=0.7, w=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - p1) ** 2) / (2 * w ** 2)) + 0.7 * np.exp(-((x - p2) ** 2) / (2 * w ** 2))
    a += 0.02
    return normalize_simplex(a)


def make_init_random_smooth(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        ph = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + ph)
    a = np.maximum(a, 0.001)
    return normalize_simplex(a)


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = safe_eval(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = safe_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: FTRL with diverse inits ===
    print(f"\n=== Phase 1: FTRL (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 500.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(normalize_simplex(parent))
            labels.append("parent")

        inits.append(make_init_asymmetric(n0, phase=0.4))
        labels.append("async04")

        inits.append(make_init_asymmetric(n0, phase=0.7))
        labels.append("async07")

        inits.append(make_init_bowl(n0, sharpness=2.5))
        labels.append("bowl25")

        inits.append(make_init_chirp(n0, rate=4.0))
        labels.append("chirp4")

        inits.append(make_init_two_gaussians(n0, 0.25, 0.7, 0.18))
        labels.append("2gauss")

        inits.append(make_init_random_smooth(n0, seed=7))
        labels.append("rand7")

        inits.append(make_init_random_smooth(n0, seed=13))
        labels.append("rand13")

        # More diverse
        x = np.linspace(0, 1, n0)
        a_exp = np.exp(-3.0 * x) + np.exp(-3.0 * (1 - x))
        inits.append(normalize_simplex(a_exp))
        labels.append("exp_edges")

        a_poly = (x * (1 - x)) ** 0.3 + 0.01
        inits.append(normalize_simplex(a_poly))
        labels.append("poly_3")

        print(f"  Running FTRL on {len(inits)} initializations...")
        t_per = t_phase1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            remaining_now = time_budget - 20 - elapsed_so_far
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now * 0.9)
            if t_avail < 10:
                break
            print(f"    FTRL '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = ftrl_optimize(init, t_avail, verbose=False)
            try_update(a_out, f"ftrl_{lbl}")

            # Quick spectral polish
            remaining_now = time_budget - 20 - (time.monotonic() - t_global)
            if remaining_now > 20:
                a_sp, _ = spectral_peak_reduction(a_out, min(25.0, remaining_now * 0.25), verbose=False)
                try_update(a_sp, f"spr_{lbl}")

    # === Phase 2: SPR on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: SPR on best (n={n_best}) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_in = normalize_simplex(best_seq)
        a_out, _ = spectral_peak_reduction(a_in, t2, verbose=True)
        try_update(a_out, "spr_best")

    # === Phase 3: FTRL on best with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: FTRL variant (n={n_best}) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 20:
        a_in = normalize_simplex(best_seq)
        rng3 = np.random.default_rng(123)
        perturbed = normalize_simplex(a_in + rng3.normal(0, 0.015, n_best))
        a_out, _ = ftrl_optimize(perturbed, t3, verbose=True)
        try_update(a_out, f"ftrl_var_n{n_best}")

        # SPR after
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sp, _ = spectral_peak_reduction(a_out, min(30.0, remaining * 0.3), verbose=False)
            try_update(a_sp, f"spr_var_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord Descent (n={n_best}) ===")
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 15:
        a_in = normalize_simplex(best_seq)
        a_out, _ = coord_descent_subgrad(a_in, t4, verbose=False)
        try_update(a_out, f"cd_n{n_best}")

    # === Phase 5: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length variation ===")

    for delta in [300, 700, 1200]:
        new_n = n_best + delta
        if new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_to_length(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t5 = min(remaining * 0.25, 120.0)
        if t5 > 15:
            a_out, _ = ftrl_optimize(a_long, t5, verbose=False)
            try_update(a_out, f"ftrl_n{new_n}")
            a_sp, _ = spectral_peak_reduction(a_out, min(20.0, (time_budget - 20) - (time.monotonic() - t_global)), verbose=False)
            try_update(a_sp, f"spr_n{new_n}")

    # Try shorter
    for delta in [400, 800]:
        new_n = n_best - delta
        if new_n < 2000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n} (shorter)...")
        a_short = interpolate_to_length(best_seq, new_n)
        try_update(a_short, f"interp_s_n{new_n}")

        t5s = min(remaining * 0.20, 100.0)
        if t5s > 15:
            a_out, _ = ftrl_optimize(a_short, t5s, verbose=False)
            try_update(a_out, f"ftrl_s_n{new_n}")

    # === Phase 6: Final intensive polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}) ===")
    t6 = min(remaining * 0.35, 250.0)
    if t6 > 20:
        a_in = normalize_simplex(best_seq)
        a_out, _ = ftrl_optimize(a_in, t6, verbose=True)
        try_update(a_out, f"ftrl_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sp, _ = spectral_peak_reduction(a_out, min(40.0, remaining * 0.5), verbose=False)
            try_update(a_sp, f"spr_final_n{n_best}")

    # === Phase 7: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 7: Final CD (n={n_best}) ===")
        t7 = min(remaining * 0.7, 100.0)
        if t7 > 10:
            a_in = normalize_simplex(best_seq)
            a_out, _ = coord_descent_subgrad(a_in, t7, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
