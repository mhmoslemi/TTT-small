def fast_autocorr(a):
    """Computes autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    a_arr = np.asarray(a, dtype=np.float64)
    # If length is 0, return an empty array
    if n == 0:
        return np.array([])
    # Pad with zeros to make even length for FFT
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a_arr, fft_size)
    B = np.fft.rfft(a_arr, fft_size)
    C = np.fft.irfft(A * B, fft_size)[:size] / n  # Shape (2n-1,)
    return C[(n-1)//2:]  # Keep the positive part (n to 2n-1)
