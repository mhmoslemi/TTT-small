"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using power-law sequences with optimized
exponent, followed by LP cutting-plane refinement and gradient-based polish.

Key insight: power-law sequences a[i] = (i+1)^p have autocorrelation peaks that
scale as O(1/n) in the "middle" lags (giving score ~2*(p+1)^2*p!^2/(2p+1)!) and
O(1/n^2) at endpoints (giving score ~2*(p+1)^2/n). By choosing p appropriately
for the sequence length, we achieve scores far below 1.5.
"""

import numpy as np
import time
import cvxpy as cp

_solver_cache = None

def get_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                return s
        except Exception:
            continue
    _solver_cache = cp.ECOS
    return _solver_cache


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def power_law_sequence(n, p):
    i = np.arange(n, dtype=np.float64) + 1.0
    return i ** p


def truncated_power_law(n, p, frac=0.5):
    i = np.arange(n, dtype=np.float64) + 1.0
    cap = int(n * frac)
    i_capped = np.minimum(i, cap)
    return i_capped ** p


def parametric_sweep(n_candidates, p_candidates, t0, time_budget):
    best_a = None
    best_score = float('inf')
    best_n = 0
    best_p = 0

    for n in n_candidates:
        for p in p_candidates:
            if time.monotonic() - t0 > time_budget:
                return best_a, best_score, best_n, best_p
            a = power_law_sequence(n, p)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a / a.sum()
                best_n = n
                best_p = p

    # Also try truncated power laws
    for n in n_candidates[:3]:
        for p in p_candidates[:15]:
            for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
                if time.monotonic() - t0 > time_budget:
                    return best_a, best_score, best_n, best_p
                a = truncated_power_law(n, p, frac)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a / a.sum()
                    best_n = n
                    best_p = p

    return best_a, best_score, best_n, best_p


def fine_tune_p(a0, n, t0, time_budget):
    """Golden-section search over p to find the optimal power-law exponent."""
    best_a = a0.copy()
    best_score = compute_score(a0)

    lo, hi = 1.0, 25.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for _ in range(30):
        if time.monotonic() - t0 > time_budget:
            break
        d = hi - lo
        c = hi - gr * d
        b = lo + gr * d

        ac = compute_score(power_law_sequence(n, c))
        ab = compute_score(power_law_sequence(n, b))

        if ac < ab:
            hi = b
            if ac < best_score:
                best_score = ac
                best_a = power_law_sequence(n, c) / power_law_sequence(n, c).sum()
        else:
            lo = c
            if ab < best_score:
                best_score = ab
                best_a = power_law_sequence(n, b) / power_law_sequence(n, b).sum()

    return best_a, best_score


def lp_cutting_plane(a0, time_limit, solver, initial_K=300, max_K=4000, add_per_iter=50, verbose=True):
    """LP-based cutting plane method to minimize max autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t_start = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    failed = 0

    for t in range(1, 2000000):
        if time.monotonic() - t_start > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[ki - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed += 1
            if failed > 5:
                constraint_list = sorted(constraint_set)[:initial_K]
                constraint_set = set(constraint_list)
                failed = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs

        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 1e-30)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            print(f"      LP iter={t} ({elapsed:.0f}s): J={J:.8f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t_start
    J = 2.0 * n * best_m
    if verbose:
        print(f"      LP done ({elapsed:.0f}s): J={J:.8f}")
    return best_a, best_m


def gradient_descent_optimize(a0, time_limit, top_k=50, beta=3.0, lr=2.0, verbose=True):
    """Mirror descent with soft-max gradient on top autocorrelation lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(123)

    stall = 0
    max_stall = 5000

    for t in range(1, 50000000):
        if time.monotonic() - t_start > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Restart with perturbation
            i = np.arange(n, dtype=np.float64) + 1.0
            p = float(rng.uniform(3, 12))
            a = norm_sum1(i ** p)
            stall = 0
            max_stall = int(max_stall * 1.3)
            if verbose:
                elapsed = time.monotonic() - t_start
                print(f"      GD restart ({elapsed:.0f}s): J={2*n*best_m:.8f}")
            continue

        # Compute soft-max gradient on top lags
        k_max = min(top_k, len(ac) - 1)
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * a[ki - idx]

        eta = lr / (np.sqrt(t) + 1.0)
        log_a = np.log(a + 1e-30) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a = a_new / a_new.sum()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            print(f"      GD t={t} ({elapsed:.0f}s): J={J:.8f}")

    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Random coordinate descent: move mass from high-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 300

    while time.monotonic() - t_start < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break

        # Try random moves
        top_lags = np.argsort(ac)[::-1][:20]
        improved = False

        for _ in range(50):
            if time.monotonic() - t_start > time_limit:
                break
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
            if src == dst or a[src] < 1e-12:
                continue

            frac = float(rng.uniform(0.01, 0.1))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1

    return best_a, best_m
