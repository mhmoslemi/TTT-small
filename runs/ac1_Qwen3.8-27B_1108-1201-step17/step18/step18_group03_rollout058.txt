    Compute gradient for mirror descent step.
    Uses soft-max weighting over top autocorrelation lags.
    Returns gradient vector of shape (n,).

        weights = weights / w_sum
    
    # Compute gradient: d(ac[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n, k_int + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * w * a[j_idx[valid]]
    
    return grad


def mirror_descent(a0, time_limit, lr_init=0.1, beta_init=3.0, top_k_init=30,
                   restart_every=5000, verbose=True, rng_seed=42):
    """
    Multiplicative update (mirror descent) with soft-max gradient.
    Nesterov acceleration in log-space.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    a_prev = a.copy()
    momentum = 0.85
    lr = lr_init
    beta = beta_init
    top_k = top_k_init
    stall = 0
    
    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov step in multiplicative space
        ratio = a / np.maximum(a_prev, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_nest = a * np.power(ratio, momentum)
        a_nest = norm_sum1(a_nest)
        
        # Compute autocorrelation and gradient
        ac = fast_autocorr(a_nest)
        grad = compute_mirror_grad(a_nest, ac, top_k, beta)
        
        # Adaptive learning rate
        eta = lr / (1.0 + 0.05 * np.sqrt(t))
        
        # Multiplicative update
        log_a = np.log(np.maximum(a_nest, 1e-30)) - eta * grad
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a_prev = a
        a = a_new
        
        # Check improvement
        ac_new = fast_autocorr(a)
        m_new = ac_new.max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive top_k: increase as we converge
        if t % 1000 == 0:
            top_k = min(top_k + 5, 150)
        
        # Restart when stuck
        if stall > restart_every:
            stall = 0
            pert_type = int(rng.integers(4))
            if pert_type == 0:
                # Gaussian perturbation
                x = np.linspace(0, 1, n)
                c = float(rng.uniform(0.3, 0.7))
                sig = float(rng.uniform(0.1, 0.4))
                pert = np.exp(-((x - c)**2) / (2 * sig**2))
            elif pert_type == 1:
                # Power law perturbation
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.5, 2.5))
                pert = (i_arr + 1.0) ** p
            elif pert_type == 2:
                # Sine perturbation
                x = np.linspace(0, 1, n)
                freq = int(rng.integers(1, 5))
                pert = np.sin(np.pi * freq * x + float(rng.uniform(0, np.pi)))**2
            else:
                # Small random perturbation
                pert = best_a * (1.0 + rng.normal(0, 0.05, n))
            
            pert = norm_sum1(pert)
            w = float(rng.uniform(0.1, 0.4))
            a = norm_sum1((1 - w) * best_a + w * pert)
            a_prev = a.copy()
            lr *= 0.85
            beta = max(beta - 0.5, 1.5)
            
            if verbose:
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, top_k={top_k}, beta={beta:.1f}")
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Mirror descent done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def batch_coord_descent(a0, time_limit, rng_seed=77, verbose=True):
    """
    Batch coordinate descent: for each top autocorrelation lag,
    try to reduce it by moving mass from high-contributing positions
    to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20000:
                break
        
        # Get top lags
        n_top = min(30, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k_pos in top_lags[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k_pos)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            if i_lo >= i_hi:
                continue
            
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            # Find top contributing pairs
            n_pairs = min(10, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]
            
            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-14:
                    continue
                
                # Compute impact score for each destination
                scores = np.zeros(n)
                for k2 in top_lags[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    if s2 < e2:
                        ii = np.arange(s2, e2)
                        jj = k2i - ii
                        scores[ii] += a[jj] * ac[k2i]
                        scores[jj] += a[ii] * ac[k2i]
                
                scores[src] = np.inf
                dst = int(np.argmin(scores))
                
                if dst == src or a[dst] > 0.8 * src_val:
                    continue
                
                # Try transfer fractions
                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            if stall % 500 == 0 and stall > 0:
                # Random perturbations
                n_moves = int(rng.integers(5, 20))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.005, 0.03))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord descent done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_flatten(a0, time_limit, lr_init=0.3, verbose=False):
    """
    Flatten the power spectrum to reduce autocorrelation peaks.
    Works by iteratively reducing spectral peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    lr = lr_init
    stall = 0
    beta_spec = 3.0
    
    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A)**2
        P_max = P.max()
        P_mean = P.mean()
        
        if P_max < 1e-30:
            break
        
        # Gradient: push energy from spectral peaks to valleys
        # In frequency domain: reduce A at peaks, increase at valleys
        P_dev = P - P_mean
        expo = beta_spec * P_dev / max(P_max, 1e-30)
        expo = np.clip(expo, -20, 20)
        w_spec = np.exp(expo)
        
        # Modify spectrum
        A_new = A * np.exp(-0.5 * lr / (1 + np.sqrt(t)) * w_spec / max(w_spec.mean(), 1e-10))
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm_sum1(a_new)
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                a = norm_sum1(best_a + rng.normal(0, 0.01, n) * best_a)
                stall = 0
                lr *= 0.8
                beta_spec = max(beta_spec - 0.5, 1.0)
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Spectral done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(x_new, x_old, f_norm)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss


def parametric_init(n, time_limit=10.0):
    """Quick parametric search for good initializations."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))
    
    # Gaussians
    for c in [0.3, 0.5, 0.7]:
        for sig in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(norm_sum1(np.exp(-((x - c)**2) / (2 * sig**2))))
    
    # Bimodal
    for sig in [0.1, 0.2, 0.3]:
        candidates.append(norm_sum1(
            np.exp(-((x - 0.25)**2) / (2*sig**2)) + np.exp(-((x - 0.75)**2) / (2*sig**2))))
    
    # Sine
    for f in [1, 2, 3]:
        candidates.append(norm_sum1(np.sin(np.pi * f * x)**2))
    
    # Exponential
    for r in [1, 2, 5]:
        candidates.append(norm_sum1(np.exp(-r * x)))
        candidates.append(norm_sum1(np.exp(r * x)))
    
    # Evaluate all
    best_cand = None
    best_val = float('inf')
    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        v = score_from_ac(c)
        if v < best_val:
            best_val = v
            best_cand = c.copy()
    
    return best_cand if best_cand is not None else np.ones(n) / n


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase optimization: parametric init -> mirror descent -> 
    coordinate descent -> spectral -> length exploration -> final polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        # Default: parametric search for good init
        n0 = 5000
        best_seq = parametric_init(n0, time_limit=5.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")
    
    # Also try uniform as baseline
    a_uni = np.ones(n0) / n0
    sc_uni = evaluate_safe(a_uni)
    if np.isfinite(sc_uni) and sc_uni < best_score:
        best_score = sc_uni
        best_seq = a_uni.copy()
        print(f"  Uniform improves: {sc_uni:.6f}")
    
    # === Phase 1: Mirror descent on parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_out, _ = mirror_descent(best_seq, t1, lr_init=0.08, beta_init=3.0,
                                  top_k_init=40, restart_every=4000, verbose=True)
        check_update(a_out, f"md_n{len(best_seq)}")
    
    # === Phase 2: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coord descent (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_out, _ = batch_coord_descent(best_seq, t2, rng_seed=123, verbose=True)
        check_update(a_out, f"coord_n{len(best_seq)}")
    
    # === Phase 3: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_out, _ = spectral_flatten(best_seq, t3, lr_init=0.3)
        check_update(a_out, f"spectral_n{len(best_seq)}")
    
    # === Phase 4: Second mirror descent pass (finer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Fine MD (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = mirror_descent(best_seq, t4, lr_init=0.03, beta_init=5.0,
                                  top_k_init=80, restart_every=3000, verbose=False)
        check_update(a_out, f"md_fine_n{len(best_seq)}")
    
    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (budget={remaining:.0f}s) ===")
    
    length_tries = []
    for delta in [-1000, -500, 500, 1000, 1500, 2000]:
        nn = n_best + delta
        if 1000 <= nn <= 8000 and nn != n_best:
            length_tries.append(nn)
    # Add some round numbers
    for rn in [3000, 4000, 5000, 6000, 7000]:
        if rn != n_best and 1000 <= rn <= 8000:
            length_tries.append(rn)
    length_tries = sorted(set(length_tries))
    
    for new_n in length_tries[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        
        a_interp = interpolate_sequence(best_seq, new_n)
        
        # Quick mirror descent on new length
        t5a = min(remaining * 0.35, 120.0)
        if t5a > 15:
            a_out, _ = mirror_descent(a_interp, t5a, lr_init=0.05, beta_init=3.0,
                                      top_k_init=50, restart_every=3000, verbose=False)
            check_update(a_out, f"md_n{new_n}")
        
        # Coord descent on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.3, 100.0)
            if t5b > 15:
                a_out, _ = batch_coord_descent(best_seq, t5b, rng_seed=new_n % 1000, verbose=False)
                check_update(a_out, f"coord_n{new_n}")
    
    # === Phase 6: Final polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.40, 200.0)
    if t6 > 20:
        a_out, _ = mirror_descent(best_seq, t6, lr_init=0.02, beta_init=6.0,
                                  top_k_init=100, restart_every=2000, verbose=False)
        check_update(a_out, f"final_md_n{n_best}")
    
    # Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7 = min(remaining * 0.5, 80.0)
        if t7 > 10:
            a_out, _ = batch_coord_descent(best_seq, t7, rng_seed=42, verbose=False)
            check_update(a_out, f"final_coord_n{len(best_seq)}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
