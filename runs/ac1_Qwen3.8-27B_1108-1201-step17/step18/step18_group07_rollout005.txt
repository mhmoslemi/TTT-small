"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based descent direction (solve max sum(g) s.t. conv(a,g) <= M, g>=0)
2. Backtracking line search for step size
3. Spectral domain smoothing as periodic restart
4. Multi-initialization and adaptations
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import math

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_sparse_conv_matrix(a, n):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_direction(a, M):
    """Solve LP to find descent direction g."""
    n = len(a)
    A_mat = make_sparse_conv_matrix(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_descent(a0, time_limit, verbose=False):
    """LP-based descent with backtracking."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    t_step = 0.5
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = fast_conv(a).max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            t_step = min(1.0, t_step * 1.3)
            stall = 0
        else:
            stall += 1
        
        if stall > 20:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            a = norm_simplex(a * (1 + rng.normal(0, 0.05, n)))
            t_step = 0.3
            stall = 0
            continue
        
        g = lp_direction(a, M)
        if g is None or g.sum() <= 0:
            break
        
        g = norm_simplex(g)
        
        # Backtracking line search
        accepted = False
        for alpha in [t_step, 0.7*t_step, 0.4*t_step, 0.2*t_step, 0.1*t_step]:
            a_trial = (1 - alpha) * a + alpha * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < M * (1 - 1e-10):
                a = a_trial
                t_step = min(1.0, t_step * 1.2)
                accepted = True
                break
            elif not accepted and alpha == 0.1 * t_step:
                # Accept even small improvement or same
                if M_trial <= M * (1 + 1e-8):
                    a = a_trial
                    accepted = True

            stall += 1

            elapsed = time.monotonic() - t0
            print(f"    LP iter={it} ({elapsed:.0f}s): M={M:.6e}, best={best_m:.6e}, t_step={t_step:.3f}")
    
    return best_a, best_m


def spectral_flatten(a, alpha=0.3):
    """Compress dynamic range of FFT amplitude."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - alpha)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return (1 - alpha) * a + alpha * (a_new / s)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_conv_matrix(a, n):
    """Build sparse matrix for conv(a, g) constraints."""
    # For each k in 0..2n-2, the non-zero cols are max(0,k-n+1)..min(n-1,k)
    # A[k, j] = a[k-j]
    
    # Use the structure: it's a banded matrix
    indptr = np.zeros(2*n, dtype=np.int32)
    indices = []
    values = []
    
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)  # exclusive
        length = j_end - j_start
        indices.extend(range(j_start, j_end))
        values.extend(a[k - j] for j in range(j_start, j_end))
    
    # Build indptr
    count = 0
    for k in range(2*n-1):
        count += min(k+1, n, 2*n-1-k)  # number of non-zeros in row k
        indptr[k+1] = count
    
    indices = np.array(indices, dtype=np.int32)
    values = np.array(values, dtype=np.float64)
    
    return csr_matrix((values, indices, indptr), shape=(2*n-1, n))
