def _spectral_flattening(a0, time_limit, verbose=False):

    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    fft_size = 1 << (2 * n - 2).bit_length()
    learning_rate = 0.3
    stall = 0
    beta_spec = 3.0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        P_dev = P - P_mean
        beta_spec += 0.002
        w_spec = np.exp(beta_spec * np.abs(P_dev) / max(P_mean, 1e-30))
        w_spec = w_spec / w_spec.sum()

        grad_freq = 2.0 * A * w_spec / max(P.max(), 1e-30)
        grad_time = np.fft.irfft(grad_freq, fft_size)[:n]

        lr = learning_rate / (1.0 + np.sqrt(t) * 0.1)
        a_new = a * np.exp(-lr * grad_time)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        M_new = _fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                a = _norm_sum1(best_a + rng.normal(0, 0.01, n) * best_a)
                stall = 0
                learning_rate *= 0.8
                beta_spec = max(2.0, beta_spec * 0.7)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Spectral done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def _interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            parent = _norm_sum1(p)
            p_score = _evaluate_safe(parent)

                best_seq = parent

        best_score = _evaluate_safe(best_seq)

    n0 = len(best_seq)

    def _check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric search ===
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 5:
        a_param, val_param = _parametric_init(n0, t1, best_seq, best_score)
        _check_and_update(a_param, "parametric")

    # === Phase 2: Main LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 30 and n_best <= 7000:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_frank_wolfe(a_in, t2, solver, initial_K=500, max_K=4000,

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Softmax gradient descent (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 15:

        a_out, _ = _softmax_grad_descend(a_in, t3, top_k=80, beta_start=3.0,
                                         beta_end=8.0, c0=4.0, restart_every=3000, verbose=True)
        _check_and_update(a_out, f"smd_n{n_best}")

    # === Phase 4: Peak reduction coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
