def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Evaluate the objective safely."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_arr = np.clip(a_arr, 0.0, 1000.0)
        s = a_arr.sum()
        if s < 0.01:
            return float('inf')
        conv = np.convolve(a_arr, a_arr)
        max_b = conv.max()
        return float(2.0 * len(a_arr) * max_b / (s * s))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize so sum = 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_linear_ramp(n):
    """Strong linear ramp: a[i] = i+1, normalized. Gives J ≈ 4/3 asymptotically."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return norm_sum1(a)


def make_power_ramp(n, p=1.0):
    """Power law ramp: a[i] = (i+1)^p, normalized."""
    a = np.arange(1, n + 1, dtype=np.float64) ** p
    return norm_sum1(a)


def make_exp_ramp(n, rate=3.0):
    """Exponential ramp: a[i] = exp(rate * i/(n-1)), normalized."""
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return norm_sum1(a)


def make_sigmoid(n, rate=5.0, center=0.7):
    """Sigmoid shape: sharp transition from low to high."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-rate * (x - center)))
    a = a - a.min() + 1e-10
    return norm_sum1(a)


def make_half_ramp(n):
    """Ramp over second half only."""
    a = np.zeros(n)
    half = n // 2
    a[half:] = np.arange(1, n - half + 1, dtype=np.float64)
    return norm_sum1(a)


def make_osc_ramp(n, p=1.0, freq=8.0, amp=0.1):
    """Oscillating ramp: a[i] = (i+1)^p * (1 + amp*cos(2*pi*freq*i/n))."""
    base = np.arange(1, n + 1, dtype=np.float64) ** p
    x = np.arange(n, dtype=np.float64)
    a = base * (1.0 + amp * np.cos(2.0 * np.pi * freq * x / n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return make_linear_ramp(n)
    return a / s


def compute_subgradient(a, ac, n_peaks=30, temperature=1.0):
    """
    Compute multi-peak weighted subgradient of max(conv).
    Uses softmax weighting over top peaks for a smoother gradient.
    """
    n = len(a)
    k_max = min(n_peaks, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    # Softmax weights
    top_vals = ac[top_idx]
    max_val = top_vals.max()
    exp_vals = np.exp((top_vals - max_val) / max(temperature, 1e-6))
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        ki = int(k)
        wi = w[idx_i]
        if wi < 1e-12:
            continue
        # d conv[k]/d a[j] = 2*a[k-j] for valid j (or a[j] if k=2j)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        contrib = 2.0 * a[pair_idx]
        # Where k = 2*j, the derivative is just a[j] not 2*a[j]
        diag_mask = (idx == pair_idx)
        contrib[diag_mask] = a[diag_mask]
        grad[idx] += wi * contrib
    return grad


def grad_optimize(a0, time_limit, n_peaks=40, temperature0=2.0, c0=3.0,
                  restart_every=3000, rng_seed=42):
    """Nesterov-accelerated mirror descent on the probability simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_temp = temperature0
    restart_count = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-18)
        sy = y.sum()
        y = y / sy
        
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, n_peaks=n_peaks, temperature=cur_temp)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_x = fast_autocorr(x).max()

            best_a = x.copy()

def spectral_flatten(a0, time_limit, iters=3000):
    """
    Gerchberg-Saxton style: flatten the power spectrum to reduce autocorrelation peaks.
    Key: work with zero-padded FFT to target LINEAR autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        
        if P_mean < 1e-30:
            break
        
        # Progressive flattening: blend between current spectrum and flat
        progress = it / max(1, iters)
        alpha = 0.3 + 0.7 * progress
        
        # Target: geometric mean of P and P_mean (pulls toward flat)
        P_target = P_mean * (P / np.maximum(P_mean, 1e-30)) ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)
        
        phase = np.angle(A)
        mag_new = np.sqrt(P_target / np.maximum(P, 1e-30))
        mag_new = np.clip(mag_new, 0.5, 2.0)
        A_new = A * mag_new
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped step
            blend = 0.2 + 0.3 * progress
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

    print(f"\n=== Phase 4: Aggressive gradient (n={n_best}) ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 20 and best_seq is not None:
        a_in = norm_sum1(best_seq)
        a_out, _ = grad_optimize(a_in, t4, n_peaks=80, temperature0=1.0,
                                  c0=5.0, restart_every=2000, rng_seed=7)
        try_update(a_out, f"grad_agg")
    
    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 5: Multi-length search ===")
    
    lengths_to_try = []
    for delta in [-500, 0, 500, 1000, 2000, 3000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 12000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))
    
    for new_n in lengths_to_try[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n) if best_seq is not None else make_linear_ramp(new_n)
        try_update(a_long, f"interp_n{new_n}")
        
        t5a = min(remaining * 0.25, 100.0)
        if t5a > 20:
            a_out, _ = grad_optimize(a_long, t5a, n_peaks=50, temperature0=2.0,
                                      c0=4.0, restart_every=3000, rng_seed=42)
            try_update(a_out, f"grad_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.15, 60.0)
            if t5b > 15:
                a_out, _ = spectral_flatten(a_long, t5b, iters=1000)
                try_update(a_out, f"spectral_n{new_n}")
    
    # === Phase 6: Final combined optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 6: Final optimization (n={n_best}) ===")
    
    if best_seq is not None:
        # Spectral + gradient combo
        t6a = min(remaining * 0.15, 80.0)
        t6b = min(remaining * 0.20, 100.0)
        
        if t6a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = spectral_flatten(a_in, t6a, iters=1500)
            try_update(a_out, f"spec_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if t6b > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = grad_optimize(a_in, min(t6b, remaining), n_peaks=100,
                                      temperature0=0.8, c0=3.0, restart_every=6000,
                                      rng_seed=123)
            try_update(a_out, f"grad_final")
    
    # === Phase 7: Peak reduction final ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15 and best_seq is not None:
        t7 = min(remaining * 0.4, 80.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction(a_in, t7, rng_seed=777)
        try_update(a_out, f"peakred_final")
    
    # === Phase 8: One more gradient pass with fresh params ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        t8 = min(remaining * 0.5, 100.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = grad_optimize(a_in, t8, n_peaks=120, temperature0=1.0,
                                  c0=4.0, restart_every=4000, rng_seed=555)
        try_update(a_out, f"grad_very_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = make_linear_ramp(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
