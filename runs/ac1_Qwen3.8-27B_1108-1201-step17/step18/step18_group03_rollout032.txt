import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def interpolate_to_n(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return norm_sum1(f_interp)

def lp_frank_wolfe(a0, time_limit, solver, initial_K=500, max_K=5000, verbose=True):
    """Improved LP-Frank-Wolfe with dual-guided constraint management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Start with top-K constraints
    ac = fast_autocorr(a)
    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    y = a.copy()  # Nesterov y
    x = a.copy()  # Nesterov x (previous y)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current autocorrelation
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()
        
        # Build constraint matrix
        constraint_list = sorted(constraint_set)
        K = len(constraint_list)
        
        # Manage constraint set size
        if K > max_K:
            # Keep constraints with smallest slack (most binding)
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
            K = len(constraint_list)
        
        # Build A matrix: A[k,j] = y[k-j] if valid, else 0
        A_mat = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                A_mat[jj, idx] = y[k - idx]
        
        # Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce constraints and retry
            constraint_list = constraint_list[:min(initial_K, len(constraint_list))]
            constraint_set = set(constraint_list)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Add violated constraints
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        for k in violations:
            constraint_set.add(int(k))
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 1e-15)
        y_new = y_new / y_new.sum()
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 1e-15)
            x_new = x_new / x_new.sum()
        
        y = y_new
        x = x_new
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    J = 2.0 * n * best_m
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  LP-FW done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def softmax_mirror_descent(a0, time_limit, beta_init=10.0, beta_growth=1.001, verbose=True):

            y = x.copy()

def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize array to be nonnegative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_to_n(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return norm_sum1(f_interp)

def lp_frank_wolfe(a0, time_limit, solver, initial_K, max_K, verbose=True):
    """
    LP-Frank-Wolfe with aggressive constraint management.
    Minimizes max(conv(a,a)) over the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Initialize constraint set with top-K lags
    ac = fast_autocorr(a)
    init_idx = np.argsort(ac)[::-1][:min(initial_K, size)]
    constraint_set = set(int(k) for k in init_idx)

    y = a.copy()
    x = a.copy()
    failed_solves = 0
    iter_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Check improvement
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        K = len(constraint_list)

        if K > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
            K = len(constraint_set)

        if K == 0:
            break

        # Build constraint matrix A where A[j, i] = y[constraint_list[j] - i]
        A_mat = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                A_mat[jj, idx] = y[k - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 3:
                constraint_list = constraint_list[:min(initial_K // 2, len(constraint_list))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints to the set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        for k in violations:
            constraint_set.add(int(k))

        # Nesterov-accelerated FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 1e-15)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 1e-15)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def softmax_mirror_descent(a0, time_limit, beta_init, beta_growth, verbose=True):
    """
    Smooth optimization using soft-max approximation of max(conv).
    Uses accelerated mirror descent on the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    beta = beta_init
    lr = 1.0
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    rng = np.random.default_rng(123)

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            mom = (t - 1.5) / (t + 1.5)
            y = x + mom * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Compute autocorrelation
        ac = fast_autocorr(y)
        ac_max = ac.max()
        if ac_max < 1e-30:
            break

        # Soft-max weights
        beta_ac = beta * ac / ac_max
        beta_ac = np.clip(beta_ac, -50.0, 50.0)
        w = np.exp(beta_ac - beta_ac.max())
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / w_sum

        # Compute gradient using top-k lags for efficiency
        n_top = min(300, len(ac))
        top_idx = np.argsort(w)[::-1][:n_top]
        grad = np.zeros(n)
        for k_int in top_idx:
            k_int = int(k_int)
            wk = w[k_int]
            if wk < 1e-12:
                continue
            j_lo = max(0, k_int - n + 1)
            j_hi = min(n - 1, k_int)
            if j_lo <= j_hi:
                js = np.arange(j_lo, j_hi + 1)
                grad[js] += 2.0 * wk * y[k_int - js]

        # Mirror descent step (multiplicative weights)
        lr_t = lr / (1.0 + 0.3 * np.sqrt(t))
        log_a = np.log(y + 1e-30) - lr_t * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stuck
        if stall > 3000:
            stall = 0
            noise = rng.normal(0, 0.02, n) * best_a
            x = norm_sum1(best_a + noise)
            x_prev = x.copy()
            beta = max(beta_init, beta * 0.5)

        # Grow beta
        beta *= beta_growth
        if beta > 5000.0:
            beta = 5000.0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SoftMax iter={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.0f}, stall={stall}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    SoftMax done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

            b1 = min(b1, n - blen)
            b2 = min(b2, n - blen)
            a_pert[b1:b1 + blen], a_pert[b2:b2 + blen] = a_pert[b2:b2 + blen].copy(), a_pert[b1:b1 + blen].copy()
            a_pert = norm_sum1(a_pert)

        # Short LP-FW on perturbed
        t_sub = min(remaining * 0.4, 60.0)
        if t_sub > 10:
            a_out, m_out = lp_frank_wolfe(a_pert, t_sub, solver,
                                           initial_K=300, max_K=2000, verbose=False)
            if m_out < best_m - 1e-10:
                best_a = a_out
                best_m = m_out
                if verbose:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    Perturb[{i}] ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def parametric_init(n, t_global, time_limit):
    """Try various parametric initializations and return the best."""
    best_a = None
    best_val = float('inf')
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr + 1.0) ** p)

    # Beta-like
    for alpha in [0.5, 1.0, 1.5]:
        for beta_p in [0.5, 1.0, 1.5]:
            candidates.append(np.maximum(i_arr + 0.5, 0.1) ** alpha *
                              np.maximum(n - i_arr + 0.5, 0.1) ** beta_p)

    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    # Bimodal
    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            for sigma in [0.08, 0.12, 0.18]:
                candidates.append(
                    np.exp(-((x - c1) ** 2) / (2 * sigma * sigma)) +
                    np.exp(-((x - c2) ** 2) / (2 * sigma * sigma))
                )

    # Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [10, 20, 50]:
            candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))

    # Sinusoidal
    for freq in [1, 2, 3]:
        for ph in [0.0, np.pi / 4, np.pi / 2]:
            candidates.append(np.maximum(np.sin(np.pi * freq * x + ph) ** 2, 0.01))

    # Exponential
    for rate in [1, 2, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1)))

    # Linear tilt
    for slope in [-0.5, -0.3, 0.3, 0.5]:
        candidates.append(1.0 + slope * (x - 0.5))

    for cand in candidates:
        if time.monotonic() - t_global > time_limit:
            break
        cand = np.maximum(np.asarray(cand, dtype=np.float64), 0.01)
        s = cand.sum()
        if s < 1e-10:
            continue
        cand_norm = cand / s
        ac = fast_autocorr(cand_norm)
        val = ac.max()
        if val < best_val:
            best_val = val
            best_a = cand_norm.copy()

    return best_a


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Uses multi-phase strategy: template -> LP-FW -> polish -> explore.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_state = [None, float('inf')]  # [sequence, score]
    rng = np.random.default_rng(2024)

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_state[0] = parent
                best_state[1] = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # Initialize if no parent
    if best_state[0] is None:
        n_base = 5000
        a_init = parametric_init(n_base, t_global, 10.0)
        if a_init is not None:
            best_state[0] = a_init
        else:
            i_arr = np.arange(n_base, dtype=np.float64)
            best_state[0] = norm_sum1((i_arr + 1.0) ** 0.7)
        best_state[1] = evaluate_safe(best_state[0])
        print(f"Initialized: n={len(best_state[0])}, score={best_state[1]:.6f}")

    # === Phase 1: Small-n LP-FW for shape template ===
    print(f"\n=== Phase 1: Small-n template ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 45.0)
    n_small = 80

    if t1 > 15:
        a_small_init = norm_sum1(best_state[0])
        if len(a_small_init) != n_small:
            a_small_init = interpolate_to_n(a_small_init, n_small)
        # LP-FW with ALL constraints (small enough for n=80)
        a_small, m_small = lp_frank_wolfe(a_small_init, t1, solver,
                                           initial_K=2 * n_small - 1,
                                           max_K=2 * n_small - 1, verbose=False)
        # Also try a second small-n with linear init
        a_lin = np.linspace(0.5, 1.5, n_small)
        a_lin = norm_sum1(a_lin)
        a_lin, m_lin = lp_frank_wolfe(a_lin, t1 * 0.5, solver,
                                       initial_K=2 * n_small - 1,
                                       max_K=2 * n_small - 1, verbose=False)
        # Use the better one as template
        if m_lin < m_small:
            a_template = a_lin
        else:
            a_template = a_small

        # Interpolate to current best length
        n_current = len(best_state[0])
        a_interp = interpolate_to_n(a_template, n_current)
        check_and_update(best_state, a_interp, "template", t_global)
        # Also try longer
        n_target = max(n_current, 5500)
        a_interp_long = interpolate_to_n(a_template, n_target)
        check_and_update(best_state, a_interp_long, "template_long", t_global)

    # === Phase 2: Main LP-FW optimization ===
    print(f"\n=== Phase 2: Main LP-FW (n={len(best_state[0])}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_main = len(best_state[0])

    if n_main > 8000:
        n_main = 7000
        best_state[0] = interpolate_to_n(best_state[0], n_main)
        best_state[1] = evaluate_safe(best_state[0])
        print(f"  Reduced to n={n_main}, score={best_state[1]:.6f}")

    t2 = remaining * 0.55
    if t2 > 60:
        a_in = norm_sum1(best_state[0])
        a_out, _ = lp_frank_wolfe(a_in, t2, solver,
                                   initial_K=600, max_K=5000, verbose=True)
        check_and_update(best_state, a_out, f"lp_main_n{len(a_in)}", t_global)

    # === Phase 3: Perturbation escape ===
    print(f"\n=== Phase 3: Perturbation escape ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.12
    if t3 > 30:
        a_in = norm_sum1(best_state[0])
        a_out, _ = perturb_and_optimize(a_in, t3, solver,
                                         n_perturb=10, rng=rng, verbose=True)
        check_and_update(best_state, a_out, "perturb", t_global)

    # === Phase 4: Soft-max polishing ===
    print(f"\n=== Phase 4: Soft-max polish (n={len(best_state[0])}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    if t4 > 20:
        a_in = norm_sum1(best_state[0])
        a_out, _ = softmax_mirror_descent(a_in, t4, beta_init=30.0,
                                           beta_growth=1.0008, verbose=True)
        check_and_update(best_state, a_out, f"softmax_n{len(a_in)}", t_global)

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_state[0])

    length_trials = []
    for delta in [500, -500, 1000, -1000, 1500, 2000]:
        new_n = n_best + delta
        if 3000 <= new_n <= 9000:
            length_trials.append(new_n)
    # Add some round numbers
    for rn in [4000, 5000, 5500, 6000, 6500, 7000]:
        if rn != n_best and 3000 <= rn <= 9000:
            length_trials.append(rn)
    length_trials = sorted(set(length_trials))

    for new_n in length_trials[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_new = interpolate_to_n(best_state[0], new_n)
        # Quick LP-FW
        t5a = min(remaining * 0.35, 100.0)
        if t5a > 20:
            a_out, _ = lp_frank_wolfe(a_new, t5a, solver,
                                       initial_K=400, max_K=3000, verbose=False)
            check_and_update(best_state, a_out, f"lp_n{new_n}", t_global)

    # === Phase 6: Final LP-FW polish on global best ===
    print(f"\n=== Phase 6: Final LP-FW polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_state[0])
    if remaining > 40 and n_final <= 8000:
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 20:
            a_in = norm_sum1(best_state[0])
            a_out, _ = lp_frank_wolfe(a_in, t6, solver,
                                       initial_K=800, max_K=5000, verbose=False)
            check_and_update(best_state, a_out, f"lp_final_n{n_final}", t_global)

    # === Phase 7: Final short soft-max ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7 = min(remaining * 0.6, 80.0)
        if t7 > 10:
            a_in = norm_sum1(best_state[0])
            a_out, _ = softmax_mirror_descent(a_in, t7, beta_init=50.0,
                                               beta_growth=1.001, verbose=False)
            check_and_update(best_state, a_out, "softmax_final", t_global)

    # === Final validation ===
    if best_state[0] is not None:
        fs = evaluate_safe(best_state[0])
        if np.isfinite(fs):
            best_state[1] = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_state[1]:.6f}, n={len(best_state[0])}, time={elapsed:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_state[0] = norm_sum1(np.sin(np.pi * x) ** 2)
        best_state[1] = evaluate_safe(best_state[0])
        print(f"Fallback: score={best_state[1]:.6f}")

    return [float(v) for v in best_state[0]]
