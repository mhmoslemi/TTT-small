def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def build_lp_matrix(binding_idx, a, n):
    """Build constraint matrix T where T[i,j] = a[k_i - j] for valid j."""
    K = len(binding_idx)
    T = np.zeros((K, n))
    for ii, k in enumerate(binding_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            T[ii, j] = a[ki - j]
    return T

            K = min(K + 20, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 50, top_K_max)
            continue
        g_dir = b_val / b_sum

        # Backtracking line search with Armijo
        base_obj = M_curr
        alpha = min(1.0, min_step + 0.5 / (iter_count + 1))
        best_alpha = alpha
        best_trial_m = base_obj + 1  # Start optimistic

        for trial_alpha in [alpha, alpha * 0.7, alpha * 0.5, alpha * 0.3,
                            alpha * 0.2, 0.1, 0.05]:
            a_trial = (1.0 - trial_alpha) * a + trial_alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = trial_alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        iter_count += 1

        # Adaptive K increase
        if iter_count % 5 == 0:
            K = min(K + 40, top_K_max)

        if stall > max_stall:
            # Perturb to escape
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.01, n) * a
            a_p = np.maximum(a + pert, 0)
            a_p = a_p / a_p.sum()
            m_p = fast_autocorr(a_p).max()
            if m_p < best_m:
                a = a_p
            else:
                a = best_a.copy()
            stall = 0

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=True, rng_seed=100):
    """Smoothed projected gradient with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = 0.5
    stall = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 800.0))
        # Temperature: start moderate, increase to capture more peaks
        tau = 5.0 + 40.0 * progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        # Compute gradient: g[j] = 2 * sum_k p[k] * a[k-j] (valid range)
        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = 2.0 * (a * S + cross)

        # Project onto tangent space
        g_tangent = g - g.mean()

        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.03, 2.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.8
            if stall > 250:
                pert = rng.normal(0, 0.004, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a_p = a_p / a_p.sum()
                m_p = fast_autocorr(a_p).max()
                if m_p < best_m:
                    a = a_p
                    best_m = m_p
                    best_a = a.copy()
                    stall = 0
                    eta = 0.3
                else:
                    a = a_new

        a = a_new if m_new <= best_m + 1e-10 else best_a
        total_iters += 1

        if verbose and total_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Smooth iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def alternating_lp_smooth(a0, time_limit, solver, verbose=True, rng_seed=300):
    """Alternate between LP-FW steps and smooth gradient steps."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    K = min(500, 2 * n - 1)
    K_max = min(2 * n - 1, 6000)
    eta = 0.3
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        phase_start = time.monotonic()

        # LP step (every other iteration, or when stalled)
        if iter_count % 2 == 0 or stall > 5:
            ac = fast_autocorr(a)
            M_curr = ac.max()
            K_sel = min(K, 2 * n - 1)
            binding_idx = np.argsort(ac)[::-1][:K_sel]
            T = build_lp_matrix(binding_idx, a, n)

            b_var = cp.Variable(n, nonneg=True)
            constraints = [T @ b_var <= M_curr]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                K = min(K + 30, K_max)
                iter_count += 1
                continue

            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                b_sum = b_val.sum()
                if b_sum > 1e-15:
                    g_dir = b_val / b_sum
                    alpha = min(0.5, 1.0 / (iter_count + 2))
                    a_new = (1.0 - alpha) * a + alpha * g_dir
                    a_new = np.maximum(a_new, 0.0)
                    s_t = a_new.sum()
                    if s_t > 1e-15:
                        a_new = a_new / s_t
                        m_new = fast_autocorr(a_new).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a_new.copy()
                            stall = 0
                        a = a_new
            K = min(K + 20, K_max)
        else:
            # Smooth step
            ac = fast_autocorr(a)
            ac_max = ac.max()
            tau = 10.0
            ac_shifted = tau * (ac - ac_max)
            exp_ac = np.exp(ac_shifted)
            p = exp_ac / exp_ac.sum()

            p_cum = np.concatenate([[0.0], np.cumsum(p)])
            k_lo = np.arange(n)
            k_hi = np.minimum(k_lo + n, 2 * n - 1)
            S = p_cum[k_hi] - p_cum[k_lo]
            cross = sp_correlate(p, a, mode='valid')
            g = 2.0 * (a * S + cross)
            g_tangent = g - g.mean()

            a_new = project_simplex(a - eta * g_tangent)
            m_new = fast_autocorr(a_new).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()
                stall = 0
            a = a_new
            eta = max(eta * 0.99, 0.05)

        stall += 1
        iter_count += 1

        if stall > 40:
            pert = rng.normal(0, 0.005, n) * best_a
            a_p = np.maximum(best_a + pert, 0.0)
            a_p = a_p / a_p.sum()
            m_p = fast_autocorr(a_p).max()
            if m_p < best_m:
                a = a_p
                best_m = m_p
                best_a = a.copy()
            else:
                a = best_a.copy()
            stall = 0
            K = min(K + 50, K_max)

        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AltLP-Smooth iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Aggressive coordinate descent with pair-swap."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 80
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s_idx = max(0, ki - n + 1)
            e_idx = min(n - 1, ki)
            idx = np.arange(s_idx, e_idx + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try moving mass from src to a low-contribution position
            for dst_candidate in rng.choice(n, size=min(5, n), replace=False):
                dst = int(dst_candidate)
                if dst == src:
                    continue
                # Check if dst is in peak region
                in_peak = False
                for k2 in top_k[:8]:
                    s2 = max(0, int(k2) - n + 1)
                    e2 = min(n - 1, int(k2))
                    if s2 <= dst <= e2:
                        in_peak = True
                        break
                if in_peak:
                    continue

                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with alternating LP/Smooth and length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = make_u_shape(n_base, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Quick AMD multi-start ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Quick AMD (n={n0}) ===")
    t_phase1 = min(remaining * 0.12, 180.0)

    if t_phase1 > 40:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3.0")
        inits.append(make_w_shape(n0, sharpness=1.5))
        labels.append("w1.5")
        inits.append(make_cosine(n0))
        labels.append("cos")
        inits.append(spectral_init(n0, n_modes=30, seed=42))
        labels.append("spec")

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                     restart_every=3000, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP-FW main (aggressive K) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.18, 320.0)
    if t2 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2500)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(2 * n_best - 1, 8000)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all, min_step=0.1)
        try_update(a_out, "lp_main")

    # === Phase 3: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smoothed gradient (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t3, verbose=True, rng_seed=100)
        try_update(a_out, "smooth1")

    # === Phase 4: Alternating LP/Smooth ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Alternating LP/Smooth (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = alternating_lp_smooth(a_in, t4, solver, verbose=True, rng_seed=300)
        try_update(a_out, "alt1")

    # === Phase 5: LP-FW round 2 with higher K ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.12, 220.0)
    if t5 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2500)
        top_K_init = min(800, 2 * n_best - 1)
        top_K_max = min(2 * n_best - 1, 10000)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all, min_step=0.05)
        try_update(a_out, "lp_r2")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate (n={n_best}) ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [1000, -1000, 2000, -2000, 1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick AMD on new length
        t7a = min(remaining * 0.3, 150.0)
        if t7a > 15:
            a_out, _ = amd_optimize(a_long, t7a, top_k=60, beta=6.0, c0=4.0,
                                     restart_every=4000, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        # LP on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 6000:
            t7b = min(remaining * 0.35, 120.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                use_all = (new_n <= 2500)
                top_K = min(5000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                           top_K_init=min(400, 2 * new_n - 1),
                                           top_K_max=top_K, use_all=use_all,
                                           min_step=0.08)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final alternating LP/Smooth on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 40 and n_best <= 8000:
        t8 = min(remaining * 0.3, 180.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = alternating_lp_smooth(a_in, t8, solver, verbose=True, rng_seed=500)
            try_update(a_out, "alt_final")

    # === Phase 9: Final smooth with high temperature ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t9 = min(remaining * 0.25, 120.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = smooth_gradient_optimize(a_in, t9, verbose=False, rng_seed=700)
            try_update(a_out, "smooth_final")

    # === Phase 10: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t10 = min(remaining * 0.3, 80.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t10, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # === Phase 11: One more LP pass with max K ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30 and n_best <= 6000:
        print(f"\n=== Phase 11: Final LP (n={n_best}) ===")
        t11 = min(remaining * 0.5, 100.0)
        if t11 > 15:
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 2500)
            top_K = min(2 * n_best - 1, 12000)
            a_out, _ = lp_frank_wolfe(a_in, t11, solver, verbose=False,
                                       top_K_init=min(1000, 2 * n_best - 1),
                                       top_K_max=top_K, use_all=use_all,
                                       min_step=0.03)
            try_update(a_out, "lp_last")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
