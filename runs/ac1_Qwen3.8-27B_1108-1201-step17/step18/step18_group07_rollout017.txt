"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Iterative Flat Spectrum Design (IFTA) - Fourier-domain amplitude shaping
2. Multiple target shapes (flat, tapered, variable softness)
3. FTRL mirror descent for local refinement
4. Multi-length exploration and multi-init strategy
5. Hybrid spectral + gradient polish

import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_target_magnitude(n, N, taper=0.0, softness=0.0):
    """Compute target |A(ω)| for desired convolution shape.
    
    taper: 0 = flat, >0 = edges higher (quadratic taper)
    softness: 0 = hard zeros, >0 = softened zeros
    """
    freqs = np.arange(N // 2 + 1)
    w = 2.0 * np.pi * freqs / N
    half_len = (2 * n - 1) / 2.0
    
    with np.errstate(divide='ignore', invalid='ignore'):
        dirichlet = np.abs(np.sin(w * half_len) / np.sin(w / 2))
    dirichlet[0] = 2.0 * n - 1.0
    
    if softness > 0:
        dirichlet = np.maximum(dirichlet, softness * (2.0 * n - 1.0))
    
    # Apply taper to the target convolution shape
    if taper > 0:
        # Target conv[k] = M * (1 + taper * (2k/(2n-2) - 1)^2)
        # Its FFT is M * (Dirichlet + taper * (Dirichlet * fourth_order_moment))
        # Approximation: just scale the target by a factor that's >1 at low freq
        # Actually, let's just modify the target magnitude directly
        # The ideal flat target has |A|^2 proportional to Dirichlet kernel
        # A tapered target would have a slightly different shape
        # For simplicity, we'll multiply the target by a smooth function
        pass  # Keep flat for now, taper is experimental
    
    target_mag = np.sqrt(dirichlet / (2.0 * n - 1.0))
    target_mag = np.maximum(target_mag, 1e-20)
    target_mag[0] = 1.0
    return target_mag


def flat_spectrum_optimize(a0, time_limit, alpha_start=0.3, alpha_end=0.05,
                           taper=0.0, softness=0.0, verbose=False):
    """Iterative Flat Spectrum Design algorithm.
    
    Drives |A(ω)| toward target that makes conv(a,a) flat.
    Uses geometric interpolation with annealed alpha.
    """
    n = len(a0)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    target_mag = compute_target_magnitude(n, N, taper=taper, softness=softness)
    
    a = norm_simplex(a0)
    c = fast_conv(a)
    best_m = c.max()
    best_a = a.copy()
    t_start = time.monotonic()
    
    total_iters = 200000
    for it in range(total_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Anneal alpha
        progress = min(1.0, it / (total_iters * 0.8))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Compute DFT
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-30)
        phase = np.angle(A)
        
        # Geometric interpolation: |A_new| = |A|^(1-alpha) * target^alpha
        # Use log-space for numerical stability
        log_mag = np.log(mag)
        log_target = np.log(np.maximum(target_mag, 1e-30))
        log_mag_new = (1.0 - alpha) * log_mag + alpha * log_target
        mag_new = np.exp(log_mag_new)
        mag_new[0] = 1.0  # Ensure sum(a) = 1
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project: non-negative, sum=1
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate
        c = fast_conv(a_new)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    FlatSpec it={it} ({elapsed:.0f}s): max_conv={M:.8f}, score={2*n*M:.6f}, alpha={alpha:.4f}")
    
    return best_a, best_m


def flat_spectrum_blend(a0, time_limit, alpha_start=0.5, alpha_end=0.1,
                        blend_rate=0.5, softness=0.0, verbose=False):
    """Flat spectrum with blending to avoid getting stuck.
    
    Instead of hard projection, blends with the previous iterate.
    """
    n = len(a0)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    target_mag = compute_target_magnitude(n, N, taper=0, softness=softness)
    log_target = np.log(np.maximum(target_mag, 1e-30))
    
    a = norm_simplex(a0)
    c = fast_conv(a)
    best_m = c.max()
    best_a = a.copy()
    t_start = time.monotonic()
    
    for it in range(300000):
        if time.monotonic() - t_start > time_limit:
            break
        
        progress = min(1.0, it / 100000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        blend = blend_rate * (1.0 - 0.5 * progress)
        
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-30)
        phase = np.angle(A)
        
        log_mag = np.log(mag)
        log_mag_new = (1.0 - alpha) * log_mag + alpha * log_target
        mag_new = np.exp(log_mag_new)
        mag_new[0] = 1.0
        
        A_new = mag_new * np.exp(1j * phase)
        a_raw = np.fft.irfft(A_new, N)[:n]
        
        # Blend with original before projecting
        a_blended = (1.0 - blend) * a + blend * a_raw
        a_new = np.maximum(a_blended, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        c = fast_conv(a_new)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    FlatSpecBlend it={it} ({elapsed:.0f}s): score={2*n*M:.6f}")
    
    return best_a, best_m


def ftrl_optimize(a0, time_limit, verbose=False):
    """FTRL mirror descent with annealed log-sum-exp smoothing."""
    n = len(a0)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t_start = time.monotonic()
    
    c0 = 1.0
    T_init = 30.0
    T_end = 5.0
    top_k = 15
    
    for t in range(1, 500000):
        if time.monotonic() - t_start > time_limit:
            break
        
        progress = min(1.0, t / 200000.0)
        beta = T_init * (T_end / T_init) ** progress
        eta = c0 / (np.sqrt(t) + 1.0)
        
        c = fast_conv(a)
        
        # Compute subgradient from top peaks
        k_max = min(top_k, size)
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = c[top_idx].copy()
        
        max_v = top_vals[0]
        if max_v < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            exponents = beta * (top_vals - max_v) / max_v
            exponents = np.maximum(exponents, -50.0)
            w = np.exp(exponents)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Periodic spectral re-flattening
        if t % 5000 == 0 and time.monotonic() - t_start < time_limit - 10:
            A = np.fft.rfft(a, N)
            mag = np.abs(A)
            mag = np.maximum(mag, 1e-30)
            phase = np.angle(A)
            # Gentle flattening
            mag_smooth = mag ** 0.95
            mag_smooth[0] = 1.0
            A_smooth = mag_smooth * np.exp(1j * phase)
            a_smooth = np.fft.irfft(A_smooth, N)[:n]
            a_smooth = np.maximum(a_smooth, 0.0)
            ss = a_smooth.sum()
            if ss > 1e-15:
                a_smooth = a_smooth / ss
                c_check = fast_conv(a_smooth)
                if c_check.max() < best_m:
                    best_m = c_check.max()
                    best_a = a_smooth.copy()
                a = 0.5 * a + 0.5 * a_smooth
                a = norm_simplex(a)
        
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    FTRL t={t} ({elapsed:.0f}s): score={2*n*best_m:.6f}")
    
    return best_a, best_m

def make_init_bowl(n, sharpness=2.0):
    """Bowl-shaped: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.2, pos2=0.8, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.1
    return norm_simplex(a)


def make_init_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_random_smooth(n, rng, num_freqs=15):
    """Random smooth initialization using sine series."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, num_freqs + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.01)
    return norm_simplex(a)


def make_init_v_shape(n, power=1.0):
    """V-shaped: minimum at center, maximum at edges."""
    x = np.linspace(0, 1, n)
    a = np.abs(2.0 * x - 1.0) ** power + 0.05
    return norm_simplex(a)


def make_init_triangular(n):
    """Triangular: peaks at edges."""
    x = np.linspace(0, 1, n)
    a = np.abs(1.0 - 2.0 * x) + 0.1
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    print(f"\n=== Phase 4: Length Variation (n_best={n_best}) ===")
    
    # Try a longer length
    for new_n in [n_best + 500, n_best + 1000, n_best - 500]:
        if new_n < 1500 or new_n > 10000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")
        
        t4 = min(remaining * 0.20, 100.0)
        if t4 > 15:
            a_out, _ = flat_spectrum_optimize(a_interp, t4,
                                               alpha_start=0.25, alpha_end=0.04, softness=0.0)
            try_update(a_out, f"fs_n{new_n}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_f, _ = ftrl_optimize(norm_simplex(best_seq), min(30.0, remaining * 0.3))
                try_update(a_f, f"ftrl_n{len(best_seq)}_v2")
    
    # === Phase 5: Intensive final optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final Intensive (n={n_best}, best={best_score:.6f}) ===")
    t5 = min(remaining * 0.40, 400.0)
    
    if t5 > 20:
        # Multiple short flat spectrum runs with different params
        params_list = [
            (0.2, 0.03, 0.0),
            (0.15, 0.02, 0.02),
            (0.3, 0.05, 0.0),
            (0.1, 0.01, 0.05),
        ]
        
        t_per5 = t5 / (len(params_list) + 2)
        
        for ps in params_list:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 10:
                break
            t_avail = min(t_per5, remaining)
            if t_avail < 8:
                break
            
            a_in = norm_simplex(best_seq)
            # Small random perturbation for diversity
            rng5 = np.random.default_rng(int(time.monotonic() * 1000) % 10000)
            a_in = norm_simplex(a_in + rng5.normal(0, 0.005, len(a_in)))
            
            a_out, _ = flat_spectrum_optimize(a_in, t_avail,
                                               alpha_start=ps[0], alpha_end=ps[1], softness=ps[2])
            try_update(a_out, f"fs_final_{ps[0]}_{ps[2]}")
        
        # FTRL on best
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = ftrl_optimize(norm_simplex(best_seq), min(60.0, remaining * 0.5))
            try_update(a_out, "ftrl_final")
            
            # Spectral polish
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_sp, _ = spectral_polish(norm_simplex(best_seq), min(20.0, remaining * 0.4))
                try_update(a_sp, "spec_final")
    
    # === Phase 6: Final short flat spectrum with very low alpha (fine tuning) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        print(f"\n=== Phase 6: Fine Tuning ({remaining:.0f}s remaining) ===")
        t6 = remaining * 0.6
        if t6 > 5:
            a_in = norm_simplex(best_seq)
            # Very gentle flat spectrum
            a_out, _ = flat_spectrum_optimize(a_in, t6,
                                               alpha_start=0.05, alpha_end=0.01, softness=0.0)
            try_update(a_out, "fs_finetune")
            
            # Also try blend variant
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 8:
                a_out2, _ = flat_spectrum_blend(a_in, remaining * 0.5,
                                                 alpha_start=0.08, alpha_end=0.02,
                                                 blend_rate=0.4, softness=0.01)
                try_update(a_out2, "blend_finetune")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_init_bowl(n_fb, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
