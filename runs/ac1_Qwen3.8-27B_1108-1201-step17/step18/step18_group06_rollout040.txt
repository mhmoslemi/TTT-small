def fast_autocorr(a):

    a = np.asarray(a, dtype=np.float64)

    A = np.fft.rfft(a, fft_size)

def make_ramp(n, alpha):
    """Create normalized (i+1)^alpha sequence."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    return a / a.sum()


def make_tapered_ramp(n, alpha, beta):
    """Create a[i] = (i+1)^alpha * (1 - (i/n)^beta) normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / n
    a = x ** alpha * (1.0 - 0.3 * t ** beta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return make_ramp(n, alpha)
    return a / s


def make_two_factor(n, alpha, nramp):
    """Ramp of length nramp followed by decay."""
    a = np.zeros(n)
    x = np.arange(1, nramp + 1, dtype=np.float64)
    a[:nramp] = x ** alpha
    if nramp < n:
        decay = np.exp(-np.arange(1, n - nramp + 1) * 2.0 / max(1, nramp))
        a[nramp:] = a[nramp - 1] * decay
    s = a.sum()
    if s < 1e-15:
        return make_ramp(n, 1.0)
    return a / s


def compute_gradient_at_peaks(a, conv, n_peaks=30):
    """Subgradient using top peaks with softmax weighting."""
    n = len(a)
    top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    # Softmax weights over top peaks
    vals = conv[top_idx]
    max_v = vals.max()
    exp_v = np.exp((vals - max_v) / max(1e-10, max_v * 0.01))
    w = exp_v / exp_v.sum()

    grad = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        ki = int(k)
        wi = w[idx_i]
        if wi < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        j_range = np.arange(j_lo, j_hi + 1)
        pair = ki - j_range
        grad[j_range] += wi * 2.0 * a[pair]
    return grad

        x_prev = x
        x = a_new

        M_new = conv.max()
        M_x = fast_autocorr(x).max()
        M_eff = min(M_new, M_x)
        if M_eff < best_m - 1e-16:
            best_m = M_eff
            best_a = (x if M_x <= M_new else y).copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.1, 0.02 + 0.01 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                print(f"      MD restart {restart_count} t={t}: J={2*n*best_m:.6f}")

        if verbose and t % 10000 == 0:
            print(f"      MD t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def spectral_flatten(a0, time_limit, verbose=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 1e-30:
            break

        # Flatten: push power from peaks to mean
        alpha = 0.3 + 0.7 * min(1.0, it / 500.0)
        P_new = P + alpha * (P_mean - P)
        P_new = np.maximum(P_new, 1e-30)

        mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.5, 2.0)
        A_new = A * mag_ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.2
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()

        if verbose and it % 500 == 0:
            print(f"      Spec it={it} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Direct coordinate moves to reduce peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 100:
        conv = fast_autocorr(a)
        k_max = int(np.argmax(conv))

        # Find pair with max contribution at k_max
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        if j_lo > j_hi:
            stall += 1
            continue
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        src = idx[np.argmax(contrib)]

        if a[src] < 1e-12:
            stall += 1
            continue

        # Find low-cost destination
        # Cost: how much a[j] contributes to top peaks
        top_ks = np.argpartition(conv, -10)[-10:]
        costs = np.zeros(n)
        for k2 in top_ks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            if lo2 <= hi2:
                valid = np.arange(lo2, hi2 + 1)
                costs[valid] += a[k2i - valid]

        # Pick destination with low cost and low value
        sorted_idx = np.argsort(costs * 0.1 + a)
        dst = sorted_idx[0]
        if dst == src:
            dst = sorted_idx[1] if len(sorted_idx) > 1 else int(rng.integers(n))

        moved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_t = fast_autocorr(a_trial).max()
            if M_t < best_m - 1e-16:
                a = a_trial
                best_m = M_t
                best_a = a.copy()
                moved = True
                stall = 0
                break

        if not moved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def safe_eval(a):
    """Safely compute the objective value."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """Main search: ramp initializations + mirror descent + spectral + coordinate descent."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Generate initial candidates
    candidates = []
    n_base = 4200

    # KEY: Linear ramp (alpha=1) gives J -> 4/3
    for alpha in [0.8, 1.0, 1.0, 1.2]:
        a = make_ramp(n_base, alpha)
        candidates.append((a, f"ramp_a{alpha}_n{n_base}"))

    # Tapered ramps
    candidates.append((make_tapered_ramp(n_base, 1.0, 2.0), "tapered_1_2"))
    candidates.append((make_tapered_ramp(n_base, 1.0, 4.0), "tapered_1_4"))

    # Different lengths
    for n_try in [5000, 6000]:
        candidates.append((make_ramp(n_try, 1.0), f"ramp_1_n{n_try}"))

    # If parent exists, include it
    if parent is not None:
        p_norm = parent / parent.sum()
        candidates.insert(0, (p_norm, "parent_norm"))

    # Evaluate all initial candidates
    print(f"Initial candidates: {len(candidates)}")
    for a, label in candidates:
        sc = safe_eval(a)
        if np.isfinite(sc):
            print(f"  Init [{label}]: {sc:.6f}")

    # Phase 1: Mirror descent on top candidates
    print(f"\n=== Phase 1: Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    # Sort by initial score
    scored = []
    for a, label in candidates:
        sc = safe_eval(a)
        scored.append((sc, a, label))
    scored.sort(key=lambda x: x[0])

    n_cands = min(6, len(scored))
    t_per = max(20.0, min(300.0, remaining * 0.40 / n_cands))

    for i in range(n_cands):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget:
            break
        t_avail = min(t_per, time_budget - elapsed)
        if t_avail < 10:
            break
        sc0, a0, label = scored[i]
        print(f"  MD [{label}] (t={t_avail:.0f}s, init={sc0:.6f})...")
        a_out, _ = mirror_descent(a0, t_avail, n_peaks=40, lr0=3.0,
                                  restart_every=3000, rng_seed=42 + i, verbose=False)
        try_update(a_out, f"md_{label}")

    # Phase 2: Spectral flattening on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Spectral Flattening ===")
    if remaining > 60 and best_seq is not None:
        t2 = min(remaining * 0.15, 200.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = spectral_flatten(a_in, t2, verbose=False)
        try_update(a_out, "spectral")

        # Gradient polish after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t2b = min(remaining * 0.10, 120.0)
            a_out, _ = mirror_descent(best_seq / best_seq.sum(), t2b,
                                       n_peaks=60, lr0=2.0, restart_every=5000,
                                       rng_seed=99, verbose=False)
            try_update(a_out, "md_post_spec")

    # Phase 3: Mirror descent with different params on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Aggressive Mirror Descent ===")
    if remaining > 60 and best_seq is not None:
        t3 = min(remaining * 0.20, 250.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = mirror_descent(a_in, t3, n_peaks=80, lr0=5.0,
                                  restart_every=2000, rng_seed=7, verbose=True)
        try_update(a_out, "md_agg")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate Descent ===")
    if remaining > 40 and best_seq is not None:
        t4 = min(remaining * 0.12, 150.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = coordinate_descent(a_in, t4, verbose=False, rng_seed=42)
        try_update(a_out, "coord")

        a_out, _ = coordinate_descent(a_in, t4, verbose=False, rng_seed=777)
        try_update(a_out, "coord2")

    # Phase 5: Try different n by interpolation
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for new_n in [n_best + 1000, n_best + 2000, max(3000, n_best - 1000)]:
            if new_n <= 0 or new_n > 12000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            a_interp = interpolate_to_length(best_seq, new_n)
            try_update(a_interp, f"interp_n{new_n}")

            t5 = min(remaining * 0.25, 150.0)
            if t5 > 20:
                a_out, _ = mirror_descent(a_interp, t5, n_peaks=50, lr0=3.0,
                                          restart_every=4000, rng_seed=100 + new_n % 100,
                                          verbose=False)
                try_update(a_out, f"md_n{new_n}")

    # Phase 6: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Polish ===")
    if remaining > 30 and best_seq is not None:
        a_in = best_seq / best_seq.sum()
        # Spectral then MD
        t6a = min(remaining * 0.30, 80.0)
        t6b = min(remaining * 0.40, 120.0)
        if t6a > 15:
            a_out, _ = spectral_flatten(a_in, t6a, verbose=False)
            try_update(a_out, "spec_final")
            a_in = a_out / max(a_out.sum(), 1e-15)
        if t6b > 15:
            a_out, _ = mirror_descent(a_in, t6b, n_peaks=100, lr0=2.0,
                                       restart_every=6000, rng_seed=555, verbose=True)
            try_update(a_out, "md_final")

    # Phase 7: One more coordinate descent pass
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        t7 = min(remaining * 0.30, 80.0)
        a_in = best_seq / best_seq.sum()
        a_out, _ = coordinate_descent(a_in, t7, verbose=False, rng_seed=12345)
        try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = make_ramp(4200, 1.0)
        best_score = safe_eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
