import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1] + 1
    theta = cssv[rho - 1] / rho
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def softmax_weights(c, top_k, temperature):
    """Compute softmax weights over top_k peaks of convolution c."""
    n_c = len(c)
    k = min(top_k, n_c)
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    max_v = top_vals.max()
    if max_v < 1e-30:
        return top_idx, np.ones(k) / k
    scaled = (top_vals - max_v) / temperature
    scaled = np.maximum(scaled, -100.0)
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / s
    return top_idx, w


def compute_subgradient(a, top_idx, weights):
    """Compute weighted subgradient: sum_k w_k * 2*a[k-i] for valid i."""
    n = len(a)
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        if wi < 1e-14:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def spectral_equalize(a, exponent=0.7, blend=0.5):
    """Spectral flattening: compress dynamic range of |FFT(a)|."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.power(ratio, exponent)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    result = (1.0 - blend) * a + blend * a_new
    return project_simplex(result)

                        stall = 0

    return best_a, best_m


def peak_targeting(a0, time_limit, verbose=False):
    """Directly target the maximum peak: push mass away from positions that feed it."""
    n = len(a0)
    a = project_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for iteration in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        k_max = int(np.argmax(c))
        M_curr = c[k_max]

        # Gradient at the single peak
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        if i_start > i_end:
            break
        idx = np.arange(i_start, i_end + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[k_max - idx]

        # Step away from peak
        step = 0.01 / (1.0 + 0.01 * iteration)
        a_new = project_simplex(a - step * grad)
        c_new = fast_conv(a_new)
        M_new = c_new.max()

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        elif M_new < M_curr:
            a = a_new
        else:
            # Try moving mass to random far positions
            src_candidates = idx[a[idx] > 1e-10]
            if len(src_candidates) == 0:
                break
            src = int(src_candidates[rng.integers(len(src_candidates))])
            dst = int(rng.integers(n))
            if dst != src:
                amt = a[src] * 0.05
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    M_t = fast_conv(a_trial).max()
                    if M_t < best_m:
                        best_m = M_t
                        best_a = a_trial.copy()
                    a = a_trial

    return best_a, best_m


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    s = a.sum()
    return a / s


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    s = a.sum()
    return a / s


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator using multiple optimization strategies.
    
    Strategy:
    - Phase 1: Subgradient optimization with Nesterov + line search (diverse inits)
    - Phase 2: Spectral equalization + peak targeting
    - Phase 3: Intensive coordinate descent polish
    - Phase 4: Length variation
    - Phase 5: Final intensive optimization on best
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Subgradient optimization from multiple initializations ===
    print(f"\n=== Phase 1: Subgradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.40, 600.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(parent.copy())
            labels.append("parent")

        inits.append(make_init_cosine(n0, mode=1))
        labels.append("cos1")
        inits.append(make_init_bowl(n0, sharpness=2.5))
        labels.append("bowl25")
        inits.append(make_init_dual_peak(n0, 0.2, 0.8, 0.15))
        labels.append("dualpeak")
        inits.append(make_init_sinc(n0, bw=7.0))
        labels.append("sinc7")

        # Random smooth inits
        rng_init = np.random.default_rng(123)
        x = np.linspace(0, 1, n0)
        for seed in [1, 2, 3]:
            a_r = np.ones(n0) * 0.5
            for f in range(1, 15):
                coeff = rng_init.normal(0, 0.15 / np.sqrt(f))
                phase = rng_init.uniform(0, 2 * np.pi)
                a_r += coeff * np.sin(2 * np.pi * f * x + phase)
            a_r = np.maximum(a_r, 0.001)
            s_r = a_r.sum()
            inits.append(a_r / s_r)
            labels.append(f"rand{seed}")

        # Perturbed parent
        if parent is not None:
            for seed in [10, 20]:
                rng_p = np.random.default_rng(seed)
                pert = rng_p.normal(0, 0.02, n0)
                a_p = np.maximum(parent + pert, 0.0)
                s_p = a_p.sum()
                if s_p > 0.01:
                    inits.append(a_p / s_p)
                    labels.append(f"parent_pert{seed}")

        # Deduplicate
        seen_set = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 20)], 5))
            if sig not in seen_set:
                seen_set.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        print(f"  {len(inits)} initializations, total budget={t_phase1:.0f}s")
        t_per = t_phase1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            remaining_now = time_budget - 30 - elapsed_so_far
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 10:
                break
            print(f"    SG '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = subgradient_optimize(init, t_avail, verbose=False)
            try_update(a_out, f"sg_{lbl}")

    # === Phase 2: Spectral + peak targeting on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Spectral + Peak Targeting (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = project_simplex(best_seq)
        # Spectral
        t2a = t2 * 0.4
        a_sp, _ = spectral_optimize_plus(a_in, t2a, verbose=False)
        try_update(a_sp, f"spectral_n{n_best}")
        # Peak targeting on the best of (best_seq, a_sp)
        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.12, 150.0)
        if t2b > 15:
            a_pt, _ = peak_targeting(project_simplex(best_seq), t2b, verbose=False)
            try_update(a_pt, f"peak_n{n_best}")

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coordinate Descent (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_in = project_simplex(best_seq)
        a_out, _ = coordinate_descent_plus(a_in, t3, verbose=False)
        try_update(a_out, f"cd_n{n_best}")

    # === Phase 4: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length variation ===")
    for delta in [200, 400, -200, -400]:
        new_n = n_best + delta
        if new_n < 2000 or new_n > 6000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_new = interpolate_sequence(best_seq, new_n)
        try_update(a_new, f"interp_n{new_n}")
        t4 = min(remaining * 0.2, 120.0)
        if t4 > 15:
            a_out, _ = subgradient_optimize(a_new, t4, verbose=False)
            try_update(a_out, f"sg_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_sp, _ = spectral_optimize_plus(a_out, min(20.0, remaining * 0.2))
                try_update(a_sp, f"spec_n{new_n}")

    # === Phase 5: Final intensive optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final intensive (n={n_best}) ===")
    t5 = min(remaining * 0.45, 300.0)
    if t5 > 20:
        a_in = project_simplex(best_seq)
        a_out, _ = subgradient_optimize(a_in, t5, verbose=True)
        try_update(a_out, f"final_sg_n{n_best}")

        # Spectral polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sp, _ = spectral_optimize_plus(a_out, min(40.0, remaining * 0.4))
            try_update(a_sp, f"final_spec_n{n_best}")

        # Peak targeting
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_pt, _ = peak_targeting(project_simplex(best_seq), min(30.0, remaining * 0.4))
            try_update(a_pt, f"final_peak_n{n_best}")

    # === Phase 6: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        print(f"\n=== Phase 6: Final CD (n={n_best}) ===")
        t6 = min(remaining * 0.7, 60.0)
        if t6 > 5:
            a_in = project_simplex(best_seq)
            a_out, _ = coordinate_descent_plus(a_in, t6, verbose=False)
            try_update(a_out, f"final_cd_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
