"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient with multiplicative (mirror) descent
2. Temperature annealing schedule
3. Multiple diverse starting points and periodic restarts
4. Length interpolation for exploring different sequence sizes
5. Coordinate-level polish targeting peak-contributing positions
"""

import numpy as np
import time
import math


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lse_value(c, T):
    """Compute log-sum-exp value: T * log(sum(exp(c_k/T)))."""
    c_max = c.max()
    return T * (np.log(np.sum(np.exp((c - c_max) / T))) + c_max / T)


def lse_gradient(a, T):
    """
    Compute gradient of 2n * V_T(conv(a,a)) w.r.t. a.
    Returns (gradient, current_max, lse_value).
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    
    # Softmax weights
    p = np.exp((c - c_max) / T)
    p_sum = p.sum()
    p = p / p_sum
    
    # Gradient: (2n/T) * cross_corr(a, p)
    # g[j] = sum_{m=0}^{n-1} a[m] * p[j+m]
    # Use Hankel matrix approach for n <= 8000, FFT for larger
    if n <= 8000:
        # g[j] = dot(a, p[j:j+n])
        g = np.empty(n)
        for j in range(n):
            g[j] = np.dot(a, p[j:j+n])
    else:
        # FFT approach
        fft_len = 1 << (2 * (2*n - 1) - 1).bit_length()
        A = np.fft.rfft(a, fft_len)
        P = np.fft.rfft(p, fft_len)
        # cross-correlation: ifft(fft(a) * conj(fft(p)))
        g_full = np.fft.irfft(A * np.conj(P), fft_len)
        g = g_full[:n]
    
    grad = (2.0 * n / T) * g
    v = lse_value(c, T)
    return grad, c_max, v


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lse_optimize(a0, time_limit, T_init, T_final, lr_init, verbose=False, seed=42):
    """
    Log-sum-exp smoothed multiplicative gradient descent on simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T = T_init
    lr = lr_init
    patience = 0
    max_patience = 500
    iter_count = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        total_time = time.monotonic() - t0
        progress = min(1.0, total_time / time_limit)
        
        # Anneal temperature
        T = T_init * (T_final / T_init) ** (progress ** 0.5)
        # Anneal learning rate
        lr = lr_init * (1.0 - 0.7 * progress)
        
        grad, c_max, v = lse_gradient(a, T)
        
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Random restart
            pert = rng.normal(0, 0.01, n)
            a_new = a * np.exp(pert)
            a = norm_sum1(a_new)
            patience = 0
            continue
        
        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check if update improved
        c_new = fast_autocorr(a_new)
        m_new = c_new.max()
        
        if m_new <= c_max + 1e-12:
            a = a_new
        else:
            # Halve learning rate
            lr *= 0.5
            if lr < 1e-8:
                lr = lr_init * (1.0 - 0.7 * progress) * 0.1
        
        iter_count += 1
        total_iters += 1
        
        if verbose and iter_count % 100 == 0:
            J = 2.0 * n * best_m
            print(f"    LSE iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, T={T:.4f}, lr={lr:.5f}")
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    LSE done ({time.monotonic()-t0:.0f}s): J={J:.6f}, iters={total_iters}")
    return best_a
