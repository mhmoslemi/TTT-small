    s = a.sum()
    if s < 1e-15:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]

    return np.maximum(v - theta, 0.0)

    """Compute gradient of (2*n/tau)*log(sum exp(tau*ac_k)) using FFT."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    ac_shifted = tau * (ac - ac_max)
    exp_ac = np.exp(ac_shifted)
    p = exp_ac / exp_ac.sum()

    A_rev = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(P * A_rev, fft_size)
    g = 4.0 * n * conv_result[n-1 : 2*n-1]
    return g, ac, p

            # Evaluate improvement
            m_new = fast_autocorr(x_new).max()

                best_m = m_new
                best_a = x_new.copy()

                    eta *= 0.7
                if stall > 100:

                    x_pert = np.maximum(best_a + pert, 0.0)

                    m_pert = fast_autocorr(x_pert).max()

Now let me think about the LP-FW more carefully. The key is to use a large K (number of binding constraints). For n=5200, the full problem has 2*5200-1 = 10399 constraints. Using all of them might be too slow, but using K=2000-5000 should give good results.

The LP-FW algorithm:
1. Compute ac = conv(a, a), M = max(ac)
2. Select top-K largest autocorrelation values as binding constraints
3. Solve LP: maximize sum(b) subject to:
   - b >= 0
   - For each selected k: sum_j T[k,j] * b_j <= M
   where T[k,j] = a[k-j] (with appropriate bounds)
4. Get direction g = b / sum(b)
5. Line search: a_new = (1-alpha)*a + alpha*g
6. Repeat

The LP has n=5200 variables and K constraints. With K=2000, this is a 5200-variable, 2000-constraint LP. Modern solvers should handle this in ~1-5 seconds.

Let me also think about a different approach: **SDD (Spectral Descent with Dreze)** or **Submodular-based methods**. Actually, these might be overkill.

Another idea: **Proximal point method** with a Bregman divergence. The Bregman divergence for the entropic regularizer gives us the mirror descent algorithm, which is essentially what AMD does.

Let me focus on making the Nesterov PG and LP-FW work well, and add AMD as well.

Actually, I realize I should also think about the problem from a different angle. The objective 2*n*max_conv / (sum)^2 is equivalent to:
- Normalize so sum(a) = sqrt(2n) (so that the denominator is 1... wait no)
- Actually: J = 2*n*M / S^2 where M = max_conv, S = sum(a)
- If we normalize S=1: J = 2*n*M
- If we normalize S=sqrt(2n): J = M

So minimizing J is equivalent to minimizing M = max_k (a*a)_k subject to sum(a) = 1 (and a >= 0).

This is a convex optimization problem! We're minimizing a convex function (max of linear functions) over a convex set (simplex). The LP-FW is exactly the right algorithm for this.

The issue is that the problem is large (n=5200, 2n-1=10399 linear functions being maxed). The LP subproblem in FW is the key computational bottleneck.

For the LP subproblem, the constraint matrix T has a special structure: each row k is a shifted version of a. This means T is (partially) a Toeplitz matrix. We could exploit this structure for faster LP solving, but cvxpy doesn't directly support this.

However, we can use the fact that the dual of the LP is:
minimize M * sum(y_k)
subject to: for each j: sum_k T[k,j] * y_k >= 1
           y_k >= 0

The dual has K variables (one per constraint) and n constraints. If K < n (which it will be since we only select top-K), the dual might be easier to solve.

Actually, for cvxpy, it auto-detects the primal/dual form, so we don't need to worry about this.

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, tau):
    """Compute gradient of (2*n/tau)*log(sum exp(tau*ac_k)) using FFT cross-correlation."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    ac_shifted = tau * (ac - ac_max)
    exp_ac = np.exp(ac_shifted)
    p = exp_ac / exp_ac.sum()
    
    # g_i = 4*n * sum_k p_k * a_{k-i} for i in [0, n-1]
    # = 4*n * conv(p, a_reversed)[n-1 : 2*n-1]
    a_rev = a[::-1]
    conv_len = len(p) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(P * A_rev, fft_size)
    g = 4.0 * n * conv_result[n - 1: 2 * n - 1]
    return g, ac, p


def nesterov_pg(a0, time_limit, verbose=True, rng_seed=42, tau_start=5.0, tau_end=500.0):
    """Nesterov-accelerated projected gradient with adaptive tau annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Adaptive tau: start smooth, end sharp
    n_phases = 8
    tau_values = tau_start * (tau_end / tau_start) ** (np.arange(n_phases) / (n_phases - 1))
    
    for phase_idx, tau in enumerate(tau_values):
        if time.monotonic() - t0 > time_limit:
            break
        phase_start = time.monotonic()
        remaining = time_limit - (phase_start - t0)
        # Allocate more time to later (sharper) phases
        phase_frac = 0.3 + 0.7 * (phase_idx / max(1, n_phases - 1))
        phase_limit = remaining * phase_frac * 0.8
        
        x = a.copy()
        x_prev = a.copy()
        t_k = 1.0
        eta = 0.5
        stall = 0
        iter_count = 0
        max_stall = 400
        
        while time.monotonic() - phase_start < phase_limit and stall < max_stall:
            # Nesterov momentum
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            beta = (t_k - 1.0) / t_next
            y = x + beta * (x - x_prev)
            
            # Project y to be valid
            y_pos = np.maximum(y, 0.0)
            y_sum = y_pos.sum()
            if y_sum < 1e-15:
                y = x.copy()
            else:
                y = y_pos / y_sum
            
            # Compute smooth gradient
            g, ac, p = compute_smooth_gradient(y, tau)
            
            # Gradient step + simplex projection
            x_trial = y - eta * g
            x_new = project_simplex(x_trial)
            
            # Evaluate
            m_new = fast_autocorr(x_new).max()
            
            if m_new < best_m - 1e-12:
                best_m = m_new
                best_a = x_new.copy()
                stall = 0
                eta = min(eta * 1.01, 3.0)
            else:
                stall += 1
                if stall > 40 and eta > 0.02:
                    eta *= 0.75
                if stall > 80:
                    # Adaptive perturbation
                    pert_scale = 0.001 + 0.005 * (stall / max(1, max_stall))
                    pert = rng.normal(0, pert_scale, n) * best_a
                    x_pert = np.maximum(best_a + pert, 0.0)
                    x_pert = x_pert / x_pert.sum()
                    m_pert = fast_autocorr(x_pert).max()
                    if m_pert < best_m:
                        best_m = m_pert
                        best_a = x_pert.copy()
                    x = x_pert
                    x_prev = x_pert.copy()
                    t_k = 1.0
                    eta = 0.3
                    stall = 0
                    continue
            
            x_prev = x
            x = x_new
            t_k = t_next
            iter_count += 1
            
            if verbose and iter_count % 2000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    NestPG tau={tau:.1f} iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}")
        
        a = norm_sum1(x)
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, K_init=500, K_max=None):
    """LP-based Frank-Wolfe with aggressive adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if K_max is None:
        K_max = min(2 * n - 1, 8000)
    K = K_init
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix row by row efficiently
        # T[k, j] = a[k-j] for valid j
        # We build it incrementally to save memory
        T_rows = []
        for k in binding_idx:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            row = np.zeros(n)
            row[j_start:j_end] = a[ki - np.arange(j_start, j_end)]
            T_rows.append(row)
        T_mat = np.array(T_rows)
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_mat @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 200, K_max)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 100, K_max)
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Line search
        base_m = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_m
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                       0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        iter_count += 1
        
        # Adaptive K growth
        if stall < 10:
            K = min(K + 300, K_max)
        elif stall < 30:
            K = min(K + 100, K_max)
        else:
            K = min(K + 50, K_max)
        
        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")
    
    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=200, beta_param=8.0, c0=5.0,
                 restart_every=5000, verbose=False, rng_seed=42):
    """Away-Maps Descent with large active set and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta_param
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute soft subgradient with large active set
        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 50, 2 * n - 1)
            cur_beta = max(cur_beta - 2.0, 3.0)
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with sensitivity-based moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 80
    n_top = 15
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            top_cands = candidates[np.argsort(sens[candidates])[:5]]
            
            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.02, 0.01, 0.03, 0.05, 0.08]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
            
            if found:
                break
        
        if not improved:
            stall += 1
            # Random perturbation when stalled
            if stall > 20:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
    
    return best_a, best_m

        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Quick multi-start AMD ===
    print(f"\n=== Phase 1: Multi-start AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.12, 200.0)
    
    if t_phase1 > 60:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_u_shape(n0, 2.0))
        labels.append("u2.0")
        inits.append(make_u_shape(n0, 4.0))
        labels.append("u4.0")
        inits.append(make_bimodal(n0, 0.2))
        labels.append("bimodal")
        inits.append(make_cosine(n0))
        labels.append("cosine")
        
        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            print(f"  AMD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=300, beta_param=10.0, c0=5.0,
                                     restart_every=4000, verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")
    
    # === Phase 2: Nesterov PG (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Nesterov PG (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_pg(a_in, t2, verbose=True, rng_seed=42,
                                tau_start=5.0, tau_end=400.0)
        try_update(a_out, "nestpg_main")
    
    # === Phase 3: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        K_init = min(500, 2 * n_best - 1)
        K_max = min(6000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, K_init=K_init, K_max=K_max)
        try_update(a_out, "lp_main")
    
    # === Phase 4: Nesterov PG round 2 (fine-tuning) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Nesterov PG r2 (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_pg(a_in, t4, verbose=True, rng_seed=200,
                                tau_start=50.0, tau_end=800.0)
        try_update(a_out, "nestpg_r2")
    
    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        K_init = min(1000, 2 * n_best - 1)
        K_max = min(10399, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True, K_init=K_init, K_max=K_max)
        try_update(a_out, "lp_r2")
    
    # === Phase 6: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coordinate polish ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=789)
        try_update(a_out, "coord_final")
    
    # === Phase 7: Final short Nesterov + LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final optimization ===")
        t7 = min(remaining * 0.5, 80.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_pg(a_in, t7 * 0.6, verbose=False, rng_seed=500,
                                    tau_start=100.0, tau_end=1000.0)
            try_update(a_out, "nestpg_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            n_best = len(best_seq)
            t7b = min(remaining * 0.6, 60.0)
            if t7b > 15 and n_best <= 6000:
                a_in = norm_sum1(best_seq)
                K_init = min(2000, 2 * n_best - 1)
                K_max = min(10399, 2 * n_best - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                           K_init=K_init, K_max=K_max)
                try_update(a_out, "lp_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
