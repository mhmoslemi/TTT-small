        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for violations and add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search
        if t <= 3:
            alpha = 1.0
        else:
            alpha, _ = line_search_alpha(y, g_dir, n)
        
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        y = y_new
        iter_count += 1
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW-MaxSum iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW-MaxSum done (" + str(elapsed) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def adaptive_subgradient(a0, time_limit, verbose=True):
    """
    Adaptive mirror descent with:
    - Nesterov acceleration
    - Adaptive top_k and beta
    - Restart with perturbation
    """
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    x_prev = x.copy()
    gamma = 5.0
    stall = 0
    cur_top_k = 30
    cur_beta = 6.0
    restart_count = 0
    restart_every = 3000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute autocorrelation and gradient
        ac = fast_autocorr(y)
        
        # Adaptive top_k: start small, grow over time
        cur_top_k = min(int(20 + t / 500.0), 80)
        cur_beta = max(6.0 - t / 2000.0, 2.0)
        
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                grad[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror descent update (exponential family)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y + 1e-30) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.02 / (1.0 + 0.03 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            gamma = max(gamma * 0.95, 1.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SGD restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))
    
    return best_a, best_m


def pair_transfer_refinement(a0, time_limit, verbose=True):
    """
    Direct refinement: identify peak-contributing pairs and transfer mass.
    More targeted than coordinate descent.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-25:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Find the "partner" index
            partner = k_int - src
            partner_val = a[partner]
            
            # Score all positions by their "danger" (how much they'd increase peaks if we add mass there)
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            # Find safe destination (low score)
            candidates = np.where(scores < scores.min() + 0.1 * (scores.max() - scores.min() + 1e-20))[0]
            if len(candidates) < 5:
                candidates = np.where(scores == scores.min())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                dst = int(candidates[(list(candidates).index(src) + 1) % len(candidates)]) if src in candidates else candidates[0]
            
            # Try transferring
            found = False
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                Tf = np.fft.rfft(a_trial, fft_size)
                ac_trial = np.fft.irfft(Tf * Tf, fft_size)[:size]
                M_trial = ac_trial.max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
            
            if not found and stall > 50:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-12:
                    frac = rng.uniform(0.001, 0.008)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        Tf = np.fft.rfft(a, fft_size)
                        ac_new = np.fft.irfft(Tf * Tf, fft_size)[:size]
                        M_new = ac_new.max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            if stall % 100 == 0:
                # More aggressive random moves
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairRefine (" + str(elapsed) + "s): J=" + repr(J) + ", stall=" + str(stall))
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Improved LP-FW with max-sum formulation (faster per solve)
2. Aggressive active set management with periodic full refresh

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:

            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  LP solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def line_search_best_alpha(y, g_dir, n):
    """Try a grid of alpha values, return the best."""
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    alphas = [0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    best_alpha = 0.1
    best_M = float('inf')
    for alpha in alphas:
        trial = (1.0 - alpha) * y + alpha * g_dir
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-15:
            continue
        trial = trial / s_t
        Tf = np.fft.rfft(trial, fft_size)
        ac = np.fft.irfft(Tf * Tf, fft_size)[:size]
        M_trial = ac.max()
        if M_trial < best_M:
            best_M = M_trial
            best_alpha = alpha
    return best_alpha, best_M


def lp_fw_maxsum(a0, time_limit, solver, initial_K=400, max_K=2500,
                 refresh_every=40, add_per_iter=50, verbose=True):
    """
    LP-FW using max-sum subproblem:
      max 1^T g  s.t.  A@g <= M_curr,  g >= 0
    This avoids an extra variable M and is typically faster per solve.
    """
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 2000
    iter_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW-M: stall " + str(no_improve) + ", stopping")
            break

        # Periodic full refresh of active set to top-K lags
        if t % refresh_every == 0:
            ac = fast_autocorr(y)
            top_lags = np.argsort(ac)[::-1][:max_K]
            active_set = set(int(k) for k in top_lags)

        M_curr = fast_autocorr(y).max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) == 0:
            break

        if len(constraint_list) > max_K:
            ac_curr = fast_autocorr(y)
            margins = np.array([M_curr - ac_curr[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        coeffs = build_constraint_matrix(y, constraint_list, n)

        # Max-sum LP: max sum(g) s.t. A@g <= M_curr, g >= 0
        g_var = cp.Variable(n, nonneg=True)
        constraints = [coeffs @ g_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Detect violations and add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        if t <= 3:
            alpha = 1.0
        else:
            alpha, _ = line_search_best_alpha(y, g_dir, n)

        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        y = y_new
        iter_count += 1

        if verbose and t % 80 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW-M t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(J) + " |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW-M done (" + str(round(elapsed, 1)) + "s) J=" + repr(J) + " iters=" + str(iter_count))
    return best_a, best_m


def adaptive_mirror_descent(a0, time_limit, verbose=False):
    """
    Mirror descent (exponential family) with Nesterov acceleration
    and adaptive focusing parameters.
    """
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    x_prev = x.copy()
    gamma = 4.0
    stall = 0
    restart_every = 2500
    restart_count = 0

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)

        # Adaptive focus
        top_k = min(int(25 + t / 400.0), 70)
        beta = max(5.0 - t / 3000.0, 2.0)

        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y + 1e-30) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise = 0.015 / (1.0 + 0.04 * restart_count)
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            gamma = max(gamma * 0.94, 1.0)

    return best_a, best_m


def pair_refinement(a0, time_limit, verbose=True):
    """
    Targeted mass transfer: identify the pair (i, k-i) contributing most
    to each peak lag, and move mass from the heavier element to a safe position.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(55)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-28:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Compute danger score for each position
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            # Pick a safe destination
            threshold = scores.min() + 0.05 * (scores.max() - scores.min() + 1e-20)
            candidates = np.where(scores <= threshold)[0]
            if len(candidates) < 3:
                candidates = np.arange(n)

            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                mask = candidates != src
                if mask.sum() > 0:
                    dst = int(candidates[mask][0])
                else:
                    continue

            found = False
            for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.12]:
                amt = src_val * frac
                if amt < 1e-17:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                Tf = np.fft.rfft(a_trial, fft_size)
                ac_t = np.fft.irfft(Tf * Tf, fft_size)[:size]
                M_t = ac_t.max()
                if M_t < best_m - 1e-16:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 80:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-13:
                    amt = a[src] * rng.uniform(0.002, 0.01)
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 150 == 0:
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-13:
                        amt = a[i] * rng.uniform(0.001, 0.015)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairRef (" + str(round(elapsed, 1)) + "s) J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main orchestrator: LP-FW max-sum as primary driver, with SGD,
    pair refinement, and multi-n exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print("Time budget: " + repr(round(time_budget, 1)) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = norm_sum1(p)
            best_score = evaluate_safe(best_seq)
            print("Parent: n=" + str(len(best_seq)) + " score=" + repr(best_score))
    except Exception:
        pass

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + " score=" + repr(best_score))

    n0 = len(best_seq)

    # --- Phase 1: Main LP-FW ---
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        print("\n=== P1: LP-FW max-sum n=" + str(n0) + " t=" + str(round(t1)) + "s ===")
        a_out, _ = lp_fw_maxsum(best_seq, t1, solver,
                                initial_K=400, max_K=2500,
                                refresh_every=35, add_per_iter=50, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("*** BEST [lp1]=" + str(sc))

    # --- Phase 2: Mirror descent for diversification ---
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 30:
        print("\n=== P2: Mirror descent n=" + str(len(best_seq)) + " ===")
        a_out, _ = adaptive_mirror_descent(best_seq, t2, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("*** BEST [md]=" + str(sc))

    # --- Phase 3: LP-FW round 2 ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 50 and n_cur <= 7000:
        print("\n=== P3: LP-FW round2 n=" + str(n_cur) + " ===")
        a_out, _ = lp_fw_maxsum(best_seq, t3, solver,
                                initial_K=500, max_K=3000,
                                refresh_every=30, add_per_iter=40, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("*** BEST [lp2]=" + str(sc))

    # --- Phase 4: Pair refinement ---
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        print("\n=== P4: Pair refine n=" + str(len(best_seq)) + " ===")
        a_out, _ = pair_refinement(best_seq, t4, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("*** BEST [pair]=" + str(sc))

    # --- Phase 5: Multi-n exploration ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    for delta in [300, -300, 500]:
        new_n = n_best + delta
        if new_n < 2000 or new_n > 8000:
            continue
        if remaining < 150:
            break
        print("\n=== P5: n=" + str(new_n) + " ===")
        a_try = interpolate_sequence(best_seq, new_n)
        sc_try = evaluate_safe(a_try)
        if np.isfinite(sc_try) and sc_try < best_score:
            best_score = sc_try
            best_seq = a_try.copy()
            print("*** BEST [interp n=" + str(new_n) + "]=" + str(sc_try))

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 180 and new_n <= 7500:
            t5 = min(remaining * 0.35, 200.0)
            if t5 > 50:
                a_out, _ = lp_fw_maxsum(a_try, t5, solver,
                                        initial_K=350, max_K=2000,
                                        refresh_every=40, add_per_iter=30, verbose=False)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print("*** BEST [lp n=" + str(new_n) + "]=" + str(sc))
                remaining = time_budget - (time.monotonic() - t_global)

    # --- Phase 6: Final LP-FW polish ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_fin = len(best_seq)
    if remaining > 100 and n_fin <= 7500:
        print("\n=== P6: Final LP-FW n=" + str(n_fin) + " ===")
        t6 = min(remaining * 0.50, 300.0)
        if t6 > 50:
            a_out, _ = lp_fw_maxsum(best_seq, t6, solver,
                                    initial_K=500, max_K=3000,
                                    refresh_every=25, add_per_iter=35, verbose=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print("*** BEST [lp_fin]=" + str(sc))

    # --- Phase 7: Final pair refinement ---
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print("\n=== P7: Final pair n=" + str(len(best_seq)) + " ===")
        t7 = min(remaining * 0.65, 180.0)
        if t7 > 20:
            a_out, _ = pair_refinement(best_seq, t7, verbose=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print("*** BEST [pair_fin]=" + str(sc))

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
