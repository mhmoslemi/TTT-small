def lse_smooth_gradient(a, ac, tau):
    """Gradient of tau*log(sum exp(ac_k/tau)) w.r.t. a via FFT."""
    n = len(a)
    ac_s = ac / tau
    ac_s -= ac_s.max()
    exp_v = np.exp(ac_s)
    p = exp_v / exp_v.sum()

    a_flip = a[::-1].copy()
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1: n - 1 + n]
    return np.maximum(grad, 0.0)

    return best_a, best_m
