"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max smoothed gradient descent with FFT-based gradient computation
2. Adaptive temperature (beta) schedule
3. Mirror descent (exponential) updates for non-negativity
4. Multi-start with diverse parametric initializations
5. Coordinate descent polishing
6. Length exploration with interpolation
"""

import numpy as np
import time
import math

def fast_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_fast(a):
    """Fast evaluation without calling evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def softmax_gradient(a, beta):
    """
    Compute gradient of softmax_beta(conv(a,a)) with respect to a.
    Returns the gradient vector.
    """
    n = len(a)
    c = fast_conv(a)
    size = len(c)
    
    # Compute soft weights
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / exp_bc.sum()
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (with corrections for boundary and diagonal)
    # This is essentially 2 * (w * a) where * denotes appropriate convolution
    # But we need to be careful about the linear convolution boundaries
    
    # The full gradient is:
    # g[i] = sum_k [dc[k]/da[i]] * w[k]
    # dc[k]/da[i] = 2*a[k-i] if 0<=k-i<n and 2i != k
    # dc[k]/da[i] = a[k-i] if 0<=k-i<n and 2i == k
    
    # For large n, the diagonal correction is negligible, so:
    # g[i] ≈ 2 * sum_k w[k] * a[k-i] (where 0 <= k-i < n)
    #        = 2 * (w * a_reversed)[i] ... 
    
    # Actually, let's think of it as:
    # g[i] = 2 * sum_{k: 0<=k<i+n, 0<=k-i<n} w[k] * a[k-i] (approximately)
    #      = 2 * sum_{j: 0<=j<n} w[i+j] * a[j] (where j = k-i, k = i+j)
    
    # So g[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1, where w is defined for k=0..2n-2
    # This is a "cross-correlation" of w and a!
    
    # We can compute this using FFT:
    # cross_corr[i] = sum_j w[i+j] * a[j]
    # = sum_j w_reversed[2n-2-(i+j)] * a[j] ... hmm, let me think differently.
    
    # Alternative: cross_corr[i] = sum_j a[j] * w[i+j]
    # Let's define v[k] = w[k] for k=0..2n-2
    # cross_corr[i] = sum_{j=0}^{n-1} a[j] * v[i+j]
    # This is a valid "sliding dot product" which can be computed via convolution.
    
    # If we reverse v: v_rev[k] = v[2n-2-k]
    # Then cross_corr[i] = sum_j a[j] * v_rev[2n-2-i-j]
    # = (a * v_rev)[2n-2-i] (where * is linear convolution)
    
    # So cross_corr = reverse(convolve(a, v_rev)) reversed = ... 
    
    # Let me just use direct FFT approach:
    # cross_corr[i] = sum_j a[j] * w[i+j]
    # Pad a to length 2n-1, pad w to length 2n-1
    # cross_corr[i] for i=0..n-1
    
    # Actually, the simplest way: 
    # cross_corr = sum over j of a[j] * w shifted by j
    # = convolve-like operation
    
    # Let's use the following trick:
    # cross_corr[i] = <a, w[i:i+n]> (dot product of a with the slice w[i:i+n])
    # This is the "valid" cross-correlation of w and reversed a.
    
    # np.correlate(w, a[::-1], mode='valid') gives us exactly this!
    # But np.correlate might be slow for large arrays. Let's use FFT.
    
    # Using FFT: cross_corr[i] = sum_j a[j] * w[i+j]
    # = IFFT[FFT(a_padded) * conj(FFT(w_padded))][i] ... 
    
    # Actually, let's just use:
    # cross_corr = np.correlate(w, a, mode='valid')
    # where w has length 2n-1 and a has length n
    # np.correlate(w, a, 'valid')[i] = sum_j w[i+j] * a[j] for j=0..n-1
    # Yes! That's exactly what we want.
    
    # For efficiency with large n, use FFT-based correlation:
    fft_size2 = 1 << (size + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size2)
    A = np.fft.rfft(a, fft_size2)
    # cross_corr[i] = sum_j w[i+j]*a[j] = (w * reverse(a))[i+n-1] in full convolution
    # Let's use: cross_corr = ifft(W * conj(A)) but we need to be careful
    
    # Actually, the standard formula:
    # correlation(x, y)[i] = sum_j x[i+j] * conj(y[j])
    # In FFT: correlation = ifft(FFT(x) * conj(FFT(y)))
    # But this gives circular correlation. For linear, we need zero-padding.
    
    # Let me just use a simpler approach: compute g directly via FFT
    # g[i] = 2 * sum_{k=max(0,i)}^{min(2n-2, i+n-1)} w[k] * a[k-i]
    # Let j = k-i: g[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] (when i+j < 2n-1)
    
    # This is the cross-correlation of w (length 2n-1) and a (length n), "valid" mode.
    
    # FFT approach:
    # zero-pad both to length >= (2n-1) + n - 1 = 3n-2
    fft_len = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, fft_len)
    Ar = np.fft.rfft(a[::-1], fft_len)  # reverse a
    # conv(w, a_rev)[k] = sum_i w[i] * a_rev[k-i] = sum_i w[i] * a[n-1-(k-i)]
    # = sum_i w[i] * a[n-1-k+i]
    # Let j = n-1-k+i, so i = j-n+1+k
    # conv(w, a_rev)[k] = sum_j w[j-n+1+k] * a[j] (for valid j, i)
    
    # Hmm, this is getting complicated. Let me just use the direct formula with FFT.
    
    # Simplest efficient approach:
    # cross_corr[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = sum_{j=0}^{n-1} a[j] * w[i+j]
    # If we define w_ext of length 3n-2 (zero-padded), then:
    # cross_corr = ifft(fft(w_ext) * fft(a_ext))[n-1 : 2n-2]
    # No, that's not right either.
    
    # Let me just use np.correlate for correctness (it's O(n*m) but for our purposes
    # with n up to ~6000, it should be OK for the number of iterations we need).
    # Actually np.correlate is O(n*m) which for n=6000 would be 36M operations per call.
    # That might be too slow. Let me use FFT.
    
    # Correct FFT approach for cross-correlation:
    # cross_corr[i] = sum_{j=0}^{n-1} a[j] * w[i+j], for i = 0, 1, ..., n-1
    # (w has length 2n-1, so i+j ranges from i to i+n-1, which is at most 2n-2)
    
    # Method: 
    # 1. Reverse w: w_rev[k] = w[2n-2-k], length 2n-1
    # 2. Compute linear convolution of a (length n) and w_rev (length 2n-1)
    #    conv[i] = sum_j a[j] * w_rev[i-j] = sum_j a[j] * w[2n-2-i+j]
    # 3. cross_corr[i] = conv[i+n-1] = sum_j a[j] * w[2n-2-(i+n-1)+j] = sum_j a[j] * w[n-1-i+j]
    #    Hmm, that gives sum_j a[j] * w[n-1-i+j], not what we want.
    
    # Let me try a different approach. Define:
    # h[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the "cross-correlation" of a with w, but with a specific convention.
    
    # Standard cross-correlation: (a ⋆ w)[i] = sum_j a[j+i] * conj(w[j])
    # That's not quite what we have.
    
    # Our formula: h[i] = sum_j a[j] * w[i+j]
    # Let's substitute k = i+j: h[i] = sum_k a[k-i] * w[k] (where k-i is in [0,n-1])
    # = sum_k a[k-i] * w[k]
    # This is the cross-correlation (w ⋆ a)[i] = sum_k w[k] * a[k-i]!
    
    # So h = cross_correlation(w, a) in "full" mode, but we only want the "valid" part.
    
    # FFT computation of cross-correlation:
    # (w ⋆ a)[i] = ifft(FFT(w) * conj(FFT(a)))[i] (circular)
    # For linear, zero-pad to length >= len(w) + len(a) - 1 = 3n-2
    
    fft_len = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, fft_len)
    A = np.fft.rfft(a, fft_len)
    xcorr = np.fft.irfft(W * np.conj(A), fft_len)
    
    # xcorr[i] = sum_k w[k] * a[(k-i) mod fft_len] (circular)
    # For linear (with zero-padding), xcorr[i] = sum_k w[k] * a[k-i] for valid k-i
    # We want h[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1
    # where a[k-i] is 0 if k-i < 0 or k-i >= n
    
    # With zero-padding to fft_len >= 3n-2, the linear cross-correlation is:
    # xcorr_linear[i] = sum_{k} w[k] * a_padded[k-i] where a_padded is a zero-padded to fft_len
    # The valid range for i is 0 to len(w)-1 = 2n-2
    # We want i = 0 to n-1
    
    g = 2.0 * xcorr[:n]
    
    # Diagonal correction (for k where 2i == k, the derivative is a[k-i] not 2*a[k-i])
    # This correction is small for large n, so we'll ignore it for now.
    # If needed: for even k, g[k//2] -= w[k] * a[k//2]
    
    return g, c.max(), w

def softmax_objective(a, beta):
    """Compute the softmax-beta smoothed objective."""
    c = fast_conv(a)
    c_max = c.max()
    return float(c_max + np.log(c.size) / beta)  # approximation

def gradient_descent_smoothed(a0, time_limit, beta0, beta_final, decay_rate, lr0, lr_decay, verbose=True):
    """
    Gradient descent on the softmax-smoothed objective.
    Uses mirror descent (exponential) updates.
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_obj = evaluate_fast(a)
    t0 = time.monotonic()
    
    iter_count = 0
    stall = 0
    max_stall = 500
    rng = np.random.default_rng(123)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        progress = min(1.0, t / 5000.0)
        beta = beta0 + (beta_final - beta0) * (progress ** decay_rate)
        
        # Learning rate
        lr = lr0 / (1.0 + lr_decay * t)
        
        # Compute gradient
        g, c_max, w = softmax_gradient(a, beta)
        
        # Mirror descent update
        # a_new[i] = a[i] * exp(-lr * g[i])
        # For numerical stability, use log domain
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - lr * g
        log_a_new -= log_a_new.max()  # normalize for stability
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        
        a = a_new
        iter_count += 1
        
        # Evaluate
        obj = evaluate_fast(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradDesc t={t} ({elapsed:.0f}s): obj={best_obj:.6f}, beta={beta:.1f}, lr={lr:.6f}, stall={stall}")
        
        if stall > max_stall:
            # Perturbation
            a = norm_sum1(best_a + rng.normal(0, 0.02, n) * best_a)
            stall = 0
            lr *= 0.5
            if lr < 1e-8:
                break
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    GradDesc done ({elapsed:.0f}s): obj={best_obj:.6f}, iters={iter_count}")
    return best_a, best_obj

def coordinate_descent_polish(a0, time_limit, verbose=True):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed.
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_obj = evaluate_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(456)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Random coordinate order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try different values for a[i]
            # The constraint is that conv(a,a)[k] <= M for all k
            # When we change a[i], the affected conv values are those where i participates
            # conv[k] involves a[i] when k-i is in [0,n-1], i.e., k in [i, i+n-1]
            # Also when i-k is in [0,n-1], i.e., k in [i-(n-1), i]
            # So k in [max(0, i-(n-1)), min(2n-2, i+n-1)] = [0, min(2n-2, i+n-1)] (since i-(n-1) <= 0 for i < n-1)
            
            # Simple approach: try increasing and decreasing a[i] by small amounts
            cur_val = a[i]
            if cur_val < 1e-15:
                continue
            
            # Try reducing a[i] and compensating elsewhere
            # Or try a local search: a[i] -> a[i] + delta for several deltas
            
            # Quick local search
            best_delta = 0.0
            best_local_obj = best_obj
            
            # Sample some deltas
            deltas = rng.uniform(-0.01, 0.01, 20) * max(cur_val, 0.01 / n)
            for delta in deltas:
                if cur_val + delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] = cur_val + delta
                # Renormalize (keep sum the same by adjusting another coordinate)
                # Actually, let's just renormalize the whole thing
                a_try = a_try / a_try.sum()
                obj_try = evaluate_fast(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_delta = delta
            
            if best_delta != 0.0:
                a[i] = a[i] + best_delta
                a = a / a.sum()
                improved = True
                stall = 0
                if a_copy_obj := evaluate_fast(a) < best_obj:
                    best_obj = a_copy_obj
                    best_a = a.copy()
        
        if not improved:
            stall += 1
            if stall > max_stall:
                break
    
    return best_a, best_obj

def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0, 1, n)
    i = np.arange(n, dtype=np.float64)
    
    candidates = []
    
    # Power laws
    for p in np.random.uniform(0.3, 3.0, 8):
        candidates.append((i + 1) ** p)
        candidates.append((n - i) ** p)
    
    # Beta-like
    for _ in range(8):
        alpha = rng.uniform(0.3, 3.0)
        beta = rng.uniform(0.3, 3.0)
        candidates.append(np.maximum(i + 1, 0.5) ** alpha * np.maximum(n - i, 0.5) ** beta)
    
    # Gaussian mixtures
    for _ in range(8):
        n_components = rng.integers(1, 4)
        a = np.zeros(n)
        for _ in range(n_components):
            c = rng.uniform(0.1, 0.9)
            s = rng.uniform(0.05, 0.4)
            amp = rng.uniform(0.5, 1.0)
            a += amp * np.exp(-((x - c) ** 2) / (2 * s * s))
        candidates.append(np.maximum(a, 0))
    
    # Sigmoid
    for _ in range(4):
        c = rng.uniform(0.3, 0.7)
        s = rng.uniform(5, 50)
        candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))
    
    # Sinusoidal
    for _ in range(4):
        freq = rng.uniform(0.5, 5.0)
        ph = rng.uniform(0, 2 * np.pi)
        candidates.append(np.maximum(np.sin(np.pi * freq * x + ph) ** 2, 0.01))
    
    # Random smooth
    for _ in range(4):
        a = np.exp(-0.5 * (np.linspace(-3, 3, n) + rng.normal(0, 1))**2)
        candidates.append(a)
    
    return candidates

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import time
import math


def fast_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize a to be non-negative with sum 1."""

def evaluate_fast(a):
    """Fast evaluation of the objective."""

    n = len(a)
    s = a.sum()

        return float('inf')

    return float(2.0 * n * c.max() / (s * s))


def interp_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def softmax_gradient_step(a, beta):
    """
    One gradient step for softmax-beta smoothed max(conv(a,a)).
    Returns (gradient, c_max).
    Uses FFT-based cross-correlation for O(n log n) cost.
    """
    n = len(a)
    size = 2 * n - 1
    c = fast_conv(a)
    c_max = c.max()

    # Soft weights (numerically stable)
    shifted = beta * (c - c_max)
    # Clip for numerical safety
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()

    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (approximately)
    # This is the cross-correlation of w (length 2n-1) with a (length n)
    # g[i] = sum_k w[k] * a[k-i] for k-i in [0, n-1]
    #      = (w cross_corr a)[i] in "valid" sense

    # FFT-based cross-correlation
    fft_len = 1 << (size + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_len)
    A = 
np.fft.rfft(a, fft_len)
    # Cross-correlation: (w ⋆ a)[i] = sum_k w[k] * a[k-i]
    # = IFFT(FFT(w) * conj(FFT(a))) evaluated at index i
    xcorr = np.fft.irfft(W * np.conj(A), fft_len)

    # The "valid" cross-correlation values we need are at indices 0..n-1
    # (corresponding to h[i] = sum_k w[k]*a[k-i] where k-i in [0,n-1])
    g = 2.0 * xcorr[:n]

    return g, c_max

        # Learning rate
        lr = lr_start / (1.0 + lr_decay * t)

        # Compute gradient
        g, c_max = softmax_gradient_step(a, beta)

        # Mirror descent (exponential) update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - lr * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-300)

                                stall = 0
                                improved = True

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PairSwap ({elapsed:.0f}s): obj={2.0*n*best_m:.6f}, stall={stall}")

    return best_a, 2.0 * n * best_m


def make_parametric_inits(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    inits = []

    # Power laws (increasing)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append((i + 1.0) ** p)

    # Power laws (decreasing)
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        inits.append((n - i) ** p)

    # Beta-like shapes
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta_p in [0.5, 1.0, 1.5, 2.0]:
            inits.append(np.maximum(i + 1.0, 0.5) ** alpha * np.maximum(n - i, 0.5) ** beta_p)

    # Gaussian
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            inits.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    # Bimodal
    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            for sigma in [0.08, 0.12, 0.18]:
                inits.append(np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) +
                             np.exp(-((x - c2) ** 2) / (2 * sigma ** 2)))

    # Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 40]:
            inits.append(1.0 / (1.0 + np.exp(-s * (x - c))))

    # Sinusoidal
    for freq in [1, 2, 3]:
        for ph in [0.0, np.pi / 4, np.pi / 2]:
            inits.append(np.maximum(np.sin(np.pi * freq * x + ph) ** 2, 0.01))

    # U-shape (concave)
    for sharp in [1.0, 2.0, 3.0, 5.0]:
        inits.append(1.0 / (1.0 + sharp * (x - 0.5) ** 2))

    # Random smooth
    for _ in range(6):
        coeffs = rng.normal(0, 1, 5)
        a = np.ones(n)
        for d, cf in enumerate(coeffs):
            a += cf * np.cos(np.pi * (d + 1) * x)
        inits.append(np.maximum(a, 0.01))

    return inits


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    rng = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_fast(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_fast(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                parent_loaded = True
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        init_list = make_parametric_inits(n_base, rng)
        # Pick a few and evaluate
        best_init = None
        best_init_score = float('inf')
        for init_a in init_list[:20]:
            init_norm = norm_sum1(init_a)
            sc = evaluate_fast(init_norm)
            if sc < best_init_score:
                best_init_score = sc
                best_init = init_norm
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Time budget: {time_budget:.0f}s")

    # === Phase 1: Multi-start short optimization ===
    print(f"\n=== Phase 1: Multi-start (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 15:
        init_list = make_parametric_inits(n0, rng)
        # Shuffle and take a diverse subset
        rng.shuffle(init_list)
        n_starts = min(12, len(init_list))
        per_start = t1 / n_starts
        for idx in range(n_starts):
            if time.monotonic() - t_global > time_budget - t1 + 30:
                break
            a_init = norm_sum1(init_list[idx])
            # Skip if too similar to current best
            t_sub = min(per_start, 12.0)
            if t_sub > 3:
                a_out, _ = soft_gradient_descent(
                    a_init, t_sub,
                    beta_start=50.0, beta_end=300.0,
                    lr_start=0.4, lr_decay=0.01,
                    stall_limit=200, verbose=False
                )
                check_update(a_out, f"start_{idx}")

    # === Phase 2: Main soft-gradient descent ===
    print(f"\n=== Phase 2: Main SGD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_out, _ = soft_gradient_descent(
            best_seq, t2,
            beta_start=100.0, beta_end=600.0,
            lr_start=0.3, lr_decay=0.005,
            stall_limit=400, verbose=True
        )
        check_update(a_out, "sgd_main")

    # === Phase 3: High-beta refinement ===
    print(f"\n=== Phase 3: High-beta SGD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = soft_gradient_descent(
            best_seq, t3,
            beta_start=500.0, beta_end=3000.0,
            lr_start=0.1, lr_decay=0.002,
            stall_limit=300, verbose=True
        )
        check_update(a_out, "sgd_highbeta")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_out, _ = subgradient_polish(
            best_seq, t4, lr_start=0.15, verbose=True
        )
        check_update(a_out, "subgrad")

    # === Phase 5: Pair swap descent ===
    print(f"\n=== Phase 5: Pair swap (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 15:
        a_out, _ = pair_swap_descent(best_seq, t5, verbose=True)
        check_update(a_out, "pairs
wap")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    length_candidates = []
    for delta in [200, -200, 500, -500, 800, -800, 1000, -1000, 1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            length_candidates.append(new_n)
    for rn in [3000, 4000, 5000, 5500, 6000, 7000]:
        if abs(rn - n_best) > 300:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    n_tried = 0
    for new_n in length_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50 or n_tried >= 5:
            break
        print(f"\n  Length n={new_n} (remaining={remaining:.0f}s)")
        a_new = interp_sequence(best_seq, new_n)
        n_tried += 1

        t6 = min(remaining * 0.30, 100.0)
        if t6 > 10:
            a_out, _ = soft_gradient_descent(
                a_new, t6,
                beta_start=200.0, beta_end=1000.0,
                lr_start=0.2, lr_decay=0.003,
                stall_limit=300, verbose=False
            )
            check_update(a_out, f"len_{new_n}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)

    if remaining > 20:
        t7a = min(remaining * 0.35, 120.0)
        if t7a > 10:
            a_out, _ = soft_gradient_descent(
                best_seq, t7a,
                beta_start=2000.0, beta_end=10000.0,
                lr_start=0.05, lr_decay=0.001,
                stall_limit=200, verbose=True
            )
            check_update(a_out, "final_sgd")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7b = min(remaining * 0.35, 80.0)
        if t7b > 8:
            a_out, _ = subgradient_polish(
                best_seq, t7b, lr_start=0.08, verbose=False
            )
            check_update(a_out, "final_subgrad")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t7c = min(remaining * 0.40, 60.0)
        if t7c > 5:
            a_out, _ = pair_swap_descent(best_seq, t7c, verbose=False)
            check_update(a_out, "final_swap")

    # Final validation with the actual evaluator
    if best_seq is not None:
        val_seq = [float(x) for x in best_seq]
        fs = evaluate_sequence(val_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_fast(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
