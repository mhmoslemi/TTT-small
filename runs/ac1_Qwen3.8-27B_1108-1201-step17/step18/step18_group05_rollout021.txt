def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)

        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def compute_lp_direction(a, M, binding_idx, solver):
    """Solve the LP to get the FW direction."""
    n = len(a)
    K = len(binding_idx)
    
    T = np.zeros((K, n))
    for ii, k in enumerate(binding_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            T[ii, j_start:j_end] = a[ki - np.arange(j_start, j_end)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        # Use tight tolerances for accuracy
        if solver == cp.PDLP:
            prob.solve(solver=solver, verbose=False, max_iter=5000, eps_abs=1e-9, eps_rel=1e-9)
        elif solver == cp.CLARABEL:
            prob.solve(solver=solver, verbose=False, tol_gap_abs=1e-10, tol_gap_rel=1e-10,
                       tol_feas=1e-10, tol_ktr_c=1e-10, tol_ktr_r=1e-10)
        else:
            prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_frank_wolfe_v2(a0, time_limit, solver, top_K_init=300, top_K_max=4000,
                       margin_init=0.95, margin_final=0.995, verbose=True, rng_seed=777):
    """Enhanced LP-Frank-Wolfe with adaptive margin and exact line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    K = top_K_init
    iter_count = 0
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Adaptive margin
        margin = margin_init + (margin_final - margin_init) * progress
        margin_threshold = M_curr * margin
        
        # Select constraints: top-K by value + all above margin
        top_idx = np.argsort(ac)[::-1][:min(K, 2 * n - 1)]
        margin_idx = np.where(ac >= margin_threshold)[0]
        
        # Combine and deduplicate
        combined = np.unique(np.concatenate([top_idx, margin_idx]))
        # Cap total constraints
        if len(combined) > top_K_max:
            # Keep top-K_K_max by value
            combined = combined[np.argsort(ac[combined])[::-1]][:top_K_max]
        
        # Solve LP
        g_dir = compute_lp_direction(a, M_curr, combined, solver)
        
        if g_dir is None:
            K = min(K + 200, top_K_max)
            stall += 1
            continue
        
        # Exact line search
        base_obj = 2.0 * n * best_m
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr

        alphas = sorted(set([min(1.0, max(0.01, al)) for al in alphas]), reverse=True)
        
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m - 1e-16:
                best_trial_m = m_trial
                best_alpha = alpha

        a = np.maximum(a, 0.0)

        if s_a < 1e-15:
            break

        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

                pert = rng.normal(0, 0.005 * progress + 0.002, n) * best_a
                a_pert = best_a + pert
                a_pert = np.maximum(a_pert, 1e-12)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                    stall = 0

        if iter_count % 15 == 0:

            print(f"    LP-FW2 iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={len(combined)}, "

        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LP-FW2 final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def smoothed_gradient_v2(a0, time_limit, tau_schedule=None, eta_init=0.5,
                         use_momentum=True, verbose=True, rng_seed=42):
    """High-precision smoothed projected gradient with Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    if tau_schedule is None:
        tau_schedule = [50.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.2]
    
    eta = eta_init
    v = np.zeros(n)  # momentum
    beta_mom = 0.9  # Nesterov momentum coefficient
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        phase_start = time.monotonic()
        phase_limit = (time_limit - (phase_start - t0)) / (tau_schedule.index(tau) + 1)
        phase_limit = min(phase_limit, time_limit / len(tau_schedule) * 1.5)
        stall = 0
        max_stall = 200
        iter_count = 0
        v = np.zeros(n)
        a_look = a.copy()  # Nesterov lookahead
        
        while time.monotonic() - phase_start < phase_limit and stall < max_stall:
            # Nesterov step
            a_look = a + beta_mom * v
            a_look = np.maximum(a_look, 1e-15)
            a_look = a_look / a_look.sum()
            
            # Compute smoothed gradient
            ac = fast_autocorr(a_look)
            ac_max = ac.max()
            ac_shifted = (ac - ac_max) / tau
            exp_ac = np.exp(ac_shifted)
            p = exp_ac / exp_ac.sum()
            
            # Gradient: g[j] = 2 * correlate(p, a_look, mode='valid')[j]
            g = 2.0 * sp_correlate(p, a_look, mode='valid')
            
            # Project onto tangent space (simplex)
            g_tangent = g - g.mean()
            
            # Gradient step
            a_new = a_look - eta * g_tangent
            a_new = project_simplex(a_new)
            
            # Update momentum
            v = a_new - a
            a = a_new
            
            # Evaluate
            m_new = fast_autocorr(a).max()
            if m_new < best_m - 1e-14:
                best_m = m_new
                best_a = a.copy()
                stall = 0
                eta = min(eta * 1.01, 2.0)
            else:
                stall += 1
                if stall > 50:
                    eta *= 0.85
                if stall > 100:
                    # Restart with perturbation
                    pert = rng.normal(0, 0.003, n) * best_a
                    a_p = np.maximum(best_a + pert, 1e-12)
                    a = a_p / a_p.sum()
                    v = np.zeros(n)
                    a_look = a.copy()
                    stall = 50
                    eta = eta_init * 0.5
            
            iter_count += 1
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad2 tau={tau} ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    
    return best_a, best_m

    t_phase1 = min(remaining * 0.10, 150.0)
    
    if t_phase1 > 30:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3")
        inits.append(make_bimodal(n0, width=0.12))
        labels.append("bimodal")
        inits.append(make_cosine_power(n0, power=3.0))
        labels.append("cos3")
        inits.append(make_trimodal(n0))
        labels.append("trimodal")
        
        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            # Quick smoothed gradient on each init
            a_out, _ = smoothed_gradient_v2(init, t_avail,
                                             tau_schedule=[20.0, 10.0, 5.0, 3.0],
                                             eta_init=0.8, verbose=False,
                                             rng_seed=100 + idx)
            try_update(a_out, f"qs_{lbl}")
    
    # === Phase 2: Main LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW Main (n={n_best}) ===")
    t2 = min(remaining * 0.22, 380.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        top_K_init = min(400, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, verbose=True,
                                      top_K_init=top_K_init, top_K_max=top_K_max,
                                      margin_init=0.92, margin_final=0.99,
                                      rng_seed=777)
        try_update(a_out, "lp_main")
    
    # === Phase 3: High-precision smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smoothed Gradient (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_gradient_v2(a_in, t3,
                                         tau_schedule=[30.0, 20.0, 15.0, 10.0, 7.0, 5.0, 4.0, 3.0, 2.5, 2.0],
                                         eta_init=0.6, verbose=True, rng_seed=200)
        try_update(a_out, "smooth_main")
    
    # === Phase 4: LP-FW Round 2 (from best) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW R2 (n={n_best}) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, verbose=True,
                                      top_K_init=min(800, 2 * n_best - 1),
                                      top_K_max=min(6000, 2 * n_best - 1),
                                      margin_init=0.95, margin_final=0.995,
                                      rng_seed=888)
        try_update(a_out, "lp_r2")
    
    # === Phase 5: Fine smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Fine Smooth (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_gradient_v2(a_in, t5,
                                         tau_schedule=[5.0, 4.0, 3.5, 3.0, 2.5, 2.2, 2.0, 1.8, 1.5],
                                         eta_init=0.3, verbose=True, rng_seed=300)
        try_update(a_out, "smooth_fine")
    
    # === Phase 6: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord Polish (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish_v2(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")
    
    # === Phase 7: Pair swap optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Pair Swap (n={n_best}) ===")
    t7 = min(remaining * 0.08, 120.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_swap_optimize(a_in, t7, verbose=True, rng_seed=555)
        try_update(a_out, "pairswap")
    
    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration (base n={n_best}) ===")
    
    trial_lengths = []
    for delta in [500, 1000, -500, 1500, 2000, -1000, 3000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]
    
    n_try = 0
    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80 or n_try >= 3:
            break
        n_try += 1
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        
        # Quick LP-FW on new length
        t8a = min(remaining * 0.3, 150.0)
        if t8a > 20:
            use_all = (new_n <= 2000)
            top_K_init = min(300, 2 * new_n - 1)
            top_K_max = min(3000, 2 * new_n - 1)
            
            if not use_all:
                a_out, _ = lp_frank_wolfe_v2(a_long, t8a, solver, verbose=False,
                                              top_K_init=top_K_init,
                                              top_K_max=top_K_max,
                                              margin_init=0.93, margin_final=0.99,
                                              rng_seed=new_n % 1000)
            else:
                a_out, _ = lp_frank_wolfe_v2(a_long, t8a, solver, verbose=False,
                                              top_K_init=2 * new_n - 1,
                                              top_K_max=2 * new_n - 1,
                                              margin_init=0.999, margin_final=0.999,
                                              rng_seed=new_n % 1000)
            try_update(a_out, f"lp_n{new_n}")
        
        # Quick smooth
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t8b = min(remaining * 0.3, 100.0)
            if t8b > 15:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out, _ = smoothed_gradient_v2(a_in, t8b,
                                                 tau_schedule=[15.0, 10.0, 7.0, 5.0],
                                                 eta_init=0.5, verbose=False,
                                                 rng_seed=new_n % 500 + 200)
                try_update(a_out, f"smooth_n{new_n}")
    
    # === Phase 9: Final LP-FW + smooth on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final optimization (n={n_best}) ===")
    if remaining > 60:
        t9a = min(remaining * 0.4, 180.0)
        if t9a > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v2(a_in, t9a, solver, verbose=False,
                                          top_K_init=min(1000, 2 * n_best - 1),
                                          top_K_max=min(7000, 2 * n_best - 1),
                                          margin_init=0.96, margin_final=0.998,
                                          rng_seed=1234)
            try_update(a_out, "lp_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t9b = min(remaining * 0.4, 120.0)
            if t9b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = smoothed_gradient_v2(a_in, t9b,
                                                 tau_schedule=[3.0, 2.5, 2.2, 2.0, 1.8, 1.5, 1.3, 1.2],
                                                 eta_init=0.2, verbose=False, rng_seed=5678)
                try_update(a_out, "smooth_final")
    
    # === Phase 10: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final polish (n={n_best}) ===")
        t10 = min(remaining * 0.6, 80.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish_v2(a_in, t10, verbose=False, rng_seed=999)
            try_update(a_out, "coord_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_u_shape(n_fb, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
