2. **Spectral post-processing**: Periodic FFT-based smoothing

3. **Multi-start strategy**: Different initializations and lengths

4. **Local search**: Small random perturbations with acceptance

Let me also think about the length question. The score is 2*n*max_c/(sum_a)^2. If we normalize sum_a = 1, it's 2*n*max_c. For the flat sequence, max_c = 1/n, so score = 2. For better sequences, we want max_c < 1/n.

The question is: does a longer sequence help? For a fixed "shape" of the normalized sequence, if we go from n to 2n:
- The new sequence is a_interp with sum still 1
- max_c scales roughly as 1/n (since each a_i is halved, products are quartered, but there are 2x more terms, so c scales as 1/2)
- So max_c_new ≈ max_c_old / 2
- Score_new = 2*2n*max_c/2 = 2n*max_c = Score_old

So the score is roughly independent of n for a fixed shape! This means the length doesn't matter much - what matters is the shape of the normalized sequence.

Wait, that's not quite right. If we interpolate a sequence of length n to length 2n, the autocorrelation changes in a non-trivial way due to the interpolation. Let me not worry about this and just try different lengths.

Actually, re-examining: the score is 2n * max(conv(a,a)) / (sum a)^2. If sum a = 1, it's 2n * max(conv(a,a)). 

For a flat sequence a_i = 1/n: conv(a,a)_k = (n-|k|)/n^2 for |k| < n. Max is 1/n at k=0. Score = 2n * 1/n = 2.

Now if we take the same "shape" (all equal) but with length 2n: a_i = 1/(2n). Max conv = 1/(2n). Score = 2*(2n)*(1/(2n)) = 2. Same!

So the score is shape-dependent, not length-dependent (for the flat case at least). This means we should focus on finding the best shape, and the length is less important.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent via log-sum-exp with Adam optimizer
2. Multiplicative (exponential weights) updates for simplex constraint
3. Adaptive temperature annealing
4. Spectral post-processing in frequency domain
5. Multi-start with diverse initializations and lengths
6. Simulated annealing for escape from local minima
"""

import numpy as np
import time

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_gradient(a, c, beta):
    """Compute gradient of 2n * LSE(beta * c) w.r.t. a.
    
    Returns grad of length n.
    """
    n = len(a)
    # LSE weights
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()
    # grad_j = 2 * sum_m w[j+m] * a[m] for m=0..n-1
    # This is 2 * correlate(w, a, 'valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, w

def norm_simplex(a):
    """Normalize to probability simplex."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lse_smooth_max(c, beta):
    """Compute smooth approximation of max(c) using log-sum-exp."""
    c_max = c.max()
    return c_max + np.log(np.exp(beta * (c - c_max)).sum()) / beta

def obj_from_c(c, n):
    """Compute objective from convolution."""
    return 2.0 * n * c.max()

class AdamOptimizer:
    def __init__(self, dim, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
    
    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return -self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

def smooth_gradient_descent(a0, time_limit, beta_init=50.0, beta_end=500.0, 
                             lr_init=0.01, lr_end=0.0001, verbose=False):
    """Core optimizer: smooth gradient descent with multiplicative updates."""
    n = len(a0)
    a = norm_simplex(a0)
    f = np.log(a + 1e-30)  # log parameters
    
    best_a = a.copy()
    best_c = fast_conv(a)
    best_score = 2.0 * n * best_c.max()
    
    t0 = time.monotonic()
    total_iters = 50000
    adam = AdamOptimizer(n, lr=lr_init)
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing
        progress = min(1.0, t / total_iters)
        beta = beta_init * (beta_end / beta_init) ** progress
        lr = lr_init * (lr_end / lr_init) ** progress
        
        # Compute convolution and gradient
        c = fast_conv(a)
        grad, w = fast_gradient(a, c, beta)
        
        # Convert to log-space gradient (natural gradient)
        mean_grad = np.dot(grad, a)
        log_grad = a * (grad - mean_grad)
        
        # Adam step in log-space
        update = adam.step(log_grad)
        f = f + update
        
        # Convert back to simplex
        f_max = f.max()
        a = np.exp(f - f_max)
        s = a.sum()
        if s < 1e-15:
            a = norm_simplex(a0)
            f = np.log(a + 1e-30)
        else:
            a = a / s
        
        # Track best
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_score / (2.0 * n) - 1e-15:
            best_score = 2.0 * n * M_new
            best_a = a.copy()
        
        # Periodic spectral smoothing
        if t % 2000 == 0:
            a = spectral_step(a, strength=0.1)
            c_check = fast_conv(a)
            if c_check.max() < best_score / (2.0 * n):
                best_score = 2.0 * n * c_check.max()
                best_a = a.copy()
                f = np.log(best_a + 1e-30)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): score={best_score:.6f}, beta={beta:.1f}")
    
    return best_a, best_score

def spectral_step(a, strength=0.1):
    """One step of spectral flattening."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    return norm_simplex((1.0 - strength) * a + strength * a_new)

def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    step = 0.2
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_new = spectral_step(a, strength=step)
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            step = min(0.4, step * 1.02)
        else:
            step = max(0.03, step * 0.8)
            if step < 0.03:
                break
    
    return best_a, 2.0 * n * best_m

def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def annealing_search(a0, time_limit, n_trials=100, verbose=False):
    """Simulated annealing with random perturbations."""
    n = len(a0)
    a = norm_simplex(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    T_init = best_m * 0.1
    T_end = best_m * 0.001
    
    for t in range(1, n_trials + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, t / n_trials)
        T = T_init * (T_end / T_init) ** progress
        
        # Random perturbation
        perturbation_scale = 0.01 * (1.0 - progress) + 0.001
        a_trial = a + rng.normal(0, perturbation_scale, n)
        a_trial = norm_simplex(a_trial)
        
        c_trial = fast_conv(a_trial)
        M_trial = c_trial.max()
        
        delta = M_trial - best_m
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a_trial.copy()
    
    return best_a, 2.0 * n * best_m
```

Wait, I need to check: the evaluate_sequence function takes a list, and it clips values to [0, 1000]. Since I'm working with normalized sequences (sum=1), all values are in [0, 1], so that's fine.

Also, I'm using `np.correlate` which for n=4000 would be correlating an array of length 7999 with one of length 4000. That's O(n*m) = O(4000*4000) = 16M operations. In numpy, this should take maybe 50-100ms. That might be too slow for 50000 iterations (50000 * 0.07s = 3500s). 

I should use the FFT-based approach instead:
- FFT of w (padded to size S): O(S log S)
- FFT of a (padded to size S): O(S log S)  
- Pointwise multiply: O(S)
- IFFT: O(S log S)

For S = 16384 (next power of 2 >= 3*4000-2 = 11998), each FFT is about 16384 * 14 ≈ 229K operations. Total per gradient step: ~1M operations. Much faster.

Let me rewrite the gradient computation using FFT:

```python
def fast_gradient_fft(a, c, beta):
    """Compute gradient using FFT for efficiency."""
    n = len(a)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()
    
    # FFT-based correlation: grad_j = 2 * sum_m w[j+m] * a[m]
    # This is correlate(w, a, 'valid')
    # Via FFT: pad to size >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    S = 1 << (3 * n - 2 - 1).bit_length()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent via log-sum-exp with FFT-based gradient computation
2. Adam optimizer with multiplicative (exponential weights) updates in log-space
3. Adaptive temperature annealing for the smooth max approximation
4. Spectral post-processing in frequency domain (amplitude compression)
5. Multi-start with diverse initializations and lengths
6. Simulated annealing and coordinate descent for final polish

import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_gradient(a, c, beta):
    """Compute gradient of LSE(beta * c) w.r.t. a using FFT-based correlation.
    
    grad_j = 2 * sum_{m=0}^{n-1} w_{j+m} * a[m] where w = softmax(beta * c)
    """
    n = len(a)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()
    
    # FFT-based cross-correlation for efficiency
    S = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, S)
    A = np.fft.rfft(a, S)
    xcorr = np.fft.irfft(W * np.conj(A), S)
    grad = 2.0 * xcorr[:n]
    return grad, w


class AdamOptimizer:
    """Adam optimizer."""
    def __init__(self, dim, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps

    def set_lr(self, lr):
        self.lr = lr

    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return -self.lr * m_hat / (np.sqrt(v_hat) + self.eps)


def smooth_gradient_descent(a0, time_limit, beta_init=50.0, beta_end=500.0,
                            lr_init=0.01, lr_end=0.0001, verbose=False):
    """Smooth gradient descent with Adam and multiplicative updates in log-space."""
    n = len(a0)
    a = norm_simplex(a0)
    f = np.log(a + 1e-30)

    best_a = a.copy()
    best_c = fast_conv(a)
    best_m = best_c.max()

    t0 = time.monotonic()
    adam = AdamOptimizer(n, lr=lr_init)

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Annealing schedule
        progress = min(1.0, t / 80000.0)
        beta = beta_init * (beta_end / beta_init) ** progress
        lr = lr_init * (lr_end / lr_init) ** progress
        adam.set_lr(lr)

        # Compute convolution and gradient
        c = fast_conv(a)
        grad, w = compute_gradient(a, c, beta)

        # Natural gradient in log-space: dL/d f_k = a_k * (grad_k - <grad, a>)
        mean_grad = np.dot(grad, a)
        log_grad = a * (grad - mean_grad)

        # Adam step in log-space
        update = adam.step(log_grad)
        f = f + update

        # Convert back to simplex
        f_max = f.max()
        a_new = np.exp(f - f_max)
        s = a_new.sum()
        if s < 1e-15:
            a = norm_simplex(a0)
            f = np.log(a + 1e-30)
        else:
            a = a_new / s

        # Track best
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()

        # Periodic spectral smoothing
        if t % 3000 == 0:
            a = spectral_step(a, strength=0.08)
            c_check = fast_conv(a)
            if c_check.max() < best_m:
                best_m = c_check.max()
                best_a = a.copy()
            f = np.log(a + 1e-30)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD t={t} ({elapsed:.0f}s): score={2*n*best_m:.6f}, beta={beta:.1f}")

    return best_a, 2.0 * n * best_m


def spectral_step(a, strength=0.1):
    """One step of spectral amplitude compression."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a

    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

    return norm_simplex((1.0 - strength * 0.5) * a + strength * 0.5 * a_new)


def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening with adaptive step size."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    step = 0.25
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        a_new = spectral_step(a, strength=step)
        c_new = fast_conv(a_new)
        M_new = c_new.max()

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            step = min(0.4, step * 1.03)
        else:
            step = max(0.02, step * 0.75)
            if step < 0.02:
                break

    return best_a, 2.0 * n * best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def annealing_search(a0, time_limit, n_trials=200, verbose=False):
    """Simulated annealing with random perturbations."""
    n = len(a0)
    a = norm_simplex(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    T_init = best_m * 0.05
    T_end = best_m * 0.001

    for t in range(1, n_trials + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, t / n_trials)
        T = T_init * (T_end / T_init) ** progress

        # Random perturbation with decreasing scale
        scale = 0.02 * (1.0 - 0.8 * progress) + 0.002
        a_trial = a + rng.normal(0, scale, n)
        a_trial = norm_simplex(a_trial)

        c_trial = fast_conv(a_trial)

        delta = M_trial - best_m

            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a_trial.copy()

def coordinate_polish(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions to reduce peak."""
    n = len(a0)
    a = norm_simplex(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        
        # Find top peaks
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try moving to a low-contribution position
            for attempt in range(3):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.02, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.02, 0.06)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, 2.0 * n * best_m

    inits.append(make_init_dual_peak(n0, 0.2, 0.8, 0.12))
    labels.append("dualpeak")
    inits.append(make_init_sinc(n0, bw=7.0))
    labels.append("sinc7")
    inits.append(make_init_waveshape(n0, freq1=3, freq2=7, phase_shift=0.5))
    labels.append("wave37")
    inits.append(make_init_waveshape(n0, freq1=5, freq2=11, phase_shift=1.0))
    labels.append("wave511")

    for seed in [7, 42]:
        inits.append(make_init_random_smooth(n0, seed=seed))
        labels.append(f"rand{seed}")
        inits.append(make_init_gaussian_mix(n0, seed=seed, n_components=4))
        labels.append(f"gauss{seed}")

    # Deduplicate
    seen = set()
    unique_inits = []
    unique_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0 // 15)], 4))
        if sig not in seen:
            seen.add(sig)
            unique_inits.append(init)
            unique_labels.append(lbl)

    print(f"  Running on {len(unique_inits)} initializations...")
    t_per = (remaining * 0.42) / len(unique_inits)

    for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
        elapsed_so_far = time.monotonic() - t_global
        remaining_now = time_budget - 30 - elapsed_so_far
        if remaining_now < 10:
            break
        t_avail = min(t_per, remaining_now)
        if t_avail < 8:
            break
        print(f"    GD '{lbl}' (budget={t_avail:.0f}s)...")
        
        # Use different beta schedules for different inits
        beta_i = 40.0 + 10.0 * (idx % 3)
        beta_e = 300.0 + 100.0 * (idx % 4)
        a_out, sc = smooth_gradient_descent(init, t_avail,
                                             beta_init=beta_i, beta_end=beta_e,
                                             lr_init=0.008, lr_end=0.0001)
        try_update(a_out, f"gd_{lbl}")

        # Quick spectral polish
        remaining_now = time_budget - 30 - (time.monotonic() - t_global)
        if remaining_now > 8:
            a_sp, sc2 = spectral_optimize(a_out, min(8.0, remaining_now * 0.25))
            try_update(a_sp, f"spec_{lbl}")

    # === Phase 2: Intensive optimization on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Intensive GD (n={n_best}) ===")
    t2 = min(remaining * 0.22, 280.0)
    if t2 > 30:
        a_in = norm_simplex(best_seq)
        a_out, sc = smooth_gradient_descent(a_in, t2,
                                             beta_init=30, beta_end=800,
                                             lr_init=0.005, lr_end=0.00003)
        try_update(a_out, "gd_intensive")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_sp, sc2 = spectral_optimize(a_out, min(25.0, remaining * 0.3))
            try_update(a_sp, "spec_intensive")

    # === Phase 3: Annealing + coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Annealing + CD (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 120.0)
    if t3 > 15:
        a_in = norm_simplex(best_seq)
        a_out, sc = annealing_search(a_in, t3 * 0.5, n_trials=300)
        try_update(a_out, "anneal")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_cd, sc2 = coordinate_polish(a_out, min(t3 * 0.5, remaining * 0.5))
            try_update(a_cd, "cd_after_anneal")

    # === Phase 4: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length variation ===")

    for new_n in [n_best + 800, n_best + 1500, n_best - 800]:
        if new_n < 2000 or new_n > 9000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")

        t4 = min(remaining * 0.28, 150.0)
        if t4 > 15:
            a_out, sc = smooth_gradient_descent(a_interp, t4,
                                                 beta_init=30, beta_end=400,
                                                 lr_init=0.006, lr_end=0.0001)
            try_update(a_out, f"gd_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_sp, sc2 = spectral_optimize(a_out, min(15.0, remaining * 0.3))
                try_update(a_sp, f"spec_n{new_n}")

    # === Phase 5: Final high-beta polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: High-beta polish (n={n_best}) ===")
    t5 = min(remaining * 0.30, 250.0)
    if t5 > 20:
        a_in = norm_simplex(best_seq)
        a_out, sc = smooth_gradient_descent(a_in, t5,
                                             beta_init=300, beta_end=3000,
                                             lr_init=0.002, lr_end=0.000005)
        try_update(a_out, "high_beta_gd")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_sp, sc2 = spectral_optimize(a_out, min(25.0, remaining * 0.4))
            try_update(a_sp, "high_beta_spec")

    # === Phase 6: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        print(f"\n=== Phase 6: Final CD (n={len(best_seq)}) ===")
        t6 = min(remaining * 0.6, 60.0)
        if t6 > 5:
            a_in = norm_simplex(best_seq)
            a_out, sc = coordinate_polish(a_in, t6)
            try_update(a_out, "final_cd")

    # === Phase 7: One more spectral + short GD cycle ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 7: Final spectral+GD cycle ===")
        a_in = norm_simplex(best_seq)
        a_sp, _ = spectral_optimize(a_in, min(15.0, remaining * 0.3))
        try_update(a_sp, "final_spec7")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, sc = smooth_gradient_descent(a_sp, min(remaining * 0.5, 30.0),
                                                 beta_init=500, beta_end=5000,
                                                 lr_init=0.001, lr_end=0.000001)
            try_update(a_out, "final_gd7")

    # Final validation
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
