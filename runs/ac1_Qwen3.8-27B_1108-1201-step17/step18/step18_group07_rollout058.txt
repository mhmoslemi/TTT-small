import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def log_sum_exp_smooth(a, beta):
    """Compute smooth approximation of max(c) using log-sum-exp."""
    c = fast_conv(a)
    max_c = c.max()
    if max_c < 1e-30:
        return 0.0
    # log-sum-exp / beta approximates max
    shifted = beta * (c - max_c)
    shifted = np.maximum(shifted, -100.0)
    lse = max_c + np.log(np.sum(np.exp(shifted))) / beta
    return lse

def log_sum_exp_gradient(a, beta, top_k=None):
    """Compute subgradient of log-sum-exp smoothing of max(conv(a,a))."""
    c = fast_conv(a)
    max_c = c.max()
    if max_c < 1e-30:
        return np.zeros_like(a), c
    
    if top_k is not None:
        top_idx = np.argsort(c)[::-1][:top_k]
        vals = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        vals = c
    
    shifted = beta * (vals - max_c)
    shifted = np.maximum(shifted, -100.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum
    
    grad = np.zeros_like(a)
    for wi, k in zip(w, top_idx):
        if wi < 1e-10:
            continue
        k_int = int(k)
        i_start = max(0, k_int - len(a) + 1)
        i_end = min(len(a), k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad, c

def adam_log_optimize(a0, time_limit, beta_start=50.0, beta_end=5.0, lr=0.01, verbose=False):
    """Adam optimizer in log-space with annealed log-sum-exp smoothing."""
    n = len(a0)
    b = np.log(np.maximum(a0, 1e-15))  # log-parameterization
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = norm_simplex(a0)
    best_m = fast_conv(best_a).max()
    t0 = time.monotonic()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing
        progress = min(1.0, t / 30000.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        cur_lr = lr * (1.0 + progress) ** (-0.5)  # decay
        
        # Current a from b
        b_max = b.max()
        a = np.exp(b - b_max)
        a = a / a.sum()
        
        # Compute gradient
        grad, c = log_sum_exp_gradient(a, beta)
        
        # Convert to log-space gradient
        # da/db_i = a_i * (delta_ij - a_j)
        # dJ/db_i = sum_j (da_j/db_i) * grad_j = a_i * (grad_i - sum_j a_j * grad_j)
        log_grad = a * (grad - np.sum(a * grad))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * log_grad
        v = beta2 * v + (1 - beta2) * log_grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        b = b - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Track best
        if t % 50 == 0:
            a_check = np.exp(b - b.max())
            a_check = a_check / a_check.sum()
            m_check = fast_conv(a_check).max()
            if m_check < best_m - 1e-15:
                best_m = m_check
                best_a = a_check.copy()
        
        if verbose and t % 5000 == 0:
            print(f"    Adam t={t}: J={2*n*best_m:.6f}, beta={beta:.1f}")
    
    return best_a, best_m

    n = len(a)

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > 60:  # time cap
            break
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        c = fast_conv(a)
        k_max = int(np.argmax(c))
        
        # Compute the "peak shape" in frequency domain
        f = np.arange(len(mag)) / fft_size
        # cos(2*pi*f*k_max) tells us which frequencies contribute positively to the peak
        peak_shape = np.cos(2 * np.pi * f * k_max)
        
        # Reduce |A| where peak_shape > 0, increase where < 0
        # This directly reduces c[k_max]
        normalization = alpha * (1.0 / (mag.mean() + 1e-30))
        mag_new = mag * (1.0 - normalization * peak_shape)
        mag_new = np.maximum(mag_new, 1e-10)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            alpha = min(0.3, alpha * 1.02)
        else:
            alpha = max(0.01, alpha * 0.8)
            a = norm_simplex(0.5 * a + 0.5 * best_a)
    
    return best_a, best_m

def spectral_flatten(a, time_limit, gamma_start=0.3, gamma_end=0.5, verbose=False):
    """Iterative spectral flattening with adaptive gamma."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    gamma = gamma_start
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: mag_new = mag_mean * (mag/mag_mean)^gamma
        mag_mean = mag.mean()
        ratio = mag / (mag_mean + 1e-30)
        mag_new = mag_mean * ratio ** gamma
        mag_new = np.maximum(mag_new, 1e-10)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            # Increase gamma (more aggressive flattening)
            gamma = min(gamma_end + 0.1, gamma + 0.01)
        else:
            gamma = max(gamma_start * 0.5, gamma * 0.9)
            a = norm_simplex(0.7 * a + 0.3 * best_a)
    
    return best_a, best_m

def make_init_cosine(n, mode=1):

    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-space Adam optimization with full log-sum-exp smoothing and beta annealing
2. Spectral peak reduction (direct frequency-domain peak targeting - new method)

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_lse_gradient(a, beta, max_weight_threshold=1e-12):
    """Compute the gradient of 2n*(1/beta)*log(sum exp(beta*c_k)) w.r.t. b (log-space params).
    
    Returns (objective_value, gradient_in_log_space, conv_result)
    """
    n = len(a)
    c = fast_conv(a)
    max_c = c.max()
    if max_c < 1e-30:
        return 0.0, np.zeros(n), c
    
    shifted = beta * (c - max_c)
    shifted = np.maximum(shifted, -100.0)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = exp_shifted / sum_exp
    
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)
    
    # grad_a[i] = 2n * sum_k w[k] * a[k-i] for valid k-i
    grad_a = np.zeros(n)
    # Only iterate over indices where w is significant
    significant = w > max_weight_threshold
    sig_idx = np.where(significant)[0]
    for k in sig_idx:
        k_int = int(k)
        wi = w[k_int]
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad_a[idx] += wi * a[j_idx]
    grad_a *= 2.0 * n
    
    # Convert to log-space: dB_i = a_i * (grad_a_i - sum(a * grad_a))
    mean_grad = np.sum(a * grad_a)
    log_grad = a * (grad_a - mean_grad)
    
    return obj, log_grad, c


def adam_log_optimize(a0, time_limit, beta_start=60.0, beta_end=3.0,
                      lr=0.03, anneal_iters=20000, verbose=False):
    """Adam optimizer in log-space with annealed log-sum-exp smoothing."""
    n = len(a0)
    a_init = norm_simplex(a0)
    b = np.log(np.maximum(a_init, 1e-15))
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a_init.copy()
    best_m = fast_conv(best_a).max()
    t0 = time.monotonic()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing schedule
        progress = min(1.0, t / float(anneal_iters))
        beta = beta_start * (beta_end / beta_start) ** progress
        cur_lr = lr * (1.0 + 0.5 * progress)
        
        # Current a from b
        b_max = b.max()
        a = np.exp(b - b_max)
        a = a / a.sum()
        
        # Compute objective and gradient
        obj, log_grad, c = compute_lse_gradient(a, beta)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * log_grad
        v = beta2 * v + (1.0 - beta2) * (log_grad * log_grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        b = b - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Track best (check every 20 iterations to save time)
        if t % 20 == 0:
            a_check = np.exp(b - b.max())
            a_check = a_check / a_check.sum()
            c_check = fast_conv(a_check)
            m_check = c_check.max()
            if m_check < best_m - 1e-15:
                best_m = m_check
                best_a = a_check.copy()
        
        if verbose and t % 10000 == 0:
            print(f"      Adam t={t}: true_max={2*n*best_m:.6f}, lse_obj={obj:.6f}, beta={beta:.1f}")
    
    return best_a, best_m


def spectral_peak_reduce(a, time_limit, alpha=0.05, verbose=False):
    """Reduce the peak of the convolution by modifying the power spectrum.
    
    Key idea: the peak c[k_max] = (1/N) sum_f |A(f)|^2 * exp(2*pi*i*f*k_max/N).
    To reduce it, decrease |A(f)| where Re(exp(2*pi*i*f*k_max/N)) > 0.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    f = np.arange(len(np.fft.rfft(a, fft_size))) / fft_size
    
    stall = 0
    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        k_max = int(np.argmax(c))
        
        # Peak shape in frequency domain
        peak_shape = np.cos(2.0 * np.pi * f * k_max)
        
        # Also consider second and third peaks for robustness
        sorted_idx = np.argsort(c)[::-1]
        for k_extra in sorted_idx[1:min(4, len(c))]:
            if c[k_extra] > 0.8 * c[k_max]:
                peak_shape += 0.3 * np.cos(2.0 * np.pi * f * int(k_extra))
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Modify magnitude: reduce where peak_shape > 0, increase where < 0
        mag_mean = mag.mean()
        scale = alpha / (mag_mean + 1e-30)
        mag_new = mag * (1.0 - scale * peak_shape)
        mag_new = np.maximum(mag_new, 1e-10)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new.copy()
            alpha = min(0.2, alpha * 1.01)
            stall = 0
        else:
            alpha = max(0.005, alpha * 0.85)
            stall += 1
            # Blend back toward best
            a = norm_simplex(0.6 * a + 0.4 * best_a)
            if stall > 100:
                break
    
    return best_a, best_m


def spectral_flatten(a, time_limit, gamma_start=0.15, gamma_end=0.7, verbose=False):
    """Iterative spectral flattening: compress dynamic range of |A(f)|."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    gamma = gamma_start
    stall = 0
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        mag_mean = mag.mean()
        ratio = mag / (mag_mean + 1e-30)
        mag_new = mag_mean * ratio ** gamma
        mag_new = np.maximum(mag_new, 1e-10)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new.copy()
            gamma = min(gamma_end, gamma * 1.02)
            stall = 0
        else:
            gamma = max(0.05, gamma * 0.95)
            stall += 1
            a = norm_simplex(0.5 * a + 0.5 * best_a)
            if stall > 50:
                break
    
    return best_a, best_m


def lbfgs_smooth_optimize(a0, time_limit, beta=30.0, max_iter=500, verbose=False):
    """Use scipy L-BFGS-B to optimize the smooth log-sum-exp approximation."""
    n = len(a0)
    a_init = norm_simplex(a0)
    b0 = np.log(np.maximum(a_init, 1e-15))
    
    best_a = a_init.copy()
    best_m = fast_conv(best_a).max()
    t0 = time.monotonic()
    
    def fun_and_grad(b):
        if time.monotonic() - t0 > time_limit:
            return 0.0, np.zeros(n)
        b_max = b.max()
        a = np.exp(b - b_max)
        a = a / a.sum()
        obj, log_grad, c = compute_lse_gradient(a, beta)
        # Update best
        nonlocal_best_m = fast_conv(a).max()
        return obj, log_grad
    
    # We can't use nonlocal, so we'll do it differently
    # Just run L-BFGS and check result at the end
    
    result = scipy_minimize(
        fun_and_grad,
        b0,
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'gtol': 1e-8, 'maxfun': 2000}
    )
    
    if result.success or len(result.x) == n:
        b_opt = result.x
        a_opt = np.exp(b_opt - b_opt.max())
        a_opt = a_opt / a_opt.sum()
        m_opt = fast_conv(a_opt).max()
        if m_opt < best_m:
            best_a = a_opt
            best_m = m_opt
    
    return best_a, best_m


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = (np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) +
         np.exp(-((x - pos2) ** 2) / (2 * width ** 2)) + 0.03)
    return norm_simplex(a)


def make_init_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_random_smooth(n, seed=42, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def combined_polish(a, time_limit):
    """Apply a sequence of polishing steps: Adam -> spectral_peak -> spectral_flatten -> Adam."""
    n = len(a)
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    def check_update(a_try, label):
        nonlocal best_a, best_m
        m_try = fast_conv(a_try).max()
        if m_try < best_m - 1e-15:
            best_m = m_try
            best_a = a_try.copy()
    
    # Step 1: Short Adam
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 20:
        a1, _ = adam_log_optimize(a, min(remaining * 0.3, 120.0),
                                  beta_start=40.0, beta_end=5.0, lr=0.03)
        check_update(a1, "adam1")
        a = best_a
    
    # Step 2: Spectral peak reduction
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 15:
        a2, _ = spectral_peak_reduce(a, min(remaining * 0.25, 90.0), alpha=0.04)
        check_update(a2, "spec_peak")
        a = best_a
    
    # Step 3: Spectral flatten
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 15:
        a3, _ = spectral_flatten(a, min(remaining * 0.25, 90.0),
                                  gamma_start=0.1, gamma_end=0.5)
        check_update(a3, "spec_flat")
        a = best_a
    
    # Step 4: Final Adam
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 15:
        a4, _ = adam_log_optimize(a, remaining * 0.5,
                                  beta_start=20.0, beta_end=2.0, lr=0.02)
        check_update(a4, "adam_final")
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator. Returns the best sequence found."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4200
        best_seq = make_init_bowl(n_base, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}, budget={time_budget:.0f}s")
    
    # === Phase 1: Adam log-space optimization with diverse inits ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam Log-Space (n={n0}) ===")
    t1 = min(remaining * 0.30, 500.0)
    
    if t1 > 60:
        inits = []
        labels = []
        
        # Always start from parent
        inits.append(norm_simplex(best_seq))
        labels.append("parent")
        
        # Diverse inits
        inits.append(make_init_bowl(n0, 2.5))
        labels.append("bowl2.5")
        
        inits.append(make_init_bowl(n0, 3.5))
        labels.append("bowl3.5")
        
        inits.append(make_init_dual_peak(n0, 0.15, 0.85, 0.18))
        labels.append("dualpeak")
        
        inits.append(make_init_cosine(n0, 1))
        labels.append("cos1")
        
        inits.append(make_init_random_smooth(n0, seed=77, n_freq=10))
        labels.append("rand_smooth")
        
        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 20)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        
        t_per = t1 / len(unique_inits)
        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 10:
                break
            print(f"  Adam '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = adam_log_optimize(init, t_avail,
                                         beta_start=80.0, beta_end=4.0, lr=0.04)
            try_update(a_out, f"adam_{lbl}")
            
            # Spectral peak reduction polish
            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now > 12:
                a_sp, _ = spectral_peak_reduce(a_out, min(20.0, remaining_now * 0.3),
                                               alpha=0.04)
                try_update(a_sp, f"spk_{lbl}")
    
    # === Phase 2: Intensive Adam on current best with different hyperparams ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Intensive Adam (n={n_best}, score={best_score:.6f}) ===")
    t2 = min(remaining * 0.25, 400.0)
    
    if t2 > 30:
        a_in = norm_simplex(best_seq)
        # Run 3 passes with different hyperparameters
        hyperparams = [
            (50.0, 3.0, 0.03),
            (100.0, 5.0, 0.05),
            (20.0, 2.0, 0.015),
        ]
        t_per2 = t2 / len(hyperparams)
        for idx, (bs, be, lr) in enumerate(hyperparams):
            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now < 15:
                break
            t_avail = min(t_per2, remaining_now)
            if t_avail < 10:
                break
            print(f"  Adam hp{idx} (beta={bs}->{be}, lr={lr}, budget={t_avail:.0f}s)...")
            a_out, _ = adam_log_optimize(a_in, t_avail,
                                         beta_start=bs, beta_end=be, lr=lr)
            try_update(a_out, f"adam_hp{idx}")
            a_in = norm_simplex(best_seq)
    
    # === Phase 3: Spectral optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Spectral Methods (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    
    if t3 > 20:
        a_in = norm_simplex(best_seq)
        
        # Peak reduction
        t3a = min(t3 * 0.5, remaining - 30)
        if t3a > 10:
            a_out, _ = spectral_peak_reduce(a_in, t3a, alpha=0.03)
            try_update(a_out, "spec_peak_main")
        
        # Flatten
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.3, 150.0)
        if t3b > 10:
            a_in2 = norm_simplex(best_seq)
            a_out2, _ = spectral_flatten(a_in2, t3b, gamma_start=0.1, gamma_end=0.5)
            try_update(a_out2, "spec_flat_main")
            
            # Peak reduce after flatten
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_out3, _ = spectral_peak_reduce(a_out2, min(30.0, remaining * 0.3),
                                                 alpha=0.02)
                try_update(a_out3, "spec_peak_after_flat")
    
    # === Phase 4: L-BFGS on smooth approximation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: L-BFGS (n={n_best}) ===")
    t4 = min(remaining * 0.10, 120.0)
    
    if t4 > 15:
        a_in = norm_simplex(best_seq)
        # Use moderate beta for L-BFGS (too high = too peaked, too low = inaccurate)
        a_out, _ = lbfgs_smooth_optimize(a_in, t4, beta=25.0, max_iter=300)
        try_update(a_out, "lbfgs_b25")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out2, _ = lbfgs_smooth_optimize(a_in, min(60.0, remaining * 0.5),
                                              beta=50.0, max_iter=200)
            try_update(a_out2, "lbfgs_b50")
    
    # === Phase 5: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length Variation ===")
    
    # Try a few different lengths
    length_options = []
    for delta in [-300, -100, 100, 300, 500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            length_options.append(new_n)
    
    for new_n in length_options[:4]:  # Limit to 4 length changes
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}...")
        a_resized = interpolate_sequence(best_seq, new_n)
        try_update(a_resized, f"resize_n{new_n}")
        
        t5 = min(remaining * 0.20, 120.0)
        if t5 > 15:
            a_out, _ = adam_log_optimize(a_resized, t5,
                                         beta_start=40.0, beta_end=4.0, lr=0.03)
            try_update(a_out, f"adam_n{new_n}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_sp, _ = spectral_peak_reduce(a_out, min(20.0, remaining * 0.3),
                                               alpha=0.03)
                try_update(a_sp, f"spk_n{new_n}")
    
    # === Phase 6: Final combined polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Polish (n={n_best}) ===")
    t6 = min(remaining * 0.40, 300.0)
    
    if t6 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = combined_polish(a_in, t6)
        try_update(a_out, "final_polish")
        
        # One more Adam pass on the result
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in2 = norm_simplex(best_seq)
            a_out2, _ = adam_log_optimize(a_in2, remaining * 0.6,
                                          beta_start=30.0, beta_end=2.0, lr=0.02)
            try_update(a_out2, "final_adam")
            
            # Spectral peak on the result
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_sp, _ = spectral_peak_reduce(a_out2, min(30.0, remaining * 0.5),
                                               alpha=0.02)
                try_update(a_sp, "final_spk")
    
    # === Phase 7: Last resort - small perturbations + quick eval ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 7: Perturbation Search ===")
        rng = np.random.default_rng(2024)
        n_best = len(best_seq)
        a_base = norm_simplex(best_seq)
        t7_start = time.monotonic()
        
        n_tries = 0
        while time.monotonic() - t7_start < remaining - 5:
            n_tries += 1
            # Random smooth perturbation
            sigma = rng.uniform(0.005, 0.02)
            pert = rng.normal(0, sigma, n_best)
            # Make it smooth (low-pass filter)
            fft_pert = np.fft.rfft(pert)
            cutoff = int(n_best * 0.1)
            fft_pert[cutoff:] *= 0.1
            pert = np.fft.irfft(fft_pert, n_best)
            
            a_try = norm_simplex(a_base + pert)
            sc = evaluate_safe(a_try)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_try.copy()
                a_base = a_try.copy()
                print(f"  Perturb {n_tries}: {sc:.6f}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_init_bowl(n_fb, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
