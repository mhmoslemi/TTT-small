import numpy as np
import time

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    """Normalize to probability simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project vector v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort-based projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n + 1) > (cssv - 1) / 1.0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

3. Spectral domain flattening (iterative amplitude compression)

import numpy as np
import time

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project onto probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1 - u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.argmax(~cond)
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def exact_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def smooth_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv(a,a)))."""
    n = len(a)
    c = fast_conv(a)
    max_c = c.max()
    exp_c = np.exp(beta * (c - max_c))
    s = exp_c.sum()
    w = exp_c / s
    
    # Cross-correlation (w ⋆ a) via FFT
    fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    
    return (2.0 / beta) * corr[:n]

            break
        
        # Current power spectrum
        A = np.fft.rfft(a, fft_size)
        mag2 = np.abs(A) ** 2
        
        # We want to flatten the autocorrelation, which means
        # concentrating the power spectrum (since flat autocorr <-> concentrated spectrum)
        # Actually: flat autocorrelation means the DFT of autocorrelation is flat
        # DFT of autocorrelation = |A|^2
        # So flat autocorrelation <-> flat |A|^2
        # We want to make |A|^2 as flat as possible
        
        # Compress the dynamic range of |A|^2
        mag2_mean = mag2.mean()
        if mag2_mean < 1e-30:
            break
        ratio = mag2 / mag2_mean
        # Compress: new_mag2 = mag2_mean * ratio^gamma
        # gamma < 1 compresses (makes flatter)
        gamma = 0.7
        new_mag2 = mag2_mean * np.maximum(ratio, 0) ** gamma
        
        # Reconstruct: keep phase, adjust magnitude
        phase = np.angle(A)
        new_mag = np.sqrt(np.maximum(new_mag2, 0))
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            break
        
        sc = exact_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

            for alpha in [0.3, 0.5, 0.7]:

                if sc_blend < best_score:

                    best_a = a_blend.copy()
