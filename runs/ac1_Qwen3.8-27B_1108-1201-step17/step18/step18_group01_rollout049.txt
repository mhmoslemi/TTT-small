"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B optimization with log-sum-exp smoothed objective
2. Softmax reparameterization for non-negativity
3. Multi-tau, multi-m sweeps for robustness
4. Coordinate descent polish on exact objective
5. Multi-length exploration via interpolation
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def safe_evaluate(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except:
        return float('inf')

def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s

def smooth_obj_and_grad(z, tau, m, n):
    """Compute smoothed max autocorrelation and its gradient w.r.t. z."""
    a = softmax(z)
    
    ac = fft_autocorr(a)
    
    # Get top m indices
    if m >= len(ac):
        top_idx = np.arange(len(ac))
        top_vals = ac.copy()
    else:
        top_idx = np.argpartition(ac, -m)[-m:]
        top_vals = ac[top_idx]
    
    # Log-sum-exp
    max_v = top_vals.max()
    shifted = tau * (top_vals - max_v)
    exp_shifted = np.exp(shifted)
    s = exp_shifted.sum()
    lse = max_v + np.log(s) / tau
    
    # Weights
    w = exp_shifted / s
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for j in range(len(top_idx)):
        k = int(top_idx[j])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad_a[idx] += 2.0 * w[j] * a[k - idx]
    
    # Transform to z-gradient (natural gradient for softmax)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    
    val = 2.0 * n * lse
    grad = 2.0 * n * grad_z
    
    return val, grad

def lfbgbs_optimize(a_init, tau, m, max_iter, time_limit, verbose=False):
    """Optimize using L-BFGS-B with smoothed objective."""
    n = len(a_init)
    a_init = norm_sum1(a_init)
    
    # Initial z: log(a_init) with clipping
    z0 = np.log(a_init + 1e-30)
    z0 = z0 - z0.mean()  # center to avoid numerical issues
    
    best_val = float('inf')
    best_z = z0.copy()
    
    counter = [0]
    t_start = time.monotonic()
    
    def fun_grad(z):
        counter[0] += 1
        if time.monotonic() - t_start > time_limit:
            raise StopIteration()
        return smooth_obj_and_grad(z, tau, m, n)
    
    try:
        res = minimize(
            fun_grad, z0, method='L-BFGS-B',
            jac=True, maxiter=max_iter,
            options={'maxcor': 20, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 40}
        )
        if res.success or res.fun < best_val:
            best_val = res.fun
            best_z = res.x.copy()
    except (StopIteration, Exception) as e:
        if verbose:
            print(f"    L-BFGS-B stopped: {e}")
    
    a_out = softmax(best_z)
    a_out = norm_sum1(a_out)
    
    if verbose:
        elapsed = time.monotonic() - t_start
        exact = 2.0 * n * fft_autocorr(a_out).max()
        print(f"    L-BFGS-B ({elapsed:.1f}s, {counter[0]} iters): smoothed={best_val:.6f}, exact={exact:.6f}")
    
    return a_out

def coord_descent_polish(a0, time_limit, verbose=False):
    """Coordinate descent on the exact objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_m = fft_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fft_autocorr(a)
        M = ac.max()
        improved = False
        
        # Find top peaks
        top_k = np.argsort(ac)[::-1][:8]
        
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Find the position with maximum contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination: minimize increase in autocorrelation
            # Approximate: move to a position that doesn't contribute to top peaks
            peak_influence = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:

                    peak_influence[idx2] += a[k2i - idx2]
            
            candidates = np.where(peak_influence < 0.3 * peak_influence.max())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            best_score_local = M
            best_move = None
            
            for dst in candidates[:50]:  # trial a few
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0 or a_trial[dst] < 0:
                        continue
                    M_trial = fft_autocorr(a_trial).max()
                    if M_trial < best_score_local:
                        best_score_local = M_trial
                        best_move = (a_trial.copy(), M_trial)
            
            if best_move is not None:
                a = best_move[0]
                M = best_move[1]
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                improved = True
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:

                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt

                M_new = fft_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                stall += 1
            else:
                stall += 1
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord descent ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a

def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = norm_sum1(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    return norm_sum1(a_new)

def parametric_families(n):
    """Generate parametric initializations."""
    x = np.linspace(0, 1, n)
    i = np.arange(n, dtype=np.float64)
    n_f = float(n)
    
    families = []
    
    # Sine-based
    for freq in [1, 2, 3]:
        for power in [1, 2, 3, 4]:
            families.append(np.maximum(np.sin(np.pi * freq * x), 0) ** power)
    
    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3]:
            families.append(np.exp(-((x - c)**2) / (2*sigma**2)))
    
    # Power laws
    for p in [0.5, 1, 2, 3]:
        families.append((i + 1) ** p)
        families.append((n_f - i) ** p)
    
    # Two-peak
    for gap in [0.1, 0.2, 0.3]:
        for w in [0.05, 0.1, 0.15]:
            families.append(np.exp(-((x - 0.5 + gap/2)**2)/(2*w**2)) + np.exp(-((x - 0.5 - gap/2)**2)/(2*w**2)))
    
    # Cosine window variations
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0]:
        families.append(np.maximum(np.cos(np.pi * alpha * (x - 0.5)), 0) ** 2)
    
    return families
