def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgrad(a, conv, top_k=40, temperature=0.005):
    n = len(a)
    conv_max = conv.max()
    threshold = conv_max - temperature * 10.0
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def golden_section(f, a, b, tol=1e-10, max_iter=50):
    """Minimize f on [a, b] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0

def simulated_annealing(a0, time_limit, seed=42, T_init=0.001, T_min=1e-7, cooldown=0.9995,

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

            except Exception:
                return 1e6

        res = scipy_minimize(obj, params0, method='Nelder-Mead',
                             options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-10,
                                      'adaptive': True})
        if res.success or res.fun < best_score:
            c_p = res.x[:M + 1]
            phi_p = res.x[M + 1:]
            a = c_p[0] * np.ones(n)
            for m in range(1, M + 1):
                a += c_p[m] * np.cos(2 * np.pi * m * x / n + phi_p[m - 1])
            a = np.exp(a - a.max())
            s = a.sum()
            if s > 1e-15:
                a = a / s
                sc = 2.0 * n * fast_conv(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
    return best_a


def make_parametric_shapes(n):
    """Generate diverse parametric initializations."""
    inits = []
    labels = []
    x = np.linspace(0, 1, n)

    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5]:
        a = (1.0 - x) ** alpha
        inits.append(norm_sum1(a))
        labels.append(f"dec{alpha}")

    for alpha in [0.3, 0.5, 0.7, 1.0]:
        a = x ** alpha
        inits.append(norm_sum1(a))
        labels.append(f"inc{alpha}")

    for center, width in [(0.3, 0.1), (0.4, 0.15), (0.5, 0.2), (0.6, 0.1), (0.7, 0.15)]:
        a = np.exp(-((x - center) / width) ** 2)
        inits.append(norm_sum1(a))
        labels.append(f"gauss_{center}_{width}")

    for bw in [3.0, 5.0, 7.0, 10.0, 15.0]:
        a = np.sinc((x - 0.5) * bw) ** 2
        a = np.maximum(a, 0.0)
        inits.append(norm_sum1(a))
        labels.append(f"sinc{bw}")

    for mode in [1, 2, 3]:
        a = np.cos(np.pi * mode * (x - 0.5)) ** 2
        inits.append(norm_sum1(a))
        labels.append(f"cos{mode}")

    for rate in [0.5, 1.0, 2.0, 3.0]:
        a = np.exp(-rate * x)
        inits.append(norm_sum1(a))
        labels.append(f"exp{rate}")

    a = 0.5 * np.exp(-((x - 0.25) / 0.1)**2) + 0.3 * np.exp(-((x - 0.65) / 0.12)**2)
    inits.append(norm_sum1(a))
    labels.append("twohump")

    for seed_r in [7, 17, 27]:
        rng_r = np.random.default_rng(seed_r)
        a = np.ones(n) * 0.5
        for f in range(1, 8):
            coeff = rng_r.normal(0, 0.15 / np.sqrt(f))
            phase = rng_r.uniform(0, 2 * np.pi)
            a += coeff * np.sin(2 * np.pi * f * x + phase)
        a = np.maximum(a, 0.001)
        inits.append(norm_sum1(a))
        labels.append(f"rnd{seed_r}")

    return inits, labels
