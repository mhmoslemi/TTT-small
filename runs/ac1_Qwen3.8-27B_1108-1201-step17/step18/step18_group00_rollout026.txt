import numpy as np
import time

def fast_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient(a, c, beta, n):
    """Compute gradient of smoothed max of convolution.
    
    g[i] = 2 * sum_m a[m] * w[i+m] where w = softmax(beta * c)
    
    Computed via: g = conv(a_reversed, w)[1:n+1]
    """
    # Softmax weights
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()
    
    # g[i] = sum_m a[m] * w[i+m] = conv(a_rev, w)[i+1]
    a_rev = a[::-1]
    # Use FFT for the cross-correlation
    size = len(a_rev) + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(Ar * W, fft_size)[:size]
    
    # g[i] = conv_result[i+1] for i=0,...,n-1
    g = conv_result[1:n+1] * 2.0
    return g, w

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def mirrored_descent_step(a, grad, eta, min_val=1e-15):
    """Mirror descent step on the simplex."""
    log_a = np.log(np.maximum(a, min_val)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def smoothed_max_descent(a0, time_limit, beta_start=50.0, beta_end=5.0,
                         eta_start=0.5, momentum=0.9, verbose=True):
    """Main optimizer: smoothed max gradient descent with mirror steps."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    v = np.zeros(n)  # momentum
    total_iters = 0
    
    # We'll do multiple "rounds" with decreasing beta
    n_rounds = 20
    for round_idx in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Beta schedule: start high, decrease
        progress = round_idx / max(1, n_rounds - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Reset momentum at start of each round
        v = np.zeros(n)
        
        iters_this_round = 0
        max_round_time = time_limit / n_rounds * 1.5  # allow some overflow
        
        round_start = time.monotonic()
        while time.monotonic() - round_start < max_round_time:
            if time.monotonic() - t0 > time_limit:
                break
            
            c = fast_conv(a)
            M_curr = c.max()
            
            if M_curr < best_m - 1e-15:
                best_m = M_curr
                best_a = a.copy()
            
            grad, w = compute_gradient(a, c, beta, n)
            
            # Nesterov momentum
            v = momentum * v + grad
            # Actually for Nesterov, we should evaluate gradient at lookahead point
            # But for simplicity, use heavy ball: v = momentum*v + grad; a -= eta*v
            
            eta = eta_start / (1.0 + 0.01 * iters_this_round)
            
            a_new = mirrored_descent_step(a, v, eta)
            
            # Check if we improved
            if iters_this_round % 50 == 0:
                c_check = fast_conv(a_new)
                if c_check.max() > best_m * 1.5:
                    # Too bad, revert
                    a_new = a
                else:
                    a = a_new
            
            a = a_new
            iters_this_round += 1
            total_iters += 1
            
            if verbose and iters_this_round % 500 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SMD round={round_idx} iter={iters_this_round} (t={elapsed:.1f}s): J={J:.6f} beta={beta:.1f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    SMD done (t={elapsed:.1f}s): J={J:.6f}, total_iters={total_iters}")
    return best_a, best_m

                scores[k2i - idx2] += a[idx2]

            candidates = np.array([i for i in range(n) if i not in peak_zones and i != src_idx])
            if len(candidates) < 5:
                candidates = np.array([i for i in range(n) if i != src_idx])
            if len(candidates) == 0:
                continue

            cand_scores = scores[candidates]
            # Add some randomness to avoid cycling
            noise = rng.random(len(candidates)) * cand_scores.max() * 0.1
            cand_scores += noise

            dst = int(candidates[np.argmin(cand_scores)])

            # Try different fractions
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src_idx] -= amt
                a_trial[dst] += amt
                a_trial = project_simplex(a_trial)
                c_trial = fast_conv(a_trial)
                if c_trial.max() < best_m - 1e-15:
                    a = a_trial
                    best_m = c_trial.max()
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
                elif c_trial.max() < M_curr:
                    a = a_trial
                    improved = True
                    break

        if not improved:
            no_improve += 1
            # Random perturbation to escape
            if no_improve > 100:
                idx1 = int(rng.integers(n))
                idx2 = int(rng.integers(n))
                if idx1 != idx2 and a[idx1] > 1e-10:
                    amt = a[idx1] * rng.uniform(0.01, 0.05)
                    a[idx1] -= amt
                    a[idx2] += amt
                    a = project_simplex(a)
                    c_new = fast_conv(a)
                    if c_new.max() < best_m:
                        best_m = c_new.max()
                        best_a = a.copy()
                        no_improve = 0

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PS iter={iter_count} (t={elapsed:.1f}s): J={J:.7f} no_improve={no_improve}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    PS done (t={elapsed:.1f}s): J={J:.7f} iters={iter_count}")
    return best_a, best_m
