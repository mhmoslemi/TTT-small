def norm_sum1(a):

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def spectral_shaping_step(a, n_iters=5, step_size=0.1):
    """Try to flatten the power spectrum of a."""
    n = len(a)
    a = a.copy()
    for _ in range(n_iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        # Target: flat magnitude
        target_mag = np.ones_like(mag)
        # Phase-preserving update: new a = IFFT(mag_new * exp(i*phase))
        phase = np.angle(A)
        # Blend current and target magnitude
        mag_new = (1 - step_size) * mag + step_size * target_mag * mag.mean()
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None, use_all=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search (more aggressive)
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        alphas = [best_alpha, min(1.0, best_alpha * 1.3), min(1.0, best_alpha * 1.8),
                  min(1.0, best_alpha * 2.5), 0.5, 0.3, 0.2, 0.1]
        for alpha in alphas:
            if alpha <= 0 or alpha >= 1.01:
                continue
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 8 == 0:
            K = min(K + 25, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def spectral_gradient_optimize(a0, time_limit, verbose=True, rng_seed=100):
    """
    New method: Use the relationship between autocorrelation and power spectrum.
    The autocorrelation is IFFT(|FFT(a)|^2). We want to minimize the peak.
    Strategy: iteratively reduce the peak by adjusting the power spectrum.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = 0.5
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find the top few peaks and their influence on a
        n_peaks = min(30, len(ac))
        peak_idx = np.argsort(ac)[::-1][:n_peaks]
        peak_vals = np.maximum(ac[peak_idx], 1e-30)

        # Compute gradient from peaks (weighted by their value)
        w = peak_vals ** 4  #sharp weighting
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, peak_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        # Subtract mean for simplex projection
        grad_tangent = grad - grad.mean()

        a_new = project_simplex(a - eta * grad_tangent)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.85
            if stall > 300:
                # Spectral perturbation to escape
                A = np.fft.rfft(a)
                noise = rng.normal(0, 0.02, len(A))
                A_mag = np.abs(A) * (1 + noise)
                A_new = np.maximum(A_mag, 1e-10) * np.exp(1j * np.angle(A))
                a_pert = np.fft.irfft(A_new, n)
                a_pert = np.maximum(a_pert, 0)
                s_p = a_pert.sum()
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    m_pert = fast_autocorr(a_pert).max()
                    if m_pert < best_m:
                        a = a_pert
                        stall = 0
                        eta = 0.3
                    else:
                        a = a_new
                else:
                    a = a_new
            elif stall > 100:
                a = a_new
            else:
                a = a_new

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpectGrad ({elapsed:.0f}s): J={J:.6f}, stall={stall}, eta={eta:.3f}")

    return best_a, best_m


def two_phase_optimize(a0, time_limit, solver, verbose=True, rng_seed=50):
    """
    Two-phase: first use spectral shaping to get a good coarse solution,
    then use AMD/LP for fine tuning.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Phase A: Spectral shaping (flatten power spectrum)
    t_phase_a = min(time_limit * 0.3, 120.0)
    t_start_a = time.monotonic()
    step_size = 0.05
    while time.monotonic() - t_start_a < t_phase_a:
        A = np.fft.rfft(a)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: make magnitude as flat as possible while keeping a non-negative
        mag_target = np.ones_like(mag) * mag.mean()
        mag_blend = (1 - step_size) * mag + step_size * mag_target
        A_new = mag_blend * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size if not improving
            step_size *= 0.95
            if step_size < 0.001:
                break
            a = a_new  # Still move to explore

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    TwoPhase spectral ({elapsed:.0f}s): J={J:.6f}")

    # Phase B: AMD refinement
    t_remaining = time_limit - (time.monotonic() - t0)
    if t_remaining > 30:
        a_out, _ = amd_optimize(best_a, t_remaining, top_k=60, beta=6.0,
                                 c0=5.0, restart_every=4000, verbose=False,
                                 rng_seed=rng_seed)
        m_out = fast_autocorr(a_out).max()
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    TwoPhase done ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Aggressive coordinate descent with pair-swap optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 150
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-contribution destinations
            dst_sens = np.zeros(n)
            for kk in top_k[:8]:
                kki = int(kk)
                ss = max(0, kki - n + 1)
                ee = min(n - 1, kki)
                ii = np.arange(ss, ee + 1)
                dst_sens[ii] += a[kki - ii]

            candidates = np.where(dst_sens < np.percentile(dst_sens, 20))[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            for dst in candidates[:5]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
    Uses multi-method approach: spectral shaping, AMD, LP-FW, coordinate descent,
    with adaptive time allocation and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = make_u_shape(n_base, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Spectral shaping (new) ===
    print(f"\n=== Phase 1: Spectral shaping (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out = spectral_shaping_step(a_in, n_iters=50, step_size=0.03)
        try_update(a_out, "spectral_shape")

        # Also try with different starting points
        a_in2 = norm_sum1(best_seq)
        # Perturb in frequency domain
        rng_sp = np.random.default_rng(777)
        A = np.fft.rfft(a_in2)
        A_mag = np.abs(A) * (1 + rng_sp.normal(0, 0.05, len(A)))
        A_new = np.maximum(A_mag, 1e-10) * np.exp(1j * np.angle(A))
        a_sp = np.fft.irfft(A_new, n0)
        a_sp = np.maximum(a_sp, 0)
        a_sp = a_sp / a_sp.sum()
        a_out2 = spectral_shaping_step(a_sp, n_iters=50, step_size=0.03)
        try_update(a_out2, "spectral_pert")

    # === Phase 2: Two-phase optimize (new) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Two-phase optimize (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = two_phase_optimize(a_in, t2, solver, verbose=True, rng_seed=50)
        try_update(a_out, "twophase")

    # === Phase 3: AMD multi-start ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: AMD multi-start (n={n_cur}) ===")
    t3 = min(remaining * 0.20, 300.0)

    if t3 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n_cur, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_inverted_u(n_cur, sharpness=1.0))
        labels.append("inv_u")
        inits.append(make_bimodal(n_cur, width=0.15))
        labels.append("bimodal")
        inits.append(spectral_init(n_cur, n_modes=30, seed=42))
        labels.append("spectral")

        if parent is not None:
            a_p = norm_sum1(parent)
            inits.append(norm_sum1(0.7 * a_p + 0.3 * make_u_shape(n_cur, 2.0)))
            labels.append("blend70")
            inits.append(norm_sum1(0.5 * a_p + 0.5 * make_bimodal(n_cur, 0.2)))
            labels.append("blend50b")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n_cur // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(10.0, t3 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                     restart_every=4000, adaptive_beta=True,
                                     verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 4: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_cur}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40 and n_cur <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_cur <= 1500)
        top_K_init = min(300, 2 * n_cur - 1)
        top_K_max = min(3000, 2 * n_cur - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all)
        try_update(a_out, "lp_main")

    # === Phase 5: Spectral gradient (new method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Spectral gradient (n={n_cur}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_gradient_optimize(a_in, t5, verbose=True, rng_seed=100)
        try_update(a_out, "spectral_grad")

    # === Phase 6: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LP-FW r2 (n={n_cur}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 30 and n_cur <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_cur <= 1500)
        top_K_init = min(500, 2 * n_cur - 1)
        top_K_max = min(5000, 2 * n_cur - 1)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all)
        try_update(a_out, "lp_r2")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (current n={n_cur}) ===")

    trial_lengths = []
    for delta in [200, 500, -200, 800, -500, 1000, -800]:
        new_n = n_cur + delta
        if 2000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_cur]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.3, 150.0)
        if t7a > 20:
            a_out, _ = amd_optimize(a_long, t7a, top_k=60, beta=6.0, c0=4.0,
                                     restart_every=4000, adaptive_beta=True,
                                     verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 5000:
            t7b = min(remaining * 0.3, 100.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                use_all = (new_n <= 1500)
                top_K = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                           top_K_init=min(300, 2*new_n-1),
                                           top_K_max=top_K, use_all=use_all)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Coordinate polish (n={n_cur}) ===")
    t8 = min(remaining * 0.08, 100.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t8, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 9: Final intensive AMD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Final AMD (n={n_cur}) ===")
    t9 = min(remaining * 0.15, 150.0)
    if t9 > 20:
        a_in = norm_sum1(best_seq)
        # Multiple AMD runs with different parameters
        for run in range(3):
            t_run = min(t9 / (4 - run), time_budget - 30 - (time.monotonic() - t_global))
            if t_run < 10:
                break
            rng_s = 1000 + run * 100
            a_out, _ = amd_optimize(a_in, t_run, top_k=50 + run*20, beta=5.0 + run,
                                     c0=4.0 + run, restart_every=3000 + run*1000,
                                     verbose=False, rng_seed=rng_s)
            try_update(a_out, f"finalamd_{run}")
            a_in = norm_sum1(best_seq)  # Refresh from best

    # === Phase 10: Final coordinate polish ===
