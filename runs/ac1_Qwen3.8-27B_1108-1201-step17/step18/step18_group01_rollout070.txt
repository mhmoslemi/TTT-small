```
def fft_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n <= 1:
        return a.astype(np.float64).copy() if n==1 else np.array([0.0])
    size = 2*n - 1
    fsz = 1 << (size - 1).bit_length()
    F = np.fft.rfft(a, fsz)
    ac = np.fft.irfft(F * F, fsz)[:size]
    # remove imaginary? irfft real
    return ac
```

```
def coord_polish(a0, time_limit, verbose=False, seed=101):
    rng = np.random.default_rng(seed)
    a = norm_sum1(a0)
    n = len(a)
    if n < 2: return a
    best_a = a.copy()
    best_m = float(np.max(fft_autocorr(a)))
    t0 = time.monotonic()
    stall = 0
    max_stall = 120
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fft_autocorr(a)
        M = float(np.max(ac))
        top_count = min(10, len(ac))
        if top_count == len(ac):
            top_lags = np.arange(len(ac))
        else:
            top_lags = np.argpartition(ac, -top_count)[-top_count:]
        top_lags = top_lags[np.argsort(ac[top_lags])[::-1]]
        improved = False
        risk = np.zeros(n, dtype=np.float64)
        risk_count = min(6, len(top_lags))
        total_w = 0.0
        weights = np.zeros(risk_count, dtype=np.float64)
        for ii in range(risk_count):
            v = float(ac[int(top_lags[ii])])
            if v > 0:
                weights[ii] = v
        total_w = float(np.sum(weights))
        if total_w > 0:
            for ii in range(risk_count):
                if weights[ii] <= 0: continue
                k = int(top_lags[ii])
                wt = weights[ii] / total_w
                i0 = max(0, k - n + 1)
                i1 = min(n - 1, k)
                if i0 <= i1:
                    idx = np.arange(i0, i1 + 1)
                    risk[idx] += wt * a[k - idx]
        else:
            risk = a.copy()
        # low risk candidates
        if n > 50:
            q = max(1, min(n - 1, int(0.1 * n)))
            thr = float(np.sort(risk)[q])
            low = np.where(risk <= thr)[0]
            if len(low) < 20:
                thr = float(np.sort(risk)[min(n - 1, int(0.35 * n))])
                low = np.where(risk <= thr)[0]
        else:
            low = np.arange(n)
        low_sorted = low[np.argsort(risk[low])]
        for lag_pos in range(min(5, len(top_lags))):
            if improved: break
            k = int(top_lags[lag_pos])
            i0 = max(0, k - n + 1)
            i1 = min(n - 1, k)
            if i0 > i1: continue
            idx = np.arange(i0, i1 + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.size == 0: continue
            src_order = np.argsort(contrib)[::-1]
            for src_pos in range(min(6, src_order.size)):
                src = int(idx[src_order[src_pos]])
                aval = a[src]
                if aval <= 1e-13:
                    continue
                # choose up to 4 destinations
                checked = 0
                accepted = False
                for dpos in range(min(50, len(low_sorted))): # scan to find 4
                    dst = int(low_sorted[dpos])
                    if dst == src:
                        continue
                    checked += 1
                    for frac in (0.05, 0.15):
                        amt = aval * frac
                        a_try = np.empty_like(a)
                        a_try[:] = a
                        a_try[src] -= amt
                        a_try[dst] += amt
                        if a_try[src] < 0.0 or a_try[dst] < 0.0:
                            continue
                        m_try = float(np.max(fft_autocorr(a_try)))
                        if m_try < best_m - 1e-15:
                            a_try = norm_sum1(a_try)
                            a[:] = a_try  # or a = a_try
                            a = a_try
                            best_m = m_try
                            best_a = a.copy()
                            improved = True
                            accepted = True
                            break
                    if accepted or checked >= 4:
                        break
                if accepted:
                    break
            if accepted:
                break
        if improved:
            stall = 0
            continue
        # random escapes
        for _ in range(25):
            if time.monotonic() - t0 > time_limit: break
            src = int(rng.integers(n))
            dst = int(low_sorted[rng.integers(len(low_sorted))]) if len(low_sorted)>0 else int(rng.integers(n))
            if src == dst or a[src] <= 1e-13: continue
            frac = float(rng.uniform(0.02, 0.1))
            amt = a[src] * frac
            a_try = np.empty_like(a)
            a_try[:] = a
            a_try[src] -= amt
            a_try[dst] += amt
            if a_try[src] < 0: continue
            m_try = float(np.max(fft_autocorr(a_try)))
            if m_try < best_m - 1e-15:
                a = norm_sum1(a_try)
                best_m = m_try
                best_a = a.copy()
                stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a
```

```python
"""
Search for low autocorrelation peak non-negative step functions.

The returned sequence minimizes the given functional
    2*n*max(np.convolve(a, a)) / (sum(a))^2
under the stated nonnegativity constraint.  The implementation combines:

1. Strong analytic seeds, especially monotone ramp families, which attain
   very low peak squared convolution for this evaluator.
2. Accelerated refinement on the simplex via softmax reparameterization
   and L-BFGS-B on a smoothed top-lag autocorrelation surrogate.
3. Multiplicative subgradient / mirror-flow polish.
4. Exact coordinate-descent polish on the unsmoothed objective.
5. Occasional perturbed restarts and length modifications.

All helper functions are top-level, and no lambdas or closures are used.
"""

import math
import time

import numpy as np
from scipy.optimize import minimize


_OPT_TAU = 10.0
_OPT_M = 128
_OPT_N = 1
_OPT_DEADLINE = 0.0
_OPT_STOP = False
_OPT_BEST_EXACT_VAL = float("inf")
_OPT_BEST_EXACT_Z = None


def fft_autocorr(a):
    """Linear autocorrelation / self-convolution computed by FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n <= 0:
        return np.array([0.0], dtype=np.float64)
    if n == 1:
        return a.copy()
    size = 2 * n - 1
    fsz = 1 << (size - 1).bit_length()
    F = np.fft.rfft(a, fsz)
    ac = np.fft.irfft(F * F, fsz)[:size]
    return ac


def norm_sum1(a):
    """Clip to nonnegative and normalize to sum one."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def internal_score(a):
    """Fast proxy score comparable to evaluate_sequence for nonnegative inputs."""
    a = norm_sum1(a)
    n = len(a)
    if n <= 0:
        return float("inf")
    ac = fft_autocorr(a)
    m = float(np.max(ac))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def safe_evaluate(a):
    """Call the provided external evaluator safely."""
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0:
            return float("inf")
        if not np.all(np.isfinite(arr)):
            return float("inf")
        vals = [float(x) for x in arr.tolist()]
        return float(evaluate_sequence(vals))
    except Exception:
        return float("inf")


def softmax(z):
    """Stable softmax returning a probability vector."""
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.ones(n, dtype=np.float64) / float(n)
    z = z - float(np.max(z))
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        e = np.exp(z)
    s = float(e.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def smoothed_fun_grad(z):
    """
    Smoothed top-lag surrogate and its gradient with respect to z.

    The actual sequence is a = softmax(z).  The returned value is
    2n * logsumexp(tau * top_lags), which approximates 2n * max_lag.
    We also track the best exact (unsmoothed) value seen during line searches.
    """
    global _OPT_BEST_EXACT_VAL, _OPT_BEST_EXACT_Z

    if _OPT_STOP or time.monotonic() > _OPT_DEADLINE:
        raise TimeoutError("time limit reached in L-BFGS-B objective")

    z = np.asarray(z, dtype=np.float64)
    n = int(z.size)
    if n <= 1:
        return 1e100, np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return 1e100, np.zeros(n, dtype=np.float64)

    a = softmax(z)
    ac = fft_autocorr(a)
    ac = np.asarray(ac, dtype=np.float64)

    exact = float(2.0 * n * np.max(ac))
    if np.isfinite(exact) and exact < _OPT_BEST_EXACT_VAL:
        _OPT_BEST_EXACT_VAL = exact
        _OPT_BEST_EXACT_Z = z.copy()

    tau = float(_OPT_TAU)
    m = int(_OPT_M)
    cnt = min(max(1, m), len(ac))

    if not np.all(np.isfinite(ac)):
        ac = np.where(np.isfinite(ac), ac, -np.inf)

    if cnt >= len(ac):
        top_idx = np.arange(len(ac), dtype=int)
        top_vals = ac.copy()
    else:
        top_idx = np.argpartition(ac, -cnt)[-cnt:].astype(int)
        top_vals = ac[top_idx]

    max_v = float(np.max(top_vals))

    if not np.isfinite(max_v):
        finite_idx = np.where(np.isfinite(ac))[0]
        if finite_idx.size == 0:
            return 1e100, np.zeros(n, dtype=np.float64)
        k = int(finite_idx[int(np.argmax(ac[finite_idx]))])
        grad_a = np.zeros(n, dtype=np.float64)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            idx = np.arange(i0, i1)
            grad_a[idx] = 2.0 * a[k - idx]
        base = grad_a - float(np.dot(a, grad_a))
        grad = 2.0 * n * (a * base)
        val = float(2.0 * n * ac[k])
        if not np.isfinite(val):
            val = 1e100
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n, dtype=np.float64)
        return float(val), grad

    shifted = tau * (top_vals - max_v)
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        e = np.exp(shifted)
    s = float(e.sum(dtype=np.float64))

    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros(cnt, dtype=np.float64)
        w[int(np.argmax(top_vals))] = 1.0
        lse = max_v
    else:
        w = e / s
        lse = max_v + math.log(s / tau) if tau > 0 else max_v

    grad_a = np.zeros(n, dtype=np.float64)
    for jj in range(cnt):
        wt = float(w[jj])
        if wt <= 1e-15:
            continue
        k = int(top_idx[jj])
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 >= i1:
            continue
        idx = np.arange(i0, i1)
        grad_a[idx] += (2.0 * wt) * a[k - idx]

    base = grad_a - float(np.dot(a, grad_a))
    grad = 2.0 * n * (a * base)
    val = float(2.0 * n * lse)

    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    if not np.isfinite(val):
        val = 1e100

    return float(val), grad


def _lfgbbs_callback(xk):
    """Callback for L-BFGS-B to stop on the wall-clock deadline."""
    global _OPT_STOP
    if time.monotonic() > _OPT_DEADLINE:
        _OPT_STOP = True


def lfgbbs_optimize(a_init, tau, m, max_iter, time_limit, verbose=False):
    """
    Optimize on the simplex using softmax reparameterization and L-BFGS-B.

    The smoothed objective is only a driver; the best exact sequence seen is
    returned.
    """
    global _OPT_TAU, _OPT_M, _OPT_N, _OPT_DEADLINE, _OPT_STOP
    global _OPT_BEST_EXACT_VAL, _OPT_BEST_EXACT_Z

    t_start = time.monotonic()
    limit = float(time_limit)
    a = norm_sum1(a_init)
    n = int(a.size)
    if n <= 1 or limit <= 0.0:
        return a

    _OPT_TAU = float(tau)
    _OPT_M = int(m)
    _OPT_N = n
    _OPT_DEADLINE = t_start + max(1e-6, limit)
    _OPT_STOP = False
    _OPT_BEST_EXACT_VAL = float("inf")
    _OPT_BEST_EXACT_Z = None

    # Softmax start close to a, with a small floor so zero coordinates are
    # not completely inaccessible.
    floor = max(1e-12, 1e-8 / max(1, float(n)))
    z0 = np.log(np.maximum(a, floor))
    z0 = np.where(np.isfinite(z0), z0, -30.0)
    z0 = z0 - float(np.mean(z0))

    res = None
    try:
        res = minimize(
            smoothed_fun_grad,
            z0,
            jac=True,
            method="L-BFGS-B",
            callback=_lfgbbs_callback,
            options={
                "maxiter": int(max_iter),
                "maxls": 15,
                "ftol": 1e-12,
                "gtol": 1e-8,
                "maxcor": 20,
            },
        )
    except Exception:
        # Timeout or numerical issue: use the best exact iterate saved.
        pass

    out = a
    if _OPT_BEST_EXACT_Z is not None:
        out = softmax(_OPT_BEST_EXACT_Z)

    try:
        if res is not None and hasattr(res, "x") and res.x is not None:
            if np.all(np.isfinite(np.asarray(res.x, dtype=np.float64))):
                ar = softmax(np.asarray(res.x, dtype=np.float64))
                if internal_score(ar) < internal_score(out):
                    out = ar
    except Exception:
        pass

    out = norm_sum1(out)
    if verbose:
        elapsed = time.monotonic() - t_start
        sc = internal_score(out)
        print(f"    L-BFGS-B ({elapsed:.1f}s): internal={sc:.6f}, tau={tau:g}, m={int(m)}")
    return out


def mgd_polish(a0, time_limit, top_k, eta0, verbose=False, seed=17):
    """
    Multiplicative gradient descent on the simplex with weighted top-lag
    subgradients.
    """
    a = norm_sum1(a0)
    n = len(a)
    if n <= 1 or time_limit <= 0.0:
        return a

    best_a = a.copy()
    best_m = float(np.max(fft_autocorr(a)))
    t0 = time.monotonic()

    eta = float(eta0)
    cur_top_k = max(8, int(top_k))
    patience = 0
    max_patience = 250
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < float(time_limit):
        ac = fft_autocorr(a)
        M = float(np.max(ac))
        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            break

        kmax = min(cur_top_k, len(ac))
        if kmax <= 0:
            break

        if kmax >= len(ac):
            top_idx = np.arange(len(ac), dtype=int)
        else:
            top_idx = np.argpartition(ac, -kmax)[-kmax:].astype(int)

        top_vals = np.maximum(ac[top_idx], 1e-300)
        w = top_vals * top_vals
        sw = float(w.sum(dtype=np.float64))
        if not np.isfinite(sw) or sw <= 1e-300:
            w = np.ones(kmax, dtype=np.float64) / float(kmax)
        else:
            w = w / sw

        grad = np.zeros(n, dtype=np.float64)
        for jj in range(kmax):
            wt = 2.0 * float(w[jj])
            if wt <= 1e-15:
                continue
            k = int(top_idx[jj])
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 >= i1:
                continue
            idx = np.arange(i0, i1)
            grad[idx] += wt * a[k - idx]

        gmax = float(np.max(np.abs(grad)))
        if not np.isfinite(gmax) or gmax < 1e-18:
            break

        grad = (grad - float(np.mean(grad))) / gmax
        with np.errstate(over="ignore", under="ignore", invalid="ignore"):
            a_new = a * np.exp(-eta * grad)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum(dtype=np.float64))
        if not np.isfinite(s) or s <= 1e-15:
            break
        a_new = a_new / s

        M_new = float(np.max(fft_autocorr(a_new)))
        if np.isfinite(M_new) and M_new <= M + 1e-13:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.5
            if eta < 0.0012:
                cur_top_k = max(8, cur_top_k // 2)
                eta = 0.014
                patience = max(0, patience - 80)
                # small stochastic escape to avoid degenerate stagnation
                if rng.random() < 0.25:
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = float(rng.uniform(0.005, 0.03))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s = float(a.sum(dtype=np.float64))
                        if s > 0:
                            a = a / s

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD ({elapsed:.1f}s): internal={J:.6f}, eta={eta:.6f}, top={cur_top_k}")
    return best_a


def coord_polish(a0, time_limit, verbose=False, seed=101):
    """
    Exact local search.  Moves small mass from peak-contributing coordinates
    to low-risk coordinates, accepting only exact improvements.
    """
    a = norm_sum1(a0)
    n = len(a)
    if n <= 1 or time_limit <= 0.0:
        return a

    best_a = a.copy()
    best_m = float(np.max(fft_autocorr(a)))
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 120

    while time.monotonic() - t0 < float(time_limit) and stall < max_stall:
        ac = fft_autocorr(a)
        M = float(np.max(ac))

        top_count = min(10, len(ac))
        if top_count >= len(ac):
            top_lags = np.arange(len(ac), dtype=int)
        else:
            top_lags = np.argpartition(ac, -top_count)[-top_count:].astype(int)
        top_lags = top_lags[np.argsort(ac[top_lags])[::-1]]

        risk = np.zeros(n, dtype=np.float64)
        rc = min(6, len(top_lags))
        ws = np.zeros(rc, dtype=np.float64)
        for ii in range(rc):
            v = float(ac[int(top_lags[ii])])
            if np.isfinite(v) and v > 0.0:
                ws[ii] = v
        total_w = float(ws.sum(dtype=np.float64))
        if total_w > 0.0:
            for ii in range(rc):
                if ws[ii] <= 0.0:
                    continue
                k = int(top_lags[ii])
                wt = ws[ii] / total_w
                i0 = max(0, k - n + 1)
                i1 = min(n - 1, k)
                if i0 <= i1:
                    idx = np.arange(i0, i1 + 1)
                    risk[idx] += wt * a[k - idx]
        else:
            risk = a.copy()

        if n > 50:
            q = max(1, min(n - 1, int(0.10 * n)))
            thr = float(np.partition(risk, q)[q])
            low = np.where(risk <= thr)[0]
            if low.size < 20:
                q2 = min(n - 1, int(0.35 * n))
                thr2 = float(np.partition(risk, q2)[q2])
                low = np.where(risk <= thr2)[0]
        else:
            low = np.arange(n, dtype=int)

        if low.size == 0:
            low = np.arange(n, dtype=int)
        low_sorted = low[np.argsort(risk[low])]

        improved = False

        for lag_pos in range(min(5, len(top_lags))):
            if improved:
                break
            k = int(top_lags[lag_pos])
            i0 = max(0, k - n + 1)
            i1 = min(n - 1, k)
            if i0 > i1:
                continue
            idx = np.arange(i0, i1 + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.size == 0:
                continue
            src_order = np.argsort(contrib)[::-1]

            for src_pos in range(min(6, src_order.size)):
                src = int(idx[src_order[src_pos]])
                aval = float(a[src])
                if aval <= 1e-13:
                    continue

                accepted = False
                checked_dsts = 0
                for dp in range(min(80, low_sorted.size)):
                    dst = int(low_sorted[dp])
                    if dst == src:
                        continue
                    checked_dsts += 1

                    for frac in (0.05, 0.15):
                        amt = aval * frac
                        a_try = np.empty_like(a)
                        a_try[:] = a
                        a_try[src] -= amt
                        a_try[dst] += amt
                        if a_try[src] < 0.0 or a_try[dst] < 0.0:
                            continue

                        m_try = float(np.max(fft_autocorr(a_try)))
                        if m_try < best_m - 1e-15:
                            a_try = norm_sum1(a_try)
                            a = a_try
                            best_m = m_try
                            best_a = a.copy()
                            improved = True
                            accepted = True
                            break

                    if accepted or checked_dsts >= 4:
                        break

                if accepted:
                    break

        if improved:
            stall = 0
            continue

        # Random escape moves from high-mass sources to low-risk destinations.
        for _ in range(25):
            if time.monotonic() - t0 > float(time_limit):
                break
            src = int(rng.integers(n))
            dst = int(low_sorted[int(rng.integers(low_sorted.size))]) if low_sorted.size else int(rng.integers(n))
            if src == dst or a[src] <= 1e-13:
                continue
            frac = float(rng.uniform(0.02, 0.10))
            amt = float(a[src]) * frac
            a_try = np.empty_like(a)
            a_try[:] = a
            a_try[src] -= amt
            a_try[dst] += amt
            if a_try[src] < 0.0:
                continue

            m_try = float(np.max(fft_autocorr(a_try)))
            if m_try < best_m - 1e-15:
                a_try = norm_sum1(a_try)
                a = a_try
                best_m = m_try
                best_a = a.copy()
                stall = 0
                break

        stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): internal={J:.6f}")
    return best_a


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    a = norm_sum1(a)
    old_n = len(a)
    new_n = int(new_n)
    if new_n <= 1:
        return np.array([1.0], dtype=np.float64)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, a)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    return norm_sum1(out)


def random_perturb(a, amp, seed):
    """Small additive relative perturbation, followed by simplex projection."""
    a = norm_sum1(a)
    rng = np.random.default_rng(seed)
    scale = float(amp) * max(1e-300, float(np.mean(a)))
    noise = rng.normal(0.0, max(1e-300, scale), size=a.size)
    out = a + noise
    out = np.where(np.isfinite(out), out, 0.0)
    return norm_sum1(out)


def parametric_initializations(n):
    """
    A collection of cheap seeds.  The monotone ramp is especially strong for
    the unconstrained nonnegative problem defined by the evaluator.
    """
    n = int(n)
    if n < 2:
        return [np.array([1.0], dtype=np.float64)]

    i = np.arange(n, dtype=np.float64)
    x = i / float(n - 1)
    out = []

    # Monotone ramps and shifted versions.
    out.append(i + 1.0)
    out.append(float(n) - i)
    out.append(i + 0.5)
    out.append(1.0 + i / float(max(1, n - 1)))

    # Power ramps.
    for p in (0.25, 0.5, 0.75, 1.0, 1.5, 2.0):
        out.append(np.power(i + 1.0, p))
        out.append(np.power(float(n) - i, p))

    # Experiments with slowly varying exponentials.
    for r in (0.3, 0.7, 1.3):
        out.append(np.exp(r * x))
        out.append(np.exp(r * (1.0 - x)))

    # Affine seeds with small left-end mass.
    for p in (0.0, 0.05, 0.10, 0.20):
        out.append(p + (1.0 - p) * x)

    # Simple smooth increasing shapes.
    out.append(np.sin(0.5 * np.pi * x) + 0.01)
    out.append(1.0 - np.cos(0.5 * np.pi * x) + 0.01)

    return out


def consider_candidate(best_a, best_score, cand):
    """
    Accept a candidate only if its fast score is already promising and its
    external safe score improves the current best.
    """
    if cand is None:
        return best_a, best_score

    cand = norm_sum1(cand)
    if cand.size == 0:
        return best_a, best_score

    try:
        iscore = internal_score(cand)
    except Exception:
        return best_a, best_score

    if not np.isfinite(iscore):
        return best_a, best_score

    if np.isinf(best_score) or iscore < float(best_score) - 1e-10:
        sc = safe_evaluate(cand)
        if np.isfinite(sc) and sc < float(best_score):
            return cand, float(sc)

    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main orchestrator.  Returns the best sequence found as a Python list.
    """
    t_global = time.monotonic()
    budget = min(float(budget_s), 1800.0) - 90.0
    if budget < 60.0:
        budget = 60.0

    best_a = None
    best_score = float("inf")

    # Try to inherit a previous construction if it is reasonably sized.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and np.all(np.isfinite(p)) and p.size <= 15000:
            cand = norm_sum1(p)
            sc = safe_evaluate(cand)
            if np.isfinite(sc):
                best_a = cand
                best_score = float(sc)
                print(f"Parent: n={len(cand)}, score={sc:.6f}")
        elif p.size > 0 and np.all(np.isfinite(p)):
            # Too large for many operations here; use a downsampled seed,
            # but keep the large parent as a final fallback candidate.
            seed_large = interpolate_to_length(norm_sum1(p), 7000)
            print(f"Parent too large ({len(p)}); using downsampled seed.")
            sc = safe_evaluate(seed_large)
            if np.isfinite(sc):
                best_a = seed_large
                best_score = float(sc)
    except Exception:
        pass

    # Choose a working length for analytic restarts.
    if best_a is not None and len(best_a) >= 1000:
        base_n = max(5200, min(len(best_a), 8000))
    else:
        base_n = 7000
    base_n = int(max(1024, min(base_n, 9000)))

    # Parametric search first.  This includes the monotone ramp, which is a
    # very strong seed for this evaluator.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 10.0:
        fams = parametric_initializations(base_n)
        best_probe = None
        best_probe_internal = float("inf")
        for fam in fams:
            cand = norm_sum1(fam)
            try:
                sc = internal_score(cand)
            except Exception:
                continue
            if np.isfinite(sc) and sc < best_probe_internal:
                best_probe_internal = float(sc)
                best_probe = cand
        if best_probe is not None:
            best_a, best_score = consider_candidate(best_a, best_score, best_probe)
            print(f"Parametric best: internal={best_probe_internal:.6f}, safe={best_score:.6f}")

    if best_a is None:
        # Extremely safe fallback: linear ramp.
        i = np.arange(7000, dtype=np.float64)
        best_a = norm_sum1(i + 1.0)
        best_score = safe_evaluate(best_a)
        print(f"Fallback ramp: safe={best_score:.6f}")

    # Main optimization loop.  Each block is time-boxed and considered safely.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 60.0:
        tt = min(remaining * 0.18, 170.0)
        if tt > 25.0:
            cand = lfgbbs_optimize(best_a, 8.0, 128, 90, tt, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

    remaining = budget - (time.monotonic() - t_global)
    if remaining > 45.0:
        tt = min(remaining * 0.14, 120.0)
        if tt > 20.0:
            cand = mgd_polish(best_a, tt, top_k=32, eta0=0.012, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

    remaining = budget - (time.monotonic() - t_global)
    if remaining > 35.0:
        tt = min(remaining * 0.11, 90.0)
        if tt > 15.0:
            cand = coord_polish(best_a, tt, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

    remaining = budget - (time.monotonic() - t_global)
    if remaining > 70.0:
        tt = min(remaining * 0.20, 180.0)
        if tt > 25.0:
            cand = lfgbbs_optimize(best_a, 24.0, 256, 80, tt, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

    remaining = budget - (time.monotonic() - t_global)
    if remaining > 40.0:
        tt = min(remaining * 0.13, 95.0)
        if tt > 18.0:
            cand = mgd_polish(best_a, tt, top_k=56, eta0=0.004, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

    remaining = budget - (time.monotonic() - t_global)
    if remaining > 35.0:
        tt = min(remaining * 0.11, 80.0)
        if tt > 15.0:
            cand = coord_polish(best_a, tt, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

    # Perturbed restart, useful if a local smooth optimum is encountered.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 170.0:
        pert = random_perturb(best_a, 0.75, seed=911)
        tt1 = min(remaining * 0.32, 130.0)
        if tt1 > 25.0:
            cand = lfgbbs_optimize(pert, 10.0, 160, 70, tt1, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

        remaining = budget - (time.monotonic() - t_global)
        if remaining > 35.0:
            tt2 = min(remaining * 0.25, 70.0)
            if tt2 > 15.0:
                cand = coord_polish(best_a, tt2, verbose=False)
                best_a, best_score = consider_candidate(best_a, best_score, cand)

    # Length exploration from the current best.
    remaining = budget - (time.monotonic() - t_global)
    new_lengths = []
    if best_a is not None:
        ncur = len(best_a)
        for delta in (600, 1200, 2000):
            nn = ncur + delta
            if 4000 <= nn <= 9500:
                new_lengths.append(nn)
    for nn in new_lengths[:2]:
        remaining = budget - (time.monotonic() - t_global)
        if remaining < 90.0:
            break
        a_long = interpolate_to_length(best_a, nn)
        tt1 = min(remaining * 0.42, 150.0)
        if tt1 > 25.0:
            cand = lfgbbs_optimize(a_long, 12.0, 160, 70, tt1, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

        remaining = budget - (time.monotonic() - t_global)
        if remaining > 40.0:
            tt2 = min(remaining * 0.25, 70.0)
            if tt2 > 15.0:
                cand = coord_polish(best_a, tt2, verbose=False)
                best_a, best_score = consider_candidate(best_a, best_score, cand)

    # Final polish.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 60.0:
        tt = min(remaining * 0.35, 150.0)
        if tt > 25.0:
            cand = lfgbbs_optimize(best_a, 40.0, 320, 70, tt, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

    remaining = budget - (time.monotonic() - t_global)
    if remaining > 40.0:
        tt = min(remaining * 0.35, 100.0)
        if tt > 18.0:
            cand = mgd_polish(best_a, tt, top_k=80, eta0=0.002, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

    remaining = budget - (time.monotonic() - t_global)
    if remaining > 30.0:
        tt = min(remaining * 0.45, 80.0)
        if tt > 15.0:
            cand = coord_polish(best_a, tt, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, cand)

    # Final forced validation and safety check.
    if best_a is None:
        i = np.arange(7000, dtype=np.float64)
        best_a = norm_sum1(i + 1.0)
    else:
        best_a = norm_sum1(best_a)

    final_score = safe_evaluate(best_a)
    if np.isfinite(final_score):
        best_score = float(min(final_score, best_score))

    elapsed = time.monotonic() - t_global
    print(f"FINAL: n={len(best_a)}, best_score={best_score:.6f}, elapsed={elapsed:.1f}s")

    return [float(x) for x in best_a.tolist()]
```
