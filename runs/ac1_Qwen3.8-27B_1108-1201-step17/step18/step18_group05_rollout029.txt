def peak_coordinate_descent(a0, time_limit, n_top=15, rng_seed=200):
    """Targeted coordinate descent focusing on the highest lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start > j_end:
                continue
            
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-16:
                continue
            
            # Find the source (highest contribution) and a good destination
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Compute sensitivity: how much does each coordinate appear in top lags?
            sens = np.zeros(n)
            for kl in top_lags[:8]:
                kli = int(kl)
                s2 = max(0, kli - n + 1)
                e2 = min(n - 1, kli)
                idx2 = np.arange(s2, e2 + 1)
                sens[idx2] += a[kli - idx2]
            
            # Find low-sensitivity destinations
            # Invert source's contribution to this lag
            # Moving mass from src to dst changes (a*a)_ki by:
            # delta = -2*amt*a[ki-src] + 2*amt*a[ki-dst] (approximately, for small amt)
            # We want delta < 0, i.e., a[ki-dst] < a[ki-src]
            # But also need to not increase other lags.
            
            ki_src = a[ki - src] if 0 <= ki - src < n else 0.0
            # Actually ki - src = ki - (some_j) and we know src is in [j_start, j_end]
            # so ki - src is in [ki-j_end, ki-j_start] which should be valid
            
            # Try moving to low-sensitivity, low-value positions
            candidates = np.where((sens < np.percentile(sens, 20)) & (np.arange(n) != src))[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Sort by a value (prefer moving to low-value positions)
            candidates = candidates[np.argsort(a[candidates])[:20]]
            
            found_move = False
            for dst in candidates:
                if found_move:
                    break
                for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        found_move = True
                        break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = max(1, n // 100)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m_new = fast_autocorr(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0
    
    return best_a, best_m

def spectral_rebalance(a0, time_limit, rng_seed=300):
    """Use spectral methods to find directions that flatten the autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

        ac = fast_autocorr(a)
        ac_max = ac.max()
        
        # Compute "excess" at each lag
        target = ac.mean()  # average level
        excess = np.maximum(ac - target, 0)
        
        # Create a "pressure" field: lags with high excess push back
        # Gradient direction: reduce mass at positions that contribute to high-excess lags
        p = excess / (excess.sum() + 1e-30)
        g = 2.0 * sp_correlate(p, a, mode='valid')
        g_tangent = g - g.mean()

            if m_trial < best_trial_m:

            a = a / a.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-temperature smoothed gradient with adaptive step size and backtracking
2. Peak-targeted coordinate descent with sensitivity-based destination selection
3. Spectral excess-driven optimization
4. Multi-length exploration with interpolation

import numpy as np
import time
from scipy.signal import correlate as sp_correlate


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def multi_temp_gradient(a, ac, taus):
    """Compute gradient as weighted combination of smoothed gradients at different temperatures."""
    n = len(a)
    grad = np.zeros(n)
    total_w = 0.0
    for tau in taus:
        ac_max = ac.max()
        if ac_max < 1e-30:
            continue
        p = np.exp(tau * (ac - ac_max))
        p_sum = p.sum()
        if p_sum < 1e-300:
            continue
        p = p / p_sum
        w = tau
        g = 2.0 * sp_correlate(p, a, mode='valid')
        grad += w * g
        total_w += w
    if total_w > 0:
        grad /= total_w
    return grad

        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:n_top]

        # Compute sensitivity for destination selection
        sens = np.zeros(n)
        for kl in top_lags[:10]:
            kli = int(kl)
            s2 = max(0, kli - n + 1)
            e2 = min(n - 1, kli)
            if s2 > e2:
                continue
            idx2 = np.arange(s2, e2 + 1)
            sens[idx2] += a[kli - idx2]

        improved = False

        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start > j_end:
                continue

            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[ki - idx]

            if contrib.max() < 1e-16:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Find low-sensitivity destinations sorted by a value
            mask = (np.arange(n) != src) & (sens < np.percentile(sens[sens > 0], 25) if (sens > 0).any() else np.inf)
            candidates = np.where(mask)[0]
            if len(candidates) < 5:
                candidates = np.where(np.arange(n) != src)[0]
            candidates = candidates[np.argsort(a[candidates])[:25]]

            found_move = False
            for dst in candidates:
                if found_move or time.monotonic() - t0 > time_limit:
                    break
                for frac in [0.001, 0.002, 0.005, 0.01, 0.02]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        found_move = True
                        break

        if not improved:
            stall += 1
            if stall > 50:
                # Multi-coordinate perturbation
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.008)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                stall = 0

    return best_a, best_m


def spectral_excess_opt(v, time_limit, rng_seed):
    """Optimize by targeting lags that exceed the mean autocorrelation level."""
    n = len(v)
    a = norm_sum1(v)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_mean = ac.mean()

        # Excess above mean
        excess = np.maximum(ac - ac_mean, 0.0)
        if excess.sum() < 1e-30:
            break
        p = excess / excess.sum()

        g = 2.0 * sp_correlate(p, a, mode='valid')
        g_tangent = g - g.mean()

        best_alpha = 0.0
        best_trial_m = ac_max
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            a_trial = project_simplex(a - alpha * g_tangent)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m - 1e-18:
                best_trial_m = m_trial
                best_alpha = alpha

        if best_alpha > 0:
            a = project_simplex(a - best_alpha * g_tangent)
            m_new = fast_autocorr(a).max()
            if m_new < best_m - 1e-16:
                best_m = m_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > 30:
                pert = rng.normal(0, 0.003, n) * a
                a = np.maximum(a + pert, 0.0)
                a = a / a.sum()
                stall = 0

    return best_a, best_m


def pair_swap_descent(v, time_limit, n_top, rng_seed):
    """Try 2-coordinate moves that simultaneously adjust two elements of a."""
    n = len(v)
    a = norm_sum1(v)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 600:
        ac = fast_autocorr(a)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False
        # Try pair moves on the top contributing pairs
        for k in top_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start > j_end:
                continue
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[ki - idx]

            # Sort by contribution, take top 5 pairs
            top_pairs = np.argsort(contrib)[::-1][:5]

            for pi in top_pairs:
                if time.monotonic() - t0 > time_limit:
                    break
                j1 = idx[pi]
                j2 = ki - j1
                if j2 < 0 or j2 >= n or j2 == j1:
                    continue
                v1 = a[j1]
                v2 = a[j2]
                if v1 < 1e-14 and v2 < 1e-14:
                    continue

                # Try reducing the larger of v1, v2 and moving mass to a low-sensitivity position
                if v1 >= v2:
                    src = j1
                    src_v = v1
                else:
                    src = j2
                    src_v = v2

                if src_v < 1e-14:
                    continue

                # Find destination
                candidates = np.where((np.arange(n) != src) & (a < np.median(a)))[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates = candidates[np.argsort(a[candidates])[:10]]

                for dst in candidates:
                    if time.monotonic() - t0 > time_limit:
                        break
                    for frac in [0.002, 0.005, 0.01]:
                        amt = src_v * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

        if not improved:
            stall += 1
            if stall > 40:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                stall = 0

    return best_a, best_m


def make_u_shape(n, sharpness):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)
