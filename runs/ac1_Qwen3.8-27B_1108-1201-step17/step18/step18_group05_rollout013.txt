def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

"""

import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    tmp = u * np.arange(1, n + 1)
    cond = tmp > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient_optimize(a0, time_limit, verbose=False, rng_seed=42):
    """Three-phase smoothed projected gradient with Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Phase parameters: (tau_start, tau_end, eta, max_iters_fraction, momentum)
    phases = [
        (3.0, 10.0, 1.5, 0.25, 0.9),    # Phase 1: aggressive
        (10.0, 40.0, 0.8, 0.5, 0.9),    # Phase 2: balanced
        (40.0, 200.0, 0.3, 1.0, 0.8),   # Phase 3: fine-tuning
    ]
    
    momentum = 0.0
    v_mom = np.zeros(n)  # momentum buffer
    
    total_elapsed_target = time_limit
    
    for phase_idx, (tau_s, tau_e, eta_max, frac, mom) in enumerate(phases):
        phase_start = time.monotonic()
        phase_limit = frac * total_elapsed_target - (phase_start - t0)
        if phase_limit < 2:
            break
        
        eta = eta_max
        stall = 0
        iter_count = 0
        
        while time.monotonic() - phase_start < phase_limit:
            elapsed_in_phase = time.monotonic() - phase_start
            progress = min(1.0, elapsed_in_phase / max(phase_limit, 1e-6))
            tau = tau_s * (tau_e / tau_s) ** progress
            
            ac = fast_autocorr(a)
            ac_max = ac.max()

            # For numerical stability

            grad_full = sp_correlate(p, a, mode='full')[:n]
            grad = 2.0 * grad_full
            
            # Project onto tangent space of simplex
            grad_t = grad - grad.mean()
            
            # Nesterov momentum
            v_mom = mom * v_mom + (1 - mom) * grad_t

            # Step

            a_new = project_simplex(a_trial)

                if stall > 50:

def lp_frank_wolfe_fast(a0, time_limit, solver, verbose=False, K_init=100):
    """Fast LP-Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    K = K_init
    max_K = min(2 * n - 1, 2000)
    iter_count = 0
    stale_LP = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix T (K x n)
        # T[ii, j] = a[ki - j] if 0 <= ki - j < n, else 0
        T = np.zeros((K_sel, n))
        for ii, ki in enumerate(binding_idx):
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, max_K)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stale_LP += 1
            if stale_LP > 5:
                K = min(K + 100, max_K)
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            stale_LP += 1
            continue
        g_dir = b_val / b_sum
        stale_LP = 0
        
        # Backtracking line search
        base_m = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_m
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()
        
        iter_count += 1
        if iter_count % 5 == 0:
            K = min(K + 20, max_K)
        
        if verbose and iter_count % 20 == 0:
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count}: J={J:.6f}, K={K_sel}")
    
    return best_a, best_m

        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        for ki in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(ki)
            # Get contributing pairs
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the top contributing source
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find low-contribution destinations
            all_idx = np.arange(n)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:min(5, len(top_k))]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = all_idx
            
            # Score candidates by their autocorrelation sensitivity
            # Lower sensitivity = better destination
            sens = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                idx2 = np.arange(s2, e2 + 1)
                sens[idx2] += a[k2i - idx2]
            
            # Try top few low-sensitivity candidates
            top_cands = candidates[np.argsort(sens[candidates])[:5]]
            
            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.05, 0.02, 0.01, 0.08, 0.03, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
            
            if found and improved:
                break  # Restart the loop with new ac
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    PeakPair final: J={J:.6f}")
    return best_a, best_m

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)

def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)

def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    return norm_sum1(a)

def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
