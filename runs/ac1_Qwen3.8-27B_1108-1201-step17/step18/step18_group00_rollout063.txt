def fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def proj_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    maxrho = int(rho[-1])
    theta = (cssv[maxrho] - 1.0) / (maxrho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return out / s


def smooth_max(a, beta):
    """Compute softmax weights over array a."""
    a = np.asarray(a, dtype=np.float64)
    am = a - a.max()
    exp_a = np.exp(beta * am)
    s = exp_a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return exp_a / s


def gradient_of_weighted_conv(a, q):
    """
    Compute gradient of sum_k q[k]*C[k] where C = conv(a,a).
    Result = 2 * IFFT(FFT(q) * conj(FFT(a)))[1:n]
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    Q = np.fft.rfft(q, fft_size)
    # IFFT of Q * conj(A) gives cross-correlation q * a
    # The gradient is 2 * (this result)[1:n]
    cross = np.fft.irfft(Q * np.conj(A), fft_size)
    grad = 2.0 * cross[1:n + 1]
    return grad


def soft_argmax_qp(a0, time_limit, beta_init=50.0, beta_final=500.0,
                   inner_iters=8, eta_init=1.0, verbose=True):
    """
    Soft-argmax QP iteration:

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    outer_count = 0
    stall = 0
    max_stall = 3000

    eta = eta_init
    for outer in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        C = fast_autocorr(a)
        curr_m = C.max()

        if curr_m < best_m - 1e-15:

            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive temperature

        if stall > 100:
            beta = min(beta * 1.1, beta_final * 2)
