        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lse_mirror_descent(a0, time_limit, beta_start=20.0, beta_end=400.0,
                       c0=3.0, n_restarts=4, inner_iters=8000, verbose=True):
    """
    Mirror descent on log-sum-exp smoothed objective with heavy-ball momentum.
    Gradient computed efficiently via FFT cross-correlation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    size = 2 * n - 1
    # Need fft_size >= 3n-2 for cross-correlation
    fft_size_corr = 1 << (3 * n - 2 - 1).bit_length()
    fft_size_conv = 1 << (size - 1).bit_length()

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        if restart > 0:
            noise = 0.025 / (restart + 1)
            pert = rng.normal(0, noise, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            s = a.sum()
            a = a / s

        x = a.copy()
        x_prev = a.copy()
        improved_since_restart = 0

        for t in range(1, inner_iters + 1):
            if time.monotonic() - t0 > time_limit:
                break

            progress = t / inner_iters
            beta = beta_start + (beta_end - beta_start) * (progress ** 0.4)

            # Heavy ball extrapolation
            if t > 1:
                mu = (t - 1.0) / (t + 1.5)
                y = x + mu * (x - x_prev)
                y = np.maximum(y, 1e-15)
                s_y = y.sum()
                y = y / s_y
            else:
                y = x.copy()

            # Compute autocorrelation
            ac = fast_autocorr(y)

            # Log-sum-exp weights (numerically stable)
            ac_clipped = np.maximum(ac, 1e-30)
            ac_max = ac_clipped.max()
            log_w = beta * (ac_clipped - ac_max) - np.log(ac_clipped)
            # Actually just use: w[k] = exp(beta * (ac[k] - max)) / sum
            exp_beta = np.exp(beta * (ac - ac_max))
            w = exp_beta / exp_beta.sum()

            # Gradient: grad[j] = 2 * sum_m w[m+j] * a[m] = 2 * cross_corr(w, y)[j]
            # Using FFT: cross_corr = ifft(conj(FFT(w)) * FFT(y))
            w_pad = np.zeros(fft_size_corr, dtype=np.float64)
            w_pad[:size] = w
            y_pad = np.zeros(fft_size_corr, dtype=np.float64)
            y_pad[:n] = y
            Wf = np.fft.rfft(w_pad)
            Yf = np.fft.rfft(y_pad)
            xcorr = np.fft.irfft(np.conj(Wf) * Yf, fft_size_corr)
            grad = 2.0 * xcorr[:n]

            # Mirror descent update
            eta = c0 / (np.sqrt(t) + 1.0)
            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            a_new = a_new / s_new

            x_prev = x
            x = a_new

            M_new = fast_autocorr(x).max()
            if M_new < best_m - 1e-14:
                best_m = M_new
                best_a = x.copy()
                improved_since_restart += 1

            if verbose and t % 1000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    LSE-MD r=" + str(restart) + " t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " beta=" + repr(round(beta, 1)))

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LSE-MD restart " + str(restart) + " done (" + str(elapsed) + "s): J=" + repr(J) + " improvements=" + str(improved_since_restart))

        if improved_since_restart == 0 and restart > 0:
            break

    return best_a, best_m


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw_improved(a0, time_limit, solver, initial_K=400, max_K=3000,
                   add_per_iter=40, verbose=True):
    """LP-FW with golden-section line search and adaptive constraint management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 1200
    iter_count = 0

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stuck for " + str(no_improve) + " iters, stopping")
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(constraint_list[i]) for i in range(min(initial_K, len(constraint_list))))
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for violations and add constraints
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Also periodically refresh: add current top peaks
        if t % 50 == 0:
            top_peaks = np.argsort(ac)[::-1][:initial_K]
            for k in top_peaks[:50]:
                active_set.add(int(k))

        # Golden section line search for optimal step size
        if t == 1:
            best_alpha = 1.0
        else:
            golden_ratio = (np.sqrt(5.0) - 1.0) / 2.0
            inv_phi = golden_ratio
            a_lo = 0.0
            a_hi = 1.0
            d = a_hi - a_lo
            c = a_hi - inv_phi * d
            b_pt = a_lo + inv_phi * d

            def trial_M(alpha_val):
                trial = (1.0 - alpha_val) * y + alpha_val * g_dir
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    return float('inf')
                trial = trial / s_t
                return fast_autocorr(trial).max()

            M_c = trial_M(c)
            M_b = trial_M(b_pt)

            for _ in range(12):
                d = a_hi - a_lo
                if d < 1e-6:
                    break
                if M_c < M_b:
                    a_hi = b_pt
                    b_pt = c
                    M_b = M_c
                    c = a_hi - inv_phi * (a_hi - a_lo)
                    M_c = trial_M(c)
                else:
                    a_lo = c
                    c = b_pt
                    M_c = M_b
                    b_pt = a_lo + inv_phi * (a_hi - a_lo)
                    M_b = trial_M(b_pt)

            best_alpha = 0.5 * (a_lo + a_hi)

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        y = y_new
        iter_count += 1

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(elapsed) + "s): J=" + repr(J) + " iters=" + str(iter_count))
    return best_a, best_m


def peak_targeted_cd(a0, time_limit, verbose=True, rng_seed=42):
    """
    Coordinate descent that specifically targets the top autocorrelation peaks.
    For each peak lag k*, find the pair (i,j) with i+j=k* contributing most,
    and move mass from the larger element to a low-impact position.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 2000
    n_fft = 1 << (2 * n - 2 - 1).bit_length()

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Compute "damage score" for each potential destination
            # We want to minimize the increase in max autocorrelation
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])
            if dst == src:
                continue

            # Try different step sizes
            found = False
            for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                amt = src_val * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 200:
                # Random exploration
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Random perturbation to escape
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PtcCD (" + str(elapsed) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def parametric_init(n, best_init=None, best_score=float('inf')):
    """Generate diverse parametric initializations."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init if best_init is not None else np.ones(n) / n
    best_val = best_score

    families = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        families.append((i_arr + 1.0) ** p)
        families.append((n - i_arr) ** p)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            families.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            families.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        families.append(x ** p + (1.0 - x) ** p)
    for freq in [1, 2, 3, 4]:
        for ph in [0.0, 0.5]:
            families.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.01))
    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50]:
            families.append(1.0 / (1.0 + np.exp(-s * (x - c))))
    for rate in [0.5, 1, 2, 5]:
        families.append(np.exp(-rate * x) + np.exp(rate * (x - 1.0)))
    families.append(np.sin(np.pi * x) ** 2)
    families.append(np.sin(np.pi * x) ** 4)
    families.append(1.0 - np.abs(2.0 * x - 1.0))

    rng = np.random.default_rng(7)
    for trial in range(50):
        families.append(rng.random(n) ** rng.uniform(0.5, 3.0))

    for a_raw in families:
        a = norm_sum1(a_raw)
        v = 2.0 * n * fast_autocorr(a).max()
        if v < best_val:
            best_val = v
            best_a = a.copy()

    return best_a, best_val


def symmetrize(a):
    """Force symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = np.zeros(n)
    for i in range(n):
        a_sym[i] = (a[i] + a[n - 1 - i]) / 2.0
    return norm_sum1(a_sym)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score) and p_score < 2.0:
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        pass

    if best_seq is None:
        n_base = 6000
        best_seq, best_score = parametric_init(n_base)
        print("Init: n=" + str(len(best_seq)) + ", score=" + repr(2.0 * len(best_seq) * fast_autocorr(best_seq).max()))

    n0 = len(best_seq)

    def try_update(best_seq, best_score, a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print("*** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")

    # Phase 1: Quick parametric refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Parametric (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, best_seq, best_score)
        try_update(best_seq, best_score, a_param, "param_n" + str(n0))
        best_seq, best_score = (best_seq, best_score)

    # Phase 2: LSE Mirror Descent (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LSE Mirror Descent (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        a_out, _ = lse_mirror_descent(norm_sum1(best_seq), t2,
                                      beta_start=20.0, beta_end=400.0,
                                      c0=3.0, n_restarts=4, inner_iters=8000,
                                      verbose=True)
        try_update(best_seq, best_score, a_out, "lse_md_n" + str(n_best))

    # Phase 3: LP-FW with golden section
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW Golden (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 60 and n_best <= 8000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t3, solver,
                                  initial_K=400, max_K=3000,
                                  add_per_iter=40, verbose=True)
        try_update(best_seq, best_score, a_out, "lp_gs_n" + str(n_best))

    # Phase 4: Peak-targeted coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak-targeted CD (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 30:
        a_out, _ = peak_targeted_cd(norm_sum1(best_seq), t4, verbose=True, rng_seed=42)
        try_update(best_seq, best_score, a_out, "ptcd_n" + str(n_best))

    # Phase 5: LSE again (second pass, sharper)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LSE MD pass 2 (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40:
        a_out, _ = lse_mirror_descent(norm_sum1(best_seq), t5,
                                      beta_start=50.0, beta_end=600.0,
                                      c0=2.5, n_restarts=3, inner_iters=6000,
                                      verbose=True)
        try_update(best_seq, best_score, a_out, "lse2_n" + str(n_best))

    # Phase 6: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length ===")
    try_lengths = []
    for delta in [-1000, -500, 500, 1000, 2000]:
        cand = n_best + delta
        if 3000 <= cand <= 9000:
            try_lengths.append(cand)

    for new_n in try_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 0)) + "s)")
        a_long = interpolate_sequence(best_seq, new_n)
        # Quick LSE
        t6 = min(remaining * 0.20, 180.0)
        if t6 > 30:
            a_out, _ = lse_mirror_descent(a_long, t6,
                                          beta_start=30.0, beta_end=300.0,
                                          c0=3.0, n_restarts=2, inner_iters=5000,
                                          verbose=False)
            try_update(best_seq, best_score, a_out, "multilens_n" + str(new_n))

    # Phase 7: Final LP-FW polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 8000:
        t7 = min(remaining * 0.35, 300.0)
        if t7 > 30:
            a_out, _ = lp_fw_improved(norm_sum1(best_seq), t7, solver,
                                      initial_K=500, max_K=3500,
                                      add_per_iter=50, verbose=True)
            try_update(best_seq, best_score, a_out, "lp_final_n" + str(n_best))

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final CD polish (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t8 = min(remaining * 0.50, 200.0)
        if t8 > 15:
            a_out, _ = peak_targeted_cd(norm_sum1(best_seq), t8, verbose=False, rng_seed=314)
            try_update(best_seq, best_score, a_out, "cd_final_n" + str(n_best))

    # Final LSE micro-pass with very sharp beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print("\n=== Phase 9: Sharp LSE micro-pass (n=" + str(n_best) + ") ===")
        t9 = min(remaining * 0.50, 200.0)
        if t9 > 20:
            a_out, _ = lse_mirror_descent(norm_sum1(best_seq), t9,
                                          beta_start=100.0, beta_end=800.0,
                                          c0=2.0, n_restarts=2, inner_iters=4000,
                                          verbose=False)
            try_update(best_seq, best_score, a_out, "sharp_lse_n" + str(n_best))

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
