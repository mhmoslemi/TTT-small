def evaluate_sequence(seq):
    if not isinstance(seq, list):
        return np.inf
    try:
        sequence = np.asarray([float(x) for x in seq], dtype=np.float64)
        sequence = np.maximum(sequence, 0.0)
        sequence = np.minimum(sequence, 1000.0)
        n = len(sequence)
        convolve_result = np.convolve(sequence, sequence)
        max_value = np.max(convolve_result)
        sum_value = np.sum(sequence)
        return float(2.0 * n * max_value / (sum_value ** 2))
    except:
        return np.inf

def norm_sum1(seq):
    """Normalize sequence to have sum 1 while enforcing non-negativity and clipping."""
    if not isinstance(seq, np.ndarray) and not hasattr(seq, 'sum'):
        try:
            npa = np.asarray(seq, dtype=np.float64)
        except:
            return None
        npa = npa + np.finfo(float).eps
        sum_npa = npa.sum()
        if sum_npa < 1e-15:
            return np.ones_like(npa) / len(npa)
        factor = 1 / sum_npa if sum_npa < 1 else np.min(npa) / (sum_npa / len(npa) * (1 if len(npa) == 1 else ...))
        return np.clip(npa * factor, 0, 1000)
    else:
        return None
