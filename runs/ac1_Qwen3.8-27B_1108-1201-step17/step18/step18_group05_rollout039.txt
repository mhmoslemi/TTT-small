def fast_autocorr(a):
    """FFT-based autocorrelation (full)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

from scipy.signal import correlate as sp_correlate

def fast_autocorr(a):
    """FFT-based full autocorrelation."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def smooth_gradient(a, tau):
    """Exact gradient of smooth approximation of max autocorrelation.
    
    Returns (gradient, softmax_weights, autocorrelation).
    """
    n = len(a)
    C = fast_autocorr(a)
    C_max = C.max()
    shifted = tau * (C - C_max)
    max_shifted = shifted.max()
    if max_shifted > 500:
        shifted = shifted - max_shifted
    exp_C = np.exp(shifted)
    total = exp_C.sum()
    if total < 1e-300:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p = exp_C / total
    # grad[j] = 2 * sum_m a[m] * p[j+m] = 2 * correlate(p, a, 'valid')[j]
    grad_half = sp_correlate(p, a, mode='valid')
    grad = 2.0 * grad_half
    return grad, p, C

def lp_frank_wolfe_step(a, M, active_idx, solver):
    """One LP-FW step: find direction b that doesn't increase autocorrelation peak."""
    n = len(a)
    K = len(active_idx)
    if K == 0 or K > 2*n-1:
        return None
    
    # Build constraint matrix: A_constr @ b <= M for each active lag
    # For lag k: sum_j a[k-j] * b[j] <= M (where a[m]=0 outside [0,n-1])
    A_constr = np.zeros((K, n))
    for ii, k in enumerate(active_idx):
        ki = int(k)
        # b[j] appears with coefficient a[ki-j] when 0 <= ki-j < n
        j_vals = np.arange(max(0, ki-n+1), min(n, ki+1))
        if len(j_vals) == 0:
            continue
        A_constr[ii, j_vals] = a[ki - j_vals]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_constr @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-10:
        return None
    return b_val / b_sum

def get_solver():
    """Get a working LP solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)

def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)

def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    return norm_sum1(a)

def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def smooth_gradient_optimize(a0, time_limit, tau_start=20.0, tau_end=300.0,
                              eta_start=1.0, solver=None, lp_every=40,
                              top_K_init=200, verbose=True, rng_seed=42):
    """Main optimization: smoothed gradient with adaptive temperature + periodic LP-FW."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = eta_start
    stall = 0
    stall_limit = 200
    iter_count = 0
    K = top_K_init
    K_max = min(2 * n - 1, 3000)
    
    while time.monotonic() - t0 < time_limit:
        # Adaptive temperature
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute gradient
        grad, p, C = smooth_gradient(a, tau)
        M_curr = C.max()
        
        # Project gradient onto tangent space
        grad_tangent = grad - grad.mean()
        
        # Gradient step with backtracking line search
        a_new = a.copy()
        best_trial_M = M_curr
        eta_trial = eta
        for ls in range(10):
            a_trial = a - eta_trial * grad_tangent
            a_trial = project_simplex(a_trial)
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                a_new = a_trial
            eta_trial *= 0.5
            if eta_trial < 1e-12:
                break
        
        a = a_new
        M_new = fast_autocorr(a).max()
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a.copy()
            stall = 0
            eta = min(eta * 1.05, 5.0)
        else:
            stall += 1
            if stall > 50:
                eta = max(eta * 0.8, 0.01)
            if stall > stall_limit:
                # Perturbation
                pert = rng.normal(0, 0.01, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a_p = a_p / a_p.sum()
                M_p = fast_autocorr(a_p).max()
                if M_p < best_M:
                    a = a_p
                    best_a = a_p.copy()
                    best_M = M_p
                    stall = 50
                    eta = eta_start * 0.5
                else:
                    a = best_a.copy()
                    stall = stall_limit // 2
                    K = min(K + 50, K_max)
        
        # Periodic LP-FW step
        iter_count += 1
        if solver is not None and iter_count % lp_every == 0 and stall > 20:
            active_idx = np.argsort(C)[::-1][:K]
            b_dir = lp_frank_wolfe_step(a, M_curr, active_idx, solver)
            if b_dir is not None:
                # Line search on LP direction
                best_alpha_M = M_curr
                best_alpha = 0.1
                for alpha in [0.3, 0.2, 0.1, 0.05, 0.5]:
                    a_lp = (1 - alpha) * a + alpha * b_dir
                    a_lp = np.maximum(a_lp, 0.0)
                    s_lp = a_lp.sum()
                    if s_lp < 1e-15:
                        continue
                    a_lp = a_lp / s_lp
                    M_lp = fast_autocorr(a_lp).max()
                    if M_lp < best_alpha_M:
                        best_alpha_M = M_lp
                        best_alpha = alpha
                a = (1 - best_alpha) * a + best_alpha * b_dir
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_lp = fast_autocorr(a).max()
                if M_lp < best_M:
                    best_M = M_lp
                    best_a = a.copy()
                    stall = 0
                K = min(K + 30, K_max)
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    SmoothGrad iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_M

def coordinate_polish(a0, time_limit, n_top=15, verbose=False, rng_seed=123):
    """Targeted coordinate descent on active lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        C = fast_autocorr(a)
        M_curr = C.max()
        top_lags = np.argsort(C)[::-1][:n_top]
        
        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Find the pair with highest contribution
            src_pos = idx[np.argmax(contrib)]
            src_val = a[src_pos]
            if src_val < 1e-14:
                continue
            
            # Find low-contribution destinations
            in_peak = np.zeros(n, dtype=bool)
            for kk in top_lags[:8]:
                s2 = max(0, int(kk) - n + 1)
                e2 = min(n, int(kk) + 1)
                in_peak[s2:e2] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Move mass to a random low-contribution position
            for attempt in range(3):
                dst = candidates[rng.integers(len(candidates))]
                for frac in [0.05, 0.02, 0.1, 0.01]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_pos] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-16:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0
        else:
            stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_M
