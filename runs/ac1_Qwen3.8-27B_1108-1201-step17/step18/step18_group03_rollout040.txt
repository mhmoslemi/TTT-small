"""
Local search by exact transfer/coordinate moves.

The objective is scale invariant after normalization sum(a)=1:
    score = 2*n*max(conv(a,a)).

The optimizer keeps the current convolution exactly and evaluates
single-coordinate scales and two-coordinate mass transfers in O(n) time.
Directions are selected from subgradients of the top convolution peaks,
with periodic broad/random escalation and length exploration.
"""

import numpy as np
import time


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def conv_sq_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    f = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(f * f, fft_size)[:size]
    c = np.maximum(c, 0.0)
    return c


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        if a.size == 0:
            return a
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    if abs(s - 1.0) > 1e-15:
        a = a / s
    return a


def score_array(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf")
    an = a / s
    c = conv_sq_fft(an)
    return float(2.0 * n * c.max())


def interpolate_seq(a, new_n):
    a = norm_sum1(a)
    old_n = int(a.size)
    if old_n == 0:
        return np.ones(int(new_n), dtype=np.float64)
    if old_n == int(new_n):
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, int(new_n))
    b = np.interp(new_x, old_x, a)
    return norm_sum1(b)


def parametric_candidates(n):
    out = []
    if n <= 0:
        return out
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in (0.35, 0.45, 0.50, 0.55, 0.60, 0.70, 0.80, 1.00):
        out.append(norm_sum1(np.power(i, -p)))

    for alpha in (0.40, 0.50, 0.60):
        arr = np.empty(n, dtype=np.float64)
        arr[0] = 1.0
        for k in range(1, n):
            arr[k] = arr[k - 1] * (float(k) + alpha - 1.0) / float(k)
        out.append(norm_sum1(arr))

    x = np.linspace(0.0, 1.0, n)
    for base_p in (0.5, 0.6):
        base = np.power(i, -base_p)
        for cut in (0.8, 0.9, 0.95):
            mask = 1.0 - (x / cut)
            mask = np.maximum(mask, 0.0)
            out.append(norm_sum1(base * mask))
    return out


def best_parametric(n):
    best = None
    best_sc = float("inf")
    for a in parametric_candidates(n):
        sc = score_array(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
    if best is None:
        i = np.arange(1, max(2, n + 1), dtype=np.float64)
        best = norm_sum1(np.power(i[:, :n] if i.size >= n else i, -0.5)) if n > 0 else np.ones(1)
    return best


def unique_sorted_floats(vals, lo, hi):
    out = []
    seen = set()
    for v in vals:
        try:
            v = float(v)
        except Exception:
            continue
        if not np.isfinite(v):
            continue
        if v < lo:
            v = float(lo)
        if v > hi:
            v = float(hi)
        if v not in seen:
            seen.add(v)
            out.append(v)
    out.sort()
    return out


def compute_h_direct(a, idx, weights):
    n = len(a)
    h = np.zeros(n, dtype=np.float64)
    for pos in range(len(idx)):
        wt = float(weights[pos])
        if wt == 0.0:
            continue
        k = int(idx[pos])
        if k < 0 or k >= 2 * n - 1:
            continue
        start = max(0, k - n + 1)
        end = min(n - 1, k)
        if start <= end:
            h[start:end + 1] += wt * a[k - end:k - start + 1]
    return h


def transfer_pair_raw(c, a, i, j, frac, inv_s2):
    i = int(i)
    j = int(j)
    if i == j:
        return float("inf")
    u = float(a[i])
    delta = u * float(frac)
    if delta <= 1e-300:
        return float("inf")
    if delta > u:
        delta = u

    n = len(a)
    Lc = len(c)
    b = c.copy()
    d = 2.0 * delta

    len1 = min(n, Lc - i)
    if len1 > 0:
        b[i:i + len1] -= d * a[:len1]

    len2 = min(n, Lc - j)
    if len2 > 0:
        b[j:j + len2] += d * a[:len2]

    d2 = delta * delta
    ki = 2 * i
    if ki < Lc:
        b[ki] += d2
    kj = 2 * j
    if kj < Lc:
        b[kj] += d2
    kj2 = i + j
    if kj2 < Lc:
        b[kj2] -= 2.0 * d2

    return float(b.max()) * float(inv_s2)


def apply_pair_transfer(c, a, i, j, frac):
    i = int(i)
    j = int(j)
    if i == j:
        return float(c.max())
    u = float(a[i])
    delta = u * float(frac)
    if delta > u:
        delta = u
    if delta <= 1e-300:
        return float(c.max())

    n = len(a)
    Lc = len(c)
    d = 2.0 * delta

    len1 = min(n, Lc - i)
    if len1 > 0:
        c[i:i + len1] -= d * a[:len1]

    len2 = min(n, Lc - j)
    if len2 > 0:
        c[j:j + len2] += d * a[:len2]

    d2 = delta * delta
    ki = 2 * i
    if ki < Lc:
        c[ki] += d2
    kj = 2 * j
    if kj < Lc:
        c[kj] += d2
    kj2 = i + j
    if kj2 < Lc:
        c[kj2] -= 2.0 * d2

    a[i] = max(0.0, u - delta)
    a[j] = a[j] + delta
    return float(c.max())


def single_scale_raw(c, a, i, factor, s0):
    i = int(i)
    u = float(a[i])
    x = u * float(factor)
    d = x - u
    if d == 0.0:
        return float(c.max()) / (s0 * s0)

    n = len(a)
    Lc = len(c)
    b = c.copy()

    len1 = min(n, Lc - i)
    if len1 > 0:
        b[i:i + len1] += (2.0 * d) * a[:len1]

    d2 = d * d
    ki = 2 * i
    if ki < Lc:
        b[ki] += d2

    s = s0 + d
    if s <= 1e-300:
        return float("inf")
    return float(b.max()) / (s * s)


def apply_single_scale(c, a, i, factor):
    i = int(i)
    u = float(a[i])
    x = u * float(factor)
    d = x - u
    if d == 0.0:
        return float(c.max())

    n = len(a)
    Lc = len(c)

    len1 = min(n, Lc - i)
    if len1 > 0:
        c[i:i + len1] += (2.0 * d) * a[:len1]

    d2 = d * d
    ki = 2 * i
    if ki < Lc:
        c[ki] += d2

    a[i] = max(0.0, x)
    return float(c.max())


def make_pairs_from_h(h, a, count, rng=None, src_top=8, dst_top=8, random_pairs=6):
    n = len(a)
    pairs = []
    seen = set()

    min_h = float(h.min())
    max_h = float(h.max())
    if max_h - min_h > 1e-30:
        src_scores = a * np.maximum(h - min_h, 0.0)
        src_order = np.argsort(src_scores)[::-1]
        dst_order = np.argsort(h)
    else:
        src_order = np.argsort(a)[::-1]
        dst_order = np.argsort(a)

    src = []
    for idx in src_order:
        idx = int(idx)
        if a[idx] > 1e-16 and len(src) < src_top:
            src.append(idx)

    dst = []
    for idx in dst_order:
        idx = int(idx)
        if len(dst) < dst_top:
            dst.append(idx)

    target = max(1, count - max(0, random_pairs))
    for ii in src:
        for jj in dst:
            if ii != jj and (ii, jj) not in seen:
                seen.add((ii, jj))
                pairs.append((ii, jj))
            if len(pairs) >= target:
                break
        if len(pairs) >= target:
            break

    if rng is not None:
        for _ in range(max(0, random_pairs) * 3):
            ii = int(rng.integers(n))
            jj = int(rng.integers(n))
            if ii != jj and a[ii] > 1e-16 and (ii, jj) not in seen:
                seen.add((ii, jj))
                pairs.append((ii, jj))

    k = 0
    while len(pairs) < count and k < 2 * n:
        ii = k % n
        jj = (ii + (k // n) + 1) % n
        if ii != jj and a[ii] > 1e-16 and (ii, jj) not in seen:
            seen.add((ii, jj))
            pairs.append((ii, jj))
        k += 1

    return pairs[:count]


def make_coord_candidates(h, a, count, rng=None):
    n = len(a)
    cand = []
    seen = set()

    for idx in (0, 1, 2, n - 3, n - 2, n - 1):
        if 0 <= idx < n and idx not in seen:
            seen.add(idx)
            cand.append(idx)

    mean_h = float(np.mean(h))
    dev = a * np.abs(h - mean_h)
    for idx in np.argsort(dev)[::-1]:
        idx = int(idx)
        if idx not in seen:
            seen.add(idx)
            cand.append(idx)
        if len(cand) >= count:
            break

    for idx in np.argsort(h):
        idx = int(idx)
        if idx not in seen:
            seen.add(idx)
            cand.append(idx)
        if len(cand) >= count:
            break

    if rng is not None:
        for _ in range(2 * count):
            idx = int(rng.integers(n))
            if idx not in seen:
                seen.add(idx)
                cand.append(idx)
            if len(cand) >= count:
                break

    k = 0
    while len(cand) < count and k < 2 * n:
        idx = k % n
        if idx not in seen:
            seen.add(idx)
            cand.append(idx)
        k += 1

    return cand[:count]


def refine_pair(c, a, i, j, frac_best, inv_s2):
    best_frac = float(frac_best)
    best_raw = float("inf")
    u = float(a[i])
    if u <= 1e-300:
        return best_frac, best_raw

    base = np.geomspace(1e-8, 1.0, 40)
    extras = [
        0.0, 1.0,
        best_frac,
        best_frac * 0.5, best_frac * 2.0,
        best_frac * 0.1, best_frac * 10.0,
        best_frac * 0.01, best_frac * 100.0,
    ]
    grid = unique_sorted_floats(list(base) + extras, 0.0, 1.0)

    for fr in grid:
        if fr <= 0.0:
            continue
        raw = transfer_pair_raw(c, a, i, j, fr, inv_s2)
        if raw < best_raw:
            best_raw = raw
            best_frac = float(fr)
    return best_frac, best_raw


def refine_coord(c, a, i, factor_best, s0):
    best_factor = float(factor_best)
    best_raw = float("inf")
    u = float(a[i])
    if u <= 1e-300:
        return best_factor, best_raw

    base = np.geomspace(1e-6, 8.0, 45)
    extras = [
        0.0, 8.0,
        best_factor,
        best_factor * 0.5, best_factor * 2.0,
        best_factor * 0.1, best_factor * 10.0,
        best_factor * 0.01, best_factor * 100.0,
    ]
    grid = unique_sorted_floats(list(base) + extras, 0.0, 8.0)

    for fr in grid:
        raw = single_scale_raw(c, a, i, fr, s0)
        if raw < best_raw:
            best_raw = raw
            best_factor = float(fr)
    return best_factor, best_raw


def local_pair_search(a0, time_limit, verbose=False, seed=11,
                      pairs_per=25, coord_count=25, refine_num=3):

    c = conv_sq_fft(a)
    cur_raw = float(c.max())
    best_a = a.copy()
    best_raw = cur_raw

    if time_limit <= 0.0:
        return best_a

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    accepts = 0
    wave = 0
    n = len(a)

        1e-7, 1e-6, 1e-5, 3e-5, 1e-4, 3e-4,

    )
    coord_factors = (
        0.0, 1e-3, 1e-2, 3e-2, 1e-1, 3e-1,
        6e-1, 1.0, 1.5, 2.5, 4.0, 8.0
    )

    if verbose:
        print(f"    local start: n={n}, score={2.0 * n * cur_raw:.9f}")

    while time.monotonic() - t0 < time_limit:
        s0 = float(a.sum())
        if not np.isfinite(s0) or s0 <= 1e-300:
            break
        if abs(s0 - 1.0) > 1e-11:
            a = a / s0
            c = c / (s0 * s0)

            ss = float(vv.sum())
            if ss <= 1e-300:

            else:

            w = vv
        else:
            idx = order
            scale = max(M * 5e-3, 1e-30)

            else:
                vv = vv / ss
            w = vv

        h = compute_h_direct(a, idx, w)
        pairs = make_pairs_from_h(h, a, max(4, int(pairs_per)), rng,
                                   src_top=8, dst_top=8, random_pairs=6)
        coords = make_coord_candidates(h, a, max(4, int(coord_count)), rng)

        inv_s2 = 1.0 / (s0 * s0)
        trials = []

        for ii, jj in pairs:
            if time.monotonic() - t0 > time_limit:

                else:
                    apply_single_scale(c, a, best_i, best_p)

                s_new = float(a.sum())
                if s_new > 1e-300:
                    a = a / s_new
                    c = c / (s_new * s_new)
                cur_raw = float(c.max())

                accepts += 1
                no_improve = 0
                if cur_raw < best_raw - 1e-18:
                    best_raw = cur_raw
                    best_a = a.copy()

                if accepts % 50 == 0:
                    a = norm_sum1(a)
                    c = conv_sq_fft(a)
                    cur_raw = float(c.max())
                    if cur_raw < best_raw:
                        best_raw = cur_raw
                        best_a = a.copy()

                if verbose and accepts % 100 == 0:
                    el = time.monotonic() - t0
                    print(f"    local accepts={accepts} t={el:.0f}s score={2.0 * n * cur_raw:.9f}")
        else:
            no_improve += 1

        wave += 1
        if verbose and wave % 300 == 0:
            el = time.monotonic() - t0
            print(f"    local wave={wave} t={el:.0f}s score={2.0 * n * cur_raw:.9f} no_improve={no_improve}")

    if verbose:
        el = time.monotonic() - t0
        print(f"    local done t={el:.0f}s best_score={2.0 * n * best_raw:.9f}")
    return norm_sum1(best_a)


def length_variant_candidates(n):
    out = []
    used = set()
    for m in (
        n - 1000, n + 1000,
        n - 2000, n + 2000,
        n - 500, n + 500,
        n - 3000, n + 3000,
        3000, 4000, 5000, 6000, 8000
    ):
        if 2500 <= m <= 9000 and abs(m - n) > 200 and m not in used:
            used.add(m)
            out.append(m)
        if len(out) >= 4:
            break
    return out


def propose_candidate():
    t_global = time.monotonic()
    budget = 1740.0

    best_a = None
    best_score = float("inf")

    try:
        parent = height_sequence_1
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-8:
            a = norm_sum1(p)
            sc = score_array(a)
            if np.isfinite(sc):
                best_a = a
                best_score = sc
                print(f"Parent loaded: n={len(a)}, score={sc:.9f}")
    except Exception:
        pass

    if best_a is None:
        n_fallback = 5200
        best_a = best_parametric(n_fallback)
        best_score = score_array(best_a)
        print(f"Generated fallback: n={len(best_a)}, score={best_score:.9f}")

    # Quick shape pool at the current length.
    n_cur = int(len(best_a))
    for a_cand in parametric_candidates(n_cur):
        sc = score_array(a_cand)
        if sc < best_score:
            best_score = sc
            best_a = a_cand
            print(f"Parametric improved: score={sc:.9f}")

    # Symmetrized version can occasionally help in a basin.
    try:
        a_sym = 0.5 * (best_a + best_a[::-1])
        sc = score_array(a_sym)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a_sym)
            print(f"Symmetrized improved: score={sc:.9f}")
    except Exception:
        pass

    # If the parent is long, inspect useful shorter lengths before optimizing.
    if n_cur > 7000:
        for m in (3500, 4500, 5500, 6500, 8000):
            av = interpolate_seq(best_a, m)
            sc = score_array(av)
            if sc < best_score:
                best_score = sc
                best_a = av
                print(f"Resized to n={m}: score={sc:.9f}")

    # Main local refinement.
    remaining = budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.55, 950.0)
    if t1 > 30.0:
        a1 = local_pair_search(
            best_a, t1, verbose=True, seed=17,
            pairs_per=30, coord_count=30, refine_num=3
        )
        sc = score_array(a1)
        if sc < best_score:
            best_score = sc
            best_a = a1
            print(f"Main local improved: score={sc:.9f}")

    # Length exploration from the best sequence.
    for mv in length_variant_candidates(len(best_a)):
        remaining = budget - (time.monotonic() - t_global)
        if remaining < 140.0:
            break
        t2 = min(remaining * 0.18, 320.0)
        if t2 < 60.0:
            break

        av = interpolate_seq(best_a, mv)
        sc = score_array(av)
        if sc < best_score:
            best_score = sc
            best_a = av
            print(f"Length {mv} initial improved: score={sc:.9f}")

        a2 = local_pair_search(
            av, t2, verbose=False,
            seed=10_000 + int(mv) % 10_000,
            pairs_per=22, coord_count=22, refine_num=2
        )
        sc = score_array(a2)
        if sc < best_score:
            best_score = sc
            best_a = a2
            print(f"Length {mv} local improved: score={sc:.9f}")

    # Final heavy polish.
    remaining = budget - (time.monotonic() - t_global) - 30.0
    if remaining > 40.0:
        a3 = local_pair_search(
            best_a, remaining, verbose=True, seed=99_983,
            pairs_per=40, coord_count=40, refine_num=4
        )
        sc = score_array(a3)
        if sc < best_score:
            best_score = sc
            best_a = a3
            print(f"Final local improved: score={sc:.9f}")

    final_a = norm_sum1(best_a)
    out_list = [float(max(0.0, x)) for x in final_a]

    try:
        validated = float(evaluate_sequence(out_list))
    except Exception:
        validated = score_array(final_a)

    if not np.isfinite(validated):
        n_fb = 5200
        final_a = norm_sum1(np.power(np.arange(1, n_fb + 1, dtype=np.float64), -0.5))
        out_list = [float(x) for x in final_a]
        validated = score_array(final_a)

    print(f"FINAL estimated score={validated:.9f}, n={len(out_list)}")
    return out_list
