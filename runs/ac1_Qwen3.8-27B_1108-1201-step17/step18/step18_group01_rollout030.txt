"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated projected subgradient (FISTA-style) on the convex max-convolution objective
2. Log-sum-exp smoothing with adaptive temperature
3. Multiple structured initializations (sine, linear, edge-peaked, parent)
4. Multi-length exploration with interpolation
5. Coordinate-level local search polish
Key insight: the problem is CONVEX (min of max of convex quadratics over simplex),
so any convergent method finds the global optimum for a fixed n.
"""

import numpy as np
import time


def fast_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2*n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Project v onto {x: x >= 0, sum(x) = s} using water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, s / n)
    rho = idx[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def conv_grad_at_lag(a, k):
    """Compute dC[k]/da for linear convolution C = conv(a,a).
    dC[k]/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        # For j in [lo, hi]: grad[j] = 2*a[k-j]
        # k-j ranges from k-lo down to k-hi
        grad[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def weighted_conv_grad(a, lags, weights):
    """Compute sum_k weights[k] * dC[k]/da."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(lags, weights):
        ki = int(k)
        if w < 1e-15:
            continue
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * w * a[ki - hi:ki - lo + 1][::-1]
    return grad


def fista_optimize(a0, time_limit, verbose=False, top_k=60, beta=None, label=""):
    """
    Nesterov-accelerated projected gradient on the (optionally smoothed) max-conv objective.
    If beta is None, uses hard max (subgradient of argmax lag + a few neighbors).
    If beta > 0, uses log-sum-exp smoothing.
    """
    n = len(a0)
    a = norm_sum1(a0)
    y = a.copy()
    t_mom = 1.0
    a_prev = a.copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    it = 0
    eta = 1.0 / (4.0 * n)  # initial step size (conservative)
    eta_min = 1e-8
    eta_decay = 0.9999
    stall = 0
    max_stall = 5000

    # If beta given, use smoothing
    use_smooth = (beta is not None and beta > 0)
    cur_beta = beta if use_smooth else 0

    while time.monotonic() - t0 < time_limit:
        # Compute convolution at Nesterov point y
        C = fast_conv(y)

        # Compute gradient
        if use_smooth and cur_beta > 1:
            # Softmax weighting
            bC = cur_beta * (C - C.max())
            p = np.exp(bC)
            p /= p.sum()
            # Top-K lags
            k_idx = np.argsort(p)[::-1][:min(top_k, 2 * n - 1)]
            w = p[k_idx]
            grad = weighted_conv_grad(y, k_idx, w)
            # Lipschitz estimate for smoothed function
            L_est = 4.0 * cur_beta * C.max() + 1.0 / (eta * n + 1e-10)
        else:
            # Hard max: use top few lags weighted by their values
            top_idx = np.argsort(C)[::-1][:min(top_k, 2 * n - 1)]
            top_vals = C[top_idx]
            # Weight by normalized values (favors the max)
            max_v = top_vals[0] + 1e-30
            w = (top_vals / max_v) ** 4  # sharp weighting toward max
            w /= w.sum()
            grad = weighted_conv_grad(y, top_idx, w)
            L_est = 4.0 * C.max() * len(top_idx) + 1.0

        # Adaptive step size with backtracking
        step = 1.0 / max(L_est, 1.0)
        # Clamp step
        step = max(step, eta_min)
        step = min(step, 1.0 / n)

        # FISTA update
        a_new = y - step * grad
        a_new = project_simplex(a_new, 1.0)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a_new.copy()

        a_prev = a.copy()
        a = a_new
        t_mom = t_new
        it += 1

        # Decay step size slowly
        eta *= eta_decay

        # Track best (check every 20 iterations)
        if it % 20 == 0:
            m = fast_conv(a).max()
            score = 2.0 * n * m
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                best_m = m
                stall = 0
            else:
                stall += 1

        # If using smoothing, increase beta periodically
        if use_smooth and it % 2000 == 0:
            cur_beta *= 1.3
            # Dampen momentum on beta change
            t_mom = 1.0
            y = a.copy()
            a_prev = a.copy()

        # Stalling: add small perturbation and reset momentum
        if stall > max_stall:
            rng = np.random.default_rng(it)
            a = a + rng.normal(0, 0.001 / n, n)
            a = norm_sum1(a)
            y = a.copy()
            a_prev = a.copy()
            t_mom = 1.0
            stall = 0
            eta = 1.0 / (4.0 * n)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA{label} it={it} ({elapsed:.0f}s): score={best_score:.6f}, step={step:.2e}")

    # Final check
    m = fast_conv(a).max()
    score = 2.0 * n * m
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a


def mgd_optimize(a0, time_limit, top_k=50, label=""):
    """
    Multiplicative Gradient Descent (mirror descent on simplex) with Nesterov momentum.
    Uses top-K lag subgradient with power weighting.
    """
    n = len(a0)
    a = norm_sum1(a0)
    a_prev = a.copy()
    y = a.copy()  # momentum buffer
    t_mom = 1.0
    best_a = a.copy()
    best_m = fast_conv(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    it = 0
    eta = 0.05
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit:
        C = fast_conv(y)
        M = C.max()

        # Top-K lags with power weighting
        k_max = min(top_k, 2 * n - 1)
        top_idx = np.argsort(C)[::-1][:k_max]
        top_vals = np.maximum(C[top_idx], 1e-30)
        # Power weighting: favor peaks more
        w = top_vals ** 3.0
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.ones(k_max) / k_max

        # Compute gradient
        grad = weighted_conv_grad(y, top_idx, w)

        # Scale gradient
        gmax = np.abs(grad).max()
        if gmax < 1e-20:
            break
        grad_scaled = grad / (gmax * n)  # normalize

        # Multiplicative update with momentum
        a_new = a * np.exp(-eta * grad_scaled)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a_new

        a_prev = a.copy()
        a = a_new
        t_mom = t_new
        it += 1

        # Track best
        if it % 10 == 0:
            m = fast_conv(a).max()
            score = 2.0 * n * m
            if score < best_score - 1e-13:
                best_score = score
                best_a = a.copy()
                best_m = m
                stall = 0
                eta = min(eta * 1.002, 0.15)  # slightly increase if making progress
            else:
                stall += 1

        # Adaptive step size
        if stall % 200 == 199 and stall > 0:
            eta *= 0.85
        if stall > max_stall:
            # Perturb and restart
            rng = np.random.default_rng(42 + it)
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = norm_sum1(a)
            y = a.copy()
            a_prev = a.copy()
            t_mom = 1.0
            stall = 0
            eta = 0.03

    m = fast_conv(a).max()
    score = 2.0 * n * m
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a


def coord_polish(a0, time_limit, label=""):
    """
    Coordinate descent: iteratively move mass from high-peak-contribution positions
    to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        C = fast_conv(a)
        M = C.max()
        # Top lags
        top_lags = np.argsort(C)[::-1][:min(8, 2 * n - 1)]
        improved = False

        for k in top_lags:
            if time.monotonic() - t0 > time_limit or improved:
                break
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo > hi:
                continue

            idx = np.arange(lo, hi + 1)
            pairs = a[idx] * a[ki - idx]
            if pairs.max() < 1e-20:
                continue

            # Find the contributing positions
            src = idx[np.argmax(pairs)]

            # Score all positions by their total peak contribution
            peak_contrib = np.zeros(n)
            for k2 in top_lags:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                if lo2 <= hi2:
                    idx2 = np.arange(lo2, hi2 + 1)
                    peak_contrib[idx2] += a[ki2 - idx2] if True else 0
                    # Actually: contribution of position j to C[k2] is a[k2-j]
                    peak_contrib[idx2] += 0  # we'll recompute below

            # Simpler: for each position j, its "peak involvement" = sum over top lags k of a[k-j]
            peak_involve = np.zeros(n)
            for k2 in top_lags:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                if lo2 <= hi2:
                    peak_involve[lo2:hi2 + 1] += a[k2i - hi2:k2i - lo2 + 1][::-1]

            # Find destination: low peak involvement
            # Exclude positions too close to src
            mask = np.ones(n, dtype=bool)
            exclude_range = max(1, n // 50)
            mask[max(0, src - exclude_range):min(n, src + exclude_range + 1)] = False
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            dst_candidates = candidates[peak_involve[candidates] < peak_involve[src]]
            if len(dst_candidates) == 0:
                dst_candidates = candidates

            dst = dst_candidates[np.argmin(peak_involve[dst_candidates])]
            if dst == src:
                continue

            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try different transfer fractions
            for frac in [0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < M - 1e-15:
                    a = a_trial
                    M = M_trial
                    improved = True
                    break

        if improved:
            if M < best_m:
                best_m = M
                best_a = a.copy()
                best_score = 2.0 * n * best_m
            stall = 0
        else:
            stall += 1
            # Random escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    best_score = 2.0 * n * best_m
                    stall = 0

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initializations(n, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)

    # Parent (interpolated to this length)
    if parent is not None and len(parent) > 0:
        p = interpolate_sequence(parent, n)
        inits.append(('parent', p))

    # Sine squared
    s1 = norm_sum1(np.sin(np.pi * x) ** 2)
    inits.append(('sine2', s1))

    # Sine (one half-wave, shifted)
    s2 = norm_sum1(np.sin(np.pi * x) ** 4)
    inits.append(('sine4', s2))

    # Linear increase
    lin = norm_sum1(i_arr + 1.0)
    inits.append(('linear', lin))

    # Edge-peaked (two Gaussians at edges)
    sigma = n * 0.15
    edge = np.exp(-(i_arr / sigma) ** 2) + np.exp(-((i_arr - n + 1) / sigma) ** 2)
    inits.append(('edge', norm_sum1(edge)))

    # Center-dipped (1 - center gaussian)
    center = 1.0 - np.exp(-((i_arr - n / 2) / (n * 0.2)) ** 2)
    center = np.maximum(center, 0.05)
    inits.append(('centerdip', norm_sum1(center)))

    # Quadratic (parabola centered)
    q = 1.0 - ((i_arr - n / 2) / (n / 2)) ** 2
    inits.append(('quad', norm_sum1(np.maximum(q, 0.01))))

    # Random
    rng = np.random.default_rng(0)
    inits.append(('rand', norm_sum1(rng.dirichlet(np.ones(n)))))

    # Sine with frequency 2
    s3 = norm_sum1(np.maximum(np.sin(2 * np.pi * x) ** 2, 0.01))
    inits.append(('sine2x', s3))

    # Cubic
    c = (i_arr / n) ** 3 + 0.01
    inits.append(('cubic', norm_sum1(c)))

    return inits


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting score: {best_score:.6f}, n={n_parent}")

    # === Phase 1: Try multiple initializations at parent length ===
    n1 = n_parent
    remaining = time_budget - (time.monotonic() - t_global)
    t_per_init = min(remaining * 0.03, 40.0)

    print(f"\n=== Phase 1: Multi-init at n={n1} (t_per={t_per_init:.0f}s) ===")
    if t_per_init > 5:
        inits = make_initializations(n1, parent=best_seq if parent is not None else None)
        for name, init_a in inits:
            if time.monotonic() - t_global > time_budget - 200:
                break
            sc = 2.0 * n1 * fast_conv(init_a).max()
            if sc < best_score:
                best_score = sc
                best_seq = init_a.copy()
                print(f"  init '{name}': score={sc:.6f} (NEW BEST)")
            else:
                print(f"  init '{name}': score={sc:.6f}")

    # === Phase 2: FISTA on best init ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: FISTA (n={n_best}, t={t2:.0f}s) ===")
    if t2 > 30:
        a_out = fista_optimize(norm_sum1(best_seq), t2, top_k=80, label=" main")
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** FISTA: {sc:.6f} (NEW BEST)")
        else:
            print(f"  FISTA: {sc:.6f}")

    # === Phase 3: MGD with momentum ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 3: MGD+momentum (n={n_best}, t={t3:.0f}s) ===")
    if t3 > 30:
        a_out = mgd_optimize(norm_sum1(best_seq), t3, top_k=50, label=" m1")
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** MGD m1: {sc:.6f} (NEW BEST)")
        else:
            print(f"  MGD m1: {sc:.6f}")

    # === Phase 4: FISTA with smoothing ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: FISTA smoothed (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 30:
        a_out = fista_optimize(norm_sum1(best_seq), t4, top_k=60, beta=50.0, label=" smooth")
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** FISTA smooth: {sc:.6f} (NEW BEST)")
        else:
            print(f"  FISTA smooth: {sc:.6f}")

    # === Phase 5: Coordinate polish ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 5: Coord polish (n={n_best}, t={t5:.0f}s) ===")
    if t5 > 20:
        a_out = coord_polish(norm_sum1(best_seq), t5, label=" c1")
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Coord: {sc:.6f} (NEW BEST)")
        else:
            print(f"  Coord: {sc:.6f}")

    # === Phase 6: Multi-length exploration ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length ===")

    # Try shorter lengths (sometimes shorter sequences have better per-element quality)
    shorter_lengths = [int(n_best * 0.7), int(n_best * 0.5)]
    shorter_lengths = [l for l in shorter_lengths if l >= 1000 and l < n_best]

    # Try longer lengths
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 8000 and l > n_best]

    all_trial_lengths = shorter_lengths + longer_lengths

    for new_n in all_trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick FISTA
        t6 = min(remaining * 0.25, 120.0)
        if t6 > 20:
            a_out = fista_optimize(a_long, t6, top_k=50, label=f" n{new_n}")
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** n={new_n}: {sc:.6f} (NEW BEST)")
            else:
                print(f"  n={new_n} FISTA: {sc:.6f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.20, 80.0)
            if t6b > 15:
                a_out = mgd_optimize(norm_sum1(best_seq), t6b, top_k=40, label=f" n{new_n} mgd")
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** n={new_n} MGD: {sc:.6f} (NEW BEST)")

    # === Phase 7: Final FISTA polish on best ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t7 = min(remaining * 0.35, 200.0)
        if t7 > 20:
            a_out = fista_optimize(norm_sum1(best_seq), t7, top_k=100, label=" final")
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** Final FISTA: {sc:.6f} (NEW BEST)")

    # === Phase 8: Final MGD ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 15:
            a_out = mgd_optimize(norm_sum1(best_seq), t8, top_k=40, label=" final_mgd")
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** Final MGD: {sc:.6f} (NEW BEST)")

    # === Phase 9: Final coord ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.50, 100.0)
        if t9 > 10:
            a_out = coord_polish(norm_sum1(best_seq), t9, label=" final_c")
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** Final coord: {sc:.6f} (NEW BEST)")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
