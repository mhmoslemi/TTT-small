    Minimizes (sum c[k]^p)^(1/p) where c = a*a.

        while time.monotonic() - t0 < time_limit:
            c = fast_autocorr(a)
            M = c.max()

            if M < best_max - 1e-15:
                best_max = M
                best_a = a.copy()
                p_patience = 0
            else:
                p_patience += 1

            if p_patience > max_patience:
                break

            # Compute gradient direction
            if p < 100:
                grad_dir = compute_pnorm_grad(c, a, p)
            else:
                # For large p, use top-k weighted subgradient for stability
                k_max = min(50, len(c))
                top_idx = np.argsort(c)[::-1][:k_max]
                top_vals = np.maximum(c[top_idx], 1e-30)
                w = top_vals ** 4
                ws = w.sum()
                if ws < 1e-30:
                    w = np.ones(k_max) / k_max
                else:
                    w = w / ws
                grad_dir = np.zeros(n)
                for wi, k in zip(w, top_idx):
                    ki = int(k)
                    js = max(0, ki - n + 1)
                    je = min(n, ki + 1)
                    if js < je:
                        idx = np.arange(js, je)
                        grad_dir[idx] += wi * a[ki - idx]

            # Project to tangent space of simplex
            grad_dir = grad_dir - grad_dir.mean()

            # Momentum
            v = momentum * v + (1 - momentum) * grad_dir
            v = v - v.mean()

            gnorm = np.sqrt(np.sum(v ** 2))
            if gnorm < 1e-18:
                break

            # Backtracking line search
            eta_try = p_eta
            a_cur = a
            f_cur = pnorm_value(c, min(p, 50))

            accepted = False
            for ls in range(15):
                a_try = a_cur * np.exp(-eta_try * v)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    eta_try *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = fast_autocorr(a_try)
                M_try = c_try.max()

                if M_try <= M + 1e-12:
                    a = a_try
                    p_eta = min(eta_try * 1.05, 0.5)
                    accepted = True
                    break
                else:
                    eta_try *= 0.5

            if not accepted:
                p_eta *= 0.7
                if p_eta < 0.001:
                    break

            p_iter += 1
            iter_count += 1

            if verbose and p_iter % 100 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max
                print(f"    pnrm p={p} iter={p_iter} ({elapsed:.0f}s): J={J:.6f}, eta={p_eta:.5f}")

    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_max
        print(f"    pnrm done ({elapsed:.0f}s): J={J:.6f}, total_iters={iter_count}")
    return best_a


def coord_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: move mass from peak-contributing positions to low-impact ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination: position with lowest peak contribution
            peak_contrib = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]

            candidates = np.argwhere(peak_contrib < peak_contrib[src] * 0.5).flatten()
            if len(candidates) == 0:
                candidates = np.argsort(peak_contrib)[:n // 4]
            dst = candidates[np.argmin(peak_contrib[candidates])]
            if dst == src:
                dst = candidates[np.argmin(peak_contrib[candidates[1:] if len(candidates) > 1 else candidates))]
                if dst == src:
                    continue

            for frac in [0.02, 0.04, 0.06, 0.10, 0.15]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < M - 1e-16:
                    a = a_trial
                    M = M_trial
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    elif stall < max_stall - 50:
                        stall -= 1  # Allow some exploration

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def block_optimize(a0, time_limit, block_size=32, verbose=False):
    """
    Block-level optimization: cyclically optimize each block using gradient steps.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    n_blocks = (n + block_size - 1) // block_size
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False

        # Process blocks in random order
        block_order = rng.permutation(n_blocks)
        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break

            start = bi * block_size
            end = min(n, start + block_size)
            if end - start < 2:
                continue

            # Compute gradient for this block
            c = fast_autocorr(a)
            k_max = int(np.argmax(c))
            # Focus on top lags
            top_lags = np.argsort(c)[::-1][:15]

            grad_block = np.zeros(end - start)
            for k in top_lags:
                ki = int(k)
                for i in range(start, end):
                    j = ki - i
                    if 0 <= j < n and j != i:
                        grad_block[i - start] += 2.0 * a[j]
                    elif 0 <= j < n and j == i:
                        grad_block[i - start] += 2.0 * a[j]

            grad_block = grad_block - grad_block.mean()
            gnorm = np.sqrt(np.sum(grad_block ** 2))
            if gnorm < 1e-18:
                continue

            # Try several step sizes
            for eta_b in [0.01, 0.03, 0.05]:
                new_block = a[start:end] * np.exp(-eta_b * grad_block / gnorm)
                new_block = np.maximum(new_block, 0.0)

                a_trial = a.copy()
                a_trial[start:end] = new_block
                a_trial = norm_sum1(a_trial)
                M_trial = fast_autocorr(a_trial).max()

                if M_trial < M - 1e-16:
                    a = a_trial
                    M = M_trial
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Perturb
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.03
                a[i] -= amt
                a[j] += amt
                a = norm_sum1(a)

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Block done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def random_init(n, seed=0):
    """Generate a random non-negative sequence with sum 1."""
    rng = np.random.default_rng(seed)
    # Mixture of exponential and uniform for diversity
    if seed % 3 == 0:
        a = rng.exponential(1.0, n)
    elif seed % 3 == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.05) + 0.1 * rng.random(n)
    else:
        # Multi-peaked
        n_peaks = rng.integers(3, 10)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            sigma = rng.uniform(0.02, 0.1)
            a += np.exp(-(np.linspace(0, 1, n) - c) ** 2 / (2 * sigma ** 2))
    return norm_sum1(a)


def sinusoidal_init(n, freq=1.0, phase=0.0, width=1.0):
    """Generate a sinusoidal-like sequence."""
    x = np.linspace(0, 1, n)
    a = np.maximum(np.sin(np.pi * freq * x + phase) ** width, 0.01)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = norm_sum1(p)
            p_score = evaluate_safe(p_norm)
            if np.isfinite(p_score):
                best_seq = p_norm
                best_score = p_score
                print(f"Parent: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = sinusoidal_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting optimization from n={n0}, score={best_score:.6f}")

    # === Phase 1: p-homotopy gradient descent from parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: p-homotopy GN (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 250.0)
    if t1 > 30:
        a_out = pnrm_optimize(best_seq, [2, 4, 8, 16, 32, 64], t1,
                              eta_init=0.08, momentum=0.80, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 1 improved: {sc:.6f}")

    # === Phase 2: Continue with higher p ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: High-p refinement (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 20:
        a_out = pnrm_optimize(best_seq, [64, 128, 256], t2,
                              eta_init=0.03, momentum=0.90, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 2 improved: {sc:.6f}")

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord polish (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out = coord_polish(best_seq, t3, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 3 improved: {sc:.6f}")

    # === Phase 4: Block optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Block opt (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_out = block_optimize(best_seq, t4, block_size=32, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 4 improved: {sc:.6f}")

    # === Phase 5: Restart with perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Perturbed restart (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        n_best = len(best_seq)
        rng = np.random.default_rng(123)
        # Small Gaussian perturbation
        perturbation_scale = 0.02 * best_seq.mean()
        a_pert = best_seq + rng.normal(0, perturbation_scale, n_best)
        a_pert = norm_sum1(a_pert)

        a_out = pnrm_optimize(a_pert, [16, 32, 64, 128], t5 * 0.6,
                              eta_init=0.04, momentum=0.85, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 5 restart improved: {sc:.6f}")
        else:
            # Try perturbation + coord polish
            remaining5 = time_budget - (time.monotonic() - t_global)
            if remaining5 > 30:
                a_out2 = coord_polish(a_pert, remaining5 * 0.5, verbose=False)
                sc2 = evaluate_safe(a_out2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_out2.copy()
                    print(f"  *** Phase 5 polish improved: {sc2:.6f}")

    # === Phase 6: Try different initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Alt init (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        n_best = len(best_seq)
        inits = [
            sinusoidal_init(n_best, freq=1.0, phase=0.0, width=1.5),
            sinusoidal_init(n_best, freq=2.0, phase=0.0, width=2.0),
            random_init(n_best, seed=7),
            random_init(n_best, seed=13),
        ]
        for ii, init in enumerate(inits[:3]):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 25:
                break
            t6 = min(remaining * 0.25, 100.0)
            if t6 < 15:
                break
            a_out = pnrm_optimize(init, [4, 8, 16, 32], t6,
                                  eta_init=0.05, momentum=0.80, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** Phase 6 init {ii} improved: {sc:.6f}")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Multi-length (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)
    longer_lengths = [n_best + 200, n_best + 500, n_best + 1000]
    longer_lengths = [l for l in longer_lengths if l <= 9000]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.35, 120.0)
        if t7a > 20:
            a_out = pnrm_optimize(a_long, [8, 16, 32, 64], t7a,
                                  eta_init=0.04, momentum=0.85, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                n_best = len(best_seq)
                print(f"  *** Phase 7 n={new_n} improved: {sc:.6f}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final polish (remaining={remaining:.0f}s, n={n_final}) ===")
    if remaining > 30:
        t8a = min(remaining * 0.40, 120.0)
        if t8a > 15:
            a_out = pnrm_optimize(best_seq, [32, 64, 128], t8a,
                                  eta_init=0.02, momentum=0.92, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** Final pnrm improved: {sc:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8b = min(remaining * 0.50, 100.0)
        if t8b > 10:
            a_out = coord_polish(best_seq, t8b, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** Final coord improved: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5200
        best_seq = sinusoidal_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}, n={n_fb}")

    # Ensure sum is large enough for evaluator
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.1 / best_seq.sum())

    return [float(x) for x in best_seq]
