    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_n)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def gradient_optimize(a0, time_limit, topk=20, c0=5.0, restart_every=3000, rng_seed=42, verbose=False):
    """Nesterov mirror descent with exact top-k subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g, _ = compute_topk_subgradient(y, topk=topk)
        
        eta = c0 / (np.sqrt(t) + 1.0)
        # Mirror step: a_new[j] ∝ y[j] * exp(-eta * g[j])
        log_ratio = np.log(np.maximum(y, 1e-30)) - eta * g
        log_ratio -= log_ratio.max()
        a_new = np.exp(log_ratio)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            strength = min(0.2, 0.05 + 0.02 * (stall // restart_every))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
        
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GRAD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def lp_direction(f, M, n_lp=None):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns normalized g or None if infeasible/expensive.
    """
    n = len(f)
    if n_lp is not None:
        n = n_lp
    
    # Build constraint matrix A (sparse)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # A is (2n-1) x n
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    if len(rows) == 0:
        return None
    
    A_sparse = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A_sparse[rows[i], cols[i]] = vals[i]
    
    c_vec = -np.ones(n)  # minimize -sum(g)
    b_vec = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c_vec, A_ub=A_sparse, b_ub=b_vec, bounds=bounds, method='highs')
    
    if not res.success:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-15:
        return None
    
    return g / s  # normalize
