Key insight: For a_j = (j+1)^p (normalized to sum=1), the self-convolution peak
is much lower than 1/n because the asymmetric shape pairs small values with large
values at the center lag, while the right-end peak (a[n-1]^2) is also small since
a[n-1] ≈ (p+1)/n after normalization.

The achievable J ≈ 2(p+1)^2 * Γ(p+1)^2 / Γ(2p+2) (center-dominated regime)

import numpy as np
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def compute_j(a):
    """Compute J = 2n * max(conv) / sum^2 for a numpy array."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    return 2.0 * n * max_b / (sum_a ** 2)


def make_poly_seq(n, p):
    """Create a_j = (j+1)^p normalized to sum=1, using log-space for stability."""
    j = np.arange(1, n + 1, dtype=np.float64)
    log_a = p * np.log(j)
    # Subtract max for numerical stability
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    for n in n_values:
        for p in p_values:
            # Asymptotic estimate to avoid computing huge convolutions unnecessarily
            # Center peak: (p+1)^2 * Gamma(p+1)^2 / Gamma(2p+2) / n
            # Right-end peak: ~ (p+1)^2 / n^2 * C where C ~ p
            # J = 2n * max(center, right_end)
            import math
            if p < 150:
                log_center = 2 * math.log(p + 1) + 2 * math.lgamma(p + 1) - math.lgamma(2 * p + 2) - math.log(n)
                center = math.exp(log_center) if log_center > -700 else 0.0
            else:
                center = 0.0  # Underflows, treat as 0
            right_end = (p + 1) ** 2 / (n * n) * max(1, p)  # rough upper bound
            est_J = 2 * n * max(center, right_end)
            candidates.append((est_J, n, p))

    # Sort by estimated J
    candidates.sort(key=lambda x: x[0])
    print("Top 20 by asymptotic estimate:")
    for i, (est, n, p) in enumerate(candidates[:20]):
        print(f"  {i+1}. n={n}, p={p}, est_J={est:.8f}")

    # Evaluate the top candidates exactly
    print("\n--- Phase 2: Exact evaluation of top candidates ---")
    evaluated = 0
    for est, n, p in candidates:
        if time.monotonic() - t_global > time_budget - 30:
            break
        if evaluated >= 40:
            break
        # Skip if estimated J is already much worse than current best
        if _best_tracker['score'] < float('inf') and est > _best_tracker['score'] * 1.1:
            # Still might be worth checking if estimate is rough
            if est > _best_tracker['score'] * 5:
                continue
        try:
            a = make_poly_seq(n, p)
            j_val = compute_j(a)
            label = f"poly_n{n}_p{p}"
            if np.isfinite(j_val):
                try_update(a, label)
            evaluated += 1
            if evaluated % 5 == 0:
                elapsed = time.monotonic() - t_global
                print(f"  Evaluated {evaluated} candidates, elapsed={elapsed:.0f}s, best={_best_tracker['score']:.8f}")
        except Exception as e:
            print(f"  Error for n={n}, p={p}: {e}")
            continue

    # Phase 3: Try non-integer p values around the best integer p
    print("\n--- Phase 3: Non-integer p refinement ---")
    if _best_tracker['seq'] is not None:
        best_n = len(_best_tracker['seq'])
        # Try p values nearby
        for p_int in range(max(1, 1), min(20, 50)):
            if time.monotonic() - t_global > time_budget - 30:
                break
            for p_frac in [p_int + 0.25, p_int + 0.5, p_int + 0.75]:
                if time.monotonic() - t_global > time_budget - 30:
                    break
                try:
                    a = make_poly_seq(best_n, p_frac)
                    j_val = compute_j(a)
                    if np.isfinite(j_val):
                        try_update(a, f"poly_n{best_n}_p{p_frac}")
                except Exception:
                    pass

    # Phase 4: Try larger n values nearby the best n
    print("\n--- Phase 4: n refinement ---")
    if _best_tracker['seq'] is not None:
        best_n = len(_best_tracker['seq'])
        # Determine best p from the best sequence
        # Just try a range of n around best_n with multiple p values
        n_neighbors = []
        for factor in [0.5, 0.7, 1.4, 2.0, 3.0, 5.0]:
            new_n = int(best_n * factor)
            if 1000 <= new_n <= 2000000 and new_n != best_n:
                n_neighbors.append(new_n)

        for new_n in n_neighbors:
            if time.monotonic() - t_global > time_budget - 30:
                break
            for p in [3, 4, 5, 6, 7, 8, 10, 12, 15, 20, 30]:
                if time.monotonic() - t_global > time_budget - 30:
                    break
                try:
                    a = make_poly_seq(new_n, p)
                    j_val = compute_j(a)
                    if np.isfinite(j_val):
                        try_update(a, f"poly_n{new_n}_p{p}")
                except Exception:
                    pass

    # Phase 5: Try some alternative shapes
    print("\n--- Phase 5: Alternative shapes ---")
    n_base = 100000
    shapes = []

    # Exponential
    for r in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]:
        x = np.arange(n_base, dtype=np.float64) / n_base
        a = np.exp(r * x)
        a = a / a.sum()
        shapes.append((a, f"exp_r{r}"))

    # Power with offset
    for p in [1, 2, 3, 5, 8, 15]:
        j = np.arange(1, n_base + 1, dtype=np.float64)
        a = j.astype(np.float64) ** p
        a = a / a.sum()
        shapes.append((a, f"offset_p{p}"))

    for a, label in shapes:
        if time.monotonic() - t_global > time_budget - 30:
            break
        try:
            j_val = compute_j(a)
            if np.isfinite(j_val):
                try_update(a, f"shape_{label}")
        except Exception:
            pass

    # Final
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={_best_tracker['score']:.8f}, n={len(_best_tracker['seq'])}, t={elapsed:.0f}s")
    else:
        # Fallback: linear sequence
        n_fb = 100000
        j = np.arange(1, n_fb + 1, dtype=np.float64)
        _best_tracker['seq'] = j / j.sum()
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback (linear): score={_best_tracker['score']:.8f}, n={n_fb}")

    return [float(v) for v in _best_tracker['seq']]
