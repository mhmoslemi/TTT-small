```python
def lp_step(a, M, time_budget=10):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k.
    Returns g normalized to have same sum as a."""
    from scipy.optimize import linprog
    n = len(a)
    
    # c = -1 (we want to maximize sum, so minimize -sum)
    c_obj = -np.ones(n)
    
    # Constraints: (a*g)[k] <= M for k = 0..2n-2
    # (a*g)[k] = sum_i a[i]*g[k-i] (for valid i, k-i)
    # For row k: g[j] * a[k-j] for j where 0<=k-j<n
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
    
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    
    g = g / g_sum * a.sum()
    return g
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))²`. The key insight is that the gradient of the log-sum-exp smoothed objective can be computed efficiently via FFT-based cross-correlation, giving full gradient information rather than the crude top-k approximation used previously.

My approach:
1. **Smooth mirror descent** with FFT-computed full gradient (O(n log n) per iteration)
2. **Adaptive temperature** schedule (broad → sharp focus on peaks)
3. **LP-based "big moves"** periodically using reduced constraints
4. **Spectral domain** periodic re-shaping
5. **Multiple restart refinements** with different parameters

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent with full FFT-based gradient
2. Adaptive temperature annealing (broad exploration to sharp exploitation)
3. LP-based big moves with constraint generation
4. Spectral domain periodic re-shaping
5. Multi-phase refinement with different parameter settings
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient(a, p):
    n = len(a)
    size = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(P * np.conjugate(A), size)
    return 2.0 * corr[:n]

def softmax_stable(x, T):
    x_shifted = (x - x.max()) / T
    x_shifted = np.clip(x_shifted, -200, 200)
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s

def normalize_simplex(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def current_M(a):
    c = fast_convolve(a)
    return c.max(), c

def mirror_descent_step(a, grad, eta, damping=0.8):
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    log_a -= log_a.max()
    a_new = np.exp(np.clip(log_a, -300, 300))
    a_new = np.maximum(a_new, 1e-30)
    a_new /= a_new.sum()
    a_out = damping * a + (1.0 - damping) * a_new
    return normalize_simplex(a_out)

def lp_big_move(a, M, top_k=80):
    n = len(a)
    c = fast_convolve(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    A_rows = []
    for k in top_indices:
        k_int = int(k)
        row = np.zeros(n)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                row[j] = a[i]
        A_rows.append(row)
    A_mat = np.array(A_rows)
    b_mat = np.full(len(top_indices), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_mat, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.status != 0 or res.x is None:
            return None
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs < 1e-15:
            return None
        g = g / gs
        return g
    except Exception:
        return None

def spectral_reshape(a, alpha=0.3):
    n = len(a)
    size = 1 << (n - 1).bit_length()
    if size < n:
        size = n
    A = np.fft.rfft(a, size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_avg = mag.mean()
    if mag_avg < 1e-30:
        return a.copy()
    ratio = mag / (mag_avg + 1e-30)
    mag_new = mag_avg * np.power(np.maximum(ratio, 1e-10), 1.0 - alpha)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    blend = 0.7 * a + 0.3 * a_new
    return normalize_simplex(blend)

def smooth_mirror_descent(a0, time_budget, T_start=0.5, T_end=0.02, eta_start=2.0,
                          eta_end=0.3, total_planned=500000, lp_every=20000,
                          verbose=True, perturbation_freq=50000):
    n = len(a0)
    a = normalize_simplex(a0.copy().astype(np.float64))
    best_a = a.copy()
    best_M, _ = current_M(a)
    t0 = time.time()
    stall = 0
    max_stall = 80000
    rng = np.random.default_rng(42)
    last_lp_time = t0
    last_perturb = t0
    damping = 0.85

    for it in range(total_planned):
        elapsed = time.time() - t0
        if elapsed > time_budget:
            break
        progress = min(1.0, it / float(total_planned))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        c, M = current_M(a)
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            a = a.copy()
            a += rng.normal(0, 0.02, n)
            a = normalize_simplex(a)
            stall = 0
            last_perturb = time.time()
        p = softmax_stable(c, max(T, 0.005))
        grad = compute_gradient(a, p)
        gmax = np.max(np.abs(grad))
        if gmax < 1e-15:
            break
        eta_scaled = eta / (gmax + 1e-10)
        eta_scaled = min(eta_scaled, 10.0)
        a = mirror_descent_step(a, grad, eta_scaled, damping=damping)
        if it > 0 and it % lp_every == 0 and (time.time() - last_lp_time) > 10:
            g = lp_big_move(a, best_M, top_k=60)
            if g is not None:
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    a_trial = (1.0 - frac) * a + frac * g
                    a_trial = normalize_simplex(a_trial)
                    c_trial = fast_convolve(a_trial)
                    M_trial = c_trial.max()
                    if M_trial < best_M - 1e-12:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        stall = 0
                        break
            last_lp_time = time.time()
        if it > 0 and it % perturbation_freq == 0 and (time.time() - last_perturb) > 30:
            a = spectral_reshape(a, alpha=0.2)
            c_chk = fast_convolve(a)
            if c_chk.max() < best_M:
                best_M = c_chk.max()
                best_a = a.copy()
            last_perturb = time.time()
        if verbose and it % 20000 == 0:
            score = 2.0 * n * best_M
            print(f"    SMD iter={it}, t={elapsed:.0f}s, M={best_M:.10f}, score={score:.6f}, T={T:.4f}, eta={eta:.4f}")
    score_final = 2.0 * n * best_M
    if verbose:
        print(f"    SMD done: score={score_final:.6f}, iters={it+1}, time={time.time()-t0:.0f}s")
    return best_a, best_M

def refine_with_cd(a0, time_budget, verbose=False):
    n = len(a0)
    a = normalize_simplex(a0.copy())
    best_a = a.copy()
    best_M, _ = current_M(a)
    t0 = time.time()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 2000

    while time.time() - t0 < time_budget and stall < max_stall:
        c, M = current_M(a)
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        top_k = min(20, len(c))
        peaks = np.argsort(c)[::-1][:top_k]
        improved = False
        for k in peaks:
            if time.time() - t0 > time_budget:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue
            for _ in range(3):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = normalize_simplex(a_trial)
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                if improved:
                    break
        if not improved and stall > 100:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.05
                a[i] -= amt
                a[j] += amt
                a = normalize_simplex(a)
        if verbose and stall % 500 == 0:
            print(f"    CD stall={stall}, M={best_M:.10f}")
    return best_a, best_M

def propose_candidate(budget_s=1800.0):
    t_global = time.time()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        c = fast_convolve(a_norm)
        sc = 2.0 * len(a_norm) * c.max()
        if sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.time() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        n_fb = 4200
        parent = np.ones(n_fb) / n_fb
        print(f"Using uniform init: n={len(parent)}")

    n0 = len(parent)
    M0, _ = current_M(parent)
    score0 = 2.0 * n0 * M0
    print(f"Starting: n={n0}, M={M0:.10f}, score={score0:.6f}")
    best_seq = parent.copy()
    best_score = score0

    remaining = time_budget - (time.time() - t_global)
    print(f"\n=== Phase 1: Smooth Mirror Descent (main) ===")
    t_phase1 = remaining * 0.55
    a1, M1 = smooth_mirror_descent(
        parent, t_phase1,
        T_start=0.8, T_end=0.01,
        eta_start=3.0, eta_end=0.2,
        total_planned=800000,
        lp_every=30000,
        perturbation_freq=80000,
        verbose=True
    )
    try_update(a1, "smd_phase1")

    remaining = time_budget - (time.time() - t_global)
    print(f"\n=== Phase 2: Refinement with different params ===")
    t_phase2 = remaining * 0.25
    a2, M2 = smooth_mirror_descent(
        best_seq, t_phase2,
        T_start=0.3, T_end=0.005,
        eta_start=1.5, eta_end=0.1,
        total_planned=400000,
        lp_every=15000,
        perturbation_freq=60000,
        verbose=True
    )
    try_update(a2, "smd_phase2")

    remaining = time_budget - (time.time() - t_global)
    print(f"\n=== Phase 3: LP-focused refinement ===")
    t_phase3 = remaining * 0.15
    if t_phase3 > 20:
        a_cur = best_seq.copy()
        M_cur, _ = current_M(a_cur)
        t_lp_start = time.time()
        lp_count = 0
        while time.time() - t_lp_start < t_phase3:
            g = lp_big_move(a_cur, M_cur, top_k=100)
            if g is None:
                break
            applied = False
            for frac in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                a_trial = (1.0 - frac) * a_cur + frac * g
                a_trial = normalize_simplex(a_trial)
                M_trial = fast_convolve(a_trial).max()
                if M_trial < M_cur - 1e-14:
                    a_cur = a_trial
                    M_cur = M_trial
                    applied = True
                    break
            lp_count += 1
            if not applied:
                a_cur = a_cur + 0.01 * np.random.randn(n0)
                a_cur = normalize_simplex(a_cur)
            if lp_count % 10 == 0:
                print(f"    LP iter {lp_count}: M={M_cur:.10f}")
        try_update(a_cur, "lp_phase3")
        # Quick smooth polish after LP
        rem = time_budget - (time.time() - t_global)
        if rem > 15:
            a3, M3 = smooth_mirror_descent(a_cur, min(rem * 0.5, 60.0),
                                           T_start=0.2, T_end=0.01,
                                           eta_start=1.0, eta_end=0.5,
                                           total_planned=100000,
                                           lp_every=999999,
                                           perturbation_freq=99999,
                                           verbose=False)
            try_update(a3, "polish_after_lp")

    remaining = time_budget - (time.time() - t_global)
    print(f"\n=== Phase 4: Final examination search ===")
    t_phase4 = remaining * 0.30
    if t_phase4 > 20:
        a4, M4 = smooth_mirror_descent(

    # Final validation

    else:
        n_fb = 4200
        best_seq = np.ones(n_fb) / n_fb
        best_score = 2.0
        print(f"ERROR fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
