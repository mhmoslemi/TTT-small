            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0, asymmetry=0.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    if asymmetry != 0.0:
        a = a * (1.0 + asymmetry * x)
        a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_bimodal(n, width=0.15, pos1=0.1, pos2=0.9):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width) ** 2) + np.exp(-((x - pos2) / width) ** 2) + 0.003
    return norm_sum1(a)


def make_cosine(n, freq=1):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    a = a ** 2
    return norm_sum1(a)


def make_asymmetric_edge(n, concentration=3.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-concentration * x) + 0.01
    return norm_sum1(a)


def make_dual_peak_asym(n, width1=0.08, width2=0.15, pos1=0.05, pos2=0.85, ratio=1.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width1) ** 2) + ratio * np.exp(-((x - pos2) / width2) ** 2) + 0.005
    return norm_sum1(a)


def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def compute_weighted_subgradient(a, ac, top_k=80, beta=4.0, weight_mode='power'):
    """Compute subgradient with weighted active set."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    
    if weight_mode == 'power':
        w = top_vals ** beta
    elif weight_mode == 'excess':
        med = np.median(ac)
        w = np.maximum(top_vals - med, 0.0) ** beta
        if w.sum() < 1e-30:
            w = top_vals ** beta
    else:
        w = np.ones(k_max)
    
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=60, beta=6.0, c0=5.0, restart_every=5000,
                 adaptive=True, verbose=False, rng_seed=42, lookahead=100):
    """Nesterov-accelerated mirror descent with lookahead averaging."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    best_m_history = []
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_weighted_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta, weight_mode='excess')

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        best_m_history.append(M_new)
        
        # Lookahead: check average of recent iterates
        if len(best_m_history) >= lookahead:
            avg_window = best_m_history[-lookahead:]
            if min(avg_window) < best_m - 1e-15:
                # Try the iterate that gave the min in the window
                pass  # We track best_a directly, so this is handled below
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if adaptive and stall_count > restart_every // 3:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 10, 200)

        if stall_count > restart_every:
            stall_count = 0
            # Diverse restart
            strategy = rng.integers(3)
            if strategy == 0:
                pert = rng.normal(0, 0.03, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                # Shrink toward uniform
                uniform = np.ones(n) / n
                x_r = 0.7 * best_a + 0.3 * uniform
            else:
                # Random spas
                mask = rng.random(n) < 0.1
                pert = np.zeros(n)
                pert[mask] = rng.normal(0, 0.05, mask.sum())
                x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.5)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}, k={cur_top_k}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=300, top_K_max=None,
                   use_all=False, line_search_pts=10):
    """LP-based Frank-Wolfe with adaptive K and improved line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            # Weighted selection: prefer peaks that are significantly above median
            med = np.median(ac)
            scores = np.maximum(ac - med, 0.0) ** 2
            binding_idx = np.argsort(scores)[::-1][:K_sel]
            binding_idx = binding_idx[scores[binding_idx] > 0]
            if len(binding_idx) == 0:
                binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((len(binding_idx), n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 30, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Improved line search: try more points including extrapolation
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        
        alphas_to_try = [best_alpha,
                         min(1.0, best_alpha * 1.5),
                         min(1.0, best_alpha * 2.0),
                         min(1.0, best_alpha * 0.5),
                         min(1.0, best_alpha * 3.0),
                         0.1, 0.2, 0.3]
        alphas_to_try = sorted(set(np.round(alphas_to_try, 6)))
        
        for alpha in alphas_to_try:
            if alpha <= 0 or alpha > 1:
                continue
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 5 == 0:
            K = min(K + 40, top_K_max)

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={len(binding_idx)}, alpha={best_alpha:.4f}")

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True, rng_seed=77):
    """Direct peak shaving: identify top contributing pairs and redistribute mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peaks
        n_peaks = min(15, 2 * n - 1)
        top_peaks = np.argsort(ac)[::-1][:n_peaks]
        
        # Build sensitivity map: for each position i, how much does it contribute to peaks vs non-peaks
        peak_contrib = np.zeros(n)
        other_contrib = np.zeros(n)
        
        for k in top_peaks[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            for i in range(s, e + 1):
                peak_contrib[i] += a[ki - i]
        
        # Total contribution from each position (across all lags)
        # Approximate: for position i, it appears in lags 0..2n-2 as a[i]*a[k-i]
        # Total = sum_k a[i]*a[k-i] = (sum a)^2 = 1 (when normalized)
        # So other_contrib[i] ≈ 1 - peak_contrib[i] (roughly)
        
        total_contrib = np.ones(n)  # normalized
        other_contrib = np.maximum(total_contrib - peak_contrib, 0.0)
        
        # Find positions with high peak contribution but low other contribution
        # These are "specialized" for the peaks and good candidates to reduce
        ratio = peak_contrib / (other_contrib + 1e-10)
        
        # Also build list of "safe" positions (low total sensitivity)
        # A position j is safe if reducing/increasing it barely affects any peak
        safe_score = np.zeros(n)
        for k in top_peaks[:5]:
            ki = int(k)
            for i in range(max(0, ki - n + 1), min(n, ki + 1)):
                safe_score[i] += a[ki - i]
        
        # Try moves: reduce from high-ratio positions, add to low-safe-score positions
        high_ratio_idx = np.argsort(ratio)[::-1][:20]
        low_safe_idx = np.argsort(safe_score)[:20]
        
        improved = False
        
        for src in high_ratio_idx[:10]:
            if time.monotonic() - t0 > time_limit:
                    break
            if a[src] < 1e-8:
                continue
                
            src_val = a[src]
            
            for dst in low_safe_idx[:5]:
                if dst == src:
                    continue
                
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                    amt = src_val * frac
                    if amt < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
            
            if improved:
                stall = 0
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-6:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=True, tau_init=10.0, tau_final=200.0,
                              eta_init=0.6, rng_seed=100):
    """Smoothed projected gradient with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    max_stall = 400
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 800.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        # Numerical safety
        ac_shifted = np.maximum(ac_shifted, -50.0)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross

        g_tangent = g - g.mean()
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            stall += 10
            continue

        # Adaptive step size
        effective_eta = eta * min(1.0, 1.0 / (g_norm * n))
        a_trial = a - effective_eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.03, 3.0)
        else:
            a = a_new
            stall += 1
            if stall > 80:
                eta *= 0.75
            if stall > 200:
                pert = rng.normal(0, 0.004, n) * a
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                    stall = 0
                    eta = eta_init * 0.6
                else:
                    a = a_new
            if stall > 300:
                # Larger perturbation
                pert = rng.normal(0, 0.02, n) * a
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert
                stall = 100

        total_iters += 1

        if verbose and total_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with importance-weighted move selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = 10
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build contribution map
        contrib = np.zeros(n)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib[idx] += a[ki - idx]
        
        # Low-contribution positions (safe to add mass)
        low_contrib = np.argsort(contrib)[:n // 10 + 10]
        
        improved = False
        
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            prod = a[idx] * a[ki - idx]
            
            if prod.max() < 1e-20:
                continue
            
            src = idx[np.argmax(prod)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try moving to low-contribution positions
            for dst_idx in range(len(low_contrib)):
                dst = low_contrib[dst_idx]
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.04, 0.07]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                stall = 0
        
        if not improved:
            stall += 1
            # Random move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init AMD (diverse starts) ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.22, 350.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3.0")
        inits.append(make_u_shape(n0, sharpness=5.0, asymmetry=0.3))
        labels.append("u_asym")
        inits.append(make_bimodal(n0, width=0.12, pos1=0.08, pos2=0.92))
        labels.append("bimodal")
        inits.append(make_dual_peak_asym(n0, width1=0.06, width2=0.2, pos1=0.03, pos2=0.8, ratio=0.5))
        labels.append("dual_asym")
        inits.append(make_asymmetric_edge(n0, concentration=2.0))
        labels.append("edge")
        inits.append(spectral_init(n0, n_modes=40, seed=42))
        labels.append("spectral")
        inits.append(make_cosine(n0, freq=2))
        labels.append("cos2")

        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, 2.0, asymmetry=0.2)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            print(f"  AMD '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                     restart_every=4000, adaptive=True,
                                     verbose=False, rng_seed=42 + idx * 7)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.22, 400.0)
    if t2 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1200)
        top_K_init = min(400, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all)
        try_update(a_out, "lp_main")

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak shaving (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 25:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t3, verbose=True, rng_seed=77)
        try_update(a_out, "peakshave")

    # === Phase 4: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Smoothed gradient (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 25:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t4, verbose=True,
                                             tau_init=10.0, tau_final=200.0,
                                             eta_init=0.6, rng_seed=100)
        try_update(a_out, "smooth1")

    # === Phase 5: LP-FW round 2 (with higher K) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1200)
        top_K_init = min(800, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all)
        try_update(a_out, "lp_r2")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate polish (n={n_best}) ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (n={n_best}) ===")

    trial_lengths = []
    for delta in [400, 800, -400, 1200, -800, 200, -200]:
        new_n = n_best + delta
        if 1000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 70:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.4, 150.0)
        if t7a > 15:
            a_out, _ = amd_optimize(a_long, t7a, top_k=60, beta=6.0, c0=4.5,
                                     restart_every=5000, adaptive=True,
                                     verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5000:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 15:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                use_all = (new_n <= 1200)
                top_K = min(2500, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                           top_K_init=min(400, 2*new_n-1),
                                           top_K_max=top_K, use_all=use_all)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final optimization on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final optimization (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.35, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            # High-temperature smooth to escape, then cool
            a_out, _ = smooth_gradient_optimize(a_in, t8 * 0.5, verbose=False,
                                                 tau_init=30.0, tau_final=300.0,
                                                 eta_init=0.4, rng_seed=500)
            try_update(a_out, "smooth_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 6000:
            t8b = min(remaining * 0.45, 130.0)
            if t8b > 20:
                a_in = norm_sum1(best_seq)
                use_all = (n_best <= 1200)
                top_K_init = min(1000, 2 * n_best - 1)
                top_K_max = min(6000, 2 * n_best - 1)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False,
                                           top_K_init=top_K_init, top_K_max=top_K_max,
                                           use_all=use_all)
                try_update(a_out, "lp_final")

    # === Phase 9: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t9b = min(remaining * 0.6, 50.0)
            if t9b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = peak_shaving(a_in, t9b, verbose=False, rng_seed=999)
                try_update(a_out, "peakshave_final")

    # === Phase 10: AMD fine-tune on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: AMD fine-tune (n={n_best}) ===")
        t10 = min(remaining * 0.7, 100.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t10, top_k=80, beta=8.0, c0=3.0,
                                     restart_every=3000, adaptive=True,
                                     verbose=False, rng_seed=1234)
            try_update(a_out, "amd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
