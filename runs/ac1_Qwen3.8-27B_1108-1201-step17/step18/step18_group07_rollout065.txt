def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_val(a):
    """Objective assuming sum(a) == 1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def compute_subgrad(a, c, n_peaks=40):
    """Weighted subgradient of max_k (a*a)_k. Heavier weight on higher peaks."""
    n = len(a)
    k_max = min(n_peaks, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()
    if len(top_vals) > 1:
        hi = top_vals[0]
        lo = top_vals[-1]
        if hi - lo < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = (top_vals - lo) / (hi - lo)
            w = np.maximum(w, 0.01)
            w = w / w.sum()
    else:
        w = np.ones(1)
    grad = np.zeros(n)
    for t in range(k_max):
        wi = w[t]
        if wi < 1e-12:
            continue
        ki = int(top_idx[t])
        s = max(0, ki - n + 1)
        e = min(n, ki + 1)
        if s >= e:
            continue
        m = np.arange(s, e)
        j = ki - m
        grad[m] += 2.0 * wi * a[j]
    return grad


def spectral_flatten(a, gamma=0.5):
    """Compress the dynamic range of |FFT(a)| (water-filling in frequency)."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.power(ratio, gamma)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def init_flat(n):
    return np.ones(n) / n


def init_cosine(n, mode=1):
    x = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return normalize(a)


def init_bowl(n, sharpness=2.0):
    x = np.linspace(0.0, 1.0, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return normalize(a)


def init_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    return normalize(x ** power)


def init_dual(n, p1=0.25, p2=0.75, w=0.1):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-((x - p1) ** 2) / (2 * w * w)) + np.exp(-((x - p2) ** 2) / (2 * w * w))
    a = a + 0.05
    return normalize(a)


def init_sinc(n, bw=5.0):
    x = np.linspace(0.0, 1.0, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return normalize(a)


def init_random_smooth(n, seed):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.full(n, 0.5)
    for f in range(1, 14):
        coef = rng.normal(0.0, 0.12 / np.sqrt(f))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a = a + coef * np.sin(2.0 * np.pi * f * x + ph)
    a = np.maximum(a, 0.001)
    return normalize(a)


def init_tilted_cosine(n, tilt=0.6):
    """Slightly asymmetric cosine bowl; bias mass toward one side to defocus center peak."""
    x = np.linspace(0.0, 1.0, n)
    base = np.cos(np.pi * (x - 0.5)) ** 2
    asym = 1.0 + tilt * (x - 0.5)
    a = np.maximum(base * asym, 0.0)
    return normalize(a)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def init_internal_ramp(n, power=1.0):
    """Ramp that rises then gently falls (avoids sharp endpoints)."""

    a = x ** power * (1.0 - 0.3 * x)
    return normalize(a)
