def fast_autocorr(a):

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgradient(a, ac, n_peaks=20):
    """Compute subgradient using exact top-K peaks."""
    n = len(a)
    k_max = min(n_peaks, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    
    # Weight peaks by their value (normalized)
    peak_vals = ac[top_idx]
    total = peak_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = peak_vals / total
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = int(top_idx[idx])
        w = weights[idx]
        if w < 1e-12:
            continue
        # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        idx_arr = np.arange(j_lo, j_hi + 1)
        pair_idx = k - idx_arr
        grad[idx_arr] += 2.0 * w * a[pair_idx]
    return grad

def spectral_shaping(a0, time_limit, verbose=False):
    """Flatten the amplitude spectrum to reduce autocorrelation peak."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Target: geometric mean of current spectrum (soft flat)
        P_log = np.log(np.maximum(P, 1e-30))
        P_target = np.exp(np.median(P_log))
        
        # Progressive: blend between current and target
        alpha = min(1.0, 0.05 + 0.01 * it / 100)
        P_new = P * (1 - alpha) + P_target * alpha
        
        # Update magnitude
        mag_ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.5, 2.0)
        A_new = A * mag_ratio
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try a smaller step
            a_try = (1 - 0.3) * a + 0.3 * a_new
            a_try = norm_sum1(a_try)
            M_try = fast_autocorr(a_try).max()
            if M_try < best_m:
                best_m = M_try
                best_a = a_try.copy()
                a = a_try
            # If even smaller step doesn't help, reduce alpha
            # (handled by the progression above)
    
    return best_a, best_m


def coordinate_descent(a0, time_limit, n_peaks=15, verbose=False, rng_seed=42):
    """Coordinate descent: move mass from peak-contributing positions to safe positions."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peaks]
        
        improved = False
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            
            # Find source: position with max contribution to this peak
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Compute "exposure" of each position to top peaks
            exposure = np.zeros(n)
            for k2 in top_k[:n_peaks]:
                k2i = int(k2)
                j2_lo = max(0, k2i - n + 1)
                j2_hi = min(n - 1, k2i)
                if j2_lo <= j2_hi:
                    v = np.arange(j2_lo, j2_hi + 1)
                    exposure[v] += a[k2i - v]
            
            # Destination: low exposure and low current value
            cost = exposure + 0.1 * a
            # Exclude source
            cost[src] = np.inf
            dst = int(np.argmin(cost))
            
            # Bisection for optimal fraction
            lo, hi = 1e-6, 0.99
            best_frac = 0.0
            best_M = best_m
            for _ in range(20):  # 20 bisection steps
                mid = (lo + hi) / 2
                amt = src_val * mid
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    hi = mid
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M:
                    best_M = M_trial
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid
            
            if best_frac > 0 and best_M < best_m - 1e-16:
                amt = src_val * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_M
                best_a = a.copy()
                improved = True
                stall = 0
        
        if not improved:
            stall += 1
            # Escape: random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m


def make_ramp(n, power=0.4):

def mirror_descent(a0, time_limit, n_peaks=50, temp0=2.0, c0=4.0,

    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_temp = temp0
    restart_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()

            y = x.copy()
        else:
            y = y / sy

        g = compute_subgradient(y, ac, n_peaks=n_peaks)

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)

        x = a_new

        M_eff = min(M_x, M_y)
        if M_eff < best_m - 1e-15:
            best_m = M_eff
            if M_x <= M_y:
                best_a = x.copy()
            else:
                best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        cur_temp = max(0.5, temp0 * (1.0 - 0.1 * min(stall / 5000.0, 1.0)))

            x = x_r

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.
