"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (solve LP for optimal descent direction)
2. FTRL mirror descent with full softmax subgradients
3. Spectral domain flattening (iterative amplitude compression)
4. Adaptive line search for step size
5. Multi-start with diverse initializations
6. Hierarchical optimization (small-n LP warm-start for large-n)
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv2(f, g):
    """Fast linear convolution f*g via FFT."""
    f = np.asarray(f, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_m(a):
    """Compute max(conv(a,a)) assuming sum(a)=1. The full objective is 2n*M."""
    return fast_conv(a).max()


def obj_full(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * obj_m(a)


def evaluate_safe(a):
    """Safely evaluate the sequence using the provided evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))

def compute_gradient(a, w):
    """Compute grad[i] = 2 * sum_j a[j] * w[i+j] using FFT.
    w has length 2n-1 (the softmax weights over convolution lags)."""
    n = len(a)
    a_flipped = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    A_f = np.fft.rfft(a_flipped, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(A_f * W, fft_size)
    return 2.0 * conv_result[n - 1: n - 1 + n]


def softmax_weights(c, beta):
    """Compute softmax weights over convolution values."""
    max_v = c.max()
    if max_v < 1e-30:
        return np.ones(len(c)) / len(c)
    exponents = beta * (c - max_v) / (max_v + 1e-30)
    exponents = np.maximum(exponents, -50.0)
    w = np.exp(exponents)
    s = w.sum()
    if s < 1e-30:
        return np.ones(len(c)) / len(c)
    return w / s


def lp_direction(f, M, K=150, solver='highs'):
    """Solve LP to find descent direction.
    Maximize sum(g) subject to g >= 0, (f*g)_k <= M for top K lags.
    Returns g (unnormalized) or None if infeasible.
    """
    n = len(f)
    c = fast_conv(f)
    # Use top K lags as constraints (most binding)
    top_lags = np.argsort(c)[::-1][:min(K, len(c))]
    
    # Build constraint matrix: C[idx, j] = f[k - j] for lag k = top_lags[idx]
    C = np.zeros((len(top_lags), n))
    for idx, k in enumerate(top_lags):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        for j in range(j_start, j_end):
            C[idx, j] = f[k_int - j]
    
    # Maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_lags), M + 1e-10)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method=solver,
                         options={"presolve": True, "time_limit": 30.0})
        if result.success:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

        grad = compute_gradient(a, w)
        
        # Momentum
        momentum = momentum_decay * momentum + (1 - momentum_decay) * grad
        eff_grad = momentum
        
        # Mirror descent update in log-space
        log_a = log_a - eta * eff_grad
        # Prevent overflow/underflow
        log_a = np.clip(log_a, -30.0, 30.0)
        # Renormalize
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        log_a = np.log(np.maximum(a_new, 1e-15))
        
        # Check improvement
        m_new = obj_m(a_new)
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert
            a = best_a.copy()
            log_a = np.log(np.maximum(a, 1e-15))
            # Reduce learning rate
            eta0 = max(0.01, eta0 * 0.9)
        
        # Periodic spectral nudge
        if iteration % 2000 == 0:
            a_nudged = spectral_step(a, 0.05)
            m_nudged = obj_m(a_nudged)
            if m_nudged < best_m:
                best_m = m_nudged
                best_a = a_nudged.copy()
                a = a_nudged
                log_a = np.log(np.maximum(a, 1e-15))
        
        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FTRL iter={iteration} ({elapsed:.0f}s): M={best_m:.8f}, J={2*n*best_m:.6f}")
    
    return best_a, 2 * n * best_m


def spectral_step(a, strength=0.2):
    """One step of spectral flattening."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return norm_simplex((1 - strength * 0.5) * a + strength * 0.5 * a_new)


def spectral_optimize(a0, time_budget, verbose=False):
    """Iterative spectral flattening with adaptive step."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = obj_m(a)
    t0 = time.monotonic()
    
    step = 0.15
    stall = 0
    while time.monotonic() - t0 < time_budget and stall < 200:
        a = spectral_step(a, step)
        m = obj_m(a)
        if m < best_m - 1e-15:
            best_m = m
            best_a = a.copy()
            step = min(0.4, step * 1.02)
            stall = 0
        else:
            a = norm_simplex(0.7 * a + 0.3 * best_a)
            step = max(0.03, step * 0.85)
            stall += 1
    
    return best_a, 2 * n * best_m


def coordinate_descent(a0, time_budget, verbose=False):
    """Local coordinate descent: move mass between positions to reduce peak."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = obj_m(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_budget and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        top_peak = int(np.argmax(c))
        
        # Find positions contributing most to the peak
        i_start = max(0, top_peak - n + 1)
        i_end = min(n, top_peak + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[top_peak - idx]
        
        improved = False
        src = idx[np.argmax(contrib)]
        src_val = a[src]
        
        if src_val > 1e-10:
            # Try moving mass to various destinations
            candidates = [int(rng.integers(n)) for _ in range(20)]
            # Also try positions that reduce other peaks
            other_peaks = np.argsort(c)[::-1][1:5]
            for op in other_peaks:
                op_int = int(op)
                i_s = max(0, op_int - n + 1)
                i_e = min(n, op_int + 1)
                idx2 = np.arange(i_s, i_e)
                if len(idx2) > 0:
                    candidates.append(idx2[np.argmin(a[idx2])])
            
            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = obj_m(a_trial)
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-6:
                frac = rng.uniform(0.05, 0.15)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = norm_simplex(a)
                m_new = obj_m(a)
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, 2 * n * best_m


def make_init_uniform(n):
    return np.ones(n) / n


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.03
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_random_smooth(n, seed=0, n_frequencies=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_frequencies + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):

    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:

    if best_seq is None:
