- But the variance of `conv(a,a)[k]` is ≈ 1/n^2 (for independent entries)
- So `max over k` of `conv(a,a)[k]` ≈ 1/n + c*sqrt(log(n)/n^2)` by extreme value theory
- This gives `max_b ≈ 1/n * (1 + c*sqrt(log(n)/n))`
- For n=5200: `c*sqrt(log(5200)/5200) ≈ c*sqrt(8.55/5200) ≈ c*0.04` (with c≈3 for Gaussian max)
- So `max_b ≈ 1/n * 1.12`
- Objective ≈ `2*n*(1/n)*1.12/n ≈ 2*1.12/n`... wait that doesn't seem right.

Let me redo. If `a[j]` are i.i.d. with mean `μ = 1/n` and variance `σ^2`:
- `conv(a,a)[k] = sum_j a[j]*a[k-j]` (about n/2 terms for interior k)
- E[conv[k]] ≈ (n/2) * μ^2 = (n/2) * 1/n^2 = 1/(2n)
- Var[conv[k]] ≈ (n/2) * (E[a^4] - (E[a^2])^2) ≈ (n/2) * σ^4 (roughly)
- For exponential: E[a^2] = 2μ^2, E[a^4] = 24μ^4, so Var ≈ n/2 * (24-4)μ^4 = 10*n*μ^4 = 10/n^3
- SD ≈ sqrt(10)/n^{3/2}
- Max over 2n lags: ≈ 1/(2n) + 3*sqrt(2*log(2n)/n) * (1/n^{3/2})... 

Hmm this is getting complicated. Let me just focus on the algorithmic approach.

**Final algorithm plan:**

1. **Initialization**: `a[j] = sin^2(pi*(j+0.5)/n)` normalized (avoids zero endpoints naturally since we use j+0.5)
   Actually, `sin^2(pi*j/(n-1))` is zero at j=0 and j=n-1. Let's use that.
   But wait, if a[0]=0 and a[n-1]=0, then conv[0]=conv[2n-2]=0, which is great.
   The conv[n-1] (center) = sum_j a[j]*a[n-1-j]. For sin^2 symmetric shape, this is positive.

2. **LP Cutting-Plane (main workhorse)**:
   - Maintain a set of active constraints (lags k where conv > threshold)
   - At each iteration, solve LP: min M s.t. sum g=1, g>=0, (y^T T_k) g <= M for active k
   - After solving, compute conv(y, g) at all lags, add all violated to active set
   - Prune constraints that are far from active (margin > threshold)
   - Use Nesterov-accelerated step
   - Track best solution

3. **Symmetry**: Force g[n-1-j] = g[j] in the LP (or just use full n variables but only add constraints for k=0,...,n-1)

4. **Polish**: After LP converges, try local moves (swap small amounts between pairs of positions)

5. **Length exploration**: Try n in {3000, 4000, 5000, 6000, 7000}

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Sinusoidal initialization with zero endpoints
2. Aggressive LP cutting-plane Frank-Wolfe with full constraint re-evaluation
3. Symmetric sequence exploitation
4. Multi-length exploration
5. Local polish with pairwise swaps
"""

import numpy as np
import time
import cvxpy as cp

_SOLVER = None

def _get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _SOLVER = cp.SCS
    print(f"  Fallback LP solver: {s}")
    return _SOLVER

def _fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _interp(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return f_new / ss

def _init_sin2(n):
    """sin^2 initialization: zero at endpoints, peak at center."""
    j = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.array([1.0])
    a = np.sin(np.pi * j / (n - 1)) ** 2
    return _norm1(a)

def _init_parametric(n, family, **kwargs):
    """Various parametric initializations."""
    j = np.arange(n, dtype=np.float64)
    x = j / max(n - 1, 1)
    if family == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif family == 'sin4':
        a = np.sin(np.pi * x) ** 4
    elif family == 'cos_domain':
        a = np.cos(np.pi * x) ** 2  # 1 at center, 0 at edges... no, cos is 1 at 0
        a = (1 - np.cos(np.pi * x)) ** 2  # 0 at edges, max at center
    elif family == 'uniform_cut':
        a = np.ones(n)
        cut = int(kwargs.get('cut', 0.05) * n)
        a[:cut] = 0
        a[-cut:] = 0
    elif family == 'triangular':
        a = 1 - np.abs(2 * x - 1)
    elif family == 'power_sym':
        p = kwargs.get('p', 0.5)
        a = ((1 - np.abs(2*x - 1)) ** p)  # triangular to power p
    else:
        a = np.sin(np.pi * x) ** 2
    return _norm1(a)

def _lp_cutting_plane(a0, time_limit, solver, initial_K=300, max_K=4000, 
                       add_all_violated=True, prune_margin=0.1, verbose=True):
    """LP cutting-plane Frank-Wolfe for minimizing max autocorrelation."""
    n = len(a0)
    y = _norm1(a0).copy()
    best_a = y.copy()
    ac = _fast_autocorr(y)
    best_m = ac.max()
    t0 = time.monotonic()
    
    # Initialize constraint set with top lags
    size_ac = len(ac)
    init_idx = np.argsort(ac)[::-1][:min(initial_K, size_ac)]
    constraint_set = set(int(k) for k in init_idx)
    
    # Nesterov momentum variable
    x_mom = y.copy()
    
    iter_count = 0
    no_improve_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = _fast_autocorr(y)
        M_curr = ac_y.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Check x_mom too
        ac_xm = _fast_autocorr(x_mom)
        if ac_xm.max() < best_m - 1e-15:
            best_m = ac_xm.max()
            best_a = x_mom.copy()
            no_improve_count = 0
        
        # Prune constraints: remove those with large margin
        if len(constraint_set) > 200:
            margins = np.array([M_curr - ac_y[k] for k in constraint_set if k < size_ac])
            # Keep constraints where margin < prune_margin * M_curr
            keep_set = set()
            for k in list(constraint_set):
                if k >= size_ac:
                    continue
                margin = M_curr - ac_y[k]
                if margin < prune_margin * M_curr:
                    keep_set.add(int(k))
            # Always keep top initial_K
            for k in init_idx[:initial_K // 2]:
                if k < size_ac:
                    keep_set.add(int(k))
            if len(keep_set) >= initial_K // 3:
                constraint_set = keep_set
        
        # Cap constraint set size
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            # Keep the most binding (smallest margin)
            ac_vals = np.array([ac_y[k] for k in constraint_list])
            margins = M_curr - ac_vals
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix A (K x n)
        # Row k: A[k, j] = y[k-j] if 0 <= k-j < n, else 0
        A = np.zeros((K, n))
        for ii, k in enumerate(constraint_list):
            k_int = int(k)
            # g[m] coefficient: y[k-m] for m in [max(0, k-n+1), min(n-1, k)]
            m_start = max(0, k_int - n + 1)
            m_end = min(n - 1, k_int)
            if m_start <= m_end:
                m_range = np.arange(m_start, m_end + 1)
                A[ii, m_range] = y[k_int - m_range]
        
        # Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception as e:
            if verbose:
                print(f"    LP solve failed: {e}")
            # Retry with fewer constraints
            constraint_list = sorted(constraint_set)[:min(200, len(constraint_set))]
            constraint_set = set(constraint_list)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check all constraints and add violated ones
        # Compute conv(y, g_dir) at all lags
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:2*n-1]
        
        if add_all_violated:
            violated = np.where(conv_yg > M_curr + 1e-10)[0]
            for k in violated:
                constraint_set.add(int(k))
        else:
            # Add top 50 violated
            viol_idx = np.where(conv_yg > M_curr + 1e-10)[0]
            if len(viol_idx) > 0:
                viol_amt = conv_yg[viol_idx] - M_curr
                top = viol_idx[np.argsort(viol_amt)[::-1][:50]]
                for k in top:
                    constraint_set.add(int(k))
        
        # Nesterov accelerated step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_mom_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_mom_new = y_new + beta_t * (y_new - y)
            x_mom_new = np.maximum(x_mom_new, 0.0)
            sx = x_mom_new.sum()
            if sx < 1e-15:
                x_mom_new = y_new.copy()
            else:
                x_mom_new = x_mom_new / sx
        
        y = y_new
        x_mom = x_mom_new
        iter_count += 1
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t
