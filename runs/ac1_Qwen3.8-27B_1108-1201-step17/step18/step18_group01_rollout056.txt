def lp_step(f, M, active_indices=None, solver='GLPK'):
    """
    One LP step: find g that maximizes sum(g) s.t. g>=0, (f*g)(k)<=M.
    If active_indices is given, only enforce constraints at those indices.
    Returns (g, active_indices_for_next).
    """
    n = len(f)
    size = 2*n - 1
    
    # Build constraint matrix
    A = build_conv_matrix(f)
    
    if active_indices is not None:
        A_sub = A[active_indices, :]
        b_sub = M * np.ones(len(active_indices))

        b_sub = M * np.ones(size)

    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A_sub @ g_var <= b_sub]
    obj = cp.Maximize(cp.sum(g_var))

    # Find new active set

    new_active = np.unique(np.concatenate([new_active, must_include]))
    
    return g, new_active

import time
import math
from scipy.sparse import lil_matrix, csr_matrix, csc_matrix

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_conv_matrix(f):
    """Build sparse matrix A (size x n) where A @ g gives f * g (full conv)."""
    n = len(f)
    size = 2 * n - 1
    # A[k, j] = f[j] for j <= k <= j+n-1
    # Efficient construction using diagonals
    A = lil_matrix((size, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = j + n  # exclusive
        A[k_start:k_end, j] = f[j]
    return A.tocsr()

def lp_optimize(f_init, time_limit, t_blend=0.2, max_active_frac=1.0, verbose=True):
    """
    LP-based optimization: iteratively solve LP to find best direction,
    blend with current, repeat.
    """
    import cvxpy as cp
    import cvxpy.conifig as conifg
    
    n = len(f_init)
    f = norm_sum1(f_init)
    best_f = f.copy()
    ac = fast_autocorr(f)
    best_M = ac.max()
    t0 = time.monotonic()
    active_indices = None
    solver_try = ['ECOS', 'SCS', 'GLPK']
    solver_idx = 0
    
    iter_count = 0
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(f)
        M = ac.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
            stall = 0
            if verbose:
                print(f"    LP iter={iter_count}: M={M:.8f}, J={2*n*M:.6f}")
        else:
            stall += 1
        
        if stall > max_stall:
            break
        
        # Determine active constraints
        if active_indices is None or len(active_indices) > max_active_frac * (2*n - 1):
            # Use all constraints
            A = build_conv_matrix(f)
            A_sub = A
            b_sub = M * np.ones(2*n - 1)
        else:
            A = build_conv_matrix(f)
            A_sub = A[active_indices, :]
            b_sub = M * np.ones(len(active_indices))
        
        # Solve LP
        g_var = cp.Variable(n)
        constraints = [g_var >= 0, A_sub @ g_var <= b_sub]
        obj = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(obj, constraints)
        
        solved = False
        for si in range(len(solver_try)):
            try:
                solver = solver_try[(solver_idx + si) % len(solver_try)]
                prob.solve(solver=solver)
                if g_var.value is not None:
                    solver_idx = (solver_idx + si) % len(solver_try)
                    solved = True
                    break
            except Exception as e:
                continue
        
        if not solved:
            if verbose:
                print(f"    LP failed at iter {iter_count}, switching to MGD")
            break
        
        g = np.maximum(g_var.value, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        # Normalize g to match f
        g_norm = g / g_sum  # sum = 1
        
        # Blend
        f_new = (1 - t_blend) * f + t_blend * g_norm
        f_new = np.maximum(f_new, 0.0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            break
        f_new = f_new / f_new_sum
        
        # Check improvement
        ac_new = fast_autocorr(f_new)
        M_new = ac_new.max()
        
        if M_new < M:
            f = f_new
            # Update active set
            if A is not None:
                conv_g = A @ g_norm
            else:
                conv_g = fast_autocorr(f)  # fallback
            slack = M - conv_g
            threshold = 0.02 * M
            new_active = np.where(slack < threshold)[0]
            # Always include key lags
            size = 2*n - 1
            must = np.concatenate([
                np.arange(min(3, size)),
                np.arange(max(0, n-3), min(n+3, size)),
                np.arange(max(0, size-3), size)
            ])
            active_indices = np.unique(np.concatenate([new_active, must]))
        else:
            # Try smaller blend
            t_blend = max(0.05, t_blend * 0.7)
        
        iter_count += 1
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    LP done: iter={iter_count}, time={elapsed:.0f}s, J={2*n*best_M:.6f}")
    return best_f
