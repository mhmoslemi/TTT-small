```python
"""
Minimize J = 2*n*max(conv(a,a))/(sum a)^2 over non-negative step functions.

Unlike the prior FTRL/DE pipeline this program drives the search from three
different engines that are *not* FTRL mirror descent + differential evolution:

  1. Spectral power flattening (iterative |A|^2 compression in the frequency
     domain) used both as a pre-processor and as a periodic "jump".
  2. A robust multiplicative (log-space) mirror-descent with an
     auto-scaled log-sum-exp focus factor, momentum and a safeguarded line
     search.  The per-coordinate direction is the cross-correlation of the
     peak-focus weights w with the current sequence, mean-centred so the
     simplex mass is preserved.  This directly attacks the autocorrelation
     peaks rather than a generic subgradient.
  3. A pair-swap coordinate local search that redistributes mass between the
     positions contributing most to the current dominant lag.

The engines are interleaved over many sequence lengths and many analytic /
random smooth initialisations, always retaining the best candidate found and
the (optional) parent construction height_sequence_1.
"""

import numpy as np
import time


# --------------------------------------------------------------------------- #
#  Core linear algebra helpers
# --------------------------------------------------------------------------- #
def fast_conv(a):
    """Linear convolution a*a via a zero-padded FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_from_a(a):
    """2*n*max(conv(a,a)) assuming the input is already on the simplex."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safe wrapper around the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float("inf")


def cross_corr_w_a(w, a):
    """W_a[j] = sum_k w[k]*a[k-j]  for j = 0..n-1,  w has length 2n-1."""
    n = len(a)
    fft_len = 1 << ((2 * n - 1) + n - 1).bit_length()
    W = np.fft.rfft(w, fft_len)
    Arev = np.zeros(fft_len)
    Arev[:n] = a[::-1]
    AR = np.fft.rfft(Arev, fft_len)
    conv = np.fft.irfft(W * AR, fft_len)
    return conv[n - 1: 2 * n - 1]


def stable_softmax(z, beta):
    """softmax(beta*z) computed stably."""
    b = beta * (z - z.max())
    e = np.exp(np.maximum(b, -60.0))
    s = e.sum()
    if s <= 0:
        return np.ones_like(z) / len(z)
    return e / s


# --------------------------------------------------------------------------- #
#  Engine 1 : spectral power flattening
# --------------------------------------------------------------------------- #
def spectral_flat(a, gamma=0.65, n_iter=80):
    """Iteratively compress |FFT(a)|^2 toward a flat (white) spectrum."""
    n = len(a)
    a = norm_simplex(a.copy())
    fft_size = 1 << (2 * n - 1).bit_length()
    for _ in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        m = mag.mean()
        if m < 1e-30:
            break
        ratio = mag / (m + 1e-30)
        mag_new = m * np.power(np.maximum(ratio, 1e-10), 1.0 - gamma)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return norm_simplex(a)


def spectral_polish(a, t_limit, gamma=0.5):
    """Greedy spectral flattening with an inner objective check."""
    a = norm_simplex(a.copy())
    best = a.copy()
    best_j = obj_from_a(best)
    t0 = time.monotonic()
    g = gamma
    while time.monotonic() - t0 < t_limit:
        cand = spectral_flat(best, gamma=g, n_iter=12)
        j = obj_from_a(cand)
        if j < best_j - 1e-15:
            best = cand
            best_j = j
            g = min(0.85, g * 1.06)
        else:
            g = max(0.2, g * 0.7)
            if g < 0.2 and obj_from_a(best) < 1.52:
                # already good; mix for diversity
                cand = norm_simplex(0.5 * best + 0.5 * np.ones_like(best))
                cj = obj_from_a(cand)
                if cj < best_j - 1e-15:
                    best, best_j = cand, cj
                break
    return best


# --------------------------------------------------------------------------- #
#  Engine 2 : multiplicative mirror descent with focus annealing
# --------------------------------------------------------------------------- #
def mirror_descent(a0, t_limit, eta0=0.25, max_iter=400000,
                   spectral_every=400, verbose=False):
    n = len(a0)
    a = norm_simplex(a0.copy())
    best = a.copy()
    best_j = obj_from_a(best)
    t0 = time.monotonic()
    eta = eta0
    v = np.zeros(n)
    rng = np.random.default_rng(hash(n) % (10 ** 6))
    momentum = 0.55
    stalled = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > t_limit:
            break
        c = fast_conv(a)
        spread = max(np.percentile(c, 99.0) - np.percentile(c, 50.0), 1e-9)
        beta = 6.0 / spread                      # auto focus scale
        w = stable_softmax(c, beta)             # peak focus weights
        W_a = cross_corr_w_a(w, a)              # where mass creates peaks
        direction = W_a - W_a.mean()            # mean-centred -> mass neutral

        v = momentum * v + (1.0 - momentum) * direction

        # safeguarded line search in log-space (multiplicative)
        step = eta
        for _ in range(8):
            a_t = a * np.exp(-step * v)
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            j = obj_from_a(a_t)
            if j < best_j - 1e-15:
                a = a_t
                if j < best_j:
                    best = a.copy()
                    best_j = j
                    stalled = 0
                eta = min(2.0, eta * 1.08)
                break
            step *= 0.5
        else:
            # no improvement at any trial -> slow the walk & escape
            stalled += 1
            eta = max(0.02, eta * 0.7)
            if stalled > 60:
                # kick: redistribute a little mass to random low v positions
                idx = np.argsort(v)[: max(2, n // 40)]
                amt = 0.02 * a.sum()
                a_r = a.copy()
                a_r[:] = a_r[:] - 0.0
                src = a > a.mean()
                pool = a[src]
                if pool.sum() > 1e-12:
                    a_r[src] = a[src] * (1.0 - 0.5)
                    a_r[idx] += 0.5 * pool.sum() / len(idx) + 1e-9
                    a = norm_simplex(a_r)
                v[:] = 0.0
                stalled = 30

        # periodic spectral jump to escape plateaus
        if it % spectral_every == 0 and it > 0:
            cand = spectral_flat(best, gamma=0.4, n_iter=8)
            cj = obj_from_a(cand)
            if cj < best_j - 1e-15:
                best, best_j = cand, cj
                a = cand.copy()
                v[:] = 0.0
            elif rng.random() < 0.25:
                a = norm_simplex(0.5 * a + 0.5 * cand)

        if it % 2000 == 0:
            a = norm_simplex(a)

    best = norm_simplex(best)
    # final micro-polish
    final = spectral_polish(best, 1.5)
    j = obj_from_a(final)
    if j < best_j:
        best = final
    return norm_simplex(best)


# --------------------------------------------------------------------------- #
#  Engine 3 : pair-swap coordinate search on the dominant lag
# --------------------------------------------------------------------------- #
def pair_swap(a0, t_limit, verbose=False):
    n = len(a0)
    a = norm_simplex(a0.copy())
    best = a.copy()
    best_j = obj_from_a(best)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < t_limit:
        c = fast_conv(best)
        k_star = int(np.argmax(c))
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        if i_start >= i_end:
            break
        idx = np.arange(i_start, i_end)
        contrib = best[idx] * best[k_star - idx]
        if contrib.sum() <= 1e-15:
            break
        order = np.argsort(contrib)[::-1]
        top_src = idx[order[: min(20, len(order))]]

        improved = False
        # move mass out of high-contrib positions into low positions
        low_idx = idx[np.argsort(contrib)[: min(20, len(order))]]
        for src in top_src:
            if time.monotonic() - t0 > t_limit:
                break
            sv = best[src]
            if sv < 1e-12:
                continue
            for dst in low_idx:
                if dst == src or best[dst] >= sv:
                    continue
                for frac in (0.05, 0.15, 0.3, 0.55, 0.8):
                    amt = sv * frac
                    a_t = best.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    a_t = a_t / a_t.sum()
                    jt = obj_from_a(a_t)
                    if jt < best_j - 1e-15:
                        best = a_t
                        best_j = jt
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # random nudge to escape
            i = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if i != j2 and best[i] > 1e-8:
                amt = best[i] * rng.uniform(0.05, 0.25)
                a_t = best.copy()
                a_t[i] -= amt
                a_t[j2] += amt
                a_t = np.maximum(a_t, 0.0)
                a_t = a_t / a_t.sum()
                jt = obj_from_a(a_t)
                if jt >= best_j:
                    break  # stuck -> stop this phase
                best = a_t
                best_j = jt

    return norm_simplex(best)

def init_cosine(n, mode=1):
    x = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    return norm_simplex(np.maximum(a, 0.0))


def init_bowl(n, sharp=2.0):
    x = np.linspace(0.0, 1.0, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharp + 0.08
    return norm_simplex(a)


def init_gauss(n, w=0.22):
    x = np.linspace(-1.0, 1.0, n)
    return norm_simplex(np.exp(-x ** 2 / (2.0 * w * w)))


def init_dip(n):
    x = np.linspace(-1.0, 1.0, n)
    a = 1.0 - 0.7 * np.exp(-x ** 2 / 0.06)
    return norm_simplex(np.maximum(a, 0.0))


def init_random_smooth(n, seed):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 0.6
    for f in range(1, 14):
        a += rng.normal(0, 0.10 / np.sqrt(f)) * np.sin(2 * np.pi * f * x
                                                        + rng.uniform(0, 6.283))
    return norm_simplex(np.maximum(a, 1e-4))


def init_parent_resample(parent, n):
    if parent is None:
        return None
    p = norm_simplex(parent)
    old_x = np.linspace(0.0, 1.0, len(p))
    new_x = np.linspace(0.0, 1.0, n)
    return norm_simplex(np.maximum(np.interp(new_x, old_x, p), 0.0))

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float("inf")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 0.01:
            return
        a = a / a.sum()
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            el = time.monotonic() - t_global
            print("  NEW BEST [%s]: %.6f (n=%d, t=%.0fs)" % (label, sc, len(a), el))

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            ps = evaluate_safe(norm_simplex(parent))
            print("Parent: n=%d score=%.6f" % (len(parent), ps))
            if np.isfinite(ps):
                best_seq = norm_simplex(parent)
                best_score = ps
    except Exception:
        pass

    if best_seq is None:
        best_seq = init_cosine(4000, 1)
        best_score = evaluate_safe(best_seq)
        print("Default init score=%.6f" % best_score)

    n0 = len(best_seq)
    print("Start n=%d score=%.6f  (target < 1.5027)" % (n0, best_score))

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # ---- candidate lengths (avoid enormous n) ---------------------------- #
    lengths = set()
    for n_c in (n0, 3000, 3500, 4000, 4500, 5000, 5500, 6000):
        if 2000 <= n_c <= 7000:
            lengths.add(int(n_c))
    lengths.add(max(2000, n0 - 500))
    lengths.add(min(7000, n0 + 500))
    lengths = sorted(lengths)

    inits_per_n = [
        ("cos", init_cosine),
        ("bowl", init_bowl),
        ("gauss", init_gauss),
        ("dip", init_dip),
        ("rands1", lambda nn: init_random_smooth(nn, 11)),
        ("rands2", lambda nn: init_random_smooth(nn, 29)),
    ]

    # budget per (n, init) pair
    n_pairs = len(lengths) * len(inits_per_n)
    per_pair = max(6.0, (time_budget * 0.70) / max(1, n_pairs))

    for n_c in lengths:
        if remaining() < 12:
            break
        for name, maker in inits_per_n:
            if remaining() < per_pair + 4:
                break
            try:
                base = maker(n_c)
            except Exception:
                continue
            # parent-aware blend for the first init of each length
            if name == "cos" and parent is not None:
                pr = init_parent_resample(parent, n_c)
                if pr is not None:
                    base = norm_simplex(0.5 * base + 0.5 * pr)

            el = min(per_pair, remaining() - 6)
            if el < 6:
                break
            # 1) spectral pre-flatten
            sp = spectral_flat(base, gamma=0.6, n_iter=25)
            # 2) mirror descent (main engine)
            md = mirror_descent(sp, el * 0.6)
            try_update(md, "md_%s_n%d" % (name, n_c))
            # 3) spectral polish on result
            if remaining() > el * 0.25 + 4:
                sp2 = spectral_polish(md, el * 0.25)
                try_update(sp2, "spmd_%s_n%d" % (name, n_c))

    # ---- targeted pair-swap on every local best we have ------------------ #
    # (rebuild a small pool from different starts, pair-swap each)
    pool_n = len(best_seq)
    if remaining() > 40:
        for name, maker in inits_per_n[:4]:
            if remaining() < 15:
                break
            base = maker(pool_n)
            sp = spectral_flat(base, gamma=0.6, n_iter=20)
            ps = pair_swap(sp, min(12.0, remaining() * 0.3))
            try_update(ps, "swap_%s_n%d" % (name, pool_n))

    # ---- final intensive polish on the global best ----------------------- #
    n_best = len(best_seq)
    if remaining() > 25:
        kb = min(remaining() * 0.35, 120.0)
        fn = mirror_descent(best_seq, kb)
        try_update(fn, "md_final_n%d" % n_best)
    if remaining() > 15:
        kb = min(remaining() * 0.30, 60.0)
        fn = spectral_polish(best_seq, kb)
        try_update(fn, "sp_final_n%d" % n_best)
    if remaining() > 12:
        kb = min(remaining() * 0.40, 50.0)
        fn = pair_swap(best_seq, kb)
        try_update(fn, "swap_final_n%d" % n_best)

    # ---- last-ditch: one more mirror descent on interpolated variants ----- #
    if remaining() > 20 and best_seq is not None:
        for d in (250, -250, 500):
            nn = n_best + d
            if nn < 2000 or nn > 7200:
                continue
            if remaining() < 12:
                break
            pr = init_parent_resample(best_seq, nn)
            if pr is None:
                continue
            kb = min(remaining() * 0.3, 40.0)
            fn = mirror_descent(spectral_flat(pr, 0.55, 15), kb)
            try_update(fn, "md_interp_n%d" % nn)

    # ---- final validation ----------------------------------------------- #
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("FINAL score=%.6f n=%d t=%.0fs" % (best_score, len(best_seq), time.monotonic() - t_global))
    else:
        best_seq = init_cosine(3500, 1)
        best_score = evaluate_safe(best_seq)
        print("Fallback score=%.6f" % best_score)

    return [float(x) for x in best_seq]
```
