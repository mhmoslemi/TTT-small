where μ is the Lagrange multiplier for `sum(a) = 1` (so μ*1 is the gradient of the constraint) and `ν_j ≥ 0` is the multiplier for `a_j ≥ 0` (with complementary slackness: `ν_j * a_j = 0`).

This gives us: for each j with `a_j > 0`:
`Σ_k λ_k * (∇b[k])_j + μ = 0`

And for j with `a_j = 0`:
`Σ_k λ_k * (∇b[k])_j + μ ≥ 0`

This is a system of equations/inequalities that characterizes the optimum. It's similar to the KKT conditions for the LP in the Frank-Wolfe step.

def _fast_conv(a):
    """Fast full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _norm1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _eval(a):
    """Evaluate the objective (assumes sum(a)=1)."""
    a = _norm1(a)
    b = _fast_conv(a)
    return 2.0 * len(a) * b.max()


def _build_lp_matrix(y, k_list, n):
    """Build the constraint matrix for the LP subproblem."""
    K = len(k_list)
    C = np.zeros((K, n))
    for jj, k in enumerate(k_list):
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            C[jj, idx] = y[k - idx]
    return C


def _fw_lp_direction(y, active_set, n, solver, max_K=3000):
    """Solve the LP subproblem to get the FW direction."""
    k_list = sorted(active_set)
    if len(k_list) > max_K:
        # Prune: keep only the most binding constraints
        ac = _fast_conv(y)
        M = ac.max()
        margins = np.array([M - ac[k] for k in k_list])
        keep = np.argsort(margins)[:max_K]
        k_list = [k_list[i] for i in keep]
    
    K = len(k_list)
    if K == 0:
        return y.copy()
    
    C = _build_lp_matrix(y, k_list, n)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [cp.sum(g) == 1, C @ g <= M]
    prob = cp.Problem(cp.Minimize(M), cons)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return y.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    s = g_val.sum()
    if s < 1e-15:
        return None
    return g_val / s


def _line_search(y, g, n, max_iters=50):
    """Find optimal step size via bisection on the 1D objective."""
    # Try full step first
    trial = _norm1(y + g)  # (1-1)*y + 1*g = g, then normalize
    M_full = _fast_conv(trial).max()
    
    # If full step is good, use it
    M_curr = _fast_conv(y).max()
    if M_full <= M_curr:
        return 1.0, trial
    
    # Bisection
    lo, hi = 0.0, 1.0
    best_alpha = 0.0
    best_M = M_curr
    best_trial = y.copy()
    
    for _ in range(20):
        mid = (lo + hi) / 2.0
        trial = _norm1((1.0 - mid) * y + mid * g)
        M_trial = _fast_conv(trial).max()
        if M_trial < best_M:
            best_M = M_trial
            best_alpha = mid
            best_trial = trial
        if M_trial < M_curr:
            lo = mid
        else:
            hi = mid
    
    return best_alpha, best_trial

def _coord_descent(a0, time_limit, top_k=20, verbose=True):
    """Coordinate descent polishing: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_M = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_conv(a)
        peaks = np.argsort(ac)[::-1][:top_k]
        
        improved = False
        for k in peaks:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue
            
            idx = np.arange(i_lo, i_hi + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            if a[src] < 1e-10:
                continue
            
            # Find a good destination: a position that doesn't contribute much to any peak
            # Score each position by its total contribution to top peaks
            scores = np.zeros(n)
            for k2 in peaks[:10]:
                k2 = int(k2)
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                idx2 = np.arange(s2, e2 + 1)
                scores[idx2] += a[k2 - idx2]
                scores[k2 - idx2] += a[idx2]
            
            dst = np.argmin(scores)
            if dst == src:
                dst = (src + 1) % n
            
            # Try to move some mass
            for frac in [0.01, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = _norm1(a_trial)
                M_trial = _fast_conv(a_trial).max()
                if M_trial < best_M - 1e-15:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    a[i] -= a[i] * frac
                    a[j] += a[i] * frac  # Bug: should use original a[i]
                    a = _norm1(a)
        
        if improved:
            stall = 0
    
    return best_a, best_M


def _interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
