def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def soft_max_gradient(a, beta):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs_matrix = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs_matrix[jj, idx] = y[k_int - idx]
    return coeffs_matrix

        active = np.where(ac >= threshold)[0]

        if stall > 0:
            stall += 1

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def final_perturbation(a0, time_limit, verbose):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue
            frac = float(rng.uniform(0.002, 0.02))
            amt = a[i] * frac
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            A_f = np.fft.rfft(a_trial, fft_size)
            b_trial = np.fft.irfft(A_f * A_f, fft_size)[:2 * n - 1]
            M_trial = b_trial.max()
            if M_trial < best_m - 1e-15:
                best_m = M_trial
                best_a = a_trial.copy()
                a = a_trial
                if verbose:
                    J = 2.0 * n * best_m
                    print(f"    Perturb: J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        i_arr = np.arange(n_base, dtype=np.float64)
        best_seq = norm_sum1((i_arr + 1.0) ** 0.7 * (n_base - i_arr + 1.0) ** 0.3)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_and_update_local(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s}")

    n0 = len(best_seq)

    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 5:
        a_param, _ = parametric_init(n0, best_seq)
        check_and_update_local(a_param, "parametric")

    n_best = len(best_seq)

    print(f"\n=== Phase 2: SoftMax Mirror Descent (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 90.0)
    if t2 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = softmax_mirror_descent(a_in, t2, 1.0, 30.0, 3.0, True)
        check_and_update_local(a_out, "softmax_md")

    n_best = len(best_seq)

    print(f"\n=== Phase 3: LP-FW main (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.35, 500.0)
    if t3 > 30 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t3, solver, 400, 3500, 60, 1.8, True)
        check_and_update_local(a_out, f"lp_main_n{n_best}")

    n_best = len(best_seq)

    print(f"\n=== Phase 4: KKT polish (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = kkt_polish(a_in, t4, solver, 40, True)
        check_and_update_local(a_out, f"kkt_n{n_best}")

    n_best = len(best_seq)

    print(f"\n=== Phase 5: Rapid relinearization (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20 and n_best <= 7000:
        a_in = norm_sum1(best_seq)
        a_out, _ = rapid_relinearize_lp(a_in, t5, solver, 8, True)
        check_and_update_local(a_out, f"rapid_n{n_best}")

    n_best = len(best_seq)

    print(f"\n=== Phase 6: Coordinate descent (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_polish(a_in, t6, True)
        check_and_update_local(a_out, f"coord_n{n_best}")

    n_best = len(best_seq)

    print(f"\n=== Phase 7: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    length_candidates = []
    for delta in [500, 1000, -500, -1000, 1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            length_candidates.append(new_n)
    for rn in [3000, 4000, 5000, 6000, 7000]:
        if rn != n_best and 2000 <= rn <= 8000:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.35, 100.0)
        if t6a > 10:
            a_out, _ = softmax_mirror_descent(a_long, t6a, 2.0, 40.0, 3.0, False)
            check_and_update_local(a_out, f"soft_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 6000:
            t6b = min(remaining * 0.5, 150.0)
            if t6b > 20:
                a_in2 = norm_sum1(best_seq) if len(best_seq) == new_n else interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_frank_wolfe_v2(a_in2, t6b, solver, 300, 2500, 50, 2.0, False)
                check_and_update_local(a_out, f"lp_n{new_n}")

    n_best = len(best_seq)

    print(f"\n=== Phase 8: Final LP-FW (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and n_best <= 7000:
        t8 = min(remaining * 0.30, 180.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v2(a_in, t8, solver, 500, 4000, 70, 2.0, True)
            check_and_update_local(a_out, f"lp_final_n{n_best}")

    n_best = len(best_seq)

    print(f"\n=== Phase 9: Final perturbation (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t9 = min(remaining * 0.40, 120.0)
        if t9 > 5:
            a_in = norm_sum1(best_seq)
            a_out, _ = final_perturbation(a_in, t9, True)
            check_and_update_local(a_out, f"perturb_n{n_best}")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    if not best_seq is not None:
        best_seq = [0.0] * 100
    return [float(x) for x in best_seq]
