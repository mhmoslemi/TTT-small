```python
def logsumexp_mgd(a0, time_limit, beta, eta0, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = eta0
    patience = 0
    max_patience = 500
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        # Compute softmax weights
        ac_shifted = ac - ac.max()  # for numerical stability
        p = np.exp(beta * ac_shifted)
        p = p / p.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        # Only use lags with significant weight
        thresh = 1.0 / (n * 10)  # threshold for significant contribution
        significant = np.where(p > thresh)[0]
        for k in significant:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += p[ki] * a[ki - idx]
        
        # Center gradient
        grad = grad - grad.mean()
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Accept if improvement
        M_new = fast_autocorr(a_new).max()
        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.5
            if eta < 0.001:
                eta = 0.01
                patience = max(0, patience - 200)
    
    return best_a
```
