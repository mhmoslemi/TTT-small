def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_value(a):

def proj_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fista_hankel(w, a0, steps, step_size, verbose=False):
    """FISTA to minimize a^T M(a) a where M_{ij} = w_{i+j}, subject to a in simplex.
    
    The gradient of f(a) = a^T M a is 2 * M @ a = 2 * hankel_mv(w, a).
    """
    n = len(a0)
    a = a0.copy()
    y = a0.copy()
    t = 1.0
    L = 2.0 * np.sum(w)  # Upper bound on Lipschitz constant of gradient (crude)
    # Actually, the Lipschitz constant of a^T M a is 2 * lambda_max(M)
    # We can bound lambda_max(M) <= sum_j w_j (trace bound) or use power iteration
    # For now, use a crude bound
    step = 1.0 / L
    if step <= 0 or not np.isfinite(step):
        step = 1.0 / max(1.0, 2.0 * np.sum(w))
    
    for k in range(steps):
        # Gradient step
        grad = 2.0 * hankel_mv(w, y)
        a_new = proj_simplex(y - step * grad)
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = proj_simplex(y)  # Ensure y stays in simplex for stability
        a = a_new
        t = t_new
    
    return a

def irls_optimize(a0, time_limit, p_power=2.0, verbose=False):
    """IRLS optimization: iteratively reweight peaks and solve QP via FISTA."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    irls_iter = 0
    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        M_curr = c.max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()
        
        # Compute weights: emphasize peaks
        # w_k = c_k^p (normalized so sum w_k = 1 for stability)
        eps = 1e-10
        w = (c + eps) ** p_power
        w_sum = w.sum()
        if w_sum < 1e-20:
            w = np.ones_like(c) / len(c)
        else:
            w = w / w_sum
        
        # Run FISTA for a few steps
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        # Estimate how many FISTA steps we can afford
        # Each step is O(n log n) ≈ O(n * log(n)) 
        # Rough: n=4000, log(n)≈12, so ~48000 ops per step
        # At ~100M ops/s, that's ~0.0005s per step, so ~2000 steps/s
        # But Python overhead makes it slower, maybe ~100-500 steps/s
        fista_steps = min(200, int(remaining * 100))
        if fista_steps < 10:
            fista_steps = 10
        
        a_new = fista_hankel(w, a, fista_steps, step_size=1.0, verbose=False)
        a_new = norm_simplex(a_new)
        
        # Check improvement
        M_new = fast_conv(a_new).max()
        if M_new < M_curr:
            a = a_new
        else:
            # Blend
            a = 0.5 * a + 0.5 * a_new
            a = norm_simplex(a)
        
        irls_iter += 1
        if verbose and irls_iter % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      IRLS iter={irls_iter} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m

            best_a = a_trial.copy()

            no_improve = 0
            # Increase aggressiveness slightly
            alpha = min(0.8, alpha * 1.02)
            blend = min(0.5, blend * 1.01)
        else:

            # Reduce aggressiveness

def subgradient_optimize(a0, time_limit, verbose=False):
    """Subgradient method with adaptive step size and momentum."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Momentum
    v = np.zeros(n)
    momentum = 0.9
    lr = 1.0 / np.sqrt(n)
    
    # Polyak-like step size
    f_best = 2.0 * n * best_m  # current best "loss"
    f_lower = 1.0  # theoretical lower bound (approximately)
    
    no_improve = 0
    max_no_improve = 500
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        c = fast_conv(a)
        M_curr = c.max()
        
        # Subgradient: for the max, pick the argmax
        # But use top-k with softmax for smoothness
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        
        # Annealed softmax
        beta = 20.0 + 50.0 * min(1.0, t / 50000.0)
        exps = beta * (top_vals - top_vals[0]) / max(top_vals[0], 1e-10)
        exps = np.maximum(exps, -50.0)
        w = np.exp(exps)
        w = w / w.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        # Adaptive step size (Polyak)
        gap = f_best - f_lower
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq > 1e-20:
            lr_polyak = min(lr, gap / (2.0 * grad_norm_sq))
        else:
            lr_polyak = lr
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr_polyak * v
        a_new = proj_simplex(a_new)
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            f_best = 2.0 * n * best_m
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        # Decay step size
        lr *= 0.9999
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def hamming_window(n):
    """Generate a Hamming window normalized to sum=1."""
    x = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
    return norm_simplex(w)

def hann_window(n):
    """Generate a Hann window normalized to sum=1."""
    x = np.arange(n)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1)))
    return norm_simplex(w)

def blackman_window(n):
    """Generate a Blackman window normalized to sum=1."""
    x = np.arange(n)
    w = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
    return norm_simplex(w)

def nuttall_window(n):
    """Generate a Nuttall window normalized to sum=1."""
    a = np.array([0.3635819, 0.4891775, 0.1365995, 0.0106411])
    x = np.arange(n)
    w = np.ones(n)
    for i, ai in enumerate(a):
        if i == 0:
            w += ai
        else:
            w += ai * np.cos(2.0 * np.pi * i * x / (n - 1))
    w = np.abs(w)
    return norm_simplex(w)

def polynomial_window(n, order=4):
    """Generate a polynomial (raised cosine) window."""
    x = np.linspace(-1, 1, n)
    w = (1 - x**2) ** order
    return norm_simplex(w)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive spectral flattening with smooth non-negativity handling
2. IRLS (Iteratively Reweighted Least Squares) with FISTA
   - Exploits Hankel matrix structure for O(n log n) matrix-vector products
3. Subgradient method with adaptive step size and momentum
4. Multi-start with diverse window-function initializations and parent sequence
"""

import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):

def proj_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[:: -1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_vals = np.nonzero(cond)[0]
    if len(rho_vals) == 0:
        return np.ones(n) / n
    rho = rho_vals[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def hankel_mv(w, a):
    """Compute H @ a where H_{ij} = w[i+j], using FFT.
    w has length 2n-1, a has length n, result has length n.
    """
    n = len(a)
    w_rev = w[::-1]
    fft_size = 1 << (3 * n).bit_length()
    A2 = np.fft.rfft(a, fft_size)
    W_rev = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A2 * W_rev, fft_size)
    result = conv_result[n - 1: 2 * n - 1][::-1]
    return result


def fista_hankel(w, a0, steps):
    """FISTA to minimize a^T M_w a where (M_w)_{ij} = w_{i+j}, subject to simplex."""
    n = len(a0)
    a = a0.copy()
    y = a0.copy()
    t_mom = 1.0
    w_sum = np.sum(w)
    L = 2.0 * max(w_sum, 1e-10)
    step = 1.0 / L

    for k in range(steps):
        grad = 2.0 * hankel_mv(w, y)
        a_new = proj_simplex(y - step * grad)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        y = proj_simplex(y)
        a = a_new
        t_mom = t_new

    return a


def irls_optimize(a0, time_limit, p_power=2.0, verbose=False):
    """IRLS optimization: iteratively reweight peaks and solve QP via FISTA."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    irls_iter = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        if no_improve > 100:
            break

        c = fast_conv(a)
        M_curr = c.max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0

        eps = 1e-10
        w = (c + eps) ** p_power
        w_sum = w.sum()
        if w_sum < 1e-20:
            w = np.ones_like(c) / len(c)
        else:
            w = w / w_sum

        remaining = time_limit - (time.monotonic() - t0)
        fista_steps = min(300, max(30, int(remaining * 50)))

        a_new = fista_hankel(w, a, fista_steps)

        M_new = fast_conv(a_new).max()
        if M_new < M_curr * (1.0 - 1e-10):
            a = a_new
            no_improve = 0
        else:

        if verbose and irls_iter % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      IRLS iter={irls_iter} ({elapsed:.0f}s): J={2 * n * best_m:.6f}")

def spectral_optimize_v2(a0, time_limit, verbose=False):
    """Enhanced spectral optimization with adaptive parameters."""
    n = len(a0)

    best_a = a.copy()

        if no_improve > 80:
            break

        mag = np.abs(A)
        phase = np.angle(A)

        mag_mean = mag.mean()
        if mag_mean < 1e-30:
            break
        ratio = mag / (mag_mean + 1e-30)
        mag_new = mag_mean * ratio ** (1.0 - alpha)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        a_soft = np.where(a_new > 0, a_new, a_new * 0.05)
        a_soft = np.maximum(a_soft, 0.0)
        s = a_soft.sum()
        if s < 1e-15:
            a_soft = np.ones(n) / n
        else:
            a_soft = a_soft / s

        a_trial = (1.0 - blend) * a + blend * a_soft
        a_trial = norm_simplex(a_trial)

        M_trial = fast_conv(a_trial).max()
        if M_trial < best_m - 1e-15:
            best_m = M_trial
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            alpha = min(0.75, alpha * 1.01)
            blend = min(0.5, blend * 1.005)
        else:
            no_improve += 1
            alpha = max(0.05, alpha * 0.97)
            blend = max(0.05, blend * 0.95)

    return best_a, best_m


def subgradient_optimize(a0, time_limit, verbose=False):
    """Subgradient method with adaptive step size and momentum."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    v = np.zeros(n)
    momentum = 0.85
    lr = 5.0 / n

    no_improve = 0
    f_best = 2.0 * n * best_m

        M_curr = c.max()

        beta = 15.0 + 60.0 * min(1.0, t / 30000.0)
        max_v = top_vals[0]
        if max_v < 1e-20:
            w = np.ones(top_k) / top_k
        else:
            exps = beta * (top_vals - max_v) / max_v
            exps = np.maximum(exps, -50.0)
            w = np.exp(exps)
            w_sum = w.sum()
            if w_sum < 1e-20:
                w = np.ones(top_k) / top_k
            else:
                w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        gap = max(f_best - f_lower, 0.1)
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq > 1e-20:
            lr_eff = min(lr, gap / (2.0 * grad_norm_sq + 1e-10))
        else:
            lr_eff = lr

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr_eff * v
        a_new = proj_simplex(a_new)

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()

            no_improve = 0
        else:

        lr *= 0.9998

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad t={t} ({elapsed:.0f}s): J={2 * n * best_m:.6f}")

    return best_a, best_m


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return norm_simplex(a)


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def hamming_window(n):
    x = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / max(n - 1, 1))
    return norm_simplex(w)


def hann_window(n):
    x = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    return norm_simplex(w)


def blackman_window(n):
    x = np.arange(n, dtype=np.float64)
    w = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / max(n - 1, 1)) + 0.08 * np.cos(4.0 * np.pi * x / max(n - 1, 1))
    return norm_simplex(w)


def nuttall_window(n):
    coeffs = np.array([0.3635819, 0.4891775, 0.1365995, 0.0106411])
    x = np.arange(n, dtype=np.float64)
    w = np.zeros(n)

        w += coeffs[i] * np.cos(np.pi * i * x / max(n - 1, 1))
    w = np.abs(w)
    return norm_simplex(w)


def polynomial_window(n, order=4):
    x = np.linspace(-1, 1, n)
    w = (1.0 - x ** 2) ** order + 1e-6
    return norm_simplex(w)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def do_try_update(best_seq, best_score, a, label, t_global):
    """Attempt to update best sequence. Returns (best_seq, best_score)."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_init_bowl(n_fb, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
