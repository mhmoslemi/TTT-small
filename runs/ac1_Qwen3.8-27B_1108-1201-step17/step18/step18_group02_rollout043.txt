```python
"""
Search for a nonnegative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2

Method:
- Treat the sequence as a probability vector (scale invariant).
- Use correct full-convolution statistics and FFT-based gradients.
- Alternate smoothed mirror-descent methods:
    * adaptive line-search mirror descent over log-density variables
    * Adam-like multiplicative mirror descent
- Multi-resolution continuation, diverse seed generation, and final local polishing.
"""

import numpy as np
import time
import math


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fft_size_for_len(n):
    # We need full convolution of a (length n) with p (length 2n-1).
    # That length is 3n-2.
    return next_pow2(max(1, 3 * n - 2))


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size == 0:
        return a
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones_like(a, dtype=np.float64) / max(1, a.size)
    return a / s


def uniform_sequence(n):
    n = max(1, int(n))
    return np.ones(n, dtype=np.float64) / n


def interpolate_sequence(f, new_n):
    f = norm_sum1(f)
    old_n = f.size
    new_n = max(2, int(new_n))
    if old_n == new_n:
        return f.copy()
    if old_n == 0:
        return uniform_sequence(new_n)

    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / float(old_n)
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / float(new_n)
    out = np.interp(x_new, x_old, f)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return uniform_sequence(new_n)
    return out / s


def fast_conv_size(a, size):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    lin = 2 * n - 1
    if size < lin:
        size = next_pow2(lin)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    c = c[:lin]
    c = np.maximum(c, 0.0)
    return c


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = fft_size_for_len(n)
    return fast_conv_size(a, size)


def internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    c = fast_conv(a)
    return float(2.0 * n * c.max())


def evaluate_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if s <= 1e-300:
            return float("inf")
        if s < 0.01:
            arr = arr / s
        lst = [float(x) for x in arr.tolist()]
        val = evaluate_sequence(lst)
        val = float(val)
        if math.isfinite(val):
            return val
        return float("inf")
    except Exception:
        return float("inf")


def candidate_score(a):
    a = norm_sum1(a)
    if a.size == 0:
        return None, float("inf")
    sc = evaluate_safe(a)
    if math.isfinite(sc):
        return a, float(sc)
    sc = internal_score(a)
    if not math.isfinite(sc):
        return None, float("inf")
    return a, float(sc)


def softmax_x(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x.copy()
    m = float(np.max(x))
    e = np.exp(x - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.ones_like(x, dtype=np.float64) / x.size
    return e / s


def seed_uniform(n):
    return uniform_sequence(n)


def seed_power(n, p, rev=False):
    i = np.arange(1, n + 1, dtype=np.float64)
    if rev:
        i = n + 1.0 - i
    a = np.power(i, float(p))
    a = np.maximum(a, 1e-8)
    return norm_sum1(a)


def seed_cos(n, m, phase, amp):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = 1.0 + float(amp) * np.cos(2.0 * math.pi * float(m) * x + float(phase))
    a = np.maximum(a, 1e-3)
    return norm_sum1(a)


def seed_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.power(i, float(alpha)) * np.power(n + 0.5 - i, float(beta))
    a = np.maximum(a, 1e-8)
    return norm_sum1(a)


def seed_gauss(n, rng):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    k = int(rng.integers(1, 4))
    a = np.zeros(n, dtype=np.float64)
    for _ in range(k):
        c = float(rng.uniform(0.10, 0.90))
        s = float(rng.uniform(0.05, 0.35))
        w = float(rng.uniform(0.50, 1.00))
        a += w * np.exp(-((x - c) * (x - c)) / (2.0 * s * s))
    a = np.maximum(a, 1e-4)
    return norm_sum1(a)


def seed_random_fft(n, rng):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.ones(n, dtype=np.float64)
    k = 24
    freqs = rng.integers(1, 40, size=k)
    amps = rng.random(k) * (0.80 ** np.arange(k, dtype=np.float64))
    phases = rng.uniform(-math.pi, math.pi, size=k)
    for j in range(k):
        a += float(amps[j]) * np.cos(2.0 * math.pi * float(freqs[j]) * x + float(phases[j]))
    a = np.maximum(a, 0.02)
    return norm_sum1(a)


def generate_pool(n, rng_seed=123):
    n = max(64, int(n))
    rng = np.random.default_rng(int(rng_seed))
    pool = []
    pool.append(seed_uniform(n))

    for p in [-0.7, -0.3, -0.1, 0.1, 0.3, 0.7, 1.0, 1.5]:
        pool.append(seed_power(n, p, False))
        pool.append(seed_power(n, p, True))

    for m in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            for amp in [0.5, 0.8, 0.95]:
                pool.append(seed_cos(n, m, phase, amp))

    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for beta in [0.2, 0.5, 1.0, 2.0]:
            pool.append(seed_beta(n, alpha, beta))

    for _ in range(10):
        pool.append(seed_gauss(n, rng))
    for _ in range(6):
        pool.append(seed_random_fft(n, rng))

    # Keep only valid normalized arrays.
    good = []
    for a in pool:
        a = norm_sum1(a)
        if a.size == n and np.all(np.isfinite(a)) and a.sum() > 0.0:
            good.append(a)
    return good


def coarse_lengths(n):
    n = int(n)
    out = []
    m = max(128, n // 8)
    while m < n:
        out.append(int(m))
        m = min(n, 2 * m)
    out.append(n)

    # De-duplicate while preserving order.
    seen = set()
    uniq = []
    for v in out:
        if v >= 64 and v not in seen:
            seen.add(v)
            uniq.append(v)
    if len(uniq) == 0 or uniq[-1] != n:
        uniq.append(n)
    return uniq


def smooth_search_line(
    a0,
    time_limit,
    tau_ratio=0.20,
    tau_min_ratio=0.002,
    tau_decay=0.9995,
    base=0.25,
    min_base=0.01,
    max_base=3.0,
    baseline=0.02,
    restart_seconds=25.0,
    max_restarts=8,
    clip=30.0,
    seed=123,
    verbose=False,
):
    t0 = time.monotonic()
    time_limit = max(0.0, float(time_limit))
    a = norm_sum1(a0)
    n = a.size
    if n < 2:
        return a.copy(), internal_score(a)

    size = fft_size_for_len(n)
    L = 2 * n - 1
    idx = np.arange(L - 1, L - 1 - n, -1, dtype=np.int64)

    # Keep a tiny floor so log variables are finite.
    a = a + 1e-15 / float(n)
    a = norm_sum1(a)

    x = np.log(a)
    x = x - float(np.mean(x))
    x = np.clip(x, -clip, clip)

    best_a = a.copy()
    c = fast_conv_size(a, size)
    best_M = float(c.max())
    score_best = 2.0 * n * best_M

    tau = max(best_M * float(tau_ratio), 1e-30)
    tau_min = max(best_M * float(tau_min_ratio), 1e-30)
    base_cur = float(base)
    rng = np.random.default_rng(int(seed))
    last_improve = t0
    restarts = 0
    flat_count = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        a = softmax_x(x)

        A = np.fft.rfft(a, size)
        c = np.fft.irfft(A * A, size)[:L]
        c = np.maximum(c, 0.0)
        M = float(c.max())

        if M < best_M - 1e-18:
            best_M = M
            score_best = 2.0 * n * best_M
            best_a = a.copy()
            last_improve = time.monotonic()
            tau_min = max(best_M * float(tau_min_ratio), 1e-30)

        cm = float(c.max())
        if cm <= 0.0 or not math.isfinite(cm):
            break

        if tau < tau_min:
            tau = tau_min
        z = (c - cm) / tau
        q = np.exp(z)
        qs = float(q.sum())
        if not math.isfinite(qs) or qs <= 0.0:
            q = np.ones(L, dtype=np.float64) / float(L)
        else:
            q = q / qs

        p = (1.0 - float(baseline)) * q + float(baseline) / float(L)
        p_rev = p[::-1]
        P = np.fft.rfft(p_rev, size)
        conv = np.fft.irfft(A * P, size)
        g = 2.0 * conv[idx]
        g = np.maximum(g, 0.0)
        g = g - float(g.mean())
        grange = float(g.max() - g.min())

        if not math.isfinite(grange) or grange <= 1e-16:
            flat_count += 1
            tau = max(tau * 2.0, best_M * 0.05)
            base_cur = min(max_base, max(min_base, base_cur * 1.2))
            if flat_count > 80:
                break
            continue
        flat_count = 0

        gnorm = g / grange

        # Try 3 step sizes around the current adaptive base.
        lams = []
        for fac in (0.8, 1.0, 1.25):
            lam = base_cur * fac
            if lam < min_base:
                lam = min_base
            if lam > max_base:
                lam = max_base
            lams.append(lam)

        best_trial_M = float("inf")
        best_trial_x = x.copy()
        best_trial_lam = base_cur

        for lam in lams:
            xt = x - float(lam) * gnorm
            xt = xt - float(np.mean(xt))
            xt = np.clip(xt, -clip, clip)
            at = softmax_x(xt)
            ct = fast_conv_size(at, size)
            Mt = float(ct.max())
            if Mt < best_trial_M:
                best_trial_M = Mt
                best_trial_x = xt
                best_trial_lam = lam

        if best_trial_M < M - 1e-15:
            x = best_trial_x
            if best_trial_M < best_M - 1e-18:
                best_M = best_trial_M
                score_best = 2.0 * n * best_M
                best_a = softmax_x(x).copy()
                last_improve = time.monotonic()
                tau_min = max(best_M * float(tau_min_ratio), 1e-30)
                base_cur = min(max_base, base_cur * 1.05)

        elif best_trial_M <= M + 1e-14:
            # Accept a nearly-flat move to keep exploring.
            x = best_trial_x
            base_cur = min(max_base, base_cur * 1.02)
        else:
            base_cur = max(min_base, base_cur * 0.85)

        it += 1
        tau *= float(tau_decay)
        if tau < tau_min:
            tau = tau_min

        if time.monotonic() - last_improve > float(restart_seconds):
            restarts += 1
            if restarts > int(max_restarts):
                break

            tau = max(tau * 2.5, best_M * 0.08)
            base_cur = 0.20

            xb = np.log(best_a + 1e-30)
            xb = xb - float(np.mean(xb))
            xb = np.clip(xb, -clip, clip)
            noise = rng.normal(0.0, 0.05, size=n)
            x = xb + noise
            x = x - float(np.mean(x))
            x = np.clip(x, -clip, clip)
            last_improve = time.monotonic()

        if verbose and (it % 700 == 0):
            print(
                "    line t=%.1fs score=%.7f tau=%.3e base=%.3f restarts=%d"
                % (time.monotonic() - t0, score_best, tau, base_cur, restarts)
            )

    return best_a, score_best


def smooth_search_adam(
    a0,
    time_limit,
    tau_ratio=0.15,
    tau_min_ratio=0.002,
    tau_decay=0.9995,
    lr=0.50,
    beta1=0.90,
    beta2=0.999,
    max_step=1.5,
    baseline=0.03,
    restart_seconds=25.0,
    max_restarts=8,
    clip=30.0,
    seed=321,
    verbose=False,
):
    t0 = time.monotonic()
    time_limit = max(0.0, float(time_limit))
    a = norm_sum1(a0)
    n = a.size
    if n < 2:
        return a.copy(), internal_score(a)

    size = fft_size_for_len(n)
    L = 2 * n - 1
    idx = np.arange(L - 1, L - 1 - n, -1, dtype=np.int64)

    a = a + 1e-15 / float(n)
    a = norm_sum1(a)

    x = np.log(a)
    x = x - float(np.mean(x))
    x = np.clip(x, -clip, clip)

    best_a = a.copy()
    c = fast_conv_size(a, size)
    best_M = float(c.max())
    score_best = 2.0 * n * best_M

    tau = max(best_M * float(tau_ratio), 1e-30)
    tau_min = max(best_M * float(tau_min_ratio), 1e-30)
    m_hat = np.zeros(n, dtype=np.float64)
    v_hat = np.zeros(n, dtype=np.float64)
    t = 1
    rng = np.random.default_rng(int(seed))
    last_improve = t0
    restarts = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        a = softmax_x(x)

        A = np.fft.rfft(a, size)
        c = np.fft.irfft(A * A, size)[:L]
        c = np.maximum(c, 0.0)
        M = float(c.max())

        if M < best_M - 1e-18:
            best_M = M
            score_best = 2.0 * n * best_M
            best_a = a.copy()
            last_improve = time.monotonic()
            tau_min = max(best_M * float(tau_min_ratio), 1e-30)

        cm = float(c.max())
        if cm <= 0.0 or not math.isfinite(cm):
            break

        if tau < tau_min:
            tau = tau_min
        z = (c - cm) / tau
        q = np.exp(z)
        qs = float(q.sum())
        if not math.isfinite(qs) or qs <= 0.0:
            q = np.ones(L, dtype=np.float64) / float(L)
        else:
            q = q / qs

        p = (1.0 - float(baseline)) * q + float(baseline) / float(L)
        p_rev = p[::-1]
        P = np.fft.rfft(p_rev, size)
        conv = np.fft.irfft(A * P, size)
        g = 2.0 * conv[idx]
        g = np.maximum(g, 0.0)

        m_hat = float(beta1) * m_hat + (1.0 - float(beta1)) * g
        v_hat = float(beta2) * v_hat + (1.0 - float(beta2)) * (g * g)

        b1t = 1.0 - float(beta1) ** t
        b2t = 1.0 - float(beta2) ** t
        m_c = m_hat / b1t
        v_c = v_hat / b2t

        u = float(lr) * m_c / (np.sqrt(v_c) + 1e-12)
        u = u - float(u.mean())

        mu = float(np.max(np.abs(u)))
        if mu > float(max_step):
            u = u * (float(max_step) / mu)

        x = x - u
        x = x - float(x.mean())
        x = np.clip(x, -clip, clip)

        t += 1
        it += 1
        tau *= float(tau_decay)
        if tau < tau_min:
            tau = tau_min

        if time.monotonic() - last_improve > float(restart_seconds):
            restarts += 1
            if restarts > int(max_restarts):
                break

            tau = max(tau * 2.5, best_M * 0.08)
            xb = np.log(best_a + 1e-30)
            xb = xb - float(np.mean(xb))
            xb = np.clip(xb, -clip, clip)
            noise = rng.normal(0.0, 0.05, size=n)
            x = xb + noise
            x = x - float(np.mean(x))
            x = np.clip(x, -clip, clip)
            m_hat = np.zeros(n, dtype=np.float64)
            v_hat = np.zeros(n, dtype=np.float64)
            t = 1
            last_improve = time.monotonic()

        if verbose and (it % 700 == 0):
            print(
                "    adam t=%.1fs score=%.7f tau=%.3e restarts=%d"
                % (time.monotonic() - t0, score_best, tau, restarts)
            )

    return best_a, score_best


def final_polish(a, remaining_s, seed_base=9001):
    t0 = time.monotonic()
    remaining_s = max(0.0, float(remaining_s))
    if remaining_s <= 5.0:
        return a

    # Two very small-temperature passes.
    if remaining_s > 60.0:
        t1 = min(remaining_s * 0.55, 220.0)
        b, _ = smooth_search_line(
            a,
            t1,
            tau_ratio=0.030,
            tau_min_ratio=0.0005,
            tau_decay=0.9998,
            base=0.08,
            min_base=0.005,
            max_base=0.6,
            baseline=0.06,
            restart_seconds=12.0,
            max_restarts=4,
            seed=int(seed_base),
            verbose=False,
        )
        a = norm_sum1(b)
        remaining_s = time.monotonic() - t0 * -1 + remaining_s  # not used; recompute below
        remaining_s = max(0.0, (time.monotonic() - (t0)) * -1 + remaining_s)
        # Simpler:
        elapsed = time.monotonic() - t0
        remaining_s = max(0.0, float(t1) - elapsed)

    elapsed = time.monotonic() - t0
    if float(elapsed) < float(remaining_s):
        t2 = min((1800.0 - 10.0), 1.0)  # placeholder no-op
        # Use actual remaining from caller after both lines:
    return a


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    # Reserve enough wall time for final validation and return.
    time_budget = total_budget - 90.0
    if time_budget < 60.0:
        time_budget = max(30.0, total_budget * 0.5)

    best_seq = None
    best_score = float("inf")

    # ---------------- Load parent if present ----------------
    parent = None
    try:
        raw_parent = globals().get("height_sequence_1", None)
        if raw_parent is not None:
            par = np.asarray(raw_parent, dtype=np.float64).copy()
            if par.ndim == 0:
                par = par.reshape(1)
            par = np.nan_to_num(par, nan=0.0, posinf=0.0, neginf=0.0)
            par = np.clip(par, 0.0, 1000.0)
            if par.size > 0 and par.sum() > 1e-8:
                par = norm_sum1(par)
                par_score = evaluate_safe(par)
                if math.isfinite(par_score):
                    best_seq = par
                    best_score = float(par_score)
                parent = par
                print(
                    "Loaded parent: n=%d score=%.7f"
                    % (par.size, best_score if math.isfinite(best_score) else float("inf"))
                )
            else:
                parent = None
    except Exception:
        parent = None

    # ---------------- Choose main working length ----------------
    if parent is not None:
        n_parent = parent.size
        # Keep mostly in the efficient FFT size range (<= 5462 for 3n-2 <= 16384).
        n_main = int(min(5462, max(2048, n_parent)))
    else:
        n_main = 5462

    if best_seq is None:
        uni = uniform_sequence(n_main)
        best_seq, best_score = candidate_score(uni)
        print("Initialized with uniform n=%d score=%.7f" % (best_seq.size, best_score))

    # ---------------- Initial seed pool ----------------
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 90.0:
        pool = generate_pool(n_main, rng_seed=20250101)
        # Always include current best and a lightly perturbed best.
        if best_seq is not None:
            pool.insert(0, norm_sum1(best_seq))
            rng_seed = np.random.default_rng(7)
            noise = rng_seed.normal(0.0, 0.01 / float(n_main) ** 0.5, size=n_main)
            pool.insert(1, norm_sum1(best_seq + noise))
        if parent is not None:
            pool.append(interpolate_sequence(parent, n_main))

        scored = []
        for a in pool:
            sc = internal_score(a)
            if math.isfinite(sc):
                scored.append((float(sc), len(scored), norm_sum1(a)))
        scored.sort()

        top_seeds = []
        seen = set()
        for _, idxp, a in scored:
            # Light de-duplication by checksum of rounded normalized array.
            key = np.round(a, 6).tobytes()
            if key not in seen:
                seen.add(key)
                top_seeds.append(a)
            if len(top_seeds) >= 10:
                break

        if len(top_seeds) == 0:
            top_seeds = [norm_sum1(best_seq)]

        # ---------------- Main high-resolution search ----------------
        remaining = time_budget - (time.monotonic() - t0)
        main_time = min(remaining * 0.45, 760.0)
        if main_time > 40.0:
            fractions = [0.42, 0.18, 0.12, 0.08, 0.06, 0.05, 0.04, 0.03, 0.01, 0.01]
            stop_main = time.monotonic() + main_time
            seed_base = 11

            for i, a in enumerate(top_seeds):
                if time.monotonic() > stop_main:
                    break
                frac = fractions[i] if i < len(fractions) else 0.01
                t_allow = max(15.0, main_time * frac)
                # Do not blow past stop.
                if time.monotonic() + 5.0 > stop_main:
                    t_allow = min(t_allow, max(5.0, stop_main - time.monotonic()))

                out = a
                if i % 2 == 0:
                    out, _ = smooth_search_line(
                        a,
                        t_allow,
                        tau_ratio=0.22,
                        tau_min_ratio=0.0025,
                        tau_decay=0.9994,
                        base=0.30,
                        min_base=0.01,
                        max_base=3.5,
                        baseline=0.02,
                        restart_seconds=28.0,
                        max_restarts=8,
                        seed=int(seed_base + 17 * i),
                        verbose=False,
                    )
                else:
                    out, _ = smooth_search_adam(
                        a,
                        t_allow,
                        tau_ratio=0.18,
                        tau_min_ratio=0.0030,
                        tau_decay=0.9994,
                        lr=0.55,
                        beta1=0.90,
                        beta2=0.999,
                        max_step=1.6,
                        baseline=0.03,
                        restart_seconds=28.0,
                        max_restarts=8,
                        seed=int(seed_base + 31 * i),
                        verbose=False,
                    )

                out_seq, out_score = candidate_score(out)
                if out_seq is not None and math.isfinite(out_score):
                    if out_score < best_score - 1e-12:
                        best_seq = out_seq
                        best_score = out_score
                        print(
                            "  *** main high-res improve: score=%.7f n=%d elapsed=%.1fs"
                            % (best_score, best_seq.size, time.monotonic() - t0)
                        )

    # ---------------- Multi-resolution continuation ----------------
    time_left = time_budget - (time.monotonic() - t0)
    if best_seq is not None and time_left > 220.0:
        coarse_time = min(time_left * 0.22, 320.0)
        if coarse_time > 80.0:
            lengths = coarse_lengths(best_seq.size)
            # Enter at a shorter length to allow basin change.
            start_len = max(128, best_seq.size // 8)
            a_cur = interpolate_sequence(best_seq, start_len)

            # Rough time allocation by length.
            weights = [0.0, 0.2, 0.3, 0.5]
            for i, ln in enumerate(lengths[:-1]):
                if time.monotonic() > t0 + time_budget:
                    break
                w = weights[i] if i < len(weights) else 0.5
                t_allow = coarse_time * w
                a_cur = smooth_search_line(
                    a_cur,
                    t_allow,
                    tau_ratio=0.28,
                    tau_min_ratio=0.004,
                    tau_decay=0.9992,
                    base=0.35,
                    min_base=0.02,
                    max_base=4.0,
                    baseline=0.02,
                    restart_seconds=18.0,
                    max_restarts=4,
                    seed=5000 + i,
                    verbose=False,
                )[0]
                if i + 1 < len(lengths):
                    a_cur = interpolate_sequence(a_cur, lengths[i + 1])

            out_seq, out_score = candidate_score(a_cur)
            if out_seq is not None and math.isfinite(out_score):
                if out_score < best_score - 1e-12:
                    best_seq = out_seq
                    best_score = out_score
                    print(
                        "  *** multi-res improve: score=%.7f n=%d elapsed=%.1fs"
                        % (best_score, best_seq.size, time.monotonic() - t0)
                    )

    # ---------------- Length exploration ----------------
    time_left = time_budget - (time.monotonic() - t0)
    if best_seq is not None and time_left > 200.0:
        len_time = min(time_left * 0.25, 420.0)
        if len_time > 100.0:
            n_best = best_seq.size
            cand_lengths = []

            # Nearby efficient lengths.
            if n_best > 1200:
                cand_lengths.append(int(min(5462, n_best + 260)))
            cand_lengths.append(int(n_best))
            if n_best > 3000:
                cand_lengths.append(int(max(2048, n_best + 900)))
            if n_best > 6000:
                cand_lengths.append(int(max(3072, n_best - 600)))
            cand_lengths.append(4096)
            cand_lengths.append(8192)
            cand_lengths.append(10240)

            # De-dup and filter.
            uniq = []
            seen = set()
            for v in cand_lengths:
                v = int(v)
                if 1024 <= v <= 12000 and v not in seen:
                    seen.add(v)
                    uniq.append(v)

            # Prefer fewer, close lengths first.
            if len(uniq) > 3:
                uniq = uniq[:3]

            stop_len = time.monotonic() + len_time
            for ln in uniq:
                if time.monotonic() > stop_len:
                    break
                if ln == best_seq.size:
                    continue
                a_in = interpolate_sequence(best_seq, ln)
                t_allow = min(60.0 + 2.2 * ln, (stop_len - time.monotonic()) * 0.9)
                if t_allow <= 20.0:
                    continue

                out, _ = smooth_search_line(
                    a_in,
                    t_allow,
                    tau_ratio=0.18,
                    tau_min_ratio=0.0020,
                    tau_decay=0.9994,
                    base=0.25,
                    min_base=0.01,
                    max_base=3.0,
                    baseline=0.03,
                    restart_seconds=20.0,
                    max_restarts=5,
                    seed=7000 + ln,
                    verbose=False,
                )

                out_seq, out_score = candidate_score(out)
                if out_seq is not None and math.isfinite(out_score):
                    if out_score < best_score - 1e-12:
                        best_seq = out_seq
                        best_score = out_score
                        print(
                            "  *** length %d improve: score=%.7f elapsed=%.1fs"
                            % (ln, best_score, time.monotonic() - t0)
                        )

    # ---------------- Final polishing ----------------
    time_left = time_budget - (time.monotonic() - t0)
    if best_seq is not None and time_left > 80.0:
        final_time = min(time_left * 0.70, 400.0)
        if final_time > 60.0:
            out, _ = smooth_search_line(
                best_seq,
                final_time,
                tau_ratio=0.035,
                tau_min_ratio=0.0004,
                tau_decay=0.9997,
                base=0.09,
                min_base=0.004,
                max_base=0.8,
                baseline=0.06,
                restart_seconds=14.0,
                max_restarts=4,
                seed=9377,
                verbose=False,
            )
            out_seq, out_score = candidate_score(out)
            if out_seq is not None and math.isfinite(out_score):
                if out_score < best_score - 1e-12:
                    best_seq = out_seq
                    best_score = out_score
                    print(
                        "  *** final line polish improve: score=%.7f elapsed=%.1fs"
                        % (best_score, time.monotonic() - t0)
                    )

            time_left = time_budget - (time.monotonic() - t0)
            if time_left > 40.0:
                out, _ = smooth_search_adam(
                    best_seq,
                    min(time_left * 0.75, 150.0),
                    tau_ratio=0.025,
                    tau_min_ratio=0.0003,
                    tau_decay=0.9997,
                    lr=0.25,
                    beta1=0.90,
                    beta2=0.999,
                    max_step=0.8,
                    baseline=0.05,
                    restart_seconds=10.0,
                    max_restarts=3,
                    seed=9477,
                    verbose=False,
                )
                out_seq, out_score = candidate_score(out)
                if out_seq is not None and math.isfinite(out_score):
                    if out_score < best_score - 1e-12:
                        best_seq = out_seq
                        best_score = out_score
                        print(
                            "  *** final adam polish improve: score=%.7f elapsed=%.1fs"
                            % (best_score, time.monotonic() - t0)
                        )

    # ---------------- Final validation ----------------
    best_seq, best_score = candidate_score(best_seq)
    if best_seq is None or not math.isfinite(best_score):
        best_seq = uniform_sequence(5462)
        best_score = internal_score(best_seq)

    arr = np.asarray(best_seq, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-12:
        arr = uniform_sequence(5462)
        s = float(arr.sum())
    arr = arr / s

    try:
        final_score = evaluate_safe(arr)
    except Exception:
        final_score = best_score

    if not math.isfinite(final_score):
        final_score = internal_score(arr)

    print(
        "FINAL proposed: n=%d score=%.7f time=%.1fs"
        % (arr.size, final_score, time.monotonic() - t0)
    )

    return [float(x) for x in arr.tolist()]
```
