def fast_conv(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad(a, conv, top_k=40, temperature=0.0005):
    """Compute subgradient of max(conv) via low-temperature softmax over active peaks."""
    n = len(a)
    conv_max = conv.max()
    # Low-temperature softmax for near-exact subgradient
    log_w = (conv - conv_max) / temperature
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # Keep only top-k for efficiency
    if top_k < len(conv):
        top_idx = np.argsort(w)[::-1][:top_k]
        w_new = np.zeros_like(conv)
        w_new[top_idx] = w[top_idx]
        ws = w_new.sum()
        if ws > 0:
            w_new /= ws
        w = w_new

    active = np.where(w > 1e-12)[0]

            # Adapt parameters
            top_k = min(top_k + 5, 100)
            temperature = max(temperature * 0.8, 0.0002)
            c0 = 4.0 * (1.0 + 0.05 * min(restart_count, 10))
            restart_every = max(3000, restart_every + 500)
            if verbose:
                print(f"    MD restart {restart_count} t={t}: score={best_score:.6f}, tk={top_k}, temp={temperature:.5f}")

    return best_x, best_score


def coord_descent(a0, time_limit, seed=42, verbose=False):
    """Targeted coordinate descent: transfer from high-grad to low-grad coordinates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        g = compute_subgrad(a, conv, top_k=50, temperature=0.0003)

        # Find source (highest gradient) and destination (lowest gradient)
        # Only consider coordinates with significant mass
        active_coords = np.where(a > 1e-6)[0]
        if len(active_coords) < 2:
            stall += 1
            continue

        # Source: coordinates that contribute most to peaks
        src_candidates = active_coords[np.argsort(g[active_coords])[::-1][:max(20, n // 50)]]
        # Destination: coordinates that contribute least
        dst_candidates = active_coords[np.argsort(g[active_coords])[:max(20, n // 50)]]

        improved = False
        for src in src_candidates[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-10:
                continue
            for dst in dst_candidates[:15]:
                dst_i = int(dst)
                if dst_i == src_i:
                    continue
                # Try several transfer fractions
                for frac in [0.03, 0.07, 0.15, 0.25, 0.4, 0.6]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_score = 2.0 * n * best_m
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if a[np.argmax(g)] * 1e-4 < a.sum() * 1e-4:
                i = int(np.argmax(g))
                j = int(np.argmin(g))
                if i != j:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] = max(0, a[i] - amt)
                    a[j] += amt
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        best_score = 2.0 * n * best_m
                        stall = 0

        if verbose and stall % 40 == 0 and stall > 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): score={best_score:.6f}, stall={stall}")

    return best_a, best_score


def golden_section_coord(a, src_i, dst_i, n_iter=12):
    """Golden section search for optimal transfer from src to dst."""
    best_amt = 0.0
    best_m = fast_conv(a).max()
    lo = 0.0
    hi = a[src_i] * 0.95
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for _ in range(n_iter):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        a1 = a.copy()
        a1[src_i] -= m1
        a1[dst_i] += m1
        a1 = np.maximum(a1, 0.0)
        a1 = a1 / a1.sum()
        M1 = fast_conv(a1).max()

        a2 = a.copy()
        a2[src_i] -= m2
        a2[dst_i] += m2
        a2 = np.maximum(a2, 0.0)
        a2 = a2 / a2.sum()
        M2 = fast_conv(a2).max()

        if M1 < M2:
            hi = m2
            if M1 < best_m:
                best_m = M1
                best_amt = m1
        else:
            lo = m1
            if M2 < best_m:
                best_m = M2
                best_amt = m2

    return best_amt, best_m
