def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_from_conv(a, conv):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def compute_adaptive_subgrad(a, conv, top_k=50, adaptive_temp=True):
    """Compute subgradient with adaptive temperature based on peak sharpness."""
    n = len(a)
    conv_max = conv.max()
    
    if adaptive_temp:
        # Adaptive temperature: look at the spread of top values
        top_vals = np.sort(conv)[::-1][:min(50, len(conv))]
        spread = conv_max - top_vals[min(10, len(top_vals)-1)]
        if spread < 1e-15:
            temperature = conv_max * 0.01
        else:
            temperature = max(spread * 0.5, conv_max * 0.001)
    else:
        temperature = conv_max * 0.003
    
    threshold = conv_max - 3.0 * temperature
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):

    s = a.sum()

def compute_subgrad_v2(a, conv, top_k=100, adaptive_temp=True):
    """Subgradient with adaptive temperature and larger active set."""
    n = len(a)
    conv_max = conv.max()
    
    if adaptive_temp:
        top_vals = np.sort(conv)[::-1][:min(100, len(conv))]
        if len(top_vals) > 20:
            spread = conv_max - top_vals[20]
        else:
            spread = conv_max - top_vals[-1]
        if spread < 1e-18:
            temperature = max(conv_max * 0.005, 1e-20)
        else:
            temperature = max(spread * 0.4, conv_max * 0.001)
    else:
        temperature = conv_max * 0.002
    
    threshold = conv_max - 2.5 * temperature
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

        return np.ones(n) / n
    return a / s


def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Default init: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    print(f"Starting: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    # === Phase 1: Fine n-sweep (250s) ===
    print(f"\n=== Phase 1: n-sweep ===")
    t1_end = t_global + min(250.0, time_budget * 0.14)
    n_parent = len(_best_tracker['seq'])
    
    # Try n values around the parent
    n_candidates = sorted(set([
        n_parent - 300, n_parent - 200, n_parent - 100,
        n_parent + 100, n_parent + 200, n_parent + 300,
        n_parent + 500, n_parent - 500,
    ]))
    n_candidates = [n for n in n_candidates if 2000 <= n <= 8000]
    
    t_per_n = max(20.0, (t1_end - time.monotonic()) / max(1, len(n_candidates)))
    
    for new_n in n_candidates:
        remaining = t1_end - time.monotonic()
        if remaining < 15:
            break
        t_avail = min(t_per_n, remaining - 5)
        if t_avail < 10:
            break
        print(f"  n={new_n}, t={t_avail:.0f}s")
        a_interp = interpolate_seq(_best_tracker['seq'], new_n)
        try_update(a_interp, f"intp{new_n}")
        a_out, _ = amd_optimize_v2(a_interp, t_avail, top_k_init=60,
                                   c0_init=2.5, restart_every=3000,
                                   seed=1000 + new_n % 100, use_adaptive_temp=True)
        try_update(a_out, f"amd_n{new_n}")

    # === Phase 2: Main AMD optimization (700s) ===
    print(f"\n=== Phase 2: Main AMD (best n) ===")
    n_best = len(_best_tracker['seq'])
    t2_end = t_global + min(950.0, time_budget * 0.54)
    t2_avail = min(t2_end - time.monotonic(), 700.0)
    if t2_avail > 30:
        print(f"  n={n_best}, t={t2_avail:.0f}s")
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = amd_optimize_v2(a_in, t2_avail, top_k_init=80,
                                   c0_init=2.0, restart_every=5000,
                                   seed=42, use_adaptive_temp=True,
                                   verbose=True)
        try_update(a_out, "amd_main")

    # === Phase 3: Gradient flow polish (300s) ===
    print(f"\n=== Phase 3: Gradient Flow ===")
    t3_end = t_global + min(1250.0, time_budget * 0.70)
    t3_avail = min(t3_end - time.monotonic(), 300.0)
    if t3_avail > 20:
        print(f"  t={t3_avail:.0f}s")
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = gradient_flow_polish(a_in, t3_avail, seed=777, verbose=True)
        try_update(a_out, "gradflow")

    # === Phase 4: Coordinate descent (200s) ===
    print(f"\n=== Phase 4: Coord Descent ===")
    t4_end = t_global + min(1450.0, time_budget * 0.82)
    t4_avail = min(t4_end - time.monotonic(), 200.0)
    if t4_avail > 15:
        print(f"  t={t4_avail:.0f}s")
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = coord_descent_v2(a_in, t4_avail, seed=31415, verbose=True)
        try_update(a_out, "cd_main")

    # === Phase 5: Final multi-n check + polish (remaining) ===
    print(f"\n=== Phase 5: Final ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"  Best so far: n={n_best}, score={_best_tracker['score']:.6f}, remaining={remaining:.0f}s")
    
    if remaining > 60:
        # Quick check of adjacent n values
        for delta in [200, -200, 400, -400]:
            new_n = n_best + delta
            if 2000 <= new_n <= 8000 and remaining > 30:
                a_interp = interpolate_seq(_best_tracker['seq'], new_n)
                try_update(a_interp, f"fin_intp{new_n}")
                a_out, _ = amd_optimize_v2(a_interp, min(40.0, remaining - 10),
                                           top_k_init=80, c0_init=2.0,
                                           restart_every=4000, seed=999)
                try_update(a_out, f"fin_amd{new_n}")
                remaining = time_budget - (time.monotonic() - t_global)
    
    if remaining > 30:
        # Final AMD polish
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = amd_optimize_v2(a_in, min(remaining * 0.5, 100.0),
                                   top_k_init=100, c0_init=1.5,
                                   restart_every=8000, seed=2024)
        try_update(a_out, "amd_final")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        # Final CD polish
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = coord_descent_v2(a_in, min(remaining * 0.6, 80.0), seed=27182)
        try_update(a_out, "cd_final")

    # Final
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
