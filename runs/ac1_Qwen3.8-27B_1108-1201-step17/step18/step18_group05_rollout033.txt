```python
"""
Search for low self-convolution probability sequences.

The objective is scale invariant, so all internally maintained sequences are
normalized to sum 1.  For a normalized sequence a of length n we minimize

    J(a) = 2*n*max_k conv(a,a)[k].

The optimizer combines:
  * multigrid global exploration at small lengths,
  * smooth log-sum-exp projected/multiplicative gradient steps,
  * a linear-cutting-plane Frank-Wolfe style step solved with HiGHS,
  * short final polishing,
  * pool-based final evaluation through the provided evaluate_sequence.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    def linprog(*args, **kwargs):
        raise RuntimeError("linprog unavailable")

try:
    import scipy.fft as _scipy_fft
except Exception:
    _scipy_fft = None

from operator import itemgetter

_fft_cache = {}


def next_fft_len(m):
    if m <= 1:
        return 1
    if m in _fft_cache:
        return _fft_cache[m]
    if _scipy_fft is not None:
        try:
            size = int(_scipy_fft.next_fast_len(m))
            _fft_cache[m] = size
            return size
        except Exception:
            pass
    size = 1
    while size < m:
        size <<= 1
    _fft_cache[m] = size
    return size


def conv_full_with_A(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64), np.zeros(1, dtype=np.complex128), 1
    need = 2 * n - 1
    N = next_fft_len(need)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:need]
    c = np.maximum(c, 0.0)
    return c, A, N


def fast_conv(a):
    c, _, _ = conv_full_with_A(a)
    return c


def hard_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return float("inf")
    c, _, _ = conv_full_with_A(a)
    if c.size == 0:
        return float("inf")
    mx = float(c.max())
    if not np.isfinite(mx):
        return float("inf")
    return float(2.0 * n * mx / (s * s))


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0:
        return np.ones(0, dtype=np.float64)
    if not np.isfinite(s) or s < 1e-12:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if new_n <= 0:
        return np.ones(0, dtype=np.float64)
    if n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if n == new_n:
        return norm_sum1(a)
    a = norm_sum1(a)
    if n == 1:
        return np.full(new_n, float(a[0]) / float(new_n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    sb = float(b.sum())
    if not np.isfinite(sb) or sb < 1e-12:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return b / sb


def generate_variety(n, seed=0):
    out = []
    if n <= 0:
        return out
    x = np.linspace(0.0, 1.0, n)

    out.append(np.ones(n, dtype=np.float64) / float(n))

    poly_terms = [
        (0.0, 0.0),
        (1.0, 0.0),
        (-1.0, 0.0),
        (2.0, 0.0),
        (-2.0, 0.0),
        (0.0, 1.5),
        (0.0, -1.5),
        (1.0, 1.0),
        (-1.0, 1.0),
    ]
    for aa, bb in poly_terms:
        val = 1.0 + aa * x + bb * x * x
        val = np.maximum(val, 1e-6)
        out.append(norm_sum1(val))

    peak_sets = [
        (0.20, 0.80, 0.15, 0.15),
        (0.10, 0.90, 0.12, 0.12),
        (0.25, 0.75, 0.20, 0.20),
        (0.15, 0.55, 0.18, 0.25),
        (0.45, 0.85, 0.22, 0.18),
        (0.35, 0.70, 0.35, 0.20),
    ]
    for c1, c2, w1, w2 in peak_sets:
        val = np.exp(-((x - c1) / w1) ** 2) + np.exp(-((x - c2) / w2) ** 2) + 1e-4
        out.append(norm_sum1(val))

    val = np.sin(np.pi * x) ** 2
    out.append(norm_sum1(val))
    val = (0.5 + 0.5 * np.cos(2.0 * np.pi * x)) ** 2
    out.append(norm_sum1(val))
    val = (0.5 + 0.5 * np.sin(np.pi * x)) ** 3
    out.append(norm_sum1(val))
    val = 1.0 + 0.75 * np.cos(np.pi * x)
    val = np.maximum(val, 1e-5)
    out.append(norm_sum1(val))
    val = 1.0 - 0.75 * np.cos(np.pi * x)
    val = np.maximum(val, 1e-5)
    out.append(norm_sum1(val))

    rng = np.random.default_rng(seed)
    for _ in range(8):
        val = np.ones(n, dtype=np.float64) * 0.6
        modes = max(1, min(18, n // 4))
        for f in range(1, modes + 1):
            amp = rng.normal(0.0, 0.22 / float(np.sqrt(f)))
            phase = rng.uniform(0.0, 2.0 * np.pi)
            val += amp * np.sin(2.0 * np.pi * float(f) * x + phase)
        val = np.maximum(val, 1e-6)
        out.append.norm_sum1(val) if False else out.append(norm_sum1(val))

    for _ in range(8):
        centers = rng.uniform(0.08, 0.92, 2)
        widths = rng.uniform(0.06, 0.30, 2)
        amps = rng.uniform(0.4, 2.0, 2)
        val = (
            amps[0] * np.exp(-((x - centers[0]) / widths[0]) ** 2)
            + amps[1] * np.exp(-((x - centers[1]) / widths[1]) ** 2)
            + 1e-4
        )
        out.append(norm_sum1(val))

    for slope in (-1.8, -1.2, -0.7, 0.7, 1.2, 1.8):
        val = 1.0 + slope * (x - 0.5)
        val = np.maximum(val, 0.02)
        out.append(norm_sum1(val))

    return out


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(idx[-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / s


def soft_weights(c, T, n):
    v = 2.0 * float(n) * np.asarray(c, dtype=np.float64)
    if v.size == 0:
        return np.zeros(0, dtype=np.float64)
    v = v - float(v.max())
    if T <= 1e-14:
        p = np.zeros_like(v)
        mx = float(v.max())
        p[np.isclose(v, mx, rtol=0.0, atol=1e-12)] = 1.0
        ps = float(p.sum())
        if ps <= 0.0:
            p[int(np.argmax(v))] = 1.0
            ps = 1.0
        return p / ps
    p = np.exp(v / float(T))
    ps = float(p.sum())
    if not np.isfinite(ps) or ps < 1e-300:
        p = np.zeros_like(v)
        p[int(np.argmax(v))] = 1.0
        return p
    return p / ps


def corr_with_A(p, N, A, n):
    P = np.fft.rfft(np.asarray(p, dtype=np.float64), N)
    h = np.fft.irfft(P * np.conjugate(A), N)
    return h[:n]


def gradient_direction(a, c, A, N, T, n):
    p = soft_weights(c, T, n)
    h = corr_with_A(p, N, A, n)
    g = 4.0 * float(n) * h
    g = g - float(g.mean())
    gmax = float(np.abs(g).max())
    if not np.isfinite(gmax) or gmax < 1e-15:
        return None
    return g / gmax


def mirror_optimize(a0, time_limit, T_list=None, eta_init=1.0, verbose=False, rng_seed=0):
    a = norm_sum1(a0)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a, hard_score(a)

    if T_list is None:
        T_list = (1.0, 0.4, 0.15, 0.05, 0.02)
    T_list = tuple(T_list)
    m = len(T_list)

    t0 = time.monotonic()
    total = float(time_limit)
    best = a.copy()
    best_s = hard_score(a)
    cur_s = best_s
    z = np.log(np.maximum(a, 1e-300))
    z = z - float(z.mean())

    rng = np.random.default_rng(rng_seed)

    for ph, T in enumerate(T_list):
        phase_budget = total / float(m)
        phase_start = time.monotonic()
        eta = float(eta_init)
        stall = 0
        it = 0

        while time.monotonic() < phase_start + phase_budget and time.monotonic() < t0 + total:
            c, A, N = conv_full_with_A(a)
            cur_s = float(2.0 * n * c.max())
            gd = gradient_direction(a, c, A, N, float(T), n)
            if gd is None:
                break

            best_trial_s = cur_s
            best_trial_a = a.copy()
            best_trial_z = z.copy()
            best_step = eta

            for fac in (1.0, 0.5, 0.25, 1.5):
                st = eta * fac
                if st <= 0.0 or not np.isfinite(st):
                    continue
                zt = z - st * gd
                zt = zt - float(zt.mean())
                zt = np.clip(zt, -600.0, 600.0)
                e = np.exp(zt)
                se = float(e.sum())
                if not np.isfinite(se) or se < 1e-300:
                    continue
                at = e / se
                ct = fast_conv(at)
                sc = float(2.0 * n * ct.max())
                if np.isfinite(sc) and sc < best_trial_s - 1e-13:
                    best_trial_s = sc
                    best_trial_a = at.copy()
                    best_trial_z = zt.copy()
                    best_step = st

            if best_trial_s < cur_s - 1e-13:
                a = best_trial_a
                z = best_trial_z
                cur_s = best_trial_s
                eta = float(min(max(best_step * 1.03, 1e-4), 10.0))
                stall = 0
                if cur_s < best_s - 1e-12:
                    best = a.copy()
                    best_s = cur_s
                    if verbose:
                        print(f"    mirror t={time.monotonic()-t0:.1f}s score={best_s:.7f}")
            else:
                eta *= 0.7
                stall += 1
                if eta < 1e-4 or stall > 80:
                    a = best.copy()
                    cur_s = best_s
                    if stall > 40:
                        pert = rng.normal(0.0, 0.03, n) * a
                        ap = np.maximum(a + pert, 1e-12)
                        ap = norm_sum1(ap)
                        sp = hard_score(ap)
                        if np.isfinite(sp) and sp < cur_s * 1.07:
                            a = ap
                            cur_s = sp
                    z = np.log(np.maximum(a, 1e-300))
                    z = z - float(z.mean())
                    eta = float(eta_init) * 0.8
                    stall = 0

            it += 1
            if verbose and it % 200 == 0:
                print(f"    mirror iter={it} T={T:.3f} cur={cur_s:.7f} best={best_s:.7f}")

    return best, best_s


def pgd_optimize(a0, time_limit, T_list=None, eta_init=0.15, verbose=False, rng_seed=0):
    a = norm_sum1(a0)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a, hard_score(a)

    if T_list is None:
        T_list = (1.0, 0.4, 0.15, 0.05)
    T_list = tuple(T_list)
    m = len(T_list)

    t0 = time.monotonic()
    total = float(time_limit)
    best = a.copy()
    best_s = hard_score(a)
    cur_s = best_s
    rng = np.random.default_rng(rng_seed)

    for ph, T in enumerate(T_list):
        phase_budget = total / float(m)
        phase_start = time.monotonic()
        eta = float(eta_init)
        stall = 0
        it = 0

        while time.monotonic() < phase_start + phase_budget and time.monotonic() < t0 + total:
            c, A, N = conv_full_with_A(a)
            old_s = float(2.0 * n * c.max())
            gd = gradient_direction(a, c, A, N, float(T), n)
            if gd is None:
                break

            best_trial_s = old_s
            best_trial_a = a.copy()
            best_step = eta

            for fac in (1.0, 0.5, 0.25, 2.0):
                st = eta * fac
                if st <= 0.0 or not np.isfinite(st):
                    continue
                at = project_simplex(a - st * gd)
                at = np.maximum(at, 0.0)
                sa = float(at.sum())
                if not np.isfinite(sa) or sa < 1e-12:
                    continue
                at = at / sa
                ct = fast_conv(at)
                sc = float(2.0 * n * ct.max())
                if np.isfinite(sc) and sc < best_trial_s - 1e-13:
                    best_trial_s = sc
                    best_trial_a = at.copy()
                    best_step = st

            if best_trial_s < old_s - 1e-13:
                a = best_trial_a
                cur_s = best_trial_s
                eta = float(min(max(best_step * 1.03, 1e-5), 3.0))
                stall = 0
                if cur_s < best_s - 1e-12:
                    best = a.copy()
                    best_s = cur_s
                    if verbose:
                        print(f"    pgd t={time.monotonic()-t0:.1f}s score={best_s:.7f}")
            else:
                eta *= 0.7
                stall += 1
                if eta < 1e-5 or stall > 100:
                    a = best.copy()
                    cur_s = best_s
                    if stall > 40:
                        pert = rng.normal(0.0, 0.025, n) * a
                        ap = np.maximum(a + pert, 1e-12)
                        ap = norm_sum1(ap)
                        sp = hard_score(ap)
                        if np.isfinite(sp) and sp < cur_s * 1.08:
                            a = ap
                            cur_s = sp
                    eta = float(eta_init) * 0.8
                    stall = 0

            it += 1
            if verbose and it % 200 == 0:
                print(f"    pgd iter={it} T={T:.3f} cur={cur_s:.7f} best={best_s:.7f}")

    return best, best_s


def heuristic_direction(a, order):
    n = len(a)
    if n == 0 or len(order) == 0:
        return None
    D = np.zeros(n, dtype=np.float64)
    for k in order:
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        if s > e:
            continue
        j_min = k - e
        j_max = k - s
        D[s:e + 1] += a[j_min:j_max + 1][::-1]
    D /= float(len(order))
    b = 1.0 / (D + float(D.mean()) + 1e-12)
    sb = float(b.sum())
    if sb <= 0.0:
        return None
    return b / sb


def lp_direction(a, order):
    n = len(a)
    K = len(order)
    if n == 0 or K == 0:
        return None

    F = np.zeros((K, n), dtype=np.float64)
    for r, k in enumerate(order):
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        if s <= e:
            j_min = k - e
            j_max = k - s
            F[r, s:e + 1] = a[j_min:j_max + 1][::-1]

    if linprog is None:
        return heuristic_direction(a, order)

    m = n + 1
    c_obj = np.zeros(m, dtype=np.float64)
    c_obj[-1] = 1.0

    A_ub = np.zeros((K, m), dtype=np.float64)
    A_ub[:, :n] = F
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(K, dtype=np.float64)

    A_eq = np.zeros((1, m), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    res = None
    tried = []
    for method in ("highs", "highs-ds", None):
        try:
            if method is None:
                res = linprog(
                    c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method="auto"
                )
            else:
                res = linprog(
                    c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method=method
                )
            if res is not None and res.x is not None and res.x.size == m:
                break
        except Exception:
            res = None
        tried.append(method)

    if res is None or getattr(res, "x", None) is None:
        return heuristic_direction(a, order)

    if getattr(res, "status", -1) not in (0, 1, 2, 3):
        x = res.x
    else:
        x = res.x

    b = np.asarray(x[:n], dtype=np.float64)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    sb = float(b.sum())
    if sb <= 1e-12:
        return heuristic_direction(a, order)
    return b / sb


def lp_optimize(a0, time_limit, K_init=None, K_max=None, verbose=True):
    a = norm_sum1(a0)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a, hard_score(a)

    if K_init is None:
        K_init = 200 if n < 2000 else 320
    max_entries = 6_000_000
    if K_max is None:
        K_max = int(max(K_init, min(2 * n - 1, max_entries // max(1, n + 1))))
    K = int(min(K_init, 2 * n - 1))
    K_max = int(max(K, min(2 * n - 1, K_max)))

    t0 = time.monotonic()
    total = float(time_limit)
    best = a.copy()
    best_s = hard_score(a)
    cur_s = best_s
    rng = np.random.default_rng(1009 + n)

    stall = 0
    it = 0

    while time.monotonic() < t0 + total:
        rem = t0 + total - time.monotonic()
        if rem < 8.0:
            break

        c, _, _ = conv_full_with_A(a)
        cur_s = float(2.0 * n * c.max())
        K = int(min(K, 2 * n - 1))
        order = np.argsort(c)[::-1][:K]
        b = lp_direction(a, order)

        if b is None:
            K = int(min(K + 100, K_max))
            stall += 1
            if K >= K_max and stall > 12:
                break
            continue

        old_s = cur_s
        best_trial_s = old_s
        best_trial_a = a.copy()

        alphas = (0.05, 0.10, 0.20, 0.35, 0.55, 1.00)
        if rem < 30.0:
            alphas = alphas[:4]

        for alpha in alphas:
            trial = (1.0 - alpha) * a + alpha * b
            trial = norm_sum1(trial)
            sc = hard_score(trial)
            if np.isfinite(sc) and sc < best_trial_s - 1e-13:
                best_trial_s = sc
                best_trial_a = trial.copy()

        if best_trial_s < old_s - 1e-13:
            a = best_trial_a
            cur_s = best_trial_s
            stall = 0
            it += 1
            if cur_s < best_s - 1e-12:
                best = a.copy()
                best_s = cur_s
                if verbose:
                    print(f"    LP iter={it} t={time.monotonic()-t0:.1f}s score={best_s:.7f} K={K}")
        else:
            stall += 1
            if stall % 3 == 0:
                K = int(min(K + 100, K_max))
            if stall > 18:
                a = best.copy()
                cur_s = best_s
                moved = False
                n_pert = max(1, n // 800)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] <= 1e-12:
                        continue
                    amt = 0.01 * a[i]
                    ap = a.copy()
                    ap[i] -= amt
                    ap[j] += amt
                    ap = norm_sum1(np.maximum(ap, 0.0))
                    sp = hard_score(ap)
                    if np.isfinite(sp) and sp < cur_s:
                        a = ap
                        cur_s = sp
                        if sp < best_s:
                            best = ap.copy()
                            best_s = sp
                        moved = True
                        break
                stall = 0
                if not moved:
                    K = int(min(K + 100, K_max))

        if it % 4 == 0:
            K = int(min(K + 60, K_max))

        if K >= K_max and stall > 45:
            break

    return best, best_s


def multigrid_small_search(current_best, time_limit):
    t0 = time.monotonic()
    total = float(time_limit)
    patterns = []
    ns = (64, 128, 256, 512)
    weights = (0.15, 0.25, 0.30, 0.30)
    seed_base = 911

    for idx, n in enumerate(ns):
        elapsed = time.monotonic() - t0
        if elapsed >= total:
            break
        phase_budget = total * weights[idx]
        if time.monotonic() - t0 + phase_budget > total:
            phase_budget = total - elapsed
        if phase_budget < 4.0:
            break

        inits = generate_variety(n, seed_base + 17 * idx)
        if current_best is not None:
            inits.insert(0, interpolate_sequence(current_best, n))

        scored = []
        for arr in inits:
            aa = norm_sum1(arr)
            if aa.size != n:
                continue
            sc = hard_score(aa)
            if np.isfinite(sc):
                scored.append((sc, aa))
        scored.sort(key=itemgetter(0))
        top = scored[:6]
        if not top:
            continue

        t_per = phase_budget / float(len(top))
        for j, (_, a_init) in enumerate(top):
            if time.monotonic() - t0 >= total:
                break
            avail = min(t_per, total - (time.monotonic() - t0))
            if avail < 1.5:
                break

            t_pg = min(avail * 0.55, 40.0)
            a_pg, s_pg = pgd_optimize(
                a_init, t_pg,
                T_list=(1.0, 0.4, 0.15),
                eta_init=0.25,
                verbose=False,
                rng_seed=seed_base + j
            )
            if a_pg is not None and np.isfinite(s_pg):
                patterns.append((s_pg, a_pg))

            rem = total - (time.monotonic() - t0)
            if rem > 1.0:
                t_mi = min(avail * 0.45, rem, 30.0)
                a_mi, s_mi = mirror_optimize(
                    a_pg if a_pg is not None else a_init,
                    t_mi,
                    T_list=(0.4, 0.15, 0.05),
                    eta_init=1.0,
                    verbose=False,
                    rng_seed=seed_base + j + 1000
                )
                if a_mi is not None and np.isfinite(s_mi):
                    patterns.append((s_mi, a_mi))

    patterns.sort(key=itemgetter(0))
    return [a for _, a in patterns[:12]]


def time_remaining(t_start, budget):
    r = float(budget) - (time.monotonic() - t_start)
    return r if r > 0.0 else 0.0


def consider_candidate(pool, best_seq, best_score, a, label, score_hint=None):
    if a is None:
        return pool, best_seq, best_score
    arr = np.asarray(a, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 1e-12:
        return pool, best_seq, best_score
    arr = arr / s

    sc = score_hint
    if sc is None or not np.isfinite(sc) or sc <= 0.0:
        sc = hard_score(arr)
    if not np.isfinite(sc):
        return pool, best_seq, best_score
    sc = float(sc)

    if best_seq is None or sc < best_score - 1e-12:
        best_seq = arr.copy()
        best_score = sc
        print(f"  new internal best [{label}] = {best_score:.7f}, n={len(arr)}")

    if pool is None:
        pool = []
    if len(pool) == 0:
        pool.append((sc, arr.copy()))
    else:
        worst_idx = 0
        worst_val = float("inf")
        for i, entry in enumerate(pool):
            if entry[0] < worst_val:
                worst_val = float(entry[0])
                worst_idx = i
        if sc <= worst_val + 0.012:
            pool.append((sc, arr.copy()))
            if len(pool) > 8:
                del pool[worst_idx]
    return pool, best_seq, best_score


def safe_evaluate(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        arr = np.minimum(arr, 1000.0)
        if arr.size == 0:
            return float("inf")
        if float(arr.sum()) < 0.01:
            return float("inf")
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        if np.isfinite(val):
            return float(val)
        return float("inf")
    except Exception:
        return float("inf")
```
