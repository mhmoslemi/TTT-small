    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    if s < 1e-15:
        return np.ones_like(a) / len(a)

def smoothed_obj_and_grad(a, tau):
    """Compute smoothed objective and gradient.
    
    Returns (obj, grad) where obj = 2n * tau * log(sum(exp(c/tau)))
    (which approximates 2n * max(c) for small tau)

    # Log-sum-exp
    c_scaled = c / tau

    conv_p_Ra = np.fft.irfft(P * R, fft_size)
    grad = (2.0 / tau) * conv_p_Ra[n-1:n-1+n]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Proper smoothed gradient (log-sum-exp) with adaptive step size
2. Mirror descent (multiplicative) with geometric annealing
3. Spectral pre-conditioning and re-weighting
4. Multi-length, multi-initialization search
5. Simulated annealing restarts to escape local minima
6. Coordinate descent polish
"""

import numpy as np
import time


def fast_conv(a):
    """FFT-based linear convolution a*a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Project to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, x)) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def smoothed_obj_grad(a, tau):
    """Compute smoothed objective and gradient via log-sum-exp.
    
    obj = 2n * tau * LSE(c/tau) ~ 2n * max(c) for small tau
    grad[i] = (2/tau) * sum_m p[i+m] * a[m]  (computed via FFT)
    """
    n = len(a)
    c = fast_conv(a)
    c_scaled = c / tau
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    p = exp_c / exp_c.sum()
    log_sum_exp = c_max + np.log(exp_c.sum())
    obj = 2.0 * n * tau * log_sum_exp
    # Gradient via cross-correlation: grad[i] = (2/tau) * sum_m p[i+m] * a[m]
    fft_size = 1 << (2 * n - 1).bit_length()
    A_ft = np.fft.rfft(a, fft_size)
    P_ft = np.fft.rfft(p, fft_size)
    Ra = a[::-1]
    R_ft = np.fft.rfft(Ra, fft_size)
    conv_p_Ra = np.fft.irfft(P_ft * R_ft, fft_size)
    grad = (2.0 / tau) * conv_p_Ra[n - 1:n - 1 + n]
    return obj, grad, p, c


def mirror_step(a, grad, eta):
    """Multiplicative (mirror) descent step maintaining a >= 0, sum(a)=1."""
    grad_clip = np.clip(grad, -50.0, 50.0)
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad_clip
    log_a = log_a - log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_flatten(a, alpha=0.3):
    """Compress dynamic range of |A(w)| to flatten power spectrum."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / mag_mean
    mag_new = mag_mean * np.power(ratio, 1.0 - alpha)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return a_new


def optimize_core(a0, time_limit, tau_init_factor=1.0, tau_min_factor=0.003,
                  eta_init=0.5, verbose=False):
    """Core smoothed gradient descent optimizer with adaptive step and annealing."""
    n = len(a0)
    a = norm_simplex(a0.copy())
    c0 = fast_conv(a)
    M0 = c0.max()
    best_a = a.copy()
    best_M = M0
    best_obj = 2.0 * n * M0

    tau_init = M0 * tau_init_factor
    tau_min = M0 * tau_min_factor
    if tau_init < 1e-30:
        tau_init = 0.01
        tau_min = 1e-5

    t0 = time.monotonic()
    step = 0
    eta = eta_init
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Annealing: geometric cooling with progress
        progress = min(1.0, step / 30000.0)
        tau = tau_init * (tau_min / tau_init) ** (progress ** 0.7)

        # Compute gradient
        obj, grad, p, c = smoothed_obj_grad(a, tau)
        true_obj = 2.0 * n * c.max()

        # Adaptive step size: target max relative change ~ 0.05
        rel_grad = np.abs(grad) / np.maximum(a, 1e-15)
        max_rel = max_rel = rel_grad.max()
        if max_rel > 1e-30:
            eta_safe = 0.05 / max_rel
        else:
            eta_safe = eta
        eta_cur = min(eta, eta_safe)

        # Take step
        a_new = mirror_step(a, grad, eta_cur)
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        obj_new = 2.0 * n * M_new

        if obj_new < best_M - 1e-15:
            best_M = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(3.0, eta * 1.005)
        else:
            a = a_new
            no_improve += 1
            eta = max(0.05, eta * 0.9)

            # Simulated annealing: accept worse moves occasionally early on
            if progress < 0.3 and True:
                pass  # we already accepted a_new above

            # Reheat if stuck
            if no_improve > 3000:
                noise_scale = 0.02 * (1.0 - progress)
                a = norm_simplex(a + rng.normal(0, noise_scale, n))
                c_reheat = fast_conv(a)
                if c_reheat.max() < best_M:
                    best_M = c_reheat.max()
                    best_a = a.copy()
                tau = tau_init * 0.5
                no_improve = 0
                eta = eta_init * 0.5

        # Periodic spectral re-weighting
        if step > 0 and step % 8000 == 1:
            a_spec = spectral_flatten(a, alpha=0.2)
            c_spec = fast_conv(a_spec)
            if c_spec.max() < best_M:
                best_M = c_spec.max()
                best_a = a_spec.copy()
                a = a_spec

        step += 1
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Step {step} ({elapsed:.0f}s): best_M={best_M:.8f}, "
                  f"score={best_M:.6f}, tau={tau:.2e}, eta={eta:.4f}")

    return best_a, best_M


def coord_descent_polish(a0, time_limit, verbose=False):
    """Local coordinate descent: try moving mass between pairs."""
    n = len(a0)
    a = norm_simplex(a0.copy())
    c = fast_conv(a)
    best_M = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M = c.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top peaks
        top_k = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:top_k]

        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Try several destinations
            for attempt in range(8):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Random escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = norm_simplex(a)
                M_new = fast_conv(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0
            stall += 1

    return best_a, best_M


def make_init_bowl(n, sharpness=2.0, floor=0.1):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + floor
    return norm_simplex(a)


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_simplex(a)


def make_init_flat(n):
    return np.ones(n) / n


def make_init_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s
