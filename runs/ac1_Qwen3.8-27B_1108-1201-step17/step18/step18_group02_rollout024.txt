    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:

def _fft_autocorr(a):
    """Compute linear self-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def _norm1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, float), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _score(a):
    """Compute J = 2n * max(a*a) / sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a_n = a / s
    c = _fft_autocorr(a_n)
    return 2.0 * n * c.max()


def _lse_smoothing_grad(a, c, tau):
    """Gradient of tau*log(sum(exp(c_k/tau))) w.r.t. a."""
    n = len(a)
    c_s = c / tau
    c_s = c_s - c_s.max()
    e = np.exp(c_s)
    p = e / e.sum()
    # grad[i] = 2 * sum_k p[k] * a[k-i] for valid indices
    # = 2 * (p * flip(a))[n-1 : 2n-2]... 
    # Actually: grad[i] = 2 * sum_k p[k] * a_padded[k-i]
    # Let's use FFT convolution
    size = len(c)  # 2n-1
    N = 1 << (size - 1).bit_length()
    # We need: for each i, grad[i] = 2 * sum_k p[k] * a[k-i] (with a out of bounds = 0)
    # This is 2 * (p convolve flip(a))[n-1 + i] ... hmm
    # Actually: sum_k p[k] * a[k-i] = (p * a_flipped)[i + (n-1)] where a_flipped = flip(a)
    # Let's verify: (p * a_f)[m] = sum_k p[k] * a_f[m-k] = sum_k p[k] * a[n-1-(m-k)] = sum_k p[k] * a[n-1-m+k]
    # We want sum_k p[k] * a[k-i]. Set n-1-m+k = k-i => n-1-m = -i => m = n-1+i.
    # So grad[i] = 2 * (p * a_f)[n-1+i]
    a_f = a[::-1]
    N2 = 1 << (len(p) + len(a_f) - 2).bit_length()
    Pf = np.fft.rfft(p, N2)
    Af = np.fft.rfft(a_f, N2)
    conv = np.fft.irfft(Pf * Af, N2)
    grad = 2.0 * conv[n-1 : n-1+n]
    return np.maximum(grad, 0)  # gradient should be non-negative for non-negative a


def lse_adam(a0, time_limit, tau_decay=0.99985, lr=0.8, beta1=0.9, beta2=0.999,
             restart_every=2000, seed=42, verbose=False):
    """LSE-smoothed multiplicative Adam."""
    n = len(a0)
    a = _norm1(a0)
    c = _fft_autocorr(a)
    best_a = a.copy()
    best_j = 2.0 * n * c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    tau = max(c.max() * 0.3, 1e-6)
    tau_min = max(c.max() * 0.002, 1e-12)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = _fft_autocorr(a)
        M = c.max()
        J = 2.0 * n * M
        
        if J < best_j - 1e-14:
            best_j = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        
        g = _lse_smoothing_grad(a, c, tau)
        
        m_hat = beta1 * m_hat + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * (g * g)
        
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        # Multiplicative update in log-space
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        if stall > restart_every:
            stall = 0
            # Aggressive restart
            noise = rng.normal(0, 0.05, n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                print(f"    LSE-Adam restart ({time.monotonic()-t0:.0f}s): J={best_j:.6f}")
        
        if verbose and t % 1000 == 0:
            print(f"    LSE-Adam t={t} ({time.monotonic()-t0:.0f}s): J={best_j:.6f}, tau={tau:.2e}")
    
    return best_a, best_j

            best_j = J
            best_a = a.copy()
        
        # Find top peaks
        k_star = int(c.argmax())
        
        # Gradient of C[k*] w.r.t. a: g[j] = 2 * a_padded[k*-j]
        # a_padded[m] = a[m] if 0<=m<n else 0
        g = np.zeros(n)
        # g[j] = 2 * a_padded[k_star - j]
        # For j in [0, n-1], k_star - j ranges from k_star to k_star - (n-1)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            g[lo : hi+1] = 2.0 * a[hi : lo-1 : -1]  # a[k_star - j] for j from lo to hi
        
        # Direction: d = mean(g) - g (negative gradient projected onto simplex tangent)
        d = g.mean() - g
        # Clip to maintain feasibility: d[i] < 0 means we decrease a[i]
        # Step size by quadratic formula
        # C[k*](a + t*d) = C[k*](a) + 2t*(a_padded * d_padded at k*) + t^2 * (d*d at k*)
        
        # Compute (d*d)[k*] = sum_i d[i]*d[k*-i]
        dd_k = 0.0
        lo2 = max(0, k_star - (n-1))
        hi2 = min(n-1, k_star)
        if lo2 <= hi2:
            dd_k = np.dot(d[lo2:hi2+1], d[hi2:lo2-1:-1])
        
        # Compute (a_padded * d_padded at k*) = sum_i a[i]*d[k*-i]
        ad_k = 0.0
        if lo <= hi:
            ad_k = np.dot(a[lo:hi+1], d[hi:lo-1:-1])
        
        if dd_k < 1e-30:
            break  # direction doesn't change C[k*], need different approach
        
        # Optimal unconstrained step
        t_uncon = -ad_k / dd_k
        
        # Feasibility: a[i] + t*d[i] >= 0 for all i
        t feas = inf
        neg_mask = d < 0
        if neg_mask.any():
            t_feas = np.min(-a[neg_mask] / d[neg_mask])
        else:
            t_feas = float('inf')
        
        t_opt = min(t_uncon, t_feas * 0.999)
        if t_opt <= 0:
            # Can't take a step in this direction, try a random perturbation
            break
        
        a = a + t_opt * d
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:

        a = a / s
    
    return best_a, best_j
