    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def spectral_preprocess(a, n_iters=50, power=1.5, verbose=False):
    """Apply spectral low-pass filtering to reduce autocorrelation peak."""
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / (mag + 1e-300)  # phase = A/|A|
        
        # Low-pass: reduce high-frequency components
        freqs = np.arange(len(mag))
        # Apply a cosine taper to the magnitude
        cutoff = len(mag) * 0.3  # keep low 30% of frequencies
        taper = np.ones_like(mag)
        # Smooth transition
        for i in range(len(mag)):
            if freqs[i] > cutoff:
                frac = (freqs[i] - cutoff) / max(1, len(mag) - cutoff)
                taper[i] = max(0.0, 1.0 - frac * power)
        
        mag_new = mag * taper
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:

        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        a = a_new
        
        if verbose and it % 10 == 0:
            print(f"  Spectral iter {it}: max_ac={m_new:.8f}, best={best_m:.8f}")

        if len(constraint_list) > max_K:
            # Remove constraints with largest margin
            margins = np.array([M_y - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail += 1
            if lp_fail > 5:
                constraint_set = set(int(constraint_list[min(initial_K, len(constraint_list))]))
                lp_fail = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail += 1
            if lp_fail > 10:
                constraint_set = set(int(constraint_list[initial_K // 2]))
                lp_fail = 0
            continue
        
        lp_fail = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Backtracking line search
        best_alpha = 2.0 / (t + 2.0)  # default (FW step size)
        best_obj = M_y
        
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            st = y_trial.sum()
            if st < 1e-15:
                continue
            y_trial = y_trial / st
            M_trial = fast_autocorr(y_trial).max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        st = y_new.sum()
        if st < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / st
        
        # Update averaged iterate (for tracking)
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        x = x_new
        y = y_new
        
        # Check x as well
        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
        
        # Update constraint set: add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_y + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_y
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        iter_count += 1
        
        # Restart if stalled
        if stall > 300:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            y = best_a + pert
            y = np.maximum(y, 1e-10)
            y = y / y.sum()
            x = y.copy()
            # Reset constraint set
            ac_new = fast_autocorr(y)
            new_init = np.argsort(ac_new)[::-1][:initial_K]
            constraint_set = set(int(k) for k in new_init)
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW-BT iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, alpha={best_alpha:.2f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    FW-BT done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m
