def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute 2*n*max_autocorr for a on the simplex (sum=1)."""

def compute_smoothed_objective(a, tau):
    """Compute smoothed objective."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    return 2.0 * n * tau * np.log(np.sum(np.exp((ac - ac_max) / tau)) + 1e-30) + 2.0 * n * ac_max


def compute_gradient_a(a, tau):
    """Compute gradient of smoothed objective w.r.t. a (on simplex).
    Returns grad such that the directional derivative in direction d (with sum(d)=0)
    is approx grad @ d.
    """
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    p = np.exp((ac - ac_max) / tau)
    p = p / p.sum()
    # grad[j] = 2 * sum_m a[m] * p[j+m]
    cross = sp_correlate(p, a, mode='valid')
    grad = 2.0 * cross
    return grad, p


def compute_gradient_x(x, tau):
    """Compute gradient in log-space x where a = softmax(x)."""
    a = softmax(x)
    grad_a, _ = compute_gradient_a(a, tau)
    # grad_x[j] = a[j] * (grad_a[j] - dot(a, grad_a))
    mean_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_grad)
    return grad_x, a

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_a(a, tau):
    """Correct gradient of smoothed objective 2*n*tau*logsumexp(ac/tau) w.r.t. a.
    Returns (2n) * 2 * cross where cross[j] = sum_m a[m]*p[j+m].
    """
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    p = np.exp((ac - ac_max) / tau)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(ac)) / len(ac)
    else:
        p = p / p_sum
    cross = sp_correlate(p, a, mode='valid')
    grad = 2.0 * cross
    return 2.0 * n * grad, p, ac


def compute_gradient_a_alt(a, tau):

    grad_tan = grad - grad.mean()
    # Backtracking
    best_alpha = eta

            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]

        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)

            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

        top_K_max = min(2 * n - 1, 5000)
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 100, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 50, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 50, top_K_max)
            continue
        g_dir = b_val / b_sum

        # Backtracking
        base_obj = 2.0 * n * M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_obj = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            obj_trial = 2.0 * n * fast_autocorr(a_trial).max()
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:

        m_new = fast_autocorr(a).max()
        obj_new = 2.0 * n * m_new
        if obj_new < 2.0 * n * best_m:
            best_m = m_new

            K = min(K + 30, top_K_max)

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): obj={2.0*n*best_m:.6f}, K={K_sel}")

    return best_a, best_m


def pair_move_polish(a0, time_limit, verbose=True, rng_seed=456):
    """Coordinate descent trying pair moves on top autocorrelation lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:15]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination
            best_dest = -1
            best_dest_obj = best_obj
            # Try a subset of low-value positions
            low_mask = a < np.median(a)
            candidates = np.where(low_mask)[0]
            if len(candidates) > 200:
                candidates = candidates[np.argsort(candidates, kind='stable')[:200]]
            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    obj_t = 2.0 * n * fast_autocorr(a_trial).max()
                    if obj_t < best_dest_obj - 1e-12:
                        best_dest_obj = obj_t
                        best_dest = dst
                        best_frac = frac

            if best_dest >= 0:
                amt = src_val * best_frac
                a[src] -= amt
                a[best_dest] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_obj = 2.0 * n * fast_autocorr(a).max()
                best_a = a.copy()
                stall = 0
                improved = True

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj_new = 2.0 * n * fast_autocorr(a).max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            print(f"    PairMove ({time.monotonic()-t0:.0f}s): obj={best_obj:.6f}, stall={stall}")

    return best_a


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / width) ** 2) + np.exp(-((x - 0.9) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_sin_sq(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    return norm_sum1(a)


def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_sin_sq(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-start with corrected gradient + AMD ===
    print(f"\n=== Phase 1: Multi-start (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)

    if t_phase1 > 60:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")
        inits.append(make_bimodal(n0, 0.2))
        labels.append("bimodal")
        inits.append(make_sin_sq(n0))
        labels.append("sinsq")
        inits.append(make_cosine(n0))
        labels.append("cosine")
        inits.append(spectral_init(n0, 30, 42))
        labels.append("spectral")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break

            # Alternate between AMD and corrected gradient
            if idx % 2 == 0:
                print(f"  [{lbl}] AMD ({t_avail:.0f}s)...")
                a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                         restart_every=4000, adaptive_beta=True,
                                         verbose=False, rng_seed=42 + idx)
            else:
                print(f"  [{lbl}] SmoothGrad ({t_avail:.0f}s)...")
                a_out = smoothed_grad_descend(init, t_avail, tau_init=10.0, tau_final=200.0,
                                               eta_init=0.3, verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"p1_{lbl}")

    # === Phase 2: L-BFGS on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: L-BFGS (n={n_best}) ===")
    t2 = min(remaining * 0.08, 120.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        for tau_lb in [30.0, 80.0]:
            remaining = time_budget - (time.monotonic() - t_global)
            t_avail = min(t2 / 2.0, remaining - 30)
            if t_avail < 20:
                break
            print(f"  L-BFGS tau={tau_lb} ({t_avail:.0f}s)...")
            a_out = lbfgs_optimize(a_in, t_avail, tau=tau_lb, maxiter=300, verbose=True)
            try_update(a_out, f"lbfgs_t{int(tau_lb)}")

    # === Phase 3: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp1")

    # === Phase 4: Corrected gradient descent (longer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: SmoothGrad long (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out = smoothed_grad_descend(a_in, t4, tau_init=5.0, tau_final=400.0,
                                       eta_init=0.5, verbose=True, rng_seed=200,
                                       use_lbfgs_every=200, lbfgs_tau=60.0)
        try_update(a_out, "smooth_long")

    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp2")

    # === Phase 6: L-BFGS round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: L-BFGS r2 (n={n_best}) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out = lbfgs_optimize(a_in, t6, tau=100.0, maxiter=300, verbose=True)
        try_update(a_out, "lbfgs_r2")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [500, 1000, -500, 1500, -1000, 2000]:
        new_n = n_best + delta
        if 1500 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]
    trial_lengths.sort(key=lambda x: abs(x - n_best))

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.4, 150.0)
        if t7a > 30:
            a_out, _ = amd_optimize(a_long, t7a, top_k=60, beta=6.0, c0=4.0,
                                     restart_every=5000, adaptive_beta=True,
                                     verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 5000:
            t7b = min(remaining * 0.3, 100.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                top_K = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=True,
                                           top_K_init=min(300, 2 * new_n - 1),
                                           top_K_max=top_K)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 30:
        # Pair move polish
        t8a = min(remaining * 0.3, 80.0)
        if t8a > 15:
            a_in = norm_sum1(best_seq)
            a_out = pair_move_polish(a_in, t8a, verbose=False, rng_seed=789)
            try_update(a_out, "pairmove")

        # Final smoothed gradient
        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.3, 60.0)
        if t8b > 15:
            a_in = norm_sum1(best_seq)
            a_out = smoothed_grad_descend(a_in, t8b, tau_init=20.0, tau_final=500.0,
                                           eta_init=0.2, verbose=False, rng_seed=500)
            try_update(a_out, "smooth_final")

        # Final L-BFGS
        remaining = time_budget - (time.monotonic() - t_global)
        t8c = min(remaining * 0.3, 50.0)
        if t8c > 15:
            a_in = norm_sum1(best_seq)
            a_out = lbfgs_optimize(a_in, t8c, tau=150.0, maxiter=200, verbose=True)
            try_update(a_out, "lbfgs_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_sin_sq(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
