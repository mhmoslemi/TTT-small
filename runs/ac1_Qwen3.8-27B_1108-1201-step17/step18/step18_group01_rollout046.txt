- Peak values in the interior that are balanced

This is similar to a "tapered" sequence that's zero at the boundaries and has some oscillatory structure in the interior.

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_max(vals, temperature):
    """Compute soft max and its 'weights'."""
    vals = np.asarray(vals)
    shifted = vals / temperature - np.max(vals) / temperature
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    soft_max_val = temperature * (np.log(exp_vals.sum()) + np.max(vals) / temperature)
    return soft_max_val, weights

def smoothed_objective_and_grad(a, temperature):
    """
    Compute the smoothed max of conv(a,a) and its gradient.
    f_tau(a) = tau * log(sum_k exp(conv(k)/tau))
    gradient = sum_k p_k * grad_k where p_k = exp(conv(k)/tau) / sum_j exp(conv(j)/tau)
    """
    ac = fast_autocorr(a)
    M = ac.max()
    shifted = (ac - M) / temperature
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    soft_max_val = M + temperature * np.log(exp_shifted.sum())
    
    # Gradient: sum_k p[k] * grad_k
    n = len(a)
    grad = np.zeros(n)
    # Only need to consider significant p[k]
    threshold = 1e-10
    significant = np.where(p > threshold)[0]
    for k in significant:
        grad += p[k] * compute_conv_gradient(a, int(k))
    
    return soft_max_val, grad

def projected_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Use the standard projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def proj_grad_descent(a0, time_limit, tau0, verbose=True):
    """
    Projected gradient descent with smoothed max objective.
    Anneals temperature tau from tau0 down to a small value.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    tau = tau0
    eta = 0.1  # step size
    iter_count = 0
    patience = 0
    max_patience = 500
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_val - 1e-15:
            best_val = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Restart with new tau
            tau *= 0.5
            if tau < 1e-4:
                break
            eta = 0.05
            patience = 0
            continue
        
        # Compute smoothed gradient
        tau_eff = max(tau, best_val * 0.01)  # don't let tau get too small relative to M
        _, grad = smoothed_objective_and_grad(a, tau_eff)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Projected gradient step
        a_new = projected_to_simplex(a - eta * gnorm * grad)
        # Hmm, I should use the actual gradient magnitude
        
        # Actually, let's use: a_new = project(a - eta * grad_unnormalized)
        _, grad_un = smoothed_objective_and_grad(a, tau_eff)
        a_new = projected_to_simplex(a - eta * grad_un)
        
        # Line search (backtracking)
        M_new = fast_autocorr(a_new).max()
        if M_new > M:
            # Backtrack
            eta *= 0.5
            if eta < 1e-6:
                patience += 50
                continue
            a_new = projected_to_simplex(a - eta * grad_un)
            M_new = fast_autocorr(a_new).max()

            a = a_new
            # Slowly decrease eta and tau
            tau *= 0.999
        else:
            eta *= 0.5

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_val
            print(f"    PGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, tau={tau_eff:.6f}, eta={eta:.6f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_val
        print(f"    PGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a

def mgd_optimize_v2(a0, time_limit, top_k, eta0, verbose):
    """
    Improved MGD with exact single-lag subgradient and adaptive step.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = eta0
    patience = 0
    max_patience = 400
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        # Use top-k lags with softmax weighting
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        # Softmax weights based on autocorr values
        top_vals = ac[top_idx]
        tau_w = (top_vals[0] - top_vals[-1]) / max(1, k_max * 0.5) + 1e-15
        w = np.exp((top_vals - top_vals[0]) / tau_w)
        w = w / w.sum()
        
        # Multi-lag gradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            grad += wi * compute_conv_gradient(a, int(k))
        
        # Center the gradient (for invariance to adding constants)
        grad = grad - grad.mean()
        
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()
        
        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.5
            if eta < 0.0003:
                patience = max(0, patience - 200)
                eta = 0.03
        
        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD2 iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD2 done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a

def local_search(a0, time_limit, verbose=False):
    """
    Local search: try small random perturbations and coordinate moves.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False
        
        for attempt in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Random two-coordinate move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            # Try moving mass from i to j or j to i
            for (src, dst) in [(i, j), (j, i)]:
                if a[src] < 1e-10:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved and attempt > 30:
                break
        
        if not improved:
            stall += 1
            # Random escape: perturb a random block
            block_size = int(rng.integers(5, 30))
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            a[start:end] += rng.normal(0, 0.01, end - start)
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        else:
            stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Local done ({elapsed:.0f}s): J={J:.6f}")
    return best_a

def geometric_init(n, r):
    """Create a geometric (exponentially decaying) sequence."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    return norm_sum1(a)

def ramp_init(n, alpha):
    """Create a ramp sequence: a[i] = (i+1)^alpha."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** alpha
    return norm_sum1(a)
