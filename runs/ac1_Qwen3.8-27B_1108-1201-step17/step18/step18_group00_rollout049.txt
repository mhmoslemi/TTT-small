1. Asymmetric parametric initialization (ramps, exponentials)
2. LP-based Frank-Wolfe with backtracking line search and away steps
3. Exact 1D coordinate descent polish
4. Multiplicative weight descent with Nesterov momentum
5. Multi-phase orchestration with time management
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  LP solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_objective(a, n):
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(elapsed) + "s)")
    return best_seq, best_score


def parametric_search(n, time_limit, best_init, best_score):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    # Asymmetric ramps - key insight: optimum is asymmetric
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        a = norm_sum1((n - i_arr) ** p)
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()
            print("    Param ramp" + str(p) + ": J=" + repr(v))
        a = norm_sum1((i_arr + 1.0) ** p)
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()
            print("    Param iramp" + str(p) + ": J=" + repr(v))

    # Exponential decays
    for rate in [0.5, 1, 2, 3, 5, 10]:
        a = norm_sum1(np.exp(-rate * x))
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()
            print("    Param exp" + str(rate) + ": J=" + repr(v))
        a = norm_sum1(np.exp(rate * (x - 1.0)))
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()
            print("    Param rexp" + str(rate) + ": J=" + repr(v))

    # Two-sided with asymmetry
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.3, 0.5, 0.7, 1.0]:
            a = np.exp(-alpha * x) + beta * np.exp(-alpha * (1 - x))
            a = norm_sum1(a)
            v = compute_objective(a, n)
            if v < best_val:
                best_val = v
                best_a = a.copy()

    # Power law combinations
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            a = norm_sum1(a)
            v = compute_objective(a, n)
            if v < best_val:
                best_val = v
                best_a = a.copy()

    # Ridge functions
    for c in [0.2, 0.3, 0.4, 0.6, 0.7]:
        for s in [0.1, 0.2, 0.3]:
            a = norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * s * s)))
            v = compute_objective(a, n)
            if v < best_val:
                best_val = v
                best_a = a.copy()

    # Blend with best_init
    for w in [0.1, 0.2, 0.3, 0.5, 0.7]:
        ramp = norm_sum1((n - i_arr) ** 1.0)
        a_blend = w * best_init / best_init.sum() + (1.0 - w) * ramp
        a_blend = norm_sum1(a_blend)
        v = compute_objective(a_blend, n)
        if v < best_val:
            best_val = v
            best_a = a_blend.copy()

    # Random search
    rng = np.random.default_rng(123)
    for trial in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        mode = trial % 4
        if mode == 0:
            a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0))
        elif mode == 1:
            a = norm_sum1(np.exp(-rng.uniform(0.5, 5.0) * x * rng.uniform(0.5, 2.0)))
        elif mode == 2:
            base = (n - i_arr) ** rng.uniform(0.5, 2.0)
            a = norm_sum1(base * (1.0 + 0.3 * np.sin(2.0 * np.pi * rng.uniform(1, 5) * x)))
        else:
            a = norm_sum1(np.maximum((n - i_arr) ** 1.0 * rng.uniform(0.8, 1.2, n), 0.01))
        v = compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    elapsed = time.monotonic() - t0
    print("  Parametric done (" + str(elapsed) + "s): best J=" + repr(best_val))
    return best_a, best_val


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs
