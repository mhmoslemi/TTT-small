def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, x)) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def enforce_mirror(a):
    n = len(a)
    half = (n + 1) // 2
    out = np.zeros(n)
    for j in range(half):
        out[j] = 0.5 * (a[j] + a[n - 1 - j])
        out[n - 1 - j] = out[j]
    return out


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    f_interp = enforce_mirror(f_interp)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    a = enforce_mirror(a)
    return norm_sum1(a)

def make_spectral(n, n_modes=30, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    a = enforce_mirror(a)
    return norm_sum1(a)


def compute_gradient(a, ac, top_k=80, beta=8.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def soft_max_gradient(a, ac, beta=20.0):
    n = len(a)
    ac_shifted = beta * (ac - ac.max())
    ac_shifted = np.minimum(ac_shifted, 700.0)
    exp_ac = np.exp(ac_shifted)
    p = exp_ac / exp_ac.sum()
    p = np.maximum(p, 0)
    grad = np.zeros(n)
    nonneg_p = p > 1e-15
    for k in np.where(nonneg_p)[0]:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * p[k] * a[j_idx]
    return grad / beta


def nesterov_subgradient(a0, time_limit, top_k=80, beta=8.0, c0=4.0,
                          restart_every=5000, verbose=False, rng_seed=42,
                          use_soft_max=False, soft_beta=20.0):
    n = len(a0)
    a = norm_sum1(a0)
    a = enforce_mirror(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = enforce_mirror(y)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        if use_soft_max:
            g = soft_max_gradient(y, ac, beta=soft_beta)
        else:
            g = compute_gradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = enforce_mirror(a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every // 3:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 5, 300)

        if stall_count > restart_every:
            stall_count = 0
            pert_scale = max(0.01, 0.05 * stall_count / restart_every)
            pert = rng.normal(0, pert_scale, n) * best_a
            x_r = best_a + pert
            x_r = enforce_mirror(x_r)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 300)
            cur_beta = max(cur_beta - 1.5, 2.0)

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSG t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}, k={cur_top_k}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200, top_K_max=None):
    n = len(a0)
    a = norm_sum1(a0)
    a = enforce_mirror(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 30, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_val if hasattr(b_var, 'value') and b_var.value is not None else np.zeros(n), dtype=np.float64), 0.0)
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 30, top_K_max)
            continue
        g_dir = b_val / b_sum
        g_dir = enforce_mirror(g_dir)
        g_dir = g_dir / g_dir.sum()

        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                      0.5, 0.3, 0.2, 0.7, 0.4]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = enforce_mirror(a_trial)
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = enforce_mirror(a)
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 5 == 0:
            K = min(K + 40, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=False, tau_init=15.0, tau_final=300.0,

    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < 800:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 400.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        ac_shifted = np.minimum(ac_shifted, 700.0)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g = enforce_mirror(g)
        g_tangent = g - g.mean()

        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)
        a_new = enforce_mirror(a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.015, 3.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.75
            if stall > 250:
                pert_scale = min(0.02, 0.005 * stall / 250.0)
                pert = rng.normal(0, pert_scale, n) * a
                a_pert = a + pert
                a_pert = enforce_mirror(a_pert)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a_pert.copy()
                    stall = 0
                    eta = eta_init * 0.5
                else:
                    a = a_new
            else:
                a = a_new

        total_iters += 1

        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def spectral_optimize(a0, time_limit, verbose=False, rng_seed=200):
    """Optimize in the frequency domain using the relation autocorr = IDFT(|A|^2)."""
    n = len(a0)
    a = norm_sum1(a0)
    a = enforce_mirror(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n).bit_length()
    eta = 0.5
    stall = 0

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 500:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2

        # Autocorrelation (circular) from power spectrum
        ac_circ = np.fft.irfft(P, fft_size)[:2 * n - 1]

        # Gradient of max(ac_circ) w.r.t. P is at the peak
        peak_idx = np.argmax(ac_circ)
        # d(ac_circ)/dP = ... (IDFT of delta at peak = exp(2*pi*i*peak*k/fft_size))
        # dP/dA = 2*Re(A) (for real a)
        # So gradient w.r.t. A is: 2*Re(A) * conj(exp(2*pi*i*peak*k/fft_size))
        # In time domain this is just the "bump" at the peak lag

        # Simpler: use subgradient in time domain
        ac = fast_autocorr(a)
        k_max = min(50, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 6.0
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        grad = enforce_mirror(grad)

        # Multiplicative update (mirror of exp(-eta*grad))
        a_new = a * np.exp(-eta * grad / max(grad.max(), 1e-15))
        a_new = enforce_mirror(a_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            stall += 1
            a = a_new
            if stall > 100:
                eta *= 0.8
            if stall > 300:
                pert = rng.normal(0, 0.01, n) * a
                a_p = enforce_mirror(a + pert)
                a_p = np.maximum(a_p, 1e-10)
                a = a_p / a_p.sum()
                stall = 150

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    a = enforce_mirror(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 150
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Mirror partner
            src_mirror = n - 1 - src

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:5]]

            found = False
            for dst in top_cands:
                if found:
                    break
                # Also try mirror of dst
                for dst_try in [dst, n - 1 - dst]:
                    if dst_try == src or dst_try == src_mirror or found:
                        continue
                    for frac in [0.01, 0.005, 0.02, 0.03, 0.05, 0.1, 0.15, 0.2]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst_try] += amt
                        a_trial[src_mirror] -= amt
                        a_trial[n - 1 - dst_try] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found = True
                            break

        if not improved:
            stall += 1
            i = int(rng.integers(n // 2))
            j = int(rng.integers(n // 2))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[n - 1 - i] -= amt
                a[j] += amt
                a[n - 1 - j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

        n_base = 5200
        best_seq = make_u_shape(n_base, sharpness=2.5)
        best_score = evaluate_safe(best_seq)

    print(f"\n=== Phase 1: Multi-start NSG (n={n0}) ===")
