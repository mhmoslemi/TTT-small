import time
import cvxpy as cp

_SOLVER = None


def _get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _SOLVER = cp.SCS
    print(f"  LP solver (fallback): SCS")
    return _SOLVER


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _interp(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return f_new / ss


def _init_sin2(n):
    if n <= 1:
        return np.array([1.0])
    j = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * j / (n - 1)) ** 2
    return _norm1(a)


def _init_family(n, family):
    x = np.linspace(0.0, 1.0, n)
    if family == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif family == 'sin4':
        a = np.sin(np.pi * x) ** 4
    elif family == 'sin6':
        a = np.sin(np.pi * x) ** 6
    elif family == 'tri':
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif family == 'tri2':
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif family == 'flat_cut':
        a = np.ones(n)
        cut = max(1, int(0.03 * n))
        a[:cut] = 0.0
        a[n - cut:] = 0.0
    elif family == 'gauss_mid':
        a = np.exp(-20.0 * (x - 0.5) ** 2)
    elif family == 'bimodal':
        a = np.exp(-50.0 * (x - 0.25) ** 2) + np.exp(-50.0 * (x - 0.75) ** 2)
    else:
        a = np.sin(np.pi * x) ** 2
    return _norm1(a)

def _spectral_refine(a0, time_limit, verbose=True):

            stall += 1

        k_peak = int(np.argmax(ac))
        m_start = max(0, k_peak - n + 1)
        m_end = min(n - 1, k_peak)
        grad = np.zeros(n)
        # Weight by autocorrelation value at each top lag
