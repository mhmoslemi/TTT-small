def soft_grad_descent(a0, time_limit, top_k=100, temperature=100, c0=2.0):
    """Multiplicative gradient descent with soft top-k objective."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    eta = c0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation and gradient
        ac = fast_autocorr(y)
        # Soft top-k gradient
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        w = np.exp(temperature * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, k in zip(w, idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx_arr = np.arange(i_start, i_end)
            grad[idx_arr] += 2.0 * wi * y[k_int - idx_arr]
        
        # Update
        eta_t = eta / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        val = 2.0 * n * fast_autocorr(x).max()
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
        
        # Periodic restart with noise
        if t % 3000 == 0:
            noise = 0.02 * rng.normal(0, 1, n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
    
    return best_a, best_val

    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:

            coeffs[jj, idx] = a[k_int - idx]

            stall = 0
        else:
            stall += 1

        if stall > 3000 and stall % 3000 == 0:
            noise = 0.015 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    GradDes t=" + str(t) + " (" + str(round(elapsed,1)) + "s) J=" + repr(best_val))

    return best_a, best_val

def coord_polish(a0, time_limit, seed=77, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 800
    fracs = [0.003, 0.006, 0.01, 0.02, 0.04, 0.08, 0.12]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                continue

            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-15:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 200:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0 and stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0:
            print("    Coord (" + str(round(time.monotonic()-t0,1)) + "s) J=" + repr(2*n*best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def deflation_step(a0, time_limit, n_targets=5, verbose=True):
    """Explicitly target the top peak lags and try to reduce them."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    for round_i in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            for _ in range(3):
                if contrib.max() < 1e-18:
                    break
                src = int(idx[np.argmax(contrib)])
                src_val = a[src]
                if src_val < 1e-12:
                    break
                # Find best destination: minimize increase in top autocorrs
                scores = np.zeros(n)
                for k2 in targets:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10
                dst = int(np.argmin(scores))
                if dst == src:
                    break
                frac = 0.005
                amt = src_val * frac
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    break
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-15:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    contrib = a[idx] * a[ki - idx]
                else:
                    break

        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.005)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()

        if verbose and round_i % 50 == 0:
            print("    Deflate round=" + str(round_i) + " J=" + repr(2*n*best_m))

    return best_a, 2.0 * n * best_m

def parametric_init(n, best_init, best_score):
    """Quick diverse initialization."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = normalize(best_init)
    best_val = obj_val(best_a)

    candidates = []
    for p in [0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr) ** p))
    for c in [0.3, 0.5, 0.7]:
        for sg in [0.1, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c)**2)/(2*sg*sg))))
    for c1 in [0.25, 0.35]:
        for c2 in [0.65, 0.75]:
            for sg in [0.08, 0.12, 0.18]:
                cands = np.exp(-((x-c1)**2)/(2*sg*sg)) + np.exp(-((x-c2)**2)/(2*sg*sg))
                candidates.append(normalize(cands))
    for f in [1, 2, 3, 5]:
        cands = np.maximum(1.0 + np.cos(2*np.pi*f*x), 0.01)
        candidates.append(normalize(cands))
    for p in [1.0, 2.0, 4.0]:
        candidates.append(normalize(x**p + (1.0-x)**p))
    # Blend with best init
    bi = normalize(best_init)
    for w in [0.3, 0.5, 0.7]:
        for cand in candidates[:4]:
            candidates.append(normalize(w*bi + (1.0-w)*cand))

    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()
            print("    Param: J=" + repr(v))

    return best_a, best_val

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    solver = get_solver()
    print("Budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n=" + str(len(p)) + " score=" + repr(best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 7000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x)**2)
        best_score = eval_safe(best_seq)
        print("Default init n=" + str(n_init) + " score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Quick parametric
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Parametric (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.03, 50.0)
    if t1 > 5:
        a_p, _ = parametric_init(n0, best_seq, best_score)
        sc = eval_safe(a_p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_p
            print("  New best param: " + repr(best_score))

    # Phase 2: Main LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: LP-FW (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.32, 500.0)
    if t2 > 60:
        a_out, v_out = lp_fw(normalize(best_seq), t2, solver,
                             init_K=500, max_K=3500, add_per=40, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW improved: " + repr(best_score))

    # Phase 3: Continue LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: LP-FW cont (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        a_out, v_out = lp_fw(normalize(best_seq), t3, solver,
                             init_K=600, max_K=4000, add_per=50, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW2 improved: " + repr(best_score))

    # Phase 4: Gradient descent at longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(n_cur + 2000, 12000)
    print("\n=== P4: GradDes n=" + str(n_long) + " ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40:
        a_long = interpolate_seq(best_seq, n_long)
        a_out, v_out = grad_descent_long(a_long, t4, top_k=80, temp=60.0, c0=2.5, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  GradDes improved: " + repr(best_score))

    # Phase 5: Gradient at even longer
    remaining = time_budget - (time.monotonic() - t_global)
    n_long2 = min(len(best_seq) + 3000, 18000)
    if n_long2 != len(best_seq):
        print("\n=== P5: GradDes n=" + str(n_long2) + " ===")
        t5 = min(remaining * 0.12, 200.0)
        if t5 > 30:
            a_long2 = interpolate_seq(best_seq, n_long2)
            a_out, v_out = grad_descent_long(a_long2, t5, top_k=100, temp=80.0, c0=2.0, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  GradDes2 improved: " + repr(best_score))

    # Phase 6: Deflation
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P6: Deflation (n=" + str(n_cur) + ") ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 20:
        a_out, v_out = deflation_step(normalize(best_seq), t6, n_targets=6, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Deflate improved: " + repr(best_score))

    # Phase 7: LP-FW on best if short enough
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 8000:
        print("\n=== P7: Final LP-FW (n=" + str(n_cur) + ") ===")
        t7 = min(remaining * 0.20, 250.0)
        if t7 > 30:
            a_out, v_out = lp_fw(normalize(best_seq), t7, solver,
                                 init_K=800, max_K=4500, add_per=60, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Final LP improved: " + repr(best_score))
    else:
        print("\n=== P7: Final GradDes (n=" + str(n_cur) + ") ===")
        t7 = min(remaining * 0.20, 250.0)
        if t7 > 30:
            a_out, v_out = grad_descent_long(normalize(best_seq), t7, top_k=100, temp=100.0, c0=1.5, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Final Grad improved: " + repr(best_score))

    # Phase 8: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P8: Coord polish (n=" + str(n_cur) + ") ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 20:
        a_out, v_out = coord_polish(normalize(best_seq), t8, seed=42, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Coord improved: " + repr(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed,1)) + "s")
    else:
        print("\nERROR fallback")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x)**2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
