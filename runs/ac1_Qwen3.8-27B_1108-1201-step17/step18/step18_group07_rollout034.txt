def spectral_peak_reduce(a, L, alpha=0.3, n_iter=1000):
    """
    Spectral method for minimizing max(conv(a,a)).
    a: non-negative sequence, sum(a) = 1
    L: FFT size (power of 2, >= 2n-1)
    alpha: peak reduction strength
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = np.max(fast_conv(a))
    
    for it in range(n_iter):
        # Compute convolution via FFT
        A = np.fft.fft(a, L)
        C = A * A  # element-wise square
        c = np.fft.ifft(C).real  # should be real
        
        # Peak reduction
        c_work = c.copy()
        c_mean = c_work[:2*n-1].mean()
        c_max = c_work[:2*n-1].max()
        if c_max > c_mean:
            # Soft peak reduction
            excess = np.maximum(c_work[:2*n-1] - c_mean, 0)
            c_work[:2*n-1] -= alpha * excess
            # Alternative: multiplicative
            # c_work[:2*n-1] = np.where(c_work[:2*n-1] > c_mean, 
            #                           c_work[:2*n-1] * (c_mean/c_work[:2*n-1])**alpha,
            #                           c_work[:2*n-1])
        
        # Reconstruct
        C_new = np.fft.fft(c_work)
        A_new = np.sqrt(np.abs(C_new)) * np.exp(1j * np.angle(C_new) / 2)
        a_full = np.fft.ifft(A_new).real
        a_new = np.maximum(a_full[:n], 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        
        # Check improvement
        M = np.max(fast_conv(a_new))
        if M < best_m:
            best_m = M
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_m

import numpy as np

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))

def spectral_peak_reduce(a, time_limit, alpha_init=0.4, alpha_min=0.05, verbose=False):
    """
    Spectral method: iteratively reduce the peak of conv(a,a) by modifying
    the spectrum and projecting back to non-negative sequences.
    """
    n = len(a)
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    
    it = 0
    alpha = alpha_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute current convolution
        A = np.fft.rfft(a, L)
        C = A * A  # complex element-wise square
        c = np.fft.irfft(C, L)[:size]
        
        # Adaptive alpha (anneal down)
        progress = min(1.0, it / 5000.0)
        alpha = alpha_init * (alpha_min / alpha_init) ** progress
        
        # Peak reduction: reduce values above the median
        c_work = c.copy()
        c_median = np.median(c_work)
        # Reduce peaks above median
        excess = np.maximum(c_work - c_median, 0)
        c_work -= alpha * excess
        
        # Ensure c_work doesn't go too negative (which would cause issues with sqrt)
        # The minimum of c should be close to 0 for non-negative a
        c_min = c_work.min()
        if c_min < -0.1 * c_median:
            c_work += (-c_min - 0.1 * c_median)  # shift up
        
        # Reconstruct: C_new = FFT(c_work), A_new = sqrt(C_new)
        C_new = np.fft.rfft(c_work, L)
        
        # Complex square root
        mag = np.sqrt(np.abs(C_new))
        phase = np.angle(C_new) / 2.0
        A_new = mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_full = np.fft.irfft(A_new, L)
        
        # Project to non-negative, supported on [0, n-1]
        a_new = np.maximum(a_full[:n], 0.0)
        s = a_new.sum()
        if s < 1e-15:
            # Fallback: use soft projection
            a_new = np.exp(np.maximum(a_full[:n] - a_full[:n].max(), -50))
            s = a_new.sum()
            if s < 1e-15:
                a_new = best_a.copy()
            else:
                a_new = a_new / s
        else:
            a_new = a_new / s
        
        # Track best
        M = fast_conv(a_new).max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a_new.copy()
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SpecPR it={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}, alpha={alpha:.4f}")
    
    return best_a, best_m


def spectral_multiplicative(a, time_limit, beta=0.3, verbose=False):
    """
    Spectral method with multiplicative peak compression.
    Instead of subtractive, multiply: c' = c * (median/c)^beta for c > median.
    """
    n = len(a)
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        A = np.fft.rfft(a, L)
        C = A * A
        c = np.fft.irfft(C, L)[:size]
        
        # Multiplicative compression
        c_work = c.copy()
        c_median = np.median(c_work)
        if c_median > 1e-30:
            # Compress values above median
            ratio = np.maximum(c_work / c_median, 1.0)
            c_work = c_work / (ratio ** beta)
        
        C_new = np.fft.rfft(c_work, L)
        mag = np.sqrt(np.abs(C_new))
        phase = np.angle(C_new) / 2.0
        A_new = mag * np.exp(1j * phase)
        a_full = np.fft.irfft(A_new, L)
        
        a_new = np.maximum(a_full[:n], 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        
        M = fast_conv(a_new).max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a_new.copy()
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SpecMult it={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m


def ftrl_optimize(a0, time_limit, verbose=False):
    """FTRL mirror descent with annealed log-sum-exp smoothing."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    c0 = 1.0
    T_init = best_m * 0.5
    T_min = best_m * 0.01
    top_k_init = 20
    top_k_min = 3
    
    x = a.copy()
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, t / 80000.0)
        T = T_init * (T_min / T_init) ** progress
        cur_top_k = max(top_k_min, int(top_k_init * (1 - 0.7 * progress) + top_k_min))
        cur_beta = 15.0 + 35.0 * progress
        
        # Nesterov
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Subgradient
        c = fast_conv(y)
        k_max = min(cur_top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = c[top_idx].copy()
        
        max_v = top_vals[0]
        if max_v < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            exponents = cur_beta * (top_vals - max_v) / max_v
            exponents = np.maximum(exponents, -50.0)
            w = np.exp(exponents)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x = a_new
        a = a_new
        
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions to reduce peak."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            for attempt in range(8):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m

def make_init_random_smooth(n, seed=0):

    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5

        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)

def make_init_triangle_asym(n):

    x = np.linspace(0, 1, n)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:

        return float('inf')


def spectral_peak_reduce(a, time_limit, alpha_init=0.4, alpha_min=0.03, verbose=False):
    """
    Spectral method: iteratively reduce the peak of conv(a,a) by modifying
    the autocorrelation in the frequency domain and projecting back to
    non-negative sequences with correct support.
    """
    n = len(a)
    a = norm_simplex(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()

    it = 0
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        it += 1

        # Compute current spectrum and convolution
        A = np.fft.rfft(a, L)
        C = A * A
        c = np.fft.irfft(C, L)[:size]

        # Adaptive alpha (anneal down)
        progress = min(1.0, it / 10000.0)
        alpha = alpha_init * (alpha_min / alpha_init) ** progress

        # Peak reduction: reduce values above the mean
        c_work = c.copy()
        c_mean = c_work.mean()
        c_median = np.median(c_work)
        # Use a blend: reduce excess above a target between mean and median
        target = 0.7 * c_median + 0.3 * c_mean
        excess = np.maximum(c_work - target, 0)
        c_work -= alpha * excess

        # Ensure c_work stays reasonable (not too negative)
        c_min = c_work.min()
        if c_min < -0.5 * max(c_median, 1e-15):
            c_work += (-c_min - 0.3 * max(c_median, 1e-15))

        # Reconstruct: C_new = FFT(c_work), A_new = sqrt(C_new)
        C_new = np.fft.rfft(c_work, L)

        # Complex square root: sqrt(re + i*im) = sqrt(|z|) * exp(i*angle(z)/2)
        mag = np.sqrt(np.maximum(np.abs(C_new), 0))
        phase = np.angle(C_new) / 2.0
        A_new = mag * np.exp(1j * phase)

        # Inverse FFT
        a_full = np.fft.irfft(A_new, L)

        # Project: enforce support [0, n-1] and non-negativity
        a_new = np.maximum(a_full[:n], 0.0)
        s = a_new.sum()
        if s < 1e-15:
            # Fallback: soft projection via exponential
            shifted = a_full[:n] - np.max(a_full[:n])
            a_new = np.exp(np.maximum(shifted, -50))
            s = a_new.sum()
            if s < 1e-15:
                a_new = best_a.copy()
            else:
                a_new = a_new / s
        else:
            a_new = a_new / s

        # Track best
        M = fast_conv(a_new).max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SpecPR it={it} ({elapsed:.0f}s): J={2*n*best_m:.6f}, alpha={alpha:.4f}")

    return best_a, best_m

def ftrl_optimize(a0, time_limit, verbose=False):
    """FTRL mirror descent with annealed subgradient."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    c0 = 1.0
    T_init = max(best_m * 0.5, 1e-6)
    T_min = max(best_m * 0.01, 1e-8)
    top_k_init = 20
    top_k_min = 3

    x = a.copy()

    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, t / 100000.0)
        T = T_init * (T_min / T_init) ** progress
        cur_top_k = max(top_k_min, int(top_k_init * (1 - 0.7 * progress) + top_k_min))
        cur_beta = 15.0 + 35.0 * progress

        # Nesterov step
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient from top peaks
        c = fast_conv(y)
        k_max = min(cur_top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = c[top_idx].copy()

        max_v = top_vals[0]
        if max_v < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            exponents = cur_beta * (top_vals - max_v) / max_v
            exponents = np.maximum(exponents, -50.0)
            w = np.exp(exponents)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        # Mirror descent step (exponential family)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x = a_new
        a = a_new

        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()

        if verbose and t % 30000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions to reduce peak."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]

        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue

            for attempt in range(10):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.003, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape local minimum
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def make_init_exp_decay(n, rate=1.0):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * x / n)
    return norm_simplex(a)


def make_init_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_reverse_exp(n, rate=1.0):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - x) / n)
    return norm_simplex(a)


def make_init_dual_peak_asym(n):
    x = np.linspace(0, 1, n)
    a = 1.0 * np.exp(-((x - 0.2) ** 2) / 0.015) + 0.6 * np.exp(-((x - 0.75) ** 2) / 0.025) + 0.03
    return norm_simplex(a)


def make_init_triangle_asym(n):
    x = np.linspace(0, 1, n)
    a = np.where(x < 0.35, x / 0.35, (1 - x) / 0.65)
    a = np.maximum(a, 0.01)
    return norm_simplex(a)


def make_init_sinc_asym(n, bw=5.0, shift=0.15):
    x = np.linspace(0, 1, n)
    t = (x - 0.5 - shift) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_random_smooth(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def make_init_chirp_envelope(n, chirp_rate=3.0):
    """Asymmetric chirp-like envelope."""
    x = np.linspace(0, 1, n)
    a = np.exp(-2.0 * x) * (1.0 + 0.5 * np.cos(2 * np.pi * chirp_rate * x))
    a = np.maximum(a, 0.01)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining spectral methods, FTRL, and local search."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_init_exp_decay(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Spectral peak reduction on diverse asymmetric inits ===
    print(f"\n=== Phase 1: Spectral Peak Reduction (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 500.0)

    if t_phase1 > 30:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_simplex(parent))
            labels.append("parent")

        inits.append(make_init_exp_decay(n0, rate=1.0))
        labels.append("exp1")

        inits.append(make_init_exp_decay(n0, rate=2.5))
        labels.append("exp25")

        inits.append(make_init_reverse_exp(n0, rate=1.5))
        labels.append("rev_exp")

        inits.append(make_init_ramp(n0, power=0.7))
        labels.append("ramp07")

        inits.append(make_init_dual_peak_asym(n0))
        labels.append("dualpeak")

        inits.append(make_init_triangle_asym(n0))
        labels.append("triangle")

        inits.append(make_init_sinc_asym(n0, bw=5.0, shift=0.15))
        labels.append("sinc_asym")

        inits.append(make_init_chirp_envelope(n0, chirp_rate=4.0))
        labels.append("chirp")

        inits.append(make_init_random_smooth(n0, seed=42))
        labels.append("rand42")

        inits.append(make_init_random_smooth(n0, seed=99))
        labels.append("rand99")

        t_per = t_phase1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            remaining_now = time_budget - 20 - elapsed_so_far
            if remaining_now < 10:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 8:
                break
            print(f"    SpecPR '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = spectral_peak_reduce(init, t_avail, alpha_init=0.4, alpha_min=0.03, verbose=False)

                try_update(a_f, f"ftrl_{lbl}")

    # === Phase 2: Combined spectral method on best ===
