This is a joint optimization over (g, t, M), which is still an LP (the constraint is linear in g and M, and t only appears in the coefficient). But it's more complex.

Actually, the constraint is: (1-t)*(y*y)(k) + t*(y*g)(k) + t^2*(g*g)(k) <= M. The t^2*(g*g)(k) term makes this nonlinear (it's quadratic in g and t). So it's not an LP.

One more idea: **proximal gradient method**. The objective f(a) = max_k (a*a)(k) is the max of convex functions (each (a*a)(k) is convex in a when a >= 0... wait, is it?).

Let's check: (a*a)(k) = sum_{i+j=k} a_i * a_j. The Hessian of this w.r.t. a is 2*B_k (where B_k is the matrix with (B_k)_{i,j} = 1 if i+j=k). For this to be PSD, we need B_k to be PSD. But B_k has eigenvalues that can be negative (it's a "reversed identity" for k=0, which is just the identity, but for other k it's a partial permutation matrix).

Actually, for k=0: B_0 has (B_0)_{0,0} = 1, rest 0. PSD. ✓
For k=1: B_1 has (B_1)_{0,1} = (B_1)_{1,0} = 1. Eigenvalues 1 and -1. NOT PSD. ✗

So (a*a)(1) = 2*a_0*a_1 is NOT convex in a. The objective is not convex.

This means gradient-based methods won't work in the usual sense. The LP-FW approach is using a "convex relaxation" that works in practice but isn't theoretically guaranteed.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with exact line search
2. Aggressive constraint management
3. Coordinate descent polishing
4. Multi-length exploration
"""

import numpy as np
import time
import cvxpy as cp

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def fast_conv(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def fast_autocorr(a):
    return fast_conv(a, a)

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_max_conv(a):
    return fast_autocorr(a).max()

def exact_line_search(yy_conv, yg_conv, gg_conv, num_points=200):
    """Find alpha in [0,1] minimizing max of combined convolution."""
    alphas = np.linspace(0, 1, num_points)
    best_alpha = 0
    best_val = float('inf')
    for alpha in alphas:
        combined = (1-alpha)**2 * yy_conv + 2*alpha*(1-alpha) * yg_conv + alpha**2 * gg_conv
        val = combined.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    # Refine
    if best_alpha > 0 and best_alpha < 1:
        lo = best_alpha - 1.0/num_points
        hi = best_alpha + 1.0/num_points
        alphas_fine = np.linspace(max(0, lo), min(1, hi), 50)
        for alpha in alphas_fine:
            combined = (1-alpha)**2 * yy_conv + 2*alpha*(1-alpha) * yg_conv + alpha**2 * gg_conv
            val = combined.max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
    return best_alpha

def lp_fw(a0, time_limit, solver, initial_K=400, max_K=3000, verbose=True):
    """LP-based Frank-Wolfe with exact line search."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    yy_conv = fast_autocorr(y)
    active_set = list(np.argsort(yy_conv)[::-1][:initial_K])
    
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        # Solve LP
        K = len(active_set)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_set):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce active set
            active_set = active_set[:max(100, K//2)]
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Exact line search
        # Precompute yg_conv and gg_conv
        yg_conv = fast_conv(y, g_dir)
        gg_conv = fast_autocorr(g_dir)
        
        alpha = exact_line_search(yy_conv, yg_conv, gg_conv)
        
        # Update
        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        
        y = y_new
        yy_conv = fast_autocorr(y)
        M_curr = yy_conv.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set: add violated constraints
        # Check which constraints are violated (yy_conv[k] > M_curr + tolerance)
        # Actually, we should check yg_conv for violations in the LP sense
        # The LP was: (y*g)(k) <= M for k in active_set
        # After the step, the new "violated" constraints for the next LP step
        # are those where the new convolution exceeds the new M.
        
        # Add new constraints: find peaks in yy_conv that are close to max
        top_indices = np.argsort(yy_conv)[::-1][:max_K]
        new_set = set(active_set)
        for k in top_indices:
            new_set.add(int(k))
        # Keep only top max_K by value
        if len(new_set) > max_K:
            vals = [yy_conv[k] for k in new_set]
            sorted_idx = np.argsort(vals)[::-1][:max_K]
            active_set = [new_set[i] for i in sorted_idx]
        else:
            active_set = list(new_set)
        
        iter_count += 1
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-FW iter={t} ({elapsed:.1f}s): M={best_m:.8f}, |S|={len(active_set)}")
    
    elapsed = time.monotonic() - t0
    print(f"  LP-FW done ({elapsed:.1f}s): M={best_m:.8f}, iters={iter_count}")
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Coordinate descent: for each coordinate, find optimal 1D move."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False
        
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if i >= n:
                continue
            
            a_i = a[i]
            if a_i < 1e-15:
                continue
            
            # Compute "gradient" for coordinate i
            # g_k = 2 * a_{k-i} for k where 0 <= k-i < n
            # So the change in ac[k] is proportional to 2*a_{k-i}*delta
            
            # We want to find delta that minimizes max_k [ac[k] + 2*delta*a_{k-i}]
            # (ignoring the delta^2 term for small delta)
            
            # The "influence" vector for coordinate i:
            # For k in [i, i+n-1] (clipped to [0, 2n-2]):
            #   influence[k] = 2 * a[k-i] (if k-i < n)
            # Plus the self-term: ac[2i] += delta^2 (if 2i < 2n-1)
            
            # Simplified: find the best delta by trying a few values
            # Move mass from i to a "safe" location
            
            # Find where to move mass: the k where ac[k] is maximal
            # and a_{k-i} is large (i.e., i contributes a lot to the peak)
            
            peak_k = int(np.argmax(ac))
            contrib = a_i * a[peak_k - i] if 0 <= peak_k - i < n else 0
            
            if contrib < 1e-20:
                continue
            
            # Try reducing a_i and adding to a "safe" coordinate
            # Safe coordinate: one where a_{k-i} is small for all peak k
            peak_ks = np.argsort(ac)[::-1][:20]
            influence = np.zeros(n)
            for k in peak_ks:
                ki = int(k) - i
                if 0 <= ki < n:
                    influence[ki] += ac[int(k)]
            
            # Find coordinate with minimum influence
            safe_j = int(np.argmin(influence))
            if safe_j == i:
                continue
            
            # Try different amounts to move
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a_i * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[safe_j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
        else:
            stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Coord ({elapsed:.1f}s): M={best_m:.8f}, stall={stall}")
    
    return best_a, best_m
```

```python
def fast_conv(a, b=None):
    if b is None:

    n = len(a)
    m = len(b)
    size = n + m - 1

def exact_line_search(yy_conv, yg_conv, gg_conv, n_coarse=100, n_fine=50):
    """Find alpha in [0,1] minimizing max of (1-a)^2*yy + 2a(1-a)*yg + a^2*gg."""
    alphas = np.linspace(0.0, 1.0, n_coarse)
    best_alpha = 0.0
    best_val = float('inf')
    for alpha in alphas:
        combined = (1.0 - alpha) ** 2 * yy_conv + 2.0 * alpha * (1.0 - alpha) * yg_conv + alpha ** 2 * gg_conv
        val = combined.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    # Refine around best
    if best_alpha > 0.0 and best_alpha < 1.0:
        step = 1.0 / n_coarse
        lo = max(0.0, best_alpha - step)
        hi = min(1.0, best_alpha + step)
        alphas_fine = np.linspace(lo, hi, n_fine)
        for alpha in alphas_fine:
            combined = (1.0 - alpha) ** 2 * yy_conv + 2.0 * alpha * (1.0 - alpha) * yg_conv + alpha ** 2 * gg_conv
            val = combined.max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
    return best_alpha


def build_band_active_set(ac, n, center_frac=0.3, min_K=200, max_K=3000):
    """Build active set as a band around the peak position, plus top global peaks."""
    size = len(ac)
    peak_pos = int(np.argmax(ac))
    peak_val = ac[peak_pos]
    # Band around peak: positions where ac >= 0.85 * peak (or within center_frac of center)
    band = np.where(ac >= 0.85 * peak_val)[0]
    # Also include top max_K positions globally

def lp_fw(a0, time_limit, solver, verbose=True, initial_K=500, max_K=3000,

    iter_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:

            break

        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj in range(K):

            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        M_var = cp.Variable()

            active_set = active_set[:max(200, K // 2)]

        # Exact line search
        yg_conv = fast_conv(y, g_dir)
        gg_conv = fast_autocorr(g_dir)
        alpha = exact_line_search(yy_conv, yg_conv, gg_conv)

def block_coordinate_descent(a0, time_limit, verbose=True, block_size=4):
    """Block coordinate descent: update groups of adjacent coordinates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

    t0 = time.monotonic()

    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if i >= n:
                continue
            a_i = a[i]
            if a_i < 1e-15:
                continue

            # Compute influence on top peaks
            top_ks = np.argsort(ac)[::-1][:15]
            influence = np.zeros(n)
            for k in top_ks:
                ki = int(k) - i
                if 0 <= ki < n:
                    influence[ki] += ac[int(k)]

            # Find safe destination
            influence[i] = np.inf
            safe_j = int(np.argmin(influence))

            if safe_j == i or not np.isfinite(influence[safe_j]):
                safe_candidates = np.where(influence == np.min(np.where(influence == np.inf, 1e30, influence)))[0]
                if len(safe_candidates) == 0:
                    continue
                safe_j = int(safe_candidates[0])
                if safe_j == i:
                    continue

            # Try moving mass
            for frac in [0.02, 0.05, 0.1, 0.15]:
                amt = a_i * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = a_i - amt
                a_trial[safe_j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        else:
            stall += 1

        if verbose and stall % 40 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD ({elapsed:.0f}s): M={best_m:.10f}, stall={stall}")

def spectral_shaping(a0, n_iter=50, verbose=False):
    """Iterative spectral flattening: flatten |A(f)|^2 and reconstruct."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        mag2 = np.abs(A) ** 2
        # Target: flat spectrum (proportional to 1)
        # But we need to keep DC component (sum(a))
        # Use: new_mag = sqrt(target) where target is smoothed mag2
        # Simple approach: new_A = A * (mean_mag2 / mag2)^(1/2)
        mag2_safe = np.maximum(mag2, 1e-30)
        target = mag2.mean()
        scale = np.sqrt(target / mag2_safe)
        # Don't scale DC too much
        scale[0] = 1.0
        new_A = A * scale
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new

    if verbose:
        print(f"    Spectral: M={best_m:.10f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        candidates.append(rng.random(n) ** rng.uniform(0.5, 3.0))

    for c in candidates:
        a = norm_sum1(c)
        m = compute_max_conv(a)
        if m < best_m:
            best_m = m
            best_a = a.copy()

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n = 5000
        a_init, _ = parametric_init(n)
        best_seq = a_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n}, score={best_score:.10f}")

    n0 = len(best_seq)

    # Phase 1: Quick spectral shaping
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Spectral shaping (n={n0}) ===")
    a_out, _ = spectral_shaping(best_seq, n_iter=100, verbose=True)
    sc = evaluate_safe(a_out)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_out
        print(f"  NEW BEST: {best_score:.10f}")

    # Phase 2: Main LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_cur}, budget={remaining:.0f}s) ===")
    t2 = remaining * 0.40
    a_out, _ = lp_fw(best_seq, t2, solver, verbose=True, initial_K=500, max_K=3000,
                     band_frac=0.85, refresh_every=40, max_no_improve=500)
    sc = evaluate_safe(a_out)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_out
        print(f"  NEW BEST after LP-FW: {best_score:.10f}")

    # Phase 3: Symmetric LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Symmetric LP-FW (n={n_cur}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 20 and n_cur <= 8000:
        a_out, _ = symmetric_lp_fw(best_seq, t3, solver, verbose=True, initial_K=300, max_K=2000,
                                   refresh_every=30, max_no_improve=250)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST after Sym-LP: {best_score:.10f}")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Block CD (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 120.0)
    if t4 > 10:
        a_out, _ = block_coordinate_descent(best_seq, t4, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST after BCD: {best_score:.10f}")

    # Phase 5: LP-FW from perturbed start
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP-FW perturbed (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 250.0)
    if t5 > 30 and len(best_seq) <= 8000:
        rng = np.random.default_rng(777)
        pert = rng.normal(0, 0.015, len(best_seq)) * best_seq
        a_pert = norm_sum1(best_seq + pert)
        a_out, _ = lp_fw(a_pert, t5, solver, verbose=True, initial_K=400, max_K=2500,
                         band_frac=0.85, refresh_every=40, max_no_improve=400)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST after LP-FW perturbed: {best_score:.10f}")

    # Phase 6: Try larger n
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    for new_n in [n_best + 400, n_best + 800]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        if new_n > 9000:
            continue
        print(f"\n=== Phase 6: Scale to n={new_n} (budget={remaining:.0f}s) ===")
        a_interp = interpolate_sequence(best_seq, new_n)
        t6 = remaining * 0.35
        a_out, _ = lp_fw(a_interp, t6, solver, verbose=True, initial_K=400, max_K=2500,
                         band_frac=0.85, refresh_every=40, max_no_improve=350)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  NEW BEST at n={new_n}: {best_score:.10f}")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 7: Final BCD (budget={remaining:.0f}s) ===")
        t7 = remaining * 0.7
        a_out, _ = block_coordinate_descent(best_seq, t7, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        n_fb = 5000
        best_seq, _ = parametric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.10f}")

    return [float(x) for x in best_seq]
```
