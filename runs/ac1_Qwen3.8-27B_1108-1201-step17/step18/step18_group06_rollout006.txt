    """Find the best direction to improve a, given current max conv M."""
    n = len(a)
    b = cp.Variable(n)
    constraints = [b >= 0]
    
    # conv(a, b)_k <= M for all k
    # conv(a,b) has 2n-1 elements
    # conv(a,b)_k = sum_j a_j * b_{k-j}

def lp_direction(a, M, top_k=200):
    """Solve LP to find best direction: max sum(b) s.t. a*b <= M, b>=0.
    Uses only top_k most active lags for speed."""
    n = len(a)
    if not HAS_CP:
        return a.copy()
    
    ac = fast_autocorr(a)
    # Get top k lags
    top_lags = np.argsort(ac)[::-1][:top_k]
    
    b = cp.Variable(n, nonneg=True)
    constraints = []
    for k in top_lags:
        ki = int(k)
        # (a*b)_ki = sum_j a_j * b_{ki-j} for valid j
        # = sum_i b_i * a_{ki-i} for valid i
        # Coefficient of b_i is a_{ki-i} if 0 <= ki-i < n
        coeffs = np.zeros(n)
        for i in range(n):
            idx = ki - i
            if 0 <= idx < n:
                coeffs[i] = a[idx]
        nonzero = coeffs > 1e-15
        if np.any(nonzero):
            constraints.append(coeffs @ b <= M)
    
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK)  # or GLOP, ECOS
    except:
        try:
            prob.solve()
        except:
            return a.copy()
    
    if b.value is None:
        return a.copy()
    
    b_val = np.maximum(b.value, 0)
    s = b_val.sum()
    if s < 1e-10:
        return a.copy()
    return b_val / (a.sum())

def autocorr_after_perturb(A, i, j, delta, n):
    """Fast autocorrelation after a[i]+=delta, a[j]-=delta, given FFT A."""
    fft_size = len(A)
    omega = 2.0 * np.pi / fft_size
    # RFFT: A[k] = sum_t a[t] * exp(-2*pi*i*t*k/fft_size) for k=0..fft_size/2

def compute_subgradient(a, ac, n_peaks=150, temperature=1.0):
    """Compute softmax-weighted subgradient of max(ac) over top n_peaks."""
    n = len(a)
    k_max = min(n_peaks, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    log_vals = np.log(np.maximum(top_vals, 1e-30))
    log_max = log_vals.max()
    exp_vals = np.exp((log_vals - log_max) / temperature)
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        k = int(top_idx[ii])
        wi = w[ii]
        if wi < 1e-12:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = k - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    return grad

            print(f"    GRAD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_guided_optimize(a0, time_limit, top_k=150, verbose=False, rng_seed=42):
    """LP-guided optimization: iteratively solve LP for direction, blend."""
    if not HAS_CP:
        return a0.copy(), fast_autocorr(a0).max()

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    # Gradient prefix for speed
    a_cum = np.concatenate([[0.0], np.cumsum(a)])
    a_rev_cum = np.concatenate([[0.0], np.cumsum(a[::-1])])

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:top_k]

        b = cp.Variable(n, nonneg=True)
        constraints = []
        for k in top_lags:
            ki = int(k)
            # (a*b)_ki = sum_i b_i * a_{ki-i} where 0<=ki-i<n
            coeffs = np.zeros(n)
            i_lo = max(0, ki - n + 1)
            i_hi = min(n - 1, ki)
            if i_lo <= i_hi:
                coeffs[i_lo:i_hi + 1] = a[ki - (i_hi):ki - i_lo - 1:-1]
                # coeffs[i] = a[ki-i] for i in [i_lo, i_hi]
                coeffs[i_lo:i_hi + 1] = a[ki - np.arange(i_lo, i_hi + 1)]
            if np.any(coeffs > 1e-15):
                constraints.append(coeffs @ b <= M_curr)

        obj = cp.Maximize(cp.sum(b))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000)
        except Exception:
            try:
                prob.solve()
            except Exception:
                break

        if b.value is None:
            break

        b_val = np.maximum(b.value, 0)
        s_b = b_val.sum()
        if s_b < 1e-10:
            break

        # Rescale b to have same sum as a
        b_norm = b_val / s_b
        a_sum = a.sum()
        b_norm = b_norm * (a_sum / b_norm.sum())

        # Blend
        for blend in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_trial = (1 - blend) * a + blend * b_norm
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                stall = 0
                break
        else:
            stall += 1
            if stall > 20:
                break

        iteration += 1
        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP it={iteration} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def fine_local_search(a0, time_limit, verbose=False, rng_seed=42):
    """Ultra-fine 2-coordinate random local search with FFT-based evaluation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Precompute FFT
    A, fft_size = compute_fft(a)
    mean_a = a.mean()

    # Adaptive delta
    delta_scale = 1e-5
    stall = 0
    trials = 0
    best_trial_count = trials

    while time.monotonic() - t0 < time_limit:
        # Batch of trials
        batch_size = 50
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break

            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue

            # Decide direction: move mass from larger to smaller, or random
            if a[i] > a[j]:
                src, dst = i, j
                sign = 1.0
            elif a[j] > a[i]:
                src, dst = j, i
                sign = 1.0
            else:
                if rng.random() < 0.5:
                    src, dst = i, j
                else:
                    src, dst = j, i
                sign = 1.0

            if a[src] * < 1e-20:
                continue

            delta = rng.uniform(0.3, 1.0) * delta_scale * mean_a
            if delta > a[src] * 0.5:
                delta = a[src] * 0.3

            # Compute new autocorrelation
            ac_new = autocorr_after_perturb(A, src, dst, delta, n)
            M_new = ac_new.max()

            if M_new < best_m - 1e-18:
                # Accept
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                s = a.sum()
                a = a / s
                # Update FFT incrementally
                k_vals = np.arange(len(A), dtype=np.float64)
                omega = 2.0 * np.pi / fft_size
                A += delta * (np.exp(-1j * omega * src * k_vals) - np.exp(-1j * omega * dst * k_vals))
                best_m = M_new
                best_a = a.copy()
                stall = 0
                trials += 1
            else:
                trials += 1

        # Check progress
        if stall > 200:
            # Reduce delta
            delta_scale *= 0.5
            if delta_scale < 1e-10:
                # Reset delta and try larger random moves
                delta_scale = 1e-4
                # Random perturbation
                idx = rng.choice(n, size=min(50, n), replace=False)
                for ii in idx:
                    a[ii] *= rng.uniform(0.95, 1.05)
                a = np.maximum(a, 0)
                a = a / a.sum()
                A, fft_size = compute_fft(a)
                best_m = fast_autocorr(a).max()
                best_a = a.copy()
            stall = 0
        elif trials - best_trial_count > 100:
            stall += 1

        if verbose and trials % 100000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LOCAL t={trials} ({elapsed:.0f}s): J={J:.6f}, delta={delta_scale:.2e}")

    return best_a, best_m


def polynomial_shape_optimize(a0, n, time_limit, degree=8, verbose=False):
    """Optimize a polynomial shape a(x) = sum c_k * x^k to minimize score."""
    x = np.linspace(0, 1, n)
    # Initial coefficients from a0
    V = np.vander(x, N=degree + 1, increasing=True)
    c, _, _, _ = np.linalg.lstsq(V, a0, rcond=None)
    c = np.maximum(c, 0)

    best_score = float('inf')
    best_c = c.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for outer in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        a = V @ c
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        score = 2.0 * n * fast_autocorr(a).max()

        if score < best_score:
            best_score = score
            best_c = c.copy()

        # Gradient step: perturb each coefficient
        step = 0.01 * np.max(np.abs(c)) + 1e-10
        improved = False
        for k in range(len(c)):
            for direction in [1, -1]:
                c_trial = c.copy()
                c_trial[k] += direction * step
                if c_trial[k] < 0:
                    c_trial[k] = 0
                a_trial = V @ c_trial
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                sc_t = 2.0 * n * fast_autocorr(a_trial).max()
                if sc_t < best_score - 1e-12:
                    c = c_trial
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Random perturbation
            c = c + rng.normal(0, step * 0.5, len(c))
            c = np.maximum(c, 0)
            step *= 0.9

    a_final = V @ best_c
    a_final = np.maximum(a_final, 0)
    a_final = a_final / a_final.sum()
    return a_final


def coordinate_sweep(a0, time_limit, verbose=False):
    """Sweep each coordinate, trying small adjustments."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    order = np.argsort(a)  # sweep from smallest to largest

    for round_num in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            ii = int(idx)
            if a[ii] < 1e-15:
                continue
            # Try reducing a[ii] slightly, adding to a random other position
            for frac in [0.001, 0.005, 0.01, 0.02]:
                amt = a[ii] * frac
                dst = int(rng.integers(n))
                if dst == ii:
                    continue
                a_trial = a.copy()
                a_trial[ii] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_autocorr(a_trial).max()
                if M_t < best_m - 1e-16:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    break

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_power_law_ramp(n_base, power=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP-guided optimization ===
    print(f"\n=== Phase 1: LP-guided optimization (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 250.0)
    if t1 > 50 and HAS_CP:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_guided_optimize(a_in, t1, top_k=150, verbose=True)
        try_update(a_out, f"lp_n{n0}")

    # === Phase 2: Fine gradient descent ===
    print(f"\n=== Phase 2: Fine gradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = grad_optimize(a_in, t2, n_peaks=150, temperature0=1.5,
                                  c0=3.0, restart_every=5000, verbose=True, rng_seed=42)
        try_update(a_out, f"grad_n{n0}")

    # === Phase 3: Ultra-fine local search ===
    print(f"\n=== Phase 3: Fine local search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fine_local_search(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, f"local_n{n0}")

    # === Phase 4: Gradient + local combo ===
    print(f"\n=== Phase 4: Grad + Local combo (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4a = min(remaining * 0.15, 200.0)
    t4b = min(remaining * 0.10, 150.0)
    if t4a > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = grad_optimize(a_in, t4a, n_peaks=200, temperature0=1.0,
                                  c0=2.0, restart_every=8000, verbose=False, rng_seed=77)
        try_update(a_out, f"grad2_n{n0}")
        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.10, 150.0)
        if t4b > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = fine_local_search(a_in, t4b, verbose=False, rng_seed=88)
            try_update(a_out, f"local2_n{n0}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    lengths_to_try = []
    for delta in [-500, -200, 200, 500, 1000]:
        new_n = n_best + delta
        if 2500 <= new_n <= 8000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))

    t_per_len = max(60, min(120, remaining / max(1, len(lengths_to_try))))
    for new_n in lengths_to_try[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t5a = min(remaining * 0.35, t_per_len)
        if t5a > 20:
            a_out, _ = grad_optimize(a_long, t5a, n_peaks=100, temperature0=2.0,
                                      c0=4.0, restart_every=4000, verbose=False, rng_seed=42)
            try_update(a_out, f"grad_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.25, 80.0)
            if t5b > 15:
                a_out, _ = fine_local_search(norm_sum1(best_seq), t5b, verbose=False, rng_seed=55)
                try_update(a_out, f"local_n{new_n}")

    # === Phase 6: Polynomial shape exploration (different structure) ===
    print(f"\n=== Phase 6: Polynomial shape (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t6 = min(remaining * 0.15, 100.0)
        a_out = polynomial_shape_optimize(norm_sum1(best_seq), n0, t6, degree=6)
        try_update(a_out, f"poly_n{n0}")
        if a_out is not None and np.isfinite(evaluate_safe(a_out)):
            # Polish the polynomial result
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_pol = norm_sum1(a_out)
                a_out2, _ = fine_local_search(a_pol, min(60, remaining * 0.5), verbose=False, rng_seed=33)
                try_update(a_out2, f"poly_local_n{n0}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7a = min(remaining * 0.4, 120.0)
        t7b = min(remaining * 0.5, 150.0)
        if t7a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = grad_optimize(a_in, t7a, n_peaks=250, temperature0=0.8,
                                      c0=1.5, restart_every=10000, verbose=True, rng_seed=123)
            try_update(a_out, f"grad_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = fine_local_search(a_in, min(remaining * 0.8, 150.0), verbose=True, rng_seed=456)
            try_update(a_out, f"local_final")

    # === Phase 8: Very last try - coordinate sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_sweep(a_in, min(remaining * 0.7, 80.0), verbose=False)
        try_update(a_out, f"sweep_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_power_law_ramp(4200, power=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
