def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    for freq in [1, 2, 3, 4]:
        for ph in [0, 0.25, 0.5, 0.75]:
            candidates.append(np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01))
    
    # Exponential
    for rate in [1, 2, 3, 5, 8, 12]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1)))
    
    # Triangle
    candidates.append(1.0 - np.abs(2.0 * x - 1.0))
    # Flat middle, tapered ends
    for t in [0.1, 0.2, 0.3]:
        candidates.append(np.clip(1.0 - np.abs(x - 0.5) / t, 0.1, 1.0))
    
    # Cosine taper
    candidates.append(np.cos(np.pi * x / 2.0) ** 2)
    candidates.append(np.sin(np.pi * x) ** 2)
    
    for cand in candidates:
        cand = norm_sum1(cand)
        val = score_normalized(cand)
        if val < best_val:
            best_val = val
            best_a = cand
    
    # Also try blends with best_init
    if best_init is not None:
        for w in [0.1, 0.2, 0.3, 0.5]:
            blend = norm_sum1(w * best_init + (1 - w) * best_a)
            val = score_normalized(blend)
            if val < best_val:
                best_val = val
                best_a = blend
    
    return best_a, best_val


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, initial_K=300, max_K=2500):
    """
    LP-based Frank-Wolfe with adaptive line search.
    Minimizes max autocorrelation subject to a>=0, sum(a)=1.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Pre-compute FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Initialize constraint set with top autocorrelation lags
    ac = fast_autocorr(a)
    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_set = list(set(int(k) for k in init_idx))
    
    iter_count = 0
    no_improve_count = 0
    last_best_m = best_m
    t_prev_val = best_m  # For patience
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Manage constraint set
        if len(constraint_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_set])
            keep = np.argsort(margins)[:max_K]
            constraint_set = [constraint_set[i] for i in keep]
        
        K = len(constraint_set)
        if K == 0:
            break
        
        # Build constraint matrix
        coeff_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_set):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeff_matrix[jj, idx] = a[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeff_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        Yf = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ag = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ag > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ag[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                if int(k) not in constraint_set:
                    constraint_set.append(int(k))
        
        # Line search: try different step sizes
        # Standard FW: alpha = 2/(t+2)
        alpha_fw = 2.0 / (t + 2.0)
        
        # Try a few step sizes and pick the best
        candidate_alphas = [alpha_fw]
        if t > 5:
            candidate_alphas.extend([alpha_fw * 1.5, alpha_fw * 2.0, min(0.3, alpha_fw * 3.0)])
        
        best_alpha = alpha_fw
        best_trial_m = fast_autocorr((1 - alpha_fw) * a + alpha_fw * g_dir).max()
        
        for alpha_c in candidate_alphas:
            alpha_c = min(alpha_c, 1.0)
            trial = (1 - alpha_c) * a + alpha_c * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            m_t = fast_autocorr(trial).max()
            if m_t < best_trial_m:
                best_trial_m = m_t
                best_alpha = alpha_c
        
        a_new = (1 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_a
        
        a = a_new
        iter_count += 1
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, alpha={best_alpha:.4f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def smooth_max_gradient(a, beta):
    """
    Compute gradient of smooth max of autocorrelation.
    Uses log-sum-exp for numerical stability.
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Log-sum-exp for stability
    max_ac = ac.max()
    log_weights = beta * (ac - max_ac)
    log_weights -= log_weights.max()  # Additional stability
    weights = np.exp(log_weights)
    weights /= weights.sum()
    
    # Gradient: grad_m = sum_k weights[k] * (a[m+k] + a[m-k]) with bounds
    # This is a convolution of weights with a (circular-ish)
    # More precisely: grad_m = sum_k weights[k] * (a_{m+k}*I(m+k<n) + a_{m-k}*I(m>=k))
    #                    + sum_{k: m-k<0} weights[k] * a_{m-k+n}  (for full conv, but we use one-sided)
    
    # Efficient computation using FFT
    # The gradient is essentially a weighted autocorrelation
    # grad = IFFT(FFT(a) * FFT(weights_padded))
    # But weights is defined on lags 0..2n-2, so we need to be careful
    
    # Direct computation via correlation
    # For each lag k, the contribution to grad is weights[k] * (shifted a)
    # This is equivalent to: grad = sum_k weights[k] * (a shifted by k + a shifted by -k)
    # Which is: grad = weights * a (cross-correlation, symmetric)
    
    # Use FFT for efficiency
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(weights, fft_size)
    # The autocorrelation of a with kernel weights
    grad = np.fft.irfft(A * W.conjugate(), fft_size)[:n]
    
    # Actually, let me think more carefully.
    # ac[k] = sum_i a[i]*a[i+k] for k >= 0 (one-sided)
    # But the full convolution is: (a*a)[k] for k = 0..2n-2
    # (a*a)[k] = sum_i a[i]*a[k-i] where both indices are in [0, n-1]
    # For k = 0: sum_i a[i]^2
    # For k = 1: sum_i a[i]*a[i+1] (i from 0 to n-2)
    # For k = n-1: sum_i a[i]*a[i-(n-1)] -> a[n-1]*a[0]
    # For k = n: sum_i a[i]*a[i-n] -> 0 (out of bounds)
    # Actually for full conv of length-n vectors:
    # (a*a)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
    
    # d(a*a)[k] / da[m] = a[k-m] (if 0 <= k-m <= n-1) + a[m-k] (if 0 <= m-k <= n-1)
    # Wait, (a*a)[k] = sum_i a[i]*a[k-i]
    # d/da[m]: the term a[i]*a[k-i] gets a derivative when i=m (gives a[k-m]) or k-i=m (i.e., i=k-m, gives a[m])
    # So d(a*a)[k]/da[m] = a[k-m]*I(0<=k-m<=n-1) + a[m]*I(0<=k-m<=n-1)
    # Wait that's not right either. Let me be more careful.
    
    # (a*a)[k] = sum_{i} a[i] * a[k-i] where i ranges over valid indices (0 to n-1 for both i and k-i)
    # d/da[m] of this = sum over i where i=m: a[k-m] + sum over i where k-i=m (i.e., i=k-m): a[m]
    # = a[k-m] * I(0<=k-m<n) + a[m] * I(0<=k-m<n)
    # = a[k-m] + a[m] (when 0 <= k-m < n)
    # But wait, if m = k-m (i.e., k = 2m), we're double counting.
    # Actually no: the sum is over i, and we're differentiating with respect to a[m].
    # The term a[i]*a[k-i] contributes a[k-m] when i=m (since d(a[m])/da[m] = 1, giving a[k-m])
    # The term a[i]*a[k-i] contributes a[m] when k-i=m, i.e., i=k-m (since d(a[k-i])/da[m] = d(a[k-(k-m)])/da[m] = d(a[m])/da[m] = 1)
    # But when i=m AND k-i=m (i.e., k=2m), both contributions come from the same term a[m]*a[m],
    # whose derivative is 2*a[m]. So the formula a[k-m] + a[m] = 2*a[m] is correct in that case.
    
    # So: grad[m] = sum_k weights[k] * (a[k-m] + a[m]) * I(0<=k-m<n)
    #              = sum_k weights[k] * a[k-m] * I(0<=k-m<n) + a[m] * sum_k weights[k] * I(0<=k-m<n)
    
    # The first term is a cross-correlation of weights and a.
    # The second term is a[m] * (sum of weights[k] for k in range [m, m+n-1])
    
    # For efficiency, let's just do the full convolution approach:
    # grad = sum_k weights[k] * (a reversed and shifted by k)
    
    # Actually, the simplest correct approach:
    # grad[m] = sum_k W[k] * (a[(k-m) % ... ])
    # This is getting complicated. Let me just use the direct formula.
    
    # For each m, grad[m] = sum_k W[k] * (a[k-m if 0<=k-m<n else 0] + a[m-k if 0<=m-k<n else 0])
    # But this is O(n^2) which is too slow for large n.
    
    # Alternative: use the fact that the gradient of max_k ac[k] (smoothed) w.r.t. a
    # can be computed as: grad = 2 * (W * a) where * denotes circular convolution
    # Wait, that's for circular autocorrelation. For linear, it's a bit different.
    
    # Let me use a different approach: directly compute the gradient via FFT
    # The linear convolution (a*a)[k] for k=0..2n-2 can be seen as the
    # linear cross-correlation of a with itself.
    # The gradient w.r.t. a is: for each k, d(a*a)[k]/da = [a reversed, shifted] + [a, shifted]
    # This is equivalent to: the "correlation" of the gradient kernel with a.
    
    # Simplest efficient approach:
    # grad[m] = sum_k W[k] * (contrib from a at position m to lag k)
    # contrib(m, k) = a[k-m] if 0 <= k-m < n, else 0  (from i=m term)
    #                + a[m-k] if 0 <= m-k < n, else 0  (from i=k-m term)
    #                - a[m] if k == 2m (to avoid double counting? No, as discussed, it's correct)
    
    # Actually I realize the cleanest way:
    # Let's define the "autocorrelation operator" T: a -> ac where ac[k] = sum_i a[i]*a[k-i]
    # Then T is a quadratic map. Its derivative at a in direction v is:
    # (Da)(v)[k] = sum_i (v[i]*a[k-i] + a[i]*v[k-i])
    # So the gradient of sum_k W[k]*ac[k] w.r.t. a is:
    # grad[m] = sum_k W[k] * (sum_i delta[m,i]*a[k-i] + sum_i a[i]*delta[m,k-i])
    #         = sum_k W[k] * a[k-m] * I(0<=k-m<n) + sum_k W[k] * a[k-m] * I(0<=k-m<n)
    # Wait, this is the same thing twice? Let me redo.
    
    # Da(v)[k] = sum_i v[i]*a[k-i] + sum_i a[i]*v[k-i]
    # Set v = e_m (unit vector at position m):
    # Da(e_m)[k] = a[k-m]*I(0<=k-m<n) + sum_i a[i]*e_m[k-i]
    #            = a[k-m]*I(0<=k-m<n) + a[m-k]*I(0<=m-k<n) [taking i=m-k, so k-i=m, i.e., i=k-m... wait]
    
    # sum_i a[i]*v[k-i] = sum_i a[i]*e_m[k-i] = a[m-k] (when k-i=m, i.e., i=k-m, and a[k-m] is the value)
    # Wait: v[k-i] = e_m[k-i] = 1 if k-i=m, i.e., i=k-m.
    # So sum_i a[i]*v[k-i] = a[k-m] (when 0 <= k-m < n)
    
    # And sum_i v[i]*a[k-i] = v[m]*a[k-m] = a[k-m] (when 0 <= k-m < n)
    
    # So Da(e_m)[k] = a[k-m]*I(0<=k-m<n) + a[k-m]*I(0<=k-m<n) = 2*a[k-m]*I(0<=k-m<n)
    
    # Hmm, that gives grad[m] = sum_k W[k] * 2 * a[k-m] * I(0<=k-m<n)
    # = 2 * sum_{k: 0<=k-m<n} W[k] * a[k-m]
    # = 2 * sum_{j: 0<=j<n} W[m+j] * a[j]  (substituting j = k-m)
    
    # So grad[m] = 2 * sum_j W[m+j] * a[j]
    
    # This is a cross-correlation! grad = 2 * correlate(W, a) where correlate is:
    # correlate(W, a)[m] = sum_j W[m+j] * a[j]
    
    # This can be computed via FFT:
    # correlate(W, a) = ifft(fft(W_reversed) * conj(fft(a)))... 
    # Actually, let me think about this differently.
    # correlate(W, a)[m] = sum_j W[m+j] * a[j] = (W * reverse(a))[m] evaluated appropriately
    # Or: it's the cross-correlation of W and a.
    # In numpy: np.correlate(W, a, mode='full') gives us this.
    # But for large n, FFT is better.
    
    # correlate(W, a)[m] = sum_j W[m+j]*a[j]
    # Let b[j] = a[j], c[j] = W[j]
    # We want sum_j c[m+j]*b[j] for m = 0, 1, ..., 2n-2 (or just 0 to n-1)
    # = sum_j c[m+j]*b[j]
    # Let's reverse c: c_rev[j] = c[-j]... no, let's think in terms of convolution.
    # (c * b_rev)[m] = sum_j c[j] * b_rev[m-j] = sum_j c[j] * b[-(m-j)] = sum_j c[j] * b[j-m]... 
    
    # This is getting confusing. Let me just use the FFT approach directly.
    # correlate(W, a)[m] = sum_j W[m+j]*a[j]
    # = sum_j W[m+j]*a[j]
    # If I define W_padded with appropriate zeros, this is a cross-correlation.
    # In numpy/scipy: np.correlate(W, a, 'valid') or using FFT.
    
    # For FFT: 
    # correlate(x, y)[k] = sum_n x[n+k]*y[n]
    # = ifft(fft(x, N) * conj(fft(y, N)))  -- this gives circular correlation
    # For linear, we need N >= len(x) + len(y) - 1
    
    # So: grad = 2 * ifft(fft(W, N) * conj(fft(a, N)))[:n] where N is large enough
    
    # Let me verify: ifft(fft(W)*conj(fft(a)))[k] = sum_n W[n]*a[n-k] (circular)
    # Hmm, that's not quite what I want. Let me use a different formulation.
    
    # Actually, I'll just do it with a direct but efficient computation using the fact
    # that for each m, grad[m] = 2 * sum_j W[m+j] * a[j]
    # This is equivalent to: for each m, take the window W[m:m+n] and dot with a.
    # This is a sliding dot product, which can be done with FFT.
    
    # Define V of length 2n-1: V[m] = sum_j W[m+j]*a[j] for m=0,...,n-1 (we only need first n)
    # V is the cross-correlation of W (length 2n-1) and a (length n).
    # V[m] = sum_{j=0}^{n-1} W[m+j]*a[j]
    #       = sum_{j=0}^{n-1} W[m+j]*a[j]
    # Let W_full = W (length 2n-1), a_padded = [a, 0, ..., 0] (length 2n-1)
    # Then V[m] = (W_full * a_padded)[m] where * is cross-correlation
    # In terms of convolution: V[m] = sum_j W_full[m+j]*a_padded[j]
    # = sum_j W_full[m+j]*a_padded[j]
    # If we reverse W_full: W_rev[j] = W_full[-1-j] = W_full[2n-2-j]
    # Then (W_rev * a_padded)[m] = sum_j W_rev[j]*a_padded[m-j] = sum_j W_full[2n-2-j]*a_padded[m-j]
    # Let j' = 2n-2-j: = sum_{j'} W_full[j']*a_padded[m-2n+2+j']
    # This is getting too complicated. Let me just use scipy.signal.correlate or do it with numpy.
    
    # Simplest: use np.correlate
    # But for n=5000, this is O(n^2) = 25M operations, which might be OK per iteration
    # but too slow if we do many iterations.
    
    # Let me use the FFT approach:
    # V[m] = sum_j W[m+j]*a[j] for j=0..n-1
    # = sum_j W[m+j]*a[j]
    # Pad W to length >= 2n-1 (it already is) and a to same length
    # V = correlate(W, a, mode='full') gives V[k] = sum_n W[n+k]*a[n] for all valid k
    # We want V[0], V[1], ..., V[n-1]
    
    # Using FFT: correlate(W, a) = ifft(fft(W, N) * conj(fft(a, N))) where N >= len(W)+len(a)-1
    # Wait, I need to be careful about the definition.
    # numpy's correlate(x, y, mode) computes:
    # c[k] = sum_n conj(x[n+k]) * y[n]  (for real: sum_n x[n+k]*y[n])
    # So correlate(W, a)[k] = sum_n W[n+k]*a[n] -- this is exactly V[k]!
    
    # FFT: correlate(W, a) = ifft(conj(fft(W, N)) * fft(a, N))... 
    # Actually for real signals: correlate(W, a) = ifft(fft(W, N) * conj(fft(a, N)))
    # Let me verify with a small example...
    # Actually, the standard result is:
    # cross-correlation R_xy[k] = sum_n x[n+k]*y[n]
    # R_xy = ifft(FFT(X) * conj(FFT(Y)))  (with appropriate zero-padding)
    
    # So: V = ifft(FFT(W, N) * conj(FFT(a, N)))[:n]
    # where N >= (2n-1) + n - 1 = 3n-2... actually N >= len(W) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    
    # Hmm, that's a large FFT. Let me just use np.correlate for correctness and accept the O(n^2) cost.
    # For n=5000, that's about 25M multiply-adds, which is ~50ms in numpy.
    # For 1000 iterations, that's 50 seconds. Acceptable.
    
    # Actually, let me reconsider. The FFT approach with N = next_power_of_2(3*n) would work.
    # For n=5000, N = 16384. That's a 16K FFT, very fast.
    
    N = 1
    while N < 3 * n - 1:
        N *= 2
    
    Wf = np.fft.rfft(weights, N)
    Af = np.fft.rfft(a, N)
    corr = np.fft.irfft(Wf * np.conjugate(Af), N)[:n]
    
    return 2.0 * corr


def smooth_max_descent(a0, time_limit, beta_start=10.0, beta_end=100.0, lr0=0.5, verbose=True):
    """
    Mirror descent with smooth max of autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 2000
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 5000.0)
        
        # Compute gradient
        grad = smooth_max_gradient(a, beta)
        grad = np.maximum(grad, 0.0)  # Gradient should be non-negative for non-negative a
        
        # Mirror descent update (multiplicative weights)
        lr = lr0 / (1.0 + np.sqrt(t))
        log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Restart with perturbation
                a = norm_sum1(best_a + rng.normal(0, 0.02, n) * best_a)
                stall = 0
                lr0 *= 0.8
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothMax iter={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")
    
    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """
    Local coordinate descent: try moving mass from high-autocorrelation-contributing
    positions to low-contributing positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    stall = 0
    max_stall = 300
