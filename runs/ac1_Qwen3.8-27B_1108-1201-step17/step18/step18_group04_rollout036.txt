    AR = np.fft.rfft(a_rev, fft_size2)
    conv_da = np.fft.irfft(D * AR, fft_size2)
    g = conv_da[n-1:n-1+n]  # g_i = conv_da[n-1+i]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp annealing AMD (smooth max with temperature annealing)
2. Variance pre-optimization
3. Aggressive multi-n exploration
4. Fine-grained peak-shaving coordinate descent

import numpy as np
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_lse_grad(a, conv, tau):
    """Compute gradient of tau * log(sum exp(conv/tau)) w.r.t. a."""
    n = len(a)
    c_scaled = conv / tau
    c_max = c_scaled.max()
    log_w = c_scaled - c_max
    log_w = np.clip(log_w, -50, 0)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / w_sum
    
    # grad_i = 2 * tau * sum_k w_k * a_{k-i}
    # Use FFT for efficiency: this is the cross-correlation of w and a
    # g_i = sum_k w_k * a_{k-i}
    # = (w * a_rev)_{n-1+i} where a_rev[j] = a[n-1-j]
    a_rev = a[::-1]
    total_len = len(conv) + n - 1
    fft_sz = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_sz)
    AR = np.fft.rfft(a_rev, fft_sz)
    g = np.fft.irfft(W * AR, fft_sz)
    g = g[n-1:n-1+n]
    
    return 2.0 * tau * g


def lse_anneal(a0, time_limit, tau_start=1.0, tau_end=0.001, seed=42, verbose=False):
    """AMD with log-sum-exp annealing."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    total_iters = 20000
    c0 = 5.0
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = (t - 1) / max(total_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute gradient
        conv_y = fast_conv(y)
        g = compute_lse_grad(y, conv_y, tau)
        g = np.maximum(g, 0)  # Gradient should be non-negative for non-negative a
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
    
    return best_x, best_m


def variance_preopt(a0, time_limit, seed=42):

    n = len(a0)
    x = norm_sum1(a0)

        g = np.fft.irfft(D * AR, fft_sz)
        g = g[n-1:n-1+n]
        grad = 4.0 * g
        
        # Gradient step with projection
        lr_t = lr / (1 + 0.01 * t)
        x_new = x - lr_t * grad
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-15:

def peak_shaving(a0, time_limit, seed=42, verbose=False):
    """Aggressive peak-shaving via coordinate moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    fracs = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        conv = fast_conv(a)
        
        # Find top peaks
        n_peaks = min(10, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_peaks]
        
        # For the main peak, find which source elements contribute most
        pk = int(top_peaks[0])
        lo = max(0, pk - n + 1)
        hi = min(n - 1, pk)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[pk - idx]
        
        # Top contributing elements
        n_top = max(5, min(15, n // 300))
        top_src = idx[np.argsort(contrib)[::-1][:n_top]]
        
        # Compute cost of moving to each position (sum of contributions to all top peaks)
        costs = np.zeros(n)
        for k2 in top_peaks[:5]:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]
        
        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            si = int(src)
            if a[si] < 1e-12:
                continue
            
            # Find best destination: low cost positions
            candidates = np.argsort(costs + 1e-10 * a)[:max(10, n // 50)]
            for dst_i in candidates:

                if di == si:
                    continue
                for frac in fracs:
                    amt = a[si] * frac
                    if amt < 1e-14:
                        break
                    a_trial = a.copy()
                    a_trial[si] -= amt
                    a_trial[di] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_t = fast_conv(a_trial).max()
                    if M_t < best_m - 1e-14:
                        a = a_trial
                        best_m = M_t
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 5 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:

                        best_a = a.copy()

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp annealing AMD (smooth max approximation with decreasing temperature)
2. Variance-based pre-optimization for landscape flattening
3. Aggressive peak-shaving coordinate descent
4. Multi-scale n exploration with interpolation

import numpy as np
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def cross_corr_fft(w, a):
    """Compute g_i = sum_k w_k * a_{k-i} for i=0..n-1 using FFT.
    w has length 2n-1, a has length n."""
    n = len(a)
    a_rev = a[::-1]
    total_len = len(w) + n - 1
    fft_sz = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_sz)
    AR = np.fft.rfft(a_rev, fft_sz)
    g = np.fft.irfft(W * AR, fft_sz)
    return g[n - 1:n - 1 + n]


def lse_anneal(a0, time_limit, tau_start=0.5, tau_end=0.001, seed=42, c0=5.0, verbose=False):
    """AMD with log-sum-exp annealing for smooth max optimization."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    total_iters = 30000
    stall_count = 0
    restart_count = 0
    restart_every = 5000

    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = (t - 1) / max(total_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        conv_y = fast_conv(y)
        c_scaled = conv_y / tau
        c_max = c_scaled.max()
        log_w = c_scaled - c_max
        log_w = np.clip(log_w, -60, 0)
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv_y)
            w[int(np.argmax(conv_y))] = 1.0
        else:
            w = w / w_sum

        g = cross_corr_fft(w, y)
        g = 2.0 * tau * g
        g = np.maximum(g, 0.0)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.04, 0.008 + 0.004 * min(restart_count, 12))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            c0 = 5.0 * (1.0 + 0.02 * min(restart_count, 10))
            restart_every = max(3000, restart_every + 800)

    return best_x, best_m


def single_peak_amd(a0, time_limit, seed=42, c0=4.0):
    """AMD using exact subgradient at the single maximum peak."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()

    t0 = time.monotonic()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        conv_y = fast_conv(y)
        k_star = int(np.argmax(conv_y))

        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        g = np.zeros(n)
        g[idx] = 2.0 * y[k_star - idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()

    return best_x, best_m


def variance_preopt(a0, time_limit, seed=42):
    """Pre-optimize using variance of autocorrelation as smooth objective."""
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    lr = 0.5

    for t in range(1, 8000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv(x)
        mu = conv.mean()
        d = conv - mu

        a_rev = x[::-1]
        total_len = len(conv) + n - 1
        fft_sz = 1 << (total_len - 1).bit_length()
        D = np.fft.rfft(d, fft_sz)
        AR = np.fft.rfft(a_rev, fft_sz)
        g = np.fft.irfft(D * AR, fft_sz)
        g = g[n - 1:n - 1 + n]
        grad = 4.0 * g

        lr_t = lr / (1.0 + 0.02 * t)
        x_new = x - lr_t * grad
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / s
        x = x_new

        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()

        if t % 500 == 0:
            pert = rng.normal(0, 0.01, n) * x
            x_p = x + pert
            x_p = np.maximum(x_p, 0.0)
            sp = x_p.sum()
            if sp > 1e-15:
                x_p = x_p / sp
                Mp = fast_conv(x_p).max()
                if Mp < best_m:
                    best_m = Mp
                    best_x = x_p.copy()
                    x = x_p.copy()

    return best_x, best_m


def peak_shaving(a0, time_limit, seed=42, verbose=False):
    """Aggressive peak-shaving via coordinate moves with quadratic approximation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    fracs = [0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.4]

    while time.monotonic() - t0 < time_limit and stall < 150:
        conv = fast_conv(a)
        n_peaks = min(12, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_peaks]

        pk = int(top_peaks[0])
        lo = max(0, pk - n + 1)
        hi = min(n - 1, pk)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[pk - idx]

        n_top = max(6, min(20, n // 250))
        top_src = idx[np.argsort(contrib)[::-1][:n_top]]

        costs = np.zeros(n)
        for k2 in top_peaks[:6]:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            si = int(src)
            if a[si] < 1e-12:
                continue

            candidates = np.argsort(costs + 1e-10 * a)[:max(12, n // 40)]
            for dst_i in candidates:
                di = int(dst_i)
                if di == si:
                    continue
                for frac in fracs:
                    amt = a[si] * frac
                    if amt < 1e-14:
                        break
                    a_trial = a.copy()
                    a_trial[si] -= amt
                    a_trial[di] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_t = fast_conv(a_trial).max()
                    if M_t < best_m - 1e-14:
                        a = a_trial
                        best_m = M_t
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 3 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_smooth_bump(n, center=0.5, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return norm_sum1(a)


def make_cosine(n, mode=1):

def make_two_hump(n, p1=0.3, p2=0.7, w=0.15):

    a = np.exp(-((x - p1) / w) ** 2) + 0.8 * np.exp(-((x - p2) / (w * 1.2)) ** 2)
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.1 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_power_law(n, alpha=0.3):
    x = np.arange(n, dtype=np.float64)
    a = (n - x + 1) ** (-alpha)
    return norm_sum1(a)


def make_bell_symmetric(n, sigma_frac=0.3):

    a = np.exp(-x ** 2 / (2 * sigma_frac ** 2))
    return norm_sum1(a)


def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 4500
        x = np.linspace(0, 1, n_base)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Default init: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    n0 = len(_best_tracker['seq'])
    print(f"Starting: n={n0}, score={_best_tracker['score']:.6f}")

    # Phase 1: Variance pre-opt + LSE annealing on diverse inits
    print("\n=== Phase 1: Variance + LSE Annealing ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.42, 700.0)

    if t_phase1 > 30:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_smooth_bump(n0, 0.5, 0.22)); labels.append("bump22")
        inits.append(make_smooth_bump(n0, 0.5, 0.30)); labels.append("bump30")
        inits.append(make_smooth_bump(n0, 0.5, 0.40)); labels.append("bump40")
        inits.append(make_cosine(n0, 1)); labels.append("cos1")
        inits.append(make_cosine(n0, 2)); labels.append("cos2")
        inits.append(make_sinc(n0, 4.0)); labels.append("sinc4")
        inits.append(make_sinc(n0, 8.0)); labels.append("sinc8")
        inits.append(make_sinc(n0, 14.0)); labels.append("sinc14")
        inits.append(make_two_hump(n0, 0.25, 0.75, 0.12)); labels.append("twohump")
        inits.append(make_two_hump(n0, 0.35, 0.65, 0.18)); labels.append("twohump2")
        inits.append(make_random_smooth(n0, seed=7, n_freq=6)); labels.append("rnd1")
        inits.append(make_random_smooth(n0, seed=42, n_freq=12)); labels.append("rnd2")
        inits.append(make_random_smooth(n0, seed=123, n_freq=20)); labels.append("rnd3")
        inits.append(make_power_law(n0, 0.2)); labels.append("powlaw02")
        inits.append(make_power_law(n0, 0.5)); labels.append("powlaw05")
        inits.append(make_bell_symmetric(n0, 0.25)); labels.append("bell25")
        inits.append(make_bell_symmetric(n0, 0.35)); labels.append("bell35")

        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 20)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(12.0, t_phase1 / n_inits)

        for idx in range(n_inits):
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget - 80:
                break
            t_avail = min(t_per, time_budget - 80 - elapsed)
            if t_avail < 10:
                break
            lbl = unique_labels[idx]

            t_var = min(t_avail * 0.12, 25.0)
            if t_var > 4:
                a_var, _ = variance_preopt(unique_inits[idx], t_var, seed=1000 + idx)
            else:
                a_var = unique_inits[idx]

            t_lse = t_avail - t_var - 3
            if t_lse > 8:
                a_out, _ = lse_anneal(a_var, t_lse, tau_start=0.4, tau_end=0.001,
                                      seed=2000 + idx, c0=5.0)
                try_update(a_out, f"lse_{lbl}")

    # Phase 2: Peak shaving on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Peak Shaving ===")
    t2 = min(remaining * 0.14, 220.0)
    if t2 > 10:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = peak_shaving(a_in, t2, seed=42)
        try_update(a_out, "peak1")

    # Phase 3: LSE refine with lower temperature
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE Refine ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 15:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = lse_anneal(a_in, t3, tau_start=0.15, tau_end=0.0005,
                              seed=9999, c0=4.0)
        try_update(a_out, "lse_refine")

    # Phase 4: Multi-n exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 4: Multi-n (best n={n_best}) ===")

    candidate_ns = []
    for delta in [-1500, -1000, -500, -200, 200, 500, 1000, 1500, 2000, 3000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 12000:
            candidate_ns.append(new_n)

    candidate_ns.sort(key=lambda x: abs(x - n_best))

    for new_n in candidate_ns[:6]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 70:
            break
        print(f"  n={new_n}")
        a_interp = interpolate_seq(_best_tracker['seq'], new_n)
        try_update(a_interp, f"intp{new_n}")

        t4a = min(remaining * 0.25, 160.0)
        if t4a > 12:
            a_out, _ = lse_anneal(a_interp, t4a, tau_start=0.3, tau_end=0.001, seed=7777)
            try_update(a_out, f"lse_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t4b = min(remaining * 0.15, 90.0)
            if t4b > 8:
                a_out, _ = peak_shaving(a_out, t4b, seed=5555)
                try_update(a_out, f"pk_n{new_n}")

    # Phase 5: Final polish - single peak + peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(_best_tracker['seq'])
    print(f"\n=== Phase 5: Final (n={n_final}) ===")

    if remaining > 80:
        t5a = min(remaining * 0.30, 180.0)
        if t5a > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = single_peak_amd(a_in, max(t5a * 0.5, 10), seed=2024, c0=3.0)
            try_update(a_out, "sp_amd")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t5a2 = min(remaining * 0.30, 120.0)
                if t5a2 > 8:
                    a_out, _ = lse_anneal(a_out, t5a2, tau_start=0.08, tau_end=0.0003,
                                          seed=3000, c0=4.0)
                    try_update(a_out, "lse_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t5b = min(remaining * 0.55, 180.0)
        if t5b > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = peak_shaving(a_in, t5b, seed=31415)
            try_update(a_out, "pk_final")

    # Final verification
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4500
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
