def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_objective(a, n):
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (a.sum() ** 2)

def build_constraint_rows(y, constraint_list, n):
    """Build constraint coefficient rows for the LP: (A^T g)_k = sum_i y[k-i]*g[i] <= M"""
    K = len(constraint_list)
    rows = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            rows[jj, idx] = y[k_int - idx]
    return rows


def lp_fw_away(a0, time_limit, solver, initial_K=400, max_K=4000,
               add_per_iter=30, verbose=True):
    """LP-based Frank-Wolfe with away-step, aggressive line search, adaptive active set."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:min(initial_K, 2 * n - 1)])

    no_improve = 0
    max_no_improve = 1500
    y_old = a.copy()  # for away step

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = build_constraint_rows(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(constraint_list[i]) for i in range(min(initial_K, len(constraint_list))))
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]

                active_set.add(int(k))

        for alpha in [default_alpha, default_alpha * 0.5, default_alpha * 1.5,

            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            candidates.append(trial)

        # Away step: move away from y_old
        if t > 10:
            away_dir = y - y_old
            for gamma in [0.1, 0.2, 0.3]:
                trial = y + gamma * away_dir
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                candidates.append(trial)

        best_trial = y.copy()
        for trial in candidates:
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial = trial

        # Finer line search around best
        if best_trial_M < float('inf'):
            # Determine which alpha gave the best
            for alpha_fine in np.linspace(0.01, 1.0, 10):
                trial = (1.0 - alpha_fine) * y + alpha_fine * g_dir
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                M_trial = fast_autocorr(trial).max()
                if M_trial < best_trial_M:
                    best_trial_M = M_trial
                    best_trial = trial

        if t == 1:
            y = g_dir.copy()
        else:
            y = best_trial

        y_old = (1.0 - 0.5) * y_old + 0.5 * y  # slow tracking for away step

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW-A iter=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW-A done (" + str(elapsed) + "s): J=" + repr(J))
    return best_a, best_m


def peak_targeting_descent(a0, time_limit, verbose=True):
    """Directly target reduction of the max convolution value via mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    max_stall = 300

        ac = fast_autocorr(a)

        # Find top contributing pairs for k_star

        i_end = min(n - 1, k_star)

        contrib = a[idx] * a[j_idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            break

        # Sort by contribution
        order = np.argsort(contrib)[::-1]
        top_count = min(20, len(order))

        for oi in range(top_count):
            if time.monotonic() - t0 > time_limit:
                break
            ii = idx[order[oi]]
            jj = j_idx[order[oi]]

            if ii == jj:

            # Try moving mass from the larger of a[ii], a[jj]
            src = ii if a[ii] >= a[jj] else jj
            dst_candidate = None

            # Find a good destination: a position that doesn't contribute much to top peaks

            impact = np.zeros(n)

                    idx2 = np.arange(s2, e2 + 1)
                    j2 = k2i - idx2
                    impact[idx2] += a[j2]
                    impact[j2] += a[idx2]

            mask[src] = False
            for k2 in top_k[:5]:

                candidates = candidates[candidates != src]

            if len(candidates) == 0:
                continue

            cand_impact = impact[candidates]
            dst = int(candidates[np.argmin(cand_impact)])

            # Try different fractions

                if s_t < 1e-15:

                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < M_curr - 1e-16:
                    a = a_trial
                    M_curr = M_trial
                    improved_this_round = True
                    found = True
                    if M_trial < best_m:
                        best_m = M_trial
                        best_a = a.copy()
                    break

            if not found:
                # Try random destination

                    dst_r = int(rng.integers(n))

                        continue
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst_r] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M_curr - 1e-16:
                        a = a_trial
                        M_curr = M_trial
                        improved_this_round = True
                        if M_trial < best_m:
                            best_m = M_trial
                            best_a = a.copy()
                        break

        if not improved_this_round:
            stall += 1
            # Perturb to escape local minimum

                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t

        if verbose and (t % 50 == 0 or stall % 100 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakDesc (" + str(elapsed) + "s): J=" + repr(J) + ", stall=" + str(stall))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    PeakDesc done (" + str(elapsed) + "s): J=" + repr(J))
    return best_a, best_m


def mirror_descent_soft(a0, time_limit, top_k=80, beta=5.0, c0=4.0,
                        restart_every=3000, verbose=True):
    """Mirror descent with soft subgradient, Nesterov acceleration, and adaptive restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0

    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov step
        beta_t = (np.sqrt(1.0 + 4.0 * (t - 1) ** 2) - 1.0) / 2.0
        beta_t = min(beta_t, 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            restart_count += 1
            noise_level = 0.02 / (1.0 + 0.03 * restart_count)
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 150)
            cur_beta = max(cur_beta - 0.2, 2.5)
            gamma = max(gamma * 0.93, 1.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    MD restart t=" + str(t) + " (" + str(elapsed) + "s): J=" + repr(J))

    return best_a, best_m


def mass_redistribute_local(a0, time_limit, verbose=True):
    """Direct local search: for each peak position, try to reduce it by redistributing mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)

    for iteration in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        # Get top 5 peaks
        top_peaks = np.argsort(ac)[::-1][:5]

        for k_peak in top_peaks:
            k_int = int(k_peak)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            peak_val = ac[k_int]

            # For the top contributing pair, try to reduce it
            order = np.argsort(contrib)[::-1]
            for oi in range(min(5, len(order))):
                ii = int(idx[order[oi]])
                jj = int(j_idx[order[oi]])
                if ii == jj:
                    continue

                src = ii if a[ii] >= a[jj] else jj
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Try moving to several candidate destinations
                # Choose destinations that minimize impact on top peaks
                trial_dests = []
                # Random candidates
                for _ in range(5):
                    d = int(rng.integers(n))
                    if d != src:
                        trial_dests.append(d)
                # Also try positions far from all peak contributors
                safe_mask = np.ones(n, dtype=bool)
                for k2 in top_peaks[:3]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    safe_mask[max(0, s2 - 2):min(n, e2 + 3)] = False
                safe_mask[src] = False
                safe_pos = np.where(safe_mask)[0]
                if len(safe_pos) > 0:
                    trial_dests.extend([int(safe_pos[idx2]) for idx2 in rng.choice(len(safe_pos), min(3, len(safe_pos)), replace=False)])

                for dst in trial_dests:
                    if dst == src:
                        continue
                    for frac in [0.003, 0.007, 0.015, 0.03]:
                        amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            break
                    else:
                        continue
                    break

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    MassRedist done (" + str(elapsed) + "s): J=" + repr(J))
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        # Non-symmetric initial guess: front-loaded
        i_arr = np.arange(n_base, dtype=np.float64)
        best_seq = norm_sum1(np.exp(-0.5 * x) + 0.3 * np.exp(-3.0 * (x - 0.7) ** 2 / 0.04))
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: LP-FW away-step (main workhorse, 45% of time)
    print("\n=== Phase 1: LP-FW Away (n=" + str(n0) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 60:
        a_out, _ = lp_fw_away(norm_sum1(best_seq), t1, solver,
                              initial_K=500, max_K=5000,
                              add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_away_n" + str(n0), t_global)

    # Phase 2: Peak-targeting descent (20% of time)
    print("\n=== Phase 2: Peak Desktop (n=" + str(len(best_seq)) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, _ = peak_targeting_descent(norm_sum1(best_seq), t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakdesc_n" + str(len(best_seq)), t_global)

    # Phase 3: LP-FW round 2 (15% of time)
    print("\n=== Phase 3: LP-FW Round 2 (n=" + str(len(best_seq)) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30 and len(best_seq) <= 7000:
        a_out, _ = lp_fw_away(norm_sum1(best_seq), t3, solver,
                              initial_K=600, max_K=5000,
                              add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_r2_n" + str(len(best_seq)), t_global)

    # Phase 4: Mirror descent (10% of time)
    print("\n=== Phase 4: Mirror Descent (n=" + str(len(best_seq)) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = mirror_descent_soft(norm_sum1(best_seq), t4, top_k=80, beta=5.0,
                                       c0=4.0, restart_every=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "md_n" + str(len(best_seq)), t_global)

    # Phase 5: Mass redistribution local search (5% of time)
    print("\n=== Phase 5: Mass Redistribute (n=" + str(len(best_seq)) + ") ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.05, 80.0)
    if t5 > 15:
        a_out, _ = mass_redistribute_local(norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "masred_n" + str(len(best_seq)), t_global)

    # Phase 6: Try different lengths
    print("\n=== Phase 6: Multi-length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    length_options = []
    for delta in [-200, -100, 100, 200, 400]:
        nl = n_best + delta
        if 5000 <= nl <= 7500 and nl not in length_options:
            length_options.append(nl)

    for new_n in length_options[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print("  Trying n=" + str(new_n))
        a_interp = interpolate_sequence(best_seq, new_n)

        t_lp = min(remaining * 0.4, 120.0)
        if t_lp > 20:
            a_out, _ = lp_fw_away(a_interp, t_lp, solver,
                                  initial_K=400, max_K=4000,
                                  add_per_iter=30, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

        t_pk = min(remaining * 0.3, 60.0)
        if t_pk > 15:
            a_out, _ = peak_targeting_descent(norm_sum1(best_seq), t_pk, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_n" + str(len(best_seq)), t_global)

    # Phase 7: Final polish with LP-FW
    print("\n=== Phase 7: Final LP-FW Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and len(best_seq) <= 7000:
        t7 = min(remaining * 0.7, 150.0)
        if t7 > 20:
            a_out, _ = lp_fw_away(norm_sum1(best_seq), t7, solver,
                                  initial_K=500, max_K=5000,
                                  add_per_iter=40, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 8: Final peak descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.8, 100.0)
        if t8 > 10:
            a_out, _ = peak_targeting_descent(norm_sum1(best_seq), t8, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(elapsed) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.exp(-2.0 * x) + 0.5 * np.exp(-5.0 * (x - 0.6) ** 2 / 0.03))
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
