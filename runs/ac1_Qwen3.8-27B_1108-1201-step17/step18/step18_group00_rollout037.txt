```python
def proj_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() <= 1 and (v >= 0).all():
        return v / v.sum() if v.sum() > 0 else np.ones(n)/n
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n+1) * ... )
    # Standard algorithm
    w = np.sort(v)[::-1]
    cw = np.cumsum(w)
    ind = np.arange(1, n+1)
    cond = w - (cw - 1) / ind > 0
    rho = ind[cond][-1]
    theta = (cw[cond][rho-1] - 1) / rho
    return np.maximum(v - theta, 0)
```

Let me also think about the "peak-peeling" algorithm more carefully:

**Peak-peeling algorithm:**
1. Compute `b = conv(a,a)`, find `k* = argmax(b)`
2. The peak is at `b[k*] = sum_i a[i] * a[k*-i]`
3. The "top contributors" are the pairs `(i, k*-i)` with the largest `a[i]*a[k*-i]`
4. To reduce the peak, we want to reduce these products
5. Strategy: pick the top contributor `(i*, k*-i*)` and move a small amount of mass from `a[i*]` to a position `j` that doesn't contribute to `b[k*]`
6. Check if the max autocorrelation decreased; if so, accept; if not, try another move

The key is choosing the destination `j` well. Good candidates:
- Positions that are "far" from the peak contributors
- Positions with small current value (so adding mass doesn't create a new peak)
- Position `j` should minimize the increase in other autocorrelation terms

A simple heuristic: `j` should be a position where `sum_k a[|k-j|]` is small (i.e., the "autocorrelation footprint" of position `j` is small).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse multi-length initialization (parametric + random)
2. Log-sum-exp projected gradient descent for fast initial convergence
3. LP-based Frank-Wolfe with line search and adaptive constraint management
4. Peak-peeling local search targeting the max autocorrelation lag
5. Coordinate descent polish
"""

import numpy as np
import time
import cvxpy as cp

_SOLVER = None

def _get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cw = np.cumsum(w)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = w - (cw - 1.0) / ind > 0
    if not cond.any():
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = (cw[cond][int(rho) - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def obj_max_conv(a):
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max()


def compute_lse_grad(a, beta):
    b = fast_autocorr(a)
    b_scaled = beta * b
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / exp_b.sum()
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, b.max()


def lse_gradient_descent(a0, time_limit, beta_init=5.0, beta_final=200.0,
                         verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    best_obj = obj_max_conv(a)
    eta = 0.05
    n_iter = 0

    for phase in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        beta = beta_init * (beta_final / beta_init) ** (phase / 19.0)
        phase_iters = 0
        for t in range(1, 50000):
            if time.monotonic() - t0 > time_limit:
                break
            grad, b_max = compute_lse_grad(a, beta)
            step = eta / (1.0 + 0.1 * t)
            a_new = a - step * grad
            a_new = proj_simplex(a_new)
            cur_m = fast_autocorr(a_new).max()
            if cur_m < best_m - 1e-14:
                best_m = cur_m
                best_a = a_new.copy()
                best_obj = obj_max_conv(a_new)
            a = a_new
            phase_iters += 1
            n_iter += 1
        if verbose:
            elapsed = time.monotonic() - t0
            print("    LSE-GD phase %d (beta=%.1f, %d iters, %.1fs): obj=%.6f" %
                  (phase, beta, phase_iters, elapsed, best_obj))
    return best_a, best_m


def _build_lp(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=400, max_K=3000,
          add_per_iter=30, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:min(initial_K, 2 * n - 1)])

    no_improve = 0
    max_no_improve = 1500
    y_momentum = None

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = _build_lp(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(constraint_list[i])
                                 for i in range(min(initial_K, len(constraint_list))))
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, default_alpha * 0.5, default_alpha * 1.5,
                  1.0 / (t + 1), min(1.0, 2.0 / (t + 1))]
        alphas = list(set(max(0.01, min(1.0, al)) for al in alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=%d (%.1fs): obj=%.6f, |S|=%d" %
                  (t, elapsed, J, len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (%.1fs): obj=%.6f" % (elapsed, J))
    return best_a, best_m


def peak_peel(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        k_star = int(np.argmax(ac))
        b_peak = ac[k_star]

        js = range(max(0, k_star - n + 1), min(n, k_star + 1))
        products = np.array([a[j] * a[k_star - j] for j in js])
        if products.max() < 1e-20:
            break

        js_arr = np.array(list(js))
        top_src = int(js_arr[np.argmax(products)])
        src_val = a[top_src]
        if src_val < 1e-12:
            break

        scores = np.zeros(n)
        top_k_list = np.argsort(ac)[::-1][:20]
        for k2 in top_k_list[:10]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            if s2 <= e2:
                idx2 = np.arange(s2, e2 + 1)
                scores[idx2] += a[k2i - idx2]
                scores[k2i - idx2] += a[idx2]

        in_peak = np.zeros(n, dtype=bool)
        for k2 in top_k_list[:10]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            in_peak[s2:e2 + 1] = True

        n_trials = 5
        improved = False
        for trial in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            if trial == 0:
                dst = int(candidates[np.argmin(cand_scores)])
            else:
                dst = int(candidates[rng.integers(len(candidates))])
            if dst == top_src:
                continue

            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[top_src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 50:
                j = int(rng.integers(n))
                k = int(rng.integers(n))
                if j != k and a[j] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[j] * frac
                    a[j] -= amt
                    a[k] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and (stall % 100 == 0 or improved):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            if stall % 100 == 0:
                print("    PeakPeel (%.1fs): obj=%.6f, stall=%d" % (elapsed, J, stall))

    return best_a, best_m

                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 30:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (%.1fs): obj=%.6f, stall=%d" % (elapsed, J, stall))

    return best_a, best_m


def diverse_init(n, existing_seq=None):
    candidates = []
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1((i_arr + 1.0) ** p))
        candidates.append(norm_sum1((n - i_arr) ** p))

    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(norm_sum1(a))

    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3]:
            candidates.append(norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))))

    for c1 in [0.1, 0.2, 0.3]:
        for c2 in [0.7, 0.8, 0.9]:
            for sigma in [0.05, 0.1, 0.15]:
                a = (np.exp(-((x - c1) ** 2) / (2 * sigma * sigma)) +
                     np.exp(-((x - c2) ** 2) / (2 * sigma * sigma)))
                candidates.append(norm_sum1(a))

    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(norm_sum1(x ** p + (1.0 - x) ** p))

    for freq in [1, 2, 3, 5]:
        for ph in [0.0, 0.5, 1.0]:
            a = np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph * np.pi)) / 2.0, 0.01)
            candidates.append(norm_sum1(a))

    for rate in [0.5, 1, 2, 5, 10]:
        candidates.append(norm_sum1(np.exp(-rate * x) + np.exp(rate * (x - 1.0))))

    for s in [5, 10, 20, 50]:
        for c in [0.3, 0.5, 0.7]:
            candidates.append(norm_sum1(1.0 / (1.0 + np.exp(-s * (x - c)))))

    rng = np.random.default_rng(7)
    for trial in range(30):
        p = rng.uniform(0.3, 3.0)
        candidates.append(norm_sum1(rng.random(n) ** p))

    for trial in range(20):
        a = np.zeros(n)
        n_peaks = rng.integers(2, 6)
        for _ in range(n_peaks):
            c = rng.uniform(0, 1)
            sigma = rng.uniform(0.02, 0.2)
            h = rng.uniform(0.5, 1.0)
            a += h * np.exp(-((x - c) ** 2) / (2 * sigma * sigma))
        candidates.append(norm_sum1(a))

    if existing_seq is not None and len(existing_seq) == n:
        candidates.append(norm_sum1(existing_seq))

    best_cand = None
    best_val = float('inf')
    for c in candidates:
        v = obj_max_conv(c)
        if v < best_val:
            best_val = v
            best_cand = c
    return best_cand, best_val


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = _get_solver()
    print("Total time budget: %.1fs" % time_budget)

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=%d, score=%.6f" % (len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))

    n0 = len(best_seq)
    print("Starting n=%d, score=%.6f" % (n0, best_score))

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sum1(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print("*** NEW BEST [%s]: %.6f (n=%d, t=%.0fs)" % (label, sc, len(a), elapsed))

    # Phase 1: Multi-length initialization
    print("\n=== Phase 1: Multi-length init ===")
    lengths_to_try = sorted(set([n0, n0 - 1000, n0 + 1000, 4000, 5000, 6000, 7000, 8000]))
    lengths_to_try = [l for l in lengths_to_try if 1000 <= l <= 10000]

    best_init_all = {}
    for n_try in lengths_to_try[:5]:
        if time.monotonic() - t_global > time_budget * 0.05:
            break
        a_init, v_init = diverse_init(n_try, parent)
        best_init_all[n_try] = (a_init, v_init)
        print("  n=%d: init obj=%.6f" % (n_try, v_init))

    # Phase 2: LSE-GD on best-length sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LSE-GD (n=%d, %.0fs) ===" % (n_best, remaining * 0.15))
    t2 = min(remaining * 0.15, 150.0)
    if t2 > 20:
        a_out, _ = lse_gradient_descent(norm_sum1(best_seq), t2,
                                        beta_init=5.0, beta_final=300.0, verbose=True)
        _try_update(a_out, "lse_n%d" % n_best)

    # Also try LSE-GD on other lengths
    for n_try, (a_init, v_init) in best_init_all.items():
        if n_try == len(best_seq):
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60 or v_init > obj_max_conv(best_seq) * 1.1:
            continue
        t2b = min(remaining * 0.1, 80.0)
        if t2b > 15:
            a_out, _ = lse_gradient_descent(a_init, t2b,
                                            beta_init=5.0, beta_final=200.0, verbose=False)
            _try_update(a_out, "lse_n%d" % n_try)

    # Phase 3: LP-FW main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW main (n=%d, %.0fs) ===" % (n_best, remaining * 0.35))
    t3 = min(remaining * 0.35, 500.0)
    if t3 > 30 and n_best <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=500, max_K=3000, add_per_iter=30, verbose=True)
        _try_update(a_out, "lp_main_n%d" % n_best)

    # Phase 4: Peak-peeling
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak-peel (n=%d, %.0fs) ===" % (n_best, remaining * 0.2))
    t4 = min(remaining * 0.2, 250.0)
    if t4 > 20:
        a_out, _ = peak_peel(norm_sum1(best_seq), t4, verbose=True)
        _try_update(a_out, "peel_n%d" % n_best)

    # Phase 5: LP-FW second pass (after peeling)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LP-FW pass 2 (n=%d, %.0fs) ===" % (n_best, remaining * 0.15))
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30 and n_best <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t5, solver,
                         initial_K=600, max_K=3500, add_per_iter=35, verbose=True)
        _try_update(a_out, "lp2_n%d" % n_best)

    # Phase 6: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Coord descent (n=%d, %.0fs) ===" % (n_best, remaining * 0.12))
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 15:
        a_out, _ = coord_descent(norm_sum1(best_seq), t6, verbose=True)
        _try_update(a_out, "coord_n%d" % n_best)

    # Phase 7: Try a different length with full pipeline
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(best_seq)
    alt_lengths = [l for l in [n_current + 500, n_current + 1000, n_current - 500]
                   if 1000 <= l <= 10000 and l != n_current]

    for alt_n in alt_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n=== Phase 7: Alt length n=%d ===" % alt_n)
        x = np.linspace(0, 1, alt_n)
        old_x = np.linspace(0, 1, len(best_seq))
        a_alt = np.interp(x, old_x, best_seq)
        a_alt = norm_sum1(a_alt)

        t7a = min(remaining * 0.3, 120.0)
        if t7a > 20:
            a_out, _ = lse_gradient_descent(a_alt, t7a, beta_init=5.0, beta_final=200.0,
                                            verbose=False)
            _try_update(a_out, "lse_alt_n%d" % alt_n)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and alt_n <= 9000:
            t7b = min(remaining * 0.4, 150.0)
            if t7b > 20:
                a_out, _ = lp_fw(norm_sum1(best_seq if len(best_seq) != alt_n else a_alt),
                                 t7b, solver, initial_K=400, max_K=2500,
                                 add_per_iter=25, verbose=True)
                if len(a_out) == alt_n:
                    _try_update(a_out, "lp_alt_n%d" % alt_n)

    # Phase 8: Final peak-peel + coord
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final polish (n=%d, %.0fs) ===" % (n_best, remaining))
    if remaining > 30:
        t8a = min(remaining * 0.5, 120.0)
        if t8a > 15:
            a_out, _ = peak_peel(norm_sum1(best_seq), t8a, verbose=True)
            _try_update(a_out, "peel_final_n%d" % n_best)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.6, 100.0)
            if t8b > 10:
                a_out, _ = coord_descent(norm_sum1(best_seq), t8b, verbose=True)
                _try_update(a_out, "coord_final_n%d" % n_best)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.0fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=%.6f" % best_score)

    return [float(x) for x in best_seq]
```
