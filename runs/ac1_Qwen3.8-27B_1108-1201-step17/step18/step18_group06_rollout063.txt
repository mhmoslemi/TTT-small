import numpy as np
import time

def fast_full_conv(a, b):
    """Full convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def smooth_max_grad(a, tau):
    """Compute smooth max of a*a autocorrelation and gradient w.r.t. a.
    Assumes a is normalized (sum = 1). Returns (value, grad)."""
    n = len(a)
    c = fast_full_conv(a, a)  # length 2n-1
    
    # Smooth max computation (numerically stable)
    tc = tau * c
    tc_max = tc.max()
    log_sum_exp = tc_max + np.log(np.exp(tc - tc_max).sum())
    val = log_sum_exp / tau
    
    # Softmax weights
    p = np.exp(tc - tc_max)
    p = p / p.sum()
    
    # Cross-correlation: r[j] = sum_k p[k] * a[k-j] for valid k-j in [0, n-1]
    # r[j] = (a * p_rev)[(2n-2)-j] where p_rev = p[::-1]
    p_rev = p[::-1]
    conv_result = fast_full_conv(a, p_rev)  # length n + (2n-1) - 1 = 3n-2
    
    # Extract r[0..n-1]
    # r[j] = conv_result[(2n-2) - j] for j = 0, ..., n-1
    # So r = conv_result[2n-2 : n-2 : -1]... 
    # positions: 2n-2, 2n-3, ..., n-1 (that's n positions)
    r = conv_result[n-1 : 2*n-1][::-1]
    
    # Gradient of smooth_max w.r.t. a
    grad = 2 * r
    
    # Project to zero-sum (for the constraint sum(a) = 1)
    grad = grad - grad.mean()
    
    return val, grad

        if s_new < 1e-15:

        else:

        # Check improvement

        if M_new < best_val - 1e-12:

            a = a_new
