import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lse_weights(conv, tau):
    """Compute softmax weights for LSE gradient."""
    tau_conv = tau * conv
    tau_conv = tau_conv - tau_conv.max()
    w = np.exp(tau_conv)
    s = w.sum()
    if s < 1e-300:
        return np.ones_like(conv) / len(conv)
    return w / s


def lse_gradient(a, tau):
    """
    Compute gradient of LSE smoothed max of convolution.
    grad[m] = 2 * sum_{d=0}^{n-1} a[d] * w[m+d]
    """
    n = len(a)
    conv = fast_autocorr(a)
    w = lse_weights(conv, tau)

    # g[m] = sum_d a[d] * w[m+d] = conv(a_rev, w)[n-1+m]
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a[::-1], fft_size)
    W = np.fft.rfft(w, fft_size)
    full_conv = np.fft.irfft(A * W, fft_size)
    g = full_conv[n - 1: 2 * n - 1]
    grad = 2.0 * g
    return grad, conv


def mgd_lse(a0, time_limit, tau_init, tau_max, tau_growth, eta_init, verbose=False):
    """
    Multiplicative gradient descent with LSE smoothed gradient.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    tau = tau_init
    eta = eta_init
    iter_count = 0
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit:
        grad, conv = lse_gradient(a, tau)
        M = conv.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            break

        # Center the gradient (remove mean for stability)
        grad_centered = grad - grad.mean()
        gmax = np.abs(grad_centered).max()
        if gmax < 1e-15:
            break
        grad_centered = grad_centered / gmax

        # Multiplicative update
        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s

        # Check if it improved
        conv_new = fast_autocorr(a_new)
        M_new = conv_new.max()

        if M_new <= M + 1e-13:
            a = a_new
            eta *= 0.9998
        else:
            # Try smaller step
            eta *= 0.5
            if eta < eta_init * 0.01:
                eta = eta_init * 0.1
                # Try with centered gradient removed
                a_new = a * np.exp(-eta * (grad - grad.mean()) / max(gmax, 1e-15))
                a_new = np.maximum(a_new, 1e-30)
                a_new = a_new / a_new.sum()
                conv_new2 = fast_autocorr(a_new)
                if conv_new2.max() < M:
                    a = a_new

        # Adapt tau
        iter_count += 1
        if iter_count % 30 == 0 and tau < tau_max:
            tau = min(tau * tau_growth, tau_max)

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MGD-LSE iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.0f}, eta={eta:.6f}")

    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"      MGD-LSE done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}, tau={tau:.0f}")
    return best_a


def pairwise_coord_search(a0, time_limit, verbose=False):
    """
    Pairwise coordinate transfer with efficient convolution updates.
    Transfers mass between coordinates to reduce the peak convolution.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    conv = fast_autocorr(a).copy()
    best_a = a.copy()
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    pass_count = 0
    stall = 0
    max_stall = 8

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        pass_count += 1
        improved = False
        M = conv.max()

        # Identify top contributing lags
        top_lags = np.argsort(conv)[::-1][:8]

        # Build a score for each coordinate based on its contribution to peaks
        peak_score = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js <= je:
                idx = np.arange(js, je + 1)
                peak_score[idx] += a[ki - idx] * a[idx]

        # Sort coordinates by peak contribution (high to low)
        order_high = np.argsort(peak_score)[::-1]
        order_low = np.argsort(peak_score)

        n_trials = min(40, n)
        for t in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick source from high, destination from low
            if t < n_trials // 2:
                src = order_high[t % min(len(order_high), 20)]
                dst_candidates = order_low[:20]
            else:
                dst = order_low[t % min(len(order_low), 20)]
                src_candidates = order_high[:20]
                # Swap roles
                src = src_candidates[t % min(len(src_candidates), 20)]
                dst_candidates = np.array([dst])

            if a[src] < 1e-10:
                continue

            dst = dst_candidates[t % len(dst_candidates)]
            if dst == src or a[src] < 1e-10:
                continue

            # Try several transfer fractions
            fracs = [0.02, 0.05, 0.1, 0.2, 0.3]
            best_frac = 0.0
            best_delta = 0.0
            best_m_trial = M

            for frac in fracs:
                delta = a[src] * frac
                if delta < 1e-12:
                    continue

                # Compute new conv
                conv_trial = conv.copy()

                # Update for src (a[src] += delta)
                k_lo = max(0, int(src))
                k_hi = min(2 * n - 2, int(src) + n - 1)
                k_range = np.arange(k_lo, k_hi + 1)
                mask = k_range != 2 * int(src)
                if mask.any():
                    conv_trial[k_range[mask]] += 2.0 * delta * a[k_range[mask] - int(src)]
                conv_trial[2 * int(src)] += 2.0 * delta * a[int(src)] + delta * delta

                # Update for dst (a[dst] -= delta)
                j_lo = max(0, int(dst))
                j_hi = min(2 * n - 2, int(dst) + n - 1)
                j_range = np.arange(j_lo, j_hi + 1)
                jmask = j_range != 2 * int(dst)
                if jmask.any():
                    conv_trial[j_range[jmask]] -= 2.0 * delta * a[j_range[jmask] - int(dst)]
                conv_trial[2 * int(dst)] -= 2.0 * delta * a[int(dst)] + delta * delta

                M_trial = conv_trial.max()
                if M_trial < best_m_trial - 1e-15:
                    best_m_trial = M_trial
                    best_frac = frac
                    best_delta = delta

            if best_delta > 0:
                # Apply the update
                a[int(src)] += best_delta
                a[int(dst)] -= best_delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = fast_autocorr(a)  # Recompute for consistency
                improved = True

                if conv.max() < best_m:
                    best_m = conv.max()
                    best_a = a.copy()

                if t > n_trials // 2:
                    break

        if not improved:
            stall += 1
            # Random escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                delta = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = fast_autocorr(a)
                if conv.max() < best_m:
                    best_m = conv.max()
                    best_a = a.copy()
                    stall = 0

        if verbose and pass_count % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Pairwise pass={pass_count} ({elapsed:.0f}s): J={J:.6f}")

    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"      Pairwise done ({elapsed:.0f}s): J={J:.6f}, passes={pass_count}")
    return best_a


def focused_local_search(a0, time_limit, verbose=False):
    """
    Focused local search: try small perturbations to the top-contributing
    coordinates with adaptive step sizes.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    conv = fast_autocorr(a)
    best_a = a.copy()
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_autocorr(a)
        M = conv.max()
        improved = False

        top_lags = np.argsort(conv)[::-1][:12]

        for k in top_lags[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find the top contributing position
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find good destination: low contribution to all top peaks
            dst_scores = np.zeros(n)
            for k2 in top_lags[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    dst_scores[idx2] += a[ki - idx2] if ki == k2i else a[k2i - idx2] * 0.5

            candidates = np.where(dst_scores < np.percentile(dst_scores, 25))[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            for _ in range(3):
                dst = candidates[int(rng.integers(len(candidates)))]
                if dst == src:
                    continue

                for frac in [0.01, 0.03, 0.05, 0.08, 0.12]:
                    delta = src_val * frac
                    if delta < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            # Escape: random redistribution
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                delta = a[i] * rng.uniform(0.005, 0.03)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"      Focused done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def structured_init(n):
    """Generate several structured initial sequences."""
    x = np.linspace(0, 1, n)
    results = []

    # Sine squared
    results.append(norm_sum1(np.sin(np.pi * x) ** 2))

    # Raised cosine
    results.append(norm_sum1(0.5 * (1 + np.cos(2 * np.pi * x))))

    # Hamming-like window
    results.append(norm_sum1(0.54 - 0.46 * np.cos(2 * np.pi * x)))

    # Triangular
    results.append(norm_sum1(np.maximum(1 - 2 * np.abs(x - 0.5), 0)))

    # Bump in middle
    for sigma in [0.15, 0.25, 0.35]:
        results.append(norm_sum1(np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))))

    # Two bumps
    for pos in [0.3, 0.35]:
        results.append(norm_sum1(
            np.exp(-((x - pos) ** 2) / (2 * 0.1 ** 2)) +
            np.exp(-((x - (1 - pos)) ** 2) / (2 * 0.1 ** 2))
        ))

    # Power law edges
    for p in [0.5, 1.0, 2.0]:
        results.append(norm_sum1(np.minimum(x, 1 - x) ** p + 0.01))

    # Exponential edges
    for rate in [2, 5, 10]:
        results.append(norm_sum1(np.exp(-rate * np.minimum(x, 1 - x))))

    return results

    # === Phase 1: MGD with LSE (low tau, smooth) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MGD-LSE smooth (n={n0}, tau=2000->50000) ===")
    t1 = min(remaining * 0.22, 300.0)
    if t1 > 30:
        a_out = mgd_lse(norm_sum1(best_seq), t1, tau_init=2000, tau_max=50000,
                        tau_growth=1.08, eta_init=0.03, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Phase1: {sc:.6f}")

    # === Phase 2: MGD with LSE (higher tau, sharper) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 2: MGD-LSE sharp (n={n_curr}, tau=20000->200000) ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 30:
        a_out = mgd_lse(norm_sum1(best_seq), t2, tau_init=20000, tau_max=200000,
                        tau_growth=1.06, eta_init=0.012, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Phase2: {sc:.6f}")

    # === Phase 3: Pairwise coordinate search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 3: Pairwise search (n={n_curr}) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 20:
        a_out = pairwise_coord_search(norm_sum1(best_seq), t3, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Phase3: {sc:.6f}")

    # === Phase 4: Focused local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 4: Focused search (n={n_curr}) ===")
    t4 = min(remaining * 0.12, 160.0)
    if t4 > 15:
        a_out = focused_local_search(norm_sum1(best_seq), t4, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Phase4: {sc:.6f}")

    # === Phase 5: MGD again on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 5: MGD-LSE refine (n={n_curr}) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_out = mgd_lse(norm_sum1(best_seq), t5, tau_init=50000, tau_max=500000,
                        tau_growth=1.04, eta_init=0.008, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Phase5: {sc:.6f}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 6: Multi-length exploration ===")

    # Try a few different lengths
    target_lengths = []
    for delta in [-500, -200, 200, 500, 1000, 2000]:
        cand_n = n_curr + delta
        if 2000 <= cand_n <= 8000 and cand_n not in target_lengths:
            target_lengths.append(cand_n)

    for new_n in target_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")

        a_long = interpolate_sequence(best_seq, new_n)
        # Quick MGD at new length
        t6 = min(remaining * 0.35, 130.0)
        if t6 > 25:
            a_out = mgd_lse(a_long, t6, tau_init=10000, tau_max=100000,
                            tau_growth=1.07, eta_init=0.02, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST n={new_n}: {sc:.6f}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t6b = min(remaining * 0.3, 80.0)
                if t6b > 15:
                    a_out2 = pairwise_coord_search(norm_sum1(best_seq), t6b, verbose=False)
                    sc2 = evaluate_safe(a_out2)
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_seq = a_out2.copy()
                        print(f"  *** NEW BEST pairwise n={new_n}: {sc2:.6f}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_curr}) ===")
    if remaining > 40:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 20:
            a_out = mgd_lse(norm_sum1(best_seq), t7, tau_init=100000, tau_max=1000000,
                            tau_growth=1.03, eta_init=0.005, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final MGD: {sc:.6f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8 = min(remaining * 0.5, 100.0)
            if t8 > 15:
                a_out = focused_local_search(norm_sum1(best_seq), t8, verbose=False)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST final focused: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
