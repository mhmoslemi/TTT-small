def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def nesterov_opt(f0, time_budget, p_init=6.0, p_final=80.0, p_growth=1.0025,
                 lr_init=0.4, momentum=0.9, verbose=True):
    n = len(f0)
    f = norm_sum1(f0)
    v = np.zeros(n)
    best_f = f.copy()
    best_M = self_conv(f).max()
    t0 = time.monotonic()
    p = p_init
    stall = 0
    step = 0

    while time.monotonic() - t0 < time_budget:
        step += 1
        y = f + momentum * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = f.copy()
        else:
            y = y / sy

        conv_y = self_conv(y)
        M_y = conv_y.max()

        if M_y < best_M - 1e-14:
            best_M = M_y
            best_f = y.copy()
            stall = 0
        else:
            stall += 1

        grad = power_smoothed_grad(y, conv_y, p)

        lr = lr_init / (1.0 + 0.0005 * step)
        v = momentum * v - lr * grad
        f = np.maximum(f + v, 0.0)
        sf = f.sum()
        if sf < 1e-15:
            f = best_f.copy()
            v = np.zeros(n)
        else:
            f = f / sf

        p = min(p * p_growth, p_final)

        if stall > 400:
            stall = 0
            rng = np.random.default_rng(step)
            pert = rng.normal(0, 0.04, n) * best_f
            f = np.maximum(best_f + pert, 0.0)
            sf2 = f.sum()
            if sf2 > 1e-15:
                f = f / sf2
            v = np.zeros(n)
            p = p_init

        if verbose and step % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Nesterov step={step} ({elapsed:.0f}s): J={J:.6f}, p={p:.1f}")

    return best
