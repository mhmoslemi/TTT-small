def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    if s < 1e-15:
        return np.ones_like(a) / len(a)

def lse_smoothed_obj_and_grad(a, tau):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_scaled = ac / tau
    ac_scaled = ac_scaled - ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()
    obj = tau * (np.log(exp_vals.sum()) + (ac.max() / tau))
    
    a_flip = a[::-1].astype(np.float64)
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return obj, np.maximum(grad, 0.0), ac

def fw_line_search(y, g_dir, ac_y, M_y, max_iter=30):
    """Backtracking line search for Frank-Wolfe step."""

    best_alpha = 2.0 / 5.0  # fallback

    for alpha in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03]:
        y_new = (1.0 - alpha) * y + alpha * g_dir

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    for freq in [1, 2, 3, 4]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.005)
            candidates.append(a / a.sum())
    
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())
    
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())
    
    # Double bump
    for c1, c2 in [(0.25, 0.75), (0.3, 0.7), (0.2, 0.8)]:
        a = np.exp(-((x - c1) ** 2) / 0.02) + np.exp(-((x - c2) ** 2) / 0.02)
        candidates.append(a / a.sum())
    
    # Windowed sinc
    for bw in [2, 3, 5]:
        a = np.abs(np.sinc(bw * (x - 0.5)))
        a = a * (np.sin(np.pi * x) ** 2 + 0.01)
        candidates.append(a / a.sum())
    
    candidates.append(np.ones(n) / n)
    
    return candidates

                check_update(cand, f"param{i}")
        # Blend with parent
        for w in [0.3, 0.5, 0.7]:
            cand = w * best_seq + (1 - w) * np.ones(n0) / n0
            check_update(cand, f"blend_{w}")
    
    # === Phase 2: Smoothed Adam ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Smoothed Adam (n={n_cur}) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam_v2(a_in, t2, verbose=True)
        check_update(a_out, f"adam_n{n_cur}")
    
    # === Phase 3: LP-FW v3 (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP-FW3 (n={n_cur}) ===")
    t3 = min(remaining * 0.30, 450.0)
    if t3 > 30 and n_cur <= 6000 and solver is not None:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v3(a_in, t3, solver, initial_K=800,
                                      max_K=5000, add_per_iter=80, verbose=True)
        check_update(a_out, f"lp3_n{n_cur}")
    
    # === Phase 4: Second LP-FW pass (refined) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-FW3 pass 2 (n={n_cur}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30 and n_cur <= 6000 and solver is not None:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v3(a_in, t4, solver, initial_K=1000,
                                      max_K=5000, add_per_iter=60, verbose=True)
        check_update(a_out, f"lp3b_n{n_cur}")
    
    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Coord descent (n={n_cur}) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_v3(a_in, t5, rng_seed=111, verbose=False)
        check_update(a_out, f"coord_n{n_cur}")
    
    # === Phase 6: Pairwise swap polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Pairwise swap (n={n_cur}) ===")
    t6 = min(remaining * 0.06, 60.0)
    if t6 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_swap(a_in, t6, rng_seed=7, verbose=False)
        check_update(a_out, f"swap_n{n_cur}")
    
    # === Phase 7: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Nearby lengths ===")
    
    nearby = []
    for delta in [300, -300, 500, -500, 200, -200, 700]:
        new_n = n_cur + delta
        if 2500 <= new_n <= 6000:
            nearby.append(new_n)
    
    seen = set()
    unique_nearby = []
    for ln in nearby:
        if ln not in seen:
            seen.add(ln)
            unique_nearby.append(ln)
    
    for new_n in unique_nearby[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_interp = interpolate_sequence(best_seq, new_n)
        
        # Quick Adam
        t7a = min(remaining * 0.20, 100.0)
        if t7a > 20:
            a_out, _ = smooth_adam_v2(a_interp, t7a, verbose=False)
            check_update(a_out, f"adam_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 5500 and solver is not None:
            t7b = min(remaining * 0.30, 200.0)
            if t7b > 25:
                a_out, _ = lp_frank_wolfe_v3(norm_sum1(best_seq), t7b, solver,
                                              initial_K=600, max_K=4000,
                                              add_per_iter=50, verbose=False)
                check_update(a_out, f"lp3_n{new_n}")
    
    # === Phase 8: Final LP-FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LP-FW (n={n_cur}) ===")
    if remaining > 60 and n_cur <= 6000 and solver is not None:
        t8 = min(remaining * 0.35, 250.0)
        if t8 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v3(a_in, t8, solver, initial_K=1200,
                                          max_K=5000, add_per_iter=100, verbose=True)
            check_update(a_out, f"lp_final_n{n_cur}")
    
    # === Phase 9: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Final coord (n={n_cur}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.40, 100.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_v3(a_in, t9, rng_seed=999, verbose=False)
            check_update(a_out, f"coord_final_n{n_cur}")
    
    # === Phase 10: Final swap polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 15:
        t10 = min(remaining * 0.30, 50.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = pairwise_swap(a_in, t10, rng_seed=123, verbose=False)
            check_update(a_out, f"swap_final_n{n_cur}")
    
    # Final validation
    if best_seq is not None:
        fs = eval_full(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3500
        best_seq = np.ones(n_fb) / n_fb
        best_score = eval_score(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
