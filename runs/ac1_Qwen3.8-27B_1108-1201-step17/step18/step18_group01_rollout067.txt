def fast_autocorr(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return result


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized non-negative sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_subgrad(a, k):
    """
    Gradient of c[k] = sum_i a[i]*a[k-i] with respect to a.
    Result: grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    j_min = max(0, k - (n - 1))
    j_max = min(n - 1, k)
    if j_min <= j_max:
        idx = np.arange(j_min, j_max + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad

        best_seq = best_seq / best_seq.sum()
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 3200
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")

    return [float(x) for x in best_seq]
