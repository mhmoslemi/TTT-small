def lp_fw_optimize(a0, time_limit, solver, top_k=None, verbose=True):
    """LP-based Frank-Wolfe with bisection line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    iter_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(top_k, total_lags)
        if K_sel >= total_lags:
            binding_idx = np.arange(total_lags)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build T matrix: T[ii, j] = a[k-i - j] for valid indices
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_end > j_start:
                T[ii, j_start:j_end] = a[ki - np.arange(j_start, j_end)]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)

            stall += 10
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stall += 5
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            stall += 5
            continue
        direction = b_val / b_sum

        # Bisection line search: find best alpha in [0, 1]
        def trial_m(alpha):
            a_t = (1.0 - alpha) * a + alpha * direction
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st < 1e-15:
                return float('inf')
            a_t = a_t / st
            return fast_autocorr(a_t).max()

        lo, hi = 0.0, 1.0
        best_alpha = 0
        best_trial = M_curr
        for _ in range(40):
            mid = 0.5 * (lo + hi)
            m_mid = trial_m(mid)
            if m_mid < best_trial:
                best_trial = m_mid
                best_alpha = mid
            # Check slope: f(mid) vs f(lo)
            m_lo = trial_m(lo) if lo > 0 else M_curr
            m_hi = trial_m(hi) if hi < 1 else float('inf')
            if m_mid < m_lo:
                lo = mid
            else:
                hi = mid

        a = (1.0 - best_alpha) * a + best_alpha * direction
        a = np.maximum(a, 0.0)

        if sa < 1e-15:
            break
        a = a / sa

        m_new = fast_autocorr(a).max()

            stall = 0

def subgradient_fw(a0, time_limit, verbose=True):
    """Fast subgradient FW with softmin direction."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 0.5
    stall = 0
    top_k = 30
    beta_sm = 8.0

    t = 0
    while time.monotonic() - t0 < time_limit and stall < 500:
        t += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        threshold = M_curr * 0.95
        active = np.where(ac >= threshold)[0]
        if len(active) < 1:
            active = np.argsort(ac)[::-1][:top_k]
        active = active[:top_k]

        # Compute subgradient: (2/|S|) * sum_{k in S} shifted_a[k]
        grad = np.zeros(n)
        for k in active:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_end > j_start:
                grad[j_start:j_end] += a[ki - np.arange(j_start, j_end)]
        grad *= 2.0 / len(active)

        # Softmin direction
        shifted = grad - grad.min()
        b_dir = np.exp(-beta_sm * shifted)
        bs = b_dir.sum()
        if bs < 1e-15:
            break
        b_dir = b_dir / bs

        # Step
        eta_t = eta / (1.0 + 0.01 * t)
        a_new = a + eta_t * (b_dir - a)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.85

        a = a_new
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SG-FW t={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m

                # add quadratic correction
                vals_lo = vals_lo.copy()
                vals_lo[k_diag] = ac_curr[k_diag] - 2 * old_val * a[j] + 2 * x_lo * a[j] + x_lo * x_lo - old_val * old_val
            max_lo = vals_lo.max()
            active_lo = np.where(np.abs(vals_lo - max_lo) < 1e-8 * max(1.0, max_lo))[0]
            if len(active_lo) > 0 and slopes[active_lo].max() < -1e-10:
                # Function is decreasing at x_lo, increase
                pass
            elif len(active_lo) > 0 and slopes[active_lo].min() > 1e-10:
                # Function is increasing at x_lo, optimal is x_lo = 0
                new_val = 0.0
            else:
                # Bisection
                for _ in range(50):
                    x_mid = 0.5 * (x_lo + x_hi)
                    vals_mid = intercepts + slopes * x_mid
                    if k_diag < 2 * n - 1:
                        vals_mid = vals_mid.copy()
                        vals_mid[k_diag] = ac_curr[k_diag] - 2 * old_val * a[j] + 2 * x_mid * a[j] + x_mid * x_mid - old_val * old_val
                    max_mid = vals_mid.max()
                    active_mid = np.where(np.abs(vals_mid - max_mid) < 1e-8 * max(1.0, max_mid))[0]
                    if len(active_mid) == 0:
                        active_mid = [int(np.argmax(vals_mid))]
                    sl = slopes[active_mid]
                    if sl.max() < -1e-10:
                        x_lo = x_mid
                    elif sl.min() > 1e-10:
                        x_hi = x_mid
                    else:
                        x_lo = x_mid
                        x_hi = x_mid
                        break
                new_val = 0.5 * (x_lo + x_hi)

            new_val = max(0.0, new_val)
            if abs(new_val - old_val) > 1e-12:
                a[j] = new_val
                m_test = fast_autocorr(a).max()
                if m_test < best_m:
                    best_m = m_test
                    best = a.copy()
                    improved = True

        if not improved:
            # Try random coordinate changes
            for _ in range(10):
                ji = int(np.random.randint(n))
                a[ji] *= np.random.uniform(0.7, 1.3)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m_test = fast_autocorr(a).max()
            if m_test < best_m:
                best_m = m_test
                best = a.copy()

    return best, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search: LP-FW + subgradient FW + 1D coordinate + multi-restart.
    Targets J < 1.5027 starting from parent (J~1.504, n=5200).
    """
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)
    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(0.3 + 0.7 * x)  # slightly increasing
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting optimization: n={n0}, J={best_score:.6f}")

    # === Phase 1: Subgradient FW (fast, many iterations) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = min(remaining * 0.15, 250.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Subgradient FW (t={t1:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_fw(a_in, t1, verbose=True)
        try_update(a_out, "sg_fw")

    # === Phase 2: LP-FW (all constraints, precise) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-FW all constraints (t={t2:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        total_lags = 2 * n0 - 1
        # Use all constraints for precision
        a_out, _ = lp_fw_optimize(a_in, t2, solver, top_k=total_lags, verbose=True)
        try_update(a_out, "lp_all")

    # === Phase 3: 1D Coordinate Polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        print(f"\n=== Phase 3: 1D Coord Polish (t={t3:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_1d_optimize(a_in, None, n_coords_per_peak=8,
                                      rounds=200, time_limit=t3)
        try_update(a_out, "coord1d")

    # === Phase 4: LP-FW round 2 on best ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 60:
        print(f"\n=== Phase 4: LP-FW r2 (t={t4:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        total_lags = 2 * n_best - 1
        a_out, _ = lp_fw_optimize(a_in, t4, solver, top_k=total_lags, verbose=True)
        try_update(a_out, "lp_r2")

    # === Phase 5: Multi-restart perturbation ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 5: Multi-restart ===")
    rng = np.random.default_rng(2024)
    n_restarts = 3
    t_per_restart = min(remaining * 0.5 / n_restarts, 120.0)

    for ri in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 40:
            break
        t_avail = min(t_per_restart, remaining - 10.0)
        if t_avail < 20:
            break

        # Perturb
        noise_scale = 0.01 / (ri + 1)
        pert = rng.normal(0, noise_scale, len(best_seq)) * best_seq
        a_pert = best_seq + pert
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()

        # Quick LP-FW
        n_p = len(a_pert)
        total_lags = 2 * n_p - 1
        t_lp = t_avail * 0.7
        a_out, _ = lp_fw_optimize(a_pert, t_lp, solver, top_k=min(total_lags, 5000),
                                   verbose=False)
        try_update(a_out, f"restart_{ri}")

        # Quick coord
        remaining2 = time_budget - (time.monotonic() - t_start)
        if remaining2 > 30:
            t_coord = min(t_avail * 0.3, 40.0)
            a_out2, _ = coord_1d_optimize(norm_sum1(a_out), None,
                                            n_coords_per_peak=5, rounds=50, time_limit=t_coord)
            try_update(a_out2, f"restart_coord_{ri}")

    # === Phase 6: Final LP-FW + Coord on absolute best ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final (t_remaining={remaining:.0f}s) ===")
    if remaining > 60:
        t6 = min(remaining * 0.5, 200.0)
        a_in = norm_sum1(best_seq)
        total_lags = 2 * n_best - 1
        a_out, _ = lp_fw_optimize(a_in, t6, solver, top_k=total_lags, verbose=True)
        try_update(a_out, "final_lp")

    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 30:
        t6b = min(remaining * 0.6, 100.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_1d_optimize(a_in, None, n_coords_per_peak=10,
                                      rounds=300, time_limit=t6b)
        try_update(a_out, "final_coord")

    # === Phase 7: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 100:
        print(f"\n=== Phase 7: Length exploration ===")
        for new_n in [int(n0 * 0.8), int(n0 * 1.2)]:
            remaining = time_budget - (time.monotonic() - t_start)
            if remaining < 60:
                break
            if not (2000 <= new_n <= 8000):
                continue
            print(f"  Trying n={new_n}...")
            a_long = interpolate_seq(best_seq, new_n)
            t7 = min(remaining * 0.4, 150.0)
            if t7 > 30:
                total_lags_7 = 2 * new_n - 1
                a_out, _ = lp_fw_optimize(a_long, t7, solver,
                                           top_k=min(total_lags_7, 6000), verbose=False)
                try_update(a_out, f"len_{new_n}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(0.3 + 0.7 * x)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
