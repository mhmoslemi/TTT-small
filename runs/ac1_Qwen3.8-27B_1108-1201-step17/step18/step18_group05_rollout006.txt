def fast_conv(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

            break
        phase_start = time.monotonic()
        phase_limit = min(time_limit - (phase_start - t0), time_limit / len(tau_schedule))
        stall = 0
        eta = eta_init * min(1.0, tau / 50.0)

        while time.monotonic() - phase_start < phase_limit and stall < max_stall:
            elapsed = time.monotonic() - t0
            if elapsed > time_limit:
                break

            if use_nesterov and iter_count > 0:
                beta = (t_nest - 1.0) / (t_nest + 1.0)
                y = a + beta * (a - a_prev)
                y = np.maximum(y, 1e-15)
                sy = y.sum()
                if sy > 0:
                    y = y / sy
            else:
                y = a

            grad, p, g = compute_spectral_gradient(y, tau)
            g_tangent = grad - grad.mean()

            a_new = project_simplex(y - eta * g_tangent)

            m_new = fast_conv(a_new).max()
            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
                stall = 0
                eta = min(eta * 1.01, 2.0)
            else:
                stall += 1
                if stall > 100:
                    eta *= 0.85
                if stall > 300:
                    pert = rng.normal(0, 0.002, n) * a
                    a_p = np.maximum(a + pert, 0.0)
                    a_p = a_p / a_p.sum()
                    m_p = fast_conv(a_p).max()
                    if m_p < best_m:
                        a = a_p
                        best_m = m_p
                        best_a = a.copy()
                        stall = 150
                    else:
                        a = a_new
                        stall = 200
                else:
                    a = a_new

            a_prev = a
            t_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2)) / 2.0
            iter_count += 1

            if verbose and iter_count % 2000 == 0:
                el = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SpectralGr tau={tau:.0f} iter={iter_count} ({el:.0f}s): J={J:.6f}, eta={eta:.4f}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    SpectralGr final: J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    prev_b = None

    while time.monotonic() - t0 < time_limit:
        ac = fast_conv(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 30, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search with more candidates
        base_M = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_M
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                  0.5, 0.3, 0.2, 0.1]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_conv(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 5 == 0:
            K = min(K + 20, top_K_max)

        if verbose and iter_count % 10 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({el:.0f}s): J={J:.6f}, K={K_sel}, a={best_alpha:.4f}")

    return best_a, best_m


def compute_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=5000,
                 adaptive_beta=True, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 80000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy > 0:
            y = y / sy
        else:
            y = x.copy()

        ac = fast_conv(y)
        g = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new > 0:
            a_new = a_new / s_new
        else:
            a_new = y.copy()

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if adaptive_beta and stall_count > restart_every // 3:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 5, 200)

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 250)
            cur_beta = max(cur_beta - 0.5, 2.0)

        if verbose and t % 20000 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({el:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def pair_exchange_polish(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k[:n_top]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if verbose and stall % 50 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairExch ({el:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    PairExch final: J={J:.6f}")
    return best_a, best_m
