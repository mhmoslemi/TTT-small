"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated subgradient computation for softmax max-convolution
2. Nesterov-accelerated mirror descent (multiplicative updates)
3. Adaptive temperature scheduling based on peak distribution
4. Multi-length, multi-init diverse search with periodic restarts
5. Final coordinate descent polish
"""

import numpy as np
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def fft_conv(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad_fft(a, conv, n_active=None, temperature=0.002):
    """
    Compute the softmax subgradient of max(conv) using FFT.
    
    The subgradient of max_k conv[k] at a is sum_k pi_k * grad_k,
    where grad_k[j] = 2 * a[(k-j) mod N] (approx).
    
    In frequency domain: FFT(grad)[w] = 2 * conj(FFT(a)[w]) * FFT(pi_padded)[w]
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Compute peak weights
    conv_max = conv.max()
    # Use threshold to select active peaks
    if n_active is None:
        # Select top peaks by threshold
        threshold = conv_max - temperature * 10.0
        active_mask = conv >= threshold
        if active_mask.sum() < 10:
            # Take top 10
            top_idx = np.argsort(conv)[::-1][:10]
            active_mask = np.zeros_like(conv, dtype=bool)
            active_mask[top_idx] = True
        n_act = active_mask.sum()
    else:
        top_idx = np.argsort(conv)[::-1][:n_active]
        active_mask = np.zeros_like(conv, dtype=bool)
        active_mask[top_idx] = True
        n_act = n_active
    
    # Softmax weights over active peaks
    conv_active = conv[active_mask]
    log_w = (conv_active - conv_max) / temperature
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(n_act) / n_act
    else:
        w = w / w_sum
    
    # Build the padded peak weight vector
    pi_padded = np.zeros(fft_size)
    pi_padded[:size] = 0.0
    pi_padded[np.where(active_mask)[0]] = w
    
    # FFT computation
    A_hat = np.fft.rfft(a, fft_size)
    Pi_hat = np.fft.rfft(pi_padded, fft_size)
    
    # Subgradient in freq domain: 2 * conj(A_hat) * Pi_hat
    # For real signals, conj(A_hat)[w] = A_hat[fft_size - w] (with A_hat[0] real)
    # But we can just compute directly
    G_hat = 2.0 * np.conj(A_hat) * Pi_hat
    
    # Wait, Pi_hat is complex (even for real input to rfft? No, rfft of real gives complex)
    # Actually np.fft.rfft returns complex values.
    # G_hat should be such that IFFT gives real result.
    # For real a and real pi_padded, G_hat[w] = 2 * conj(A_hat[w]) * Pi_hat[w]
    # The IFFT of this should be real (up to numerical error).
    
    g = np.fft.irfft(G_hat, fft_size)[:n]
    return g


def compute_subgrad_spatial(a, conv, n_active=50, temperature=0.002):
    """
    Compute the softmax subgradient in the spatial domain.
    For small n_active, this is faster than FFT.
    """
    n = len(a)
    top_idx = np.argsort(conv)[::-1]
    if n_active is not None:
        top_idx = top_idx[:n_active]
    
    conv_max = conv.max()
    log_w = (conv[top_idx] - conv_max) / temperature
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    
    return grad


def amd_fft(a0, time_limit, seed=42, verbose=False):
    """
    Accelerated Mirror Descent with FFT-based subgradient.
    """
    n = len(a0)
    x = norm1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fft_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adaptive parameters
    temperature = 0.003
    c0 = 3.0
    top_k = 30
    stall_count = 0
    restart_count = 0
    restart_every = 5000
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute convolution and subgradient
        conv_y = fft_conv(y)
        
        # Choose between FFT and spatial subgradient based on n_active
        conv_max = conv_y.max()
        # Estimate number of active peaks
        n_above = (conv_y >= conv_max - temperature * 8).sum()
        
        if n_above > 200:
            g = compute_subgrad_fft(y, conv_y, temperature=temperature)
        else:
            g = compute_subgrad_spatial(y, conv_y, n_active=top_k, temperature=temperature)
        
        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent step (multiplicative)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M = fft_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive restart
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.08, 0.02 + 0.01 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            top_k = min(top_k + 5, 100)
            temperature = max(temperature * 0.95, 0.0008)
            c0 = 3.0 * (1.0 + 0.03 * min(restart_count, 15))
            restart_every = max(3000, restart_every + 300)
            
            if verbose and restart_count % 5 == 0:
                J = 2.0 * n * best_m
                print(f"    AMD restart {restart_count} t={t}: J={J:.6f}, temp={temperature:.5f}")
    
    return best_x, best_m

        for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9]:
            amt = a[src_i] * frac
            a_trial = a.copy()
            a_trial[src_i] -= amt
            a_trial[dst_i] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fft_conv(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fft_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


# --- Initial sequence generators ---

def init_uniform(n):
    return np.ones(n) / n

def init_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm1((n - x + 1) ** power)

def init_increasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm1((x + 1) ** power)

def init_bell(n, center=0.5, width=0.2):
    x = np.linspace(0, 1, n)
    return norm1(np.exp(-((x - center) / width) ** 2))

def init_u_shape(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    return norm1(((x - 0.5) / 0.3) ** 2 * sharpness + 0.1)

def init_sinc(n, bw=5.0, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - center) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm1(a)

def init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm1(a)

def init_two_peak(n, p1=0.2, p2=0.8, w=0.08):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - p1) / w) ** 2) + np.exp(-((x - p2) / (w * 1.3)) ** 2)
    return norm1(a)

def init_three_peak(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / 0.06) ** 2) + np.exp(-((x - 0.5) / 0.08) ** 2) + np.exp(-((x - 0.85) / 0.06) ** 2)
    return norm1(a)

def init_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm1(a)

def init_power_centered(n, power=0.5):
    x_c = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm1(x_c ** (-power))

def init_exponential(n, rate=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm1(np.exp(-rate * x) + np.exp(-rate * (1 - x)))
