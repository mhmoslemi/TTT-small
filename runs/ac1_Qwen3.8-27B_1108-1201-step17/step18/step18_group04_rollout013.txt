def fast_conv(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    conv = fast_conv(a)
    # Shift for numerical stability
    c_max = conv.max()
    exp_conv = np.exp((conv - c_max) / temp)
    log_sum = c_max + temp * np.log(exp_conv.sum())

    weights = exp_conv / exp_conv.sum()

    n = len(a)
    grad = np.zeros(n)

        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * weights[k] * a[k - idx]

    # Actually, we optimize on the simplex, so we need to project
    return log_sum, grad

    if s < 1e-15:
        return np.inf
    return 2.0 * n * fast_conv(a).max() / (s * s)

def update_best(a, label=""):
    """Try to update the global best."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    if a.sum() < 0.01:
        return
    try:
        sc = evaluate_sequence([float(v) for v in a])
    except Exception:
        return
    if np.isfinite(sc) and sc < _best['score']:
        _best['score'] = sc
        _best['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

# --- Initialization functions ---

def init_sinc(n, bw=5.0, center=0.5):
    x = np.arange(n) / n
    a = np.sinc((x - center) * bw) ** 2
    return normalize(np.maximum(a, 0))

def init_bessel(n, center=0.45):
    from scipy.special import j0
    x = np.arange(n) / n
    a = j0((x - center) * 8) ** 2
    return normalize(np.maximum(a, 0))

def init_gaussian(n, mu=0.3, sigma=0.15, mu2=0.7, sigma2=0.12, w2=0.8):
    x = np.arange(n) / n
    a = np.exp(-((x - mu) / sigma) ** 2) + w2 * np.exp(-((x - mu2) / sigma2) ** 2)
    return normalize(a)

def init_power_decay(n, p=0.7, center=0.3):
    x = np.abs(np.arange(n) / n - center) + 0.01
    return normalize(x ** (-p))

def init_cosine(n, mode=2):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    return normalize(np.maximum(a, 0))

def init_asymmetric_gaussian(n):
    x = np.arange(n) / n
    # Asymmetric: different widths on each side
    a = np.exp(-x / 0.3) * np.exp(-x / 0.1)  # product of two exponentials
    return normalize(a)

def init_two_bumps(n):
    x = np.arange(n) / n
    a = 0.6 * np.exp(-((x - 0.25) / 0.08) ** 2) + 0.4 * np.exp(-((x - 0.75) / 0.12) ** 2)
    return normalize(a)

def init_smooth_random(n, seed=42, n_harmonics=12):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for k in range(1, n_harmonics + 1):
        coeff = rng.normal(0, 0.1 / np.sqrt(k))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * k * x + phase)
    return normalize(np.maximum(a, 0.01))

def lse_adam_optimize(a0, time_limit, seed=42, temp0=0.08, temp_min=0.0005):
    """
    Optimize using Adam on the log-sum-exp surrogate.
    Uses softmax parameterization: a_i = exp(b_i) / sum(exp(b_j))
    """
    n = len(a0)
    b = np.log(np.maximum(np.asarray(a0, dtype=np.float64), 1e-10))
    b = b - b.mean()  # center for numerical stability
    
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step = 0
    lr = 0.003
    
    a = normalize(np.exp(b - b.max()))
    best_a = a.copy()
    best_M = fast_conv(a).max()
    stall = 0
    restarts = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit - 2:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (time_limit * 0.7))
        temp = temp0 * (temp_min / temp0) ** progress
        
        a = np.exp(b - b.max())
        a = a / a.sum()
        
        conv = fast_conv(a)
        conv_max = conv.max()
        scaled_temp = max(temp * conv_max, 1e-10)
        
        # Softmax weights
        z = (conv - conv_max) / scaled_temp
        exp_z = np.exp(z)
        w = exp_z / exp_z.sum()
        
        # Gradient of weighted sum of conv w.r.t. a
        grad_a = np.zeros(n)
        sig = w > 1e-10
        for k in np.where(sig)[0]:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        
        # Gradient w.r.t. b (softmax Jacobian)
        # d a_i / d b_j = a_i (delta_ij - a_j)
        # grad_b_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
        #           = a_j * grad_a_j - a_j * sum_i grad_a_i * a_i
        ga_dot_a = np.dot(grad_a, a)
        grad_b = a * grad_a - a * ga_dot_a
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * grad_b ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check improvement
        a_new = np.exp(b - b.max())
        a_new = a_new / a_new.sum()
        M = fast_conv(a_new).max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Adapt learning rate and restart
        if stall > 800:
            lr *= 0.6
            stall = 0
            if lr < 1e-9:
                restarts += 1
                if restarts > 30:
                    break
                # Perturb from best
                pert = rng.normal(0, 0.005, n)
                b = np.log(np.maximum(best_a * (1 + pert), 1e-15))
                m = np.zeros(n)
                v = np.zeros(n)
                step = 0
                lr = max(0.002, lr * 5)
                stall = 0
    
    return best_a
