def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s

def lse_gradient(a, c, tau):
    """Gradient of tau*log(sum exp(c_k/tau)) w.r.t. a. Returns the gradient vector."""
    n = len(a)
    c_scaled = c / tau
    c_scaled = c_scaled - c_scaled.max()
    exp_vals = np.exp(c_scaled)
    p = exp_vals / exp_vals.sum()
    
    # g[i] = (2/tau) * sum_k p[k] * a[k-i] (valid k-i in [0,n-1])
    # This equals (2/tau) * conv(p, reverse(a))[n-1 : 2n-1]
    b = a[::-1].copy()
    fft_size = 1 << (len(c) + n - 1 - 1).bit_length()
    Pf = np.fft.rfft(p, fft_size)
    Bf = np.fft.rfft(b, fft_size)
    conv_pb = np.fft.irfft(Pf * Bf, fft_size)
    grad = (2.0 / tau) * conv_pb[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)


def multiplicative_adam(a0, time_limit, verbose=True, seed=42):
    """Log-sum-exp smoothed multiplicative Adam with adaptive tau (cosine annealing)."""
    n = len(a0)
    a = norm_sum1(a0)
    c = fast_conv(a)
    best_a = a.copy()
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    tau_max = max(c.max() * 1.0, 1e-4)
    tau_min = max(c.max() * 0.002, 1e-8)
    lr = 1.0
    beta1 = 0.9
    beta2 = 0.999
    restart_every = 1500
    stall = 0
    cycle_len = 8000  # iterations per cosine cycle
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Cosine annealing for tau
        cycle_pos = (t % cycle_len) / cycle_len
        tau = tau_min + (tau_max - tau_min) * 0.5 * (1.0 + math.cos(2.0 * math.pi * cycle_pos))
        
        g = lse_gradient(a, c, tau)
        
        # Adam moments
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - step * m_c
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        a = a_new
        
        # Restart on stall
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      MulAdam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MulAdam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}, stall={stall}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"      MulAdam done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def pressure_balance(a0, time_limit, n_hot=20, eta_init=0.5, verbose=True, seed=123):
    """Iterative pressure balancing: redistribute mass from high-pressure to low-pressure positions."""
    n = len(a0)

        c = fast_conv(a)
        M = c.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find hot lags
        hot_idx = np.argsort(c)[::-1][:n_hot]
        hot_set = set(int(k) for k in hot_idx)
        
        # Compute pressure at each position
        # p[i] = sum_{k in hot} a[k-i] (valid indices)
        p = np.zeros(n)
        for k in hot_set:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                p[idx] += a[k_int - idx]
        
        # Normalize pressure
        p_mean = p.mean()
        if p_mean < 1e-15:
            break
        
        # Multiplicative pressure balancing update
        eta = eta_init * max(0.1, 1.0 - stall / 500.0)
        ratio = p_mean / np.maximum(p, 1e-15)
        a_new = a * np.power(ratio, eta)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Occasionally mix in random perturbations to escape local optima
        if stall > 100:
            j1 = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if j1 != j2:
                frac = 0.01
                amt = a_new[j1] * frac
                a_new[j1] -= amt
                a_new[j2] += amt
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
        
        a = a_new

            break
        
        if verbose and t % 200 == 0:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
