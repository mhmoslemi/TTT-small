Let me think about the frequency-domain perturbation more carefully:

Given a (sum=1, length n):
1. Compute A = FFT(a, padded to size N >= 2n-1)
2. Compute P = |A|^2 (power spectrum)
3. The autocorrelation is IFFT(P) (approximately, for the circular case)
4. To reduce the peak of the autocorrelation, we want to flatten P
5. But we need to maintain: IFFT(A) >= 0 and sum(IFFT(A)) = 1

Approach: 
- Compute P = |A|^2
- P_new = P^alpha for alpha in (0, 1) (flattening)
- But we can't just change |A| without changing the phase... 
- Alternative: A_new = A * (P / P_new)^(-1/2) * exp(0j) ... no this doesn't work.

Better approach:
- Keep the phase of A: phase_A = A / |A|
- Modify the magnitude: |A_new| = |A|^alpha (for alpha < 1)
- A_new = |A_new| * phase_A
- a_new = IFFT(A_new) (real part)
- Project: a_new = max(a_new, 0), normalize to sum 1

This is a valid approach! By flattening the power spectrum, we reduce the peak of the autocorrelation (since the autocorrelation is the inverse FFT of the power spectrum).

"""

2. Adaptive multiplicative gradient descent (MGD) with multi-peak weighting

import numpy as np
import time
import math


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_gradient_multi_lag(a, lags, weights):
    """Compute the weighted subgradient for multi-lag peaks."""
    n = len(a)
    grad = np.zeros(n)
    for wi, k in zip(weights, lags):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    return grad


def mgd_optimize(a0, time_limit, top_k, eta0, verbose, seed=42):
    """
    Multiplicative Gradient Descent on the simplex with adaptive multi-peak weighting.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = eta0
    patience = 0
    max_patience = 400
    cur_top_k = top_k
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        # Multi-peak weighting: use top lags with exponential decay weighting
        k_max = min(cur_top_k, len(ac))
        sorted_idx = np.argsort(ac)[::-1][:k_max]
        sorted_vals = ac[sorted_idx]
        # Weight by squared values (emphasize dominant peaks)
        w = sorted_vals ** 2.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = compute_gradient_multi_lag(a, sorted_idx, w)
        # Center the gradient (remove mean) for better simplex dynamics
        grad = grad - grad.mean()

        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()

        # Adaptive step size
        if M_new <= M + 1e-12:
            a = a_new
            # Increase eta slightly if making good progress
            eta = min(eta * 1.001, 0.1)
        else:
            eta *= 0.5
            if eta < 0.0003:
                # Restart with different top_k
                cur_top_k = max(5, cur_top_k // 2)
                eta = 0.03
                patience = max(0, patience - 150)
                # Small random perturbation to escape
                a = a + rng.normal(0, 0.001 * a.mean(), n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}, tk={cur_top_k}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def freq_domain_perturb(a0, time_limit, verbose, seed=123):
    """
    Frequency-domain perturbation: flatten the power spectrum to reduce autocorrelation peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    for iteration in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        # Try various alpha values (flattening strength)
        # alpha < 1 flattens the spectrum
        alphas = [0.7, 0.8, 0.85, 0.9, 0.95]
        alpha = alphas[int(rng.integers(len(alphas)))]

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        mag[mag < 1e-30] = 1e-30
        phase = A / mag

        # Flatten magnitude
        log_mag = np.log(mag)
        log_mag_mean = log_mag.mean()
        log_mag_centered = log_mag - log_mag_mean
        # Reduce dynamic range
        log_mag_new = log_mag_mean + alpha * log_mag_centered
        mag_new = np.exp(log_mag_new)

        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try a smaller perturbation
            alpha_small = 0.95 + 0.04 * rng.random()
            log_mag_small = log_mag_mean + alpha_small * log_mag_centered
            mag_small = np.exp(log_mag_small)
            A_small = mag_small * phase
            a_small = np.fft.irfft(A_small, fft_size)[:n]
            a_small = np.maximum(a_small, 0.0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                ac_small = fast_autocorr(a_small)
                if ac_small.max() < best_m:
                    best_m = ac_small.max()
                    best_a = a_small.copy()
                    a = a_small.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Freq done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def basin_hopping(a0, time_limit, verbose, seed=99):
    """
    Basin hopping: alternate between local optimization and large perturbations.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    num_kicks = 0
    max_kicks = 20

    while time.monotonic() - t0 < time_limit and num_kicks < max_kicks:
        # Local optimization (short MGD)
        kick_time_left = time_budget_remaining = time_limit - (time.monotonic() - t0)
        local_time = min(kick_time_left * 0.6, 30.0)
        if local_time > 3:
            a = mgd_optimize(a, local_time, top_k=20, eta0=0.02, verbose=False,
                             seed=seed + num_kicks)
            ac = fast_autocorr(a)
            if ac.max() < best_m:
                best_m = ac.max()
                best_a = a.copy()

        # Kick: large perturbation
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        kick_type = int(rng.integers(4))
        if kick_type == 0:
            # Random smooth perturbation (low-pass noise)
            noise = rng.normal(0, 1, n)
            # Low-pass filter in frequency domain
            N = 1 << (n - 1).bit_length()
            Nf = np.fft.rfft(noise, N)
            cutoff = N // 8
            for i in range(cutoff + 1, len(Nf)):
                Nf[i] *= 0.1
            noise = np.fft.irfft(Nf, N)[:n]
            noise = noise - noise.mean()
            a = a + 0.3 * noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        elif kick_type == 1:
            # Block swap: swap two random blocks
            bsize = max(5, n // 50)
            i1 = int(rng.integers(0, n - 2 * bsize))
            i2 = int(rng.integers(i1 + bsize, n - bsize))
            a[i1:i1 + bsize], a[i2:i2 + bsize] = a[i2:i2 + bsize].copy(), a[i1:i1 + bsize].copy()
        elif kick_type == 2:
            # Frequency domain kick
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag[mag < 1e-30] = 1e-30
            phase = A / mag
            # Add random phase perturbation
            phase_pert = np.exp(1j * rng.normal(0, 0.2, len(phase)))
            A_new = mag * phase * phase_pert
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a = a_new / s
            else:
                a = norm_sum1(a)
        else:
            # Concentration kick: increase contrast
            ac = fast_autocorr(a)
            peak_lag = int(np.argmax(ac))
            # Reduce values that contribute to the peak
            js = max(0, peak_lag - n + 1)
            je = min(n - 1, peak_lag)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[peak_lag - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:max(1, n // 10)]]
            reduce_amt = 0.1 * a[top_contrib].mean()
            a[top_contrib] -= reduce_amt
            a[top_contrib] = np.maximum(a[top_contrib], 0.0)
            # Redistribute to low-contributing positions
            low_contrib = idx[np.argsort(contrib)[:max(1, n // 10)]]
            total_reduce = reduce_amt * len(top_contrib)
            a[low_contrib] += total_reduce / max(1, len(low_contrib))
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # Check improvement
        ac = fast_autocorr(a)
        if ac.max() < best_m:
            best_m = ac.max()
            best_a = a.copy()

        num_kicks += 1
        if verbose and num_kicks % 2 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Basin kick={num_kicks} ({elapsed:.0f}s): J={J:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Basin done ({elapsed:.0f}s): J={J:.6f}, kicks={num_kicks}")
    return best_a


def coord_polish(a0, time_limit, verbose, seed=314):
    """
    Coordinate descent: move mass from peak-contributing positions to low-impact ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find destination: position with minimal total peak contribution
            peak_contrib_total = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib_total[idx2] += a[k2i - idx2]
                    peak_contrib_total[k2i - idx2] += a[idx2]

            # Choose destination with minimum total contribution
            dst = int(np.argmin(peak_contrib_total))
            if dst == src:
                dst = int(np.argmin(np.where(np.arange(n) == src, np.inf, peak_contrib_total)))
            if dst == src:
                continue

            for frac in [0.01, 0.02, 0.04, 0.06]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < M - 1e-15:
                    a = a_trial
                    M = M_trial
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def parametric_init(n, time_limit, verbose):
    """Try many parametric families for initialization."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    results = []

    candidates = []

    # Power laws
    for p in [0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr - center) ** 2 * 0 + (1 + (i_arr - center) ** 2 / (n / 4) ** 2) ** (-p))
        candidates.append((1 + (i_arr - center) ** 2 / (n / 8) ** 2) ** (-p))

    # Gaussian windows
    for sigma_frac in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        candidates.append(np.exp(-((i_arr - center) ** 2) / (2 * sigma ** 2)))

    # Cosine windows
    for beta in [0.5, 1.0, 2.0, 3.0]:
        candidates.append(0.5 * (1 + np.cos(np.pi * (2 * (i_arr - center) / (n - 1)))) ** beta)

    # Sine-based
    for freq in [1, 2, 3]:
        candidates.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))
    candidates.append(np.sin(np.pi * x) ** 4)
    candidates.append(np.sin(np.pi * x) ** 6)

    # Two-hump (anti-symmetric structure)
    for ratio in [0.3, 0.35, 0.4]:
        c1 = ratio
        c2 = 1.0 - ratio
        sigma = 0.15
        candidates.append(np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2)))

    # Cosine with offset (creates asymmetry)
    for offset in [0.3, 0.4, 0.7]:
        for freq in [1, 2]:
            candidates.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * x + 2 * np.pi * offset), 0.01))

    # Exponential decay from center
    for rate in [0.5, 1.0, 2.0, 5.0]:
        candidates.append(np.exp(-rate * np.abs(x - 0.5) * 2))

    best_a = None
    best_score = float('inf')

    for cand in candidates:
        if time.monotonic() - time.limit_start > time_limit:
            break
        cand_norm = norm_sum1(cand)
        ac = fast_autocorr(cand_norm)
        score = 2.0 * n * ac.max()
        if score < best_score:
            best_score = score
            best_a = cand_norm.copy()

    return best_a, best_score

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST MGD: {sc:.6f}")

    # === Phase 3: Frequency domain optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Freq domain (n={n_best}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_out = freq_domain_perturb(norm_sum1(best_seq), t3, verbose=True, seed=123)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Freq: {sc:.6f}")

    # === Phase 4: MGD round 2 with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: MGD r2 (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_out = mgd_optimize(norm_sum1(best_seq), t4, top_k=80, eta0=0.01, verbose=True, seed=77)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST MGD r2: {sc:.6f}")

    # === Phase 5: Basin hopping ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Basin hopping (n={n_best}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_out = basin_hopping(norm_sum1(best_seq), t5, verbose=True, seed=99)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Basin: {sc:.6f}")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord polish (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15:
        a_out = coord_polish(norm_sum1(best_seq), t6, verbose=True, seed=314)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Coord: {sc:.6f}")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Multi-length exploration ===")

    # Try nearby lengths
    length_options = []
    for delta in [-1000, -500, 500, 1000, 2000]:
        nl = n_best + delta
        if 2000 <= nl <= 8000:
            length_options.append(nl)

    for new_n in length_options[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  Trying n={new_n} (current best n={n_best}, score={best_score:.6f})")
        a_long = interpolate_sequence(best_seq, new_n)

        # Short MGD
        t7a = min(remaining * 0.25, 100.0)
        if t7a > 20:
            a_out = mgd_optimize(a_long, t7a, top_k=30, eta0=0.02, verbose=False, seed=55)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                n_best = new_n
                print(f"    *** NEW BEST n={new_n}: {sc:.6f}")
            else:
                # Try freq domain on this
                t7b = min(remaining * 0.08, 30.0)
                if t7b > 10:
                    a_out2 = freq_domain_perturb(a_out, t7b, verbose=False, seed=66)
                    sc2 = evaluate_safe(a_out2)
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_seq = a_out2.copy()
                        n_best = new_n
                        print(f"    *** NEW BEST freq n={new_n}: {sc2:.6f}")

    # === Phase 8: Final intensive MGD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final MGD (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.35, 200.0)
        if t8 > 20:
            a_out = mgd_optimize(norm_sum1(best_seq), t8, top_k=50, eta0=0.005, verbose=True, seed=200)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final MGD: {sc:.6f}")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.50, 120.0)
        if t9 > 10:
            a_out = coord_polish(norm_sum1(best_seq), t9, verbose=True, seed=500)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final coord: {sc:.6f}")

    # === Phase 10: Final freq polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t10 = min(remaining * 0.60, 80.0)
        if t10 > 8:
            a_out = freq_domain_perturb(norm_sum1(best_seq), t10, verbose=False, seed=700)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final freq: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
