def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    n = len(a)
    if n == new_n:

    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return result / s

                print(f"  New length n={new_n}: {score:.6f}")

    # Phase 4: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t4 = min(remaining * 0.5, 150.0)
        a_opt = mirror_descent_optimize(best_seq, t4, top_k=60, eta0=0.01)
        score = evaluate_safe(a_opt)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Final polish: {score:.6f}")

    # Final validation
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
        return [float(x) for x in best_seq]
    else:
        a = np.ones(5000)
        print("Fallback to uniform sequence, n=5000")
        return [float(x) for x in a]
