        improved = False
        for _ in range(20):  # Try up to 20 transfers
            if time.monotonic() - t0 > time_limit:
                break
            src = int(source_candidates[int(rng.integers(len(source_candidates)))])
            dst = int(sink_candidates[int(rng.integers(len(sink_candidates)))])
            if src == dst or a[src] < 1e-12:
                continue
            
            for frac in [0.1, 0.2, 0.3, 0.5]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                M = fast_conv(a_trial).max()
                if M < best_m - 1e-14:
                    a = a_trial
                    best_m = M
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        if not improved:
            stall += 1
            if stall % 50 == 0 and verbose:
                J = 2.0 * n * best_m
                print(f"    CDB {stall}: J={J:.6f}")
    
    return best_a, best_m


def spectral_diversify(a0, n_new, seed=42):
    """Generate a diverse initialization by modifying the spectrum of a0 and resampling."""
    n_old = len(a0)
    a = norm_sum1(a0)
    
    # Upsample/Downsample via FFT
    fft_size_old = 1 << (2 * n_old - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size_old)
    
    # Resample spectrum to new length
    fft_size_new = 1 << (2 * n_new - 1 - 1).bit_length()
    n_freq_old = len(A)
    n_freq_new = fft_size_new // 2 + 1
    
    # Interpolate magnitude, keep phase
    mag_old = np.abs(A)
    phase_old = np.angle(A)
    
    freq_old = np.linspace(0, 1, n_freq_old)
    freq_new = np.linspace(0, 1, n_freq_new)
    mag_new = np.interp(freq_new, freq_old, mag_old)
    phase_new = np.interp(freq_new, freq_old, phase_old)
    
    # Add random phase perturbation for diversity
    rng = np.random.default_rng(seed)
    phase_pert = rng.normal(0, 0.3, n_freq_new)
    phase_new = phase_new + phase_pert
    
    A_new = mag_new * np.exp(1j * phase_new)
    a_new = np.fft.irfft(A_new, fft_size_new)
    a_new = a_new[:n_new]
    a_new = np.maximum(a_new, 0)
    a_new = norm_sum1(a_new)
    
    return a_new


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


# Various initialization functions
def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_increasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((x + 1) ** power)


def make_mixed(n, pos1=0.2, pos2=0.65, width=0.1, ratio=1.0):
    x = np.arange(n) / n
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / (width * 1.5)) ** 2)
    return norm_sum1(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_centered_power(n, power=0.5):
    x_c = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm_sum1(x_c ** (-power))


def make_modulated(n, envelope_power=0.5, freq=3.0, amp=0.3, phase=0.0):
    """Modulated envelope: a_i = envelope * (1 + amp * cos(2*pi*freq*x + phase))"""
    x = np.arange(n, dtype=np.float64) / n
    env = (x * (1 - x)) ** envelope_power  # Bump function
    mod = 1.0 + amp * np.cos(2 * np.pi * freq * x + phase)
    a = env * mod
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_two_gaussian(n, p1=0.3, p2=0.7, w1=0.05, w2=0.05, ratio=1.0):
    x = np.arange(n) / n
    a = np.exp(-((x - p1) / w1) ** 2) + ratio * np.exp(-((x - p2) / w2) ** 2)
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_exponential_decay(n, rate=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm_sum1(np.exp(-rate * x))


def make_symmetric_perturbed(n, base='sinc', bw=5.0, pert_amp=0.1, seed=42):
    """Take a symmetric base and add small asymmetric perturbation."""
    rng = np.random.default_rng(seed)
    if base == 'sinc':
        a = make_sinc(n, bw)
    elif base == 'cos':
        a = make_cosine(n, 1)
    else:
        a = make_centered_power(n, 0.5)
    # Add small random perturbation (breaks symmetry)
    pert = rng.normal(0, pert_amp, n) / n
    a = a + pert
    a = np.maximum(a, 0)
    return norm_sum1(a)
