import numpy as np
import time
from scipy.optimize import linprog
from scipy import signal

def fast_convolve(a):
    """Compute linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_soft_gradient(a, beta, n_top=50):
    """Compute soft-max subgradient of max conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    # Use top-k lags with soft weighting
    k_max = min(n_top, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_max]
    # Soft weights based on exp(beta * conv_value)
    shifted = conv[top_idx] - conv[top_idx[0]]  # shift so max is 0
    w = np.exp(beta * shifted)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        # d(conv[k])/da[j] = 2*a[k-j] for valid k-j
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            idx = np.arange(j_start, j_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    return grad / max(np.abs(grad).max(), 1e-30)

def mgd_step(a, eta, grad):
    """One MGD step."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

def lp_direction(a, peak_frac=0.99, max_constraints=200):
    """
    Find direction g via LP that decreases the peak.
    Maximizes sum(g) subject to g >= 0, sum(g) = 1,
    and conv(a,g)[k] <= 0 at peak positions.
    """
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max()
    
    # Find peak positions
    threshold = M * peak_frac
    peak_positions = np.where(conv_a >= threshold)[0]
    
    # Limit number of constraints
    if len(peak_positions) > max_constraints:
        # Keep the most important ones
        peak_vals = conv_a[peak_positions]
        order = np.argsort(peak_vals)[::-1]
        peak_positions = peak_positions[order[:max_constraints]]
    
    # Build LP
    # Variables: g[0], ..., g[n-1]
    # Objective: minimize -sum(g) (i.e., maximize sum(g))
    c = -np.ones(n)
    
    # Constraints: conv(a, g)[k] <= 0 for k in peak_positions
    # conv(a, g)[k] = sum_i a[i] * g[k-i] for valid i, k-i
    A_ub = np.zeros((len(peak_positions), n))
    for row, k in enumerate(peak_positions):
        ki = int(k)
        for j in range(max(0, ki - n + 1), min(n, ki + 1)):
            A_ub[row, j] = a[ki - j]  # coefficient of g[j] in conv(a,g)[ki]
    
    b_ub = np.zeros(len(peak_positions))
    
    # Equality constraint: sum(g) = 1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def lp_optimize(a0, time_limit, verbose=True):

    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()

        M_new = fast_convolve(a_new).max()

import numpy as np

def fast_convolve(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_soft_gradient(a, beta, n_top=50):
    """Compute soft-max subgradient of max conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    k_max = min(n_top, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_max]
    shifted = conv[top_idx] - conv[top_idx[0]]
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            idx = np.arange(j_start, j_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    gmax = np.abs(grad).max()
    if gmax < 1e-30:
        return np.zeros(n)
    return grad / gmax


def mgd_step(a, eta, grad):
    """One MGD step on the simplex."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, False
    return a_new / s, True


def solve_lp_direction(a, peak_frac=0.995, max_constraints=80):
    """
    Solve LP to find direction g>=0, sum(g)=1, such that conv(a,g)[k]<=0 at peaks.
    Returns g or None if infeasible.
    """
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max()

    if M < 1e-30:
        return None

    threshold = M * peak_frac
    peak_positions = np.where(conv_a >= threshold)[0]

    if len(peak_positions) == 0:
        return None

    if len(peak_positions) > max_constraints:
        peak_vals = conv_a[peak_positions]
        order = np.argsort(peak_vals)[::-1]
        peak_positions = peak_positions[order[:max_constraints]]

    # Build constraint matrix: conv(a,g)[k] = sum_j a[k-j]*g[j] for valid j
    n_cons = len(peak_positions)
    A_ub = np.zeros((n_cons, n))
    for row, k in enumerate(peak_positions):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            A_ub[row, j_start:j_end] = a[ki - np.arange(j_start, j_end)]

    c = -np.ones(n)
    b_ub = np.zeros(n_cons)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

5. Multi-length exploration via interpolation and re-optimization
The LP approach is the key novelty: it finds a feasible direction that provably
reduces all active peaks simultaneously, leading to more stable optimization.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_soft_gradient(a, beta, n_top=50):
    """Compute soft-max subgradient of max conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    k_max = min(n_top, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_max]
    shifted = conv[top_idx] - conv[top_idx[0]]
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            idx = np.arange(j_start, j_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    gmax = np.abs(grad).max()
    if gmax < 1e-30:
        return np.zeros(n)
    return grad / gmax


def mgd_step(a, eta, grad):
    """One MGD step on the simplex."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, False
    return a_new / s, True


def solve_lp_direction(a, peak_frac=0.997, max_constraints=60):
    """
    Solve LP to find direction g>=0, sum(g)=1, with conv(a,g)[k]<=0 at near-peak positions.
    Returns g or None if infeasible.
    """
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max()
    if M < 1e-30:
        return None

    threshold = M * peak_frac
    peak_positions = np.where(conv_a >= threshold)[0]
    if len(peak_positions) == 0:
        return None

    if len(peak_positions) > max_constraints:
        peak_vals = conv_a[peak_positions]
        order = np.argsort(peak_vals)[::-1]
        peak_positions = peak_positions[order[:max_constraints]]

    n_cons = len(peak_positions)
    A_ub = np.zeros((n_cons, n))
    for row, k in enumerate(peak_positions):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            A_ub[row, j_start:j_end] = a[ki - np.arange(j_start, j_end)]

    c = -np.ones(n)
    b_ub = np.zeros(n_cons)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, verbose=True):
    """LP-guided optimization with adaptive mixing parameter."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    iter_count = 0
    t_mix = 0.02
    patience = 0
    max_patience = 800
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        t_lp_start = time.monotonic()
        g = solve_lp_direction(a, peak_frac=0.998, max_constraints=60)
        t_lp = time.monotonic() - t_lp_start

        if g is not None:
            stuck_count = 0
            a_new = (1.0 - t_mix) * a + t_mix * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            M_new = fast_convolve(a_new).max()

            if M_new < M - 1e-16:
                a = a_new
                if t_mix < 0.12:
                    t_mix = min(t_mix * 1.015, 0.12)
            else:
                t_mix *= 0.5
                if t_mix < 0.003:
                    grad = compute_soft_gradient(a, beta=18.0, n_top=40)
                    a_trial, ok = mgd_step(a, 0.006, grad)
                    if ok:
                        M_trial = fast_convolve(a_trial).max()
                        if M_trial < M:
                            a = a_trial
                            t_mix = 0.02
                        else:
                            patience += 50
                    else:
                        break
        else:
            grad = compute_soft_gradient(a, beta=15.0, n_top=35)
            a_new, ok = mgd_step(a, 0.008, grad)
            if not ok:
                break
            M_new = fast_convolve(a_new).max()
            if M_new < M - 1e-15:
                a = a_new
                stuck_count = 0
            else:
                stuck_count += 1
                patience += 20
                if stuck_count > 80:
                    break

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP it={iter_count} ({elapsed:.0f}s): J={J:.6f}, t_mix={t_mix:.4f}, lp_ms={t_lp*1000:.0f}")

    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"    LP done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def mgd_optimize(a0, time_limit, top_k=50, eta0=0.015, verbose=False):
    """MGD with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    patience = 0
    max_patience = 500
    cur_top_k = top_k
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        grad = compute_soft_gradient(a, beta=10.0, n_top=cur_top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break

        a_new, ok = mgd_step(a, eta, grad)
        if not ok:
            break
        M_new = fast_convolve(a_new).max()

        if M_new <= M + 1e-13:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.5
            if eta < 0.0003:
                cur_top_k = max(5, cur_top_k // 2)
                eta = max(eta0 * 0.5, 0.004)
                patience = max(0, patience - 150)

        iter_count += 1
        if verbose and iter_count % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD it={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}, tk={cur_top_k}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def coord_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass from peak-contributing positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_convolve(a)
        M = conv.max()
        n_top = min(12, len(conv))
        top_k = np.argsort(conv)[::-1][:n_top]
        improved = False

        for k in top_k[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            peak_contrib = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2+1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(peak_contrib[candidates])]
            if dst == src:
                continue

            for frac in [0.01, 0.02, 0.035, 0.06]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_convolve(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a
