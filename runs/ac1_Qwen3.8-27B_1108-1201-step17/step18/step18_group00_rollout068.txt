def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(max(len(a), 1)) / max(len(a), 1)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_symmetric(a):
    n = len(a)
    out = np.zeros(n)
    for i in range(n):
        out[i] = (a[i] + a[n - 1 - i]) / 2.0
    return norm_sum1(out)


def compute_soft_grad(a, ac, top_k, beta):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / ws
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - (n - 1))
        i_end = min(n - 1, ki)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_vals = ki - idx
        grad[idx += 0] += wi * 2.0 * a[j_vals]
        m_start = max(0, ki)
        m_end = min(n - 1, ki + n - 1)
        if m_start <= m_end:
            m_idx = np.arange(m_start, m_end + 1)
            grad[m_idx] += wi * 2.0 * a[m_idx - ki]
    return grad

        candidates.append(norm_sum1(c2))
    for w1 in [0.1, 0.2, 0.3, 0.5]:
        c = np.maximum((1.0 - w1) + w1 * (i_arr / max(n - 1, 1)) ** 2, 0.01)
        candidates.append(norm_sum1(c))
        c2 = np.maximum((1.0 - w1) + w1 * (1.0 - i_arr / max(n - 1, 1)) ** 2, 0.01)
        candidates.append(norm_sum1(c2))
    for w1 in [0.1, 0.2, 0.3]:
        c = np.maximum((1.0 - w1) + w1 * (i_arr / max(n - 1, 1)) ** 0.5, 0.01)
        candidates.append(norm_sum1(c))
        c2 = np.maximum((1.0 - w1) + w1 * (1.0 - i_arr / max(n - 1, 1)) ** 0.5, 0.01)
        candidates.append(norm_sum1(c2))
    rng = np.random.default_rng(123)
    for _ in range(120):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 4.0))
        candidates.append(a)
    best_a = None
    best_val = float('inf')
    for c in candidates:
        v = fast_autocorr(c).max()
        if v < best_val:
            best_val = v
            best_a = c.copy()
    return best_a, 2.0 * n * best_val


def check_and_update(a, label, t_global):
    if a is None:
        return None, float('inf'), False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return None, float('inf'), False
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc):
        elapsed = time.monotonic() - t_global
        print("  Candidate [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
        return a_norm, sc, True
    return None, float('inf'), False
