Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Iterative Magnitude Substitution (IMS) with over-relaxation
2. FTRL mirror descent with subgradient
3. Multi-initialization strategy

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def ims_optimize(a0, time_limit, verbose=False):
    """Iterative Magnitude Substitution with over-relaxation."""
    n = len(a0)
    a = norm_simplex(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    omega = 1.0  # relaxation parameter
    target_power = 0.5  # how much to flatten (0 = no change, 1 = full flatten)
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flatten the magnitude
        # DC component is A[0] = sum(a) = 1
        # We want other components to be as close to a target as possible
        mag_dc = mag[0]
        mag_ac = mag[1:]
        target_ac = np.mean(mag_ac)
        
        # Blend current and target
        new_mag = mag.copy()
        new_mag[1:] = (1 - target_power) * mag[1:] + target_power * target_ac
        # Keep DC fixed
        new_mag[0] = mag_dc
        
        # Reconstruct
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        
        # Over-relaxation
        a_new = omega * a_new + (1 - omega) * a
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        
        # Renormalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        c = fast_conv(a_new)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive target_power: increase if improving, decrease if stuck
        if it % 100 == 0:
            if M < best_m * 0.999:
                target_power = min(0.8, target_power * 1.02)
            elif M > best_m * 1.001:
                target_power = max(0.1, target_power * 0.98)
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Iterative Magnitude Substitution (IMS) with adaptive geometric flattening and over-relaxation
2. FTRL mirror descent with subgradient from top autocorrelation peaks

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_max(a):
    """Return only max of convolution a*a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return float(c[:size].max())


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def ims_step(a, fft_size, alpha, omega):
    """Single IMS step: flatten magnitude spectrum, project to non-negative simplex."""
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Compute geometric mean of AC magnitudes (more stable for positive values)
    mag_ac = mag[1:]
    eps = 1e-30
    log_mag = np.log(mag_ac + eps)
    target_ac = np.exp(np.mean(log_mag))
    
    # Geometric interpolation: new_mag = mag^(1-alpha) * target^alpha
    new_mag = np.empty_like(mag)
    new_mag[0] = mag[0]  # Keep DC component fixed
    new_mag[1:] = (mag[1:] + eps) ** (1.0 - alpha) * (target_ac ** alpha)
    
    # Reconstruct
    new_A = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(new_A, fft_size)[:len(a)]
    
    # Over-relaxation
    if omega != 1.0:
        a_new = omega * a_new + (1.0 - omega) * a
    
    # Project to non-negative
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def subgradient_top(a, c, top_k=10, beta=30.0):
    """Compute weighted subgradient using top peaks with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()
    
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        # For self-convolution, gradient is 2*a[k-j] (counting both a_j*a_{k-j} and a_{k-j}*a_j)
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def ftrl_optimize(a0, time_limit, verbose=False, top_k_init=15, top_k_min=5):
    """FTRL mirror descent with annealed subgradient."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv_max(a)
    t0 = time.monotonic()
    
    c0 = 2.0
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, t / 30000.0)
        cur_top_k = max(top_k_min, int(top_k_init * (1 - 0.6 * progress) + top_k_min))
        cur_beta = 15.0 + 35.0 * progress
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        c = fast_conv(y)
        g = subgradient_top(y, c, top_k=cur_top_k, beta=cur_beta)
        
        # Mirror descent step (exponential family)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x = a_new
        a = a_new
        
        # Track best
        if t % 5 == 0:
            M_new = fast_conv_max(a)
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        FTRL t={t} ({elapsed:.0f}s): M={best_m:.8f}")
    
    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions to reduce max autocorrelation."""

    a = norm_simplex(a0)

        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try moving mass to several candidate destinations
            candidates = []
            # Random positions
            for _ in range(3):
                candidates.append(int(rng.integers(n)))
            # Positions far from the peak contributors
            far_positions = np.argsort(a[i_start:i_end])[:2]
            candidates.extend([i_start + int(fp) for fp in far_positions])
            
            for dst in candidates:
                if dst == src or dst < 0 or dst >= n:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv_max(a_trial)
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape local minimum
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv_max(a)
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m


def make_init_cosine(n, mode=1):
    """Cosine-squared initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_init_bowl(n, sharpness=2.0):
    """Bowl shape: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.12):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.03
    return norm_simplex(a)


def make_init_random_smooth(n, seed=0):
    """Random smooth positive sequence via Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.8
    for f in range(1, 20):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.01)
    return norm_simplex(a)


def make_init_ramp(n, power=0.5):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_ideal_spectrum(n):
    """Initialize from an ideal flat-spectrum signal, clipped to be non-negative."""
    fft_size = 1 << (2 * n - 1).bit_length()
    # Start with flat magnitude
    A = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
    A[0] = 1.0  # DC
    # Random phases for AC components
    rng = np.random.default_rng(123)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
    A[1:] = (1.0 / np.sqrt(n)) * np.exp(1j * phases)
    # Enforce conjugate symmetry (already handled by rfft convention)
    a = np.fft.irfft(A, fft_size)[:n]
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        try_update(a_out, f"cd_n{n_best}")
    
    # === Phase 5: Final intensive optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final optimization (n={n_best}) ===")
    if remaining > 30:
        # IMS again on the best
        t5a = min(remaining * 0.4, 200.0)
        if t5a > 15:
            a_in = norm_simplex(best_seq)
            a_out, _ = ims_optimize(a_in, t5a, verbose=False, alpha_init=0.4)
            try_update(a_out, "ims_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t5b = min(remaining * 0.5, 150.0)
            if t5b > 10:
                a_in = norm_simplex(best_seq)
                a_out, _ = ftrl_optimize(a_in, t5b, verbose=True)
                try_update(a_out, "ftrl_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t5c = min(remaining * 0.6, 60.0)
            if t5c > 5:
                a_in = norm_simplex(best_seq)
                a_out, _ = coordinate_descent(a_in, t5c, verbose=False)
                try_update(a_out, "cd_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
