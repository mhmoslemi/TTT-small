def fast_autocorr(a):

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_ramp(n, power=2.0):
    x = np.linspace(0.01, 1.0, n)
    a = x ** power
    return norm_sum1(a)


def make_cosine_asym(n, offset=0.3):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - offset))
    a = np.maximum(a, 0.0) ** 2
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lp_frank_wolfe(a0, time_limit, solver, top_K_init=200, top_K_max=None,
                    verbose=False, warm_K=None):
    """LP-based Frank-Wolfe with adaptive K and backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    if warm_K is not None:
        K = warm_K
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start >= j_end:
                continue
            idx = np.arange(j_start, j_end)
            T_sel[ii, idx] = a[ki - idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 50, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + 50, top_K_max)
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                      0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 5 == 0:
            K = min(K + 40, top_K_max)

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=False, tau_list=None,
                              eta_init=0.5, rng_seed=100):
    """Smoothed projected gradient with multi-tau annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    if tau_list is None:
        tau_list = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]

    eta = eta_init
    stall = 0
    max_stall_per_tau = 200

    for tau in tau_list:
        if time.monotonic() - t0 > time_limit:
            break
        phase_start = time.monotonic()
        phase_limit = (time_limit - (phase_start - t0))
        stall = 0

        while time.monotonic() - phase_start < phase_limit and stall < max_stall_per_tau:
            if time.monotonic() - t0 > time_limit:
                break
            ac = fast_autocorr(a)
            ac_max = ac.max()
            ac_shifted = tau * (ac - ac_max)
            ac_shifted = np.clip(ac_shifted, -500, 0)
            exp_ac = np.exp(ac_shifted)
            p = exp_ac / exp_ac.sum()

            p_cum = np.concatenate([[0.0], np.cumsum(p)])
            k_lo = np.arange(n)
            k_hi = np.minimum(k_lo + n, 2 * n - 1)
            S = p_cum[k_hi] - p_cum[k_lo]
            cross = sp_correlate(p, a, mode='valid')
            g = a * S + cross
            g_tangent = g - g.mean()

            g_norm = np.sqrt(np.sum(g_tangent ** 2))
            if g_norm < 1e-15:
                break
            g_tangent = g_tangent / g_norm

            a_trial = a - eta * g_tangent
            a_new = project_simplex(a_trial)

            m_new = fast_autocorr(a_new).max()
            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                eta = min(eta * 1.01, 10.0)
            else:
                if stall < 50:
                    a = a_new
                stall += 1
                if stall > 50:
                    eta *= 0.8
                if stall > 150:
                    pert = rng.normal(0, 0.003, n) * a
                    a_p = np.maximum(a + pert, 0.0)
                    a = a_p / a_p.sum()
                    stall = 80

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad tau={tau:.0f} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=False, rng_seed=42):
    """Directly attack the peak lag by redistributing mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 50:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_indices = np.argsort(ac)[::-1][:20]

        improved = False
        for k in top_indices[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            total = contrib.sum()
            if total < 1e-20:
                continue

            # Find the pair with the largest contribution
            max_pair = np.argmax(contrib)
            src = idx[max_pair]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-contribution positions to move mass to
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_indices[:5]:
                ki2 = int(k2)
                s2 = max(0, ki2 - n + 1)
                e2 = min(n - 1, ki2)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 3:
                candidates = np.arange(n)

            # Sort by how much they contribute to peaks (want low)
            sens = np.zeros(n)
            for k2 in top_indices[:8]:
                ki2 = int(k2)
                s2 = max(0, ki2 - n + 1)
                e2 = min(n - 1, ki2)
                idx2 = np.arange(s2, e2 + 1)
                sens[idx2] += a[ki2 - idx2]

            top_cands = candidates[np.argsort(sens[candidates])[:5]]

            for dst in top_cands:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShaving ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=60, beta=6.0, c0=5.0,
                 restart_every=5000, verbose=False, rng_seed=42):
    """Accelerated Mirror Descent on simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 5, len(ac))

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.04, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 200)
            cur_beta = max(cur_beta - 1.5, 2.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with focused LP-FW and multi-length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(0.5 + 0.5 * np.cos(2 * np.pi * x))
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Aggressive LP-FW on parent (main workhorse) ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        top_K_init = min(300, 2 * n0 - 1)
        top_K_max = min(3000, 2 * n0 - 1)
        a_out, m_out = lp_frank_wolfe(a_in, t1, solver, verbose=True,
                                        top_K_init=top_K_init, top_K_max=top_K_max)
        # Score in normalized form
        sc_raw = 2.0 * n0 * m_out
        J = sc_raw  # since sum=1, J = 2n*max_b
        print(f"  LP-FW raw: J={J:.6f}")
        try_update(a_out, "lp1")

    # === Phase 2: Smoothed gradient polish ===
    print(f"\n=== Phase 2: SmoothGrad (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t2, verbose=True,
                                             tau_list=[10, 20, 50, 100, 200],
                                             eta_init=0.5, rng_seed=100)
        try_update(a_out, "smooth1")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: PeakShaving (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, "peakshave")

    # === Phase 4: Second LP-FW with larger K ===
    print(f"\n=== Phase 4: LP-FW r2 (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 40 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    warm_K=500)
        try_update(a_out, "lp2")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    trial_lengths = []
    for delta in [-300, -200, -100, 100, 200, 300, 400, -400, 500]:
        new_n = n_best + delta
        if 3000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick AMD
        t5a = min(remaining * 0.3, 150.0)
        if t5a > 20:
            a_out, _ = amd_optimize(a_long, t5a, top_k=60, beta=6.0, c0=5.0,
                                     restart_every=4000, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        # LP-FW on interpolated
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 6000:
            t5b = min(remaining * 0.4, 180.0)
            if t5b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                top_K = min(3000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=False,
                                           top_K_init=min(400, 2*new_n-1),
                                           top_K_max=top_K)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final combined optimization on best ===
    print(f"\n=== Phase 6: Final optimization (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    if remaining > 60:
        t6a = min(remaining * 0.35, 200.0)
        if t6a > 20:
            a_in = norm_sum1(best_seq)
            top_K_max = min(6000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t6a, solver, verbose=True,
                                        top_K_init=min(600, 2*n_best-1),
                                        top_K_max=top_K_max, warm_K=800)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.3, 100.0)
            if t6b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = smooth_gradient_optimize(a_in, t6b, verbose=False,
                                                     tau_list=[50, 100, 200, 400],
                                                     eta_init=0.3, rng_seed=500)
                try_update(a_out, "smooth_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6c = min(remaining * 0.3, 80.0)
            if t6c > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = peak_shaving(a_in, t6c, verbose=False, rng_seed=789)
                try_update(a_out, "peak_final")

    # === Phase 7: Last LP-FW squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30 and n_best <= 8000:
        print(f"\n=== Phase 7: Final LP squeeze (remaining={remaining:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        t7 = remaining - 10
        if t7 > 15:
            top_K_max = min(8000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=False,
                                        top_K_init=min(800, 2*n_best-1),
                                        top_K_max=top_K_max, warm_K=1000)
            try_update(a_out, "lp_last")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(0.5 + 0.5 * np.cos(2 * np.pi * x))
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Ensure we return a list of floats
    return [float(x) for x in best_seq]
