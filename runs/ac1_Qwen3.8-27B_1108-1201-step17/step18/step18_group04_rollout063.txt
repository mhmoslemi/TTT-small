Now, the question is: can we do even better than 4/3?

The theoretical lower bound is 2n/(2n-1) ≈ 1. To approach this, we'd need the convolution to be nearly uniform. This is related to "difference sets" in combinatorics and "Barker sequences" in signal processing.

For continuous analogs, we want a function f on [0,1] such that the autocorrelation ∫ f(x)f(x+t) dx is as constant as possible. This is related to "flat spectrum" signals.

One approach: use a Fourier-based method. If F is the Fourier transform of a, then conv(a,a) in the time domain corresponds to |F|^2 in the frequency domain (for the circular convolution). For linear convolution, it's more complex, but the idea is similar.

If we want conv(a,a) to be constant (= (sum a)^2/(2n-1)), then in the frequency domain, |F(ω)|^2 should be the Fourier transform of a constant, which is a delta function at 0. This means f should be a constant (uniform), which gives a triangular autocorrelation, not a constant one. So we can't achieve a constant autocorrelation with a real-valued signal.

The best we can do is minimize the variance of the autocorrelation. This is related to "minimum variance" or "flat spectrum" problems.

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s

    conv = fast_conv(a)

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def coord_redistribute(a0, time_limit, seed=42, verbose=False):
    """
    Coordinate descent: move mass from high-blame to low-blame positions.
    Focuses on reducing the top convolution peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        conv = fast_conv(a)
        conv_max = conv.max()

        # Find top peaks
        n_peaks = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        # Compute blame vector (how much each position contributes to peaks)
        blame = np.zeros(n)
        for k in peak_indices:
            k_int = int(k)
            lo = max(0, k_int - n + 1)
            hi = min(n - 1, k_int)
            idx = np.arange(lo, hi + 1)
            blame[idx] += a[k_int - idx]

        # Find positions with high blame (sources) and low blame (destinations)
        # Sources: positions that contribute a lot to peaks
        # We want to REDUCE sources and INCREASE destinations

        # Top sources by blame * value (actual contribution)
        contrib = blame * a
        n_src = max(3, min(10, n // 200))
        top_src = np.argsort(contrib)[::-1][:n_src]

        # Destinations: low blame positions (safe to increase)
        n_dst = max(5, min(15, n // 100))
        low_blame = np.argsort(blame)[:max(2 * n_dst, n // 50)]
        top_dst = low_blame[np.argsort(a[low_blame])]  # prefer positions already having some mass

        improved = False
        for src_i in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_int = int(src_i)
            if a[src_int] < 1e-10:
                continue

            for dst_i in list(top_dst)[:12]:
                dst_int = int(dst_i)
                if dst_int == src_int:
                    continue

                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    amt = a[src_int] * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src_int] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

                if improved:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def alternating_lp(a0, time_limit, n_iters=None, seed=42, verbose=False):

            A = np.fft.rfft(a, fft_size)
            B = np.fft.rfft(b, fft_size)
            conv_ab = np.fft.irfft(A * B, fft_size)[:size]

            violations = conv_ab - M
            if violations.max() < 1e-10:
                break

            # Find the most violated constraint
            k_bad = int(np.argmax(violations))
            # Reduce b entries that contribute most to this violation
            lo = max(0, k_bad - n + 1)
            hi = min(n - 1, k_bad)
            idx = np.arange(lo, hi + 1)
            contrib = a[k_bad - idx] * b[idx]
            # Reduce the largest contributor
            j_red = idx[np.argmax(contrib)]
            reduction = min(b[j_red] * 0.5, (conv_ab[k_bad] - M) / max(a[k_bad - j_red], 1e-15))
            b[j_red] = max(0, b[j_red] - reduction)

        s_b = b.sum()
        if s_b < 1e-10:
            b = np.ones(n) / n
            s_b = 1.0
        b = b / s_b

        # Blend
        t_blend = 0.3
        a_new = (1 - t_blend) * a + t_blend * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_m


def parametric_scan(time_limit, base_n=4000):
    """Scan over parametric families to find good starting points."""
    t0 = time.monotonic()
    best_a = None
    best_score = float('inf')

    candidates = []
    labels = []

    # Power ramps
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        candidates.append(make_power_ramp(base_n, alpha))
        labels.append(f"pow_{alpha}")

    # Exponential ramps
    for beta in [1.0, 2.0, 3.0, 5.0]:
        candidates.append(make_exp_ramp(base_n, beta))
        labels.append(f"exp_{beta}")

    # Log ramp
    candidates.append(make_log_ramp(base_n))
    labels.append("log")

    # Sine ramps
    for mode in [0.5, 1.0, 1.5]:
        candidates.append(make_sine_ramp(base_n, mode))
        labels.append(f"sin_{mode}")

    # Two phase
    for frac in [0.2, 0.3, 0.5]:
        candidates.append(make_two_phase(base_n, frac))
        labels.append(f"2ph_{frac}")

    # Reverse ramps (decreasing)
    for alpha in [1.0, 1.5, 2.0]:
        a = make_power_ramp(base_n, alpha)
        candidates.append(a[::-1])
        labels.append(f"rpow_{alpha}")

    for a, lbl in zip(candidates, labels):
        if time.monotonic() - t0 > time_limit:
            break
        score = compute_obj(a)
        a_list = [float(v) for v in a]
        try:
            sc = evaluate_sequence(a_list)
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    Parametric [{lbl}]: {sc:.6f}")

    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    global _best_tracker
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    # Phase 1: Parametric scan (150s)
    print(f"\n=== Phase 1: Parametric Scan ===")
    t_p1 = min(150.0, time_budget)
    if _best_tracker['seq'] is None:
        t_p1 = min(200.0, time_budget)

    best_param_a = None
    best_param_score = float('inf')
    if _best_tracker['seq'] is not None:
        best_param_a = norm_sum1(_best_tracker['seq'])
        best_param_score = _best_tracker['score']
    else:
        best_param_a, best_param_score = parametric_scan(t_p1, base_n=4000)
        if best_param_a is not None:
            update_best(best_param_a, "parametric")

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"  After Phase 1: best={_best_tracker['score']:.6f}, remaining={remaining:.0f}s")

    # Phase 2: MD optimization on best candidate (500s)
    print(f"\n=== Phase 2: Mirror Descent ===")
    t_p2 = min(500.0, remaining * 0.35)
    if t_p2 > 30:
        n0 = len(best_param_a) if best_param_a is not None else 4000
        # Try multiple initializations
        inits = []
        labels = []
        if best_param_a is not None:
            inits.append(best_param_a)
            labels.append("best_param")
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_power_ramp(n0, 1.0))
        labels.append("ramp1")
        inits.append(make_power_ramp(n0, 1.5))
        labels.append("ramp15")
        inits.append(make_exp_ramp(n0, 2.0))
        labels.append("exp2")

        t_per = t_p2 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  MD v2 '{lbl}' t={t_avail:.0f}s")
            a_out, _ = md_optimize_v2(init, t_avail, top_k=3, eta0=1.5, seed=100 + idx)
            update_best(a_out, f"md2_{lbl}")

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"  After Phase 2: best={_best_tracker['score']:.6f}, remaining={remaining:.0f}s")

    # Phase 3: Coordinate redistribution (300s)
    print(f"\n=== Phase 3: Coord Redistribution ===")
    t_p3 = min(300.0, remaining * 0.25)
    if t_p3 > 20:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = coord_redistribute(a_in, t_p3, seed=42)
        update_best(a_out, "coord1")

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"  After Phase 3: best={_best_tracker['score']:.6f}, remaining={remaining:.0f}s")

    # Phase 4: MD v1 with different params (300s)
    print(f"\n=== Phase 4: MD v1 Refinement ===")
    t_p4 = min(300.0, remaining * 0.30)
    if t_p4 > 20:
        a_in = norm_sum1(_best_tracker['seq'])
        # Try different top_k and eta
        for tk, et in [(3, 0.8), (5, 0.5), (10, 0.3)]:
            t_avail = t_p4 / 3
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_avail = min(t_avail, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  MD v1 (tk={tk}, eta={et}) t={t_avail:.0f}s")
            a_out, _ = md_optimize(a_in, t_avail, top_k=tk, eta0=et, seed=200 + tk)
            update_best(a_out, f"md1_{tk}_{et}")

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"  After Phase 4: best={_best_tracker['score']:.6f}, remaining={remaining:.0f}s")

    # Phase 5: Multi-length search (200s)
    print(f"\n=== Phase 5: Multi-Length ===")
    t_p5 = min(200.0, remaining * 0.30)
    if t_p5 > 30:
        n_best = len(_best_tracker['seq'])
        for delta in [-1000, -500, 500, 1000]:
            new_n = n_best + delta
            if 2000 <= new_n <= 8000:
                if time.monotonic() - t_global > time_budget - 30:
                    break
                a_interp = interpolate_seq(_best_tracker['seq'], new_n)
                update_best(a_interp, f"intp{new_n}")
                t_avail = min(40.0, time_budget - 30 - (time.monotonic() - t_global))
                if t_avail > 10:
                    a_out, _ = md_optimize_v2(a_interp, t_avail, top_k=3, eta0=1.0, seed=500)
                    update_best(a_out, f"md{new_n}")

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"  After Phase 5: best={_best_tracker['score']:.6f}, remaining={remaining:.0f}s")

    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")
    if remaining > 50:
        a_in = norm_sum1(_best_tracker['seq'])
        t_p6a = min(remaining * 0.5, 200.0)
        if t_p6a > 10:
            a_out, _ = md_optimize_v2(a_in, t_p6a, top_k=2, eta0=0.5, seed=9999)
            update_best(a_out, "final_md")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_p6b = min(remaining * 0.6, 150.0)
            if t_p6b > 10:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = coord_redistribute(a_in, t_p6b, seed=7777)
                update_best(a_out, "final_cd")

    # Final
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        _best_tracker['seq'] = make_power_ramp(n_fb, 1.0)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
