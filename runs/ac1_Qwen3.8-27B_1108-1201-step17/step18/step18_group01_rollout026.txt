"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Threshold-based subgradient mirror descent (multiplicative updates on simplex)
   with adaptive fraction threshold and backtracking step size
2. Nesterov-momentum-augmented multiplicative gradient descent
3. Fast coordinate descent with peak-targeted mass transfers
4. Multi-length exploration via interpolation
5. Aggressive final polishing with tight thresholds

Key insight: Using all lags above a fraction of max (rather than fixed top-k)
gives a richer subgradient signal, especially when multiple peaks are near-ties.
"""

import numpy as np
import time
import math


def fast_autocorr(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def grad_threshold(a, c, frac):
    """
    Compute subgradient using all lags above frac*max(c).
    Weight by excess above threshold.
    """
    n = len(a)
    M = c.max()
    if M < 1e-30:
        return np.zeros(n)
    threshold = frac * M
    active = c > threshold
    if not active.any():
        top_idx = np.argsort(c)[::-1][:10]
        active = np.zeros(len(c), dtype=bool)
        active[top_idx] = True
    excess = np.where(active, c - threshold, 0.0)
    w_sum = excess.sum()
    if w_sum < 1e-30:
        active_idx = np.where(active)[0]
        w = np.ones(len(active_idx)) / len(active_idx)
    else:
        w = excess[active] / w_sum
    active_idx = np.where(active)[0]
    grad = np.zeros(n)
    for wi, k in zip(w, active_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    return grad


def mgd_optimize(a0, time_limit, frac_init=0.85, eta0=0.03, verbose=True):
    """
    Mirror gradient descent with threshold-based subgradient.
    Adaptive fraction threshold and backtracking step size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = eta0
    frac_current = frac_init
    patience = 0
    max_patience = 400
    iter_count = 0
    momentum = np.zeros(n)
    momentum_coeff = 0.3

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        M = c.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            frac_current = max(0.5, frac_current - 0.1)
            eta = max(0.008, eta0 * 0.5)
            momentum *= 0.0
            patience = 0
            max_patience = 300
            continue

        grad = grad_threshold(a, c, frac_current)
        grad = grad - grad.mean()

        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        # Momentum
        momentum = momentum_coeff * momentum + (1 - momentum_coeff) * grad

        # Multiplicative mirror step
        a_trial = a * np.exp(-eta * momentum)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            break
        a_trial = a_trial / s

        c_trial = fast_autocorr(a_trial)
        M_trial = c_trial.max()

        if M_trial <= M + 1e-12:
            a = a_trial
            eta = min(eta * 1.001, 0.08)
        else:
            eta *= 0.4
            momentum *= 0.3
            if eta < 0.0002:
                eta = 0.01
                frac_current = max(0.5, frac_current - 0.05)
                patience = max(0, patience - 150)

        iter_count += 1
        if verbose and iter_count % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}, frac={frac_current:.2f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def mgd_softmax(a0, time_limit, beta_init=50.0, beta_end=500.0, eta0=0.02, verbose=True):
    """
    Mirror gradient descent using softmax approximation of max.
    Anneals beta from low (smooth) to high (sharp).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta0
    patience = 0
    max_patience = 350
    iter_count = 0
    momentum = np.zeros(n)
    mom_coeff = 0.2

    total_time_est = time_limit

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / total_time_est)
        beta = beta_init + (beta_end - beta_init) * (elapsed_frac ** 0.7)

        c = fast_autocorr(a)
        M = c.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            eta = max(0.005, eta0 * 0.4)
            momentum *= 0.0
            patience = 0
            continue

        # Softmax weights
        c_shifted = c - c.max()
        exp_c = np.exp(beta * c_shifted)
        p = exp_c / exp_c.sum()

        # Gradient: grad[i] = sum_k p[k] * a[k-i] * 2
        grad = np.zeros(n)
        # Only process significant lags
        thresh = 1.0 / (len(c) * 50)
        significant = p > thresh
        sig_idx = np.where(significant)[0]
        for k in sig_idx:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * p[k] * a[ki - idx]

        grad = grad - grad.mean()
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        momentum = mom_coeff * momentum + (1 - mom_coeff) * grad

        a_trial = a * np.exp(-eta * momentum)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            break
        a_trial = a_trial / s

        c_trial = fast_autocorr(a_trial)
        M_trial = c_trial.max()

        if M_trial <= M + 1e-12:
            a = a_trial
            eta = min(eta * 1.002, 0.06)
        else:
            eta *= 0.4
            momentum *= 0.3
            if eta < 0.0002:
                eta = 0.008
                patience = max(0, patience - 100)

        iter_count += 1
        if verbose and iter_count % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SoftMGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.0f}, eta={eta:.5f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    SoftMGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def coord_desc(a0, time_limit, verbose=True):
    """
    Coordinate descent: move mass from peak-contributing positions to low-impact ones.
    Uses targeted selections based on current peak structure.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    max_stall = 80
    sweep = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        c = fast_autocorr(a)
        M = c.max()
        improved = False

        top_k = np.argsort(c)[::-1][:min(30, len(c))]

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-15:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-impact destination
            # Score each position by total peak contribution
            peak_contrib = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]

            in_peak = peak_contrib > np.median(peak_contrib)
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)

            # Pick from candidates with lowest peak contribution
            cand_subset = candidates[:min(500, len(candidates))]
            dst_scores = peak_contrib[cand_subset]
            dst = cand_subset[np.argmin(dst_scores)]
            if dst == src:
                dst = int(rng.integers(n))
                if dst == src:
                    continue

            for frac in [0.02, 0.04, 0.06, 0.08]:
                amt = src_val * frac
                a_try = a.copy()
                a_try[src] -= amt
                a_try[dst] += amt
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t < 1e-15:
                    continue
                a_try = a_try / s_t
                M_try = fast_autocorr(a_try).max()
                if M_try < M - 1e-14:
                    a = a_try
                    M = M_try
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord sweep={sweep} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}, sweeps={sweep}")
    return best_a


def block_perturb(a0, time_limit, block_size=20, verbose=True):
    """
    Block-level perturbation: modify small contiguous blocks.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M = c.max()
        improved = False
        top_lags = np.argsort(c)[::-1][:8]

        for attempt in range(40):
            if time.monotonic() - t0 > time_limit:
                break

            if attempt < 8:
                k_pick = int(top_lags[attempt % len(top_lags)])
                js = max(0, k_pick - n + 1)
                je = min(n - 1, k_pick)
                center = (js + je) // 2
                start = max(0, min(n - block_size, center - block_size // 2 + int(rng.integers(-10, 10))))
            else:
                start = int(rng.integers(0, max(1, n - block_size + 1)))

            if bs < 3:
                continue

            mean_b = block.mean() + 1e-10
            scale = rng.uniform(0.03, 0.10)
            pert = rng.normal(0, scale * mean_b, bs)
            new_block = np.maximum(block + pert, 0.0)

            a_try = a.copy()
            a_try[start:end] = new_block
            a_try = a_try / a_try.sum()
            M_try = fast_autocorr(a_try).max()

            if M_try < M - 1e-14:
                a = a_try
                M = M_try
                if M < best_m:

                best_seq = a_out.copy()
                print(f"  *** FINAL COORD: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
