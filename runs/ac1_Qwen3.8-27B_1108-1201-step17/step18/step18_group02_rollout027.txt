    L = 2*n - 1
    # Actually, use next power of 2 for FFT efficiency
    L = 1 << (2*n-1).bit_length()
    
    # Random phase initialization
    a = rng.uniform(0, 1, n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    
    for it in range(iters):
        A = np.fft.rfft(a, L)
        phase = np.angle(A)
        magnitude = 1.0 / np.sqrt(L)  # constant magnitude
        A_new = magnitude * np.exp(1j * phase)
        # Ensure conjugate symmetry for real output
        a_full = np.fft.irfft(A_new, L)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    
    return a

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def objective_score(a):
    """Compute 2*n*max(conv)/sum^2 for a normalized sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def lse_gradient(a, ac, tau):
    """Compute gradient of tau*log(sum exp(ac_k/tau)) w.r.t. a using FFT."""
    n = len(a)
    ac_scaled = ac / tau
    ac_scaled -= ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()

    a_flip = a[::-1].astype(np.float64)
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)


def spectral_init(n, iters=500, seed=42):
    """Gerchberg-Saxton style initialization for near-flat autocorrelation."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()

    # Start with a smooth bump
    x = np.linspace(0, 1, n)
    a = np.exp(-4.0 * (x - 0.5) ** 2)
    a = np.maximum(a, 0.01)
    a = a / a.sum()

    best_a = a.copy()
    best_score = objective_score(best_a)

    for it in range(iters):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        # Target: flat magnitude
        target_mag = np.ones_like(mag) / np.sqrt(fft_size)
        # Blend: don't fully project, use a convex combination
        new_mag = 0.7 * target_mag + 0.3 * mag
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_full = np.fft.irfft(A_new, fft_size)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 50 == 0:
            sc = objective_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a


def smooth_adam_v2(a0, time_limit, tau_decay=0.9997, lr=0.8, beta1=0.9, beta2=0.999,
                   restart_every=2000, verbose=True, seed=123):
    """Adaptive multiplicative Adam with log-sum-exp smoothing."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    tau = max(ac.max() * 0.8, 1e-6)
    tau_min = max(ac.max() * 0.001, 1e-10)

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    t = 0

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive tau: decrease faster when making progress
        if stall < 10:
            tau = max(tau * tau_decay * 0.999, tau_min)
        else:
            tau = max(tau * tau_decay, tau_min)

        g = lse_gradient(a, ac, tau)

        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)

        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2

        # Adaptive learning rate: larger when tau is large (smoother), smaller when sharp
        effective_lr = lr * min(tau / (ac.max() + 1e-10), 2.0)
        effective_lr = max(effective_lr, 0.1)

        step = effective_lr / (np.sqrt(v_c) + 1e-8)

        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new

        if stall > restart_every:
            stall = 0
            # More aggressive restart
            noise_scale = 0.05 + 0.05 * (stall / restart_every)
            pert = rng.normal(0, noise_scale, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Adam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}, stall={stall}")

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    Adam done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def lp_frank_wolfe_v3(a0, time_limit, solver, initial_K=300, max_K=6000,
                      add_per_iter=80, verbose=True):
    """Enhanced LP Frank-Wolfe with adaptive constraint management and warm-start scaling."""
    n = len(a0)
    a = norm_sum1(a0)
    ac_init = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac_init.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    # Initialize with top peaks
    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    lp_fail_count = 0
    last_lp_time = time.monotonic()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            # Keep tightest constraints and some spread
            keep_tight = np.argsort(margins)[:max_K * 2 // 3]
            # Keep some from the spread
            remaining = [k for k in range(len(margins)) if k not in set(keep_tight)]
            keep_spread = remaining[len(remaining) // 4: len(remaining) * 3 // 4]
            keep = np.concatenate([keep_tight, keep_spread])[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 8 and len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 12:
                constraint_set = set(int(k) for k in constraint_list[:max(200, initial_K // 2)])
                # Also add the current violation
                viol_idx = np.argmax(ac)
                constraint_set.add(int(viol_idx))
            continue

        lp_fail_count = 0
        last_lp_time = time.monotonic()
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Step size
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            lp_time = time.monotonic() - last_lp_time
            print(f"    LP-FW3 iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, lp_t={lp_time:.2f}s")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW3 done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_descent_v3(a0, time_limit, rng_seed=42, verbose=True):
    """Peak-targeted coordinate descent with adaptive transfer amounts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        improved = False
        # Focus on the top peaks
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        for k in top_k[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue

            # Find the position that contributes most to this peak
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find good destinations: positions that minimize impact on all top peaks
            scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2] * 0.5
                    scores[k2i - idx2] += a[idx2] * 0.5

            # Prefer positions with low score (low impact on peaks)
            # Also prefer positions that are "far" from current peak
            dist_to_peak = np.abs(np.arange(n) - (k_int // 2))
            combined_score = scores + 0.1 * dist_to_peak / n

            # Exclude positions too close to source
            excluded = np.abs(np.arange(n) - src) < 3
            candidates_mask = ~excluded
            if candidates_mask.sum() == 0:
                candidates_mask = np.ones(n, dtype=bool)
                candidates_mask[src] = False

            cand_indices = np.where(candidates_mask)[0]
            cand_scores = combined_score[cand_indices]
            # Pick from the lowest scoring candidates with some randomness
            n_cand = min(20, len(cand_indices))
            if n_cand > 0:
                thresh = np.sort(cand_scores)[:n_cand].max()
                eligible = cand_indices[cand_scores <= thresh]
                dst = eligible[rng.integers(len(eligible))]
            else:
                dst = cand_indices[rng.integers(len(cand_indices))]

            if dst == src:
                continue

            # Try progressive transfer fractions
            found = False
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 80:
                # Random exploration
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            if stall % 50 == 1:
                # Random perturbation of multiple positions
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord3 ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def multi_peak_redistribution(a, num_peaks=8, width_factor=0.3):
    """Create a multi-peaked distribution to spread autocorrelation energy."""
    n = len(a)
    x = np.linspace(0, 1, n)
    result = np.zeros(n)
    for i in range(num_peaks):
        center = (i + 0.5) / num_peaks + 0.05 * np.sin(i * 2.1)
        sigma = width_factor / num_peaks
        result += np.exp(-((x - center) ** 2) / (2 * sigma * sigma))
    result = np.maximum(result, 0.01)
    return result / result.sum()


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-resolution strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Small problem for structure discovery (n=1000) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Small problem n=1000 ===")
    n_small = 1000
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 20:
        # Spectral init
        a_spec = spectral_init(n_small, iters=300, seed=7)
        check_and_update(a_spec, "spectral_1000")

        # Multi-peak init
        a_mp = multi_peak_redistribution(a_spec, num_peaks=10, width_factor=0.4)
        check_and_update(a_mp, "multipeak_1000")

        # Quick Adam
        if best_seq is not None and len(best_seq) == n_small:
            a_out, _ = smooth_adam_v2(best_seq, min(t1 * 0.6, 80), tau_decay=0.9995, lr=1.0,
                                      restart_every=1000, verbose=False, seed=55)
            check_and_update(a_out, "adam_small")
        else:
            a_out, _ = smooth_adam_v2(a_spec, min(t1 * 0.6, 80), tau_decay=0.9995, lr=1.0,
                                      restart_every=1000, verbose=False, seed=55)
            check_and_update(a_out, "adam_small")

    # === Phase 2: Medium problem (n=3000) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Medium problem n=3000 ===")
    n_med = 3000
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        # Get starting point for medium: interpolate parent or use best small
        if best_seq is not None:
            a_med = interpolate_sequence(best_seq, n_med)
            check_and_update(a_med, f"interp_med")

            # Spectral init for medium
            a_spec_med = spectral_init(n_med, iters=200, seed=13)
            check_and_update(a_spec_med, "spectral_med")

            # Use the better one
            ac_med = fast_autocorr(a_med)
            ac_spec = fast_autocorr(a_spec_med)
            if ac_spec.max() / len(a_spec_med) < ac_med.max() / len(a_med):
                a_med = a_spec_med
                check_and_update(a_med, "spectral_med_wins")

            # Adam on medium
            t2a = min(t2 * 0.4, 150.0)
            if t2a > 20:
                a_out, _ = smooth_adam_v2(a_med, t2a, tau_decay=0.9997, lr=0.7,
                                          restart_every=2000, verbose=False, seed=77)
                check_and_update(a_out, "adam_med")

            # LP-FW on medium
            remaining = time_budget - (time.monotonic() - t_global)
            t2b = min(remaining * 0.25, 200.0)
            if t2b > 30:
                a_in = norm_sum1(best_seq) if len(best_seq) == n_med else a_med
                a_out, _ = lp_frank_wolfe_v3(a_in, t2b, solver, initial_K=200,
                                             max_K=2000, add_per_iter=50, verbose=False)
                check_and_update(a_out, "lp_med")
        else:
            a_med = spectral_init(n_med, iters=200, seed=13)
            check_and_update(a_med, "spectral_med_default")
            a_out, _ = smooth_adam_v2(a_med, min(t2 * 0.5, 200), tau_decay=0.9997, lr=0.7,
                                      restart_every=2000, verbose=False, seed=77)
            check_and_update(a_out, "adam_med_default")

    # === Phase 3: Large problem (n=5200) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Large problem n=5200 ===")
    n_lg = 5200
    t3 = min(remaining * 0.40, 700.0)
    if t3 > 60:
        # Interpolate best to large
        a_lg = interpolate_sequence(best_seq, n_lg)
        check_and_update(a_lg, "interp_lg")

        # Also try spectral init at large scale
        a_spec_lg = spectral_init(n_lg, iters=200, seed=23)
        check_and_update(a_spec_lg, "spectral_lg")

        # Pick the better starting point
        ac_lg = fast_autocorr(a_lg)
        ac_spec_lg = fast_autocorr(a_spec_lg)
        if ac_spec_lg.max() / (ac_spec_lg.sum() / (2 * n_lg - 1)) < ac_lg.max() / (ac_lg.sum() / (2 * n_lg - 1)):
            a_lg = a_spec_lg
            print("  Spectral init wins for large")

        # Adam refinement
        remaining = time_budget - (time.monotonic() - t_global)
        t3a = min(remaining * 0.25, 300.0)
        if t3a > 30:
            a_out, _ = smooth_adam_v2(a_lg, t3a, tau_decay=0.9998, lr=0.6,
                                      restart_every=3000, verbose=True, seed=99)
            check_and_update(a_out, "adam_lg")

        # LP-FW on large
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.40, 400.0)
        if t3b > 60:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v3(a_in, t3b, solver, initial_K=400,
                                         max_K=5000, add_per_iter=60, verbose=True)
            check_and_update(a_out, "lp_lg")

    # === Phase 4: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Final polish ===")
    n_best = len(best_seq) if best_seq is not None else n_lg

    if remaining > 40:
        # Coordinate descent
        t4a = min(remaining * 0.4, 200.0)
        if t4a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_v3(a_in, t4a, rng_seed=271, verbose=True)
            check_and_update(a_out, "coord_final")

        # Another LP-FW round
        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.4, 250.0)
        if t4b > 30 and n_best <= 5500:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v3(a_in, t4b, solver, initial_K=600,
                                         max_K=6000, add_per_iter=80, verbose=True)
            check_and_update(a_out, "lp_polish")

        # Final coord
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t4c = min(remaining * 0.5, 100.0)
            if t4c > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coord_descent_v3(a_in, t4c, rng_seed=555, verbose=False)
                check_and_update(a_out, "coord_polish2")

    # === Phase 5: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_lg
    if remaining > 120 and n_best < 8000:
        n_try = min(n_best + 1000, 8000)
        print(f"\n=== Phase 5: Try n={n_try} ===")
        a_try = interpolate_sequence(best_seq, n_try)
        check_and_update(a_try, f"interp_{n_try}")

        t5 = min(remaining * 0.5, 300.0)
        if t5 > 30:
            a_out, _ = smooth_adam_v2(a_try, min(t5 * 0.5, 150), tau_decay=0.9998, lr=0.5,
                                      restart_every=3000, verbose=False, seed=314)
            check_and_update(a_out, f"adam_{n_try}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60 and n_try <= 6000:
                t5b = min(remaining * 0.4, 200.0)
                if t5b > 30:
                    a_in = norm_sum1(best_seq)
                    a_out, _ = lp_frank_wolfe_v3(a_in, t5b, solver, initial_K=400,
                                                 max_K=4000, add_per_iter=50, verbose=False)
                    check_and_update(a_out, f"lp_{n_try}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
