def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, p):
    """Compute gradient of log-sum-exp smoothed max convolution.
    grad[i] = 2 * correlate(p, a, 'valid')[i] - p[2i] * a[i]
    """
    n = len(a)
    cross = sp_correlate(p, a, mode='valid')
    # cross[i] = sum_j p[i+j] * a[j], length n
    diag_correction = np.zeros(n)
    two_i = 2 * np.arange(n)
    valid = two_i < len(p)
    diag_correction[valid] = p[two_i[valid]] * a[valid]
    grad = 2.0 * cross - diag_correction
    return grad

def nesterov_smoothed(a0, time_limit, tau_init=10.0, tau_final=500.0,
                       verbose=True, rng_seed=42):
    """Nesterov accelerated smoothed gradient descent with backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    eta = 1.0
    stall = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_init * (tau_final / tau_init) ** progress
        t_iter = iter_count + 1.0

        # Nesterov momentum
        beta_t = (t_iter - 1.0) / (t_iter + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute smoothed gradient
        ac = fast_autocorr(y)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        grad = compute_smoothed_gradient(y, p)
        # Project gradient onto tangent space of simplex
        grad = grad - grad.mean()

        # Backtracking line search
        M_curr = fast_autocorr(y).max()
        found_step = False
        for trial_eta in [eta, eta * 0.5, eta * 0.25, eta * 0.1, eta * 0.05]:
            a_trial = y - trial_eta * grad
            a_new = project_simplex(a_trial)
            if a_new.sum() < 1e-15:
                continue
            M_new = fast_autocorr(a_new).max()
            if M_new <= M_curr + 1e-15:
                found_step = True
                break
        if not found_step:
            a_new = project_simplex(y - eta * 0.01 * grad)
            eta *= 0.5

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.05, 50.0)
        else:
            stall += 1
            if stall % 50 == 0 and stall > 100:
                # Random perturbation
                pert = rng.normal(0, 0.005, n) * x
                x_p = x + pert
                x_p = np.maximum(x_p, 1e-10)
                x_p = x_p / x_p.sum()
                M_p = fast_autocorr(x_p).max()
                if M_p < best_m:
                    x = x_p
                    x_prev = x_p.copy()
                    best_m = M_p
                    best_a = x.copy()
                    stall = 0
                    eta = max(eta, 1.0)

        x_prev = x
        x = a_new
        iter_count += 1

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NestSmooth iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.2f}, stall={stall}")

def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None):
    """Improved LP-based Frank-Wolfe with dual-informed step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    prev_M = best_m

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            # Select top K by value, but also include some "nearly tight" ones
            sorted_idx = np.argsort(ac)[::-1]
            binding_idx = sorted_idx[:K_sel]

        # Build constraint matrix: T[i,j] = a[k_i - j] for valid j
        T_sel = np.zeros((min(K_sel, 2 * n - 1), n))
        for ii, k in enumerate(binding_idx[:min(K_sel, 2*n-1)]):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K * 2, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 50, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking with more candidates
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = fast_autocorr(a).max()
        alphas = [best_alpha, min(1.0, best_alpha * 0.5), min(1.0, best_alpha * 1.5),
                  0.5, 0.3, 0.2, 0.1, 0.05]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        # Adaptive K growth
        if iter_count % 8 == 0:
            K = min(K + 40, top_K_max)

        # Stall detection
        if m_new >= prev_M - 1e-12:
            K = min(K + 20, top_K_max)
        prev_M = m_new

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW2 iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

        if K >= top_K_max and iter_count > 200 and (m_new - best_m) < 1e-10:
            break

    return best_a, best_m


def peak_coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent focusing on peak-reducing moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Compute sensitivity: how much each coordinate contributes to peaks
        sens = np.zeros(n)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if len(contrib) == 0 or contrib.max() < 1e-20:
                continue

            # Find the pair contributing most to this peak
            peak_idx_in = np.argmax(contrib)
            src = idx[peak_idx_in]
            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Find low-sensitivity destination
            candidates = np.argsort(sens)[:20]
            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.02, 0.01, 0.05, 0.08, 0.15]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t < 1e-15:
                        a[src] += amt
                        a[dst] -= amt
                        break
                    a = a / s_t
                    M_trial = fast_autocorr(a).max()
                    if M_trial < best_m - 1e-16:
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        # Undo
                        a[dst] -= amt
                        a[src] += amt
                        a = a / a.sum()

            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0,
                 restart_every=5000, verbose=False, rng_seed=42):
    """Accelerated Mirror Descent for the smoothed subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.0)

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        best_seq = make_ramp(n_base, direction=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-start AMD with diverse inits ===
    print(f"\n=== Phase 1: Multi-start AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.18, 280.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_ramp(n0, direction=1))
        labels.append("ramp1")
        inits.append(make_ramp(n0, direction=-1))
        labels.append("ramp2")
        inits.append(make_off_center(n0, center=0.25, width=0.15))
        labels.append("offcenter")
        inits.append(make_u_shape(n0, sharpness=2.0))
        labels.append("u_shape")
        inits.append(make_asymmetric_double(n0))
        labels.append("asym_double")
        inits.append(make_w_shape(n0, sharpness=3.0))
        labels.append("w_shape")
        inits.append(make_power_ramp(n0, alpha=1.5))
        labels.append("power_ramp")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 90:
                break
            t_avail = min(t_per, time_budget - 90 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                     restart_every=4000, verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW v2 (n={n_best}) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, verbose=True,
                                       top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_main")

    # === Phase 3: Nesterov smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Nesterov Smoothed (n={n_best}) ===")
    t3 = min(remaining * 0.15, 220.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smoothed(a_in, t3, tau_init=15.0, tau_final=400.0,
                                      verbose=True, rng_seed=200)
        try_update(a_out, "nesterov1")

    # === Phase 4: Second LP-FW round ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW round 2 (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, verbose=True,
                                       top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_r2")

    # === Phase 5: Peak coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak Coord (n={n_best}) ===")
    t5 = min(remaining * 0.12, 160.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_coordinate_descent(a_in, t5, verbose=True, rng_seed=300)
        try_update(a_out, "peak_coord")

    # === Phase 6: Nesterov smoothed round 2 (different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Nesterov v2 (n={n_best}) ===")
    t6 = min(remaining * 0.10, 130.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smoothed(a_in, t6, tau_init=50.0, tau_final=1000.0,
                                      verbose=False, rng_seed=500)
        try_update(a_out, "nesterov2")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.4, 150.0)
        if t7a > 20:
            a_out, _ = nesterov_smoothed(a_long, t7a, tau_init=20.0, tau_final=300.0,
                                          verbose=False, rng_seed=new_n % 500)
            try_update(a_out, f"nest_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5000:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out, _ = lp_frank_wolfe_v2(a_in, t7b, solver, verbose=False,
                                               top_K_init=min(300, 2*new_n-1),
                                               top_K_max=min(2500, 2*new_n-1))
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.4, 120.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_smoothed(a_in, t8 * 0.5, tau_init=100.0, tau_final=2000.0,
                                          verbose=False, rng_seed=800)
            try_update(a_out, "nest_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 6000:
            t8b = min(remaining * 0.5, 100.0)
            if t8b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe_v2(a_in, t8b, solver, verbose=False,
                                               top_K_init=min(600, 2*n_best-1),
                                               top_K_max=min(5000, 2*n_best-1))
                try_update(a_out, "lp_final")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 9: Final coord polish ===")
        t9 = min(remaining * 0.6, 60.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_coordinate_descent(a_in, t9, verbose=False, rng_seed=900)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = make_ramp(n_fb, direction=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
