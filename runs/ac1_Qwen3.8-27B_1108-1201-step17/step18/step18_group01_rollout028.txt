"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative optimization.
The main algorithm solves a sequence of LPs to find directions that reduce
the peak autocorrelation, combined with multi-length exploration and polish.
"""

import numpy as np
from scipy import optimize, sparse
import time


def fast_autocorr(a):
    """Compute linear autocorrelation (convolution) via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (S * S)


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    num_lags = 2 * n - 1
    
    row_parts = []
    col_parts = []
    val_parts = []
    
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        row_parts.append(np.arange(k_start, k_end))
        col_parts.append(np.full(length, j, dtype=np.int32))
        val_parts.append(np.full(length, a[j]))
    
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    
    return sparse.csr_matrix((vals, (rows, cols)), shape=(num_lags, n))


def solve_lp(a, M, A=None):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    num_lags = 2 * n - 1
    
    if A is None:
        A = build_conv_matrix(a)
    
    c = -np.ones(n)
    b_ub = np.full(num_lags, M)
    bounds = [(0, None)] * n
    
    res = optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(a0, time_limit, verbose=True):
    """
    LP-based optimization: iteratively solve LP and mix in the solution.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    
    if verbose:
        print(f"    LP opt: n={n}, initial J={best_J:.6f}")
    
    iteration = 0
    A_matrix = build_conv_matrix(a)
    
    # Line search parameters
    t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        S = a.sum()
        J = 2.0 * n * M / (S * S)
        
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
        
        # Rebuild constraint matrix (since a changed)
        A_matrix = build_conv_matrix(a)
        
        # Solve LP
        t_lp_start = time.monotonic()

        t_lp_end = time.monotonic()

                print(f"    LP failed at iter {iteration}")
            break

                print(f"    LP degenerate at iter {iteration}")

        best_t = 0.0

        for t in t_candidates:

            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial

                best_M_new = M_trial
                best_t = t
        
        if best_t <= 0:

    elapsed = time.monotonic() - t0

def mgd_optimize(a0, time_limit, top_k=30, eta0=0.03, verbose=False):
    """Multiplicative gradient descent (kept from previous approach as polish)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    max_patience = 200
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 2.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * a[ki - idx]
        
        grad = grad - grad.mean()
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()
        
        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.5
            if eta < 0.001:
                eta = 0.02
                patience = max(0, patience - 50)
    
    if verbose:
        elapsed = time.monotonic() - t0

            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination
            peak_contrib = np.zeros(n)
            for k2 in top_k[:5]:

                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]
            
            candidates = np.where(peak_contrib < peak_contrib.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n):
    """Try various parametric families and return the best."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = None
    best_J = float('inf')
    
    families = []
    # Sine variants
    for freq in [1, 2, 3]:
        families.append(np.sin(np.pi * freq * x) ** 2)
    # Exponential
    for rate in [0.5, 1, 2, 3, 5]:
        families.append(np.exp(-rate * x))
        families.append(np.exp(rate * (x - 1.0)))
    # Polynomial
    for p in [0.5, 1, 2, 3]:
        families.append((i_arr + 1) ** p)
        families.append((n - i_arr) ** p)
    # Gaussian
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.1, 0.2, 0.3]:
            families.append(np.exp(-((x - c) ** 2) / (2 * sigma ** 2)))
    # Triangle
    families.append(1.0 - np.abs(2 * x - 1))
    # Asymmetric
    families.append(np.exp(-3*x) + 0.1)
    families.append(1.0 / (x + 0.5))
    
    for fam in families:
        a = norm_sum1(fam)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    return best_a, best_J
