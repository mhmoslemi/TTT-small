    # grad_j = 2 * sum_m a[m] * w[m+j]
    # Use FFT: this is cross-correlation of a and w
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_vec, fft_size)
    # Cross-correlation: result[j] = sum_m a[m] * w[m+j]
    # = ifft(conj(A) * W)[j] for real sequences (adjusted)
    cross = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * cross


def softmax_mirror_descent(a0, time_limit, beta_init=5.0, beta_max=50.0,
                           eta_init=2.0, use_gradient=False, verbose=True):
    """Softmax mirror descent with exact gradient and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    x = a.copy()
    x_prev = a.copy()
    beta = beta_init
    eta = eta_init
    stall = 0
    stall_limit = 5000

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov step
        if t > 1:
            moe = (t - 1.0) / (t + 1.5)
            y = x + moe * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Compute gradient
        top_k = min(300, len(ac))
        if n * top_k < 2000000:
            grad = compute_grad_direct(y, ac, beta, top_k)
        else:
            grad = compute_grad_fft(y, ac, beta, top_k)

        # Mirror descent step in log domain
        eta_t = eta / (1.0 + 0.5 * np.sqrt(t))
        log_y = np.log(y)
        log_new = log_y - eta_t * grad
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Adaptive beta: increase slowly
        if t % 50 == 0:
            beta = min(beta * 1.02, beta_max)

        # Restart if stalled
        if stall > stall_limit:
            stall = 0
            stall_limit = int(stall_limit * 1.2)
            pert_type = int(rng.integers(4))
            if pert_type == 0:
                noise = rng.normal(0, 0.02, n) * best_a
                x_r = norm_sum1(best_a + noise)
            elif pert_type == 1:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.3, 2.5))
                x_r = norm_sum1((1 - 0.3) * best_a + 0.3 * (i_arr + 1.0) ** p)
            elif pert_type == 2:
                x_arr = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.1, 0.4))
                gauss = norm_sum1(np.exp(-((x_arr - c) ** 2) / (2 * sig * sig)))
                x_r = norm_sum1(0.8 * best_a + 0.2 * gauss)
            else:
                # Block swap perturbation
                x_r = best_a.copy()
                block_size = max(10, n // 20)
                idx = int(rng.integers(0, n - 2 * block_size))
                tmp = x_r[idx:idx + block_size].copy()
                x_r[idx:idx + block_size] = x_r[idx + block_size:idx + 2 * block_size] * 0.5
                x_r[idx + block_size:idx + 2 * block_size] = tmp * 0.5 + x_r[idx + block_size:idx + 2 * block_size] * 0.5
                x_r = norm_sum1(x_r)
            x = x_r
            x_prev = x_r.copy()
            eta = max(eta * 0.8, 0.1)

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    SMD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=2500,
                   add_per_iter=60, verbose=True):
    """LP-based Frank-Wolfe with cutting plane constraint management."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_autocorr(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    # Initialize constraint set with top lags
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    x = y.copy()
    M_curr = best_m
    failed_solves = 0
    iter_count = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        cx = fast_autocorr(x)
        if cx.max() < best_m - 1e-15:
            best_m = cx.max()
            best_a = x.copy()

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))

            k_int = int(k)
            # (2*a*(g))[k] = 2*sum_i a[i]*g[k-i], linearized as 2*sum_i a[k-i]*g[i]
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
