def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_max(x, T=0.01):
    """Log-sum-exp smooth max with temperature T."""
    x_max = x.max()
    if x_max < 1e-30:
        return 0.0
    shifted = (x - x_max) / T
    shifted = np.clip(shifted, -50, 0)
    return x_max + T * np.log(np.sum(np.exp(shifted)))

def compute_smooth_grad(a, ac, T=0.01):
    """Compute gradient of smooth max of autocorrelation w.r.t. a."""
    n = len(a)
    ac_max = ac.max()
    if ac_max < 1e-30:
        return np.zeros(n)
    
    # Softmax weights
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -50, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    # Gradient: d(dk)/da[j] = 2*a[j] for the k where contribution is from pair (j, k-j)
    # Actually for conv[k] = sum_j a[j]*a[k-j], d(conv[k])/da[j] = a[k-j] + a[j-(k-n+1)] (if valid)
    # More carefully: conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
    # d(conv[k])/da[j] = a[k-j] (if 0<=k-j<n) + a[j-(k-n+1)] (if 0<=j-(k-n+1)<n and k-j >= n... no)
    
    # Let me be more careful. conv[k] = sum_j a[j]*a[k-j] where 0<=j<n and 0<=k-j<n
    # i.e., max(0,k-n+1) <= j <= min(n-1,k)
    # d(conv[k])/da[j] = a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    # But also, if we're computing the full convolution (not just autocorrelation), 
    # d(conv[k])/da[j] also gets a contribution from the "other side"
    
    # Actually for conv(a,a)(k) = sum_j a[j]a[k-j]:
    # d/d(a[j]) conv(k) = a[k-j] (when 0 <= k-j < n)
    # But conv is symmetric: conv(k) = conv(2n-2-k) for a=a
    # So also d/d(a[j]) conv(k) includes a[j-(k-n+1)] if that's valid... 
    # No, the derivative is just a[k-j] for valid k-j.
    
    # Wait, let me re-derive. conv(a,a)(k) = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * a[k-j]
    # d/d(a[j0]) conv(k) = a[k-j0] if j0 is in the range, else 0
    
    # So the gradient contribution from peak k to position j is: a[k-j] (if 0<=k-j<n)
    
    grad = np.zeros(n)
    # For efficiency, only process peaks with significant weight
    threshold = w.max() * 1e-3
    active = np.where(w > threshold)[0]
    
    for k in active:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-10:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    
    return grad

def adam_optimize(a0, time_limit, T=0.01, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  restart_every=5000, verbose=False, rng_seed=42):
    """Adam optimizer with smooth max objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        # Compute smooth gradient
        g = compute_smooth_grad(a, ac, T=T)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        t += 1
        lr_t = lr / math.sqrt(t)  # Adaptive LR
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        J_new = 2.0 * n * fast_autocorr(a_new).max()
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        if stall > restart_every:
            stall = 0
            # Perturbation
            noise = rng.normal(0, 0.01, n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(0.001, lr * 0.5)
            T = max(0.003, T * 0.7)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      Adam restart t={t} ({elapsed:.0f}s): J={best_J:.6f}")
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={t} ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}, T={T:.4f}")
    
    return best_a, best_J

def make_exponential_ramp(n, rate=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return norm_sum1(a)

def make_power_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)

def make_linear_ramp(n, alpha=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * x
    return norm_sum1(a)

def make_quadratic_ramp(n, alpha=0.2, beta=0.1):
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * x + beta * x * x
    return norm_sum1(a)
