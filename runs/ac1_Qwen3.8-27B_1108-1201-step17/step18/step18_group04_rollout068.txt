def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad(a, conv, top_k=40, temperature=0.002):
    n = len(a)
    conv_max = conv.max()
    spread = max(temperature * 10.0, conv_max * 0.001)
    threshold = conv_max - spread
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine_bump(n, width=0.6, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return norm_sum1(a)


def make_power_asym(n, power=0.3, flip=False):
    x = np.arange(n, dtype=np.float64)
    if not flip:
        a = (x + 1) ** power
    else:
        a = (n - x) ** power
    return norm_sum1(a)


def make_mixed_two(n, pos1=0.2, pos2=0.7, w1=0.1, w2=0.15, ratio=0.7):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - pos1) / w1) ** 2) + ratio * np.exp(-((x - pos2) / w2) ** 2)
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_flat_top(n, flat_frac=0.3, edge_width=0.2):
    x = np.linspace(0, 1, n)
    a = np.clip((x - edge_width) / (flat_frac / 2 + 1e-10), 0, 1)
    a = np.clip((1 - x - edge_width) / (flat_frac / 2 + 1e-10), 0, 1)
    a = np.minimum(a[0], a[1]) if n > 1 else np.array([1.0])
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.20, 80.0)
            if t6b > 8:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = coord_descent_polish(a_in, t6b, seed=777 + new_n % 100, verbose=False)
                try_update(a_out, f"cd_n{new_n}")

    # Phase 7: Final aggressive polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(_best_tracker['seq'])
    print(f"\n=== Phase 7: Final Polish (n={n_final}) ===")
    if remaining > 30:
        t7a = min(remaining * 0.45, 200.0)
        if t7a > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = amd_optimize(a_in, t7a, top_k_init=60,
                                    temp_init=0.0008, c0_init=2.0,
                                    restart_every=8000, seed=9999, verbose=False)
            try_update(a_out, "amd_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t7b = min(remaining * 0.40, 150.0)
        if t7b > 8:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = coord_descent_polish(a_in, t7b, seed=27182, verbose=False)
            try_update(a_out, "cd_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7c = min(remaining * 0.40, 100.0)
        if t7c > 5:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = flatten_peaks(a_in, t7c, seed=12345)
            try_update(a_out, "flatten_final")

    # Final evaluation
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        a_fb = make_cosine_bump(n_fb, 0.5, 0.5)
        _best_tracker['seq'] = a_fb
        _best_tracker['score'] = evaluate_sequence([float(v) for v in a_fb])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
