import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

BUDGET = 1740.0


def elapsed():
    return time.monotonic() - T_START


def time_left():
    return BUDGET - elapsed()


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_from_ac(a):
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max()


def softmax_from_b(b):
    bs = b - b.max()
    a = np.exp(bs)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(b) / len(b)
    return a / s


def smooth_obj_grad(b, beta):
    n = len(b)
    a = softmax_from_b(b)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]

    m = ac.max()
    shifted = beta * (ac - m)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    Z = e.sum()
    p = e / Z
    obj = 2.0 * n * (m + np.log(Z) / beta)

    S0 = np.dot(p, ac)

    fft_len = 1 << (3 * n - 1).bit_length()
    A_full = np.fft.rfft(a, fft_len)
    P_full = np.fft.rfft(p, fft_len)
    S1 = np.fft.irfft(np.conj(A_full) * P_full, fft_len)[:n]
    S2 = np.fft.irfft(P_full * A_full, fft_len)[:n]

    grad = 2.0 * n * (S1 + S2 - 2.0 * S0) * a
    return obj, grad


def smooth_obj_only(b, beta):
    obj, _ = smooth_obj_grad(b, beta)
    return obj


def lbfgs_smooth(a0, time_limit, betas=None, max_iters_each=300):
    if betas is None:
        betas = [15, 50, 150, 500, 2000, 8000, 30000]
    a = norm_sum1(a0)
    b = np.log(np.maximum(a, 1e-15))
    best_a = a.copy()
    best_true = obj_from_ac(a)
    t0 = time.monotonic()

    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        n = len(b)
        it_count = [0]

        def fun(bx):
            it_count[0] += 1
            o, _ = smooth_obj_grad(bx, beta)
            return o

        def fun_grad(bx):
            it_count[0] += 1
            return smooth_obj_grad(bx, beta)

        try:
            res = scipy_minimize(fun_grad, b, method='L-BFGS-B',
                                 options={'maxiter': max_iters_each, 'maxfun': max_iters_each * 5,
                                          'ftol': 1e-12, 'gtol': 1e-8, 'iprint': 0})
            if res.x is not None:
                a_new = softmax_from_b(res.x)
                tv = obj_from_ac(a_new)
                if tv < best_true - 1e-10:
                    best_true = tv
                    best_a = a_new.copy()
                b = res.x.copy()
        except Exception:
            pass
        print("    L-BFGS beta={}: iters={}, true_obj={:.8f}".format(beta, it_count[0], best_true))

    return best_a, best_true


def subgradient_momentum(a0, time_limit, top_k=80, beta_w=6.0, c0=4.0,
                         restart_every=3000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_k = top_k
    cur_beta = beta_w

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)
        k_max = min(cur_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / ws

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * y[ki - idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            noise = 0.04 / (1.0 + 0.08 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r > 1e-15:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            cur_k = min(cur_k + 15, 150)
            cur_beta = max(cur_beta - 0.4, 2.0)
            gamma = max(gamma * 0.9, 1.0)

        if verbose and t % 2000 == 0:
            J = 2.0 * n * best_m
            print("    SGD t={} ({:.0f}s): J={:.8f}".format(t, time.monotonic() - t0, J))

    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 800
    step_scales = [0.003, 0.007, 0.015, 0.03, 0.06, 0.12, 0.2]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(40, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])
            if dst == src:
                continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 150:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m - 1e-15:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 30 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.012)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m - 1e-15:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 200 == 0 and stall > 0:
            J = 2.0 * n * best_m
            print("    Coord ({:.0f}s): J={:.8f}, stall={}".format(time.monotonic() - t0, J, stall))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def parametric_init(n, parent=None):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)

    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))

    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for sigma in [0.05, 0.1, 0.15, 0.2]:
                candidates.append(np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) +
                                  np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma)))

    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(x ** p + (1.0 - x) ** p)

    for freq in [1, 2, 3, 5]:
        for ph in [0.0, 0.5, 1.0]:
            candidates.append(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.01))

    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50, 100]:
            candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))

    for rate in [0.5, 1, 2, 5, 10]:
        candidates.append(np.exp(-rate * x) + np.exp(rate * (x - 1.0)))

    for amp in [0.1, 0.3, 0.5]:
        for freq in [2, 5, 10, 20]:
            base = np.sin(np.pi * x) ** 2
            candidates.append(base * (1.0 + amp * np.cos(2.0 * np.pi * freq * x)))

    if parent is not None:
        candidates.append(parent)

    best_a = None
    best_val = float('inf')
    for cand in candidates:
        a = norm_sum1(cand)
        v = obj_from_ac(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    rng = np.random.default_rng(123)
    for _ in range(200):
        a = norm_sum1(rng.random(n) ** rng.uniform(0.3, 4.0))
        v = obj_from_ac(a)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label):

        return best_seq, best_score

        best_seq = a_norm.copy()
        print("*** NEW BEST [{}]: {:.8f} (n={}, t={:.0f}s)".format(label, sc, len(a), elapsed()))
    return best_seq, best_score

    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_out, _ = coordinate_polish(best_seq, t4, rng_seed=77, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n{}".format(n_best))

    # Phase 5: L-BFGS again (second pass with higher betas)
    remaining = time_left()
    n_best = len(best_seq)
    print("\n=== Phase 5: L-BFGS second pass (n={}) ===".format(n_best))
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 60:
        a_out, _ = lbfgs_smooth(best_seq, t5, betas=[100, 500, 2000, 10000, 50000],
                                max_iters_each=250)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs2_n{}".format(n_best))

    # Phase 6: Multi-length exploration
    remaining = time_left()
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length exploration ===")
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    shorter_lengths = [n_best - 500, n_best - 1000]
    all_lengths = [l for l in longer_lengths if l <= 10000] + [l for l in shorter_lengths if l >= 2000]

    for new_n in all_lengths[:4]:
        remaining = time_left()
        if remaining < 100:
            break
        print("  Trying n={}...".format(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.15, 100.0)
        if t6a > 30:
            a_out, _ = lbfgs_smooth(a_long, t6a, betas=[50, 200, 1000, 5000],
                                    max_iters_each=200)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_n{}".format(new_n))

        remaining = time_left()
        if remaining > 60:
            t6b = min(remaining * 0.10, 80.0)
            if t6b > 20:
                a_out, _ = subgradient_momentum(best_seq, t6b, top_k=60, beta_w=5.0,
                                                c0=3.0, restart_every=2000, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_n{}".format(new_n))

    # Phase 7: Final L-BFGS + subgradient + coordinate
    remaining = time_left()
    n_best = len(best_seq)
    print("\n=== Phase 7: Final refinement (n={}) ===".format(n_best))

    if remaining > 120:
        t7a = min(remaining * 0.30, 250.0)
        if t7a > 40:
            a_out, _ = lbfgs_smooth(best_seq, t7a, betas=[200, 1000, 5000, 20000, 100000],
                                    max_iters_each=200)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs3_n{}".format(n_best))

    remaining = time_left()
    if remaining > 80:
        t7b = min(remaining * 0.25, 200.0)
        if t7b > 30:
            a_out, _ = subgradient_momentum(best_seq, t7b, top_k=100, beta_w=5.0,
                                            c0=3.5, restart_every=2000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_final_n{}".format(n_best))

    remaining = time_left()
    if remaining > 40:
        t7c = min(remaining * 0.35, 200.0)
        if t7c > 20:
            a_out, _ = coordinate_polish(best_seq, t7c, rng_seed=314, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final_n{}".format(n_best))

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("\nFINAL: score={:.8f}, n={}, time={:.0f}s".format(best_score, len(best_seq), elapsed()))
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
