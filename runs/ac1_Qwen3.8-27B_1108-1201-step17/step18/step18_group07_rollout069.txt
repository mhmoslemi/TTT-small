"""
Search for a non-negative step sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))**2

The program combines:
- LP-based iterative optimization (max-sum and min-cross-peak variants),
- log-sum-exp smoothed mirror descent using FFT-accelerated gradients,
- subgradient mirror descent on the largest autocorrelation peaks,
- spectral-flattening moves,
- coordinate-descent local polish,
- multi-initialization and length resampling.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


_LP_CACHE = {}


def next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (size - 1).bit_length()


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return out


def norm_simplex(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(1, len(a))
    return a / s


def score_internal(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if len(a) == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    c = fast_conv(a)
    cmax = float(np.max(c)) if c.size else 0.0
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * len(a) * cmax / (s * s))


def consider_candidate(best_seq, best_score, a, label, t0):
    if a is None:
        return best_seq, best_score
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0 or not np.isfinite(arr).all():
        return best_seq, best_score
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return best_seq, best_score
    arr = arr / s
    sc = score_internal(arr)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t0
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(arr)}, t={elapsed:.0f}s)")
        return arr, sc
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    new_n = int(max(1, new_n))
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = len(f)
    if old_n == 0:
        return np.ones(new_n) / new_n
    if old_n == new_n:
        return norm_simplex(f)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return out / ss


def make_init_cosine(n, mode=1):
    x = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0.0, 1.0, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.08
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.28, pos2=0.72, width=0.12):
    x = np.linspace(0.0, 1.0, n)
    a = (
        np.exp(-((x - pos1) ** 2) / (2.0 * width ** 2)) +
        np.exp(-((x - pos2) ** 2) / (2.0 * width ** 2)) +
        0.04
    )
    return norm_simplex(a)


def make_init_sinc(n, bw=6.0):
    x = np.linspace(0.0, 1.0, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.02)
    return norm_simplex(a)


def make_init_tukey(n, r=0.35):
    a = np.ones(n, dtype=np.float64)
    if n > 2:
        m = int(max(1, r * n / 2))
        m = min(m, n // 2)
        t = np.arange(m, dtype=np.float64)
        taper = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / m - 1.0)))
        a[:m] = taper
        a[-m:] = taper[::-1]
    return norm_simplex(a)


def make_init_random(n, seed=1):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n, dtype=np.float64) * 0.45
    for f in range(1, 12):
        coeff = float(rng.normal(0.0, 0.14 / np.sqrt(float(f))))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        a += coeff * np.sin(2.0 * np.pi * f * x + phase)
    a = np.maximum(a, 0.02)
    return norm_simplex(a)


def get_lp_arrays(n):
    key = int(n)
    if key not in _LP_CACHE:
        total = key * key
        indices = np.empty(total, dtype=np.int32)
        for j in range(key):
            st = j * key
            en = st + key
            indices[st:en] = np.arange(j, j + key, dtype=np.int32)
        indptr = np.arange(key + 1, dtype=np.int64) * key
        _LP_CACHE[key] = (indices, indptr)
    return _LP_CACHE[key]


def build_T_csc(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n <= 0:
        return sparse.csc_matrix((0, 0), dtype=np.float64)
    indices, indptr = get_lp_arrays(n)
    m = 2 * n - 1
    data = np.tile(a, n)
    return sparse.csc_matrix((data, indices, indptr), shape=(m, n))


def lp_options(time_limit):
    return {
        "disp": False,
        "presolve": True,
        "time_limit": max(1.0, float(time_limit)),
    }


def lp_maxsum(a, M, time_limit):
    n = len(a)
    if n <= 0:
        return None
    if M <= 0.0 or not np.isfinite(M):
        M = 1e-12
    T = build_T_csc(a)
    m = 2 * n - 1
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, float(M), dtype=np.float64)
    try:

            c_obj,
            A_ub=T,
            b_ub=b_ub,

            method="highs",
            options=lp_options(time_limit),
        )
    except Exception:
        return None
    if res is None or getattr(res, "status", 1) != 0:
        return None
    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np maximum if False else np.maximum(g, 0.0)
    s = g.sum()
    if s <= 1e-12:
        return None
    return g / s


def lp_minmax(a, time_limit):
    n = len(a)
    if n <= 0:
        return None
    T = build_T_csc(a)
    m = 2 * n - 1
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    neg_col_data = -np.ones(m, dtype=np.float64)
    M_col = sparse.csc_matrix(
        (neg_col_data, np.arange(m, dtype=np.int32), np.arange(m + 1, dtype=np.int64)),
        shape=(m, 1),
    )
    A_ub = sparse.hstack((T, M_col), format="csc")
    b_ub = np.zeros(m, dtype=np.float64)

    row = np.zeros(n, dtype=np.int32)
    col = np.arange(n, dtype=np.int32)
    A_eq = sparse.csc_matrix(
        (np.ones(n, dtype=np.float64), (row, col)),
        shape=(1, n + 1),
    )
    b_eq = np.array([1.0], dtype=np.float64)

    try:
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=(0.0, None),
            method="highs",
            options=lp_options(time_limit),
        )
    except Exception:
        return None
    if res is None or getattr(res, "status", 1) != 0:
        return None
    x = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.maximum(x[:n], 0.0)
    s = g.sum()
    if s <= 1e-12:
        return None
    return g / s


def blend_search(a, g, grid=42):
    a = np.asarray(a, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    Caa = fast_conv(a)
    Cag = fast_conv(a, g)
    Cgg = fast_conv(g)

    best_m = float(np.max(Caa)) if Caa.size else 0.0
    if best_m < 0.0:
        best_m = 0.0
    best_t = 0.0

    ts = np.linspace(0.0, 1.0, grid)
    for t in ts:
        coeff00 = (1.0 - t) * (1.0 - t)
        coeff01 = 2.0 * t * (1.0 - t)
        coeff11 = t * t
        Ct = coeff00 * Caa + coeff01 * Cag + coeff11 * Cgg
        mval = float(np.max(Ct)) if Ct.size else 0.0
        if mval < 0.0:
            mval = 0.0
        if mval < best_m:
            best_m = mval
            best_t = float(t)

    if 0.0 < best_t < 1.0:
        w = 1.0 / max(1, grid - 1)
        lo = max(0.0, best_t - w)
        hi = min(1.0, best_t + w)
        ts2 = np.linspace(lo, hi, 32)
        for t in ts2:
            coeff00 = (1.0 - t) * (1.0 - t)
            coeff01 = 2.0 * t * (1.0 - t)
            coeff11 = t * t
            Ct = coeff00 * Caa + coeff01 * Cag + coeff11 * Cgg
            mval = float(np.max(Ct)) if Ct.size else 0.0
            if mval < 0.0:
                mval = 0.0
            if mval < best_m:
                best_m = mval
                best_t = float(t)

    out = (1.0 - best_t) * a + best_t * g
    return norm_simplex(out), best_m


def lp_iterate(a0, time_limit, mode="minmax", m_factor=1.0, max_stall=9, seed=0, verbose=False):
    if time_limit <= 0:
        a = norm_simplex(a0)
        return a, score_internal(a)

    t0 = time.monotonic()
    a = norm_simplex(a0)
    best = a.copy()
    c = fast_conv(best)
    best_m = float(np.max(c)) if c.size else 0.0
    if best_m < 0.0:
        best_m = 0.0
    work_m = best_m

    solve_time = 6.0
    stall = 0
    rng = np.random.default_rng(20240 + int(seed))
    n = len(a)

    while time.monotonic() - t0 < time_limit:
        st_solve = time.monotonic()

        c_work = fast_conv(a)
        Mcur = float(np.max(c_work)) if c_work.size else 0.0
        if Mcur < 0.0:
            Mcur = 0.0
        if Mcur < best_m - 1e-15:
            best = a.copy()
            best_m = Mcur

        if mode == "minmax":
            g = lp_minmax(a, solve_time)
        else:
            Muse = Mcur * float(m_factor)
            if Muse <= 0.0:
                Muse = 1e-12
            g = lp_maxsum(a, Muse, solve_time)

        el_solve = time.monotonic() - st_solve
        if g is not None and el_solve < solve_time * 0.85:
            solve_time = min(28.0, solve_time * 1.18)
        elif g is None:
            solve_time = max(2.0, solve_time * 0.75)
            stall += 1
        else:
            g = norm_simplex(g)
            cand, cand_m = blend_search(a, g)

            if cand_m < work_m - 1e-15:
                a = cand
                work_m = cand_m
                if cand_m < best_m - 1e-15:
                    best = a.copy()
                    best_m = cand_m
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

                if stall >= 3:
                    jitter = 0.004 + 0.002 * (stall % 5)
                    pert = a + rng.normal(0.0, jitter, n)
                    pert = norm_simplex(pert)
                    pm = score_internal(pert)
                    if pm < work_m * 1.03 and pm <= best_m + 1e-4:
                        a = pert
                        work_m = pm
                        if pm < best_m:
                            best = a.copy()
                            best_m = pm
                            stall = 0

        if verbose and stall % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"        LP mode={mode} stall={stall} best={best_m:.6f} t={elapsed:.0f}s")

        if stall >= max_stall:
            break

    return best, best_m


def smooth_grad_fft(a, c, beta_rel):
    a = np.asarray(a, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    n = len(a)
    if c.size == 0:
        return np.zeros(n, dtype=np.float64)
    max_c = float(np.max(c))
    if max_c <= 1e-15:
        return np.zeros(n, dtype=np.float64)

    beta = float(beta_rel) / max(max_c, 1e-12)
    exponents = beta * (c - max_c)
    exponents = np.maximum(exponents, -50.0)
    p = np.exp(exponents)
    ps = p.sum()
    if ps <= 1e-15:
        p = np.ones_like(c) / len(c)
    else:
        p = p / ps

    L = next_pow2(3 * n - 2)
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    R = np.fft.irfft(P * np.conj(A), L)[:n]
    return 4.0 * n * R


def smooth_ftrl(a0, time_limit, beta0=12.0, beta1=450.0, eta0=0.28):
    if time_limit <= 0:
        a = norm_simplex(a0)
        return a, score_internal(a)

    t0 = time.monotonic()
    a = norm_simplex(a0)
    best = a.copy()
    c = fast_conv(best)
    best_m = float(np.max(c)) if c.size else 0.0
    if best_m < 0.0:
        best_m = 0.0

    x = a.copy()
    t = 1

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, t / 50000.0)
        beta_rel = beta0 + (beta1 - beta0) * progress

        bt = (t - 1.0) / (t + 2.0)
        y = x + bt * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        c = fast_conv(y)
        grad = smooth_grad_fft(y, c, beta_rel)
        eta = eta0 / (np.sqrt(float(t)) + 1.0)

        loga = np.log(np.maximum(y, 1e-15)) - eta * grad
        loga = loga - loga.max()
        an = np.exp(loga)
        an = np.maximum(an, 1e-30)
        s = an.sum()
        if s < 1e-15:
            an = y.copy()
        else:
            an = an / s

        x = an
        a = an
        c = fast_conv(a)
        m = float(np.max(c)) if c.size else 0.0
        if m < 0.0:
            m = 0.0
        if m < best_m - 1e-15:
            best_m = m
            best = a.copy()
        t += 1

    return best, best_m


def ftrl_subgrad(a0, time_limit, eta0=0.32, beta0=18.0, beta1=90.0, top0=28, top1=3):
    if time_limit <= 0:
        a = norm_simplex(a0)
        return a, score_internal(a)

    t0 = time.monotonic()
    a = norm_simplex(a0)
    n = len(a)
    best = a.copy()
    c = fast_conv(best)
    best_m = float(np.max(c)) if c.size else 0.0
    if best_m < 0.0:
        best_m = 0.0

    x = a.copy()
    t = 1

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, t / 45000.0)
        beta = beta0 + (beta1 - beta0) * progress
        topk = int(max(top1, top0 * (1.0 - 0.85 * progress)))
        topk = min(topk, 2 * n - 1)

        bt = (t - 1.0) / (t + 2.0)
        y = x + bt * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        c = fast_conv(y)
        if c.size:
            if topk < len(c):
                idx = np.argpartition(c, len(c) - topk)[-topk:]
            else:
                idx = np.arange(len(c))
            vals = c[idx]
            maxv = float(np.max(vals))
            if maxv > 1e-15:
                ex = beta * (vals - maxv) / maxv
                ex = np.maximum(ex, -50.0)
                w = np.exp(ex)
                ws = w.sum()
                if ws > 1e-15:
                    w = w / ws
                else:
                    w = np.ones_like(w) / len(w)
            else:
                w = np.full(len(idx), 1.0 / max(1, len(idx)))
        else:
            idx = np.array([], dtype=int)
            w = np.array([], dtype=float)

        grad = np.zeros(n, dtype=np.float64)
        factor = 4.0 * n
        for wi, k in zip(w, idx):
            if wi < 1e-12:
                continue
            ki = int(k)
            st = max(0, ki - n + 1)
            en = min(n, ki + 1)
            if st >= en:
                continue
            ii = np.arange(st, en, dtype=np.int64)
            jj = ki - ii
            grad[ii] += factor * float(wi) * y[jj]

        eta = eta0 / (np.sqrt(float(t)) + 1.0)
        loga = np.log(np.maximum(y, 1e-15)) - eta * grad
        loga = loga - loga.max()
        an = np.exp(loga)
        an = np.maximum(an, 1e-30)
        s = an.sum()
        if s < 1e-15:
            an = y.copy()
        else:
            an = an / s

        x = an
        a = an
        c = fast_conv(a)
        m = float(np.max(c)) if c.size else 0.0
        if m < 0.0:
            m = 0.0
        if m < best_m - 1e-15:
            best_m = m
            best = a.copy()
        t += 1

    return best, best_m


def spectral_smooth_step(a, alpha=0.28):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    fft_size = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean <= 1e-15:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.power(ratio, 1.0 - alpha)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    a_new = norm_simplex(a_new)
    out = (1.0 - alpha) * a + alpha * a_new
    return norm_simplex(out)


def spectral_optimize(a0, time_limit):
    if time_limit <= 0:
        a = norm_simplex(a0)
        return a, score_internal(a)

    t0 = time.monotonic()
    a = norm_simplex(a0)
    best = a.copy()
    c = fast_conv(best)
    best_m = float(np.max(c)) if c.size else 0.0
    if best_m < 0.0:
        best_m = 0.0

    step = 0.25
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 140:
        cand = spectral_smooth_step(a, step)
        c = fast_conv(cand)
        m = float(np.max(c)) if c.size else 0.0
        if m < 0.0:
            m = 0.0
        if m < best_m - 1e-15:
            best_m = m
            best = cand.copy()
            a = cand.copy()
            step = min(0.58, step * 1.06)
            stall = 0
        else:
            a = norm_simplex(0.65 * a + 0.35 * best)
            step = max(0.04, step * 0.82)
            stall += 1

    return best, best_m


def coordinate_descent(a0, time_limit, seed=42):
    if time_limit <= 0:
        a = norm_simplex(a0)
        return a, score_internal(a)

    t0 = time.monotonic()
    a = norm_simplex(a0)
    n = len(a)
    best = a.copy()
    c = fast_conv(best)
    best_m = float(np.max(c)) if c.size else 0.0
    if best_m < 0.0:
        best_m = 0.0

    rng = np.random.default_rng(int(seed))
    stall = 0
    max_stall = 220

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        work_m = float(np.max(c)) if c.size else 0.0
        if work_m < 0.0:
            work_m = 0.0
        if work_m < best_m - 1e-15:
            best_m = work_m
            best = a.copy()

        pcount = min(5, len(c))
        if pcount <= 0:
            break
        if pcount >= len(c):
            peaks = np.arange(len(c))
        else:
            peaks = np.argpartition(c, len(c) - pcount)[-pcount:]

        low_idx = int(np.argmin(a)) if n else 0
        improved = False

        for k in peaks:
            ki = int(k)
            st = max(0, ki - n + 1)
            en = min(n, ki + 1)
            if st >= en:
                continue
            ii = np.arange(st, en, dtype=np.int64)
            contrib = a[ii] * a[ki - ii]
            src = int(ii[int(np.argmax(contrib))])
            if a[src] < 1e-10:
                continue

            dest_list = []
            candidates = [
                int(rng.integers(n)),
                low_idx,
                (src + int(rng.integers(1, n))) % n,
                (n - 1 - src) % n,
            ]
            for cand in candidates:
                if 0 <= cand < n and cand != src and cand not in dest_list:
                    dest_list.append(cand)

            for dst in dest_list:
                for frac in (0.002, 0.005, 0.01, 0.02, 0.05):
                    amt = a[src] * frac
                    if amt <= 0.0:
                        continue
                    trial = a.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    if trial.min() < -1e-15:
                        continue
                    cm = fast_conv(trial)
                    mval = float(np.max(cm)) if cm.size else 0.0
                    if mval < 0.0:
                        mval = 0.0
                    if mval < best_m - 1e-15:
                        a = trial
                        best_m = mval
                        best = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if n > 1:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = float(rng.uniform(0.004, 0.03))
                    trial = a.copy()
                    amt = a[i] * frac
                    trial[i] -= amt
                    trial[j] += amt
                    if trial.min() >= -1e-15:
                        cm = fast_conv(trial)
                        mval = float(np.max(cm)) if cm.size else 0.0
                        if mval < 0.0:
                            mval = 0.0
                        if mval < best_m - 1e-15:
                            a = trial
                            best_m = mval
                            best = a.copy()
                            stall = 0

    return best, best_m


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    total_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float("inf")
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            parent_n = len(parent)
            best_seq = norm_simplex(parent)
            best_score = score_internal(best_seq)
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        parent = None
        parent_n = 0

    if best_seq is None:
        n_default = 3600
        best_seq = make_init_cosine(n_default, mode=1)
        best_score = score_internal(best_seq)
        print(f"No parent. Default init: n={n_default}, score={best_score:.6f}")

    print(f"Starting: n={len(best_seq)}, score={best_score:.6f}, budget={total_budget:.0f}s")

    # Smaller length for LP-heavy optimization.
    if parent is not None:
        n_lp = int(min(2200, max(1200, len(parent))))
    else:
        n_lp = 2000

    a_lp = interpolate_sequence(best_seq, n_lp)
    best_seq, best_score = consider_candidate(best_seq, best_score, a_lp, "init_lp", t0)

    # LP phase on several initializations at moderate length.
    raw_lp = [
        (a_lp, "parent_lp", "minmax", 1.00),
        (a_lp, "parent_ms", "maxsum", 1.00),
        (make_init_cosine(n_lp, 1), "cos_lp", "minmax", 1.00),
        (make_init_tukey(n_lp, 0.35), "tukey_lp", "maxsum", 0.95),
        (make_init_bowl(n_lp, 2.2), "bowl_lp", "minmax", 1.00),
        (make_init_dual_peak(n_lp, 0.28, 0.72, 0.12), "dp_lp", "maxsum", 0.95),
    ]

    lp_inits = []
    lp_labels = []
    lp_modes = []
    lp_factors = []
    lp_seeds = []
    seen_sigs = []

    step_sig = max(1, n_lp // 24)
    for idx, (arr, lbl, mode, fac) in enumerate(raw_lp):
        arr = norm_simplex(arr)
        vals = [float(v) for v in arr[::step_sig]]
        sig = tuple(vals)
        if sig in seen_sigs:
            continue
        seen_sigs.append(sig)
        lp_inits.append(arr)
        lp_labels.append(lbl)
        lp_modes.append(mode)
        lp_factors.append(float(fac))
        lp_seeds.append(idx + 17)

    elapsed = time.monotonic() - t0
    remaining = total_budget - elapsed
    t_lp = min(remaining * 0.42, 560.0)
    if t_lp > 80 and len(lp_inits) > 0:
        per_lp = t_lp / float(len(lp_inits))
        print(f"\n=== Phase 1: LP at n={n_lp}, tasks={len(lp_inits)}, budget={t_lp:.0f}s ===")
        for idx, init in enumerate(lp_inits):
            elapsed = time.monotonic() - t0
            remaining = total_budget - elapsed
            t_avail = min(per_lp, remaining - 8.0)
            if t_avail < 18.0:
                break
            lbl = lp_labels[idx]
            mode = lp_modes[idx]
            fac = lp_factors[idx]
            seed = lp_seeds[idx]
            print(f"  LP {lbl} mode={mode} factor={fac:.2f} budget={t_avail:.0f}s")
            out, _ = lp_iterate(
                init,
                t_avail,
                mode=mode,
                m_factor=fac,
                max_stall=9,
                seed=seed,
                verbose=False,
            )
            best_seq, best_score = consider_candidate(best_seq, best_score, out, f"lp_{lbl}", t0)

            # Quick smooth polish after each useful LP segment.
            elapsed = time.monotonic() - t0
            remaining = total_budget - elapsed
            quick = min(25.0, remaining * 0.10)
            if quick > 8.0:
                q = smooth_ftrl(out, quick, beta0=12.0, beta1=300.0, eta0=0.22)
                best_seq, best_score = consider_candidate(best_seq, best_score, q, f"sm_{lbl}", t0)

    # Smooth/FTRL at the moderate LP length.
    elapsed = time.monotonic() - t0
    remaining = total_budget - elapsed
    t_quick = min(remaining * 0.07, 70.0)
    if t_quick > 10.0:
        base = interpolate_sequence(best_seq, n_lp)
        print(f"\n=== Phase 2: Smooth/FTRL at n={n_lp}, budget={t_quick:.0f}s ===")
        out = smooth_ftrl(base, t_quick, beta0=10.0, beta1=380.0, eta0=0.25)
        best_seq, best_score = consider_candidate(best_seq, best_score, out, "smooth_lp", t0)

    # Try useful resampled lengths.
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    target_lengths.extend([2200, 2800, 3400, 4200, 4800])

    seen_targets = []
    for tn in target_lengths:
        tn = int(max(1200, min(5000, tn)))
        if tn in seen_targets:
            continue
        seen_targets.append(tn)
        if gn := True:
            pass
        if abs(tn - len(best_seq)) > max(80, int(0.12 * len(best_seq))):
            cand = interpolate_sequence(best_seq, tn)
            best_seq, best_score = consider_candidate(best_seq, best_score, cand, f"resamp_{tn}", t0)

    # Local polishing at the current best length.
    n_work = len(best_seq)
    print(f"\nCurrent best after LP/resampling: n={n_work}, score={best_score:.6f}")

    elapsed = time.monotonic() - t0
    remaining = total_budget - elapsed
    t_spec = min(remaining * 0.10, 95.0)
    if t_spec > 8.0:
        print(f"\n=== Phase 3: Spectral polish (n={n_work}, budget={t_spec:.0f}s) ===")
        out = spectral_optimize(best_seq, t_spec)
        best_seq, best_score = consider_candidate(best_seq, best_score, out, "spectral", t0)

    elapsed = time.monotonic() - t0
    remaining = total_budget - elapsed
    t_ft = min(remaining * 0.25, 250.0)
    if t_ft > 12.0:
        print(f"\n=== Phase 4: FTRL subgradient (n={n_work}, budget={t_ft:.0f}s) ===")
        out = ftrl_subgrad(best_seq, t_ft, eta0=0.34, beta0=18.0, beta1=92.0, top0=28, top1=3)
        best_seq, best_score = consider_candidate(best_seq, best_score, out, "ftrl", t0)

    elapsed = time.monotonic() - t0
    remaining = total_budget - elapsed
    t_sm = min(remaining * 0.18, 185.0)
    if t_sm > 12.0:
        print(f"\n=== Phase 5: Smooth FTRL (n={len(best_seq)}, budget={t_sm:.0f}s) ===")
        out = smooth_ftrl(best_seq, t_sm, beta0=10.0, beta1=520.0, eta0=0.26)
        best_seq, best_score = consider_candidate(best_seq, best_score, out, "smooth_final", t0)

    elapsed = time.monotonic() - t0
    remaining = total_budget - elapsed
    t_cd = min(remaining * 0.16, 145.0)
    if t_cd > 8.0:
        print(f"\n=== Phase 6: Coordinate descent (n={len(best_seq)}, budget={t_cd:.0f}s) ===")
        out = coordinate_descent(best_seq, t_cd, seed=101)
        best_seq, best_score = consider_candidate(best_seq, best_score, out, "cd", t0)

    # If we currently do not share length with the parent, try a last local run at parent length.
    if parent is not None and abs(len(best_seq) - len(parent)) > max(150, int(0.15 * len(parent))):
        elapsed = time.monotonic() - t0
        remaining = total_budget - elapsed
        if remaining > 90:
            cand = interpolate_sequence(best_seq, len(parent))
            best_seq, best_score = consider_candidate(best_seq, best_score, cand, "parent_len", t0)
            t_p = min(remaining * 0.20, 120.0)
            if t_p > 10:
                print(f"\n=== Phase 7: Local polish at parent length (n={len(parent)}, budget={t_p:.0f}s) ===")
                out = ftrl_subgrad(cand, t_p * 0.6, eta0=0.32, beta0=20, beta1=85, top0=26, top1=3)
                best_seq, best_score = consider_candidate(best_seq, best_score, out, "ftrl_parent_len", t0)
                if remaining > t_p * 0.25 + 7:
                    out2 = smooth_ftrl(out, min(25.0, remaining - t_p * 0.6), beta0=10, beta1=350, eta0=0.24)
                    best_seq, best_score = consider_candidate(best_seq, best_score, out2, "smooth_parent_len", t0)

    # Final short coordinate search if time remains.
    elapsed = time.monotonic() - t0
    remaining = total_budget - elapsed
    t_final_cd = min(remaining * 0.45, 80.0)
    if t_final_cd > 5.0 and best_seq is not None:
        print(f"\n=== Phase 8: Final coordinate descent (budget={t_final_cd:.0f}s) ===")
        out = coordinate_descent(best_seq, t_final_cd, seed=555)
        best_seq, best_score = consider_candidate(best_seq, best_score, out, "cd_final", t0)

    if best_seq is None:
        best_seq = make_init_cosine(3600, mode=1)
        best_score = score_internal(best_seq)
        print(f"Fallback sequence: n={len(best_seq)}, score={best_score:.6f}")

    final_list = [float(x) for x in best_seq]
    try:
        exact = evaluate_sequence(final_list)
        if np.isfinite(exact):
            best_score = float(exact)
        print(f"FINAL internal={score_internal(best_seq):.6f}, exact={best_score:.6f}, n={len(final_list)}, time={time.monotonic()-t0:.0f}s")
    except Exception as exc:
        print(f"FINAL internal={best_score:.6f}, n={len(final_list)}, exact eval failed: {exc}")

    return final_list
