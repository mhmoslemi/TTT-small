def fast_autocorr(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def subgradient_polish(a0, time_limit, verbose=True):
    """
    Projected subgradient with heavy-ball momentum.
    Minimizes max_k conv[k] on the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    
    v = np.zeros(n)  # momentum
    step = 0.01
    beta = 0.9  # heavy ball
    
    # Adaptive step: start moderate, decay
    step_init = step
    step_decay = 0.999
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
        
        # Find active lags (within 1% of max)
        threshold = M_curr * 0.99
        active = np.where(ac >= threshold)[0]
        
        # Compute subgradient: average of active lag gradients
        g = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                g[idx] += 2.0 * a[k_int - idx]
        g /= max(len(active), 1)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-20:
            break
        
        # Adaptive step
        step = step_init * step_decay ** t
        
        # Heavy ball update
        v = beta * v - step * g / g_norm
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_a
        
        a = a_new
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad t={t} ({elapsed:.0f}s): J={J:.6f}, step={step:.2e}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    SubGrad done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f_norm)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return out / ss
