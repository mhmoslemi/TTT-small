import numpy as np

def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    # Duchi et al. algorithm
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


def log_sum_exp_smoothing(conv, temperature):
    """Compute smoothed max via log-sum-exp."""
    c_max = conv.max()
    shifted = conv - c_max
    return c_max + temperature * np.log(np.sum(np.exp(shifted / temperature)))


def compute_smooth_gradient(a, conv, temperature, top_k=None):
    """Compute gradient of log-sum-exp smoothed max convolution."""
    n = len(a)
    num_lags = len(conv)
    
    if top_k is not None and top_k < num_lags:
        top_idx = np.argsort(conv)[::-1][:top_k]
        c_top = conv[top_idx]
        w = np.exp(c_top / temperature - c_top.max() / temperature)
        w = w / w.sum()
        # Convert to full weights
        weights = np.zeros(num_lags)
        weights[top_idx] = w
    else:
        shifted = conv / temperature - (conv / temperature).max()
        weights = np.exp(shifted)
        weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for k in range(num_lags):
        if weights[k] < 1e-15:
            continue
        wi = weights[k]
        k_int = int(k)
        # grad[j] += 2 * wi * a[k-j] for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        j_idx = np.arange(lo, hi + 1)
        a_idx = k_int - j_idx
        grad[j_idx] += 2.0 * wi * a[a_idx]
    
    return grad

        # Compute marginal cost of adding mass to each position
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]
        
        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            
            # Find best destination
            candidates = np.argsort(costs + 1e-10 * a)[:max(15, n // 40)]
            candidates = [int(c) for c in candidates if int(c) != src_i]
            
            for dst_i in candidates[:12]:
                found = False
                for frac in [0.03, 0.08, 0.15, 0.25, 0.4, 0.6]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_simplex((n - x + 1) ** power)


def make_increasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_simplex((x + 1) ** power)


def make_mixed(n, pos1=0.2, pos2=0.65, width=0.1, ratio=1.0):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / (width * 1.5)) ** 2)
    return norm_simplex(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)

    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_centered_power(n, power=0.5):
    x_c = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm_simplex(x_c ** (-power))


def make_random_smooth(n, seed=0, n_freq=8):

    a = np.maximum(a, 0.001)
    return norm_simplex(a)
