"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initializations (power-law ramps, Gaussian bumps, two-phase)
2. Nesterov mirror descent with adaptive-temperature softmax multi-peak subgradient
3. LP-based direction finding (active-set SOTA approach with scipy HiGHS)
4. Spectral domain shaping (Gerchberg-Saxton)
5. Multi-length search with interpolation
6. Final coordinate polish

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_autocorr(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a unit-sum sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient(a, ac, n_peaks=80, temperature=1.0):
    """Compute softmax-weighted multi-peak subgradient of max(ac)."""
    n = len(a)
    k_max = min(n_peaks, len(ac))
    # Softmax weighting over all lags
    ac_safe = np.maximum(ac, 1e-30)
    log_ac = np.log(ac_safe)
    log_max = log_ac.max()
    exp_vals = np.exp((log_ac - log_max) / temperature)
    w = exp_vals / exp_vals.sum()
    
    # Focus on top peaks
    top_idx = np.argsort(ac)[::-1][:k_max]
    grad = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    return grad
