def lp_best_response(f, M):

    c = -np.ones(n)

Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response optimization (Gerchberg-Saxton variant)
2. Power-law ramp initializations with various exponents
3. Subgradient refinement for final polish
4. Multi-length and multi-shape exploration

Key innovation: The LP finds the optimal "best response" direction that
maximizes total mass while keeping the cross-convolution bounded,
providing a principled descent direction.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_ramp(n, power):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def lp_best_response(f, M, verbose=False):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    
    # Build sparse constraint matrix
    # Row k: nonzeros at j = max(0, k-n+1) .. min(n-1, k), value = f[k-j]
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None


def lp_iterate(a0, time_limit, verbose=True, rng_seed=42):
    """Iterative LP-based optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    iter_count = 0
    
    t_blend = 0.5
    
    while time.monotonic() - t0 < time_limit:
        iter_start = time.monotonic()
        M = fast_autocorr(a).max()
        
        g = lp_best_response(a, M)
        if g is None:
            verbose and print(f"    LP failed at iter {iter_count}")
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Blend
        a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller blend
            t_blend = t_blend * 0.7
            a_new = (1 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
                a = a_new
            else:
                # No improvement, restart with perturbation
                if t_blend < 0.05:
                    break
        
        if M_new < best_m * 0.999:
            t_blend = min(0.8, t_blend * 1.1)
        
        iter_count += 1
        elapsed = time.monotonic() - t0
        if verbose:
            J = 2.0 * n * best_m
            print(f"    LP iter {iter_count} ({elapsed:.0f}s): best_m={best_m:.8f}, J={J:.6f}, blend={t_blend:.3f}")
        
        if time.monotonic() - t0 > time_limit - 30:
            break
    
    return best_a, best_m

def interp_to(a, new_n):

        return np.ones(new_n) / new_n
    f_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)

    f_interp = np.interp(new_x, old_x, f_norm)

    return norm_sum1(a)


def make_mild_ramp(n, center_frac=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 + center_frac * x
    return norm_sum1(a)


def make_exp_ramp(n, rate=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return norm_sum1(a)


def interpolate_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def compute_gradient(a, ac, n_peaks=50, temperature=1.0):
    n = len(a)
    k_max = min(n_peaks, len(ac))
    ac_clipped = np.minimum(ac, ac.max() + 1e-30)
    log_ac = np.log(np.maximum(ac_clipped, 1e-30))
    exp_vals = np.exp((log_ac - log_ac.max()) / temperature)
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    top_idx = np.argsort(ac)[::-1][:k_max]
    for k in top_idx:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-10:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    return grad


def grad_optimize(a0, time_limit, n_peaks=80, temperature0=3.0, c0=5.0,
                  restart_every=5000, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_temp = temperature0
    restart_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        ac = fast_autocorr(y)
        g = compute_gradient(y, ac, n_peaks=n_peaks, temperature=cur_temp)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        x_prev = x
        x = a_new
        M_x = float(fast_autocorr(x).max())
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        cur_temp = max(0.8, temperature0 * (1.0 - 0.1 * min(stall_count / 2000.0, 1.0)))
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.01 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_temp = max(0.8, temperature0 - 0.2 * restart_count)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Grad restart ({elapsed:.0f}s): J={2.0*n*best_m:.6f}")
    return best_a, best_m


def build_lp_matrix(f, k_indices=None):
    """Build sparse constraint matrix for the LP. Returns (A_ub_data, rows, cols, shape)."""
    n = len(f)
    if k_indices is None:
        k_indices = list(range(2 * n - 1))
    else:
        k_indices = [int(k) for k in k_indices]
    n_rows = len(k_indices)
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(k_indices):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        rows.extend([row_idx] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_rows, n))
    return A_ub, k_indices


def solve_lp(f, M, k_indices=None, max_cutting_planes=5):
    """Solve LP with cutting plane method for constraint handling."""
    n = len(f)
    c = -np.ones(n)

    if k_indices is None:
        # Use top constraints based on f*f
        ac_f = fast_autocorr(f)
        n_top = min(3000, 2 * n - 1)
        k_indices = np.argsort(ac_f)[::-1][:n_top]
        k_indices = sorted([int(k) for k in k_indices])

    current_k = list(k_indices)
    g = None

    for cp_iter in range(max_cutting_planes):
        A_ub, _ = build_lp_matrix(f, current_k)
        b_ub = np.full(len(current_k), M)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': 120})
        if not res.success:
            if g is not None:
                return g
            return None
        g = res.x
        # Check if all constraints are satisfied
        if len(current_k) >= 2 * n - 1:
            return g
        # Check remaining constraints
        g_conv = fast_autocorr(f)  # This is f*f, not f*g
        # We need f*g, compute it
        fg = np.convolve(f, g, mode='full')
        # Check which constraints are violated
        all_ks = set(range(2 * n - 1))
        current_set = set(current_k)
        remaining = all_ks - current_set
        if remaining:
            remaining_list = sorted(remaining)
            violated = [k for k in remaining_list if fg[k] > M + 1e-10]
            if violated:
                # Add the most violated constraints
                violated.sort(key=lambda k: fg[k], reverse=True)
                current_k.extend(violated[:100])
                continue
        return g

    return g


def lp_iterate(a0, time_limit, verbose=True, rng_seed=42, max_iters=200,
               n_constraints=None, time_per_lp=60.0):
    """Iterative LP-based optimization with adaptive blending."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    iter_count = 0
    t_blend = 0.4
    no_improve = 0

    # Pre-select k_indices based on initial autocorrelation
    ac_init = fast_autocorr(a)
    if n_constraints is not None and n_constraints < 2 * n - 1:
        k_indices = np.argsort(ac_init)[::-1][:n_constraints]
        k_indices = sorted([int(k) for k in k_indices])
    else:
        k_indices = None

    while time.monotonic() - t0 < time_limit and iter_count < max_iters:
        if no_improve > 40:
            break

        M = float(fast_autocorr(a).max())

        t_lp_start = time.monotonic()
        g = solve_lp(a, M, k_indices=k_indices, max_cutting_planes=3)
        lp_time = time.monotonic() - t_lp_start

        if g is None:
            if verbose:
                print(f"    LP solve failed at iter {iter_count}")
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Try blending
        found = False
        blend_candidates = [t_blend, t_blend * 0.6, t_blend * 0.3, 0.1, 0.05, 0.02]
        for tb in blend_candidates:
            a_new = (1.0 - tb) * a + tb * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            M_new = float(fast_autocorr(a_new).max())
            if M_new < best_m - 1e-12:
                best_m = M_new
                best_a = a_new.copy()
                a = a_new
                t_blend = min(0.7, tb * 1.3)
                found = True
                no_improve = 0
                break

        if not found:
            no_improve += 1
            t_blend = max(0.02, t_blend * 0.6)
            if no_improve > 15 and no_improve % 5 == 0:
                # Perturb to escape
                pert = rng.normal(0, 0.03 + 0.01 * (no_improve // 5), n)
                a = a + pert * a
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                # Update k_indices
                ac_now = fast_autocorr(a)
                if n_constraints is not None and n_constraints < 2 * n - 1:
                    k_indices = np.argsort(ac_now)[::-1][:n_constraints]
                    k_indices = sorted([int(k) for k in k_indices])

        iter_count += 1
        elapsed = time.monotonic() - t0
        if verbose and (iter_count <= 3 or iter_count % 5 == 0 or elapsed > time_limit * 0.85):
            J = 2.0 * n * best_m
            print(f"    LP[{n}] iter {iter_count} ({elapsed:.0f}s, solve={lp_time:.1f}s): "
                  f"J={J:.6f}, blend={t_blend:.3f}, no_imp={no_improve}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except (NameError, Exception):
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_ramp(n_base, 0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Quick gradient with diverse inits (20% time) ===
    print(f"\n=== Phase 1: Gradient optimization (diverse inits) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_p1 = remaining * 0.20

    if t_p1 > 30:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        for pw in [0.3, 0.4, 0.5, 0.7]:
            inits.append(make_ramp(n0, pw))
            labels.append(f"ramp_{pw}")
        inits.append(make_mild_ramp(n0, 0.3))
        labels.append("mild")
        inits.append(make_exp_ramp(n0, 2.0))
        labels.append("exp")

        t_per = max(8.0, t_p1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_avail = min(t_per, remaining - 60)
            if t_avail < 5:
                break
            a_out, _ = grad_optimize(init, t_avail, n_peaks=80, temperature0=3.0,
                                      c0=5.0, rng_seed=42 + idx)
            try_update(a_out, f"grad_{lbl}")

    # === Phase 2: LP at coarser scale (n_lp=1500) (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP optimization at coarse scale ===")
    t_p2 = remaining * 0.35

    if t_p2 > 30:
        n_lp = 1500
        a_coarse = interpolate_to(best_seq, n_lp)
        print(f"  LP at n={n_lp}, budget={t_p2:.0f}s")
        a_out, _ = lp_iterate(a_coarse, t_p2, verbose=True, rng_seed=42,
                              n_constraints=2500, max_iters=300)
        # Interpolate back
        a_back = interpolate_to(a_out, n0)
        try_update(a_back, f"lp_coarse_{n_lp}_to_{n0}")
        # Also try evaluating at coarse scale directly
        try_update(a_out, f"lp_coarse_{n_lp}")

    # === Phase 3: LP at medium scale (n_lp=2500) (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP at medium scale ===")
    t_p3 = remaining * 0.25

    if t_p3 > 30:
        n_lp2 = 2500
        a_med = interpolate_to(best_seq, n_lp2)
        print(f"  LP at n={n_lp2}, budget={t_p3:.0f}s")
        a_out, _ = lp_iterate(a_med, t_p3, verbose=True, rng_seed=77,
                              n_constraints=3500, max_iters=200)
        a_back = interpolate_to(a_out, n0)
        try_update(a_back, f"lp_med_{n_lp2}_to_{n0}")
        try_update(a_out, f"lp_med_{n_lp2}")

    # === Phase 4: Gradient refinement at full scale (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Gradient refinement ===")
    t_p4 = remaining * 0.15

    if t_p4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = grad_optimize(a_in, t_p4, n_peaks=120, temperature0=1.5,
                                  c0=4.0, restart_every=3000, rng_seed=99)
        try_update(a_out, f"grad_refine_{len(a_in)}")

    # === Phase 5: LP at full scale with limited constraints (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP at full scale ===")
    t_p5 = remaining * 0.10

    if t_p5 > 20:
        a_in = norm_sum1(best_seq)
        n_full = len(a_in)
        nc = min(3000, 2 * n_full - 1)
        a_out, _ = lp_iterate(a_in, t_p5, verbose=True, rng_seed=200,
                              n_constraints=nc, max_iters=50)
        try_update(a_out, f"lp_full_{n_full}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length ===")

    if remaining > 50:
        n_best = len(best_seq)
        for delta in [-1000, -500, 500, 1000, 2000, 3000]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            new_n = n_best + delta
            if new_n < 800 or new_n > 12000:
                continue
            a_try = interpolate_to(best_seq, new_n)
            try_update(a_try, f"interp_{new_n}")
            t5 = min(remaining * 0.25, 60.0)
            if t5 > 8:
                a_out, _ = grad_optimize(a_try, t5, n_peaks=80, temperature0=2.0,
                                          c0=5.0, rng_seed=42 + new_n % 100)
                try_update(a_out, f"grad_{new_n}")

    # === Phase 7: Final combined LP + gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final optimization ===")

    if remaining > 40:
        # LP at n=2000 then interpolate
        n_final_lp = 2000
        a_cf = interpolate_to(best_seq, n_final_lp)
        t_lp = min(remaining * 0.5, 120.0)
        if t_lp > 15:
            a_out, _ = lp_iterate(a_cf, t_lp, verbose=True, rng_seed=555,
                                  n_constraints=3000, max_iters=100)
            a_back = interpolate_to(a_out, len(best_seq))
            try_update(a_back, f"final_lp_{n_final_lp}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = grad_optimize(a_in, min(remaining - 5, 60.0),
                                      n_peaks=150, temperature0=1.0,
                                      c0=3.0, rng_seed=2024)
            try_update(a_out, "final_grad")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_ramp(n_fb, 0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
