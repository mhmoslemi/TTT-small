def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def subgradient_multi_peak(a, c, top_k=10, beta=30.0):
    """Compute weighted subgradient using top peaks with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()
    
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent_backtrack(a0, time_limit, verbose=False):
    """Mirror descent with backtracking line search on the simplex.
    
    Uses the structure: subgradient at peak k is a[k-i] (reversed/shifted).
    Step: a_new[i] ∝ a[i] * exp(-eta * g[i])

    a = norm_simplex(a0)
    best_a = a.copy()
    c = fast_conv(a)
    best_m = c.max()
    t0 = time.monotonic()
    
    eta = 1.0  # base step size
    top_k = 20
    beta = 30.0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:

        # Annealing
        progress = min(1.0, t / 30000.0)
        cur_beta = 15.0 + 60.0 * progress
        cur_top_k = max(5, int(30 * (1 - 0.7 * progress)))
        cur_eta = eta / (1.0 + 0.05 * np.sqrt(t))
        
        # Compute current convolution
        c = fast_conv(a)
        M_curr = c.max()
        
        # Compute subgradient
        g = subgradient_multi_peak(a, c, top_k=cur_top_k, beta=cur_beta)
        
        # Mirror descent step: a_new[i] ∝ a[i] * exp(-eta * g[i])
        log_factor = -cur_eta * g
        # Clip for numerical stability
        log_factor = np.clip(log_factor, -10.0, 10.0)
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        
        # Evaluate and backtrack if needed
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < M_curr:
            a = a_new
            # Try to increase eta slightly
            eta = min(eta * 1.02, 5.0)
        else:
            # Backtrack: take a smaller step
            a_new_bt = (1.0 - 0.5 * cur_eta) * a + 0.5 * cur_eta * a_new
            a_new_bt = norm_simplex(a_new_bt)
            c_bt = fast_conv(a_new_bt)
            if c_bt.max() < M_curr:
                a = a_new_bt
            # Decrease eta
            eta = max(eta * 0.85, 0.01)
        
        # Track best
        c_check = fast_conv(a)
        M_check = c_check.max()
        if M_check < best_m - 1e-15:

            print(f"      MD t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, eta={eta:.4f}")
    
    return best_a, best_m


def spectral_conv_smooth(a, strength=0.3):
    """Spectral smoothing specifically for convolution flattening.
    
    Key: we want IDFT[A(f)^2] to be flat, so we compress |A(f)|.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    # Compress dynamic range of amplitude
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    # Blend
    return norm_simplex((1.0 - strength * 0.5) * a + strength * 0.5 * a_new)

def parametric_shape(n, shape_type, params):
    """Generate a parametric asymmetric shape."""
    x = np.linspace(0.0, 1.0, n)
    
    if shape_type == 'power':
        alpha = params[0]
        a = x ** alpha + 1e-6
    elif shape_type == 'exp':
        lam = params[0]
        a = np.exp(lam * x)
    elif shape_type == 'power_exp':
        alpha, lam = params[0], params[1]
        a = (x ** alpha + 1e-6) * np.exp(lam * x)
    elif shape_type == 'poly2':
        c1, c2 = params[0], params[1]
        a = 1.0 + c1 * x + c2 * x ** 2
        a = np.maximum(a, 1e-6)
    elif shape_type == 'gauss_tail':
        mu, sigma = params[0], params[1]
        a = np.exp(-((x - mu) ** 2) / (2 * sigma ** 2)) + 1e-6
    elif shape_type == 'sigmoid':
        lam, center = params[0], params[1]
        a = 1.0 / (1.0 + np.exp(-lam * (x - center)))
    elif shape_type == 'combined':
        alpha, lam, c1 = params[0], params[1], params[2]
        a = (x ** alpha + 1e-6) * np.exp(lam * x) * (1.0 + c1 * x)
    else:
        a = np.ones(n)
    
    return norm_simplex(a)

def fiddle_optimize(a0, time_limit, verbose=False):
    """FIDDLE-style optimization: alternating between time and frequency domain.
    
    In time domain: multiplicative weight update (mirror descent step)
    In frequency domain: amplitude compression to flatten A(f)^2
    """
    n = len(a0)
    a = norm_simplex(a0)

    for t in range(1, 200000):

            break
        
        if (t % 10) < int(10 * time_in_time):

            M_curr = c.max()

            c_new = fast_conv(a_new)
            if c_new.max() < M_curr:

            else:

        else:

def coordinate_descent_v2(a0, time_limit, verbose=False):
    """Improved coordinate descent: move mass from peak-contributing positions."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    c = fast_conv(a)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        
        # Top peaks
        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            # Move mass from highest contributor to a "safe" position
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Find destination: position with minimum contribution to all top peaks
            dst_scores = np.zeros(n)
            for kp in top_peaks[:5]:
                kp_int = int(kp)
                js = max(0, kp_int - n + 1)
                je = min(n, kp_int + 1)
                for j in np.arange(js, je):
                    dst_scores[j] += a[j] * a[kp_int - j]
            
            # Exclude source
            dst_scores[src] = np.inf
            # Prefer positions far from peak-contributing indices
            dst = int(np.argmin(dst_scores))
            
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                amt = src_val * frac
                if amt > a[src]:
                    amt = a[src] * 0.5
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def diversified_search(a0, n, time_limit, pop_size=8, verbose=False):
    """Diversified local search: run multiple short optimizations from different perturbations."""
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    best_a = norm_simplex(a0)
    best_m = fast_conv(best_a).max()
    
    t_per = time_limit / pop_size
    
    for i in range(pop_size):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Create perturbed start
        pert_scale = 0.02 + 0.03 * i / pop_size
        pert = rng.normal(0, pert_scale, n)
        start = norm_simplex(a0 + pert)
        
        # Short mirror descent
        a_out, m_out = mirror_descent_backtrack(start, t_per, verbose=False)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4000
        # Use an asymmetric initialization
        x = np.linspace(0, 1, n_base)
        best_seq = norm_simplex(x ** 1.5 + 0.01)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Parametric shape search at various lengths (15%) ===
    print(f"\n=== Phase 1: Parametric search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.15
    
    lengths_to_try = sorted(set([max(2000, n0 - 500), n0, n0 + 500, n0 + 1000, 6000, 8000]))
    lengths_to_try = [ln for ln in lengths_to_try if 2000 <= ln <= 10000]
    
    t_per_len = t1 / len(lengths_to_try)
    for ln in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        print(f"  Parametric search n={ln} (budget={min(t_per_len, remaining*0.9):.0f}s)...")
        a_p, sc_p = parametric_search(ln, min(t_per_len, remaining * 0.9), verbose=False)
        check = evaluate_safe(a_p)
        if np.isfinite(check):
            try_update(a_p, f"param_n{ln}")
    
    # === Phase 2: Mirror descent from best (30%) ===
    print(f"\n=== Phase 2: Mirror descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.30
    
    if t2 > 30:
        print(f"  MD from best (n={n_best}, budget={t2:.0f}s)...")
        a_md, _ = mirror_descent_backtrack(norm_simplex(best_seq), t2 * 0.6, verbose=True)
        try_update(a_md, f"md_n{n_best}")
        
        # FIDDLE from best
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t2b = remaining * 0.4
            print(f"  FIDDLE from best (n={n_best}, budget={t2b:.0f}s)...")
            a_fid, _ = fiddle_optimize(norm_simplex(best_seq), t2b, verbose=True)
            try_update(a_fid, f"fiddle_n{n_best}")
    
    # === Phase 3: Diversified search (20%) ===
    print(f"\n=== Phase 3: Diversified search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.20
    
    if t3 > 30:
        a_div, _ = diversified_search(norm_simplex(best_seq), n_best, t3, pop_size=8, verbose=False)
        try_update(a_div, f"div_n{n_best}")
    
    # === Phase 4: Spectral + CD polish (15%) ===
    print(f"\n=== Phase 4: Spectral + CD polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    if remaining > 30:
        t4a = remaining * 0.5
        print(f"  Spectral polish (n={n_best}, budget={t4a:.0f}s)...")
        a_sp, _ = spectral_targeted(norm_simplex(best_seq), t4a, verbose=False)
        try_update(a_sp, f"spectral_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t4b = remaining * 0.6
            print(f"  CD polish (n={n_best}, budget={t4b:.0f}s)...")
            a_cd, _ = coordinate_descent_v2(norm_simplex(best_seq), t4b, verbose=False)
            try_update(a_cd, f"cd_n{n_best}")
    
    # === Phase 5: Try longer sequences (15%) ===
    print(f"\n=== Phase 5: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    for delta in [1000, 2000, 3000]:
        new_n = n_best + delta
        if new_n > 12000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")
        
        if remaining > 50:
            t5 = remaining * 0.35
            a_out, _ = mirror_descent_backtrack(a_long, t5, verbose=False)
            try_update(a_out, f"md_n{new_n}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_sp, _ = spectral_targeted(a_out, min(remaining * 0.4, 60.0), verbose=False)
                try_update(a_sp, f"spectral_n{new_n}")
    
    # === Phase 6: Final intensive optimization (5% remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}) ===")
    
    if remaining > 20:
        t6 = remaining * 0.6
        a_out, _ = mirror_descent_backtrack(norm_simplex(best_seq), t6, verbose=False)
        try_update(a_out, f"final_md_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out2, _ = fiddle_optimize(norm_simplex(best_seq), remaining * 0.7, verbose=False)
            try_update(a_out2, f"final_fiddle_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_simplex(x ** 1.5 + 0.01)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
