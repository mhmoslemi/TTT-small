"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Refined LP-Frank-Wolfe with pure FW steps and aggressive constraint management
2. Surgical local search targeting top convolution peaks
3. Multi-length exploration with interpolated starting points
4. Subgradient refinement with simplex projection
"""

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix

_SOLVER = None


def get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(200, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _SOLVER = cp.SCS
    print(f"  Fallback solver: SCS")
    return _SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_conv_matrix(y, lag_list, n):
    """Build sparse matrix A where A[k, i] = y[k-i] if valid, for each lag k in lag_list."""
    rows = []
    cols = []
    vals = []
    for jj, k in enumerate(lag_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeff = y[k_int - idx]
            mask = coeff > 1e-15
            if mask.any():
                rows.extend([jj] * mask.sum())
                cols.extend(list(idx[mask]))
                vals.extend(list(coeff[mask]))
    if len(rows) == 0:
        return csr_matrix((len(lag_list), n))
    return csr_matrix((np.array(vals), (np.array(rows), np.array(cols))),
                      shape=(len(lag_list), n))


def lp_frank_wolfe_refined(a0, time_limit, solver, init_K=400, max_K=4000,
                           add_max=50, use_nesterov_after=0, verbose=True):
    """
    Refined LP-Frank-Wolfe:
    - Pure FW steps (no Nesterov) for stability
    - Adaptive constraint set management
    - Tracks both x and y iterates
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    # Initialize with top lags
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    init_idx = np.argsort(ac_y)[::-1][:min(init_K, 2 * n - 1)]
    constraint_set = sorted(int(k) for k in init_idx)

    iter_count = 0
    stalled = 0
    fail_count = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            stalled = 0
        else:
            stalled += 1

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-16:
            best_m = ac_x.max()
            best_a = x.copy()
            stalled = 0

        if stalled > 200:
            # Reset with perturbation
            rng = np.random.default_rng(iter_count)
            a_pert = norm_sum1(best_a + rng.normal(0, 0.005, n) * best_a)
            y = a_pert
            x = a_pert.copy()
            ac_y = fast_autocorr(y)
            M_curr = ac_y.max()
            init_idx = np.argsort(ac_y)[::-1][:min(init_K, 2 * n - 1)]
            constraint_set = sorted(int(k) for k in init_idx)
            stalled = 0
            iter_count = 0

        # Manage constraint set
        if len(constraint_set) > max_K:
            ac_y_curr = fast_autocorr(y)
            margins = np.array([M_curr - ac_y_curr[k] for k in constraint_set])
            keep = np.argsort(margins)[:max_K]
            constraint_set = sorted(constraint_set[i] for i in keep)

        K = len(constraint_set)
        if K == 0:
            break

        # Build constraint matrix
        A_mat = build_conv_matrix(y, constraint_set, n)

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            fail_count += 1
            if fail_count > 10:
                constraint_set = constraint_set[:init_K]
                fail_count = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            fail_count += 1
            if fail_count > 5:
                constraint_set = constraint_set[:max(100, init_K // 2)]
                fail_count = 0
            continue

        fail_count = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints and add
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_yg > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_max]]
            for k in top_v:
                constraint_set.append(int(k))
            constraint_set = sorted(set(constraint_set))

        # FW step
        t = iter_count + 1
        alpha = 2.0 / (t + 2.0)

        if t > use_nesterov_after:
            # Nesterov-accelerated step
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            y_new = y_new / sy if sy > 1e-15 else y.copy()

            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            x_new = x_new / sx if sx > 1e-15 else y_new.copy()
        else:
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            y_new = y_new / sy if sy > 1e-15 else y.copy()
            x_new = y_new.copy()

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, stall={stalled}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def surgical_local_search(a0, time_limit, top_lags=15, verbose=True):
    """
    Targeted local search: for each top convolution lag, find the best
    mass transfer that reduces the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            # Random perturbation
            n_pert = int(rng.integers(2, 8))
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.01))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
                if fast_autocorr(a).max() < best_m:
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
                    stall = 0
            if stall > 1000:
                break

        # Find top lags
        top_k = np.argsort(ac)[::-1][:top_lags]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            pair_vals = a[idx] * a[k_int - idx]

            # Find top contributing pairs
            n_pairs = min(5, len(idx))
            pair_order = np.argsort(pair_vals)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                other = k_int - src
                other_val = a[other]
                if src_val < 1e-12 or other_val < 1e-12:
                    continue

                # Score all positions for receiving mass
                # A position j is good if a[j] is low AND it doesn't
                # contribute much to any top lag
                scores = np.zeros(n)
                for k2 in top_k[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * (ac[k2i] / M_curr)
                        scores[k2i - idx2] += a[idx2] * (ac[k2i] / M_curr)

                scores[src] = np.inf
                scores[other] = np.inf

                # Try best candidates
                cand_order = np.argsort(scores)[:20]
                found = False
                for dst in cand_order:
                    if dst == src or dst == other:
                        continue
                    if a[dst] > 0.3 * src_val:
                        continue

                    for frac in [0.01, 0.02, 0.04, 0.06, 0.1]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            break
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    Surgical ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    return best_a, best_m


def subgradient_refine(a0, time_limit, N_samples=20, verbose=True):
    """
    Subgradient method with variance reduction:
    - Sample a random subset of critical lags each iteration
    - Take a projected gradient step
    - Use Polyak-type step size with lower bound J_lb = 2n/(2n-1)
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    J_lb = 2.0 * n / (2 * n - 1)
    lr = 0.1
    momentum = 0.9
    vel = np.zeros(n)
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            a = norm_sum1(best_a + rng.normal(0, 0.003, n) * best_a)
            stall = 0
            lr *= 0.5

        # Compute subgradient: sample random critical lags
        threshold = M_curr * 0.99
        critical = np.where(ac >= threshold)[0]
        if len(critical) == 0:
            critical = np.array([np.argmax(ac)])
        if len(critical) > N_samples:
            critical = critical[rng.choice(len(critical), N_samples, replace=False)]

        # Subgradient of max at sampled lags
        grad = np.zeros(n)
        for k in critical:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            # d(a*a)[k]/da[i] = a[k-i] + a[i-k] (if both valid and different)
            contrib1 = a[j_idx]
            # a[i-k] term
            i_k = idx - k_int
            valid = (i_k >= 0) & (i_k < n) & (i_k != j_idx)
            if valid.any():
                contrib1 = contrib1.copy()
                contrib1[valid] += a[i_k[valid]]
            grad[idx] += contrib1

        grad = grad / len(critical)

        # Momentum + step
        vel = momentum * vel + (1 - momentum) * grad
        lr_t = lr / (1.0 + 0.01 * np.sqrt(t))
        a_new = a - lr_t * vel
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Subgrad t={t} ({elapsed:.0f}s): J={J:.6f}, lr={lr_t:.4f}, stall={stall}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    Subgrad done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m
