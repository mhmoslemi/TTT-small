def _evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _simplex_project(v):
    n = len(v)
    mu = np.sort(v)[::-1]
    cv = 1.0 + mu * np.arange(1, n + 1)
    rho = np.where(mu * np.arange(1, n + 1) - cv > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    lam = cv[rho] / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _compute_objective(a, n):
    ac = _fast_autocorr(a)
    return 2.0 * n * ac.max()


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = _evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(round(sc, 8)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def _parametric_init(n, time_limit, best_init, best_score):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_init.copy()
    best_val = best_score

    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        if time.monotonic() - t0 > time_limit:
            break
        a = _norm_sum1((i_arr + 1.0) ** p)
        v = _compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()
        a = _norm_sum1((n - i_arr) ** p)
        v = _compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            if time.monotonic() - t0 > time_limit:
                break
            a = _norm_sum1(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
            v = _compute_objective(a, n)
            if v < best_val:
                best_val = v
                best_a = a.copy()

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            if time.monotonic() - t0 > time_limit:
                break
            a = _norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
            v = _compute_objective(a, n)
            if v < best_val:
                best_val = v
                best_a = a.copy()

    for p in [0.5, 1.0, 1.5, 2.0]:
        a = _norm_sum1(x ** p + (1.0 - x) ** p)
        v = _compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    for freq in [1, 2, 3, 5]:
        for ph in [0.0, np.pi / 2]:
            if time.monotonic() - t0 > time_limit:
                break
            a = _norm_sum1(np.maximum((1.0 + np.cos(2.0 * np.pi * freq * x + ph)) / 2.0, 0.01))
            v = _compute_objective(a, n)
            if v < best_val:
                best_val = v
                best_a = a.copy()

    rng = np.random.default_rng(123)
    for trial in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        a = _norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0))
        v = _compute_objective(a, n)
        if v < best_val:
            best_val = v
            best_a = a.copy()

    print("  Parametric done (" + str(round(time.monotonic() - t0, 1)) + "s): best J=" + repr(best_val))
    return best_a, best_val


def _fienup_init(n, iters=80, target_peak=1.05):
    a = np.ones(n) / n
    for it in range(iters):
        fft_a = np.fft.fft(a, 2 * n)
        mag = np.abs(fft_a)
        phase = fft_a / np.maximum(mag, 1e-15)
        new_mag = np.ones(2 * n)
        new_mag[0] = target_peak * n
        new_mag[-1] = target_peak * n
        new_fft = new_mag * phase
        a_new = np.real(np.fft.ifft(new_fft))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return _norm_sum1(a)


def _nesterov_softmax(a0, time_limit, beta_start=5.0, beta_end=150.0,
                       c0=3.0, verbose=True):
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    N = 1 << (3 * n - 2 - 1).bit_length()
    size = 2 * n - 1

    x = a.copy()
    x_prev = a.copy()
    beta_curr = beta_start

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = (t - 1.0) / max(t + 500.0, 1.0)
        beta_curr = beta_start * (beta_end / beta_start) ** frac

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = _fast_autocorr(y)
        log_vals = beta_curr * ac - beta_curr * ac.max()
        p = np.exp(log_vals)
        p = p / (p.sum() + 1e-30)

        Af = np.fft.rfft(y, N)
        Pf = np.fft.rfft(p, N)
        grad_full = np.fft.irfft(2.0 * np.real(Af) * Pf, N)
        grad = grad_full[:n]
        grad = np.maximum(grad, 0.0)

        eta = c0 / (np.sqrt(t) + 1.0)
        v = y - eta * grad
        a_new = _simplex_project(v)
        a_new = _norm_sum1(a_new)

        x_prev = x
        x = a_new

        M_new = _fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Nesterov t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 8)) + " beta=" + str(round(beta_curr, 1)))

    return best_a, best_m


def _build_constraint_coeffs(y, constraint_list, n):
    K = len(constraint_list)
    rows = []
    for k in constraint_list:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            rows.append((idx, y[k_int - idx]))
    return rows


def _lp_fw_soad(a0, time_limit, solver, initial_K=2000, max_K=3000,
                add_per_iter=40, verbose=True):
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = _fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 500
    N_fft = 1 << (2 * n - 1 - 1).bit_length()
    size = 2 * n - 1

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: no improve for " + str(no_improve) + " iters, stopping")
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        rows = _build_constraint_coeffs(y, constraint_list, n)
        rows = [r for r in rows if len(r[0]) > 0]
        if len(rows) < 10:
            continue

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1]
        for idx, vals in rows:
            constraints.append(cp.sum(cp.multiply(g_var[idx], vals)) <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(list(active_set)[:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        Yf = np.fft.rfft(y, N_fft)
        Gf = np.fft.rfft(g_dir, N_fft)
        conv_ys = np.fft.irfft(Yf * Gf, N_fft)[:size]

        violations = np.where(conv_ys > M_curr + M_curr * 1e-6 + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)

        best_alpha = default_alpha
        best_trial_M = M_curr + 1.0
        for alpha in [default_alpha, 0.5 * default_alpha, 1.5 * default_alpha,
                      1.0 / (t + 1.0), 2.5 / (t + 2.0)]:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = _fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

            # SOAD: check opposite direction
            y_op = (1.0 - best_alpha) * y - best_alpha * g_dir + y
            y_op = y + best_alpha * (y - g_dir)
            y_op = np.maximum(y_op, 0.0)
            s_op = y_op.sum()
            if s_op > 1e-15:
                y_op = y_op / s_op
                M_op = _fast_autocorr(y_op).max()
                if M_op < best_trial_M:
                    blend = 0.3
                    y_new = (1.0 - blend) * y_new + blend * y_op
                    y_new = np.maximum(y_new, 0.0)
                    s_y = y_new.sum()
                    if s_y > 1e-15:
                        y_new = y_new / s_y

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 8)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def _coord_polish(a0, time_limit, rng_seed=77, verbose=True):
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400
    step_scales = [0.003, 0.007, 0.015, 0.03, 0.06, 0.12]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = _fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 80:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = _fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 30 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.012)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = _fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 150 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 8)) + " stall=" + str(stall))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = _get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_score = _evaluate_safe(p)
            if np.isfinite(p_score):
                best_seq = _norm_sum1(p)
                best_score = p_score
                print("Parent: n=" + str(len(p)) + ", score=" + repr(round(p_score, 8)))
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = _norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = _evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 8)))

    n0 = len(best_seq)

    # --- Phase 1: Parametric + Fienup init ---
    print("\n=== Phase 1: Initialization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 10:
        a_param, _ = _parametric_init(n0, t1 * 0.7, best_seq, best_score)
        best_seq, best_score = _try_update(best_seq, best_score, a_param, "parametric", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 5:
        a_fien = _fienup_init(n0, iters=50)
        best_seq, best_score = _try_update(best_seq, best_score, a_fien, "fienup", t_global)

    # --- Phase 2: Nesterov softmax ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov softmax (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 30:
        a_out, _ = _nesterov_softmax(_norm_sum1(best_seq), t2,
                                     beta_start=5.0, beta_end=120.0, c0=3.0, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "nesterov_n" + str(n_best), t_global)

    # --- Phase 3: Main LP-FW with SOAD ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW SOAD (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.28, 450.0)
    if t3 > 40 and n_best <= 7500:
        a_out, _ = _lp_fw_soad(_norm_sum1(best_seq), t3, solver,
                               initial_K=2500, max_K=3500,
                               add_per_iter=50, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lp_soad_n" + str(n_best), t_global)

    # --- Phase 4: Coordinate polish ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Coordinate polish (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.05, 80.0)
    if t4 > 15:
        a_out, _ = _coord_polish(_norm_sum1(best_seq), t4, rng_seed=77, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "coord_n" + str(n_best), t_global)

    # --- Phase 5: Multi-n exploration ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-n exploration ===")
    longer_lengths = []
    for dn in [300, 600, -300, -600]:
        nl = n_best + dn
        if 4000 <= nl <= 7800 and nl != n_best:
            longer_lengths.append(nl)
    longer_lengths = longer_lengths[:4]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n  --- Exploring n=" + str(new_n) + " ---")
        a_long = _interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.20, 120.0)
        if t5a > 30:
            a_out, _ = _nesterov_softmax(a_long, t5a,
                                         beta_start=10.0, beta_end=100.0, c0=2.5, verbose=False)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100 and new_n <= 7200:
            t5b = min(remaining * 0.30, 180.0)
            if t5b > 30:
                a_out, _ = _lp_fw_soad(_norm_sum1(best_seq), t5b, solver,
                                       initial_K=2000, max_K=2800,
                                       add_per_iter=40, verbose=True)
                best_seq, best_score = _try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

    # --- Phase 6: Second LP-FW pass on best ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Second LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 80 and n_best <= 7500:
        t6 = min(remaining * 0.25, 250.0)
        if t6 > 30:
            a_out, _ = _lp_fw_soad(_norm_sum1(best_seq), t6, solver,
                                   initial_K=3000, max_K=4000,
                                   add_per_iter=60, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "lp2_n" + str(n_best), t_global)

    # --- Phase 7: Final coord polish ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final coord (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t7 = min(remaining * 0.50, 120.0)
        if t7 > 15:
            a_out, _ = _coord_polish(_norm_sum1(best_seq), t7, rng_seed=314, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # --- Phase 8: One more short Nesterov pass ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t8 = min(remaining * 0.40, 100.0)
        if t8 > 20:
            a_out, _ = _nesterov_softmax(_norm_sum1(best_seq), t8,
                                         beta_start=50.0, beta_end=200.0, c0=2.0, verbose=False)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 8)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = _norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = _evaluate_safe(best_seq)
        print("Fallback: score=" + repr(round(best_score, 8)))

    return [float(x) for x in best_seq]
