1. Projected gradient descent with Nesterov momentum on the simplex
2. Log-sum-exp smooth approximation of max, tracking top-K lags
3. Adaptive K (grows from 5 to 50) and beta (grows from 10 to 1000)

import numpy as np
import time
import math

def fast_conv(a):
    """Compute linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_gradient(a, top_lags, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*c[k])) for k in top_lags.
    Returns the gradient vector of length n.
    """
    n = len(a)
    c_vals = []
    for k in top_lags:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        c_k = np.dot(a[i_start:i_end], a[k_int - i_end + 1 : k_int - i_start + 1][::-1])
        # Simpler: c_k = sum_{i=i_start}^{i_end-1} a[i] * a[k_int - i]
        idx = np.arange(i_start, i_end)
        c_k = np.dot(a[idx], a[k_int - idx])
        c_vals.append(c_k)
    
    c_vals = np.array(c_vals)
    # Stable log-sum-exp
    c_max = c_vals.max()
    log_sum_exp = c_max + np.log(np.sum(np.exp(beta * (c_vals - c_max))))
    weights = np.exp(beta * (c_vals - c_max)) / np.sum(np.exp(beta * (c_vals - c_max)))
    
    # Gradient
    grad = np.zeros(n)
    for wi, k in zip(weights, top_lags):
        k_int = int(k)
        # grad[j] += 2 * wi * a[k_int - j] for valid j
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        idx = np.arange(j_start, j_end)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    
    return grad, log_sum_exp / beta

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Projected gradient descent on the simplex with adaptive step size
2. Log-sum-exp smooth approximation of max, tracking top-K lags (K grows adaptively)
3. Backtracking line search for stable convergence
4. Hot-spot local refinement targeting peak-contributing position pairs
5. Multi-length exploration and final coordinate descent polish

import numpy as np

def fast_conv(a):
    """Compute linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_obj_and_grad(a, lags, beta):
    """
    Compute (1/beta)*log(sum exp(beta*c[k])) and its gradient for k in lags.
    Returns (obj_value, gradient_vector).
    """
    n = len(a)
    c_vals = np.zeros(len(lags))
    for ii, k in enumerate(lags):
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        idx = np.arange(s, e)
        c_vals[ii] = np.dot(a[idx], a[k - idx])

    c_max = c_vals.max()
    shifted = beta * (c_vals - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    exp_sum = exps.sum()
    if exp_sum < 1e-300:
        weights = np.ones(len(lags)) / len(lags)
    else:
        weights = exps / exp_sum
    obj_val = (c_max + np.log(exp_sum)) / beta

    grad = np.zeros(n)
    for w, k in zip(weights, lags):
        if w < 1e-15:
            continue
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        idx = np.arange(s, e)
        grad[idx] += 2.0 * w * a[k - idx]

    return obj_val, grad

def main_optimize(a0, time_limit, verbose=True):
    """
    Main optimization: projected gradient descent with LSE smooth approximation.
    """
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    c = fast_conv(a)
    best_obj = 2.0 * n * c.max()

    t0 = time.monotonic()
    K = 5
    beta = 15.0
    eta = 2.0 / (n * K)
    momentum = 0.0
    v = np.zeros(n)  # velocity for heavy ball momentum

    iter_count = 0
    stall = 0
    max_stall = 500
    last_improve_iter = 0
    growth_counter = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        M = c.max()
        obj = 2.0 * n * M

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            stall = 0
            last_improve_iter = iter_count
        else:
            stall += 1

        if stall > max_stall:
            break

        # Identify top-K lags
        c_sorted_idx = np.argsort(c)[::-1]
        top_lags = c_sorted_idx[:K]

        # Compute gradient
        obj_lse, grad = compute_lse_obj_and_grad(a, top_lags, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Line search
        step_accepted = False
        eta_trial = eta
        for ls in range(8):
            a_trial = project_simplex(a - eta_trial * grad)
            c_trial = fast_conv(a_trial)
            obj_trial = 2.0 * n * c_trial.max()
            if obj_trial < obj - 1e-15 * eta_trial * grad_norm:
                step_accepted = True
                break
            eta_trial *= 0.5

        if step_accepted:
            a = a_trial
            # Update momentum
            v = momentum * v + (a - (a + eta_trial * grad))  # not quite right
            # Actually, let's just use simple GD without momentum for stability
            momentum = min(0.5, momentum + 0.001)
        else:
            # Try with smaller eta
            eta *= 0.5
            if eta < 1e-8:
                break

        # Adapt K and beta
        growth_counter += 1
        if growth_counter % 80 == 0:
            if K < 40 and stall < max_stall // 2:
                K = min(K + 2, 40)
            if beta < 500:
                beta *= 1.15

        # Periodic restart with small random perturbation
        if stall > max_stall // 2 and stall % 50 == 0:
            rng = np.random.default_rng(iter_count)
            perturb = rng.normal(0, 0.001, n)
            a_perturbed = project_simplex(a + perturb)
            c_p = fast_conv(a_perturbed)
            if 2.0 * n * c_p.max() < obj:
                a = a_perturbed
                stall = max(0, stall - 50)

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MainOpt iter={iter_count} ({elapsed:.0f}s): obj={obj:.6f}, best={best_obj:.6f}, K={K}, beta={beta:.1f}, eta={eta:.2e}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    MainOpt done ({elapsed:.0f}s): best_obj={best_obj:.6f}, iters={iter_count}")
    return best_a

def hotspot_refine(a0, time_limit, verbose=True):
    """
    Local refinement: identify hot-spot pairs contributing to peak lags,
    and move mass from them to cold positions.
    """
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    c = fast_conv(a)
    best_obj = 2.0 * n * c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 300
    n_top_lags = 8

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        obj = 2.0 * n * c.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        top_lags = np.argsort(c)[::-1][:n_top_lags]
        improved = False

        for k in top_lags:
            k = int(k)
            s = max(0, k - n + 1)
            e = min(n, k + 1)
            idx = np.arange(s, e)
            products = a[idx] * a[k - idx]
            total = products.sum()

            if total < 1e-20:
                continue

            # Find the contributing pairs
            # For the top pairs, try moving mass from the larger contributor
            pair_contribs = products.argsort()[::-1][:5]

            for pi in pair_contribs:
                i = idx[pi]
                j = k - i
                # i and j are the two positions contributing to lag k
                # Try reducing a[i] or a[j] and adding to a "cold" position

                for src in [i, j]:
                    src_val = a[src]
                    if src_val < 1e-10:
                        continue

                    # Find cold positions (not contributing to any top lag)
                    in_hot = np.zeros(n, dtype=bool)
                    for k2 in top_lags:
                        k2 = int(k2)
                        s2 = max(0, k2 - n + 1)
                        e2 = min(n, k2 + 1)
                        in_hot[s2:e2] = True

                    cold_mask = ~in_hot
                    if not cold_mask.any():
                        cold_mask = np.ones(n, dtype=bool)
                        cold_mask[src] = False

                    # Prefer cold positions with currently low a values
                    cold_idx = np.where(cold_mask)[0]
                    if len(cold_idx) == 0:
                        continue
                    # Score: prefer positions with lower a values
                    scores = a[cold_idx]
                    dst = cold_idx[np.argmin(scores)]
                    if dst == src:
                        continue

                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        s_sum = a_trial.sum()
                        if s_sum < 1e-15:
                            continue
                        a_trial = a_trial / s_sum
                        c_trial = fast_conv(a_trial)
                        obj_trial = 2.0 * n * c_trial.max()
                        if obj_trial < best_obj - 1e-14:
                            a = a_trial
                            best_obj = obj_trial
                            best_a = a.copy()
                            improved = True
                            break

            if improved:
                stall = 0
                break

        if not improved:
            # Small random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Hotspot done ({elapsed:.0f}s): best_obj={best_obj:.6f}")
    return best_a

def coord_descent_refine(a0, time_limit, verbose=True):
    """
    Coordinate descent: for each pair (i, j), try moving a small amount
    from i to j and keep if it improves.
    """
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    c = fast_conv(a)
    best_obj = 2.0 * n * c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        obj = 2.0 * n * c.max()
        top_lags = np.argsort(c)[::-1][:5]

        # Identify hot positions
        hot = np.zeros(n, dtype=bool)
        for k in top_lags:
            k = int(k)
            s = max(0, k - n + 1)
            e = min(n, k + 1)
            hot[s:e] = True

        # Try moving mass from hot to cold positions
        hot_positions = np.where(hot & (a > 1e-8))[0]
        cold_positions = np.where(~hot)[0]

        n_tries = 0
        max_tries = min(500, n // 5)
        improved = False

        while n_tries < max_tries and time.monotonic() - t0 < time_limit:
            if len(hot_positions) == 0 or len(cold_positions) == 0:
                break
            src = hot_positions[int(rng.integers(len(hot_positions)))]
            dst = cold_positions[int(rng.integers(len(cold_positions)))]
            frac = rng.uniform(0.01, 0.15)
            amt = a[src] * frac

            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_sum = a_trial.sum()
            if s_sum < 1e-15:
                n_tries += 1
                continue
            a_trial = a_trial / s_sum

            # Quick check: only compute the affected lags
            # For efficiency, compute full conv (n is manageable)
            c_trial = fast_conv(a_trial)
            obj_trial = 2.0 * n * c_trial.max()

            if obj_trial < best_obj - 1e-14:
                a = a_trial
                best_obj = obj_trial
                best_a = a.copy()
                improved = True
                # Rebuild hot/cold
                c = fast_conv(a)
                top_lags = np.argsort(c)[::-1][:5]
                hot = np.zeros(n, dtype=bool)
                for k in top_lags:
                    k = int(k)
                    s = max(0, k - n + 1)
                    e = min(n, k + 1)
                    hot[s:e] = True
                hot_positions = np.where(hot & (a > 1e-8))[0]
                cold_positions = np.where(~hot)[0]

            n_tries += 1

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.005, 0.03)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
        else:
            stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDesc done ({elapsed:.0f}s): best_obj={best_obj:.6f}")

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase 5 improved: {obj5:.6f}")

    # === Phase 6: Final hotspot ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.50, 100.0)
    if t6 > 15:
        print(f"\n=== Phase 6: Final Hotspot ===")
        a6 = hotspot_refine(best_seq, t6, verbose=False)
        n6 = len(a6)
        obj6 = 2.0 * n6 * fast_conv(a6).max()
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase 6 improved: {obj6:.6f}")

    # Final validation with evaluate_sequence
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
