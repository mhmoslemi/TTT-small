```
def top_lags(c, K):
    L=len(c)
    if L==0 or K<=0: return np.array([], dtype=int)
    K=min(K,L)
    if K>=L:
       order=np.argsort(c)[::-1]
       return order.astype(int)
    idx=np.argpartition(c, L-K)[-K:]
    vals=c[idx]
    order=np.argsort(vals)[::-1]
    return idx[order].astype(int)
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2.

Algorithm:
- Start from the supplied parent when available, plus a broad parametric/random
  multi-start phase over several sequence lengths.
- Polishing uses an FFT-based mirror-flow method on the simplex.  The non-smooth
  max-convolution objective is approximated by a softmax over the most active
  convolution lags, with adaptive smoothing/step size.
- A cutting-plane LP phase attempts to find normalized non-negative directions
  whose mixed convolution with the current sequence is flatter on the active
  lags.
- Final local refinement uses coordinate mass transfer.
"""

import numpy as np
import time

    from scipy import optimize, sparse
    HAS_SCIPY = True
except Exception:
    HAS_SCIPY = False


def next_power_of_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_one(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.shape, dtype=np.float64) / float(a.size)
    return a / s


def evaluate_safe(a):
    try:
        lst = [float(x) for x in a]
        val = float(evaluate_sequence(lst))
        return val
    except Exception:
        return float("inf")


def fast_autocorr(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = 2 * n - 1
    if fft_size is None or int(fft_size) < size:
        N = next_power_of_two(size)
    else:
        N = int(fft_size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.maximum(c, 0.0)


def cross_correlation(weights, a, fft_size=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = 2 * n - 1
    if fft_size is None or int(fft_size) < size:
        N = next_power_of_two(size)
    else:
        N = int(fft_size)

    w = np.asarray(weights, dtype=np.float64)
    if w.size < size:
        wp = np.zeros(size, dtype=np.float64)
        wp[:w.size] = w
    else:
        wp = w[:size]

    W = np.fft.rfft(wp, N)
    A = np.fft.rfft(a, N)
    h = np.fft.irfft(W * np.conj(A), N)[:n]
    return np.maximum(h, 0.0)


def select_top(c, K):
    L = int(c.size)
    K = int(K)
    if L == 0 or K <= 0:
        return np.empty(0, dtype=int)
    K = min(K, L)
    if K >= L:
        order = np.argsort(c)[::-1]
        return order.astype(int)
    idx = np.argpartition(c, L - K)[-K:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    return idx[order].astype(int)


def make_weights(c, K, T):
    L = int(c.size)
    K = int(K)
    if L == 0 or K <= 0:
        return np.zeros(L, dtype=np.float64)
    K = min(K, L)
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, L - K)[-K:]
    vals = c[idx]
    vmax = float(vals.max())
    T = max(float(T), 1e-300)
    z = (vals - vmax) / T
    z = np.clip(z, -745.0, 0.0)
    ww = np.exp(z)
    sw = float(ww.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        ww = np.ones(K, dtype=np.float64) / float(K)
    else:
        ww = ww / sw
    w = np.zeros(L, dtype=np.float64)
    w[idx] = ww
    return w


def interp_sequence(parent, n):
    parent = np.asarray(parent, dtype=np.float64)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.maximum(parent, 0.0)
    m = parent.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if m == n:
        return normalize_one(parent)
    old_x = np.linspace(0.0, 1.0, m)
    new_x = np.linspace(0.0, 1.0, n)
    vals = np.interp(new_x, old_x, parent)
    return normalize_one(np.maximum(vals, 0.0))


def make_initials(n, parent=None):

def mgd_smooth(a0, time_limit, K, smooth_start, smooth_end, eta_start,
               max_patience=1200, seed=1):
    a = normalize_one(a0)
    n = a.size
    if n <= 2 or float(time_limit) <= 1.0:
        return a

    t0 = time.monotonic()
    fft_size = next_power_of_two(2 * n - 1)
    c = fast_autocorr(a, fft_size)
    M = float(c.max())

    best_a = a.copy()
    best_m = M
    eta = float(eta_start)
    eta_min = 1e-4
    eta_max = 1.0
    patience = 0
    rng = np.random.default_rng(seed)

    K = int(max(20, min(K, 2 * n - 1)))
    L = 2 * n - 1
    smooth_start = max(float(smooth_start), 1e-8)
    smooth_end = max(float(smooth_end), 1e-9)

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a, fft_size)
        M = float(c.max())

        if M < best_m - 1e-18:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.25, n))
            a = a * noise
            a = normalize_one(a)
            patience = 0
            eta = max(0.05, float(eta_start) * 0.7)
            continue

        if K >= L:
            spread = float(c.max() - c.min())
        else:
            tmp_idx = np.argpartition(c, L - K)[-K:]
            tmp_vals = c[tmp_idx]
            spread = float(tmp_vals.max() - tmp_vals.min())

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-9))
        smooth = smooth_start * (smooth_end / smooth_start) ** progress
        T = max(M * smooth, spread * 0.30, M * 1e-9, 1e-300)

        w = make_weights(c, K, T)
        h = cross_correlation(w, a, fft_size)
        grad_raw = 2.0 * h
        center = float(np.dot(grad_raw, a))
        grad = grad_raw - center
        scale = float(np.max(np.abs(grad)))

        if not np.isfinite(scale) or scale < 1e-20:
            break

        direction = grad / scale
        accepted = False

        for _trial in range(3):
            z = -eta * direction
            z = np.clip(z, -12.0, 12.0)
            cand = a * np.exp(z)
            cand = np.maximum(cand, 0.0)
            s = float(cand.sum())
            if s <= 1e-300:
                eta *= 0.5
                continue
            cand = cand / s
            M_cand = float(fast_autocorr(cand, fft_size).max())

            if M_cand <= M + 1e-17:
                a = cand
                accepted = True
                if M_cand < M - 1e-18:
                    eta = min(eta * 1.04, eta_max)
                else:
                    eta *= 0.98
                break
            eta *= 0.5

        if not accepted:
            eta = max(eta * 0.5, eta_min)
            z = rng.normal(0.0, 0.02, n)
            a = a * np.exp(z)
            a = normalize_one(a)
            patience = max(0, patience - 250)

    return best_a


def build_active_matrix(f, active):
    n = int(f.size)
    K = int(active.size)
    cols_parts = []
    data_parts = []
    row_parts = []

    for r, k in enumerate(active):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1, dtype=np.int64)
            cols_parts.append(idx)
            data_parts.append(np.asarray(f[k - idx], dtype=np.float64))
            row_parts.append(np.full(idx.size, r, dtype=np.int64))

    if cols_parts:
        cols = np.concatenate(cols_parts)
        data = np.concatenate(data_parts)
        rows = np.concatenate(row_parts)
        return sparse.csr_matrix((data, (rows, cols)), shape=(K, n))

    return sparse.csr_matrix((K, n))


def lp_active_refine(a0, time_limit, max_lags=None, verbose=False):
    a = normalize_one(a0)
    n = a.size
    if n <= 2 or float(time_limit) <= 5.0 or not HAS_SCIPY:
        return a

    if max_lags is None:
        K = int(max(80, min(600, n // 20)))
    else:
        K = int(max(80, min(max_lags, 2 * n - 1)))

    fft_size = next_power_of_two(2 * n - 1)
    c = fast_autocorr(a, fft_size)
    best_a = a.copy()
    best_m = float(c.max())

    active = select_top(c, K)
    eq = np.ones(n, dtype=np.float64)
    eq_t = np.concatenate([eq, [0.0]])
    box = [(0.0, 1.0)] * n
    box_t = [(0.0, 1.0)] * (n + 1)

    t0 = time.monotonic()
    fail = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a, fft_size)
        M = float(c.max())

        if M < best_m - 1e-18:
            best_m = M
            best_a = a.copy()

        if active.size < K:
            all_lags = np.unique(np.concatenate([active, select_top(c, K)]))
            vals = c[all_lags]
            order = np.argsort(vals)[::-1]
            active = all_lags[order[:K]]

        A = build_active_matrix(a, active)
        Kc = int(active.size)
        g = None
        t_val = M

        try:
            target = M * 0.99995
            res = optimize.linprog(
                c=-a,
                A_ub=A,
                b_ub=np.full(Kc, target, dtype=np.float64),
                A_eq=eq,
                b_eq=1.0,
                bounds=box,
                method="highs"
            )
            if res.status == 0:
                g = res.x
                t_val = 0.0
        except Exception:
            g = None

        if g is None:
            try:
                A_t = sparse.hstack(
                    [A, sparse.csr_matrix(np.full((Kc, 1), -1.0))],
                    format="csr"
                )
                c_obj = np.empty(n + 1, dtype=np.float64)
                c_obj[:n] = -1e-8 * a
                c_obj[n] = 1.0

                res = optimize.linprog(
                    c=c_obj,
                    A_ub=A_t,
                    b_ub=np.zeros(Kc, dtype=np.float64),
                    A_eq=eq_t,
                    b_eq=1.0,
                    bounds=box_t,
                    method="highs"
                )
                if res.status == 0:
                    g = res.x[:n]
                    t_val = float(res.x[-1])
                else:
                    fail += 1
                    active = select_top(c, K)
                    if fail >= 6:
                        break
                    continue
            except Exception:
                fail += 1
                active = select_top(c, K)
                if fail >= 6:
                    break
                continue

        if t_val != 0.0 and t_val >= M - 1e-12:
            fail += 1
            if fail >= 4:
                break
            active = select_top(c, K)
            continue

        g = np.asarray(g, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs <= 1e-300:
            fail += 1
            continue
        g = g / gs

        best_alpha = 0.0
        best_obj = M
        for alpha in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05):
            cand = (1.0 - alpha) * a + alpha * g
            cand = normalize_one(cand)
            mc = float(fast_autocorr(cand, fft_size).max())
            if mc < best_obj - 1e-16:
                best_obj = mc
                best_alpha = alpha

        if best_alpha > 0.0:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = normalize_one(a)
            fail = 0
            iters += 1

            c_new = fast_autocorr(a, fft_size)
            all_lags = np.unique(np.concatenate([active, select_top(c_new, K)]))
            vals = c_new[all_lags]
            order = np.argsort(vals)[::-1]
            active = all_lags[order[:K]]

            if verbose and (iters % 5 == 0):
                print(f"      LP iter={iters}, score={2.0 * n * best_m:.6f}")
        else:
            fail += 1
            cg = fast_autocorr(g, fft_size)
            all_lags = np.unique(np.concatenate([active, select_top(cg, K)]))
            vals = cg[all_lags]
            order = np.argsort(vals)[::-1]
            active = all_lags[order[:K]]
            if fail >= 8:
                break

    return best_a


def coordinate_polish(a0, time_limit, verbose=False):
    a = normalize_one(a0)
    n = a.size
    if n <= 2 or float(time_limit) <= 3.0:
        return a

    fft_size = next_power_of_two(2 * n - 1)
    c = fast_autocorr(a, fft_size)
    best_a = a.copy()
    best_m = float(c.max())

    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    fail = 0
    K_top = min(12, 2 * n - 1)
    K_src = 5
    K_dst = 6

    while time.monotonic() - t0 < time_limit and fail < 400:
        c = fast_autocorr(a, fft_size)
        M = float(c.max())
        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()

        top = select_top(c, K_top)
        src_scores = np.zeros(n, dtype=np.float64)
        peak_contrib = np.zeros(n, dtype=np.float64)

        for k in top:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            pair = a[k - idx]
            src_scores += a[idx] * pair
            peak_contrib[idx] += pair
            peak_contrib[k - idx] += a[idx]

        src_order = np.argsort(src_scores)[::-1][:K_src]
        dst_order = np.argsort(peak_contrib)[:K_dst]

        improved = False
        for s in src_order:
            s = int(s)
            if a[s] < 1e-13:
                continue
            for d in dst_order:
                d = int(d)
                if d == s:
                    continue
                for frac in (0.01, 0.03, 0.08):
                    amt = a[s] * frac
                    cand = a.copy()
                    cand[s] -= amt
                    cand[d] += amt
                    cand = normalize_one(cand)
                    mc = float(fast_autocorr(cand, fft_size).max())
                    if mc < best_m - 1e-16:
                        a = cand
                        best_m = mc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            fail = 0
        else:
            fail += 1
            s = int(rng.integers(n))
            d = int(dst_order[0]) if dst_order.size else int(rng.integers(n))
            cand = None

            if s != d and a[s] > 1e-13:
                amt = a[s] * float(rng.uniform(0.002, 0.02))
                c1 = a.copy()
                c1[s] -= amt
                c1[d] += amt
                cand = normalize_one(c1)

            if cand is None:
                z = rng.normal(0.0, 0.01, n)
                cand = a * np.exp(z)
                cand = normalize_one(cand)

            mc = float(fast_autocorr(cand, fft_size).max())
            if mc < best_m - 1e-16:
                a = cand
                best_m = mc
                best_a = a.copy()
                fail = 0

        if verbose and (time.monotonic() - t0) % 30 < 2:
            print(f"      Coord score={2.0 * n * best_m:.6f}, stall={fail}")

    return best_a


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    total_budget = float(budget_s)

    if total_budget <= 20.0:
        x = np.linspace(0.0, 1.0, 3000)
        a = normalize_one(np.exp(2.0 * x))
        return [float(v) for v in a]

    best_seq = None
    best_score = float("inf")

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        p = normalize_one(p)
        sc = evaluate_safe(p)
        if np.isfinite(sc):
            best_seq = p
            best_score = sc
            print(f"Parent: n={len(p)}, score={sc:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 5000
        x = np.linspace(0.0, 1.0, n0)
        best_seq = normalize_one(np.exp(2.0 * x))
        best_score = evaluate_safe(best_seq)
        print(f"Initial parent fallback: n={n0}, score={best_score:.6f}")

    parent_n = int(best_seq.size) if best_seq is not None else 0
    if parent_n >= 1000:
        cand_list = [parent_n, parent_n + 600, parent_n + 1500, 5000, 8000]
    else:
        cand_list = [3000, 4500, 6000, 8000]

    cleaned = []
    seen = set()
    for v in cand_list:
        iv = int(v)
        if 1000 <= iv <= 9000 and iv not in seen:
            seen.add(iv)
            cleaned.append(iv)
    cleaned.sort()
    if len(cleaned) > 5:
        cleaned = cleaned[:5]

    p1_time = max(0.0, min(total_budget * 0.22 - (time.monotonic() - t_global), 380.0))
    if p1_time > 30.0:
        t1_end = time.monotonic() + p1_time
        per_n = p1_time / max(1, len(cleaned))
        print(f"\n=== Phase 1: multi-start, budget={p1_time:.0f}s, lengths={cleaned} ===")

        for n in cleaned:
            if time.monotonic() >= t1_end:
                break

            inits = make_initials(n, best_seq)
            if not inits:
                continue

            scored = []
            for idx, a in enumerate(inits):
                sc = 2.0 * float(n) * float(fast_autocorr(a).max())
                if np.isfinite(sc):
                    scored.append((float(sc), idx, a))

            if not scored:
                continue

            scored.sort()
            local_a = scored[0][2]
            local_sc = float(scored[0][0])

            nof = min(3, len(scored))
            t_sub = min(per_n / max(1, nof), 25.0)
            Kq = int(max(40, min(400, n // 15)))

            for r in range(nof):
                if time.monotonic() >= t1_end:
                    break
                a_init = scored[r][2]
                if t_sub > 3.0:
                    a_out = mgd_smooth(
                        a_init,
                        t_sub,
                        Kq,
                        0.10,
                        0.010,
                        0.07,
                        seed=1000 + 31 * n + 7 * r
                    )
                    sc = 2.0 * float(n) * float(fast_autocorr(a_out).max())
                    if np.isfinite(sc) and sc < local_sc:
                        local_a = a_out
                        local_sc = float(sc)

            sc_actual = evaluate_safe(local_a)
            if np.isfinite(sc_actual) and sc_actual < best_score:
                best_score = sc_actual
                best_seq = local_a.copy()
                print(f"  Phase1 n={n}: score={sc_actual:.6f}")

    used = time.monotonic() - t_global
    p2_time = max(0.0, min(total_budget * 0.38 - used, 750.0))
    if p2_time > 30.0:
        n = int(best_seq.size)
        phase_end = time.monotonic() + p2_time
        pass_time = min(p2_time / 3.0, 320.0)
        print("\n=== Phase 2: deep smooth mirror descent ===")

        pass_configs = [
            (n // 8, 0.08, 0.008, 0.06),
            (n // 5, 0.05, 0.003, 0.04),
            (n // 3, 0.02, 0.0008, 0.025),
        ]
        seeds = [41, 42, 43]

        for pi in range(3):
            remaining = phase_end - time.monotonic()
            if remaining <= 15.0:
                break
            t_pass = min(pass_time, remaining)
            Kp = int(max(80, min(2000, pass_configs[pi][0])))
            sp = pass_configs[pi][1]
            se = pass_configs[pi][2]
            et = pass_configs[pi][3]

            a_out = mgd_smooth(best_seq, t_pass, Kp, sp, se, et, seed=seeds[pi])
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Phase2 pass {pi + 1}: score={sc:.6f}")

    used = time.monotonic() - t_global
    p3_time = max(0.0, min(total_budget * 0.22 - used, 420.0))
    if p3_time > 60.0:
        n = int(best_seq.size)
        Kl = int(max(100, min(600, n // 20)))
        print(f"\n=== Phase 3: LP active set, budget={p3_time:.0f}s ===")
        a_out = lp_active_refine(best_seq, p3_time, max_lags=Kl, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase3: score={sc:.6f}")

    used = time.monotonic() - t_global
    p4_time = max(0.0, min(total_budget * 0.12 - used, 220.0))
    if p4_time > 20.0:
        print(f"\n=== Phase 4: coordinate polish, budget={p4_time:.0f}s ===")
        a_out = coordinate_polish(best_seq, p4_time, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase4: score={sc:.6f}")

    remaining = total_budget - 30.0 - (time.monotonic() - t_global)
    if remaining > 60.0:
        n = int(best_seq.size)
        t5 = min(remaining * 0.5, 180.0)
        Kf = int(max(120, min(2200, n // 3)))
        print(f"\n=== Phase 5: final MGD, budget={t5:.0f}s ===")
        a_out = mgd_smooth(best_seq, t5, Kf, 0.015, 0.0003, 0.015, seed=97)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase5: score={sc:.6f}")

        remaining = total_budget - 30.0 - (time.monotonic() - t_global)
        if remaining > 30.0:
            t6 = min(remaining * 0.6, 120.0)
            print(f"\n=== Phase 6: final coordinate, budget={t6:.0f}s ===")
            a_out = coordinate_polish(best_seq, t6, verbose=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Phase6: score={sc:.6f}")

    if best_seq is None:
        x = np.linspace(0.0, 1.0, 4000)
        best_seq = normalize_one(np.exp(2.0 * x))
        best_score = evaluate_safe(best_seq)

    final_score = evaluate_safe(best_seq)
    if np.isfinite(final_score):
        best_score = final_score

    if best_seq is None or best_seq.size == 0 or not np.all(np.isfinite(best_seq)):
        x = np.linspace(0.0, 1.0, 4000)
        best_seq = normalize_one(np.exp(2.0 * x))

    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.maximum(best_seq, 0.0)
    ss = float(best_seq.sum())
    if ss <= 1e-12:
        best_seq = np.ones(max(500, int(best_seq.size)), dtype=np.float64) / 500.0
    else:
        best_seq = best_seq / ss

    elapsed = time.monotonic() - t_global
    print(f"\nFINAL: score={best_score:.6f}, n={best_seq.size}, time={elapsed:.0f}s")
    return [float(v) for v in best_seq]
```
