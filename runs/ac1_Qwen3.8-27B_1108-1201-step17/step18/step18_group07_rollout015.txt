```
def _spectral_smooth(a, power=0.3, mix=0.5):
    n=len(a)
    size=_next_pow2(2*n)
    A=np.fft.rfft(a,size)
    mag=np.abs(A); phase=np.angle(A)
    mm=float(np.mean(mag))
```

```python
"""
Search for a non-negative step-function height sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2

Strategy:
  - Start from the parent construction if available.
  - Normalize to the probability simplex and optimize with fast FFT-based
    smoothed-gradient moves.
  - Use several complementary local moves:
      * log-domain multiplicative descent from peak-weighted subgradients,
      * signed balance weights (peaks vs valleys) to flatten the square,
      * targeted pair transfers guided by gradient magnitude,
      * smooth random multiplicative perturbations,
      * spectral magnitude compression,
      * occasional random geometry restarts.
  - Try diverse starts and modest length variations, then a final polish.
"""

import numpy as np
import time


def _next_power_two(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _norm_simplex(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _conv_square(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    m = 2 * n - 1
    size = _next_power_two(m)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)[:m]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def _adjoint_weights(a, w):
    """
    For c_k = sum_i a_i a_{k-i}, return h_j = sum_k w_k a_{k-j}
    for the valid convolution adjoint.  Works with signed weights too.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    L = w.size
    if n == 0 or L == 0:
        return np.zeros(n, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    w = np.where(np.isfinite(w), w, 0.0)
    m = n + L - 1
    size = _next_power_two(m)
    Af = np.fft.rfft(a, size)
    Wf = np.fft.rfft(np.flip(w), size)
    P = np.fft.irfft(Af * Wf, size)[:m]
    idx = (L - 1) - np.arange(n, dtype=np.int64)
    h = P[idx]
    h = np.where(np.isfinite(h), h, 0.0)
    return h


def _logsum_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.array([], dtype=np.float64)
    max_c = float(np.max(c))
    if max_c <= 1e-300:
        return np.ones(L, dtype=np.float64) / float(L)
    x = beta * (c / max_c)
    mx = float(np.max(x))
    x = x - mx
    x = np.clip(x, -700.0, 0.0)
    w = np.exp(x)
    s = float(np.sum(w))
    if s <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    floor = 0.02
    w = (1.0 - floor) * w + floor / float(L)
    ws = float(np.sum(w))
    if ws <= 1e-300:
        return np.ones(L, dtype=np.float64) / float(L)
    return w / ws


def _balance_weights(c):
    """Signed weights: positive at high coefficients, negative at low ones."""
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.array([], dtype=np.float64)
    s = float(np.sum(c))
    if s <= 1e-300:
        return np.zeros(L, dtype=np.float64)
    p = c / s
    base = p - (1.0 / float(L))
    base = np.where(np.isfinite(base), base, 0.0)
    shaped = np.sign(base) * np.power(np.abs(base) + 1e-15, 0.8)
    shaped = np.where(shaped > 0.0, shaped, 0.6 * shaped)
    denom = float(np.sum(np.abs(shaped)))
    if denom <= 1e-15:
        return np.zeros(L, dtype=np.float64)
    return shaped / denom


def _log_update(a, g, eta):
    a = _norm_simplex(a)
    g = np.asarray(g, dtype=np.float64)
    if g.size != a.size:
        return a.copy()
    if eta <= 0.0:
        return a.copy()
    g = np.where(np.isfinite(g), g, 0.0)
    gc = g - float(np.mean(g))
    gmin = float(np.min(gc))
    gmax = float(np.max(gc))
    if not np.isfinite(gmax) or (gmax - gmin) <= 1e-15:
        return a.copy()
    gn = (gc - gmin) / (gmax - gmin)
    base = np.maximum(a, 1e-15)
    lv = np.log(base) - eta * gn
    lv = lv - float(np.max(lv))
    lv = np.clip(lv, -700.0, 0.0)
    u = np.exp(lv)
    return _norm_simplex(u)


def _spectral_smooth(a, power=0.3, mix=0.5):
    a = _norm_simplex(a)
    n = a.size
    if n == 0:
        return a
    size = _next_power_two(2 * n)
    A = np.fft.rfft(a, size)
    mag = np.abs(A)
    phase = np.angle(A)
    mm = float(np.mean(mag))
    if mm <= 1e-300:
        return a.copy()
    ratio = mag / (mm + 1e-300)
    ratio = np.where(np.isfinite(ratio), ratio, 1.0)
    ratio = np.minimum(ratio, 1e6)
    exponent = max(0.0, min(1.0, 1.0 - power))
    mag_new = mm * np.power(ratio, exponent)
    A_new = mag_new * np.exp(1j * phase)
    b = np.fft.irfft(A_new, size)[:n]
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = float(np.sum(b))
    if s <= 1e-15:
        return a.copy()
    b = b / s
    out = (1.0 - mix) * a + mix * b
    return _norm_simplex(out)


def _random_direction(n, rng):
    phi = np.zeros(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    for m in range(1, 8):
        amp = rng.normal(0.0, 0.6 / np.sqrt(float(m)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        phi += amp * np.sin(2.0 * np.pi * m * x + ph)
    phi = phi + rng.normal(0.0, 0.4, n)
    phi = phi - float(np.mean(phi))
    amax = float(np.max(np.abs(phi)))
    if amax <= 1e-15:
        phi = rng.normal(0.0, 1.0, n)
        amax = float(np.max(np.abs(phi)))
        if amax > 1e-15:
            phi = phi / amax
    else:
        phi = phi / amax
    return phi


def _random_smooth_init(n, seed, base=None):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    v = np.ones(n, dtype=np.float64)
    for m in range(1, 16):
        amp = rng.normal(0.0, 0.5 / np.sqrt(float(m)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        v += amp * np.sin(2.0 * np.pi * m * x / float(n) + ph)
    v = np.maximum(v, 0.01)
    s = float(np.sum(v))
    if s <= 1e-15:
        v = np.ones(n, dtype=np.float64)
        s = float(n)
    v = v / s
    if base is not None:
        b = _norm_simplex(base)
        if b.size == n:
            v = 0.2 * v + 0.8 * b
    return _norm_simplex(v)


def _interpolate_sequence(f, new_n):
    f = _norm_simplex(f)
    n = f.size
    if n == 0:
        return np.ones(max(1, new_n), dtype=np.float64) / float(max(1, new_n))
    if n == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    return _norm_simplex(g)


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    S = float(np.sum(a))
    if S <= 1e-15:
        return float("inf")
    c = _conv_square(a)
    M = float(np.max(c))
    return float(2.0 * n * M / (S * S))


def _true_score(a):
    try:
        func = globals().get("evaluate_sequence", None)
        if func is None:
            return _fast_score(a)
        return float(func([float(x) for x in a]))
    except Exception:
        return _fast_score(a)


def _local_search(a0, deadline, seed=0, verbose=False, beta_start=3.0, beta_end=80.0):
    a0 = np.asarray(a0, dtype=np.float64)
    n = a0.size
    if n == 0:
        return a0, 0.0, float("inf")

    a_work = _norm_simplex(a0)
    c = _conv_square(a_work)
    best_M = float(np.max(c))
    best_a = a_work.copy()
    current_M = best_M

    t0 = time.monotonic()
    limit = max(deadline - t0, 1e-6)
    if limit <= 0.0:
        return best_a, best_M, float(2.0 * n * best_M)

    rng = np.random.default_rng(seed)
    iter_count = 0
    stall = 0

    while time.monotonic() < deadline:
        iter_count += 1
        progress = min(1.0, (time.monotonic() - t0) / limit)

        if current_M > best_M * 1.01 + 1e-14:
            a_work = best_a.copy()
            current_M = best_M
            stall = 0

        if iter_count < 50:
            beta = beta_start + (8.0 - beta_start) * min(1.0, float(iter_count) / 50.0)
        else:
            beta = 8.0 + (beta_end - 8.0) * progress
            if stall > 30:
                beta = min(300.0, beta + 4.0)

        c = _conv_square(a_work)
        M_work = float(np.max(c))
        if M_work < current_M:
            current_M = M_work

        best_updated = False
        moved = False
        if M_work < best_M - 1e-15:
            best_M = M_work
            best_a = a_work.copy()
            stall = 0
            best_updated = True
        else:
            stall += 1

        if best_updated:
            if verbose and iter_count % 1000 == 0:
                print(
                    "      local iter={} t={:.0f}s best={:.7f} stall={} beta={:.1f}".format(
                        iter_count, time.monotonic() - t0, 2.0 * n * best_M, stall, beta
                    )
                )
            continue

        accept_tol = max(
            1e-14,
            best_M * (1e-5 + 1e-4 * min(1.0, float(stall) / 100.0)) * (1.0 - 0.5 * progress),
        )

        # 1) Peak-focused log-sum-exp subgradient.
        if not best_updated and not moved:
            w = _logsum_weights(c, beta)
            g = 2.0 * _adjoint_weights(a_work, w)
            gmin = float(np.min(g))
            gmax = float(np.max(g))
            old_current = current_M
            if np.isfinite(gmax) and (gmax - gmin) > 1e-14:
                gnorm = (g - gmin) / (gmax - gmin)
                for eta in (0.2, 0.5, 1.0, 2.0, 0.1, 0.05):
                    if time.monotonic() >= deadline:
                        break
                    cand = _log_update(a_work, gnorm, eta)
                    Mc = float(np.max(_conv_square(cand)))
                    if Mc < best_M - 1e-15:
                        best_M = Mc
                        best_a = cand.copy()
                        a_work = cand.copy()
                        current_M = Mc
                        stall = 0
                        best_updated = True
                        moved = True
                        break
                    elif (
                        Mc < old_current + accept_tol
                        and Mc < best_M * 1.005
                        and (Mc < old_current or stall > 10)
                    ):
                        a_work = cand
                        current_M = Mc
                        moved = True
                        break

        # 2) Signed peak-vs-valley balancing move.
        if not best_updated and not moved and (iter_count % 2 == 0 or stall > 5):
            old_current = current_M
            wb = _balance_weights(c)
            abssum = float(np.sum(np.abs(wb)))
            if abssum > 1e-15:
                gb = 2.0 * _adjoint_weights(a_work, wb)
                gbmin = float(np.min(gb))
                gbmax = float(np.max(gb))
                if np.isfinite(gbmax) and (gbmax - gbmin) > 1e-14:
                    gbn = (gb - gbmin) / (gbmax - gbmin)
                    for eta in (0.1, 0.25, 0.5):
                        if time.monotonic() >= deadline:
                            break
                        cand = _log_update(a_work, gbn, eta)
                        Mc = float(np.max(_conv_square(cand)))
                        if Mc < best_M - 1e-15:
                            best_M = Mc
                            best_a = cand.copy()
                            a_work = cand.copy()
                            current_M = Mc
                            stall = 0
                            best_updated = True
                            moved = True
                            break
                        elif (
                            Mc < old_current + accept_tol
                            and Mc < best_M * 1.005
                            and (Mc < old_current or stall > 10)
                        ):
                            a_work = cand
                            current_M = Mc
                            moved = True
                            break

        # 3) Targeted pair transfers from high-gradient to low-gradient coordinates.
        if not best_updated and not moved and n > 1:
            old_current = current_M
            kh = min(3, n)
            high = np.argpartition(g, n - kh)[-kh:]
            kl = min(3, n)
            if kl == 1:
                low = np.array([int(np.argmin(g))], dtype=np.int64)
            else:
                low = np.argpartition(g, kl - 1)[:kl]

            for si in range(len(high)):
                if best_updated or moved:
                    break
                s = int(high[si])
                if s < 0 or s >= n or a_work[s] <= 1e-12:
                    continue
                for di in range(len(low)):
                    if best_updated or moved:
                        break
                    d = int(low[di])
                    if d < 0 or d >= n or d == s:
                        continue
                    for frac in (0.3, 0.1, 0.6):
                        if time.monotonic() >= deadline:
                            break
                        cand = a_work.copy()
                        amt = a_work[s] * frac
                        cand[s] -= amt
                        cand[d] += amt
                        cand = np.maximum(cand, 0.0)
                        ss = float(np.sum(cand))
                        if ss <= 1e-15:
                            continue
                        cand = cand / ss
                        Mc = float(np.max(_conv_square(cand)))
                        if Mc < best_M - 1e-15:
                            best_M = Mc
                            best_a = cand.copy()
                            a_work = cand.copy()
                            current_M = Mc
                            stall = 0
                            best_updated = True
                            moved = True
                            break
                        elif (
                            Mc < old_current + accept_tol
                            and Mc < best_M * 1.005
                            and (Mc < old_current or stall > 10)
                        ):
                            a_work = cand
                            current_M = Mc
                            moved = True
                            break

        # 4) Smooth random multiplicative perturbations.
        if not best_updated and not moved and (stall > 5 or iter_count % 3 == 0):
            old_current = current_M
            max_sigma = 0.05 + 0.25 * progress + 0.02 * min(stall, 20)
            if max_sigma > 0.8:
                max_sigma = 0.8
            sigmas = (0.05, max_sigma, 0.15)
            for att in range(2):
                if best_updated or moved:
                    break
                phi = _random_direction(n, rng)
                for sigma in sigmas:
                    if time.monotonic() >= deadline:
                        break
                    if sigma <= 0.0:
                        continue
                    lb = np.log(np.maximum(a_work, 1e-12)) + sigma * phi
                    lb = lb - float(np.max(lb))
                    u = np.exp(np.clip(lb, -700.0, 0.0))
                    u = u + (0.005 / float(n)) * rng.random(n)
                    cand = _norm_simplex(u)
                    Mc = float(np.max(_conv_square(cand)))
                    if Mc < best_M - 1e-15:
                        best_M = Mc
                        best_a = cand.copy()
                        a_work = cand.copy()
                        current_M = Mc
                        stall = 0
                        best_updated = True
                        moved = True
                        break
                    elif (
                        Mc < old_current + accept_tol
                        and Mc < best_M * 1.005
                        and (Mc < old_current or stall > 10)
                    ):
                        a_work = cand
                        current_M = Mc
                        moved = True
                        break

        # 5) Spectral magnitude compression and symmetrization.
        if not best_updated and not moved and (stall > 15 or iter_count % 5 == 0):
            old_current = current_M
            for pwr, mix in ((0.25, 0.4), (0.5, 0.6)):
                if best_updated or moved:
                    break
                if time.monotonic() >= deadline:
                    break
                cand = _spectral_smooth(a_work, pwr, mix)
                Mc = float(np.max(_conv_square(cand)))
                if Mc < best_M - 1e-15:
                    best_M = Mc
                    best_a = cand.copy()
                    a_work = cand.copy()
                    current_M = Mc
                    stall = 0
                    best_updated = True
                    moved = True
                    break
                elif (
                    Mc < old_current + accept_tol
                    and Mc < best_M * 1.005
                    and (Mc < old_current or stall > 10)
                ):
                    a_work = cand
                    current_M = Mc
                    moved = True
                    break
            if not best_updated and not moved:
                cand = _norm_simplex(0.5 * (a_work + a_work[::-1]))
                Mc = float(np.max(_conv_square(cand)))
                if Mc < best_M - 1e-15:
                    best_M = Mc
                    best_a = cand.copy()
                    a_work = cand.copy()
                    current_M = Mc
                    stall = 0
                    best_updated = True
                    moved = True
                elif (
                    Mc < old_current + accept_tol
                    and Mc < best_M * 1.005
                    and (Mc < old_current or stall > 10)
                ):
                    a_work = cand
                    current_M = Mc
                    moved = True

        # 6) Rare larger geometric restart when stuck.
        if not best_updated and not moved and stall > 60 and iter_count % 20 == 0:
            seed_j = int(rng.integers(0, 2147483646))
            cand = _norm_simplex(0.5 * best_a + 0.5 * _random_smooth_init(n, seed_j))
            Mc = float(np.max(_conv_square(cand)))
            if Mc < best_M - 1e-15:
                best_M = Mc
                best_a = cand.copy()
                a_work = cand.copy()
                current_M = Mc
                stall = 0
                best_updated = True
                moved = True

        if best_updated:
            stall = 0

        if verbose and iter_count % 1000 == 0:
            print(
                "      local iter={} t={:.0f}s best={:.7f} stall={} beta={:.1f}".format(
                    iter_count, time.monotonic() - t0, 2.0 * n * best_M, stall, beta
                )
            )

    return best_a, best_M, float(2.0 * n * best_M)


def propose_candidate():
    t_start = time.monotonic()
    hard_deadline = t_start + 1760.0
    max_search_n = 12000

    parent = None
    parent_obj = globals().get("height_sequence_1", None)
    if parent_obj is not None:
        try:
            p = np.asarray(parent_obj, dtype=np.float64)
            if p.ndim == 1 and p.size > 0:
                p = np.where(np.isfinite(p), p, 0.0)
                p = np.clip(p, 0.0, 1000.0)
                if float(np.sum(p)) > 1e-10:
                    parent = p
        except Exception:
            parent = None

    if parent is None:
        parent = _random_smooth_init(4200, 0)

    if parent.size > max_search_n:
        parent = _interpolate_sequence(parent, max_search_n)

    best_seq = _norm_simplex(parent)
    best_score = _true_score(best_seq)
    if not np.isfinite(best_score):
        best_score = _fast_score(best_seq)
    n0 = best_seq.size
    print("Initial: n={} score={:.7f}".format(n0, best_score))

    # Phase 1: focused local search from the parent.
    rem = hard_deadline - time.monotonic()
    if rem > 100:
        phase_time = min(600.0, 0.35 * rem)
        dl = time.monotonic() + phase_time
        if dl > hard_deadline - 300:
            dl = hard_deadline - 300
        if dl - time.monotonic() > 30:
            a_out, m_out, s_out = _local_search(
                best_seq, dl, seed=7, verbose=True, beta_start=3.0, beta_end=120.0
            )
            if np.isfinite(s_out) and s_out < best_score:
                best_seq = a_out
                best_score = s_out
                print("Phase 1 improved: score={:.7f} n={}".format(best_score, best_seq.size))
            else:
                print("Phase 1 finished: best={:.7f}".format(best_score))

    # Phase 2: diverse starts at the same length.
    n_best = best_seq.size
    sym_start = _norm_simplex(0.5 * (best_seq + best_seq[::-1]))
    rand_mix_start = _random_smooth_init(n_best, 101, base=best_seq)
    spec_start = _spectral_smooth(best_seq, 0.6, 0.7)
    pure_rand_start = _random_smooth_init(n_best, 202)

    starts = [sym_start, rand_mix_start, spec_start, pure_rand_start]
    for idx, st in enumerate(starts):
        rem = hard_deadline - time.monotonic()
        if rem < 160:
            break
        phase_time = min(130.0, 0.20 * rem)
        dl = time.monotonic() + phase_time
        if dl > hard_deadline - 250:
            dl = hard_deadline - 250
        if dl - time.monotonic() < 30:
            break
        a_out, m_out, s_out = _local_search(
            st,
            dl,
            seed=1000 + idx * 17,
            verbose=False,
            beta_start=2.0,
            beta_end=150.0,
        )
        if np.isfinite(s_out) and s_out < best_score:
            best_seq = a_out
            best_score = s_out
            print(
                "Phase 2 start={} improved: score={:.7f} n={}".format(
                    idx, best_score, best_seq.size
                )
            )

    # Phase 3: modest length variations.
    base_n = best_seq.size
    lengths = set()
    for factor in (1.25, 1.5, 2.0):
        nn = int(base_n * factor)
        if 2000 <= nn <= max_search_n and nn != base_n:
            lengths.add(nn)
    if base_n > 5000:
        for factor in (0.8, 0.6):
            nn = int(base_n * factor)
            if 2000 <= nn <= max_search_n and nn != base_n:
                lengths.add(nn)
    if max_search_n > base_n:
        lengths.add(max_search_n)

    for new_n in sorted(lengths):
        rem = hard_deadline - time.monotonic()
        if rem < 250:
            break
        phase_time = min(180.0, 0.25 * rem)
        dl = time.monotonic() + phase_time
        if dl > hard_deadline - 180:
            dl = hard_deadline - 180
        if dl - time.monotonic() < 50:
            break

        a_interp = _interpolate_sequence(best_seq, new_n)
        s_interp = _fast_score(a_interp)
        if np.isfinite(s_interp) and s_interp < best_score:
            best_seq = a_interp
            best_score = s_interp
            print("Length interp n={} improved: score={:.7f}".format(new_n, best_score))

        a_out, m_out, s_out = _local_search(
            a_interp,
            dl,
            seed=5000 + (new_n % 1000),
            verbose=False,
            beta_start=3.0,
            beta_end=150.0,
        )
        if np.isfinite(s_out) and s_out < best_score:
            best_seq = a_out
            best_score = s_out
            print("Length n={} improved: score={:.7f}".format(new_n, best_score))

    # Phase 4: final polish on the global best.
    rem = hard_deadline - time.monotonic()
    if rem > 60:
        dl = hard_deadline - 25.0
        a_out, m_out, s_out = _local_search(
            best_seq,
            dl,
            seed=99991,
            verbose=True,
            beta_start=2.0,
            beta_end=200.0,
        )
        if np.isfinite(s_out) and s_out < best_score:
            best_seq = a_out
            best_score = s_out
        print("Final polish: best={:.7f} n={}".format(best_score, best_seq.size))

    # Safety cleanup.
    best_seq = _norm_simplex(best_seq)
    if best_seq.size == 0 or not np.isfinite(_fast_score(best_seq)):
        best_seq = np.ones(4200, dtype=np.float64) / 4200.0
        best_score = _fast_score(best_seq)

    # Optional final validation with the actual evaluator if time permits.
    if hard_deadline - time.monotonic() > 80 and best_seq.size <= 12000:
        try:
            fs = float(evaluate_sequence([float(x) for x in best_seq]))
            if np.isfinite(fs) and fs < best_score:
                best_score = fs
                print("Final evaluator score: {:.7f}".format(best_score))
        except Exception:
            pass

    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.maximum(best_seq, 0.0)
    s = float(np.sum(best_seq))
    if s < 0.01:
        best_seq = np.ones(1000, dtype=np.float64)
        s = 1.0
    best_seq = best_seq / s
    best_seq = np.minimum(best_seq, 1000.0)

    return [float(x) for x in best_seq]
```
