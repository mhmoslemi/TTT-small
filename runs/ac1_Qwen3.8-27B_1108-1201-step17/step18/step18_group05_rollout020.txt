"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate as sp_correlate

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_water_fill(a, time_limit, p_init=4.0, p_final=0.1, n_iters=None, rng_seed=0):
    """Iteratively flatten the power spectrum using water-filling."""
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if n_iters is None:
        n_iters = 5000
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, 2 * n)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        
        if P_mean < 1e-30:
            break
        
        # Progressive: start with aggressive flattening, end with gentle
        progress = it / n_iters
        p = p_init * (p_final / p_init) ** progress
        
        # Water-filling: A_new = A * (P_mean/P)^(1/p)
        P_safe = np.maximum(P, P_mean * 1e-6)
        ratio = (P_mean / P_safe) ** (1.0 / p)
        A_new = A * ratio
        
        a_new = np.fft.irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_m

def nesterov_soft_subgrad(a0, time_limit, top_k=80, beta=8.0, verbose=False, rng_seed=42):
    """Nesterov-accelerated soft-subgradient with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 5.0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_every = 3000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 0.5) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute soft subgradient
        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        # Adaptive step size
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Exponential update (multiplicative weights)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            gamma = min(gamma * 1.005, 15.0)
        else:
            stall += 1
            
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            gamma = 3.0
            cur_beta = max(cur_beta - 2.0, 3.0)
            cur_top_k = min(cur_top_k + 20, 200)
        
        # Adaptive: if improving, be more aggressive
        if stall < 100 and t % 100 == 0:
            cur_top_k = min(cur_top_k + 5, 200)
            cur_beta = max(cur_beta * 0.95, 3.0)
    
    return best_a, best_m

def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K_init=200, use_dual_pressure=True):
    """LP-Frank-Wolfe with dual-pressure adaptive K selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K = top_K_init
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        K_sel = min(K, 2 * n - 1)
        
        if use_dual_pressure:
            # Select constraints by "pressure" = ac value / local contribution
            # This gives more weight to constraints that are hard to satisfy
            thresh = M_curr * 0.5
            candidates = np.where(ac >= thresh)[0]
            if len(candidates) > K_sel:
                # Among candidates, prioritize those with high ac
                binding_idx = candidates[np.argsort(ac[candidates])[::-1][:K_sel]]
            else:
                binding_idx = np.sort(candidates)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        K_sel = len(binding_idx)
        
        if K_sel == 0:
            binding_idx = np.argsort(ac)[::-1][:top_K_init]
            K_sel = len(binding_idx)
        
        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, 3000)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Backtracking
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()
        
        iter_count += 1
        if iter_count % 10 == 0:
            K = min(K + 20, 3000)
        
        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPv2 iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")
    
    return best_a, best_m

def pair_swap_cd(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with pair-swap moves (move mass from i to j while adjusting k to l)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    n_top = 30
    
    while time.monotonic() - t0 < time_limit and stall < 50:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find low-contribution destination
            # A position j contributes to peak k if j is in [k-n+1, k]
            # We want j NOT in any top peak's range
            in_peak = np.zeros(n, dtype=bool)
            for kk in top_k[:8]:
                kk_i = int(kk)
                in_peak[max(0, kk_i-n+1):min(n, kk_i+1)] = True
            
            low_contrib = np.where(~in_peak)[0]
            if len(low_contrib) == 0:
                low_contrib = np.arange(n)
            
            # Sort by a[j] (prefer moving to positions with less mass)
            top_dst = low_contrib[np.argsort(a[low_contrib])[:5]]
            
            for dst in top_dst:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved and stall > 20:
                # Random exploration
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

def smooth_gradient_v2(a0, time_limit, verbose=True, rng_seed=100):
    """Improved smoothed gradient with multi-temperature and momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    momentum = np.zeros(n)
    beta_mom = 0.9
    eta = 0.5
    stall = 0
    
    total_iters = 0
    while time.monotonic() - t0 < time_limit and stall < 300:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 600.0))
        tau = 10.0 * (200.0 / 10.0) ** progress
        
        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()
        
        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()
        
        # Momentum
        momentum = beta_mom * momentum + g_tangent
        a_trial = a - eta * momentum
        a_new = project_simplex(a_trial)
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            stall += 1
            if stall > 50:
                eta *= 0.8
                beta_mom = max(beta_mom * 0.95, 0.5)
            if stall > 150:
                pert = rng.normal(0, 0.005, n) * a
                a_pert = np.maximum(a + pert, 0.0)
                a = a_pert / a_pert.sum()
                stall = 80
                momentum = np.zeros(n)
            else:
                a = a_new
        
        total_iters += 1
        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad2 iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")
    
    return best_a, best_m

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_water_fill(a0, time_limit, p_init=8.0, p_final=0.3, n_iters=5000, verbose=False):

def nesterov_amd(a0, time_limit, top_k=80, beta=8.0, gamma_init=5.0,
                 restart_every=3000, verbose=False, rng_seed=42):
    """
    Nesterov-accelerated multiplicative-weights subgradient method.
    Uses soft subgradient with adaptive top_k and beta.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = gamma_init
    stall = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 0.5) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute soft subgradient at y
        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        # Multiplicative weights update
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            gamma = min(gamma * 1.003, 20.0)
        else:
            stall += 1

        # Adaptive parameters
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            gamma = gamma_init * 0.7
            cur_beta = max(cur_beta * 0.8, 3.0)
            cur_top_k = min(cur_top_k + 15, 250)

        if stall < 100 and t % 200 == 0:
            cur_top_k = min(cur_top_k + 3, 250)
            cur_beta = max(cur_beta * 0.98, 3.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NAMD t={t} ({elapsed:.0f}s): J={J:.6f}, k={cur_top_k}, beta={cur_beta:.1f}, g={gamma:.1f}")

    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=4000):
    """LP-Frank-Wolfe with dual-pressure based constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)

        # Dual-pressure selection: prefer constraints near the max
        thresh = M_curr * 0.6
        candidates = np.where(ac >= thresh)[0]
        if len(candidates) > K_sel:
            # Among candidates, sort by ac value (highest first)
            binding_idx = candidates[np.argsort(ac[candidates])[::-1][:K_sel]]
        elif len(candidates) > 0:
            binding_idx = np.sort(candidates)
            K_sel = len(binding_idx)
        else:
            binding_idx = np.argsort(ac)[::-1][:top_K_init]
            K_sel = top_K_init

        if K_sel == 0:
            binding_idx = np.argsort(ac)[::-1][:top_K_init]
            K_sel = top_K_init

        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 50, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                  0.5, 0.3, 0.2, 0.1, 0.05]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 8 == 0:
            K = min(K + 25, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPv2 it={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, a={best_alpha:.4f}")

    return best_a, best_m


def smooth_gradient_momentum(a0, time_limit, verbose=True, rng_seed=100):
    """Smoothed gradient with heavy-ball momentum and adaptive step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    momentum = np.zeros(n)
    beta_mom = 0.90
    eta = 0.6
    stall = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < 400:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 500.0))
        tau = 10.0 * (250.0 / 10.0) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        momentum = beta_mom * momentum + g_tangent
        a_trial = a - eta * momentum
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.01, 2.5)
        else:
            stall += 1
            if stall > 60:
                eta *= 0.8
                beta_mom = max(beta_mom * 0.97, 0.5)
            if stall > 200:
                pert = rng.normal(0, 0.004, n) * a
                a_pert = np.maximum(a + pert, 0.0)
                a = a_pert / a_pert.sum()
                stall = 100
                momentum = np.zeros(n)
                eta = 0.4
            else:
                a = a_new

        total_iters += 1
        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGM it={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.3f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123, n_top=25):
    """Coordinate descent with peak-avoidance mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 80:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        # Compute in-peak mask
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            in_peak[max(0, ki - n + 1):min(n, ki + 1)] = True

        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Destination: low in_peak probability, low current mass
            low_mask = ~in_peak
            candidates = np.where(low_mask)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            top_dst = candidates[np.argsort(a[candidates])[:5]]

            found = False
            for dst in top_dst:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

            if not found and stall > 30:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.015)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main orchestrator: combines spectral, subgradient, LP, and coordinate methods
    in a multi-phase pipeline with adaptive time allocation.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Spectral water-filling ===
    print(f"\n=== Phase 1: Spectral water-filling (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_water_fill(a_in, t1, p_init=10.0, p_final=0.3,
                                       n_iters=4000, verbose=True)
        try_update(a_out, "spec_wf")

    # === Phase 2: Main Nesterov AMD (long run) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Nesterov AMD main (n={n_cur}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_amd(a_in, t2, top_k=80, beta=8.0, gamma_init=5.0,
                                restart_every=3000, verbose=True, rng_seed=42)
        try_update(a_out, "namd_main")

    # === Phase 3: Smoothed gradient with momentum ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: SmoothGrad momentum (n={n_cur}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_momentum(a_in, t3, verbose=True, rng_seed=100)
        try_update(a_out, "sgm")

    # === Phase 4: LP-FW v2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-FW v2 (n={n_cur}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30 and n_cur <= 5500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, verbose=True,
                                      top_K_init=200, top_K_max=4000)
        try_update(a_out, "lpv2")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Coord descent (n={n_cur}) ===")
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, rng_seed=123, n_top=25)
        try_update(a_out, "coord1")

    # === Phase 6: Multi-init exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-init exploration ===")
    if remaining > 120:
        inits = [
            (make_u_shape(n0, sharpness=1.5), "u15"),
            (make_u_shape(n0, sharpness=3.0), "u30"),
            (make_lorentzian(n0, gamma=0.25), "lorentz"),
            (make_multi_peak(n0, n_peaks=9, width=0.05), "multi9"),
            (make_cosine_mod(n0, freq=1.0, power=3.0), "cos3"),
        ]
        t_per = max(20.0, remaining * 0.45 / len(inits))
        for idx, (init, lbl) in enumerate(inits):
            rem = time_budget - (time.monotonic() - t_global)
            if rem < 50:
                break
            t_avail = min(t_per, rem - 40)
            if t_avail < 15:
                break
            print(f"  Init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_amd(init, t_avail, top_k=60, beta=6.0,
                                     gamma_init=4.0, restart_every=2000,
                                     verbose=False, rng_seed=1000 + idx * 7)
            try_update(a_out, f"init_{lbl}")

    # === Phase 7: Second Nesterov AMD pass on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: NAMD pass 2 (n={n_cur}) ===")
    if remaining > 50:
        t7 = min(remaining * 0.25, 250.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_amd(a_in, t7, top_k=100, beta=10.0, gamma_init=6.0,
                                restart_every=4000, verbose=True, rng_seed=777)
        try_update(a_out, "namd_p2")

    # === Phase 8: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_cur}) ===")
    if remaining > 30 and n_cur <= 5500:
        t8 = min(remaining * 0.30, 150.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t8, solver, verbose=True,
                                      top_K_init=400, top_K_max=5000)
        try_update(a_out, "lp_final")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 9: Final polish (n={len(best_seq)}) ===")
        t9 = min(remaining * 0.6, 60.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t9, verbose=False, rng_seed=999, n_top=30)
        try_update(a_out, "polish")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
