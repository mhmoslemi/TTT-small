def build_constraint_matrix(a, binding_idx):
    """Build the K x n matrix where row k has a[k-j] in position j."""
    n = len(a)
    K = len(binding_idx)
    T = np.zeros((K, n))
    for ii, k in enumerate(binding_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        T[ii, j_start:j_end] = a[ki - (np.arange(j_start, j_end))]
    return T

2. LP-based Frank-Wolfe with exact 1D line search and adaptive K
3. Aggressive peak-shaving coordinate descent
4. Multi-phase time allocation with length exploration
5. Subspace perturbation to escape local minima
"""

import numpy as np
import time
import cvxpy as cp

_lp_solver = None

def _get_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _lp_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _lp_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, b):
    """Compute conv(a, b) using FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u * np.arange(1, n + 1) - (cssv - 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_sine_envelope(n, k=1):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** (2 * k)
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def build_constraint_matrix(a, binding_idx):
    n = len(a)
    K = len(binding_idx)
    T = np.zeros((K, n))
    for ii in range(K):
        ki = int(binding_idx[ii])
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            cols = np.arange(j_start, j_end)
            T[ii, j_start:j_end] = a[ki - cols]
    return T


def lp_frank_wolfe(a0, time_limit, solver, K_init=300, K_max=None, K_growth=100,
                   verbose=True, exact_line_search=True):
    """LP-based Frank-Wolfe with adaptive K and optional exact line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if K_max is None:
        K_max = min(2 * n - 1, 8000)
    K = K_init
    iter_count = 0
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_mat = build_constraint_matrix(a, binding_idx)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_mat @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + K_growth, K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + K_growth, K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            K = min(K + K_growth, K_max)
            continue
        g_dir = b_val / b_sum

        # Line search
        if exact_line_search:
            # Precompute cross-correlation terms
            ab_cross = fast_cross_corr(a, g_dir)
            bb_conv = fast_autocorr(g_dir)
            aa_conv = ac  # already computed

            def objective(alpha):
                trial = (1.0 - alpha) * a + alpha * g_dir
                # c_k(alpha) = (1-a)^2 * aa_k + 2*a*(1-a) * ab_k + a^2 * bb_k
                vals = (1.0 - alpha) ** 2 * aa_conv + 2.0 * alpha * (1.0 - alpha) * ab_cross + alpha ** 2 * bb_conv
                return vals.max()

            # Golden section search on [0, 1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, 1.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = objective(c)
            fd = objective(d)
            for _ in range(40):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = objective(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = objective(d)
            best_alpha = (lo + hi) / 2.0
        else:
            best_alpha = 2.0 / (iter_count + 2.0)
            best_trial_m = M_curr
            for alpha in [best_alpha, min(1.0, best_alpha * 1.5), 0.5, 0.3, 0.2, 0.7, 0.8]:
                a_trial = (1.0 - alpha) * a + alpha * g_dir
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        iter_count += 1
        if iter_count % 3 == 0:
            K = min(K + K_growth, K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}, stall={stall}")

    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=80, beta=6.0, c0=5.0, restart_every=5000,
                 verbose=False, rng_seed=42):
    """AMD with Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    v = np.zeros(n)  # momentum
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    mu = 0.8  # Nesterov momentum

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov step
        y = x + mu * v
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * y[k_int - idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        # Update momentum
        v = a_new - x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 2000:
            cur_beta = max(cur_beta * 0.95, 2.0)
            cur_top_k = min(cur_top_k + 10, 200)

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            v = np.zeros(n)
            cur_top_k = min(cur_top_k + 20, 300)
            cur_beta = max(cur_beta - 1.0, 2.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=False, rng_seed=99):
    """Aggressive peak-shaving: move mass away from peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 80:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Get top peaks
        n_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        for k in top_peaks[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s_start = max(0, ki - n + 1)
            s_end = min(n, ki + 1)
            idx = np.arange(s_start, s_end)
            contrib = a[idx] * a[ki - idx]
            total = contrib.sum()
            if total < 1e-20:
                continue

            # Find the position with maximum contribution
            src_pos_in_idx = np.argmax(contrib)
            src = idx[src_pos_in_idx]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-contribution positions to move mass to
            # Low contribution means a[j] * a[ki-j] is small
            contrib_sorted = np.argsort(contrib)
            # Try moving to the lowest-contribution position
            for dest_offset in contrib_sorted[:5]:
                dst = idx[dest_offset]
                if dst == src:
                    continue
                for frac in [0.02, 0.01, 0.005, 0.03, 0.05, 0.01, 0.002]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 20:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a = best_a.copy()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def subspace_perturb(a0, time_limit, n_basis=30, rng_seed=77):
    """Optimize in a low-dimensional perturbation subspace."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Build basis: mix of Fourier modes and random smooth perturbations
    x = np.linspace(0, 1, n)
    basis_funcs = []
    n_fourier = min(n_basis // 2, 40)
    for freq in range(1, n_fourier + 1):
        basis_funcs.append(np.sin(2 * np.pi * freq * x))
    while len(basis_funcs) < n_basis:
        # Smooth random perturbation (low-pass filtered)
        r = rng.normal(0, 1, n)
        # Simple low-pass: FFT, zero high freq
        R = np.fft.rfft(r)
        cutoff = max(1, n // 4)
        R[cutoff:] = 0
        r_smooth = np.fft.irfft(R, n)
        basis_funcs.append(r_smooth)

    basis = np.array(basis_funcs[:n_basis])
    # Orthonormalize
    try:
        Q, R, _ = np.linalg.svd(basis, full_matrices=False)
        basis = Q[:, :n_basis]
    except Exception:
        basis = np.array([f / max(np.linalg.norm(f), 1e-10) for f in basis_funcs[:n_basis]])

    w = np.zeros(n_basis)
    eta = 0.02
    stall = 0

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 200:
            # Restart with new basis
            w = rng.normal(0, 0.01, n_basis)
            eta = 0.02
            stall = 0
            # Regenerate part of basis
            for i in range(n_basis // 2):
                r = rng.normal(0, 1, n)
                R = np.fft.rfft(r)
                cutoff = max(1, n // 4)
                R[cutoff:] = 0
                basis[i] = np.fft.irfft(R, n)

        a_trial = a + basis @ w
        a_trial = np.maximum(a_trial, 0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            w = np.zeros(n_basis)
            continue
        a_trial = a_trial / s_t
        m = fast_autocorr(a_trial).max()

        if m < best_m - 1e-15:
            best_m = m
            best_a = a_trial.copy()
            eta = min(eta * 1.01, 0.1)
            stall = 0
        else:
            stall += 1
            if stall > 50:
                eta *= 0.9

        # Subgradient of max autocorrelation
        ac = fast_autocorr(a_trial)
        k_star = int(np.argmax(ac))
        grad_a = np.zeros(n)
        for j in range(n):
            g = 0.0
            if j + k_star < n:
                g += a_trial[j + k_star]
            if j - k_star >= 0:
                g += a_trial[j - k_star]
            grad_a[j] = g

        grad_w = basis.T @ grad_a
        gw_norm = np.linalg.norm(grad_w)
        if gw_norm > 1e-15:
            grad_w = grad_w / gw_norm
        w -= eta * grad_w
        w = np.clip(w, -0.05, 0.05)

    return best_a, best_m


def propose_candidate():
    """Main search orchestrator with multi-phase time allocation."""
    t_global = time.monotonic()
    time_budget = 1770.0  # Leave 30s buffer

    solver = _get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            print(f"Parent loaded: n={len(parent)}, sum={p.sum():.4f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
    else:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = make_sine_envelope(n_base, k=1)
        print(f"No parent, using sine init: n={len(best_seq)}")

    best_score = evaluate_safe(best_seq)
    print(f"Initial score: {best_score:.6f}, n={len(best_seq)}")
    n0 = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick AMD polish (60s) ===
    print(f"\n=== Phase 1: AMD polish (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 60.0)
    if t1 > 10:
        a_out, _ = amd_optimize(best_seq, t1, top_k=80, beta=6.0, c0=5.0,
                                restart_every=5000, verbose=False, rng_seed=42)
        try_update(a_out, "amd_p1")

    # === Phase 2: LP Frank-Wolfe (400s) - main workhorse ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        K_init = min(400, 2 * n_best - 1)
        K_max = min(2 * n_best - 1, 8000)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, K_init=K_init, K_max=K_max,
                                   K_growth=150, verbose=True, exact_line_search=True)
        try_update(a_out, "lp_p2")

    # === Phase 3: Peak shaving (120s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak shaving (n={n_best}) ===")
    t3 = min(remaining * 0.07, 120.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t3, verbose=False, rng_seed=99)
        try_update(a_out, "peak_p3")

    # === Phase 4: LP Frank-Wolfe round 2 (350s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW r2 (n={n_best}) ===")
    t4 = min(remaining * 0.22, 350.0)
    if t4 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        K_init = min(600, 2 * n_best - 1)
        K_max = min(2 * n_best - 1, 10000)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, K_init=K_init, K_max=K_max,
                                   K_growth=200, verbose=True, exact_line_search=True)
        try_update(a_out, "lp_p4")

    # === Phase 5: Subspace perturbation (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Subspace perturb (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = subspace_perturb(a_in, t5, n_basis=40, rng_seed=77)
        try_update(a_out, "subspace_p5")

    # === Phase 6: AMD with restarts (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: AMD restarts (n={n_best}) ===")
    t6 = min(remaining * 0.13, 200.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t6, top_k=100, beta=5.0, c0=4.0,
                                restart_every=4000, verbose=False, rng_seed=200)
        try_update(a_out, "amd_p6")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [400, -400, 800, -800, 1200, -1200]:
        new_n = n_best + delta
        if 3000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.3, 150.0)
        if t7a > 20:
            a_out, _ = amd_optimize(a_long, t7a, top_k=80, beta=6.0, c0=5.0,
                                     restart_every=5000, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 5500:
            t7b = min(remaining * 0.4, 150.0)
            if t7b > 30:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                K_init = min(400, 2 * new_n - 1)
                K_max = min(2 * new_n - 1, 6000)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, K_init=K_init, K_max=K_max,
                                           K_growth=200, verbose=False, exact_line_search=True)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final LP on best (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        K_init = min(800, 2 * n_best - 1)
        K_max = min(2 * n_best - 1, 12000)
        a_out, _ = lp_frank_wolfe(a_in, remaining - 30, solver, K_init=K_init, K_max=K_max,
                                   K_growth=300, verbose=True, exact_line_search=True)
        try_update(a_out, "lp_final")

    # === Final: Quick peak shaving on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Final peak shaving (remaining={remaining:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, remaining - 10, verbose=False, rng_seed=555)
        try_update(a_out, "peak_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
