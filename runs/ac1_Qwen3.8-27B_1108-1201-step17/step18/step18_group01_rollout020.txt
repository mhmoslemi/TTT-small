"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed multiplicative gradient descent with temperature annealing
2. Asymmetric parametric initialization (exponential, power-law, shifted sine)
3. Adaptive step-size with backtracking

6. Deflation-based peak suppression

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def objective(a):
    """Compute J = 2*n*max(ac)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:

        a_list = [float(x) for x in a_arr]

    except Exception:

def lse_gradient(ac, a, tau):
    """
    Compute the gradient of the log-sum-exp smoothed objective.
    L(tau) = (1/tau) * log(sum_k exp(tau * ac[k]))
    dL/da[i] = (1/Z) * sum_k exp(tau*ac[k]) * a[k-i]
    where Z = sum_k exp(tau*ac[k])
    Returns gradient vector of length n.
    """
    n = len(a)
    size = len(ac)
    # Numerically stable log-sum-exp
    ac_max = ac.max()
    exp_ac = np.exp(tau * (ac - ac_max))
    Z = exp_ac.sum()
    if Z < 1e-300:
        return np.ones(n) / n
    weights = exp_ac / Z  # length: 2n-1

    # Gradient: grad[i] = sum_k weights[k] * a[k-i] (valid when 0<=k-i<n)
    grad = np.zeros(n)
    for k in range(size):
        if weights[k] < 1e-15:
            continue
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += weights[k] * a[k - idx]
    return grad


def lse_gradient_fast(ac, a, tau, top_m=200):
    """
    Faster LSE gradient: only use top-m lags by weight.
    """
    n = len(a)
    size = len(ac)
    ac_max = ac.max()
    log_w = tau * (ac - ac_max)
    log_w_max = log_w.max()
    exp_w = np.exp(log_w - log_w_max)
    Z = exp_w.sum()
    if Z < 1e-300:
        return np.ones(n) / n
    weights = exp_w / Z

    # Only use top m lags
    idx = np.argsort(weights)[::-1][:top_m]
    w_sum = weights[idx].sum()
    if w_sum < 1e-300:
        return np.ones(n) / n
    weights_use = np.zeros(size)
    weights_use[idx] = weights[idx] / w_sum

    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        w = weights_use[k]
        if w < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx2 = np.arange(i_start, i_end)
            grad[idx2] += w * a[k - idx2]
    return grad


def mgd_lse(a0, time_limit, tau_start=2.0, tau_end=40.0, eta0=0.05, verbose=True, seed=42):
    """

    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = eta0
    patience = 0
    max_patience = 500
    total_iters = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        # Anneal temperature
        tau = tau_start + (tau_end - tau_start) * progress

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            patience = 0

        # Compute LSE gradient

        else:
            top_m = min(200, n)
            grad = lse_gradient_fast(ac, a, tau, top_m)

        # Center the gradient (remove mean for invariance)
        grad = grad - grad.mean()

        gmax = np.abs(grad).max()
        if gmax < 1e-16:

        grad = grad / gmax

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:

        M_new = ac_new.max()

            a = a_new
            eta *= 0.999
        else:
            eta *= 0.7
            if eta < 0.001:
                eta = 0.02
                tau = max(tau_start, tau * 0.7)
                patience = max(0, patience - 200)

    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top 3 peaks
        top_lags = np.argsort(ac)[::-1][:3]
        improved = False

        for k in top_lags:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            total = contrib.sum()
            if total < 1e-20:
                continue

            # Find the pair with maximum contribution
            max_i = idx[np.argmax(contrib)]
            max_j = ki - max_i
            if max_i == max_j:
                # Middle element: reduce it and spread to neighbors
                val = a[max_i]
                if val < 1e-12:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = val * frac
                    a_try = a.copy()
                    a_try[max_i] -= amt
                    # Spread to non-peak positions
                    peak_mask = np.zeros(n, dtype=bool)
                    for k2 in top_lags:
                        k2i = int(k2)
                        s2 = max(0, k2i - n + 1)
                        e2 = min(n - 1, k2i)
                        peak_mask[s2:e2+1] = True
                    candidates = np.where(~peak_mask)[0]
                    if len(candidates) == 0:
                        candidates = np.arange(n)
                    # Spread uniformly to candidates
                    a_try[candidates] += amt / len(candidates)
                    a_try = np.maximum(a_try, 0.0)
                    s_t = a_try.sum()
                    if s_t < 1e-15:
                        continue
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < M - 1e-14:
                        a = a_try
                        M = M_try
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        improved = True
                        break
            else:
                # Two distinct elements: reduce one, increase the other's "complement"
                vi = a[max_i]
                vj = a[max_j]
                # Move mass from max_i to a position that doesn't contribute to any peak
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_lags:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    peak_mask[s2:e2+1] = True
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)
                # Pick candidate farthest from peak-contributing positions
                dst = candidates[np.argmin(peak_mask[candidates].astype(int))]

                for frac in [0.01, 0.03, 0.06, 0.1]:
                    amt = vi * frac
                    a_try = a.copy()
                    a_try[max_i] -= amt
                    a_try[dst] += amt
                    a_try = np.maximum(a_try, 0.0)
                    s_t = a_try.sum()
                    if s_t < 1e-15:
                        continue
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < M - 1e-14:
                        a = a_try
                        M = M_try
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        improved = True
                        break

        if not improved:
            # Random escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                        break

    if verbose:
        J = 2.0 * n * best_M
        print(f"      Deflation done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a


def coord_refine(a0, time_limit, verbose=True, seed=55):
    """
    Coordinate refinement: for positions contributing to peaks,
    try small multiplicative adjustments.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        top_lags = np.argsort(ac)[::-1][:8]
        stall += 1
        if stall > max_stall:
            break

        improved = False
        # Identify all positions contributing to peaks
        contrib_positions = set()
        for k in top_lags:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            for i in range(i_start, i_end + 1):
                if a[i] > 1e-10:
                    contrib_positions.add(i)

        # Try adjustments at contributing positions
        positions = list(contrib_positions)
        np.random.shuffle = None  # avoid lambda
        rng.shuffle(positions)

        for pos in positions[:30]:
            if time.monotonic() - t0 > time_limit:
                break
            for scale in [0.7, 0.85, 1.15, 1.3]:
                a_try = a.copy()
                old_val = a_try[pos]
                a_try[pos] = old_val * scale
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                M_try = fast_autocorr(a_try).max()
                if M_try < best_M - 1e-15:
                    a = a_try
                    best_M = M_try
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and stall > 20:
            # Random perturbation escape
            idx = rng.integers(n, size=10)
            deltas = rng.normal(0, 0.01, 10) * a[idx]
            a[idx] += deltas
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        J = 2.0 * n * best_M
        print(f"      Coord refine done ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a


def parametric_search(n, best_a, best_score, time_limit=30.0):
    """Try various parametric families for good initialization."""
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    results = []

    candidates = []

    # Exponential decay from one end
    for alpha in [0.5, 1, 1.5, 2, 3, 5, 8, 12, 20]:
        candidates.append(np.exp(-alpha * x))
        candidates.append(np.exp(-alpha * (1 - x)))

    # Power law
    for p in [0.2, 0.5, 1, 2, 3, 5]:
        for q in [0.2, 0.5, 1, 2, 3, 5]:
            candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** q)

    # Shifted sine
    for delta in [-0.1, -0.05, 0, 0.05, 0.1, 0.15]:
        shifted = (i_arr + n * delta) / n
        candidates.append(np.maximum(np.sin(np.pi * shifted) ** 2, 1e-6))

    # Product of different freqs
    for f1 in [1, 2, 3]:
        for f2 in [1, 2]:
            candidates.append(np.maximum(np.sin(np.pi * f1 * x) ** 2 * np.cos(np.pi * f2 * x / 2) ** 2, 1e-6))

    # Gaussian bump (off-center)
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
            candidates.append(np.exp(-((x - center) ** 2) / (2 * sigma ** 2)))

    # Triangle (off-center)
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        width = 0.4
        tri = np.maximum(0, 1 - np.abs(x - center) / width)
        candidates.append(tri)

    best_local = best_a.copy()
    best_local_score = best_score

    for c in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        c = norm_sum1(c)
        ac = fast_autocorr(c)
        score = 2.0 * n * ac.max()
        if score < best_local_score:
            best_local_score = score
            best_local = c.copy()

    if np.isfinite(best_local_score):
        elapsed = time.monotonic() - t0
        print(f"    Parametric: best score={best_local_score:.6f} ({elapsed:.1f}s)")

    return best_local, best_local_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    try:

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = norm_sum1(p)
            p_score = evaluate_safe(p_norm)
            if np.isfinite(p_score):
                best_seq = p_norm
                best_score = p_score
                print(f"Parent: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000

        # Start with slightly asymmetric exponential
        best_seq = np.exp(-1.5 * x)
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    t1 = min(remaining * 0.03, 40.0)
    if t1 > 5:
