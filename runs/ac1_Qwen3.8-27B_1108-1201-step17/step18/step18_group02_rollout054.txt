def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_obj(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_adam_v2(a0, time_limit, tau_decay=0.9998, lr=0.5, verbose=True):
    """Smoothed multiplicative Adam with LSE objective."""
    n = len(a0)
    a = normalize(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_val = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    tau = max(ac.max() * 0.3, 1e-8)
    tau_min = max(ac.max() * 0.01, 1e-12)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_val - 1e-15:
            best_val = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        
        # LSE gradient
        ac_scaled = (ac - ac.max()) / tau
        p = np.exp(ac_scaled)
        p = p / p.sum()
        
        # grad[i] = 2 * sum_j p[i+j] * a[j] (with appropriate indexing)
        a_flip = a[::-1]
        size = len(ac)
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a_flip, fft_size)
        Pf = np.fft.rfft(p, fft_size)
        conv_full = np.fft.irfft(Af * Pf, fft_size)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0.0)
        
        # Adam update
        m_hat = 0.9 * m_hat + 0.1 * grad
        v_hat = 0.999 * v_hat + 0.001 * (grad * grad)
        bc1 = 1.0 - 0.9 ** t
        bc2 = 1.0 - 0.999 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        log_a = np.log(np.maximum(a, 1e-30)) - step * grad
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        if stall > 2000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            a = x / x.sum()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_val
            print(f"    Adam2 t={t} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2e}")
    
    return best_a, best_val

def multi_resolution_refine(a_coarse, n_fine, time_limit, verbose=True):
    """Refine a coarse solution to a finer resolution."""
    # Upsample the coarse solution
    n_coarse = len(a_coarse)
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n_fine)
    a_fine = np.interp(x_fine, x_coarse, a_coarse)

def spectral_init(n, mode=1):
    """Generate initialization based on spectral considerations."""
    # For non-negative sequences with low autocorrelation peaks,
    # we want the power spectrum to be as flat as possible (away from DC)
    # A good starting point is a windowed sinusoid
    
    x = np.arange(n, dtype=np.float64) / n
    # Raised cosine window times a sinusoid
    w = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = w * (1 + 0.5 * np.cos(2 * np.pi * mode * x))
    a = np.maximum(a, 0.01)
    return a / a.sum()

def diverse_init(n, seed=0):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    inits = []
    
    # Sine-squared
    for f in [1, 2, 3]:
        a = np.sin(np.pi * f * x) ** 2
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    # Cosine window
    a = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(a / a.sum())
    
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for s in [0.1, 0.2, 0.3]:
            a = np.exp(-((x - c) ** 2) / (2 * s * s))
            inits.append(a / a.sum())
    
    # Power law
    for p in [0.5, 1.0, 2.0]:
        a = np.maximum(x, 0.01) ** p
        inits.append(a / a.sum())
        a = np.maximum(1 - x, 0.01) ** p
        inits.append(a / a.sum())
    
    # Spectral
    inits.append(spectral_init(n, 1))
    inits.append(spectral_init(n, 2))
    
    # Random smooth
    for _ in range(5):
        a = rng.uniform(0.5, 1.0, n)
        # Smooth it
        for _ in range(3):
            a = np.convolve(a, np.ones(5) / 5, mode='same')
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral balance (iterative multiplicative weight adjustment targeting autocorrelation peaks)
2. Enhanced LP Frank-Wolfe with violation-driven constraint management and backtracking line search
3. Focused pairwise descent with peak-aware destination scoring
4. Smoothed multiplicative Adam with LSE objective as fallback
5. Multi-n exploration with interpolation

import numpy as np
import time
import math
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.PDLP
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_balance(a0, time_limit, eta=0.08, verbose=True):
    """Iteratively balance the spectrum via multiplicative updates targeting autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_val - 1e-15:
            best_val = M
            best_a = a.copy()

        top_k = np.argsort(ac)[::-1][:4]

        W = np.ones(n)
        for k in top_k:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                contrib = a[k_int - idx]
                W[i_start:i_end] -= eta * contrib / (M + 1e-30)

        W = np.maximum(W, 0.05)
        a_new = a * W
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()
        a = a_new

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_val
            print(f"    SpecBal t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_val


def lp_frank_wolfe(a0, time_limit, solver, initial_K=1000, max_K=6000, verbose=True):
    """Enhanced LP Frank-Wolfe with violation-driven constraint management and backtracking."""
    n = len(a0)
    a = normalize(a0)
    ac_init = fast_autocorr(a)
    M_init = ac_init.max()

    best_a = a.copy()
    best_val = M_init
    t0 = time.monotonic()

    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    # Add neighbors of top peaks
    peak_idx = int(np.argmax(ac_init))
    for dk in range(-10, 11):
        kk = peak_idx + dk
        if 0 <= kk < len(ac_init):
            constraint_set.add(kk)

    y = a.copy()
    no_improve = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()

        if M_curr < best_val - 1e-14:
            best_val = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > 80 and t > 50:
            rng_r = np.random.default_rng(t)
            pert = rng_r.normal(0, 0.01, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            y = x / x.sum()
            no_improve = 0
            constraint_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
            continue

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > 500:
                constraint_set = set(int(k) for k in constraint_list[:500])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if len(constraint_list) > 500:
                constraint_set = set(int(k) for k in constraint_list[:500])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs

        # Check violations and add new constraints
        ac_g = fast_autocorr(g_dir)
        violations = np.where(ac_g > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = ac_g[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:80]]
            for k in top_v:
                constraint_set.add(int(k))

        # Add constraints near current peak
        pk = int(np.argmax(ac_y))
        for dk in range(-5, 6):
            kk = pk + dk
            if 0 <= kk < len(ac_y):
                constraint_set.add(kk)

        # Backtracking line search
        best_alpha = 2.0 / (t + 2.0)
        best_line_val = M_curr
        for alpha in [0.05, 0.08, 0.12, 0.15, 0.2, 0.3, 0.5]:
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            st = y_trial.sum()
            if st < 1e-15:
                continue
            y_trial = y_trial / st
            M_trial = fast_autocorr(y_trial).max()
            if M_trial < best_line_val:
                best_line_val = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_val
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_val
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_val


def focused_descent(a0, time_limit, verbose=True):
    """Peak-aware pairwise descent with destination scoring."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 250:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_val - 1e-15:
            best_val = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        top_peaks = np.argsort(ac)[::-1][:6]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            max_i = np.argmax(contrib)
            src = idx[max_i]

            if a[src] < 1e-10:
                continue

            # Score destinations by danger
            danger = np.zeros(n)
            for k2 in top_peaks[:4]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                if s2 < e2:
                    idx2 = np.arange(s2, e2)
                    danger[idx2] += a[k2i - idx2]
                    danger[k2i - idx2] += a[idx2]

            in_support = np.zeros(n, dtype=bool)
            in_support[i_start:i_end] = True
            safe = (~in_support)
            safe_idx = np.where(safe)[0]
            if len(safe_idx) < 20:
                safe_idx = np.arange(n)

            dst = safe_idx[np.argmin(danger[safe_idx])]
            if src == dst:
                continue

            for frac in [0.0005, 0.001, 0.003, 0.008, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_val - 1e-16:
                    a = a_trial
                    best_val = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.0005, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_val:
                    best_val = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_val
            print(f"    Focus ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_val


def smooth_adam(a0, time_limit, tau_decay=0.9999, lr=0.4, verbose=True):
    """Smoothed multiplicative Adam with LSE objective."""
    n = len(a0)
    a = normalize(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_val = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    tau = max(ac.max() * 0.2, 1e-10)
    tau_min = max(ac.max() * 0.005, 1e-14)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_val - 1e-15:
            best_val = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        tau = max(tau * tau_decay, tau_min)

        ac_scaled = (ac - ac.max()) / tau
        p = np.exp(ac_scaled)
        p = p / p.sum()

        a_flip = a[::-1]
        size = len(ac)
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a_flip, fft_size)
        Pf = np.fft.rfft(p, fft_size)
        conv_full = np.fft.irfft(Af * Pf, fft_size)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0.0)

        m_hat = 0.9 * m_hat + 0.1 * grad
        v_hat = 0.999 * v_hat + 0.001 * (grad * grad)
        bc1 = 1.0 - 0.9 ** t
        bc2 = 1.0 - 0.999 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2

        step = lr / (np.sqrt(v_c) + 1e-8)
        log_a = np.log(np.maximum(a, 1e-30)) - step * grad
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        if stall > 2000:
            stall = 0
            pert = rng.normal(0, 0.015, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            a = x / x.sum()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_val
            print(f"    Adam t={t} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2e}")

    return best_a, best_val


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        t6 = min(remaining * 0.4, 120.0)
        if t6 > 10:
            a_out, _ = focused_descent(best_seq, t6, verbose=False)
            try_update(a_out, f"focus_final_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 8:
            a_out, _ = spectral_balance(best_seq, t7, eta=0.03, verbose=False)
            try_update(a_out, f"specbal_final_n{n_cur}")

    # Final validation
    if best_seq is not None:
        fs = eval_obj(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = np.sin(np.pi * np.linspace(0, 1, n_fb)) ** 2
        best_seq = normalize(best_seq)
        best_score = eval_obj(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
