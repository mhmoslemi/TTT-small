import numpy as np
import time

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    """Compute J = 2n*max(conv(a,a))/(sum(a))^2 for nonnegative a."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max() / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_ramp(n, p=1.0):

    return normalize(x ** p)
