```python
"""
Search for a non-negative step-function coefficient sequence minimizing
    2 * n * max(conv(a, a)) / (sum(a)^2).

The implementation combines:
  1. A fast multiplicative gradient-descent engine on the simplex.
     The subgradient for the current peak autocorrelation lags is computed
     with FFT-based weighted cross-correlation.
  2. Periodic adaptive temperature / top-lag / step-size control and
     noise restarts from the best known point.
  3. A linear-programming direction step based on the state-of-art idea:
     maximize sum(b) subject to conv(a, b) <= current peak on selected
     active lags, then blend b with a by an exact line search.
  4. Very fast local pair-mass moves using exact rank-two convolution updates.
  5. Multi-length exploration by interpolation and re-optimization.
"""

import numpy as np
import time

try:
    import scipy.optimize as _SCIPY_OPT
except Exception:
    _SCIPY_OPT = None


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fft_size(n):
    # Enough zero padding for both a*a and cross-correlation a with weights.
    return next_pow2(3 * int(n) - 2)


def norm1(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 1e-15:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    return v / s


def score_candidate(a):
    a = norm1(a)
    n = int(a.size)
    if n == 0:
        return float("inf"), a, float("inf")
    N = fft_size(n)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    M = float(c.max())
    return (2.0 * n * M), a, M


def evaluate_safe(seq):
    try:
        lst = [float(x) for x in seq]
        val = evaluate_sequence(lst)
        if np.isfinite(val):
            return float(val)
        return float("inf")
    except NameError:
        sc, arr, _ = score_candidate(seq)
        return float(sc)
    except Exception:
        return float("inf")


def interp_sequence(f, new_n):
    f = norm1(f)
    old_n = int(f.size)
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    return norm1(g)


def parametric_sequence(n, kind, seed=0):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    if kind == "uniform":
        seq = np.ones(n, dtype=np.float64)
    elif kind == "sine2":
        seq = np.sin(np.pi * x) ** 2
    elif kind == "sine":
        seq = np.abs(np.sin(np.pi * x))
    elif kind == "ramp_up":
        seq = x + 0.05
    elif kind == "ramp_down":
        seq = (1.0 - x) + 0.05
    elif kind == "quad_up":
        seq = x * x + 0.05
    elif kind == "quad_down":
        seq = (1.0 - x) * (1.0 - x) + 0.05
    elif kind == "sqrt_up":
        seq = np.sqrt(x) + 0.05
    elif kind == "exp_up":
        seq = np.exp(2.5 * x)
    elif kind == "exp_down":
        seq = np.exp(2.5 * (1.0 - x))
    elif kind == "gauss_mid":
        seq = np.exp(-((x - 0.5) ** 2) / (2.0 * 0.2 * 0.2))
    elif kind == "gauss_left":
        seq = np.exp(-((x - 0.3) ** 2) / (2.0 * 0.25 * 0.25))
    elif kind == "gauss_right":
        seq = np.exp(-((x - 0.7) ** 2) / (2.0 * 0.25 * 0.25))
    elif kind == "triangle":
        seq = 1.0 - np.abs(2.0 * x - 1.0)
    elif kind == "double":
        seq = np.sin(2.0 * np.pi * x) ** 2 + 0.1
    else:
        seq = np.ones(n, dtype=np.float64)
    return norm1(seq)

def refine_sequence(a0, deadline, top_k=96, eta0=0.06, temp_ratio=0.04, tag=""):

    conv = np.maximum(conv, 0.0)
    M = float(conv.max())

    best_a = a.copy()
    best_M = M
    eta = float(eta0)
    temp = max(M * float(temp_ratio), M * 1e-12)
    patience = 0
    restart_count = 0
    top_k_max = int(min(L, max(16, top_k)))
    iter_count = 0
    rng = np.random.default_rng(20240521)

    while time.monotonic() < deadline - 25.0:
        if not np.isfinite(M) or M <= 0.0:
            break

        if patience > 3500 and restart_count < 12:
            noise_std = 0.015 + 0.015 * float(min(2, restart_count // 2))
            a = best_a * np.exp(rng.normal(0.0, noise_std, n))
            a = norm1(a)
            A = np.fft.rfft(a, N)
            conv = np.fft.irfft(A * A, N)[:L].copy()
            conv = np.maximum(conv, 0.0)
            M = float(conv.max())
            temp = max(M * 0.15, best_M * 0.02)
            eta = float(eta0)
            patience = 0
            restart_count += 1
            continue

        if patience > 8000:
            a = best_a.copy()
            A = np.fft.rfft(a, N)
            conv = np.fft.irfft(A * A, N)[:L].copy()
            conv = np.maximum(conv, 0.0)
            M = float(conv.max())
            temp = max(M * 0.25, best_M * 0.05)
            eta = float(eta0)
            top_k_max = min(L, top_k_max + 24)
            patience = 2000
            continue

        k = top_k_max
        if k <= 0:
            k = 1
        if k >= L:
            top = np.arange(L)
            vals = conv
        else:
            top = np.argpartition(conv, -k)[-k:]
            vals = conv[top]

        if vals.size == 0:
            break
        vmax = float(np.max(vals))
        if vmax <= 0.0 or not np.isfinite(vmax):
            break

        denom = max(temp, vmax * 1e-12)
        x = (vals - vmax) / denom
        x = np.clip(x, -60.0, 0.0)
        w = np.exp(x)
        sw = float(w.sum())
        if sw <= 0.0 or not np.isfinite(sw):
            w = np.full(w.size, 1.0 / float(w.size), dtype=np.float64)
        else:
            w = w / sw

        W = np.zeros(L, dtype=np.float64)
        W[top] = w
        Wf = np.fft.rfft(W, N)
        corr = np.fft.irfft(A * np.conj(Wf), N)[:n]
        grad = 2.0 * corr
        maxg = float(grad.max())

        if not np.isfinite(maxg) or maxg <= 1e-20:
            top_k_max = min(L, top_k_max + 16)
            patience += 50
            if patience > 5000:
                break
            continue

        gradn = grad / maxg
        M_old = M
        eta_eff = eta
        accepted = False

        for _attempt in range(3):
            a_new = a * np.exp(-eta_eff * gradn)
            a_new = np.maximum(a_new, 0.0)
            snew = float(a_new.sum())
            if not np.isfinite(snew) or snew <= 1e-15:
                eta_eff *= 0.4
                continue
            a_new = a_new / snew

            A_new = np.fft.rfft(a_new, N)
            conv_new = np.fft.irfft(A_new * A_new, N)[:L]
            conv_new = np.maximum(conv_new, 0.0)
            M_new = float(conv_new.max())

            if np.isfinite(M_new) and M_new <= M_old + 1e-13:
                a = a_new
                A = A_new
                conv = conv_new
                M = M_new
                accepted = True
                break

            eta_eff *= 0.4

        if accepted:
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            if M < M_old - 1e-14:
                eta = min(eta_eff * 1.12, 0.35)
            else:
                eta = max(eta_eff * 0.999, 0.001)
            temp = max(temp * 0.995, M * 1e-8)
        else:
            patience += 1
            eta = max(eta_eff * 0.5, 0.001)
            temp = min(M, temp * 1.35 + M * 0.001)
            if patience > 1000:
                top_k_max = min(L, top_k_max + 12)

        iter_count += 1
        if tag and iter_count % 4000 == 0:
            print(
                tag +
                " refine iter={:d} elapsed={:.0f}s internalJ={:.8f} curJ={:.8f} eta={:.5f}".format(
                    iter_count,
                    time.monotonic() - t_ref,
                    2.0 * n * best_M,
                    2.0 * n * M,
                    eta,
                )
            )

    return best_a


def pair_trial_delta(conv, a, s, d, amt):
    n = int(a.size)
    L = 2 * n - 1
    c = conv.copy()
    c[s:s + n] -= 2.0 * amt * a
    c[d:d + n] += 2.0 * amt * a
    q = amt * amt
    c[2 * s] += q
    if 0 <= s + d < L:
        c[s + d] -= 2.0 * q
    if 0 <= 2 * d < L:
        c[2 * d] += q
    m = float(c.max())
    return c, m


def commit_pair_move(a, c_trial, s, d, amt):
    a = a.copy()
    a[s] -= amt
    a[d] += amt
    a = np.maximum(a, 0.0)
    ss = float(a.sum())
    if not np.isfinite(ss) or ss <= 1e-15:
        return None, None, None, False
    if abs(ss - 1.0) > 1e-12:
        a = a / ss
        c_trial = c_trial / (ss * ss)
    conv = np.maximum(c_trial, 0.0)
    M = float(conv.max())
    return a, conv, M, True


def pair_search(a0, deadline, tag=""):
    a = norm1(a0)
    n = int(a.size)
    if n <= 1:
        return a
    L = 2 * n - 1
    N = fft_size(n)
    t_ref = time.monotonic()

    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[:L].copy()
    conv = np.maximum(conv, 0.0)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M
    move_count = 0
    stall = 0
    K = min(32, L)
    rng = np.random.default_rng(31337)

    while time.monotonic() < deadline - 15.0:
        if move_count > 0 and move_count % 250 == 0:
            A = np.fft.rfft(a, N)
            conv = np.fft.irfft(A * A, N)[:L].copy()
            conv = np.maximum(conv, 0.0)
            M = float(conv.max())
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()

        if K >= L:
            top = np.arange(L)
        else:
            top = np.argpartition(conv, -K)[-K:]

        H = np.zeros(n, dtype=np.float64)
        for kk in top:
            ki = int(kk)
            istart = max(0, ki - n + 1)
            iend = min(n, ki + 1)
            if istart < iend:
                idx = np.arange(istart, iend)
                H[idx] += a[ki - idx]

        positive = np.where(a > 1e-9)[0]
        if positive.size == 0:
            break
        hpos = H[positive]
        if hpos.size <= 8:
            high_local = np.arange(hpos.size)
        else:
            high_local = np.argpartition(hpos, -8)[-8:]
        high_idx = positive[high_local]

        if n <= 14:
            low_idx = np.arange(n)
        else:
            low_idx = np.argpartition(H, 14)[:14]

        improved = False
        for s in high_idx:
            if improved:
                break
            s = int(s)
            if a[s] <= 1e-10:
                continue
            for d in low_idx:
                if improved:
                    break
                d = int(d)
                if d == s:
                    continue
                for f in (0.05, 0.15, 0.35):
                    amt = float(a[s]) * f
                    if amt <= 0.0:
                        continue
                    c_trial, m_trial = pair_trial_delta(conv, a, s, d, amt)
                    if np.isfinite(m_trial) and m_trial < M - 1e-15:
                        res = commit_pair_move(a, c_trial, s, d, amt)
                        if res[3]:
                            a, conv, M, _ = res
                            if M < best_M - 1e-15:
                                best_M = M
                                best_a = a.copy()
                            improved = True
                            move_count += 1
                            stall = 0
                            break
                        else:
                            a = best_a.copy()
                            conv = np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(a, N), N)[:L]
                            conv = np.maximum(conv, 0.0)
                            M = float(conv.max())

        if not improved:
            if stall % 8 == 0:
                for _ in range(24):
                    s = int(rng.integers(0, n))
                    d = int(rng.integers(0, n))
                    if s == d or a[s] <= 1e-8:
                        continue
                    amt = float(a[s]) * 0.03
                    c_trial, m_trial = pair_trial_delta(conv, a, s, d, amt)
                    if np.isfinite(m_trial) and m_trial < M - 1e-15:
                        res = commit_pair_move(a, c_trial, s, d, amt)
                        if res[3]:
                            a, conv, M, _ = res
                            if M < best_M - 1e-15:
                                best_M = M
                                best_a = a.copy()
                            improved = True
                            move_count += 1
                            stall = 0
                            break

            if not improved:
                stall += 1
                if stall > 700:
                    break

        if tag and move_count > 0 and move_count % 2000 == 0:
            print(
                tag +
                " pair moves={:d} elapsed={:.0f}s internalJ={:.8f}".format(
                    move_count,
                    time.monotonic() - t_ref,
                    2.0 * n * best_M,
                )
            )

    return best_a


def lp_improve(a0, deadline, K=96, tag=""):
    a = norm1(a0)
    n = int(a.size)
    if n <= 0:
        return a
    L = 2 * n - 1
    N = fft_size(n)
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[:L].copy()
    conv = np.maximum(conv, 0.0)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M

    if _SCIPY_OPT is None or time.monotonic() > deadline - 30.0:
        return best_a
    # Keep LP size manageable.
    if n > 8500:
        return best_a

    Kcur = int(min(L, max(24, K)))
    fails = 0
    good = 0
    t_ref = time.monotonic()

    while time.monotonic() < deadline - 20.0:
        if not np.isfinite(M) or M <= 0.0:
            break
        if fails > 25:
            break

        if Kcur >= L:
            top = np.arange(L)
        else:
            top = np.argpartition(conv, -Kcur)[-Kcur:]

        A_ub = np.zeros((Kcur, n), dtype=np.float64)
        for r, kk in enumerate(top):
            ki = int(kk)
            if ki < n:
                if ki >= 0:
                    A_ub[r, 0:ki + 1] = a[ki::-1]
            else:
                jstart = ki - n + 1
                if jstart < n:
                    A_ub[r, jstart:n] = a[n - 1:jstart - 1:-1]

        A_ub *= float(n)
        current = conv[top] * float(n)
        rhs = np.maximum(M * float(n) * (1.0 + 1e-9), current + 1e-9)
        c = -np.ones(n, dtype=np.float64)
        bounds = np.empty((n, 2), dtype=np.float64)
        bounds[:, 0] = 0.0
        bounds[:, 1] = 1.0

        res = None
        try:
            res = _SCIPY_OPT.linprog(
                c,
                A_ub=A_ub,
                b_ub=rhs,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "dual_feasible": True, "time_limit": 8.0},
            )
        except Exception:
            try:
                res = _SCIPY_OPT.linprog(
                    c,
                    A_ub=A_ub,
                    b_ub=rhs,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True, "dual_feasible": True},
                )
            except Exception:
                res = None

        if res is None or not hasattr(res, "status") or res.status != 0 or res.x is None:
            fails += 1
            Kcur = max(24, Kcur // 2)
            continue

        x = np.asarray(res.x, dtype=np.float64)
        x = np.maximum(x, 0.0)
        S = float(x.sum())
        if S <= 1.0 + 1e-7:
            fails += 1
            Kcur = min(L, int(Kcur * 1.25) + 1)
            continue

        g = x / S
        G = np.fft.rfft(g, N)
        cross = np.fft.irfft(A * G, N)[:L]
        cross = np.maximum(cross, 0.0)
        conv_g = np.fft.irfft(G * G, N)[:L]
        conv_g = np.maximum(conv_g, 0.0)

        best_t = 0.0
        best_m = M
        for t in np.linspace(0.0, 1.0, 49):
            tt = float(t)
            mm = float(((1.0 - tt) * (1.0 - tt) * conv +
                        2.0 * tt * (1.0 - tt) * cross +
                        tt * tt * conv_g).max())
            if np.isfinite(mm) and mm < best_m:
                best_m = mm
                best_t = tt

        if best_t > 0.02 and best_m < best_M - 1e-12:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.05)
            for t in np.linspace(lo, hi, 25):
                tt = float(t)
                mm = float(((1.0 - tt) * (1.0 - tt) * conv +
                            2.0 * tt * (1.0 - tt) * cross +
                            tt * tt * conv_g).max())
                if np.isfinite(mm) and mm < best_m:
                    best_m = mm
                    best_t = tt

            if best_t > 1e-4 and best_m < best_M - 1e-12:
                a = (1.0 - best_t) * a + best_t * g
                a = norm1(a)
                A = np.fft.rfft(a, N)
                conv = np.fft.irfft(A * A, N)[:L].copy()
                conv = np.maximum(conv, 0.0)
                M = float(conv.max())
                if M < best_M - 1e-14:
                    best_M = M
                    best_a = a.copy()
                good += 1
                fails = 0
                if tag and good % 5 == 0:
                    print(
                        tag +
                        " lp accepts={:d} elapsed={:.0f}s internalJ={:.8f}".format(
                            good,
                            time.monotonic() - t_ref,
                            2.0 * n * best_M,
                        )
                    )
            else:
                fails += 1
                Kcur = min(L, int(Kcur * 1.2) + 1)
        else:
            fails += 1
            Kcur = min(L, int(Kcur * 1.2) + 1)

    return best_a


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 1800.0
    total = min(total, 1800.0)
    global_deadline = t0 + max(30.0, total - 60.0)

    parent = None
    try:
        parent = height_sequence_1
    except Exception:
        parent = None

    init_deadline = min(global_deadline, t0 + 45.0)
    cands = choose_initial_candidates(parent, init_deadline)

    if cands:
        best_score, best_a, _ = cands[0]
        best_a = norm1(best_a)
    else:
        fa = parametric_sequence(4000, "sine2")
        best_score, best_a, _ = score_candidate(fa)

    if not np.isfinite(best_score):
        fa = parametric_sequence(4000, "sine2")
        best_score, best_a, _ = score_candidate(fa)

    t_init = time.monotonic()
    usable = max(30.0, global_deadline - t_init)

    now = time.monotonic()
    p1_deadline = min(global_deadline, now + max(30.0, usable * 0.42))
    if p1_deadline - now > 25.0:
        out = refine_sequence(best_a, p1_deadline, top_k=96, eta0=0.06, temp_ratio=0.04, tag="main")
        sc, out, _ = score_candidate(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = out

    now = time.monotonic()
    p2_deadline = min(global_deadline, now + max(20.0, usable * 0.12))
    if _SCIPY_OPT is not None and p2_deadline - now > 35.0:
        out = lp_improve(best_a, p2_deadline, K=110, tag="lp")
        sc, out, _ = score_candidate(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = out

    now = time.monotonic()
    p3_deadline = min(global_deadline, now + max(15.0, usable * 0.09))
    if p3_deadline - now > 20.0:
        out = pair_search(best_a, p3_deadline, tag="pair")
        sc, out, _ = score_candidate(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = out

    now = time.monotonic()
    if len(cands) >= 2:
        p4_deadline = min(global_deadline, now + max(20.0, usable * 0.14))
        if p4_deadline - now > 25.0:
            _, sec_a, _ = cands[1]
            out = refine_sequence(sec_a, p4_deadline, top_k=80, eta0=0.05, temp_ratio=0.03, tag="second")
            sc, out, _ = score_candidate(out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = out

    now = time.monotonic()
    if global_deadline - now > 50.0:
        current_n = int(best_a.size)
        tried = set([current_n])
        for new_n in (current_n + 800, current_n + 1600, current_n - 800, 8000, 9000, 10000):
            if global_deadline - time.monotonic() < 50.0:
                break
            if not (3000 <= new_n <= 10000) or new_n in tried:
                continue
            alloc = min(65.0, max(25.0, (global_deadline - time.monotonic()) * 0.35))
            sub_deadline = time.monotonic() + alloc
            if sub_deadline - time.monotonic() < 25.0:
                continue
            tried.add(new_n)

            a_init = interp_sequence(best_a, new_n)
            sc0, a_init, _ = score_candidate(a_init)
            a_r = parametric_sequence(new_n, "ramp_up")
            sc1, a_r, _ = score_candidate(a_r)
            if sc1 < sc0:
                a_init = a_r

            out = refine_sequence(a_init, sub_deadline, top_k=80, eta0=0.05, temp_ratio=0.04, tag="len")
            sc, out, _ = score_candidate(out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = out

    now = time.monotonic()
    if global_deadline - now > 25.0:
        final_deadline = min(global_deadline, now + max(15.0, (global_deadline - now) * 0.6))
        out = refine_sequence(best_a, final_deadline, top_k=120, eta0=0.025, temp_ratio=0.02, tag="final")
        sc, out, _ = score_candidate(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = out

    now = time.monotonic()
    if global_deadline - now > 18.0:
        final_pair_deadline = min(global_deadline, now + max(10.0, (global_deadline - now) * 0.8))
        out = pair_search(best_a, final_pair_deadline, tag="final_pair")
        sc, out, _ = score_candidate(out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = out

    try:
        final_seq = [float(x) for x in best_a]
    except Exception:
        final_seq = [1.0]

    official_best_seq = final_seq
    official_best_score = evaluate_safe(final_seq)

    if global_deadline - time.monotonic() > 12.0:
        try:
            p_arr = norm1(height_sequence_1)
            p_seq = [float(x) for x in p_arr]
            p_score = evaluate_safe(p_seq)
            if np.isfinite(p_score) and (not np.isfinite(official_best_score) or p_score < official_best_score):
                official_best_seq = p_seq
                official_best_score = p_score
        except Exception:
            pass

    if len(cands) >= 2:
        try:
            _, sec_a, _ = cands[1]
            sec_seq = [float(x) for x in sec_a]
            sec_score = evaluate_safe(sec_seq)
            if np.isfinite(sec_score) and (not np.isfinite(official_best_score) or sec_score < official_best_score):
                official_best_seq = sec_seq
                official_best_score = sec_score
        except Exception:
            pass

    if not np.isfinite(official_best_score):
        fb = parametric_sequence(3000, "sine2")
        official_best_seq = [float(x) for x in fb]
        official_best_score = evaluate_safe(official_best_seq)

    return official_best_seq
```
