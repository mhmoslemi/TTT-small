def interpolate_sequence(f,new_n):
   old_n=len(f)
   if old_n==new_n: return f.copy()
   f = normalize_sum(f)
   if old_n==1: return np.ones(new_n)/new_n
   old_x=np.linspace(0,1,old_n)
   new_x=np.linspace(0,1,new_n)
   out=np.interp(new_x, old_x, f)

def softmax_u(u):
    m=u.max()
    e=np.exp(u-m)
    s=e.sum()
    if s <=0 or not np.isfinite(s):
       # fallback: set large negative to 0
       e = np.where(e>0, e, 0)
       s=e.sum()
       if s <=0: return np.ones_like(u)/u.size
    return e/s

    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fast_conv(a):
    """Fast linear convolution a * a using zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    size = 1 << (max(1, L) - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:L]


def max_scaled_conv(a):
    """Return n * max(conv(a, a)), assuming a is non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return 0.0
    c = fast_conv(a)
    return float(n * c.max())


def evaluate_safe(a):
    """Call the external evaluator safely."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:
        return float("inf")


def softmax_weights(d, beta):
    """Stable softmax weights over scaled convolution coefficients."""
    d = np.asarray(d, dtype=np.float64).ravel()
    L = d.size
    if L == 0:
        return np.empty(0, dtype=np.float64)
    if beta <= 1e-8:
        return np.ones(L, dtype=np.float64) / float(L)
    m = float(d.max())
    x = beta * (d - m)
    x = np.maximum(x, -50.0)
    z = np.exp(x)
    s = float(z.sum())
    if s <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[d >= m - 1e-12] = 1.0
        s = float(w.sum())
        if s <= 0.0:
            w[0] = 1.0
        else:
            w /= s
        return w
    return z / s


def smooth_value(d, beta):
    """Annealed log-sum-exp value for d."""
    d = np.asarray(d, dtype=np.float64).ravel()
    if d.size == 0:
        return 0.0
    if beta <= 1e-8:
        return float(d.sum() / d.size)
    m = float(d.max())
    x = beta * (d - m)
    x = np.maximum(x, -50.0)
    return float(m + np.log(float(np.exp(x).sum())) / beta)


def weight_grad(a, w):
    """
    Gradient of sum_k w_k * c_k with respect to a, where c = a * a.
    Returns 2 * sum_j a_j w_{i+j}.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

    wl = 2 * n - 1
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size < wl:
        w2 = np.zeros(wl, dtype=np.float64)
        if w.size > 0:
            w2[:w.size] = w
        w = w2
    elif w.size > wl:
        w = w[:wl]

    L = 3 * n - 2
    size = 1 << (max(1, L) - 1).bit_length()
    b = a[::-1]
    B = np.fft.rfft(b, size)
    W = np.fft.rfft(w, size)
    conv = np.fft.irfft(B * W, size)[:L]
    off = n - 1
    return 2.0 * conv[off:off + n]


def exp_softplus_normalize(u):
    """Stable softmax u -> probabilities."""
    u = np.asarray(u, dtype=np.float64).ravel()
    n = u.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = float(u.max())
    e = np.exp(np.maximum(u - m, -70.0))
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def smooth_noise(rng, n, n_modes=24):
    """Low-frequency smooth random perturbation."""
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    x = np.arange(n, dtype=np.float64)
    noise = np.zeros(n, dtype=np.float64)
    nd = int(max(1, n_modes))
    for m in range(nd):
        freq = rng.uniform(0.5, 12.0)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = 1.0 / (1.0 + float(m))
        noise += amp * np.sin(2.0 * np.pi * freq * x / float(n) + phase)
    sd = float(noise.std())
    if sd < 1e-15:
        return noise
    return noise / sd


def update_elites(elites, seq, d, max_keep=6):
    """Maintain a small list of best sequences (top-level, no lambdas)."""
    seq = normalize_sum(np.asarray(seq, dtype=np.float64).ravel())
    d = float(d)
    for i in range(len(elites)):
        if abs(elites[i][0] - d) <= 1e-13:
            if seq.size == elites[i][1].size:
                if float(np.max(np.abs(elites[i][1] - seq))) <= 1e-9:
                    return
    elites.append((d, seq.copy()))
    m = len(elites)
    for i in range(m):
        mi = i
        for j in range(i + 1, m):
            if elites[j][0] < elites[mi][0]:
                mi = j
        if mi != i:
            elites[i], elites[mi] = elites[mi], elites[i]
    while len(elites) > max_keep:
        elites.pop()


def poly_sequence(n, alpha, coeffs):
    """
    Sequence a_i proportional to (i+1)^(-alpha) * exp(sum_m c_m x_i^m),
    x_i = i/(n-1), normalized to sum 1.
    """
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    i = np.arange(1, n + 1, dtype=np.float64)
    base = np.power(i, -float(alpha))
    if n > 1 and len(coeffs) > 0:
        x = np.arange(n, dtype=np.float64) / float(n - 1)
        q = np.zeros(n, dtype=np.float64)
        p = np.ones(n, dtype=np.float64)
        for c in coeffs:
            q += float(c) * p
            p = p * x
        q -= float(q.max())
        base = base * np.exp(q)
    return normalize_sum(base)


def decode_poly_params(params, D):
    """Decode tanh-bounded parameters to alpha in (0,1) and bounded coeffs."""
    p = np.asarray(params, dtype=np.float64).ravel()
    if p.size == 0:
        return 0.5, [0.0] * int(D)
    alpha = 0.5 + 0.5 * float(np.tanh(p[0]))
    alpha = float(np.clip(alpha, 1e-4, 1.0 - 1e-6))
    coeffs = []
    for k in range(int(D)):
        c = 2.0 * float(np.tanh(p[1 + k])) if (1 + k) < p.size else 0.0
        coeffs.append(c)
    return alpha, coeffs


def encoded_poly_score(params, n, D):
    """Objective for scipy: 2 * max_scaled_conv(polynomial sequence)."""
    alpha, coeffs = decode_poly_params(params, D)
    seq = poly_sequence(int(n), alpha, coeffs)
    return float(2.0 * max_scaled_conv(seq))


def fit_poly_params(a, D):
    """Initialize tanh parameters by polynomial fitting log(a / singular base)."""
    full = np.zeros(int(D) + 1, dtype=np.float64)
    seq = np.asarray(a, dtype=np.float64).ravel()
    n = seq.size
    if D <= 0 or n < 3:
        return full
    seq = normalize_sum(seq)
    base = np.power(np.arange(1, n + 1, dtype=np.float64), -0.5)
    base = base / base.sum()
    ratio = np.maximum(seq, 1e-300) / np.maximum(base, 1e-300)
    y = np.log(ratio)
    x = np.arange(n, dtype=np.float64) / float(n - 1)
    A = np.ones((n, D), dtype=np.float64)
    xp = np.ones(n, dtype=np.float64)
    for m in range(D):
        A[:, m] = xp
        xp = xp * x
    try:
        sol = np.linalg.lstsq(A, y, rcond=None)[0]
    except Exception:
        sol = np.zeros(D, dtype=np.float64)
    sol = np.asarray(sol, dtype=np.float64)
    sol = np.nan_to_num(sol, nan=0.0, posinf=0.0, neginf=0.0)
    sol = np.clip(sol, -1.9, 1.9)
    full[1:1 + D] = np.arctanh(sol / 2.0)
    return full


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length, preserving sum 1."""
    f = normalize_sum(np.asarray(f, dtype=np.float64).ravel())
    old_n = f.size
    new_n = int(new_n)
    if new_n <= 0:
        return np.ones(1, dtype=np.float64)
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    out = np.interp(nx, ox, f)
    return normalize_sum(out)


def optimize_sequence(a_init, time_limit, seed=1, verbose=False):
    """
    Local optimizer for fixed length.

    Uses:
      - u = log(a) mirror/exponentiated gradient on an annealed logsumexp
        approximation to max(conv(a,a)),
      - FFT-based full gradient for all convolution coefficients,
      - small Frank-Wolfe-like mass transfers from high-gradient to low
        gradient positions,
      - smooth random restarts and elite retention.

    Returns (best_a, best_d, best_score) where best_score = 2*best_d.
    """
    t0 = time.monotonic()
    limit = float(time_limit)
    if limit <= 1e-3:
        a = normalize_sum(a_init)
        d = max_scaled_conv(a)
        return a, d, float(2.0 * d)

    a = normalize_sum(np.asarray(a_init, dtype=np.float64).ravel())
    n = int(a.size)
    if n <= 0:
        a = np.ones(1, dtype=np.float64)
        n = 1

    u = np.log(np.maximum(a, 1e-300))
    u = u - float(u.mean())
    u = np.clip(u, -70.0, 70.0)

    c = fast_conv(a)
    d = n * c
    cur_d = float(d.max())
    best_d = cur_d
    best_a = a.copy()

    elites = [(cur_d, best_a.copy())]
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    reset_stall = 0
    beta = 20.0

    while time.monotonic() - t0 < limit:
        # Optional early exit to leave a tiny margin to caller.
        if time.monotonic() - t0 > limit - 1e-3:
            break

        a = exp_softplus_normalize(u)
        c = fast_conv(a)
        d = n * c
        cur_d = float(d.max())

        if cur_d < best_d - 1e-13:
            best_d = cur_d
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        smooth_cur = smooth_value(d, beta)

        if cur_d > best_d + 0.005:
            reset_stall += 1
            if reset_stall > 40:
                a = best_a.copy()
                u = np.log(np.maximum(a, 1e-300))
                u = u - float(u.mean())
                u = np.clip(u, -70.0, 70.0)
                c = fast_conv(a)
                d = n * c
                cur_d = float(d.max())
                smooth_cur = smooth_value(d, beta)
                reset_stall = 0
            else:
                it += 1
                continue
        else:
            reset_stall = 0

        w = softmax_weights(d, beta)
        grad = weight_grad(a, w)
        g = float(n) * (grad - float(grad.mean()))
        denom = float(np.std(g) + 1e-12)
        if not np.isfinite(denom) or denom < 1e-15:
            denom = 1.0
        gu = np.tanh(g / denom)

        step = 0.25 / (1.0 + float(it) / 3000.0) + 0.02
        global_accept = False

        for mult in (0.5, 1.0, 2.0, 4.0):
            u_try = u - step * mult * gu
            u_try = u_try - float(u_try.mean())
            u_try = np.clip(u_try, -70.0, 70.0)
            a_try = exp_softplus_normalize(u_try)
            c_try = fast_conv(a_try)
            d_try = n * c_try
            dm = float(d_try.max())
            sv = smooth_value(d_try, beta)

            if (sv < smooth_cur - 1e-10 and dm <= best_d + 0.005) or (dm < best_d - 1e-13):
                u = u_try
                a = a_try
                d = d_try
                cur_d = dm
                smooth_cur = sv
                global_accept = True
                if dm < best_d - 1e-13:
                    best_d = dm
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
                break

        transfer_accept = False
        if not global_accept:
            best_trial_d = cur_d
            best_trial_s = smooth_cur
            best_trial_a = None

            if n > 1:
                k = min(4, n)
                if k < n:
                    srcs = np.argpartition(g, -k)[-k:]
                    dsts = np.argpartition(g, k)[:k]
                else:
                    srcs = np.arange(n)
                    dsts = np.arange(n)

                srcs = srcs[np.argsort(-g[srcs])]
                dsts = dsts[np.argsort(g[dsts])]
                srcs = srcs[:min(3, len(srcs))]
                dsts = dsts[:min(3, len(dsts))]

                transfer_fracs = (0.0005, 0.002, 0.01, 0.05)
                got = False

                for src in srcs:
                    src_val = float(a[src])
                    if src_val <= 1e-12:
                        continue
                    for dst in dsts:
                        if dst == src:
                            continue
                        for frac in transfer_fracs:
                            amt = src_val * float(frac)
                            if amt <= 0.0:
                                continue
                            a_t = a.copy()
                            a_t[src] -= amt
                            a_t[dst] += amt
                            if a_t[src] < 0.0:
                                a_t[src] = 0.0
                            # Sum remains 1 to floating-point accuracy.
                            c_t = fast_conv(a_t)
                            d_t = n * c_t
                            dm_t = float(d_t.max())
                            sv_t = smooth_value(d_t, beta)

                            if (dm_t < best_trial_d - 1e-13) or \
                               (sv_t < best_trial_s - 1e-10 and dm_t <= best_d + 0.005):
                                best_trial_d = dm_t
                                best_trial_s = sv_t
                                best_trial_a = a_t
                                got = True

                if got and best_trial_a is not None:
                    a = best_trial_a
                    u = np.log(np.maximum(a, 1e-300))
                    u = u - float(u.mean())
                    u = np.clip(u, -70.0, 70.0)
                    cur_d = best_trial_d
                    smooth_cur = best_trial_s
                    transfer_accept = True
                    if cur_d < best_d - 1e-13:
                        best_d = cur_d
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall += 1

        if not global_accept and not transfer_accept:
            if stall > 80 and (it % 40 == 0):
                amp = 0.04 / (1.0 + float(stall) / 200.0)
                nz = smooth_noise(rng, n)
                u_try = u + amp * nz
                u_try = u_try - float(u_try.mean())
                u_try = np.clip(u_try, -70.0, 70.0)
                a_try = exp_softplus_normalize(u_try)
                c_try = fast_conv(a_try)
                d_try = n * c_try
                dm_try = float(d_try.max())

                if dm_try <= best_d + 0.005:
                    u = u_try
                    a = a_try
                    d = d_try
                    cur_d = dm_try
                    smooth_cur = smooth_value(d, beta)
                else:
                    u = np.log(np.maximum(best_a, 1e-300))
                    u = u - float(u.mean())
                    u = np.clip(u, -70.0, 70.0)
                    a = best_a.copy()
                    c = fast_conv(a)
                    d = n * c
                    cur_d = float(d.max())
                    smooth_cur = smooth_value(d, beta)

            if stall > 800 and len(elites) > 1 and (it % 400 == 0):
                idx = int(rng.integers(1, len(elites)))
                e_a = elites[idx][1]
                u = np.log(np.maximum(e_a, 1e-300))
                u = u - float(u.mean())
                u = np.clip(u, -70.0, 70.0)
                a = exp_softplus_normalize(u)
                c = fast_conv(a)
                d = n * c
                cur_d = float(d.max())
                smooth_cur = smooth_value(d, beta)

        if it % 200 == 0:
            update_elites(elites, a, cur_d, 6)

        if it < 1500:
            beta = 20.0 * (25.0 ** (float(it) / 1500.0))
        elif it < 5000:
            beta = 500.0
        else:
            beta = 100.0 + 200.0 * (0.5 + 0.5 * float(np.sin(0.01 * it)))

        if verbose and it % 5000 == 0:
            print(f"    optimize it={it} t={time.monotonic()-t0:.0f}s "
                  f"best={2.0*best_d:.8f} cur={2.0*cur_d:.8f} beta={beta:.1f}")

        it += 1

    best_a = normalize_sum(best_a)
    final_score = float(2.0 * best_d)
    return best_a, best_d, final_score


def propose_candidate(budget_s=1800.0):
    """Main search routine; always returns a list of non-negative heights."""
    t_start = time.monotonic()
    remaining_budget = max(10.0, float(budget_s) - 25.0)
    deadline = t_start + remaining_budget

    D = 6
    params = np.zeros(D + 1, dtype=np.float64)

    best_a = None
    best_d = float("inf")
    best_score = float("inf")
    n_main = 4200
    parent = None

    # Load previously supplied construction if available.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0 and bool(np.all(np.isfinite(p))):
            p = np.clip(p, 0.0, 1000.0)
            if float(p.sum()) > 1e-8:
                parent = normalize_sum(p)
                best_a = parent.copy()
                best_d = max_scaled_conv(best_a)
                best_score = float(2.0 * best_d)
                n_main = int(best_a.size)
                print(f"Parent loaded: n={n_main}, local_score={best_score:.8f}")
    except Exception:
        parent = None

    if best_a is None:
        n_main = 4200
        best_a = poly_sequence(n_main, 0.5, [0.0] * D)
        best_d = max_scaled_conv(best_a)
        best_score = float(2.0 * best_d)
        print(f"No parent: using power-poly fallback, n={n_main}, score={best_score:.8f}")

    # If a supplied parent is too long, shrink to a safe length.
    if n_main > 5500:
        best_a = interpolate_sequence(best_a, 5000)
        n_main = 5000
        best_d = max_scaled_conv(best_a)
        best_score = float(2.0 * best_d)
        print(f"Capped length to {n_main}, score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1: bounded polynomial tapered singular power search.
    # ------------------------------------------------------------------
    try:
        if parent is not None:
            params = fit_poly_params(parent, D)
        else:
            params = np.zeros(D + 1, dtype=np.float64)
            params[1] = -0.20
            params[2] = 0.10
            params[3] = -0.05
    except Exception:
        params = np.zeros(D + 1, dtype=np.float64)

    poly_ns = []
    for nn in [512, 1536, n_main]:
        if nn not in poly_ns:
            poly_ns.append(nn)

    for n_poly in poly_ns:
        now = time.monotonic()
        if now > deadline - 30.0:
            break
        maxiter = 90
        if n_poly < 1000:
            maxiter = 160
        elif n_poly < 3000:
            maxiter = 110
        else:
            maxiter = 80

        res = None
        try:
            res = scipy_minimize(
                encoded_poly_score,
                np.asarray(params, dtype=np.float64),
                args=(n_poly, D),
                method="Nelder-Mead",
                options={"maxiter": int(maxiter), "xatol": 1e-5, "fatol": 1e-8}
            )
        except Exception:
            res = None

        if res is not None and hasattr(res, "x"):
            rx = np.asarray(res.x, dtype=np.float64).ravel()
            if rx.size >= D + 1 and bool(np.all(np.isfinite(rx))):
                params = rx

        alpha, coeffs = decode_poly_params(params, D)
        seq = poly_sequence(n_poly, alpha, coeffs)
        sd = max_scaled_conv(seq)
        ss = float(2.0 * sd)
        print(f"Poly search n={n_poly}: score={ss:.8f}")

        if ss < best_score - 1e-12:
            if n_poly == n_main:
                best_score = ss
                best_d = sd
                best_a = seq.copy()
            else:
                seq_main = interpolate_sequence(seq, n_main)
                smd = max_scaled_conv(seq_main)
                ssm = float(2.0 * smd)
                if ssm < best_score - 1e-12:
                    best_score = ssm
                    best_d = smd
                    best_a = seq_main.copy()

    # ------------------------------------------------------------------
    # Phase 2: main intensive local optimizer on current best length.
    # ------------------------------------------------------------------
    now = time.monotonic()
    if now < deadline - 60.0:
        t_main = min((deadline - now) * 0.55, 900.0)
        if t_main > 30.0:
            seq, d, sc = optimize_sequence(best_a, t_main, seed=1, verbose=True)
            if sc < best_score - 1e-12:
                best_score = sc
                best_d = d
                best_a = seq.copy()
                n_main = int(best_a.size)
                print(f"Main optimizer improved: score={best_score:.8f}, n={n_main}")
            else:
                print(f"Main optimizer did not improve; best remains {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3: second local optimizer from a smooth perturbation.
    # ------------------------------------------------------------------
    now = time.monotonic()
    if now < deadline - 60.0:
        rng = np.random.default_rng(2)
        nz = smooth_noise(rng, int(best_a.size))
        u_pert = np.log(np.maximum(best_a, 1e-300))
        u_pert = u_pert + 0.05 * nz
        u_pert = u_pert - float(u_pert.mean())
        u_pert = np.clip(u_pert, -70.0, 70.0)
        a_pert = exp_softplus_normalize(u_pert)
        t2 = min((deadline - now) * 0.30, 420.0)
        if t2 > 30.0:
            seq, d, sc = optimize_sequence(a_pert, t2, seed=2, verbose=False)
            if sc < best_score - 1e-12:
                best_score = sc
                best_d = d
                best_a = seq.copy()
                n_main = int(best_a.size)
                print(f"Perturbed restart improved: score={best_score:.8f}, n={n_main}")

    # ------------------------------------------------------------------
    # Phase 4: local optimizer from the polynomial parametrization.
    # ------------------------------------------------------------------
    now = time.monotonic()
    if now < deadline - 90.0:
        try:
            alpha, coeffs = decode_poly_params(params, D)
        except Exception:
            alpha, coeffs = (0.5, [0.0] * D)
        rng3 = np.random.default_rng(3)
        nz3 = smooth_noise(rng3, int(best_a.size))
        seq3 = poly_sequence(int(best_a.size), alpha, coeffs)
        seq3 = normalize_sum(0.98 * seq3 + 0.02 * sz_no 
                              if False else seq3)  # no-op guard to avoid accidental neg
        # More sensible positive perturbation:
        u3 = np.log(np.maximum(seq3, 1e-300)) + 0.03 * nz3
        u3 = u3 - float(u3.mean())
        u3 = np.clip(u3, -70.0, 70.0)
        seq3 = exp_softplus_normalize(u3)

        t3 = min((deadline - now) * 0.18, 240.0)
        if t3 > 30.0:
            seq, d, sc = optimize_sequence(seq3, t3, seed=3, verbose=False)
            if sc < best_score - 1e-12:
                best_score = sc
                best_d = d
                best_a = seq.copy()
                n_main = int(best_a.size)
                print(f"Poly restart improved: score={best_score:.8f}, n={n_main}")

    # ------------------------------------------------------------------
    # Phase 5: length trials.
    # ------------------------------------------------------------------
    tried = set()
    now = time.monotonic()
    for n_try in [min(5200, int(n_main) + 400), max(2500, int(n_main) - 400), 5000]:
        now = time.monotonic()
        if now > deadline - 70.0:
            break
        n_try = int(n_try)
        if n_try <= 100 or n_try > 5500 or n_try in tried:
            continue
        tried.add(n_try)

        seq_try = interpolate_sequence(best_a, n_try)
        d_try = max_scaled_conv(seq_try)
        s_try = float(2.0 * d_try)
        print(f"Length trial n={n_try}: interp_score={s_try:.8f}")
        if s_try < best_score - 1e-12:
            best_score = s_try
            best_d = d_try
            best_a = seq_try.copy()
            n_main = n_try

        t4 = min((deadline - time.monotonic()) * 0.20, 180.0)
        if t4 > 25.0:
            seq, d, sc = optimize_sequence(seq_try, t4, seed=10 + (n_try % 7), verbose=False)
            if sc < best_score - 1e-12:
                best_score = sc
                best_d = d
                best_a = seq.copy()
                n_main = int(best_a.size)
                print(f"Length trial optimized n={n_main}: score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6: final short polish on best sequence.
    # ------------------------------------------------------------------
    now = time.monotonic()
    if now < deadline - 20.0:
        t_final = min((deadline - now) * 0.75, 130.0)
        if t_final > 10.0:
            seq, d, sc = optimize_sequence(best_a, t_final, seed=42, verbose=False)
            if sc < best_score - 1e-12:
                best_score = sc
                best_d = d
                best_a = seq.copy()
                n_main = int(best_a.size)
                print(f"Final polish improved: score={best_score:.8f}, n={n_main}")

    # Final sanitation and exact evaluator validation.
    best_a = normalize_sum(np.asarray(best_a, dtype=np.float64).ravel())
    local_score = float(2.0 * max_scaled_conv(best_a))

    try:
        exact_score = evaluate_safe(best_a)
    except Exception:
        exact_score = float("inf")

    if np.isfinite(exact_score):
        if exact_score < best_score:
            best_score = exact_score
    else:
        # If external evaluator cannot be used for some reason, use local value.
        exact_score = local_score
        best_score = local_score

    if not np.isfinite(best_score):
        fallback_n = max(1000, min(4200, int(n_main)))
        best_a = poly_sequence(fallback_n, 0.5, [0.0] * D)
        local_score = float(2.0 * max_scaled_conv(best_a))
        best_score = local_score
        print("Fallback sequence used.")

    best_a = np.maximum(np.asarray(best_a, dtype=np.float64), 0.0)
    best_a = normalize_sum(best_a)
    elapsed = time.monotonic() - t_start
    print(f"FINAL: n={len(best_a)}, local_score={local_score:.8f}, "
          f"eval_score={exact_score:.8f}, elapsed={elapsed:.1f}s")

    return best_a.tolist()
