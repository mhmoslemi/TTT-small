def fast_autocorr(a):
    """FFT-based linear autocorrelation (full convolution)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Evaluate safely, converting to list."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_u_shape(n, sharpness=2.0):
    """Create a U-shaped initialization."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)

def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    """Compute soft subgradient of max autocorrelation."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0,

    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:

        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if adaptive and stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)

            x = x_r

            cur_top_k = min(cur_top_k + 10, 150)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.maximum(v, 0.0)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def lp_frank_wolfe(a0, time_limit, solver, top_K_init=200, top_K_max=None,
                   K_growth=40, verbose=True):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                idx = np.arange(j_start, j_end)
                T_sel[ii, idx] = a[ki - idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                      0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        if iter_count % 10 == 0:
            K = min(K + K_growth, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, rng_seed=0, n_peaks=15, n_trials_per_round=30):
    """
    Fast coordinate descent with analytical step sizing.
    Uses the fact that for a transfer move (i -> j, amount eps),
    the first-order change in c[k] is 2*eps*(a[k-j] - a[k-i]).
    """
    n = len(a0)
    a = norm_sum1(a0)
    c = fast_autocorr(a)
    best_m = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300
    round_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        round_count += 1
        M = c.max()

        # Find top peak lags
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        # Build sensitivity vector: how much each position contributes to peaks
        sens = np.zeros(n)
        for k in peak_idx[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        # Identify hot sources and cold destinations
        hot_count = min(30, n)
        hot_sources = np.argsort(sens)[::-1][:hot_count]
        cold_thresh = np.percentile(sens, 30)
        cold_dests = np.where(sens <= cold_thresh)[0]
        if len(cold_dests) < 5:
            cold_dests = np.argsort(sens)[:max(5, n // 10)]

        found = False
        for trial in range(n_trials_per_round):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick source (biased toward hot)
            if rng.random() < 0.7 and len(hot_sources) > 0:
                src = int(hot_sources[rng.integers(len(hot_sources))])
            else:
                src = int(rng.integers(n))

            if a[src] < 1e-12:
                continue

            # Pick destination (biased toward cold)
            if rng.random() < 0.7 and len(cold_dests) > 0:
                dst = int(cold_dests[rng.integers(len(cold_dests))])
            else:
                dst = int(rng.integers(n))

            if dst == src:
                continue

            # Compute analytical step size
            # Δc[k] ≈ 2*eps*(a[k-dst] - a[k-src])
            # Need: c[k] + 2*eps*(a[k-dst] - a[k-src]) <= M for all k
            # For k where (a[k-dst] - a[k-src]) > 0: eps <= (M - c[k]) / (2*(a[k-dst] - a[k-src]))

            # Determine k range where either term is non-zero
            k_lo = min(src, dst)
            k_hi = min(max(src + n - 1, dst + n - 1), 2 * n - 2)

            ks = np.arange(k_lo, k_hi + 1)
            n_ks = len(ks)

            # a[k-src] valid for k in [src, src+n-1]
            d_src = np.zeros(n_ks)
            vs = (ks >= src) & (ks < src + n)
            if vs.any():
                d_src[vs] = a[ks[vs] - src]

            # a[k-dst] valid for k in [dst, dst+n-1]
            d_dst = np.zeros(n_ks)
            vd = (ks >= dst) & (ks < dst + n)
            if vd.any():
                d_dst[vd] = a[ks[vd] - dst]

            diff = d_dst - d_src  # a[k-dst] - a[k-src]
            c_vals = c[k_lo:k_hi + 1]

            pos_mask = diff > 1e-15
            if pos_mask.any():
                # eps <= (M - c[k]) / (2 * diff[k])
                numer = M - c_vals[pos_mask]
                denom = 2.0 * diff[pos_mask]
                valid = numer > 0
                if valid.any():
                    eps_cand = (numer[valid] / denom[valid]).min()
                else:
                    eps_cand = 0.0
            else:
                eps_cand = a[src]

            eps_max = min(eps_cand, a[src] * 0.4)
            if eps_max < 1e-14:
                continue

            # Try the move with a safety factor
            for sf in [0.95, 0.7, 0.4, 0.2]:
                eps = eps_max * sf
                if eps < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= eps
                a_trial[dst] += eps
                c_trial = fast_autocorr(a_trial)
                M_trial = c_trial.max()

                if M_trial < best_m - 1e-16:
                    a = a_trial
                    c = c_trial
                    best_m = M_trial
                    best_a = a.copy()
                    stall = 0
                    found = True
                    break

        if not found:
            stall += 1
            # Random perturbation to escape stagnation
            if stall % 30 == 0:
                n_pert = rng.integers(2, 6)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.008)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                c = fast_autocorr(a)
                if c.max() < best_m:
                    best_m = c.max()
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, tau_init=10.0, tau_final=100.0,
                              eta_init=0.5, rng_seed=100):
    """Smoothed projected gradient with temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    eta = eta_init
    stall = 0
    max_stall = 400
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        a_new = project_simplex(a - eta * g_tangent)
        m_new = fast_autocorr(a_new).max()

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.7
            if stall > 200:
                pert = rng.normal(0, 0.003, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a = a_p / a_p.sum()
                stall = 80
        a = a_new
        total_iters += 1

        if total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = 1.0 + 2.0 * x ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick AMD from parent + variants ===
    print(f"\n=== Phase 1: AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.15, 240.0)

    if t_phase1 > 30:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_u_shape(n0, sharpness=2.0))
        labels.append("u2")
        inits.append(make_u_shape(n0, sharpness=3.5))
        labels.append("u3.5")
        inits.append(make_bimodal(n0, width=0.18))
        labels.append("bimodal")

        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, 2.5)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                     restart_every=4000, adaptive=True, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, top_K_init=top_K_init,
                                    top_K_max=top_K_max, K_growth=50, verbose=True)
        try_update(a_out, "lp_main")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord Descent (n={n_best}) ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t3, rng_seed=1, n_peaks=15, n_trials_per_round=40)
        try_update(a_out, "cd1")

    # === Phase 4: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Smooth Gradient (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t4, tau_init=10.0, tau_final=80.0,
                                             eta_init=0.6, rng_seed=100)
        try_update(a_out, "smooth")

    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW r2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, top_K_init=top_K_init,
                                    top_K_max=top_K_max, K_growth=60, verbose=False)
        try_update(a_out, "lp_r2")

    # === Phase 6: Coordinate descent round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord Descent r2 (n={n_best}) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t6, rng_seed=777, n_peaks=20, n_trials_per_round=50)
        try_update(a_out, "cd2")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    trial_lengths = []
    for delta in [500, 1000, -500, 1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.4, 150.0)
        if t7a > 20:
            a_out, _ = amd_optimize(a_long, t7a, top_k=60, beta=6.0, c0=4.0,
                                     restart_every=5000, adaptive=True, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.5, 120.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out, _ = coordinate_descent(a_in, t7b, rng_seed=new_n, n_peaks=15,
                                              n_trials_per_round=30)
                try_update(a_out, f"cd_n{new_n}")

    # === Phase 8: Final intensive coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 8: Final polish (n={n_best}, remaining={remaining:.0f}s) ===")
        t8 = min(remaining * 0.7, 120.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent(a_in, t8, rng_seed=999, n_peaks=25, n_trials_per_round=60)
            try_update(a_out, "cd_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(1.0 + 2.0 * x ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
