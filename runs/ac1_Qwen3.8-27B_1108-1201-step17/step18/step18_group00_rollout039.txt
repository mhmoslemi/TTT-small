def top_gradient(p, c, top_k, beta):
   n=len(p)
   if top_k > len(c): top_k=len(c)
   idx=np.argpartition(c, -top_k)[-top_k:]
   vals=c[idx]
   w=vals**beta
   ws=w.sum()
   if ws<=0: w=np.ones(len(idx))/len(idx)
   else: w/=ws
   G=np.zeros(n)
   for kk, ww in zip(idx, w):
      k=int(kk)
      start=max(0,k-n+1); end=min(n-1,k)

          # p[k-end : k-start+1] reversed
          seg=p[k-end:k-start+1][::-1]
          G[start:end+1] += (2.0*ww)*seg
   return idx,w,G

```python
"""
Search for non-negative step-function heights minimizing

    2 * n * max(conv(a, a)) / sum(a)^2

by combining:
- FFT-based local surrogate evaluation
- dense Frank-Wolfe linearized LP directions (SciPy HiGHS)
- gradient-guided coordinate moves with exact O(n) convolution updates
- exponential mirror descent on selected active convolution peaks
- interpolation to longer sequences and low-dimensional parametric starts

No closures or lambda functions are used; helper functions are top-level.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    _HAS_LINPROG = True
except Exception:
    _HAS_LINPROG = False


def fast_conv(a):
    """Linear convolution of a zero-padded real vector via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return np.ascontiguousarray(out, dtype=np.float64)


def normalize_seq(a):
    """Clip to nonnegative and rescale to sum 1."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.array([], dtype=np.float64)
    if not np.isfinite(a).all():
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def score_fft(a):
    """Score using FFT convolution; scale-invariant."""
    p = normalize_seq(a)
    if p.size == 0:
        return float("inf")
    c = fast_conv(p)
    return float(2.0 * p.size * float(np.max(c)))


def evaluate_safe(a):
    """Call the provided evaluator, if available, without failing the search."""
    try:
        arr = np.asarray(a, dtype=np.float64).reshape(-1)
        vals = [float(x) for x in arr.tolist()]
        v = evaluate_sequence(vals)
        if v is None:
            return float("inf")
        v = float(v)
        if np.isnan(v) or np.isinf(v):
            return float("inf")
        return v
    except Exception:
        return float("inf")


def interpolate_seq(a, m):
    """Resample a probability vector to length m and renormalize."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    m = int(m)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if n == 0:
        return np.ones(m, dtype=np.float64) / m
    if n == m:
        return a.copy()
    p = normalize_seq(a)
    if n == 1:
        return p.copy()
    x = np.linspace(0.0, 1.0, n)
    y = np.linspace(0.0, 1.0, m)
    q = np.interp(y, x, p)
    q = np.maximum(q, 0.0)
    qs = float(q.sum())
    if qs < 1e-15:
        return np.ones(m, dtype=np.float64) / m
    return q / qs


def top_indices(arr, k):
    """Indices of the k largest values."""
    n = int(arr.size)
    k = int(k)
    if n == 0 or k <= 0:
        return np.array([], dtype=int)
    if k >= n:
        return np.arange(n, dtype=int)
    return np.argpartition(arr, n - k)[-k:].astype(int, copy=False)


def select_top_bottom(values, ktop):
    """Return top-ktop and bottom-ktop indices of values."""
    n = int(values.size)
    ktop = int(min(ktop, n))
    if ktop <= 0 or n == 0:
        empty = np.array([], dtype=int)
        return empty, empty
    if ktop >= n:
        allidx = np.arange(n, dtype=int)
        return allidx, allidx
    top = np.argpartition(values, n - ktop)[-ktop:].astype(int, copy=False)
    bot = np.argpartition(values, ktop - 1)[:ktop].astype(int, copy=False)
    return top, bot


def top_gradient(p, c, top_k, beta):
    """
    Approximate subgradient of max_k c_k using top_k coefficients.
    For c_k = sum_i p_i p_{k-i}, derivative is 2 p_{k-i} on the diagonal.
    """
    p = np.asarray(p, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = int(p.size)
    idx = top_indices(c, top_k)
    G = np.zeros(n, dtype=np.float64)
    if idx.size == 0:
        return idx, np.array([], dtype=np.float64), G

    vals = np.maximum(c[idx], 1e-300)
    w = vals ** float(beta)
    ws = float(w.sum())
    if ws <= 0.0 or not np.isfinite(ws):
        w = np.ones(idx.size, dtype=np.float64) / idx.size
    else:
        w = w / ws

    for k, ww in zip(idx, w):
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        if s > e:
            continue
        ii = np.arange(s, e + 1, dtype=np.int64)
        seg = p[k - ii]
        G[s:e + 1] += (2.0 * float(ww)) * seg

    return idx, w, G


def shape_score_direct(shape):
    """Normalize a shape, score it by FFT convolution."""
    p = normalize_seq(shape)
    if p.size == 0:
        return p, float("inf")
    c = fast_conv(p)
    sc = float(2.0 * p.size * float(np.max(c)))
    return p, sc


def consider_shape(shape, best_p, best_sc):
    """Update local best from a candidate shape."""
    p, sc = shape_score_direct(shape)
    if np.isfinite(sc) and sc < best_sc:
        return p, sc
    return best_p, best_sc


def parametric_search_n(n, tlimit, parent_p=None, seed=2025):
    """
    Try many low-dimensional parametric shapes of length n.
    Returns best parameteric sequence and FFT score.
    """
    n = int(n)
    if n <= 0:
        return None, float("inf")
    t_start = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)

    best_p = None
    best_sc = float("inf")

    # Simple monotone / endpoint shapes.
    for alpha in (0.25, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0):
        shape = (1.0 + x) ** alpha
        best_p, best_sc = consider_shape(shape, best_p, best_sc)
        shape = (1.0 + (1.0 - x)) ** alpha
        best_p, best_sc = consider_shape(shape, best_p, best_sc)

    # Exponential ramps.
    for r in (0.8, 1.5, 2.5, 4.0, 7.0, 12.0, 20.0, 35.0):
        shape = np.exp(-r * x)
        best_p, best_sc = consider_shape(shape, best_p, best_sc)
        shape = np.exp(r * (x - 1.0))
        best_p, best_sc = consider_shape(shape, best_p, best_sc)

    # Single first-harmonic shapes, including clipped envelopes and powers.
    phases = np.linspace(0.0, 2.0 * np.pi, 16, endpoint=False)
    for A in (0.0, 0.3, 0.6, 0.9, 1.1, 1.4):
        for ph in phases:
            base = 1.0 + A * np.cos(2.0 * np.pi * x + ph)
            base = np.maximum(base, 0.0)
            if base.max() <= 0.0:
                continue
            for pw in (0.5, 1.0, 1.5, 2.0):
                shape = base ** pw
                best_p, best_sc = consider_shape(shape, best_p, best_sc)

    # First and second/third/fourth harmonic random Fourier-like shapes.
    rng = np.random.default_rng(seed)
    for _ in range(180):
        if time.monotonic() - t_start > max(1.0, tlimit * 0.7):
            break
        a1 = float(rng.uniform(0.3, 1.25))
        ph1 = float(rng.uniform(0.0, 2.0 * np.pi))
        base = 1.0 + a1 * np.cos(2.0 * np.pi * x + ph1)

        if rng.random() < 0.75:
            f2 = int(rng.integers(2, 5))
            a2 = float(rng.uniform(-0.85, 0.85))
            ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
            base = base + a2 * np.cos(2.0 * np.pi * f2 * x + ph2)

        if rng.random() < 0.35:
            f3 = int(rng.integers(3, 6))
            a3 = float(rng.uniform(-0.45, 0.45))
            ph3 = float(rng.uniform(0.0, 2.0 * np.pi))
            base = base + a3 * np.cos(2.0 * np.pi * f3 * x + ph3)

        shape = np.maximum(base, 0.0)
        shape = shape ** float(rng.uniform(0.5, 2.2))
        best_p, best_sc = consider_shape(shape, best_p, best_sc)

    # Asymmetric linear + sinusoid.
    for sl in (-0.8, -0.4, 0.0, 0.4, 0.8):
        for A in (0.4, 0.8, 1.2):
            for ph in (0.0, np.pi / 3.0, np.pi / 2.0, 2.0 * np.pi / 3.0, np.pi):
                base = 1.0 + sl * (x - 0.5) + A * np.cos(2.0 * np.pi * x + ph)
                shape = np.maximum(base, 0.0)
                best_p, best_sc = consider_shape(shape, best_p, best_sc)

    # Blend with parent if available.
    if parent_p is not None:
        ppar = interpolate_seq(parent_p, n)
        if ppar.size == n:
            for w in (0.25, 0.45, 0.65, 0.85):
                shape = w * ppar + (1.0 - w) * (1.0 / float(n))
                best_p, best_sc = consider_shape(shape, best_p, best_sc)
            if n >= 7:
                kern = np.ones(3, dtype=np.float64) / 3.0
                shape = np.convolve(ppar, kern, mode="same")
                best_p, best_sc = consider_shape(shape, best_p, best_sc)

    return best_p, best_sc


def mirror_run(a0, tlimit, top_k=70, beta=3.0, step=0.14, max_step=0.45, min_step=0.01):
    """
    Exponential mirror descent on log probabilities using active-peak gradients.
    Returns best sequence and FFT max-convolution.
    """
    p = normalize_seq(a0)
    n = int(p.size)
    if n == 0:
        return p, float("inf")
    c = fast_conv(p)
    M = float(np.max(c))
    best_p = p.copy()
    best_M = M

    t0 = time.monotonic()
    rng = np.random.default_rng(910)
    eta = float(step)
    stall = 0
    restarts = 0
    it = 0

    while time.monotonic() - t0 < tlimit:
        it += 1

        idx, w, G = top_gradient(p, c, top_k, beta)
        if idx.size == 0:
            break

        gmax = float(np.max(G))
        gmin = float(np.min(G))
        if not np.isfinite(gmax) or not np.isfinite(gmin) or (gmax - gmin) <= 1e-18:
            stall += 1
        else:
            G = G - float(np.mean(G))
            scale = float(np.max(np.abs(G)))
            if scale > 1e-18:
                G = G / scale
            else:
                stall += 1

        if stall < 2000:
            logp = np.log(p + 1e-300)
            best_trial_M = M
            best_trial_p = p
            for mult in (1.0, 0.5, 0.25, 0.1):
                logq = logp - (eta * mult) * G
                mq = float(np.max(logq))
                q = np.exp(logq - mq)
                qs = float(q.sum())
                if qs <= 1e-15:
                    continue
                q = q / qs
                Mq = float(np.max(fast_conv(q)))
                if Mq < best_trial_M:
                    best_trial_M = Mq
                    best_trial_p = q

            if best_trial_M < M - 1e-15:
                p = best_trial_p.copy()
                c = fast_conv(p)
                M = best_trial_M
                stall = 0
                if M < best_M - 1e-15:
                    best_M = M
                    best_p = p.copy()
                eta = min(eta * 1.03, max_step)
            else:
                stall += 1
                eta = max(eta * 0.90, min_step)

        if stall > 250:
            noise_scale = 0.025 / (1.0 + float(restarts))
            p = best_p + rng.normal(0.0, noise_scale, n) * best_p
            p = normalize_seq(p)
            c = fast_conv(p)
            M = float(np.max(c))
            if M < best_M:
                best_M = M
                best_p = p.copy()
            stall = 0
            eta = step
            restarts += 1

    c = fast_conv(best_p)
    best_M = float(np.max(c))
    return best_p, best_M


def coordinate_run(a0, tlimit, top_k=70, beta=3.0, max_select=60):
    """
    Sparse coordinate descent with exact incremental convolution updates.
    Moves mass from high-gradient to low-gradient coordinates, with line search.
    """
    p = normalize_seq(a0)
    n = int(p.size)
    if n == 0:
        return p, float("inf")
    c = fast_conv(p)
    M = float(np.max(c))
    L = c.size

    best_p = p.copy()
    best_M = M

    t0 = time.monotonic()
    rng = np.random.default_rng(1618)
    fractions = (0.003, 0.008, 0.025, 0.08, 0.20)
    min_p = 1e-9
    stall = 0
    restarts = 0
    it = 0
    recompute_every = 35

    while time.monotonic() - t0 < tlimit:
        it += 1

        # Rescale to avoid slow drift from incremental updates.
        s = float(p.sum())
        if s > 1e-20 and abs(s - 1.0) > 1e-8:
            p = p / s
            c = c / (s * s)

        if it % recompute_every == 0:
            c = fast_conv(p)
        M = float(np.max(c))

        if M < best_M - 1e-15:
            best_M = M
            best_p = p.copy()

        idx, w, G = top_gradient(p, c, top_k, beta)
        if idx.size == 0:
            break

        top_idx, bot_idx = select_top_bottom(G, max_select)
        if top_idx.size == 0:
            break

        src_list = []
        for ii in top_idx:
            ii = int(ii)
            if p[ii] > min_p:
                src_list.append(ii)
        if len(src_list) == 0:
            src_list = [int(top_idx[np.argmax(G[top_idx])])]

        dst_list = [int(jj) for jj in bot_idx if int(jj) not in src_list]
        if len(dst_list) == 0:
            dst_list = [int(bot_idx[0])]

        # Find the most promising high->low gradient pair.
        best_diff = -1e100
        bi = src_list[0]
        bj = dst_list[0]
        for ii in src_list:
            gi = float(G[ii])
            for jj in dst_list:
                if ii == jj:
                    continue
                diff = gi - float(G[jj])
                if diff > best_diff:
                    best_diff = diff
                    bi = int(ii)
                    bj = int(jj)

        if best_diff <= 1e-14:
            # Random perturbation candidate to escape plateaus.
            cand = np.flatnonzero(p > min_p)
            if cand.size > 1:
                bi = int(cand[rng.integers(cand.size)])
                bj = int(rng.integers(n))
            if bi == bj:
                bj = (bi + 1) % n

        # Line search over a few move sizes using exact O(n) update.
        pi = float(p[bi])
        if pi <= min_p:
            stall += 1
            continue

        best_trial_M = M
        best_amt = 0.0
        for frac in fractions:
            amt = frac * pi
            if amt <= 1e-18:
                continue
            c_tr = c.copy()
            two_amt = 2.0 * amt
            c_tr[bj:bj + n] += two_amt * p
            c_tr[bi:bi + n] -= two_amt * p
            c_tr[2 * bj] += amt * amt
            c_tr[2 * bi] += amt * amt
            c_tr[bi + bj] -= 2.0 * amt * amt
            mt = float(np.max(c_tr))
            if mt < best_trial_M:
                best_trial_M = mt
                best_amt = amt

        if best_amt > 0.0 and best_trial_M < M - 1e-15:
            amt = float(best_amt)
            two_amt = 2.0 * amt

            c[bj:bj + n] += two_amt * p
            c[bi:bi + n] -= two_amt * p
            c[2 * bj] += amt * amt
            c[2 * bi] += amt * amt
            c[bi + bj] -= 2.0 * amt * amt

            p[bi] -= amt
            p[bj] += amt
            M = float(best_trial_M)
            stall = 0

            if M < best_M - 1e-15:
                best_M = M
                best_p = p.copy()
        else:
            stall += 1

            if stall > 30 and stall % 30 == 0:
                # Small random coordinate move with line search.
                cand = np.flatnonzero(p > min_p)
                if cand.size > 1:
                    rr = int(cand[rng.integers(cand.size)])
                    rrj = int(rng.integers(n))
                    if rr != rrj:
                        ppii = float(p[rr])
                        for frac in (0.003, 0.01, 0.05):
                            amt = frac * ppii
                            if amt <= 1e-18:
                                continue
                            c_tr = c.copy()
                            two_amt = 2.0 * amt
                            c_tr[rrj:rrj + n] += two_amt * p
                            c_tr[rr:rr + n] -= two_amt * p
                            c_tr[2 * rrj] += amt * amt
                            c_tr[2 * rr] += amt * amt
                            c_tr[rr + rrj] -= 2.0 * amt * amt
                            mt = float(np.max(c_tr))
                            if mt < M - 1e-15:
                                c[bj:bj + n] += two_amt * p
                                c[rr:rr + n] -= two_amt * p
                                c[2 * rrj] += amt * amt
                                c[2 * rr] += amt * amt
                                c[rr + rrj] -= 2.0 * amt * amt
                                p[rr] -= amt
                                p[rrj] += amt
                                M = mt
                                stall = 0
                                break

            if stall > 320:
                noise_scale = 0.02 / (1.0 + float(restarts))
                p = best_p + rng.normal(0.0, noise_scale, n) * best_p
                p = normalize_seq(p)
                c = fast_conv(p)
                M = float(np.max(c))
                stall = 0
                restarts += 1

    c = fast_conv(best_p)
    best_M = float(np.max(c))
    return best_p, best_M


def lp_fw_run(a0, tlimit, top_K=220, max_K=560):
    """
    Frank-Wolfe style linearized LP descent.

    At current p, solve:
        min M
        s.t. sum g = 1, g >= 0
             2 (p*g)_k - c_k <= M   for k in active set
    then line-search along p + lambda (g - p).
    """
    p = normalize_seq(a0)
    n = int(p.size)
    if n == 0:
        return p, float("inf")

    c = fast_conv(p)
    M = float(np.max(c))
    best_p = p.copy()
    best_M = M
    t0 = time.monotonic()

    rng = np.random.default_rng(733)
    current_K = int(top_K)
    stall = 0
    max_stall = 35
    it = 0

    cobj = np.zeros(n + 1, dtype=np.float64)
    cobj[-1] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * n + [(None, None)]
    line_values = (0.05, 0.2, 0.5, 0.8, 1.0)

    while time.monotonic() - t0 < tlimit:
        it += 1
        if stall > max_stall:
            # Restart from current best with slight multiplicative noise.
            noise_scale = 0.015 / (1.0 + float(it // 100))
            p = best_p * (1.0 + rng.normal(0.0, noise_scale, n))
            p = normalize_seq(p)
            c = fast_conv(p)
            M = float(np.max(c))
            current_K = int(top_K)
            stall = 0
            continue

        K = min(int(current_K), (2 * n - 1))
        idx = top_indices(c, K)
        if idx.size == 0:
            break

        A = np.zeros((K, n + 1), dtype=np.float64)
        b = np.empty(K, dtype=np.float64)

        for kk, k in enumerate(idx):
            k = int(k)
            s = max(0, k - n + 1)
            e = min(n - 1, k)
            if s <= e:
                ii = np.arange(s, e + 1, dtype=np.int64)
                A[kk, s:e + 1] = 2.0 * p[k - ii]
            b[kk] = c[k]

        A[:, n] = -1.0

        ok = False
        try:
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b,
                A_eq=Aeq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True,
                         "dual_feasibility_tolerance": 1e-8,
                         "primal_feasibility_tolerance": 1e-8}
            )
            if res is not None and res.success and res.x is not None:
                ok = True
        except Exception:
            ok = False

        if not ok:
            stall += 1
            continue

        g = np.asarray(res.x[:n], dtype=np.float64)
        M_lin = float(res.x[-1])
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs <= 1e-12:
            stall += 1
            continue
        g = g / gs

        # If all active first-order increases are nonnegative, small moves may still help
        # due to nonconvex curvature, so try a few lambda values.
        best_trial_M = M
        best_trial_p = p
        for lam in line_values:
            q = (1.0 - lam) * p + lam * g
            q = np.maximum(q, 0.0)
            qs = float(q.sum())
            if qs <= 1e-12:
                continue
            q = q / qs
            Mq = float(np.max(fast_conv(q)))
            if Mq < best_trial_M:
                best_trial_M = Mq
                best_trial_p = q

        if best_trial_M < M - 1e-15:
            p = best_trial_p.copy()
            c = fast_conv(p)
            M = best_trial_M
            stall = 0
            if M < best_M - 1e-15:
                best_M = M
                best_p = p.copy()
            if it % 15 == 0 and current_K < max_K:
                current_K = min(int(max_K), int(current_K) + 25)
            if (it % 25) == 0:
                print("    LP-FW it=%d: FFT_score=%.10f, K=%d" %
                      (it, 2.0 * n * M, K), flush=True)
        else:
            stall += 1
            if stall % 5 == 0 and current_K < max_K:
                current_K = min(int(max_K), int(current_K) + 25)

    c = fast_conv(best_p)
    best_M = float(np.max(c))
    return best_p, best_M


def try_update_candidate(best_p, best_score, pool, cand, label, t_global):
    """
    Consider a candidate sequence. Updates best and a small pool of near-best
    candidates for final exact validation.
    """
    if cand is None:
        return best_p, best_score, pool
    try:
        arr = np.asarray(cand, dtype=np.float64).reshape(-1)
    except Exception:
        return best_p, best_score, pool
    if arr.size == 0:
        return best_p, best_score, pool

    p = normalize_seq(arr)
    n = int(p.size)
    if n == 0:
        return best_p, best_score, pool

    c = fast_conv(p)
    sc = float(2.0 * n * float(np.max(c)))
    if not np.isfinite(sc):
        return best_p, best_score, pool

    improved = sc < best_score
    if improved:
        best_score = sc
        best_p = p.copy()
        elapsed = time.monotonic() - t_global
        print("NEW BEST [%s]: %.10f, n=%d, t=%.1fs" % (label, sc, n, elapsed), flush=True)

    # Keep a few close candidates so final exact eval can pick robustly.
    if sc <= best_score + 0.008 and len(pool) < 10:
        pool.append(p.copy())

    return best_p, best_score, pool


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    budget = min(float(budget_s) - 60.0, 1740.0)
    if budget < 60.0:
        budget = 60.0

    best_p = None
    best_score = float("inf")
    pool = []

    # Use previously discovered construction if present and valid.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and float(p.sum()) > 1e-10:
            best_p = normalize_seq(p)
            best_score = score_fft(best_p)
            print("Parent: n=%d, FFT_score=%.10f" % (best_p.size, best_score), flush=True)
    except Exception:
        best_p = None

    # Safe fallback that is still more informative than pure uniform.
    if best_p is None:
        n_fallback = 7000
        x = np.linspace(0.0, 1.0, n_fallback)
        init_shape = np.maximum(1.0 + 0.8 * np.cos(2.0 * np.pi * x + np.pi / 3.0), 0.0)
        best_p = normalize_seq(init_shape)
        best_score = score_fft(best_p)
        print("Fallback init: n=%d, FFT_score=%.10f" % (best_p.size, best_score), flush=True)

    pool.append(best_p.copy())
    base_n = int(best_p.size)

    # Quick length tests: longer sequences often lower finite-resolution effect.
    length_targets = (7800, 9000, 10500, 12000)
    for nn in length_targets:
        remaining = budget - (time.monotonic() - t_global)
        if remaining < 45:
            break
        if nn <= base_n or nn > 12000:
            continue
        cand = interpolate_seq(best_p, nn)
        best_p, best_score, pool = try_update_candidate(best_p, best_score, pool, cand,
                                                        "interp_%d" % nn, t_global)

    # Parametric search at a moderate length and the current best length.
    n_param1 = min(10000, max(7000, base_n))
    n_param2 = int(best_p.size)
    for nn, seed in ((n_param1, 12345), (n_param2, 54321)):
        remaining = budget - (time.monotonic() - t_global)
        if remaining < 80 or nn <= 0:
            break
        t_param = min(90.0, remaining * 0.12)
        p_par, sc_par = parametric_search_n(nn, t_param, parent_p=best_p, seed=seed)
        best_p, best_score, pool = try_update_candidate(best_p, best_score, pool, p_par,
                                                        "param_n%d" % nn, t_global)

    # LP-FW on a manageable length. If global best is too long, interpolate down.
    if _HAS_LINPROG:
        remaining = budget - (time.monotonic() - t_global)
        if remaining > 180:
            n_lp = min(10000, max(5000, int(best_p.size)))
            p_lp_in = interpolate_seq(best_p, n_lp)
            t_lp = min(370.0, remaining * 0.30)
            p_out, m_out = lp_fw_run(p_lp_in, t_lp, top_K=220, max_K=560)
            best_p, best_score, pool = try_update_candidate(best_p, best_score, pool, p_out,
                                                            "lp_fw_n%d" % n_lp, t_global)

    # Coordinate descent on the global best sequence.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 120:
        t_cd = min(320.0, remaining * 0.25)
        p_out, m_out = coordinate_run(best_p, t_cd, top_k=75, beta=3.0, max_select=65)
        best_p, best_score, pool = try_update_candidate(best_p, best_score, pool, p_out,
                                                        "coord_global", t_global)

    # Mirror descent can move the shape more globally.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 70:
        t_mir = min(140.0, remaining * 0.12)
        p_out, m_out = mirror_run(best_p, t_mir, top_k=70, beta=3.0, step=0.14)
        best_p, best_score, pool = try_update_candidate(best_p, best_score, pool, p_out,
                                                        "mirror_global", t_global)

    # One more round of longer-sequence refinement, capped for evaluator runtime.
    for nn in (10000, 11000, 12000):
        remaining = budget - (time.monotonic() - t_global)
        if remaining < 180 or nn > 12000:
            break
        q = interpolate_seq(best_p, nn)
        best_p, best_score, pool = try_update_candidate(best_p, best_score, pool, q,
                                                        "scale_%d" % nn, t_global)
        t_cd = min(180.0, remaining * 0.16)
        if t_cd > 30:
            p_out, m_out = coordinate_run(q, t_cd, top_k=70, beta=3.0, max_select=60)
            best_p, best_score, pool = try_update_candidate(best_p, best_score, pool, p_out,
                                                            "coord_n%d" % nn, t_global)

    # Final polish on the best sequence found so far.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 60:
        t_final_cd = min(260.0, remaining * 0.50)
        p_out, m_out = coordinate_run(best_p, t_final_cd, top_k=85, beta=3.0, max_select=80)
        best_p, best_score, pool = try_update_candidate(best_p, best_score, pool, p_out,
                                                        "final_coord", t_global)

    # Exact validation with the actual evaluator for a few near-best candidates.
    final_best_p = best_p.copy()
    final_best_score = best_score

    candidates = []
    if pool:
        # Prefer recent near-best candidates, but never evaluate too many.
        for cand in pool[-4:]:
            if cand is not None and cand.size > 0:
                candidates.append(cand)
    candidates.append(best_p)

    seen_lens = set()
    for cand in candidates:
        if len(candidates) > 5:
            break
        key = int(cand.size)
        if key in seen_lens:
            continue
        seen_lens.add(key)
        try:
            sc_exact = evaluate_safe(cand)
        except Exception:
            sc_exact = float("inf")
        if np.isfinite(sc_exact) and sc_exact < final_best_score:
            final_best_score = sc_exact
            final_best_p = cand.copy()

    # If something failed, return a guaranteed simple sequence.
    if final_best_p is None or final_best_p.size == 0 or not np.isfinite(final_best_score):
        n_fb = 6000
        x = np.linspace(0.0, 1.0, n_fb)
        final_best_p = normalize_seq(np.maximum(1.0 + 0.75 * np.cos(2.0 * np.pi * x), 0.0))
        final_best_score = evaluate_safe(final_best_p)

    elapsed = time.monotonic() - t_global
    exact_final = evaluate_safe(final_best_p)
    if np.isfinite(exact_final):
        print("FINAL exact: %.10f, n=%d, elapsed=%.1fs" %
              (exact_final, final_best_p.size, elapsed), flush=True)
    else:
        print("FINAL exact failed; FFT score=%.10f, n=%d" %
              (final_best_score, final_best_p.size), flush=True)

    return [float(x) for x in final_best_p.tolist()]
```
