"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Reduced-constraint LP-based iterative method (active set approach)
2. FTRL mirror descent with adapted subgradients
3. Spectral domain power spectrum shaping
4. Multi-start with diverse initializations and adaptive length selection
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_from_conv(n, c):
    """Compute objective 2*n*max(c) assuming sum(a)=1."""
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely evaluate sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def subgradient_top(a, c, top_k=20, beta=40.0):
    """Compute weighted subgradient from top peaks."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()

    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def ftrl_optimize(a0, time_limit, verbose=False):
    """FTRL mirror descent with adaptive subgradients."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    c0 = 1.5
    T_init = 0.5
    T_min = 0.02
    top_k_init = 40
    top_k_min = 5
    beta_init = 15.0
    beta_max = 80.0

    x = a.copy()
    eta_schedule = 'sqrt'

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, t / 30000.0)
        cur_top_k = max(top_k_min, int(top_k_init * (1 - 0.85 * progress) + top_k_min))
        cur_beta = beta_init + (beta_max - beta_init) * progress

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 3.0)
        y = x + beta_t * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Subgradient
        c = fast_conv(y)
        g = subgradient_top(y, c, top_k=cur_top_k, beta=cur_beta)

        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)

        # Mirror descent update (multiplicative weights)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x = a_new
        a = a_new

        # Track best
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()

        # Periodic spectral smoothing
        if t % 3000 == 0:
            a = spectral_smooth(a, strength=0.2)
            c_check = fast_conv(a)
            if c_check.max() < best_m:
                best_m = c_check.max()
                best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): score={2*n*best_m:.6f}")

    return best_a, best_m


def spectral_smooth(a, strength=0.3):
    """Spectral flattening: compress dynamic range of amplitude spectrum."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a

    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

    a_out = (1.0 - strength * 0.4) * a + strength * 0.4 * a_new
    return norm_simplex(a_out)


def lp_iteration_reduced(a, threshold_frac=0.85, max_constraints=100):
    """LP iteration with reduced constraints (only peaks above threshold).
    Given fixed 'a', find b maximizing sum(b) subject to conv(a,b)_k <= M for active peaks.
    """
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    if M < 1e-15:
        return a.copy(), M

    # Select active constraints: peaks above threshold
    threshold = threshold_frac * M
    active_idx = np.where(c >= threshold)[0]
    if len(active_idx) > max_constraints:
        # Keep top max_constraints
        top_order = c[active_idx].argsort()[::-1][:max_constraints]
        active_idx = active_idx[top_order]

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []

    for idx, k in enumerate(active_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            rows.append(idx)
            cols.append(j)
            vals.append(a[i])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), M)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs',
                      options={'presolve': True})
    except Exception:
        return a.copy(), M

    if res.status != 0 or res.x is None:
        return a.copy(), M

    b = np.maximum(res.x, 0.0)
    s = b.sum()
    if s < 1e-15:
        return a.copy(), M
    return b / s, M


def lp_optimize(a0, time_limit, blend=0.5, verbose=True):
    """Iterative LP-based optimization with reduced constraints."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    stall_count = 0
    max_stall = 50
    threshold = 0.85

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        if stall_count > max_stall:
            break

        t_start = time.monotonic()
        b, M_curr = lp_iteration_reduced(a, threshold_frac=threshold)
        t_lp = time.monotonic() - t_start

        if t_lp > time_limit * 0.5:
            # LP too slow, reduce constraints
            threshold = max(0.95, threshold + 0.02)
            continue

        # Blend
        t_blend = blend * (1.0 - 0.3 * min(1.0, it / 100.0))
        a_new = (1.0 - t_blend) * a + t_blend * b
        a_new = norm_simplex(a_new)

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a_new.copy()
            stall_count = 0
            # Slowly tighten threshold
            threshold = max(0.7, threshold - 0.005)
        else:
            stall_count += 1
            # Loosen threshold to escape
            threshold = min(0.95, threshold + 0.01)
            # Try smaller blend
            t_blend = t_blend * 0.5
            a_new = (1.0 - t_blend) * a + t_blend * b
            a_new = norm_simplex(a_new)
            M_new2 = fast_conv(a_new).max()
            if M_new2 < best_m - 1e-12:
                best_m = M_new2
                best_a = a_new.copy()
                stall_count = 0

        a = a_new

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it} ({elapsed:.0f}s, LP={t_lp:.2f}s): "
                  f"score={2*n*best_m:.6f}, thr={threshold:.3f}, stall={stall_count}")

    return best_a, best_m


def lp_ftrl_hybrid(a0, time_limit, verbose=True):
    """Alternate between LP steps and FTRL steps."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    for phase in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        # LP step (short time budget)
        lp_budget = min(30.0, (time_limit - (time.monotonic() - t0)) * 0.4)
        if lp_budget > 5:
            a_lp, _ = lp_optimize(a, lp_budget, blend=0.4, verbose=False)
            M_lp = fast_conv(a_lp).max()
            if M_lp < best_m:
                best_m = M_lp
                best_a = a_lp.copy()
                a = a_lp

        # FTRL step (short time budget)
        ftrl_budget = min(30.0, (time_limit - (time.monotonic() - t0)) * 0.4)
        if ftrl_budget > 5:
            a_f, _ = ftrl_optimize(a, ftrl_budget, verbose=False)
            M_f = fast_conv(a_f).max()
            if M_f < best_m:
                best_m = M_f
                best_a = a_f.copy()
                a = a_f

        if verbose and phase % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hybrid phase {phase} ({elapsed:.0f}s): score={2*n*best_m:.6f}")

    return best_a, best_m


def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    step = 0.25
    for it in range(50000):

            step = min(0.4, step * 1.02)

            if step < 0.05:

    return best_a, best_m


def make_init_flat(n):
    return np.ones(n) / n


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.12):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.03
    return norm_simplex(a)


def make_init_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_rand_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.1 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        try_update(a_short, f"interp_short_n{new_n}")

        t4s = min(remaining * 0.15, 80.0)
        if t4s > 15:
            a_out, _ = ftrl_optimize(a_short, t4s * 0.6, verbose=False)
            try_update(a_out, f"ftrl_short_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_lp, _ = lp_optimize(a_out, min(t4s * 0.4, remaining * 0.4), blend=0.4, verbose=False)
                try_update(a_lp, f"lp_short_n{new_n}")

    # === Phase 5: Final intensive optimization on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final intensive (n={n_best}) ===")

    if remaining > 60:
        # Intensive LP
        t5 = min(remaining * 0.35, 250.0)
        if t5 > 20:
            a_in = norm_simplex(best_seq)
            a_out, _ = lp_optimize(a_in, t5, blend=0.3, verbose=True)
            try_update(a_out, f"lp_final_n{n_best}")

        # FTRL polish
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.25, 150.0)
        if t5b > 10:
            a_in = norm_simplex(best_seq)
            a_out, _ = ftrl_optimize(a_in, t5b, verbose=False)
            try_update(a_out, f"ftrl_final_n{n_best}")

        # Spectral polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_in = norm_simplex(best_seq)
            a_out, _ = spectral_optimize(a_in, min(30.0, remaining * 0.3), verbose=False)
            try_update(a_out, f"spec_final_n{n_best}")

        # One more LP with tighter threshold
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_simplex(best_seq)
            a_out, _ = lp_optimize(a_in, min(remaining * 0.4, 80.0), blend=0.2, verbose=False)
            try_update(a_out, f"lp_tight_n{n_best}")

    # === Phase 6: Final FTRL with perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 6: Perturbed restart (n={n_best}) ===")
        a_in = norm_simplex(best_seq)
        rng = np.random.default_rng(2024)
        for pert_scale in [0.01, 0.03, 0.05]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 10:
                break
            perturbed = norm_simplex(a_in + rng.normal(0, pert_scale, n_best))
            t6 = min(remaining * 0.3, 80.0)
            if t6 < 5:
                break
            a_out, _ = ftrl_optimize(perturbed, t6, verbose=False)
            try_update(a_out, f"pert_{pert_scale}_n{n_best}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_lp, _ = lp_optimize(a_out, min(remaining * 0.3, 40.0), blend=0.3, verbose=False)
                try_update(a_lp, f"lp_pert_{pert_scale}_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
