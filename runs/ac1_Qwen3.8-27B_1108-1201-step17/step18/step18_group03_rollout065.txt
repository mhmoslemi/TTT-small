def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    return best_a, best_val

def _spectral_flattening(a0, time_limit):
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    fft_size = 1 << (2 * n - 2).bit_length()
    learning_rate = 0.3
    stall = 0
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean() + 1e-30
        beta_spec = 3.0 + t * 0.0005
        P_dev = (P - P_mean) / P_mean
        w_spec = np.exp(beta_spec * np.abs(P_dev))
        w_spec = w_spec / w_spec.sum()
        grad_freq = 2.0 * A * w_spec / max(P.max(), 1e-30)
        grad_time = np.fft.irfft(grad_freq, fft_size)[:n]
        lr = learning_rate / (1.0 + np.sqrt(t))
        a_new = a * np.exp(-lr * grad_time)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        M_new = _fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                a = _norm_sum1(best_a + rng.normal(0, 0.01, n) * best_a)
                stall = 0
                learning_rate *= 0.6
                if learning_rate < 0.01:
                    break
    return best_a, best_m

    n_total = 50000
    for step in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break
        progress = step / n_total
        T = T_start * (T_end / T_start) ** progress
        a_trial = a.copy()
        for _ in range(int(rng.integers(1, 5))):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = a_trial[i] * float(rng.uniform(0.001, 0.02))
                a_trial[i] -= amt
                a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        if a_trial.sum() < 1e-15:
            continue
        a_trial = a_trial / a_trial.sum()
        M_trial = _fast_autocorr(a_trial).max()
        M_curr = _fast_autocorr(a).max()
        delta = M_trial - M_curr
        if delta < 0 or np.random.random() < np.exp(-delta / (T * M_curr + 1e-15)):
            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a_trial.copy()
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = _get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = _norm_sum1(p)
            p_score = _evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        best_seq = _norm_sum1((np.arange(n_base, dtype=np.float64) + 1.0) ** 0.7)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def _try(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 0.01:
            return
        a_n = a / a.sum()
        sc = _score_norm(a_n)
        if np.isfinite(sc) and sc < best_score:
            sc_real = _evaluate_safe(a_n)
            if np.isfinite(sc_real) and sc_real < best_score:
                best_score = sc_real
                best_seq = a_n.copy()
                print(f"  *** NEW BEST [{label}]: {sc_real:.6f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    # Phase 1: parametric init
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Parametric (n={n0}) ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 8:
        out, _ = _parametric_search(n0, t1, best_seq, 2.0 * n0 * _fast_autocorr(best_seq).max())
        _try(out, "parametric")

    # Phase 2: LP-FW main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.28, 480.0)
    if t2 > 30 and n_best <= 7000:
        out, _ = _lp_frank_wolfe(_norm_sum1(best_seq), t2, solver, initial_K=400, max_K=2500, add_per_iter=80)
        _try(out, f"lp_n{n_best}")

    # Phase 3: smooth mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smooth MD (n={n_best}) ===")
    t3 = min(remaining * 0.14, 240.0)
    if t3 > 20:
        out, _ = _smooth_mirror_descent(_norm_sum1(best_seq), t3, top_k=100, beta_start=4.0, c0=5.0, restart_every=2500)
        _try(out, f"smd_n{n_best}")

    # Phase 4: LP-FW pass 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW pass2 (n={n_best}) ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 30 and n_best <= 7000:
        out, _ = _lp_frank_wolfe(_norm_sum1(best_seq), t4, solver, initial_K=600, max_K=3000, add_per_iter=100)
        _try(out, f"lp2_n{n_best}")

    # Phase 5: coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: CoordDesc (n={n_best}) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 15:
        out, _ = _coordinate_descent(_norm_sum1(best_seq), t5, rng_seed=77, verbose=True)
        _try(out, f"coord_n{n_best}")

    # Phase 6: spectral flattening
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Spectral (n={n_best}) ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 15:
        out, _ = _spectral_flattening(_norm_sum1(best_seq), t6)
        _try(out, f"spec_n{n_best}")

    # Phase 7: length exploration (current best length only as anchor, plus neighbors)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (anchor n={n_best}) ===")
    cands = sorted(set([n_best + d for d in [-500, 500, -1000, 1000, 1500] if 1500 <= n_best + d <= 6500]))
    for new_n in cands[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 70:
            break
        print(f"  n={new_n} (remaining={remaining:.0f}s)")
        a_long = _interpolate_sequence(best_seq, new_n)
        tt = min(remaining * 0.4, 90.0)
        if tt > 12:
            out, _ = _smooth_mirror_descent(a_long, tt, top_k=80, beta_start=4.0, c0=4.0, restart_every=2000)
            _try(out, f"smd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 5500:
            tt = min(remaining * 0.5, 120.0)
            if tt > 20:
                out, _ = _lp_frank_wolfe(a_long, tt, solver, initial_K=300, max_K=2000, add_per_iter=60)
                _try(out, f"lp_n{new_n}")

    # Phase 8: final LP-FW polish on the global best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP-FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 6500:
        t8 = min(remaining * 0.4, 180.0)
        if t8 > 20:
            out, _ = _lp_frank_wolfe(_norm_sum1(best_seq), t8, solver, initial_K=700, max_K=3200, add_per_iter=120)
            _try(out, f"lpf_n{n_best}")

    # Phase 9: annealing micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t9 = min(remaining * 0.5, 90.0)
        if t9 > 5:
            out, _ = _annealing_polish(_norm_sum1(best_seq), t9, rng_seed=314)
            _try(out, f"anneal_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = _norm_sum1(best_seq)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        xa = np.linspace(0, 1, n_fb)
        best_seq = _norm_sum1(np.sin(np.pi * xa) ** 2)
        best_score = _evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
