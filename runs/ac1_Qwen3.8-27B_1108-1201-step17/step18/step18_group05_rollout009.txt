    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = a ** 2
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=6000,
                 adaptive_beta=True, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if adaptive_beta and stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)
            cur_top_k = min(cur_top_k + 5, len(ac))

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 1.0, 2.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def build_conv_matrix(a, k_indices):
    """Build the matrix T where T[ii,j] = a[k_indices[ii] - j] if valid, else 0."""
    n = len(a)
    K = len(k_indices)
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(k_indices):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            v = a[ki - j]
            if v > 1e-18:
                rows.append(ii)
                cols.append(j)
                vals.append(v)
    if len(vals) == 0:
        return np.zeros((K, n))
    from scipy.sparse import coo_matrix
    T = coo_matrix((vals, (rows, cols)), shape=(K, n)).tocsr()
    return T


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=300,
                    top_K_max=None, use_all=False, shrinkage=0.1):
    """Improved LP-based Frank-Wolfe with shrinkage and better line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T = build_conv_matrix(a, binding_idx)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Improved line search: more trial points including very small steps
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        alpha_candidates = sorted(set(
            [base_obj * 0.3, base_obj * 0.5, base_obj * 0.7,
             best_alpha, best_alpha * 0.5, best_alpha * 1.5,
             min(1.0, best_alpha * 2.0), 0.1, 0.05, 0.02, 0.01,
             0.3, 0.2, 0.15, 0.08]
        ))
        alpha_candidates = [al for al in alpha_candidates if 0 < al <= 1.0]

        for alpha in alpha_candidates:
            # Shrinkage variant: move toward both g_dir and best_a
            if shrinkage > 0 and iter_count > 10:
                a_trial = (1.0 - alpha * (1 - shrinkage)) * a + alpha * (1 - shrinkage) * g_dir + alpha * shrinkage * best_a
            else:
                a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        # Apply shrinkage update
        if shrinkage > 0 and iter_count > 10:
            a = (1.0 - best_alpha * (1 - shrinkage)) * a + best_alpha * (1 - shrinkage) * g_dir + best_alpha * shrinkage * best_a
        else:
            a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        # Adapt K
        if iter_count % 8 == 0:
            K = min(K + 40, top_K_max)

        # Increase shrinkage over time for stability
        cur_shrink = min(shrinkage + iter_count * 0.0005, 0.3)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def peak_shaving_lp(a0, time_limit, solver, verbose=False):
    """Targeted LP that focuses on reducing the top few peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    num_peaks = min(10, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Focus on top peaks with a margin
        top_idx = np.argsort(ac)[::-1][:num_peaks]
        # Use a slightly relaxed threshold to include near-peak constraints
        threshold = M_curr * 0.9
        active_idx = np.where(ac >= threshold)[0]
        if len(active_idx) < num_peaks:
            active_idx = np.concatenate([active_idx, top_idx[:num_peaks]])
            active_idx = np.unique(active_idx)

        T = build_conv_matrix(a, active_idx)
        b_var = cp.Variable(n, nonneg=True)
        # Allow slight violation to make LP feasible
        constraints = [T @ b_var <= M_curr * 1.001]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        g_dir = b_val / b_sum

        # Small step for peak shaving
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m - 1e-14:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                break

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def smoothed_gradient_optimize(a0, time_limit, verbose=False, tau_init=15.0, tau_final=200.0,
                               eta_init=0.5, rng_seed=100):
    """Smoothed projected gradient with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    max_stall = 400
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit * 0.8, 600.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        # Adagrad-like step
        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 2.0)
        else:
            stall += 1
            if stall > 50:
                eta *= 0.85
            if stall > 150:
                pert = rng.normal(0, 0.004, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a = a_p / a_p.sum()
                stall = 80
            else:
                a = a_new

        total_iters += 1

        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Focused coordinate descent targeting the global peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Focus on the top 5 peaks
        top_k = np.argsort(ac)[::-1][:5]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Find low-impact destination
            # Compute "sensitivity" at each position
            sens = np.zeros(n)
            for kk in top_k:
                ki2 = int(kk)
                s2 = max(0, ki2 - n + 1)
                e2 = min(n - 1, ki2)
                idx2 = np.arange(s2, e2 + 1)
                sens[idx2] += a[ki2 - idx2] * a[idx2]

            # Try moving to low-sensitivity positions
            low_sens = np.argsort(sens)[:20]
            candidates = [j for j in low_sens if j != src and sens[j] < sens[src] * 0.5]

            found = False
            for dst in candidates[:5]:
                if found:
                    break
                for frac in [0.005, 0.01, 0.003, 0.02, 0.002]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 20:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with time-budgeted phases."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-start AMD ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.15, 250.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3.0")
        inits.append(make_bimodal(n0, width=0.2))
        labels.append("bimodal")
        inits.append(make_cosine(n0))
        labels.append("cosine")
        inits.append(spectral_init(n0, n_modes=30, seed=42))
        labels.append("spectral")
        inits.append(spectral_init(n0, n_modes=50, seed=99))
        labels.append("spectral2")

        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, 2.0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                     restart_every=3000, adaptive_beta=True,
                                     verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: Main LP-FW (most important) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Main LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 40:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all, shrinkage=0.1)
        try_update(a_out, "lp_main")

    # === Phase 3: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smoothed gradient (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_gradient_optimize(a_in, t3, verbose=True,
                                               tau_init=10.0, tau_final=200.0,
                                               eta_init=0.6, rng_seed=100)
        try_update(a_out, "smooth1")

    # === Phase 4: LP-FW round 2 (higher K) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW r2 (n={n_best}) ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(800, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all, shrinkage=0.15)
        try_update(a_out, "lp_r2")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak shaving (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving_lp(a_in, t5, solver, verbose=True)
        try_update(a_out, "peakshave")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 200, -200]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.3, 120.0)
        if t7a > 20:
            a_out, _ = lp_frank_wolfe(a_long, t7a, solver, verbose=False,
                                       top_K_init=min(400, 2*new_n-1),
                                       top_K_max=min(2000, 2*new_n-1),
                                       use_all=(new_n <= 1500), shrinkage=0.1)
            try_update(a_out, f"lp_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.3, 80.0)
            if t7b > 15:
                a_out, _ = coordinate_polish(a_long, t7b, verbose=False, rng_seed=new_n % 1000)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 8: Final optimization cycle ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final optimization (n={n_best}) ===")

    if remaining > 60:
        t8a = min(remaining * 0.4, 200.0)
        if t8a > 30:
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 1500)
            top_K_init = min(1000, 2 * n_best - 1)
            top_K_max = min(6000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t8a, solver, verbose=False,
                                       top_K_init=top_K_init, top_K_max=top_K_max,
                                       use_all=use_all, shrinkage=0.2)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t8b = min(remaining * 0.4, 100.0)
            if t8b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = peak_shaving_lp(a_in, t8b, solver, verbose=False)
                try_update(a_out, "peakshave_final")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
        t9 = min(remaining * 0.6, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
