"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth L-BFGS-B optimization with log-sum-exp surrogate (softmax parameterization)
2. LP-based Frank-Wolfe with golden-section line search and adaptive constraints
3. Multi-phase annealing (beta schedule in smooth phase)
4. Coordinate descent polish
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize, minimize_scalar

# ---------- Top-level helpers ----------

def _fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _eval_obj(a):
    b = _fast_conv(a)
    return 2.0 * len(a) * b.max()

def _evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def _get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            pass
    return cp.ECOS

def _interpolate(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_int = np.interp(new_x, old_x, f)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_int / ss

# ---------- Smooth optimizer ----------

def _softmax(x):
    x_shift = x - x.max()
    ex = np.exp(x_shift)
    return ex / ex.sum()

def _smooth_obj_grad(x, beta, n, use_fft=True):
    """Compute logsumexp(beta*b)/beta and its gradient w.r.t. x (softmax params)."""
    a = _softmax(x)
    b = _fast_conv(a)
    b_shift = b - b.max()
    exp_b = np.exp(beta * b_shift)
    logsum = np.log(exp_b.sum())
    val = b.max() + logsum / beta

    p = exp_b / exp_b.sum()

    if use_fft:
        N = 1 << (3 * n - 2).bit_length()
        Af = np.fft.rfft(a, N)
        Pf = np.fft.rfft(p, N)
        cross = np.fft.irfft(Af * Pf.conj(), N)
        grad_a = 2.0 * cross[:n]
    else:
        grad_a = np.zeros(n)
        K = min(200, len(p))
        top_idx = np.argsort(p)[::-1][:K]
        for kk in top_idx:
            ki = int(kk)
            pk = p[ki]
            if pk < 1e-14:
                continue
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                indices = ki - np.arange(i_start, i_end + 1)
                grad_a[i_start:i_end + 1] += 2.0 * pk * a[indices]

    va = grad_a * a
    dval_dx = va - a * va.sum()
    return val, dval_dx

def _smooth_optimize(a0, time_limit, beta_schedule, verbose=True):
    """Run L-BFGS-B with annealed beta on softmax parameterization."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_val = _eval_obj(a)
    t0 = time.monotonic()

    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        x = np.where(a > 1e-15, np.log(a + 1e-30), -15.0)

        def _obj(x_in):
            return _smooth_obj_grad(x_in, beta, n)[0]

        def _objg(x_in):
            return _smooth_obj_grad(x_in, beta, n)

        phase_limit = min(300.0, time_limit - (time.monotonic() - t0))
        try:
            result = minimize(_objg, x, method='L-BFGS-B',
                              options={'maxiter': 300, 'maxfun': 10000,
                                       'ftol': 1e-14, 'gtol': 1e-10})
        except Exception:
            continue

        if result.fun < _obj(x):
            a_new = _softmax(result.x)
            v = _eval_obj(a_new)
            if v < best_val:
                best_val = v
                best_a = a_new.copy()
                a = a_new
        if time.monotonic() - t0 > time_limit:
            break

    if verbose:
        print(f"  Smooth done ({time.monotonic()-t0:.1f}s): J={best_val:.8f}")
    return best_a, best_val

# ---------- LP-FW with golden section line search ----------

def _build_constraint_rows(y, constraint_list, n):
    K = len(constraint_list)
    rows = []
    for k in constraint_list:
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n - 1, ki)
        if i_start > i_end:
            rows.append((np.array([], dtype=np.int32), np.array([], dtype=np.float64)))
            continue
        idx = np.arange(i_start, i_end + 1, dtype=np.int32)
        vals = y[ki - idx].astype(np.float64)
        rows.append((idx, vals))
    return rows

def _lp_fw(a0, time_limit, solver, initial_K=300, max_K=1800,
           add_per_iter=30, verbose=True):
    """LP-Frank-Wolfe with golden-section line search."""
    n = len(a0)
    y = _normalize(a0)
    best_a = y.copy()
    best_m = _fast_conv(y).max()
    t0 = time.monotonic()

    ac_y = _fast_conv(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 1200
    iter_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print(f"    LP-FW: stall {no_improve}, stop at iter {t}")
            break

        ac = _fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix (dense for simplicity)
        coeffs = np.zeros((K, n), dtype=np.float64)
        for jj, k in enumerate(constraint_list):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                coeffs[jj, idx] = y[ki - idx]

        # Solve LP: min M s.t. sum(g)=1, g>=0, coeffs@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(k) for k in constraint_list[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for kk in top_v:
                active_set.add(int(kk))

        # Golden section line search
        if t == 1:
            y_new = g_dir.copy()
        else:
            def _line_obj(alpha):
                trial = (1.0 - alpha) * y + alpha * g_dir
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    return 1e10
                trial = trial / s_t
                return _fast_conv(trial).max()

            res = minimize_scalar(_line_obj, bounds=(1e-5, 0.8),
                                  method='bounded',
                                  options={'xatol': 0.005})
            best_alpha = float(res.x) if res.fun < 1e10 else 2.0 / (t + 2.0)
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
            y = y_new

        iter_count += 1
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.8f}, |S|={len(active_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.8f}, iters={iter_count}")
    return best_a, best_m

def _coord_polish(a0, time_limit, rng_seed=42, verbose=True):
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_conv(a)
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-impact destination
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            candidates = np.where(scores < scores.max() * 0.3)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                continue

            for frac in [0.003, 0.006, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = _fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                a[i] -= a[i] * frac
                a[j] += a[i] * frac
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = _fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.8f}, stall={stall}")

    return best_a, best_m

        a_out, _ = _smooth_optimize(best_seq, t1, [20.0, 50.0], verbose=True)
        sc = _evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  P1 new best: {sc:.8f}")

    # --- Phase 2: LP-FW main ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW n={n_best} ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 60 and n_best <= 8500:
        a_out, _ = _lp_fw(best_seq, t2, solver, initial_K=350, max_K=2000,

        sc = _evaluate_safe(a_out)

            best_seq = a_out

    # --- Phase 3: Smooth refinement ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smooth (beta=100,200) n={n_best} ===")

    print(f"\n=== Phase 6: Coord polish n={n_best} ===")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    for new_n in [n_best + 500, n_best + 1200]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:

        if new_n > 10000:
            break
        print(f"\n=== Phase 7: Scale to n={new_n} ===")

        if t8b > 10:

                print(f"  P8 final: {sc:.8f}")

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    else:
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        best_seq = _normalize(0.5 + 0.5 * np.cos(np.pi * x))

    elapsed = time.monotonic() - t_global
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.0f}s")
    return [float(v) for v in best_seq]
