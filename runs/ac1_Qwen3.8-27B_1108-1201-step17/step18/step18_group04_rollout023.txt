- The gradient tells us the direction of steepest increase
- We move in the opposite direction (decrease high-gradient, increase low-gradient)
- The step size is found by line search

```python
def frank_wolfe_step(a, K=5, max_line_search=20):
    n = len(a)
    conv = fast_conv(a)
    peaks = np.argsort(conv)[::-1][:K]
    peak_values = conv[peaks]
    peak_max = conv.max()
    
    # Compute subgradient (weighted by softmax over peaks)
    temp = 0.01 * (peak_max - peak_values.min() + 1e-15)
    w = np.exp((peak_values - peak_max) / temp)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, peaks):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += wi * a[k - i]
    
    # Frank-Wolfe: move from argmax(grad) to argmin(grad)
    src = int(np.argmax(grad * a))  # element contributing most
    dst = int(np.argmin(grad * a))  # element contributing least
    # Actually for FW, we should project onto the simplex
    # The FW step is: move from current x to the vertex e_dst (or a combination)
    # x_new = (1-gamma) * x + gamma * e_dst
    # But this might create zeros. Better: move a small amount from src to dst.
    
    # Line search: try different fractions
    best_frac = 0
    best_val = conv.max()
    for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        amt = a[src] * frac
        if amt <= 0:
            continue
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial /= s
        M = fast_conv(a_trial).max()
        if M < best_val:
            best_val = M
            best_frac = frac
    
    if best_frac > 0:
        amt = a[src] * best_frac
        a[src] -= amt
        a[dst] += amt
        a = np.maximum(a, 0)
        a /= a.sum()
    
    return a
```

3. Structured mass redistribution (block transfers)

import numpy as np
import time


_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_peak_gradient(a, conv, K, temperature):
    """Compute weighted subgradient from top K peaks."""
    n = len(a)
    peaks = np.argsort(conv)[::-1][:K]
    peak_vals = conv[peaks]
    peak_max = peak_vals[0]

    if temperature < 1e-12:
        w = np.zeros(K)
        w[0] = 1.0
    else:
        w = np.exp((peak_vals - peak_max) / temperature)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(K) / K
        else:
            w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, peaks):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += wi * 2.0 * a[k - idx]
    return grad

                      restart_every=800, seed=42, verbose=False):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    eta = eta_init
    temp = temp_init
    stall = 0
    iteration = 0

        conv = fast_conv(a)
        M = conv.max()

        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

            noise = rng.normal(0, strength, n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            x_prev = a.copy()
            stall = 0
            eta = max(eta * 0.6, 0.5)
            temp = max(temp - 0.0003, 0.0005)
            if verbose:
                print(f"    MG restart: J={2*n*best_m:.6f}, eta={eta:.3f}, temp={temp:.5f}")
            continue

        # Compute gradient
        grad = compute_peak_gradient(a, conv, K, temp)

        # Normalize gradient for stable updates
        g_mean = grad[grad > 0].mean() if np.any(grad > 0) else 1.0
        if g_mean < 1e-30:
            g_mean = 1.0
        grad_norm = grad / g_mean

        # Backtracking line search
        improved = False
        for trial_scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            trial_eta = eta * trial_scale
            # Multiplicative update (work in log space for stability)
            log_a = np.log(np.maximum(a, 1e-30)) - trial_eta * grad_norm
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()

            M_new = fast_conv(a_new).max()
            if M_new < M - 1e-16:
                a = a_new
                eta = min(eta * 1.02, 200.0)
                improved = True
                break

        if not improved:
            eta = max(eta * 0.5, 0.01)
            # If stuck, try a direct mass transfer
            if stall > 50:
                conv_a = fast_conv(a)
                peaks = np.argsort(conv_a)[::-1][:3]
                # Find the element contributing most to the top peak
                k_top = int(peaks[0])
                lo = max(0, k_top - n + 1)
                hi = min(n - 1, k_top)
                idx = np.arange(lo, hi + 1)
                contrib = a[idx] * a[k_top - idx]
                src_i = idx[np.argmax(contrib)]
                # Find least-contributing element overall
                all_stress = np.zeros(n)
                for pk in peaks:
                    pk = int(pk)
                    lo2 = max(0, pk - n + 1)
                    hi2 = min(n - 1, pk)
                    idx2 = np.arange(lo2, hi2 + 1)
                    all_stress[idx2] += a[idx2] * a[pk - idx2]
                dst_i = int(np.argmin(all_stress))

                if src_i != dst_i and a[src_i] > 1e-10:
                    frac = rng.uniform(0.02, 0.15)
                    amt = a[src_i] * frac
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 1e-12)
                    a /= a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        iteration += 1
        if verbose and iteration % 5000 == 0:
            print(f"    MG iter={iteration}, t={time.monotonic()-t0:.0f}s, "
                  f"J={2*n*best_m:.6f}, eta={eta:.2f}, stall={stall}")

    return best_a, best_m


def block_redistribute(a0, time_limit, K=5, verbose=False):
    """Block-based mass redistribution targeting peak reduction."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        M = conv.max()

        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            break

        # Compute multi-peak stress
        peaks = np.argsort(conv)[::-1][:K]
        peak_vals = conv[peaks]
        temp = 0.005
        w = np.exp((peak_vals - peak_vals[0]) / temp)
        w = w / w.sum()

        stress = np.zeros(n)
        for wi, k in zip(w, peaks):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            stress[idx] += wi * a[idx] * a[k - idx]

        # Identify hot and cold regions
        stress_thresh_hi = np.percentile(stress, 85)
        stress_thresh_lo = np.percentile(stress, 30)
        hot = stress > stress_thresh_hi
        cold = stress < stress_thresh_lo

        hot_mass = a[hot].sum()
        if hot_mass < 1e-10:
            break

        # Try transferring different fractions
        improved = False
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
            transfer = hot_mass * frac
            a_trial = a.copy()
            a_trial[hot] *= (1.0 - frac)
            cold_weights = a[cold] + 1e-15
            cw_sum = cold_weights.sum()
            if cw_sum < 1e-15:
                break
            a_trial[cold] += transfer * cold_weights / cw_sum
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial /= s
            M_new = fast_conv(a_trial).max()
            if M_new < M - 1e-16:
                a = a_trial
                improved = True
                break

        if not improved:
            # Try element-level move
            k_top = int(peaks[0])
            lo = max(0, k_top - n + 1)
            hi = min(n - 1, k_top)
            idx = np.arange(lo, hi + 1)
            contrib = a[idx] * a[k_top - idx]
            src_i = idx[np.argmax(contrib)]

            # Find best destination: minimize impact on all peaks
            dst_costs = np.zeros(n)
            for pk in peaks[:3]:
                pk = int(pk)
                lo2 = max(0, pk - n + 1)
                hi2 = min(n - 1, pk)
                idx2 = np.arange(lo2, hi2 + 1)
                dst_costs[idx2] += a[idx2] * a[pk - idx2]

            candidates = np.argsort(dst_costs + 1e-10 * a)[:20]
            candidates = [int(c) for c in candidates if int(c) != src_i]

            found = False
            for dst_i in candidates[:5]:
                for frac2 in [0.05, 0.1, 0.2, 0.3]:
                    amt = a[src_i] * frac2
                    if amt < 1e-15:
                        continue
                    a_trip = a.copy()
                    a_trip[src_i] -= amt
                    a_trip[dst_i] += amt
                    a_trip = np.maximum(a_trip, 0.0)
                    a_trip /= a_trip.sum()
                    M_trip = fast_conv(a_trip).max()
                    if M_trip < M - 1e-16:
                        a = a_trip
                        improved = True
                        found = True
                        break
                if found:
                    break


def spectral_reshape(a0, time_limit, n_iter_max=200, verbose=False):
    """Gerchberg-Saxton style spectral reshaping to flatten convolution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break

        # Forward transform
        A = np.fft.rfft(a, fft_size)
        power = A.real**2 + A.imag**2

        # We want |A|^2 to be more concentrated (peakier)
        # which makes the autocorrelation (iffT of |A|^2) flatter
        # Strategy: amplify the peak, suppress the rest
        peak_frac = np.percentile(power, 95)
        power_mod = np.where(power > peak_frac,
                            power,
                            power * 0.7)  # suppress non-peak frequencies

        # Recover new a from modified power spectrum
        # We need A_new such that |A_new|^2 = power_mod
        # Keep the phase from original A
        phase = np.angle(A)
        A_new = np.sqrt(power_mod) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_m


def coord_descent_polish(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        M = conv.max()

        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Focus on top peaks
        n_peaks = min(8, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_peaks]

        # Compute stress for destination selection
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        improved = False

        # Try moving mass from peak-contributing elements
        for k2 in top_peaks[:3]:
            if time.monotonic() - t0 > time_limit:
                break
            k2i = int(k2)
            lo = max(0, k2i - n + 1)
            hi = min(n - 1, k2i)
            idx = np.arange(lo, hi + 1)
            contrib = a[idx] * a[k2i - idx]

            # Top contributors
            n_top = max(5, min(15, n // 300))
            top_src_idx = idx[np.argsort(contrib)[::-1][:n_top]]

            for src_i in top_src_idx:
                if time.monotonic() - t0 > time_limit:
                    break
                src_i = int(src_i)
                if a[src_i] < 1e-12:
                    continue

                # Best destinations (lowest cost)
                candidates = np.argsort(costs + 1e-10 * a)[:max(10, n // 40)]
                candidates = [int(c) for c in candidates if int(c) != src_i]

                for dst_i in candidates[:6]:
                    found = False
                    for frac in [0.03, 0.06, 0.1, 0.15, 0.25, 0.4]:
                        amt = a[src_i] * frac
                        a_trial = a.copy()
                        a_trial[src_i] -= amt
                        preject_n = int(dst_i)
                        a_trial[preject_n] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial /= s_t
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found = True
                            stall = 0
                            break
                    if found:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Small random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_asymmetric_expon(n, rate=2.0):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-rate * x)
    return norm_sum1(a)


def make_asymmetric_ramp(n, power=0.7):
    x = np.arange(n, dtype=np.float64) / n
    a = (1.0 - x) ** power + 0.05
    return norm_sum1(a)


def make_two_hump_asym(n):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - 0.2) / 0.12) ** 2) + 0.6 * np.exp(-((x - 0.7) / 0.15) ** 2)
    return norm_sum1(np.maximum(a, 0.0))


def make_edge_concentrated(n, width=0.15):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - 0) / width) ** 2) + 0.7 * np.exp(-((x - 1.0) / (width * 1.3)) ** 2) + 0.1
    return norm_sum1(a)


def make_sinc_offset(n, center=0.3, bw=6.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - center) * bw) ** 2
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

                try_update(a_out, "final_mgd2")

        # Final coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t6d = min(remaining * 0.5, 100.0)
            if t6d > 5:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = coord_descent_polish(a_in, t6d, seed=16180, verbose=False)
                try_update(a_out, "final_cd2")

    # Final evaluation
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, "
              f"t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
