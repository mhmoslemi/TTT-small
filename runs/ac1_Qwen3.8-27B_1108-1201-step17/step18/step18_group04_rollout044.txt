def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_score(a):
    """Compute the objective value directly."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def compute_conv_gradient(a, conv, n_peaks=20, temperature=0.01):
    """
    Compute a gradient-like vector for reducing max conv.
    For each position i, the 'cost' of having mass there is the sum of
    partner values at active peaks.
    """
    n = len(a)
    conv_max = conv.max()
    
    # Find active peaks
    threshold = conv_max - temperature * 10.0
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Weight by how close to max
    w = np.exp((conv[active] - conv_max) / temperature)
    w = w / w.sum()
    
    # For each position i, compute total "load" = sum over active peaks k of w_k * a[k-i]
    load = np.zeros(n)
    for wi, k in zip(w, active):
        k = int(k)
        for i in range(n):
            partner = k - i
            if 0 <= partner < n:
                load[i] += wi * a[partner]
    
    return load

def mass_transport(a, n_steps=100, transport_rate=0.1, n_peaks=10, temperature=0.01):
    """
    Move mass from high-load to low-load positions.
    This directly reduces the peak values.
    """
    n = len(a)
    a = norm_sum1(a.copy())
    best_a = a.copy()
    best_score = evaluate_score(a)
    
    for step in range(n_steps):
        conv = fast_conv(a)
        load = compute_conv_gradient(a, conv, n_peaks=n_peaks, temperature=temperature)
        
        # Move mass from high-load to low-load
        # Target: make load more uniform
        target = np.full(n, load.mean())
        # Direction: increase where load < target, decrease where load > target
        direction = target - load  # positive where we want to add mass
        
        # Apply transport
        a_new = a + transport_rate * direction
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        score = evaluate_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Halve the step
            transport_rate *= 0.7
            a_new = a + transport_rate * direction
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            score = evaluate_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
            else:
                transport_rate = max(transport_rate * 0.5, 1e-6)
    
    return best_a, best_score

def subgrad_descent(a0, time_limit, top_k=20, temperature=0.005, lr=1.0, decay=0.999, 
                    restart_every=2000, seed=42):
    """
    Nesterov-like subgradient descent on the normalized sequence.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    lr_current = lr
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta = (t - 1) / (t + 1.0)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute subgradient
        conv_y = fast_conv(y)
        conv_max = conv_y.max()
        
        # Active set
        threshold = conv_max - temperature * 8.0
        active = np.where(conv_y >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_y)])
        if len(active) > top_k:
            order = np.argsort(conv_y[active])[::-1][:top_k]
            active = active[order]
        
        w = np.exp((conv_y[active] - conv_max) / temperature)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(active)) / len(active)
        else:
            w = w / w_sum
        
        # Subgradient: for peak k, grad_i = 2 * a[k-i]
        grad = np.zeros(n)
        for wi, k in zip(w, active):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * wi * a[k - idx] if False else 2.0 * wi * y[k - idx]
        
        # Multiplicative update (mirror descent)
        log_x = np.log(y) - lr_current * grad
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        M = fast_conv(x).max()
        if M < best_m - 1e-12:
            best_m = M
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Decay learning rate
        lr_current *= decay
        
        # Restart
        if stall > restart_every:
            stall = 0
            strength = 0.03
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-8)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            lr_current = lr * 0.5
    
    return best_x, best_m

def fourier_smooth(a, alpha=0.1):
    """Apply spectral smoothing in Fourier domain."""
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    freqs = np.fft.rfftfreq(fft_size)
    # Low-pass filter
    cutoff = 1.0 / (2.0 * np.pi * alpha)
    mask = np.exp(-2.0 * np.pi**2 * freqs**2 * alpha**2 * fft_size**2)
    A_smooth = A * mask
    a_smooth = np.fft.irfft(A_smooth, fft_size)[:n]
    a_smooth = np.maximum(a_smooth, 0.0)
    return a_smooth / a_smooth.sum()

def coordinate_descent_direct(a0, time_limit, seed=42):

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss
