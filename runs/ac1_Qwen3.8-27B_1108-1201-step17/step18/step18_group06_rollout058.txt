def single_peak_gradient(a, ac):
    """Gradient at the single maximum peak."""
    n = len(a)
    k = int(np.argmax(ac))
    # d(conv[k])/d(a[i]) = 2*a[k-i] for valid i
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    grad = np.zeros(n)
    idx = np.arange(j_lo, j_hi + 1)
    partner = k - idx
    grad[idx] = 2.0 * a[partner]
    return grad

def topk_gradient(a, ac, K=3):
    """Gradient as weighted sum over top K peaks."""
    n = len(a)
    top_idx = np.argsort(ac)[::-1][:K]
    # Use softmax weighting
    vals = ac[top_idx]
    log_vals = np.log(np.maximum(vals, 1e-30))
    log_vals -= log_vals.max()
    exp_vals = np.exp(log_vals)
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for ki, wi in zip(top_idx, w):
        ki = int(ki)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        partner = ki - idx
        grad[idx] += wi * 2.0 * a[partner]
    return grad

def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent: move mass between pairs of positions."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M = ac.max()
        k_max = int(np.argmax(ac))
        
        # Find the pair with maximum contribution to the peak
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]
        
        if contrib.max() < 1e-20:
            break
        
        # Try reducing the maximum contributing pair
        src = int(idx[np.argmax(contrib)])
        src_val = a[src]
        
        # Try moving mass to random destinations
        improved = False
        n_tries = 10
        for _ in range(n_tries):
            dst = int(rng.integers(n))
            if dst == src:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize(a_trial)
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Try a random move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.001, 0.02)
                a[i] -= amt
                a[j] += amt
                a = normalize(a)
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive mirror descent focusing on single max peak (K=1) with mode switching
2. Log-sum-exp smoothed optimization for escape from plateaus
3. Targeted coordinate descent reducing peak-contributing pairs
4. Multi-length search with interpolation
Key insight: The problem is convex on the simplex. Focusing gradient on the
single maximum peak converges faster when peaks are well-separated; switching
to distributed gradient when peaks are comparable prevents creating new peaks.

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safe wrapper around evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Project onto probability simplex (non-negative, sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def single_peak_gradient(a, ac):
    """Subgradient at the single maximum autocorrelation peak."""
    n = len(a)
    k = int(np.argmax(ac))
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo > j_hi:
        return grad
    idx = np.arange(j_lo, j_hi + 1)
    partner = k - idx
    grad[idx] = 2.0 * a[partner]
    return grad


def topk_gradient(a, ac, K=5):
    """Weighted subgradient over top K peaks with softmax weighting."""
    n = len(a)
    K = min(K, len(ac))
    top_idx = np.argsort(ac)[::-1][:K]
    vals = ac[top_idx]
    log_vals = np.log(np.maximum(vals, 1e-30))
    log_vals -= log_vals.max()
    exp_vals = np.exp(log_vals)
    w = exp_vals / exp_vals.sum()

    grad = np.zeros(n)
    for ki, wi in zip(top_idx, w):
        ki = int(ki)
        if wi < 1e-8:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        partner = ki - idx
        grad[idx] += wi * 2.0 * a[partner]
    return grad


def logsumexp_gradient(a, ac, temperature):
    """Smooth gradient using log-sum-exp over all lags."""
    n = len(a)
    t = temperature
    log_ac = np.log(np.maximum(ac, 1e-30))
    log_ac -= log_ac.max()
    exp_ac = np.exp(t * log_ac)
    w = exp_ac / exp_ac.sum()

    grad = np.zeros(n)
    threshold = 1e-7
    active = np.where(w > threshold)[0]
    for ki in active:
        ki = int(ki)
        wi = w[ki]
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        partner = ki - idx
        grad[idx] += wi * 2.0 * a[partner]
    return grad

def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """
    Coordinate descent: target the max peak's dominant contributing pairs
    and move mass to positions with low autocorrelation cost.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M = ac.max()
        k_max = int(np.argmax(ac))

        # Find top contributing pairs to the max peak
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)
        idx = np.arange(j_lo, j_hi + 1)
        contrib = a[idx] * a[k_max - idx]

        if contrib.max() < 1e-20:
            break

        # Get top 5 contributing source positions
        top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

        improved = False
        for src in top_contrib_idx:
            if time.monotonic() - t0 > time_limit:
                break
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute "cost" of each destination (how much it would increase other peaks)
            # Approximate: sum of a[j] for j involved in top peaks
            top_peaks = np.argsort(ac)[::-1][:8]
            costs = np.zeros(n)
            for kp in top_peaks:
                kpi = int(kp)
                j2_lo = max(0, kpi - n + 1)
                j2_hi = min(n - 1, kpi)
                if j2_lo <= j2_hi:
                    valid2 = np.arange(j2_lo, j2_hi + 1)
                    costs[valid2] += a[kpi - valid2]

            # Try several destinations: lowest cost positions
            candidates = np.argsort(costs)[:min(n // 4, 200)]
            # Filter out positions near src
            candidates = [c for c in candidates if c != src][:50]

            if not candidates:
                candidates = [int(rng.integers(n)) for _ in range(10)]
                candidates = [c for c in candidates if c != src]

            for dst in candidates[:10]:
                if time.monotonic() - t0 > time_limit:
                    break
                for frac in [0.003, 0.008, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = normalize(a_trial)
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random escape move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.001, 0.01)
                a[i] -= amt
                a[j] += amt
                a = normalize(a)
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m

def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, a_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        labels.append("exp25")
        inits.append(make_mild_ramp(n0, slope=0.15))
        labels.append("mild015")
        inits.append(make_mild_ramp(n0, slope=0.4))
        labels.append("mild04")

        t_per = max(25.0, t1_total / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 90:
                break
            t_avail = min(t_per, time_budget - 90 - (time.monotonic() - t_global))
            if t_avail < 20:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = adaptive_mirror_descent(init, t_avail, K=1,
                                                verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: Log-sum-exp optimization on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LogSumExp (n={n_best}) ===")
    t2 = min(remaining * 0.20, 350.0)

    if t2 > 40:
        a_in = normalize(best_seq)
        # Run 1: moderate annealing
        a_out, _ = lsexp_optimize(a_in, t2, t_start=3.0, t_end=15.0,
                                  eta0=0.12, verbose=False, rng_seed=200)
        try_update(a_out, f"lsexp1_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.10, 150.0)
        if t2b > 20:
            # Run 2: sharper (more exploitation)
            a_in = normalize(best_seq)
            a_out, _ = lsexp_optimize(a_in, t2b, t_start=5.0, t_end=25.0,
                                      eta0=0.08, verbose=False, rng_seed=201)
            try_update(a_out, f"lsexp2_{n_best}")

    # === Phase 3: Coordinate descent for peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)

    if t3 > 30:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_descent(a_in, t3, verbose=False, rng_seed=300)
        try_update(a_out, f"coord_{n_best}")

    # === Phase 4: More adaptive MD with longer convergence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Long AMD (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)

    if t4 > 30:
        a_in = normalize(best_seq)
        a_out, _ = adaptive_mirror_descent(a_in, t4,
                                           verbose=False, rng_seed=400)
        try_update(a_out, f"amd_long_{n_best}")

    # === Phase 5: Multi-length search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length (base n={n_best}) ===")

    lengths_to_try = []
    for delta in [-1000, -500, 500, 1000, 1500, 2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))

    for new_n in lengths_to_try[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_to_length(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")

        t5 = min(remaining * 0.30, 120.0)
        if t5 > 20:
            a_out, _ = adaptive_mirror_descent(a_interp, t5,
                                               verbose=False, rng_seed=500 + new_n)
            try_update(a_out, f"amd_n{new_n}")

    # === Phase 6: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}) ===")

    if remaining > 60:
        # AMD with very small steps for fine convergence
        t6a = min(remaining * 0.45, 200.0)
        if t6a > 20:
            a_in = normalize(best_seq)
            a_out, _ = adaptive_mirror_descent(a_in, t6a,
                                               verbose=True, rng_seed=700)
            try_update(a_out, f"final_amd_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            # Coordinate descent finish
            t6b = min(remaining * 0.7, 100.0)
            if t6b > 15:
                a_in = normalize(best_seq)
                a_out, _ = coordinate_descent(a_in, t6b,
                                              verbose=False, rng_seed=800)
                try_update(a_out, f"final_coord_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            # One more round of logsumexp with very sharp temperature
            t6c = min(remaining * 0.8, 80.0)
            if t6c > 15:
                a_in = normalize(best_seq)
                a_out, _ = lsexp_optimize(a_in, t6c, t_start=8.0, t_end=40.0,
                                          eta0=0.05, verbose=False, rng_seed=900)
                try_update(a_out, f"final_lsexp_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_ramp(4000, power=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
