def fast_conv(a, b=None):
    a = np.asarray(a, dtype=np.float64)
    na = len(a)
    if b is None:
        b = a
        nb = na
    else:
        b = np.asarray(b, dtype=np.float64)
        nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interp_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    res = np.interp(new_x, old_x, a_norm)
    res = np.maximum(res, 0.0)
    rs = res.sum()
    if rs < 1e-15:
        return np.ones(new_n) / new_n
    return res / rs


def get_subgradient(a, ac, rel_tol=1e-8):
    n = len(a)
    M = ac.max()
    thr = M - max(M * rel_tol, 1e-16)
    active = np.where(ac >= thr)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(ac))])
    g = np.zeros(n)
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        if js < je:
            jsa = np.arange(js, je)
            g[jsa] += 2.0 * a[np.abs(k - jsa)]
    g /= len(active)
    return g, M

def local_search(a0, time_limit, rng_seed=123, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a, a).max()
    t0 = time.monotonic()

    stall = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        n_tries = int(rng.integers(20, 60))
        for _ in range(n_tries):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue
            frac = float(rng.uniform(0.01, 0.15))
            amt = a[i] * frac
            a2 = a.copy()
            a2[i] -= amt
            a2[j] += amt
            a2 = np.maximum(a2, 0.0)
            s2 = a2.sum()
            if s2 < 1e-15:
                continue
            a2 = a2 / s2
            m2 = fast_conv(a2, a2).max()
            if m2 < best_m - 1e-15:
                a = a2
                best_m = m2
                best_a = a.copy()
                improved = True
        if not improved:
            stall += 1
            if stall > 400:
                break
            if stall % 20 == 0:
                k = int(rng.integers(3, 12))
                for _ in range(k):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))

                        a[i] -= amt

                a = np.maximum(a, 0.0)

                    a = a / s2
                    m2 = fast_conv(a, a).max()
                    if m2 < best_m:

                        stall = 0
        else:

        if verbose and stall % 200 == 0:
            el = time.monotonic() - t0
            print(f"    LocalSearch ({el:.0f}s): J={2.0*n*best_m:.6f} stall={stall}")
    print(f"    LocalSearch done ({time.monotonic()-t0:.0f}s): J={2.0*n*best_m:.6f}")

def parametric_sweep(n, time_limit, seed_a, seed_m, verbose=True):
    iarr = np.arange(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    best_a = norm_sum1(seed_a)
    best_m = seed_m
    t0 = time.monotonic()

        nonlocal best_a, best_m
        a = norm_sum1(a)
        m = fast_conv(a, a).max()
        if m < best_m:
            best_m = m
            best_a = a
            if verbose:
                print(f"      param {tag}: J={2.0*n*best_m:.6f}")

    for p in [0.2, 0.5, 0.7, 1.0, 1.3, 2.0, 3.0]:
        consider((iarr + 1.0) ** p, f"ramp{p}")
        consider((n - iarr + 1.0) ** p, f"rampR{p}")
    for a1 in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for b1 in [0.5, 1.0, 1.5, 2.0, 3.0]:
            consider((iarr + 1.0) ** a1 * (n - iarr + 1.0) ** b1, f"beta{a1}_{b1}")
    for c in [0.2, 0.3, 0.5, 0.7]:
        for sgm in [0.08, 0.15, 0.25, 0.4]:
            consider(np.exp(-((x - c) ** 2) / (2.0 * sgm * sgm)), f"gauss{c}_{sgm}")
    for c in [0.2, 0.3, 0.8]:
        for sgm in [0.06, 0.1, 0.18]:
            a2 = np.exp(-((x - 0.1) ** 2) / (2.0 * sgm ** 2)) + np.exp(-((x - c) ** 2) / (2.0 * sgm ** 2))
            consider(a2, f"bimod{c}_{sgm}")
    for fr in [1, 2, 3]:
        for ph in [0.0, 0.25, 0.5]:
            consider(np.maximum(np.sin(np.pi * fr * x + ph) ** 2, 1e-4), f"sin{fr}_{ph}")
    for r in [1, 2, 4, 8]:
        consider(np.exp(-r * x), f"exp{-r}")
        consider(np.exp(r * x), f"exp{r}")
    if time.monotonic() - t0 < time_limit * 0.7:
        base = norm_sum1(seed_a)
        for w in [0.2, 0.4, 0.6, 0.8]:
            lin = (iarr + 1.0) / (iarr + 1.0).sum()
            consider(w * base + (1.0 - w) * lin, f"blend{w}")
    if verbose:
        print(f"    Param sweep done ({time.monotonic()-t0:.0f}s): J={2.0*n*best_m:.6f}")
    return best_a, best_m

    return [float(v) for v in best_seq]
