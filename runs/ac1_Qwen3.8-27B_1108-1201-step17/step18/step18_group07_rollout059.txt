"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full softmax-gradient FTRL on the probability simplex (FFT-based gradient)
2. LP-based iterative peak reduction (radar waveform peaking algorithm)
3. Spectral (frequency-domain) amplitude reshaping
4. Multi-start, multi-length strategy with annealing
5. Targeted coordinate descent for final polish

Key improvement over previous: exact full-softmax gradient via FFT cross-correlation,
plus LP-based peak targeting for the final sub-0.1% improvement.

import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely evaluate the sequence using the provided evaluate_sequence."""
    try:

        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_softmax_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv(a,a)_k)) wrt a.
    Uses FFT cross-correlation for efficiency.
    grad[i] = 2 * sum_k softmax(beta*conv)[k] * a[k-i]  (for valid indices)
    """
    n = len(a)
    conv = fast_conv(a)  # length 2n-1
    # Stable softmax
    max_c = conv.max()
    exp_c = np.exp(beta * (conv - max_c))
    w_sum = exp_c.sum()
    if w_sum < 1e-30:
        w = np.ones_like(conv) / len(conv)
    else:
        w = exp_c / w_sum
    # Cross-correlation: R[i] = sum_k w[k] * a[k-i] for 0<=k-i<n
    # R = ifft(fft(w, N) * conj(fft(a, N)), N)[:n]
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    return 2.0 * R


def ftrl_softmax_optimize(a0, time_limit, beta_init=10.0, beta_max=100.0,
                          eta0=0.5, verbose=False):
    """
    FTRL mirror descent with full softmax gradient.
    Minimizes log-sum-exp approximation of max(conv(a,a)) on the simplex.
    """
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    c0 = fast_conv(a)
    best_m = c0.max()
    best_obj = 2.0 * n * best_m
    t0 = time.monotonic()

    eta = eta0
    # Use a "running" gradient accumulation for FTRL
    G = np.zeros(n)  # cumulative gradient

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal beta
        progress = min(1.0, t / 30000.0)
        beta = beta_init * (beta_max / beta_init) ** progress

        # Compute gradient
        grad = compute_softmax_gradient(a, beta)

        # FTRL-style: accumulate gradient, use mirror update
        G += grad
        # Decay
        G *= 0.999

        # Learning rate schedule
        eta_t = eta0 / (np.sqrt(t + 1.0))

        # Mirror descent update on simplex (KL projection)
        # a_new ∝ a * exp(-eta_t * G)
        # Use log-space for stability
        log_a = np.log(np.maximum(a, 1e-30)) - eta_t * G
        # Clip for stability
        log_a = np.clip(log_a, -50.0, 50.0)
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a = a_new

        # Evaluate
        c_new = fast_conv(a)
        M_new = c_new.max()
        obj_new = 2.0 * n * M_new

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_m = M_new
            best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL-softmax t={t} ({elapsed:.0f}s): obj={best_obj:.8f}, beta={beta:.1f}")

    return best_a, best_obj

def lp_peak_reduction(a0, time_limit, verbose=False):
    """
    Iterative LP-based peak reduction (peaking algorithm).
    At each step, target the highest peak and solve an LP to find the
    best direction to reduce it while respecting non-negativity.
    """
    from scipy.optimize import linprog
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()

    step_sizes = [0.05, 0.02, 0.01]

    for outer in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()

        # Find the index of the maximum peak
        k_star = int(np.argmax(conv))

        # The gradient of conv[k_star] wrt a is: 2*a[k_star - i] for valid i
        # We want to minimize this: minimize sum_i 2*a[k_star-i]*delta[i]
        # subject to sum_i delta[i] = 0 (preserve sum)
        # and -a[i] <= delta[i] <= a[i] (keep a+delta >= 0, bounded)

        # Build the gradient vector
        c_vec = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        for i in range(i_start, i_end):
            j = k_star - i
            if 0 <= j < n:
                c_vec[i] = 2.0 * a[j]

        # LP: minimize c_vec @ delta
        # subject to: sum(delta) = 0  (equality)
        #             -a[i] <= delta[i] <= a[i]  (bounds)

        c_lp = c_vec.copy()
        # Equality constraint: sum delta = 0
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bounds = [(-a[i], a[i]) for i in range(n)]

        # For large n, this LP might be slow. Use a simplified version:
        # Only optimize over a subset of coordinates (the most important ones)
        if n > 2000:
            # Use a reduced set: top indices by |c_vec| and indices near the peak
            abs_c = np.abs(c_vec)
            top_k = np.argsort(abs_c)[::-1][:min(500, n)]
            # Also include indices near the peak
            center_idx = np.arange(max(0, k_star - n // 4), min(n, k_star + n // 4 + 1))
            active_set = np.unique(np.concatenate([top_k, center_idx]))
            if len(active_set) > 800:
                active_set = top_k[:800]
            m = len(active_set)
            c_r = c_lp[active_set]
            A_eq_r = np.ones((1, m))
            b_eq_r = np.zeros(1)
            bounds_r = [(-a[active_set[i]], a[active_set[i]]) for i in range(m)]
            try:
                res = linprog(c_r, A_eq=A_eq_r, b_eq=b_eq_r, bounds=bounds_r,
                              method='highs', options={'presolve': True})
            except Exception:
                break
            if res.success and res.fun < -1e-12:
                delta = np.zeros(n)
                delta[active_set] = res.x
            else:
                delta = np.zeros(n)
        else:
            try:
                res = linprog(c_lp, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                              method='highs', options={'presolve': True})
            except Exception:
                break
            if res.success and res.fun < -1e-12:
                delta = res.x
            else:
                delta = np.zeros(n)

        # Apply step with different sizes
        improved = False
        for frac in step_sizes:
            a_trial = a + frac * delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            obj_trial = obj_value(a_trial)
            if obj_trial < best_obj - 1e-15:
                a = a_trial
                best_obj = obj_trial
                best_a = a.copy()
                improved = True
                break
            elif obj_trial < obj - 1e-15:
                a = a_trial
                improved = True
                break

        if not improved:
            # Try random perturbation
            rng = np.random.default_rng(hash(tuple(np.round(a[::100], 4))) % 2**31)
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * 0.01
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s

        if verbose and outer % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-peak iter={outer} ({elapsed:.0f}s): obj={best_obj:.8f}")

    return best_a, best_obj


def spectral_reshape(a0, time_limit, verbose=False):
    """
    Iterative spectral (amplitude) reshaping to flatten the convolution.
    In frequency domain: modify |FFT(a)| to make IFFT(|FFT(a)|^2) flatter.
    """
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    strength = 0.3
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: make |A|^2 proportional to something that gives flat IFFT
        # Strategy: compress dynamic range of |A|
        mag_min = mag[mag > 1e-15].min() if np.any(mag > 1e-15) else 1e-15
        mag_max = mag.max()
        if mag_max < 1e-15:
            break

        # Power compression
        mid = np.sqrt(mag_min * mag_max)
        mag_new = mid * (mag / mid) ** (1.0 - strength)
        mag_new = np.maximum(mag_new, 1e-30)

        # Ensure DC component gives correct sum
        # The sum of a is proportional to A[0] (DC)
        # After modifying mag, we need to preserve A[0]
        A_new = mag_new * np.exp(1j * phase)
        A_new[0] = np.abs(A[0])  # preserve DC
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Blend
        a_trial = (1.0 - 0.5 * strength) * a + 0.5 * strength * a_new
        a_trial = norm_simplex(a_trial)
        obj_trial = obj_value(a_trial)

        if obj_trial < best_obj - 1e-15:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial
            strength = min(0.6, strength * 1.02)
        else:
            strength = max(0.05, strength * 0.8)
            if strength < 0.05:
                break

    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=False):
    """
    Targeted coordinate descent: move mass from contributors to the worst peak.
    """
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 200:
            break

        conv = fast_conv(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Target top 3 peaks
        top_peaks = np.argsort(conv)[::-1][:3]
        improved = False

        for k in top_peaks:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            # Contribution of each pair to this peak
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try moving to a few candidate destinations
            for attempt in range(8):
                # Prefer destinations far from the peak
                if attempt < 4:
                    dst = int(rng.integers(n))
                else:
                    # Try positions that are "far" from k_int
                    candidates = np.where(np.abs(np.arange(n) - k_int) > n // 3)[0]
                    if len(candidates) == 0:
                        dst = int(rng.integers(n))
                    else:
                        dst = int(candidates[int(rng.integers(len(candidates)))])

                if dst == src:
                    continue

                for frac in [0.02, 0.05, 0.1, 0.15]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    obj_trial = 2.0 * n * M_trial
                    if obj_trial < best_obj - 1e-15:
                        a = a_trial
                        best_obj = obj_trial
                        best_a = a.copy()
                        improved = True
                        break
                    elif obj_trial < 2.0 * n * M - 1e-10:
                        a = a_trial
                        improved = True
                        break
                if improved:
                    break
            if improved:
                stall = 0
                break

        if not improved:
            stall += 1

    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_simplex(p)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.8f}")

    # === Phase 1: FTRL-softmax optimization with multiple inits ===
    print(f"\n=== Phase 1: FTRL-softmax (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.40, 650.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(parent)
            labels.append("parent")

        # Diverse inits at parent length
        inits.append(make_init_cosine(n0, mode=1))
        labels.append("cos1")
        inits.append(make_init_bowl(n0, sharpness=2.0))
        labels.append("bowl2")
        inits.append(make_init_antisym(n0))
        labels.append("antisym")
        inits.append(make_init_two_bump(n0))
        labels.append("2bump")
        inits.append(make_init_random_smooth(n0, seed=1))
        labels.append("rand1")
        inits.append(make_init_random_smooth(n0, seed=2))
        labels.append("rand2")

        # Perturbed parent
        rng_p = np.random.default_rng(777)
        pert_parent = norm_simplex(parent + rng_p.normal(0, 0.005, n0)) if parent is not None else None
        if pert_parent is not None:
            inits.append(pert_parent)
            labels.append("parent_pert")

        print(f"  Running FTRL-softmax on {len(inits)} inits, total budget={t_phase1:.0f}s")
        t_per = t_phase1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            remaining_now = time_budget - 30 - elapsed_so_far
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 15:
                break
            print(f"    FTRL-softmax '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = ftrl_softmax_optimize(init, t_avail, beta_init=15.0, beta_max=80.0,
                                              eta0=0.3, verbose=False)
            try_update(a_out, f"ftrl_{lbl}")

            # Quick spectral polish
            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now > 20:
                a_sp, _ = spectral_reshape(a_out, min(20.0, remaining_now * 0.2))
                try_update(a_sp, f"spectral_{lbl}")

    # === Phase 2: LP-based peak reduction on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP Peak Reduction (n={n_best}) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = lp_peak_reduction(a_in, t2, verbose=False)
        try_update(a_out, f"lp_n{n_best}")

        # FTRL polish after LP
        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.10, 100.0)
        if t2b > 15:
            a_out2, _ = ftrl_softmax_optimize(a_out, t2b, beta_init=30.0, beta_max=100.0,
                                              eta0=0.2, verbose=False)
            try_update(a_out2, f"ftrl_after_lp_n{n_best}")

    # === Phase 3: Further FTRL with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: FTRL high-beta (n={n_best}) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 20:
        a_in = norm_simplex(best_seq)
        # Higher beta for sharper max approximation
        a_out, _ = ftrl_softmax_optimize(a_in, t3, beta_init=50.0, beta_max=200.0,
                                         eta0=0.15, verbose=False)
        try_update(a_out, f"ftrl_highbeta_n{n_best}")
        a_sp, _ = spectral_reshape(a_out, min(30.0, (time_budget - (time.monotonic() - t_global)) * 0.3))
        try_update(a_sp, f"spectral_highbeta_n{n_best}")

    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate Descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 120.0)
    if t4 > 10:
        a_in = norm_simplex(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=False)
        try_update(a_out, f"cd_n{n_best}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length Exploration ===")

    for new_n in [n_best + 500, n_best + 1000, n_best - 500, n_best - 1000]:
        if new_n < 1500 or new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 45:
            break
        print(f"  Trying n={new_n}...")
        a_new = interpolate_sequence(best_seq, new_n)
        try_update(a_new, f"interp_n{new_n}")

        t5 = min(remaining * 0.20, 100.0)
        if t5 > 20:
            a_out, _ = ftrl_softmax_optimize(a_new, t5, beta_init=20.0, beta_max=80.0, eta0=0.3)
            try_update(a_out, f"ftrl_n{new_n}")
            a_sp, _ = spectral_reshape(a_out, min(15.0, (time_budget - (time.monotonic() - t_global)) * 0.1))
            try_update(a_sp, f"spectral_n{new_n}")

    # === Phase 6: Final intensive optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Polish (n={n_best}) ===")
    t6 = min(remaining * 0.35, 200.0)
    if t6 > 15:
        a_in = norm_simplex(best_seq)
        a_out, _ = ftrl_softmax_optimize(a_in, t6, beta_init=40.0, beta_max=150.0,
                                         eta0=0.1, verbose=False)
        try_update(a_out, f"ftrl_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_cd, _ = coordinate_polish(a_out, min(30.0, remaining * 0.5))
            try_update(a_cd, f"cd_final_n{n_best}")

    # === Phase 7: Final LP pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 7: Final LP (n={len(best_seq)}) ===")
        a_in = norm_simplex(best_seq)
        a_out, _ = lp_peak_reduction(a_in, min(remaining * 0.7, 60.0))
        try_update(a_out, f"lp_final_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.8f}")

    return [float(x) for x in best_seq]
