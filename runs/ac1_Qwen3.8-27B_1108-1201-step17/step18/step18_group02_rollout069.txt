def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_of(a):

    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_conv(a)
    return 2.0 * n * ac.max() / (s * s)

def get_solver():
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s, verbose=False)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            pass
    return cp.ECOS


def lp_solve(y, k_list, solver):
    import cvxpy as cp
    n = len(y)
    K = len(k_list)
    if K == 0:
        return np.ones(n) / n
    C = np.zeros((K, n))
    for jj, k in enumerate(k_list):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            C[jj, lo:hi] = y[k - np.arange(lo, hi)]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [cp.sum(g) == 1, C @ g <= M]
    prob = cp.Problem(cp.Minimize(M), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    gv = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    gs = gv.sum()
    if gs < 1e-15:
        return None
    return gv / gs

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

        a_out, _ = coord_descent(best_seq, t3, verbose=True)
        check_update(a_out, f"coord_n{n_best}")

    # Phase 4: Try nearby lengths
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Nearby lengths ===")
    nearby = []
    for d in [300, -300, 600, -600, 400, -400]:
        nn = n_best + d
        if 2500 <= nn <= 6000:
            nearby.append(nn)
    seen = set()
    nearby_uniq = []
    for nn in nearby:
        if nn not in seen:
            seen.add(nn)
            nearby_uniq.append(nn)

    for new_n in nearby_uniq[:3]:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 200:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_new = interpolate_seq(best_seq, new_n)
        # Quick LP-FW
        t4 = min(remaining * 0.30, 250.0)
        if t4 > 60 and new_n <= 5500:
            a_out, _ = lp_fw(a_new, t4, solver, max_K=2000, verbose=False)
            check_update(a_out, f"lpfw_n{new_n}")

    # Phase 5: Final LP-FW + coordination on overall best
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final polish (n={n_best}) ===")
    t5 = min(remaining * 0.45, 400.0)
    if t5 > 60 and n_best <= 5500:
        a_out, _ = lp_fw(best_seq, t5, solver, max_K=3000, verbose=True)
        check_update(a_out, f"final_lp_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    t6 = min(remaining * 0.40, 200.0)
    if t6 > 20:
        a_out, _ = coord_descent(best_seq, t6, verbose=True)
        check_update(a_out, f"final_coord_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = score_of(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm1(np.sin(np.pi * x) ** 2)
        best_score = score_of(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
