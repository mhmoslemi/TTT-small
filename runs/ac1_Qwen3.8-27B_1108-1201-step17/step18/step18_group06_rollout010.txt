"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based multi-peak softmax subgradient with Nesterov acceleration
2. scipy L-BFGS-B refinement on soft-max smoothed objective
3. Spectral (Gerchberg-Saxton) flattening
4. Coordinate descent polish
5. Multi-shape initialization and multi-length search
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1 << (2 * n - 2).bit_length()
    if size < 2:
        size = 2
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(1, len(a))
    return a / s

def softmax_gradient(a, tau=10.0):
    """Compute smooth-max value and its gradient w.r.t. a, for J = 2*n*max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    # Softmax weights over all 2n-1 lags
    tc = tau * c
    tc_shifted = tc - tc.max()
    exp_tc = np.exp(tc_shifted)
    w = exp_tc / exp_tc.sum()
    smooth_max = c.max() + np.log(exp_tc.sum()) / tau
    
    # grad[i] = 2 * (w ⊛ a)[i] = 2 * IFFT(FFT(w) * conj(FFT(a)))[i]
    size = 1 << (2 * n - 2).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)
    grad = 2.0 * corr[:n]
    
    return smooth_max, grad

def nesterov_subgrad(a0, time_limit, tau0=20.0, c0=1.0, rng_seed=42, verbose=False):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2.0 * n * best_M  # assuming sum=1
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    cur_tau = tau0
    stall = 0
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        t = step
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        M_y, grad_y = softmax_gradient(y, tau=cur_tau)
        
        # Subgradient step in log-space (mirror descent on simplex)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_y = np.log(y) - eta * grad_y
        log_y -= log_y.max()
        a_new = np.exp(log_y)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_x = fast_conv(x).max()
        M_eff = min(M_y, M_x)
        if M_eff < best_M - 1e-12:
            best_M = M_eff
            best_a = (x if M_x <= M_y else y).copy()
            stall = 0
        else:
            stall += 1
        
        # Adapt tau: increase over time to sharpen the max
        cur_tau = min(200.0, tau0 * (1.0 + 0.01 * min(stall, 500)))
        
        # Restart if stalled
        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.05, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            cur_tau = tau0
            c0 = c0 * 1.1
            
        if verbose and step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad step={step} ({elapsed:.0f}s): J={2*n*best_M:.6f}, tau={cur_tau:.1f}")
    
    return best_a, best_M

def l_bfgs_refine(a0, time_limit, tau=50.0, max_iter=200, verbose=False):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based softmax multi-peak subgradient with Nesterov acceleration (O(n log n) per step)
2. scipy L-BFGS-B local refinement on soft-max smoothed objective in log-space
3. Spectral (Gerchberg-Saxton) iterative flattening
4. Coordinate descent polish targeting peak-contributing positions
5. Multi-shape initialization (ramps, sigmoids, Gaussians) and multi-length search
6. Simulated annealing acceptance for escape from local minima

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_conv(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(1, len(a))
    return a / s


def softmax_grad(a, tau=50.0):
    """
    Compute (smooth_max_value, gradient) for 2*n*max(conv(a,a)) when sum(a)=1.
    Uses softmax over all 2n-1 lags for smooth approximation to max.
    Gradient computed via FFT-based cross-correlation.
    """
    n = len(a)
    c = fast_conv(a)
    # Softmax weights
    tc = tau * c
    tc_shifted = tc - tc.max()
    exp_tc = np.exp(tc_shifted)
    w = exp_tc / exp_tc.sum()
    smooth_max = c.max() + np.log(exp_tc.sum()) / tau

    # grad[i] = 2 * (w cross-corr a)[i] = 2 * IFFT(FFT(w) * conj(FFT(a)))[i]
    size = 1 << (2 * n - 2).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)
    grad = 2.0 * corr[:n]

    return smooth_max, grad


def nesterov_subgrad(a0, time_limit, tau0=20.0, c0=1.5, rng_seed=42, verbose=False):
    """Nesterov-accelerated mirror descent (exponential weighting) on simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    cur_tau = tau0
    stall = 0
    step = 0
    n_restarts = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        t = step
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        M_y, grad_y = softmax_grad(y, tau=cur_tau)

        # Mirror descent step (exponential weights)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_y = np.log(y) - eta * grad_y
        log_y -= log_y.max()
        a_new = np.exp(log_y)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_x = fast_conv(x).max()
        M_eff = min(M_y, M_x)
        if M_eff < best_M - 1e-14:
            best_M = M_eff
            best_a = (x if M_x <= M_y else y).copy()
            stall = 0
        else:
            stall += 1

        # Adaptive tau: slowly increase to sharpen
        cur_tau = min(300.0, tau0 * (1.0 + 0.005 * min(stall, 600)))

        # Restart mechanism
        if stall > 5000:
            stall = 0
            n_restarts += 1
            strength = min(0.12, 0.03 + 0.01 * min(n_restarts, 10))
            pert = rng.normal(0, strength, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            cur_tau = tau0 * (1.0 + 0.1 * n_restarts)
            c0 = c0 * (1.0 + 0.03 * min(n_restarts, 5))

        if verbose and step % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad step={step} ({elapsed:.0f}s): J={2*n*best_M:.6f}, tau={cur_tau:.1f}, stall={stall}")

    return best_a, best_M

def spectral_flatten(a0, time_limit, verbose=False):
    """Gerchberg-Saxton: iteratively flatten the power spectrum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = 0.4

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Aggressiveness increases over iterations
        alpha = min(0.95, 0.3 + 0.001 * it)
        P_target = P_mean * np.power(P / np.maximum(P_mean, 1e-30), 1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)

        mag_ratio = np.sqrt(P_target / np.maximum(P, 1e-30))
        mag_ratio = np.clip(mag_ratio, 0.75, 1.33)
        A_new = A * mag_ratio

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)
            M_curr = fast_conv(a).max()
            if M_curr < best_M:
                best_M = M_curr
                best_a = a.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Spectral iters={it} ({elapsed:.0f}s): J={2*n*best_M:.6f}")

    return best_a, best_M


def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Targeted coordinate moves to reduce peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_conv(a)
        M_curr = c.max()
        n_top = min(15, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:n_top]

        improved = False
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)

            # Find position with max contribution to this peak
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-cost destination
            costs = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                j2_lo = max(0, k2i - n + 1)
                j2_hi = min(n - 1, k2i)
                valid = np.arange(j2_lo, j2_hi + 1)
                if len(valid) > 0:
                    costs[valid] += a[k2i - valid]

            candidates = np.argsort(costs + 5.0 * a)[:n // 4]
            best_dst = candidates[np.argmin(a[candidates])]
            if best_dst == src:
                best_dst = int(rng.integers(n))
                if best_dst == src:
                    continue

            for frac in [0.03, 0.06, 0.12, 0.25]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-14:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDescent ({elapsed:.0f}s): J={2*n*best_M:.6f}")

    return best_a, best_M


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_ramp(n_base, power=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}, budget={time_budget:.0f}s")

    # === Phase 1: Multi-init Nesterov subgradient (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = remaining * 0.30
    print(f"\n=== Phase 1: Multi-init subgrad (t={t_phase1:.0f}s) ===")

    inits = []
    labels = []
    if parent is not None:
        inits.append(norm_sum1(parent))
        labels.append("parent")

    for pw in [0.3, 0.4, 0.5, 0.6, 0.8]:
        inits.append(make_ramp(n0, power=pw))
        labels.append(f"ramp_{pw}")
    inits.append(make_sigmoid_ramp(n0, sharpness=2.5, center=0.4))
    labels.append("sigmoid_25")
    inits.append(make_sigmoid_ramp(n0, sharpness=4.0, center=0.5))
    labels.append("sigmoid_40")
    inits.append(make_gauss_asym(n0, center=0.7, width=0.3))
    labels.append("gauss_asym")
    inits.append(make_cosine_ramp(n0, phase=0.3))
    labels.append("cosine")

    t_per = max(10.0, t_phase1 / len(inits))
    for idx, (init_a, lbl) in enumerate(zip(inits, labels)):
        elapsed_so_far = time.monotonic() - t_global
        if elapsed_so_far > time_budget - 60:
            break
        t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
        if t_avail < 8:
            break
        print(f"  Subgrad '{lbl}' (t={t_avail:.0f}s)...", end=" ")
        a_out, M_out = nesterov_subgrad(init_a, t_avail, tau0=20.0, c0=1.5,
                                         rng_seed=42 + idx * 7, verbose=False)
        print(f"M={M_out:.6f}")
        try_update(a_out, f"subgrad_{lbl}")

    # === Phase 2: L-BFGS refinement on best (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = remaining * 0.15
    print(f"\n=== Phase 2: L-BFGS refinement (t={t_phase2:.0f}s) ===")

    if t_phase2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, M_out = l_bfgs_refine(a_in, t_phase2, verbose=True)
        try_update(a_out, f"lbfgs_n{n0}")

    # === Phase 3: Spectral flattening (12% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase3 = remaining * 0.12
    print(f"\n=== Phase 3: Spectral flattening (t={t_phase3:.0f}s) ===")

    if t_phase3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, M_out = spectral_flatten(a_in, t_phase3, verbose=True)
        try_update(a_out, f"spectral_n{n0}")

        # L-BFGS after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t2b = min(remaining * 0.10, 80.0)
            a_out2, _ = l_bfgs_refine(norm_sum1(best_seq), t2b, verbose=False)
            try_update(a_out2, f"lbfgs_post_spec")

    # === Phase 4: Coordinate descent polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase4 = remaining * 0.10
    print(f"\n=== Phase 4: Coordinate descent (t={t_phase4:.0f}s) ===")

    if t_phase4 > 10:
        a_in = norm_sum1(best_seq)
        a_out, M_out = coordinate_descent(a_in, t_phase4, verbose=True, rng_seed=99)
        try_update(a_out, f"coord_n{n0}")

    # === Phase 5: Extended subgradient + L-BFGS cycle (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase5 = remaining * 0.20
    print(f"\n=== Phase 5: Extended optimization cycle (t={t_phase5:.0f}s) ===")

    if t_phase5 > 30:
        a_in = norm_sum1(best_seq)
        # First: longer subgradient with different params
        t5a = t_phase5 * 0.5
        a_out, _ = nesterov_subgrad(a_in, t5a, tau0=30.0, c0=2.0,
                                     rng_seed=123, verbose=True)
        try_update(a_out, f"subgrad_ext")

        # Then: L-BFGS polish
        remaining5 = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining5 * 0.3, 60.0)
        if t5b > 10:
            a_out2, _ = l_bfgs_refine(norm_sum1(best_seq), t5b, verbose=False)
            try_update(a_out2, f"lbfgs_ext")

    # === Phase 6: Multi-length search (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length search (t={remaining:.0f}s) ===")

    lengths_to_try = []
    for delta in [-300, -500, 300, 500, 800, -800]:
        new_n = n_best + delta
        if 2500 <= new_n <= 8000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))

    for new_n in lengths_to_try[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t6a = min(remaining * 0.35, 120.0)
        if t6a > 10:
            a_out, _ = nesterov_subgrad(a_long, t6a, tau0=25.0, c0=1.5,
                                         rng_seed=77, verbose=False)
            try_update(a_out, f"subgrad_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.2, 40.0)
            if t6b > 8:
                a_out2, _ = l_bfgs_refine(norm_sum1(best_seq) if new_n == n_best else
                                          interpolate_sequence(best_seq, new_n),
                                          t6b, verbose=False)
                try_update(a_out2, f"lbfgs_n{new_n}")

    # === Phase 7: Final polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (t={remaining:.0f}s) ===")

    if remaining > 20:
        # L-BFGS with aggressive tau
        t7a = min(remaining * 0.4, 80.0)
        if t7a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = l_bfgs_refine(a_in, t7a, verbose=True)
            try_update(a_out, f"lbfgs_final")

        # Coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.3, 50.0)
        if t7b > 8:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent(a_in, t7b, verbose=False, rng_seed=555)
            try_update(a_out, f"coord_final")

        # One more subgradient with very high tau
        remaining = time_budget - (time.monotonic() - t_global)
        t7c = min(remaining * 0.4, 60.0)
        if t7c > 8:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_subgrad(a_in, t7c, tau0=100.0, c0=0.8,
                                         rng_seed=999, verbose=False)
            try_update(a_out, f"subgrad_hightau")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_ramp(n_fb, power=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
