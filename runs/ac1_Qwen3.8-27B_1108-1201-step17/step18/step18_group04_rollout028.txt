def fast_conv(a):
    """Fast convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    """Evaluate J for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def compute_subgrad_exact(a, conv, n_active=50, tol=1e-6):
    """Compute exact subgradient of max conv using top active lags."""
    n = len(a)
    conv_max = conv.max()
    # Find all lags within tol of max
    active = np.where(conv >= conv_max - tol)[0]
    if len(active) > n_active:
        # Take top n_active
        idx = np.argsort(conv[active])[::-1][:n_active]
        active = active[idx]
    # Equal weights on active lags
    w = np.ones(len(active)) / len(active)
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k = int(k)
        # conv(k) = sum_i a_i * a_{k-i}
        # d/dj conv(k) = a_{k-j} + a_{j-k} (when valid)
        for j in range(n):
            term = 0.0
            if 0 <= k - j < n:
                term += a[k - j]
            if 0 <= j - k < n:
                term += a[j - k]
            grad[j] += wi * term
    return grad

def fast_conv(a):
    """Fast full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad_vec(a, conv, n_active=30, rel_tol=0.01):
    """
    Vectorized subgradient of max conv.
    Uses the top n_active lags near the max with softmax weights.
    """
    n = len(a)
    conv_max = conv.max()
    threshold = conv_max * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > n_active:
        order = np.argsort(conv[active])[::-1]
        active = active[order[:n_active]]
    
    # Softmax weights
    temp = conv_max * 0.005
    w = np.exp((conv[active] - conv_max) / max(temp, 1e-15))
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k = int(k)
        # First term: a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            # grad[lo1:hi1+1] += wi * a[k-lo1 : k-hi1 : -1]... 
            # For j=lo1: a[k-lo1], j=lo1+1: a[k-lo1-1], ..., j=hi1: a[k-hi1]
            # This is a[k-hi1 : k-lo1+1] reversed
            vals = a[k - hi1 : k - lo1 + 1][::-1]
            grad[lo1:hi1+1] += wi * vals
        
        # Second term: a[j-k] for j in [max(0,k), n-1]
        lo2 = max(0, k)
        hi2 = n - 1
        if lo2 <= hi2:
            vals = a[lo2 - k : hi2 - k + 1]
            grad[lo2:hi2+1] += wi * vals
    
    return grad


def eg_optimize(a0, time_limit, n_active_init=30, rel_tol_init=0.01,
                eta0=0.5, decay=0.999, seed=42, verbose=False):
    """Exponentiated gradient descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.time()
    rng = np.random.default_rng(seed)
    n_active = n_active_init
    rel_tol = rel_tol_init
    eta = eta0
    stall = 0
    
    while time.time() - t0 < time_limit:
        conv = fast_conv(a)
        M = conv.max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Restart with perturbation
                pert = rng.normal(0, 0.02, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-15)
                a_r /= a_r.sum()
                a = a_r
                stall = 0
                n_active = min(n_active + 5, 80)
                rel_tol = max(rel_tol - 0.001, 0.002)
        
        g = compute_subgrad_vec(a, conv, n_active=n_active, rel_tol=rel_tol)
        
        # Multiplicative update
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        a = a_new
        
        # Adaptive learning rate
        eta *= decay
        if eta < 0.001:
            eta = 0.001
            # Try to restart with fresh eta
            if stall > 1000:
                eta = 0.5
                n_active = min(n_active + 10, 100)
    
    return best_a, best_M

Actually, for the 1D subproblem min_{x>=0} max_k (C_k + d_k*x + q_k*x^2), there's a simpler approach:

If all q_k = 0 (no quadratic terms, which happens when 2i is not in the active set):

This is a linear programming problem in 1D. The solution is:
- If all d_k <= 0: x* = 0, optimal value = max(C_k)
- Otherwise: the optimal x* is where two lines (C_i + d_i*x) and (C_j + d_j*x) cross, and this crossing point minimizes the max.
  We can find it by: x* = min_{i: d_i > 0} max_{j: d_j < 0 or d_j = 0} (C_j - C_i) / (d_i - d_j)
  Wait, that's not quite right either.
  
  The correct approach: the function f(x) = max_k (C_k + d_k*x) is the upper envelope of lines. We want to minimize it over x >= 0.
  - f(0) = max_k C_k
  - If the maximum at x=0 is achieved by a line with d_k < 0, then moving x to the right decreases that line, so f might decrease.
  - The minimum is at the "lowest point" of the upper envelope.
  
  For a set of lines, the minimum of the upper envelope over x >= 0 can be found by:
  1. Compute f(0) = max C_k
  2. If the line achieving the max at x=0 has d <= 0, the min might be at x=0 or at an intersection point.
  3. Find all intersection points of pairs of lines, keep those with x >= 0, and evaluate f at each.
  
  For m lines, there are O(m^2) intersection points, and evaluating f at each is O(m), so total O(m^3) or O(m^2 log m) with sorting.
  
  For m ~ 30-50, this is fast enough (O(15000) operations per coordinate).

If there's one quadratic term (q_k = 1 for some k, i.e., 2i is in the active set):
The problem is min_{x>=0} max(x^2 + C_q, max_{k≠q} (C_k + d_k*x))
This can also be solved by checking:
- The point where x^2 + C_q = C_k + d_k*x for each k (quadratic equation, 2 solutions)
- x = 0

    Solve min_{x>=0} max_i (C[i] + d[i]*x + q[i]*x^2)

    Returns optimal x.
    """

    q_idx = -1

            q_idx = i
            break

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponentiated gradient descent with exact active-set subgradient (uniform weights)
2. Exact 1D coordinate descent solving the piecewise-linear/quadratic subproblem
3. Adaptive active-set size and KKT-guided refinement
4. Multi-length search with diverse initialization
Key differences from prior: exact subgradient (no softmax bias), exact 1D solves,
tighter active-set management, and alternating EG/CD schedule.

import numpy as np
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    """Fast full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_active_lags(conv, n_active=30, rel_tol=0.01):
    """Get active lags (near max) with uniform or top-K selection."""
    conv_max = conv.max()
    threshold = conv_max * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > n_active:
        order = np.argsort(conv[active])[::-1]
        active = active[order[:n_active]]
    return active, conv_max


def compute_subgrad_exact(a, conv, active):
    """Exact subgradient: uniform average over active lags."""
    n = len(a)
    grad = np.zeros(n)
    w = 1.0 / len(active)
    for k in active:
        k = int(k)
        # First term: for j in [max(0,k-n+1), min(n-1,k)]: grad[j] += w * a[k-j]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            # a[k-j] for j=lo1,...,hi1 gives a[k-lo1], a[k-lo1-1], ..., a[k-hi1]
            # i.e., a[k-hi1 : k-lo1+1] reversed
            grad[lo1:hi1 + 1] += w * a[k - hi1:k - lo1 + 1][::-1]
        # Second term: for j in [max(0,k), n-1]: grad[j] += w * a[j-k]
        lo2 = max(0, k)
        hi2 = n - 1
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += w * a[lo2 - k:hi2 - k + 1]
    return grad


def eg_optimize(a0, time_limit, seed=42, verbose=False):
    """
    Exponentiated gradient descent with exact subgradient.
    Adaptively grows active set and decays learning rate.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.time()
    rng = np.random.default_rng(seed)

    K = 20
    rel_tol = 0.005
    eta = 0.3
    eta_decay = 0.9995
    restart_count = 0
    stall = 0

    while time.time() - t0 < time_limit:
        conv = fast_conv(a)
        M = conv.max()

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            restart_count += 1
            K = min(K + 10, 80)
            rel_tol = max(rel_tol - 0.001, 0.001)
            eta = 0.3 * (1.0 + 0.1 * min(restart_count, 5))
            # Perturb
            strength = 0.01 * (1 + 0.5 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-15)
            a_r /= a_r.sum()
            a = a_r

        active, conv_max = get_active_lags(conv, K, rel_tol)
        g = compute_subgrad_exact(a, conv, active)

        # Multiplicative update
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        a = a_new

        eta *= eta_decay
        if eta < 0.0005:
            eta = 0.0005

    return best_a, best_M


def solve_1d_linear(C, d, x_old):
    """
    Solve min_{x>=0} max_i (C[i] + d[i]*x) using bisection on M.
    Returns (x_opt, M_opt).
    """
    m = len(C)
    # Upper bound: value at x=0
    hi_M = C.max()
    lo_M = 0.0

    for _ in range(50):
        mid_M = (lo_M + hi_M) / 2.0
        x_lo = 0.0
        x_hi = np.inf
        feasible = True
        for i in range(m):
            if d[i] > 1e-30:
                upper = (mid_M - C[i]) / d[i]
                if upper < -1e-15:
                    feasible = False
                    break
                x_hi = min(x_hi, upper)
            elif d[i] < -1e-30:
                lower = (C[i] - mid_M) / (-d[i])
                if lower > 0:
                    x_lo = max(x_lo, lower)
            else:
                if C[i] > mid_M + 1e-14:
                    feasible = False
                    break

        if feasible and x_lo <= x_hi + 1e-10:
            lo_M = mid_M
        else:
            hi_M = mid_M

    # Find valid x at M = lo_M
    x_lo = 0.0
    x_hi = np.inf
    for i in range(m):
        if d[i] > 1e-30:
            x_hi = min(x_hi, (lo_M - C[i]) / d[i])
        elif d[i] < -1e-30:
            lower = (C[i] - lo_M) / (-d[i])
            if lower > 0:
                x_lo = max(x_lo, lower)

    if x_lo > x_hi:
        x_opt = max(0.0, (x_lo + x_hi) / 2.0)
    else:
        x_opt = max(0.0, (x_lo + x_hi) / 2.0)
    return x_opt, lo_M


def solve_1d_mixed(C, d, q, x_old):
    """
    Solve min_{x>=0} max_i (C[i] + d[i]*x + q[i]*x^2) where at most one q[i]=1.
    Uses candidate evaluation + bisection hybrid.
    """
    m = len(C)
    q_idx = -1
    for i in range(m):
        if q[i] > 0.5:
            q_idx = i
            break

    if q_idx < 0:
        return solve_1d_linear(C, d, x_old)

    # Candidate points: x=0, intersections of quadratic with each line
    Cq = C[q_idx]
    candidates = [0.0]
    for k in range(m):
        if k == q_idx:
            continue
        # x^2 + Cq = C[k] + d[k]*x => x^2 - d[k]*x + (Cq - C[k]) = 0
        disc = d[k] * d[k] - 4.0 * (Cq - C[k])
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (d[k] - sq) / 2.0
            r2 = (d[k] + sq) / 2.0
            if r1 >= 0:
                candidates.append(r1)
            if r2 >= 0:
                candidates.append(r2)
    candidates.append(max(0.0, x_old))

    # Also add a few scaled versions
    for scale in [0.3, 0.7, 1.3, 2.0]:
        candidates.append(max(0.0, x_old * scale))

    best_val = np.inf
    best_x = 0.0
    for x_c in candidates:
        if x_c < 0:
            x_c = 0.0
        val = np.max(C + d * x_c + q * x_c * x_c)
        if val < best_val:
            best_val = val
            best_x = x_c

    # Refine with bisection around best_x
    if best_x > 0:
        lo = max(0.0, best_x * 0.5)
        hi = best_x * 2.0
        for _ in range(30):
            mid = (lo + hi) / 2.0
            val_mid = np.max(C + d * mid + q * mid * mid)
            val_lo = np.max(C + d * lo + q * lo * lo)
            val_hi = np.max(C + d * hi + q * hi * hi)
            if val_mid <= val_lo and val_mid <= val_hi:
                # Keep mid, narrow
                if val_mid <= val_lo:
                    hi = mid
                else:
                    lo = mid
            elif val_lo <= val_mid and val_lo <= val_hi:
                hi = mid
            else:
                lo = mid
        best_x = max(0.0, (lo + hi) / 2.0)
        best_val = np.max(C + d * best_x + q * best_x * best_x)

    # Check x=0
    val_0 = np.max(C)
    if val_0 < best_val:
        best_x = 0.0
        best_val = val_0

    return best_x, best_val


def exact_coord_descent(a0, time_limit, seed=42, verbose=False):
    """
    Exact 1D coordinate descent with piecewise-linear/quadratic subproblem solving.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.time()
    rng = np.random.default_rng(seed)
    stall = 0
    sweep = 0

    while time.time() - t0 < time_limit and stall < 80:
        sweep += 1
        conv = fast_conv(a)
        M = conv.max()

        # Active lags: within 3% of max, top 40
        conv_max = M
        threshold = conv_max * 0.97
        active_arr = np.where(conv >= threshold)[0]
        if len(active_arr) > 40:
            order = np.argsort(conv[active_arr])[::-1]
            active_arr = active_arr[order[:40]]
        active = [int(k) for k in active_arr]
        m = len(active)

        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.time() - t0 > time_limit:
                break

            x_old = a[int(i)]
            if x_old < 1e-14:
                continue

            ci = int(i)
            C = np.zeros(m)
            d = np.zeros(m)
            q = np.zeros(m)

            for idx_k, k in enumerate(active):
                if k == 2 * ci:
                    C[idx_k] = conv[k] - x_old * x_old
                    d[idx_k] = 0.0
                    q[idx_k] = 1.0
                elif 0 <= k - ci < n:
                    C[idx_k] = conv[k] - 2.0 * a[k - ci] * x_old
                    d[idx_k] = 2.0 * a[k - ci]
                    q[idx_k] = 0.0
                else:
                    C[idx_k] = conv[k]
                    d[idx_k] = 0.0
                    q[idx_k] = 0.0

            x_opt, M_opt = solve_1d_mixed(C, d, q, x_old)

            if abs(x_opt - x_old) > 1e-13:
                a[ci] = max(0.0, x_opt)
                improved = True

        # Recompute and check
        conv = fast_conv(a)
        M = conv.max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Small random perturbation to escape
            if stall > 20:
                n_pert = max(10, n // 100)
                idx = rng.choice(n, size=n_pert, replace=False)
                amt = a[idx] * rng.uniform(0.001, 0.01, n_pert)
                a[idx] -= amt
                a[idx] = np.maximum(a[idx], 0)
                dst = rng.integers(n, n_pert)
                a[dst] += amt
                a = a / a.sum()
                stall = 0

        if verbose and sweep % 5 == 0:
            print(f"    ECD sweep {sweep}: J={2.0*n*best_M:.6f}, stall={stall}")

    return best_a, best_M


def interpolate_seq(a, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_shape(n, shape_type='sinc', **kwargs):
    """Create various initial shapes."""
    x = np.linspace(0, 1, n)
    if shape_type == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif shape_type == 'cos2':
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif shape_type == 'gauss':
        bw = kwargs.get('bw', 2.0)
        a = np.exp(-bw * (x - 0.5) ** 2)
    elif shape_type == 'sinc':
        bw = kwargs.get('bw', 5.0)
        a = np.sinc((x - 0.5) * bw) ** 2
        a = np.maximum(a, 0)
    elif shape_type == 'exp_decay':
        rate = kwargs.get('rate', 2.0)
        a = np.exp(-rate * x)
    elif shape_type == 'two_hump':
        a = np.exp(-40*(x-0.2)**2) + 0.7*np.exp(-25*(x-0.7)**2)
    elif shape_type == 'flat_edges':
        a = np.sin(np.pi * x) ** 0.5
    else:
        a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)

        t4a = min(remaining * 0.35, 120.0)
        if t4a > 10:
            a_out, _ = eg_optimize(a_interp, t4a, seed=777 + new_n, verbose=False)
            try_update(a_out, f"eg_n{new_n}")

        # Short ECD
        remaining = time_budget - (time.time() - t_global)
        if remaining > 20:
            t4b = min(remaining * 0.4, 100.0)
            if t4b > 10:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = exact_coord_descent(a_in, t4b, seed=888 + new_n, verbose=False)
                try_update(a_out, f"ecd_n{new_n}")

    # === Phase 5: Final polish (5% time) ===
    remaining = time_budget - (time.time() - t_global)
    n_final = len(_best_tracker['seq'])
    print(f"\n=== Phase 5: Final polish (n={n_final}) ===")
    if remaining > 20:
        t5 = min(remaining * 0.7, 120.0)
        if t5 > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = exact_coord_descent(a_in, t5, seed=31415, verbose=False)
            try_update(a_out, "ecd_final")

        remaining = time_budget - (time.time() - t_global)
        if remaining > 10:
            t5b = min(remaining * 0.8, 60.0)
            if t5b > 5:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = eg_optimize(a_in, t5b, seed=271, verbose=False)
                try_update(a_out, "eg_final")

    # Final verification
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, "
              f"t={time.time()-t_global:.0f}s")
    else:
        n_fb = 5000
        _best_tracker['seq'] = make_shape(n_fb, 'sin2')
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
