def fast_conv(a):

    size = 2 * n - 1

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize so sum equals 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

        return np.ones_like(a) / len(a)
    return a / s

def logsumexp_smoothing(a, T):
    """Compute logsumexp(conv(a,a)/T) and its gradient w.r.t. a."""
    n = len(a)
    conv = fast_conv(a)
    max_c = conv.max()
    exp_vals = np.exp((conv - max_c) / T)
    lse = np.log(exp_vals.sum()) + max_c / T
    
    # Gradient: w = softmax(conv/T), grad[j] = (1/T) * sum_k w[k] * a[k-j]
    w = exp_vals / exp_vals.sum()
    
    # Cross-correlation via FFT: cross[j] = sum_m a[m] * w[m+j]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
    
    grad_a = (1.0 / T) * cross

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization on log-sum-exp smoothed objective with temperature annealing
2. FFT-based exact gradient computation (cross-correlation)
3. Peak-targeted coordinate descent with cost landscape
4. Multi-start exploration with diverse smooth initializations
5. Hybrid alternating polish (L-BFGS + CD)
"""

import numpy as np
import time
from scipy.optimize import minimize

_best_tracker = {'seq': None, 'score': float('inf')}
_LBFGS_T = 0.01  # global temperature for L-BFGS wrapper


def fast_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize so sum equals 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_score(a):
    """Compute 2*n*max(conv) for normalized a."""
    conv = fast_conv(a)
    return 2.0 * len(a) * conv.max()


def logsumexp_conv_and_grad(a, T):
    """Compute 2*n*logsumexp(conv(a,a)/T) and gradient w.r.t. a.
    Uses FFT-based cross-correlation for the gradient.
    """
    n = len(a)
    conv = fast_conv(a)
    max_c = conv.max()
    shifted = (conv - max_c) / T
    # Clip for numerical safety
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-30:
        lse = max_c / T
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        lse = np.log(exp_sum) + max_c / T
        w = exp_vals / exp_sum

    f = 2.0 * n * lse

    # Gradient: grad_a[j] = (1/T) * sum_m a[m] * w[m+j]
    # This is cross-correlation of a and w at lag j
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(np.conj(A) * W, fft_size)[:n]

    grad_a = (1.0 / T) * cross
    return f, grad_a


def objective_and_grad_z(z, T):
    """Objective and gradient w.r.t. z (log-parameterization).
    a = softmax(z), f = 2*n*logsumexp(conv(a,a)/T)
    Returns (f, grad_z).
    """
    n = len(z)
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    if a_sum < 1e-30:
        return 1e10, np.zeros(n)
    a = a / a_sum

    f, grad_a = logsumexp_conv_and_grad(a, T)

    # Convert to gradient w.r.t. z:
    # df/dz[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    center = np.dot(grad_a, a)
    grad_z = a * (grad_a - center)

    return f, grad_z


def _lbfgs_fun(z):
    """Top-level wrapper for scipy L-BFGS (uses global _LBFGS_T)."""
    return objective_and_grad_z(z, _LBFGS_T)


def lbfgs_phase(a0, T_start, T_end, time_limit, n_segments=6, maxiter_per_seg=30000):
    """Run L-BFGS with geometric temperature annealing."""
    n = len(a0)
    z = np.log(a0 + 1e-30)
    z = z - z.max()
    t0 = time.monotonic()

    Ts = np.geomspace(T_start, T_end, n_segments)
    seg_time = time_limit / n_segments

    for seg in range(n_segments):
        if time.monotonic() - t0 > time_limit:
            break
        T = Ts[seg]
        _LBFGS_T = T

        seg_start = time.monotonic()
        seg_elapsed = seg_start - t0
        remaining_seg = time_limit - seg_elapsed

        try:
            result = minimize(
                _lbfgs_fun, z, method='L-BFGS-B',
                jac=True,
                options={
                    'maxiter': maxiter_per_seg,
                    'gtol': 1e-7,
                    'maxfun': int(min(15000, max(500, remaining_seg * 8000))),
                    'maxls': 30,
                    'iprint': -1,
                }
            )
            z = result.x
            a_check = norm_sum1(np.exp(z - z.max()))
            sc = compute_score(a_check)
            print(f"    L-BFGS seg{seg}: T={T:.5f}, 2n*logsumexp={result.fun:.4f}, score={sc:.6f}, nfev={result.nfev}")
        except Exception as e:
            print(f"    L-BFGS seg{seg} error: {e}")

    a_final = norm_sum1(np.exp(z - z.max()))
    return a_final

    """Coordinate descent that explicitly targets the convolution peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))

        # Find the dominant contributing pairs to the peak
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]

        # Get top contributors
        n_top = min(15, len(idx))
        top_order = np.argsort(contrib)[::-1][:n_top]
        top_pairs = idx[top_order]

        # Build cost landscape from top 20 peaks
        n_peaks_for_cost = min(20, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:n_peaks_for_cost]
        costs = np.zeros(n)
        for k2 in peak_indices[:15]:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        improved = False
        # Try moving mass from each top contributor to low-cost positions
        for src_i in top_pairs:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src_i)
            if a[src_i] < 1e-14:
                continue

            # Find best destinations (low cost, not near peak)
            dst_mask = np.ones(n, dtype=bool)
            dst_mask[src_i] = False
            # Avoid positions too close to source (to not create new adjacent peaks)
            dst_mask[max(0, src_i - 2):min(n, src_i + 3)] = False
            dst_candidates = np.argsort(costs)[np.argsort(costs) != src_i]
            dst_candidates = [int(d) for d in dst_candidates[:30] if dst_mask[d]]

            for dst_i in dst_candidates[:10]:
                for frac in [0.03, 0.07, 0.12, 0.2, 0.3, 0.5]:
                    amt = a[src_i] * frac
                    if amt >= a[src_i]:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Periodic random perturbation
            if stall % 8 == 0:
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def swap_cd(a0, time_limit, seed=7):
    """Coordinate descent using swap moves (exchange between two positions)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        top_k_peaks = np.argsort(conv)[::-1][:10]

        # Cost landscape
        costs = np.zeros(n)
        for k2 in top_k_peaks[:8]:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        # Find best swap: high-cost, high-value → low-cost, low-value
        order = np.argsort(costs)
        low_end = order[:n // 4]
        high_end = order[-n // 4:]

        improved = False
        # Try a sample of swaps
        n_tries = min(200, n // 10)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.choice(high_end))
            j = int(rng.choice(low_end))
            if i == j or abs(i - j) < n // 50:
                continue
            ai, aj = a[i], a[j]
            if ai < 1e-12 or aj > ai:
                continue
            # Try partial swap: move frac*ai from i to j, and frac*aj from j to i
            for frac in [0.1, 0.3, 0.5, 0.7]:
                amt_i = ai * frac
                amt_j = aj * frac * 0.5  # asymmetric
                a_trial = a.copy()
                a_trial[i] = ai - amt_i
                a_trial[j] = aj + amt_i * 0.5 - amt_j
                if a_trial[j] < 0:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                for _ in range(max(2, n // 300)):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        amt = a[i] * rng.uniform(0.02, 0.1)
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def try_update(a, label=""):
    """Track best sequence via evaluate_sequence."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def make_perturbed(a, rng, strength=0.05):
    """Create a perturbed version of a."""
    n = len(a)
    pert = rng.normal(0, strength, n) * a
    result = a + pert
    result = np.maximum(result, 1e-10)
    return result / result.sum()


def make_smooth_random(n, rng, n_freq=12):
    """Create a random smooth non-negative sequence."""
    x = np.linspace(0, 1, n)
    base = np.ones(n) * 0.6
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.18 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        base += coeff * np.sin(2 * np.pi * f * x + phase)
    base = np.maximum(base, 0.005)
    return base / base.sum()


def make_cosine_taper(n, alpha=0.5):
    """Cosine taper: (1 + alpha*cos(pi*x))/(1+alpha), alpha in [0,1]."""
    x = np.linspace(0, 1, n)
    a = (1.0 + alpha * np.cos(np.pi * x)) / (1.0 + alpha)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_super_gaussian(n, p=2.0):
    """Super-Gaussian: exp(-|x-0.5|^p)."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) * 2) ** p)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_parabola(n, sharp=1.0):
    """Parabolic: 1 - sharp*(x-0.5)^2."""
    x = np.linspace(0, 1, n)
    a = 1.0 - sharp * (x - 0.5) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_asymmetric(n, skew=0.3):
    """Asymmetric smooth: exp(-((x-c)/w)^2) with center offset."""
    x = np.linspace(0, 1, n)
    c = 0.5 + skew * 0.3
    a = np.exp(-((x - c) / 0.25) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Default init: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    n0 = len(_best_tracker['seq'])
    print(f"Starting: n={n0}, score={_best_tracker['score']:.6f}, budget={time_budget:.0f}s")

    # === Phase 1: L-BFGS with temperature annealing (main workhorse) ===
    print(f"\n=== Phase 1: L-BFGS Annealing ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.38, 600.0)

    if t_phase1 > 60:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out = lbfgs_phase(a_in, T_start=0.008, T_end=0.0008,
                            time_limit=t_phase1, n_segments=7, maxiter_per_seg=25000)
        try_update(a_out, "lbfgs_anneal")

    # === Phase 2: Peak-targeted coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Peak-Targeted CD ===")
    t_phase2 = min(remaining * 0.15, 250.0)

    if t_phase2 > 20:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = peak_targeted_cd(a_in, t_phase2, seed=42)
        try_update(a_out, "peak_cd")

    # === Phase 3: Multi-start exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Multi-Start ===")
    t_phase3 = min(remaining * 0.25, 400.0)

    if t_phase3 > 60:
        rng = np.random.default_rng(2024)
        starts = []
        labels = []

        starts.append(make_perturbed(norm_sum1(_best_tracker['seq']), rng, 0.03))
        labels.append("pert3")
        starts.append(make_perturbed(norm_sum1(_best_tracker['seq']), rng, 0.10))
        labels.append("pert10")
        starts.append(make_cosine_taper(n0, 0.7))
        labels.append("cos07")
        starts.append(make_super_gaussian(n0, 2.5))
        labels.append("sg25")
        starts.append(make_parabola(n0, 0.8))
        labels.append("parab")
        starts.append(make_smooth_random(n0, rng, 15))
        labels.append("rnd15")
        starts.append(make_asymmetric(n0, 0.4))
        labels.append("asym4")

        n_starts = len(starts)
        t_per_start = t_phase3 / n_starts

        for i in range(n_starts):
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_avail = min(t_per_start, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  Start '{labels[i]}' t={t_avail:.0f}s")
            a_out = lbfgs_phase(starts[i], T_start=0.005, T_end=0.001,
                                time_limit=t_avail, n_segments=4, maxiter_per_seg=15000)
            try_update(a_out, f"ms_{labels[i]}")

            # Quick CD polish on the result
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_cd, _ = peak_targeted_cd(a_out, min(remaining * 0.15, 30.0), seed=100 + i)
                try_update(a_cd, f"ms_cd_{labels[i]}")

    # === Phase 4: Swap-based CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Swap CD ===")
    t_phase4 = min(remaining * 0.12, 150.0)

    if t_phase4 > 15:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = swap_cd(a_in, t_phase4, seed=77)
        try_update(a_out, "swap_cd")

    # === Phase 5: Final alternating polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 5: Final Polish (n={n_best}) ===")

    if remaining > 60:
        # L-BFGS with very low temperature
        t5a = min(remaining * 0.35, 120.0)
        if t5a > 15:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out = lbfgs_phase(a_in, T_start=0.003, T_end=0.0004,
                                time_limit=t5a, n_segments=4, maxiter_per_seg=20000)
            try_update(a_out, "lbfgs_final")

        # Peak CD
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.30, 100.0)
        if t5b > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = peak_targeted_cd(a_in, t5b, seed=31415)
            try_update(a_out, "cd_final")

        # One more L-BFGS
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out = lbfgs_phase(a_in, T_start=0.0015, T_end=0.0003,
                                time_limit=min(remaining * 0.4, 40.0),
                                n_segments=3, maxiter_per_seg=10000)
            try_update(a_out, "lbfgs_ultra")

    # Final evaluation
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, "
              f"t={time.monotonic() - t_global:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
