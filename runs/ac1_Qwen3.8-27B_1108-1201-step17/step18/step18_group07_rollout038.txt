But we only need the first n values, and for those, we need N >= 2n-1 (since p has length 2n-1 and the maximum shift we need is n-1, so we need N >= (2n-1) + (n-1) = 3n-2 to avoid aliasing).

Actually wait. For the first n values of the cross-correlation:
h[j] = sum_i a[i] * p[i+j] for j = 0, ..., n-1
The maximum index of p used is (n-1) + (n-1) = 2n-2 (when i=n-1, j=n-1).
The minimum index is 0 (when i=0, j=0).
So we need p[0] through p[2n-2], which is the full length of p.
For the circular correlation to match the linear one for the first n outputs, we need N >= 2n-1 + n - 1 = 3n - 2.

For n=4200, that's about 12600, and the next power of 2 is 16384. That's fine for FFT.

Actually wait, I think I'm overcomplicating this. Let me use a simpler approach:

Since p is concentrated (it's a softmax over the convolution peaks), only a few p[k] values are significantly non-zero. I can just loop over the non-negligible p[k] values and accumulate the gradient directly.

For each k where p[k] > threshold:
    grad[i] += p[k] * a[k-i] for valid i (i.e., 0 <= k-i <= n-1)
    which means i ranges from max(0, k-n+1) to min(n-1, k)

This is O(n * num_significant_peaks) which is fine if num_significant_peaks is small (say < 50).

Let me use this approach - it's simpler and faster for concentrated p.

Now let me write the full solution:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed log-sum-exp objective with projected mirror descent
2. Adaptive peak concentration (annealing)
3. Periodic spectral domain projection (amplitude flattening)  
4. Multi-start with diverse initializations and random restarts
5. Targeted coordinate descent on active peaks
6. Length exploration
"""

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_smooth_gradient(a, c, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a.
    
    grad[j] = sum_k p[k] * a[k-j] where p[k] = softmax(beta*c)[k]
    Only uses terms where p[k] is significant.
    """
    n = len(a)
    c_max = c.max()
    exponents = beta * (c - c_max)
    exponents = np.clip(exponents, -50, 0)
    p = np.exp(exponents)
    p_sum = p.sum()
    if p_sum < 1e-30:
        return np.zeros(n)
    p = p / p_sum
    
    # Use only significant terms
    threshold = p.max() * 1e-6
    significant = np.where(p > threshold)[0]
    
    grad = np.zeros(n)
    for k in significant:
        pk = p[k]
        if pk < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += pk * a[j_idx]
    return grad

def spectral_project(a, power=0.7, blend=0.5):
    """Project toward flat power spectrum while maintaining non-negativity."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.power(ratio, 1.0 - power)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    
    a_out = (1.0 - blend) * a + blend * a_new
    return norm_simplex(a_out)

def mirror_descent_optimize(a0, time_limit, beta_init=30.0, beta_max=200.0,
                            eta_init=0.1, eta_decay=0.9995, verbose=False):
    """Mirror descent on the simplex with log-sum-exp smoothed objective."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Log-barrier parameter for strict positivity
    log_a = np.log(np.maximum(a, 1e-15))
    eta = eta_init
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta (concentration of the softmax)
        progress = min(1.0, t / 100000.0)
        beta = beta_init + (beta_max - beta_init) * progress
        
        # Compute smoothed gradient
        c = fast_conv(a)
        g = compute_smooth_gradient(a, c, beta)
        
        # Mirror descent step: a_new[j] ∝ a[j] * exp(-eta * g[j])
        # with normalization
        log_step = -eta * g
        log_step = np.clip(log_step, -5, 5)  # clip for stability
        a_new = a * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Don't accept worsening steps (greedy)
            # Occasionally accept to escape local minima
            if np.random.random() < 0.01 * (1 - progress):
                a = a_new
            # Reduce step size
            eta *= 0.95
        
        # Periodic spectral projection
        if t % 2000 == 0:
            a_sp = spectral_project(a, power=0.3 + 0.4 * progress, blend=0.2 + 0.3 * progress)
            c_sp = fast_conv(a_sp)
            if c_sp.max() < best_m:
                best_m = c_sp.max()
                best_a = a_sp.copy()
                a = a_sp
        
        eta *= eta_decay
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={beta:.1f}, eta={eta:.4f}")
    
    return best_a, best_m

def targeted_peak_reduction(a0, time_limit, verbose=False):
    """Aggressively reduce the top peaks by adjusting contributing elements."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        
        # Get top 3 peaks
        top_k = min(3, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        improved = False
        
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            # Find pairs (i, k_int-i) with large product
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            # Try to reduce the largest contribution by moving mass
            top_contrib_idx = np.argsort(contrib)[::-1][:5]
            
            for ci in top_contrib_idx:
                if improved:
                    break
                i_pos = idx[ci]
                j_pos = k_int - i_pos
                prod = a[i_pos] * a[j_pos]
                if prod < 1e-10:
                    continue
                
                # Try reducing a[i_pos] or a[j_pos] and moving to a "quiet" position
                # Find a position that doesn't contribute much to any peak
                for src in [i_pos, j_pos]:
                    if improved:
                        break
                    src_val = a[src]
                    if src_val < 1e-8:
                        continue
                    
                    # Find "quiet" destination: position with low total contribution
                    # Approximate: position where a is already low
                    quiet_positions = np.argsort(a)[:10]
                    
                    for dst in quiet_positions:
                        if dst == src or dst == (k_int - src):
                            continue
                        for frac in [0.02, 0.05, 0.1, 0.2]:
                            amt = src_val * frac
                            if amt < 1e-10:
                                break
                            a_trial = a.copy()
                            a_trial[src] -= amt
                            a_trial[dst] += amt
                            a_trial = np.maximum(a_trial, 0.0)
                            s_t = a_trial.sum()
                            if s_t < 1e-15:
                                continue
                            a_trial = a_trial / s_t
                            M_trial = fast_conv(a_trial).max()
                            if M_trial < best_m - 1e-15:
                                a = a_trial
                                best_m = M_trial
                                best_a = a.copy()
                                improved = True
                                break
                    
        if improved:
            stall = 0
            eta_mult = 1.0
        else:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m

def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0) + 0.01
    return norm_simplex(a)

def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)

def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.03
    return norm_simplex(a)

def make_init_hann(n):
    """Hann window shape."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0.0) + 0.01
    return norm_simplex(a)

def make_init_flat(n):
    return np.ones(n) / n

def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0) + 0.01
    return norm_simplex(a)

def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
```

```python
def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_smooth_gradient(a, c, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a.

    Uses only significant terms for efficiency.
    """

    c_max = c.max()
    exponents = beta * (c - c_max)
    exponents = np.clip(explacements_ if False else (exponents, -50.0, 0.0)[1] if False else exponents, -50.0, 0.0)
    p = np.exp(exponents)
    p_sum = p.sum()
    if p_sum < 1e-30:
        return np.zeros(n)
    p = p / p_sum

    threshold = max(p.max() * 1e-7, 1e-14)
    significant = np.where(p > threshold)[0]

    grad = np.zeros(n)
    for k in significant:
        pk = p[k]
        if pk < 1e-14:
            continue
        k_int = int(k)

        if i_start >= i_end:

        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += pk * a[j_idx]
    return grad


def spectral_project(a, power=0.5, blend=0.3):
    """Project toward flat power spectrum while maintaining non-negativity."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()

    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * np.power(ratio, 1.0 - power)
    mag_new = np.maximum(mag_new, 1e-30)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s

    a_out = (1.0 - blend) * a + blend * a_new
    return norm_simplex(a_out)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            pert_scale = 0.015 + 0.008 * r
            perturbed = norm_simplex(best_seq + rng.normal(0, pert_scale, n_best))
            t_avail = min(t_per_r, remaining_now)
            beta_base = 50.0 + 20.0 * r
            a_out, _ = mirror_descent_optimize(perturbed, t_avail,
                                              beta_init=beta_base, beta_max=beta_base + 150.0,
                                              eta_init=0.025, eta_decay=0.9996, verbose=False)
            try_update(a_out, f"restart{r}_n{n_best}")

    # === Phase 5: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length Variation ===")
    for delta in [300, -300, 500, -500]:
        new_n = n_best + delta
        if new_n < 2000 or new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}...")
        a_new_len = interpolate_sequence(best_seq, new_n)
        try_update(a_new_len, f"interp_n{new_n}")

        t5 = min(remaining * 0.18, 90.0)
        if t5 > 15:
            a_out, _ = mirror_descent_optimize(a_new_len, t5,
                                              beta_init=40.0, beta_max=250.0,
                                              eta_init=0.03, eta_decay=0.9997, verbose=False)
            try_update(a_out, f"md_n{new_n}")

    # === Phase 6: Final intensive polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Polish (n={n_best}) ===")
    t6 = min(remaining * 0.35, 250.0)
    if t6 > 20:
        a_curr = norm_simplex(best_seq)
        t_start = time.monotonic()
        best_m_now = fast_conv(a_curr).max()

        while time.monotonic() - t_start < t6:
            # MD burst
            t_burst = min(40.0, t6 - (time.monotonic() - t_start))
            if t_burst < 5:
                break
            a_out, m_out = mirror_descent_optimize(a_curr, t_burst,
                                                  beta_init=120.0, beta_max=400.0,
                                                  eta_init=0.004, eta_decay=0.99995, verbose=False)
            a_curr = a_out

            # Spectral step
            remaining_inner = t6 - (time.monotonic() - t_start)
            if remaining_inner < 5:
                break
            a_sp = spectral_project(a_curr, power=0.5, blend=0.25)
            m_sp = fast_conv(a_sp).max()
            if m_sp < m_out:
                a_curr = a_sp
                m_out = m_sp

            if m_out < best_m_now:
                best_m_now = m_out
                try_update(a_curr, f"final_n{n_best}")

            # Check if we're still making progress
            if time.monotonic() - t_start > t6 * 0.8:
                break

        # Final targeted reduction
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_simplex(best_seq)
            a_out, _ = targeted_peak_reduction(a_in, min(remaining * 0.5, 50.0), verbose=False)
            try_update(a_out, f"final_peak_n{n_best}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_out2, _ = mirror_descent_optimize(a_out, min(remaining * 0.7, 30.0),
                                                   beta_init=200.0, beta_max=500.0,
                                                   eta_init=0.003, eta_decay=0.99995, verbose=False)
                try_update(a_out2, f"final_md2_n{n_best}")

    # === Phase 7: Ultimate final sprint ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 7: Final Sprint (n={n_best}, {remaining:.0f}s left) ===")
        a_in = norm_simplex(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, min(remaining * 0.7, 60.0),
                                          beta_init=150.0, beta_max=500.0,
                                          eta_init=0.003, eta_decay=0.99995, verbose=False)
        try_update(a_out, f"sprint_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
