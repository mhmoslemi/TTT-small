"""
Minimizes J = 2*n*max(full_conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (maximize sum(b) s.t. a*b <= M, b>=0)
2. Smoothed gradient descent with log-sum-exp
3. Coordinate descent polish
Starting from parent sequence or power-law ramp.

import numpy as np
import time
import scipy.optimize
from scipy.optimize import linprog, minimize

def fast_autocorr_full(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_fast(a):
    """Fast evaluation of J without calling evaluate_sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_autocorr_full(a)
    return float(2.0 * len(a) * c.max() / (s * s))

def norm_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_i = np.interp(new_x, old_x, f_n)
    return norm_sum1(f_i)

def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    # Build constraint matrix: (2n-1) x n
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n
    # b >= 0 handled by bounds
    
    # c vector: minimize -sum(b)
    c_vec = -np.ones(n)
    
    # Build A_ub efficiently
    # For constraint k (0 <= k <= 2n-2) and variable j (0 <= j <= n-1):
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # This is a Toeplitz-like matrix. Let's build it row by row.
    A_rows = []
    for k in range(2 * n - 1):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        row = np.zeros(n)
        if j_lo <= j_hi:
            indices = np.arange(j_lo, j_hi + 1)
            row[indices] = a[k - indices]
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    
    result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: return uniform
        return np.ones(n) / n

def lp_optimize(a0, time_limit, verbose=True, rng_seed=42):
    """LP-based optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = evaluate_fast(a)
    best_M = fast_autocorr_full(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    lp_count = 0
    last_lp_time = 0
    
    while time.monotonic() - t0 < time_limit:
        # Get LP direction (re-solve every few iterations or first time)
        if lp_count == 0 or (time.monotonic() - last_lp_time > 5.0):
            c = fast_autocorr_full(a)
            M = c.max()
            t_lp_start = time.monotonic()
            b = solve_lp_direction(a, M)
            t_lp = time.monotonic() - t_lp_start
            last_lp_time = t_lp_start
            lp_count += 1
            b = norm_sum1(b)
            
            if verbose and lp_count % 10 == 1:
                elapsed = time.monotonic() - t0
                print(f"    LP solve {lp_count} ({t_lp:.2f}s, total {elapsed:.0f}s): J={best_J:.6f}")
        
        # Line search: try different step sizes
        best_step = None
        best_step_J = best_J
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = (1 - t) * a + t * b
            a_trial = norm_sum1(a_trial)
            J_trial = evaluate_fast(a_trial)
            if J_trial < best_step_J:
                best_step_J = J_trial
                best_step = t
        
        if best_step is not None:
            a = (1 - best_step) * a + best_step * b
            a = norm_sum1(a)
            J_new = evaluate_fast(a)
            if J_new < best_J - 1e-9:
                best_J = J_new
                best_a = a.copy()
            
            # Occasionally add small perturbation to escape plateaus
            if lp_count % 20 == 0 and J_new > best_J * 0.999:
                # Skip - just continue with LP
                pass
        else:
            # No improvement, perturb
            pert = rng.normal(0, 0.01, n)
            a = a + pert * a
            a = norm_sum1(a)
    
    return best_a, best_J

def smoothed_gradient_step(a, K=50.0):
    """One step of smoothed gradient descent using log-sum-exp."""
    n = len(a)
    c = fast_autocorr_full(a)
    
    # Log-sum-exp approximation of max
    c_max = c.max()
    exp_c = np.exp(K * (c - c_max))
    weights = exp_c / exp_c.sum()
    
    # Gradient of (1/K)*log(sum exp(K*c[k])) w.r.t. a[i]
    # = sum_k weights[k] * dc[k]/da[i]
    # dc[k]/da[i] = 2*a[k-i] (symmetric)
    grad = np.zeros(n)
    # For each k, dc[k]/da[i] = a[k-i] + a[i]*(if k-i != i) ... 
    # Actually dc[k] = sum_j a[j]*a[k-j]
    # d/dc[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if k-i != i
    #                = a[i] (which is a[k-i] when k=2i)
    # Simplified: dc[k]/da[i] = 2*a[k-i] for all valid (i, k-i)
    # (the factor 2 accounts for i appearing in both terms of a[j]*a[k-j])
    
    # Use the weighted sum
    active_k = np.where(weights > 1e-10)[0]
    for k in active_k:
        ki = int(k)
        w = weights[ki]
        if w < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair = ki - idx
        grad[idx] += 2.0 * w * a[pair]
    
    return grad, c.max()

def gradient_optimize_smooth(a0, time_limit, K=50.0, lr0=0.01, verbose=True, rng_seed=42):
    """Smoothed gradient descent with log-sum-exp."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = evaluate_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    lr = lr0
    stall = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute smoothed gradient
        K_cur = min(K * (1 + it / 1000.0), 200.0)  # Anneal K
        grad, M = smoothed_gradient_step(a, K_cur)
        
        # Gradient step in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - lr * grad / a  # Chain rule: dx = da/a
        # Actually, if we work directly in a-space:
        # a_new = a - lr * grad, then clip and renormalize
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        J_new = evaluate_fast(a_new)
        if J_new < best_J - 1e-10:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # Still move (momentum-like)
            stall += 1
        
        # Adaptive learning rate
        if stall > 50:
            lr *= 0.8
            stall = 0
        elif stall == 0 and lr < lr0:
            lr *= 1.05
        
        # Periodic perturbation
        if it > 0 and it % 500 == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.05
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradSmooth it={it} ({elapsed:.0f}s): J={best_J:.6f}, K={K_cur:.0f}, lr={lr:.5f}")
    
    return best_a, best_J

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Aggressive coordinate descent polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = evaluate_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_autocorr_full(a)
        top_k = np.argsort(c)[::-1][:15]
        
        improved = False
        for ki in top_k:
            ki = int(ki)
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find contributing pairs
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            # The pair with max contribution
            src = idx[np.argmax(contrib)]
            partner = ki - src
            
            # Try moving mass from src to a random cold position
            # Cold positions: those not in top peaks
            peak_positions = set()
            for k2 in top_k[:10]:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                for p in range(lo2, hi2 + 1):
                    if a[p] * a[k2i - p] > 0.5 * c.max():
                        peak_positions.add(p)
            
            cold = [i for i in range(n) if i not in peak_positions and i != src]
            if not cold:
                cold = list(range(n))
            
            dst = cold[int(rng.integers(len(cold)))]
            
            # Try various fractions
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                J_trial = evaluate_fast(a_trial)
                if J_trial < best_J - 1e-12:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Larger random perturbation
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.02, 0.15)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    J_try = evaluate_fast(a)
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        stall = 0
                        break
    
    return best_a, best_J

"""
Minimizes J = 2*n*max(full_conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (maximize sum(b) s.t. a*b <= M, b>=0) with line search
2. Smoothed gradient descent with log-sum-exp annealing
3. Aggressive coordinate descent polish
4. Multi-length interpolation check
Key: LP gives optimal "spreading direction" to reduce convolution peaks.
"""
import numpy as np
import time
import scipy.optimize
from scipy.optimize import linprog


def fast_autocorr_full(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_fast(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_autocorr_full(a)
    return float(2.0 * len(a) * c.max() / (s * s))


def norm_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_i = np.interp(new_x, old_x, f_n)
    return norm_sum1(f_i)


def build_lp_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = a[k - np.arange(j_lo, j_hi + 1)]
    return A


def solve_lp_direction(a, M):
    n = len(a)
    c_vec = -np.ones(n)
    A_ub = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        b = result.x
        if b.sum() < 1e-15:
            return np.ones(n) / n
        return b / b.sum()
    return np.ones(n) / n


def lp_optimize(a0, time_limit, verbose=True, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = evaluate_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    b_dir = None
    last_solve_it = -10
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if b_dir is None or (it - last_solve_it) >= 5:
            c = fast_autocorr_full(a)
            M = c.max()
            ts = time.monotonic()
            b_dir = solve_lp_direction(a, M)
            last_solve_it = it
            if verbose and it % 25 == 1:
                elapsed = time.monotonic() - t0
                print(f"    LP it={it} ({time.monotonic()-ts:.2f}s, tot {elapsed:.0f}s): J={best_J:.6f}")

        best_step = None
        best_step_J = best_J
        for t in [0.03, 0.05, 0.1, 0.15, 0.25, 0.4]:
            a_try = (1.0 - t) * a + t * b_dir
            a_try = norm_sum1(a_try)
            J_try = evaluate_fast(a_try)
            if J_try < best_step_J:
                best_step_J = J_try
                best_step = t

        if best_step is not None:
            a = (1.0 - best_step) * a + best_step * b_dir
            a = norm_sum1(a)
            J_new = evaluate_fast(a)
            if J_new < best_J - 1e-12:
                best_J = J_new
                best_a = a.copy()
        else:
            if it % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * 0.03
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    J_try = evaluate_fast(a)
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()

    return best_a, best_J


def smoothed_grad_step(a, K):
    n = len(a)
    c = fast_autocorr_full(a)
    c_max = c.max()
    exp_c = np.exp(K * (c - c_max))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for ki in active:
        ki = int(ki)
        wi = w[ki]
        if wi < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair = ki - idx
        grad[idx] += 2.0 * wi * a[pair]
    return grad, c_max


def gradient_optimize_smooth(a0, time_limit, K0=30.0, lr0=0.005, verbose=True, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = evaluate_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    lr = lr0
    stall = 0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        K_cur = min(K0 + it * 0.1, 200.0)
        grad, M = smoothed_grad_step(a, K_cur)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        J_new = evaluate_fast(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
        if stall > 80:
            lr *= 0.7
            stall = 0
        elif stall == 0 and lr < lr0 * 2:
            lr = min(lr * 1.03, lr0 * 3)
        if it > 0 and it % 300 == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.04
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradSmooth it={it} ({elapsed:.0f}s): J={best_J:.6f}, K={K_cur:.0f}, lr={lr:.5f}")

    return best_a, best_J


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = evaluate_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 150:
        c = fast_autocorr_full(a)
        top_k = np.argsort(c)[::-1][:12]
        improved = False

        for ki in top_k:
            ki = int(ki)
            if time.monotonic() - t0 > time_limit:
                break
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_pos = set()
            for k2 in top_k[:8]:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                for pp in range(lo2, hi2 + 1):
                    if a[pp] * a[k2i - pp] > 0.4 * c.max():
                        peak_pos.add(pp)

            cold_list = [i for i in range(n) if i not in peak_pos and i != src]
            if not cold_list:
                cold_list = list(range(n))
            dst = cold_list[int(rng.integers(len(cold_list)))]

            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                J_t = evaluate_fast(a_t)
                if J_t < best_J - 1e-13:
                    a = a_t
                    best_J = J_t
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            for _ in range(8):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.02, 0.12)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    J_try = evaluate_fast(a)
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        stall = 0
                        break

    return best_a, best_J

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global

            parent = p.copy()
            parent = parent / parent.sum()
            p_score = evaluate_fast(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_ramp(n_base, power=0.5)
        best_score = evaluate_fast(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Phase 1: LP optimization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP optimization (budget {remaining:.0f}s) ===")
    t1 = min(remaining * 0.45, 750.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_optimize(a_in, t1, verbose=True, rng_seed=42)
        do_update(a_out, f"lp1_{n0}")

    # Phase 2: Smoothed gradient
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Smoothed gradient (budget {remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_optimize_smooth(a_in, t2, K0=30.0, lr0=0.004,
                                            verbose=True, rng_seed=7)
        do_update(a_out, f"gs_{n0}")

    # Phase 3: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP round 2 (budget {remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_optimize(a_in, t3, verbose=True, rng_seed=99)
        do_update(a_out, f"lp2_{n0}")

    # Phase 4: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate polish (budget {remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=False, rng_seed=123)
        do_update(a_out, f"coord_{n0}")

    # Phase 5: Final gradient + LP
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 5: Final round (budget {remaining:.0f}s) ===")
        t5a = remaining * 0.45
        t5b = remaining * 0.40
        if t5a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_optimize(a_in, t5a, verbose=True, rng_seed=555)
            do_update(a_out, f"lp_final_{n0}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t5b = min(remaining * 0.6, 120.0)
            if t5b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = gradient_optimize_smooth(a_in, t5b, K0=80.0, lr0=0.002,
                                                    verbose=True, rng_seed=777)
                do_update(a_out, f"gs_final_{n0}")

    # Phase 6: Multi-length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 6: Multi-length (budget {remaining:.0f}s) ===")
        for new_n in [4000, 5000, 4500]:
            if abs(new_n - n0) < 100:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            a_long = interpolate_seq(best_seq, new_n)
            do_update(a_long, f"interp_n{new_n}")
            t_alloc = min(remaining * 0.35, 150.0)
            if t_alloc > 15:
                a_out, _ = lp_optimize(a_long, t_alloc, verbose=False, rng_seed=42 + new_n % 100)
                do_update(a_out, f"lp_n{new_n}")

    if best_seq is not None:
        fs = evaluate_fast(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        best_seq = make_ramp(4200, power=0.5)
        best_score = evaluate_fast(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
