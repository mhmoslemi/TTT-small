"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based optimization (maximize sum(g) s.t. a*g <= M, g >= 0) with active-set
2. Ramp-based initializations (key insight: ramp gives ~4/3 vs uniform's 2)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_obj(a):
    """Compute 2n * max(conv) / sum^2 for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return 2.0 * n * c.max() / (s * s)


def evaluate_safe(a):
    """Safely call the external evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def precompute_lp_structure(n):
    """Precompute row/col structure and index mapping for LP constraints."""
    rows = []
    cols = []
    i_vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(k - i)
            i_vals.append(i)
    return (np.array(rows, dtype=np.int32),
            np.array(cols, dtype=np.int32),
            np.array(i_vals, dtype=np.int32))


def lp_solve(a, M, row_idx, col_idx, i_vals, n, top_k=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for selected k, g >= 0."""
    if top_k is not None and top_k < 2 * n - 1:
        aa = fast_conv(a)
        top_set = set(np.argsort(aa)[::-1][:top_k].tolist())
        mask = np.array([k in top_set for k in row_idx])
        r = row_idx[mask]
        c = col_idx[mask]
        v = a[i_vals[mask]]
        # Remap row indices to 0..top_k-1
        unique_rows = np.unique(r)
        row_map = {int(ur): idx for idx, ur in enumerate(unique_rows)}
        r_remapped = np.array([row_map[int(x)] for x in r], dtype=np.int32)
        A = csr_matrix((v, (r_remapped, c)), shape=(len(unique_rows), n))
        b = np.full(len(unique_rows), M)
    else:
        v = a[i_vals]
        A = csr_matrix((v, (row_idx, col_idx)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M)

    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60})
        if result.status != 0:
            return None
        return result.x
    except Exception:
        return None


def lp_optimize(a0, time_limit, verbose=False):
    """Iterative LP-based optimization with adaptive active set."""
    n = len(a0)
    a = norm_simplex(a0)
    c = fast_conv(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()

    row_idx, col_idx, i_vals = precompute_lp_structure(n)

    # Adaptive top_k: start with all, reduce over time
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 50.0)
        if progress < 0.1:
            cur_top_k = None  # All constraints
        elif progress < 0.4:
            cur_top_k = int((2 * n - 1) * (1 - 0.5 * progress))
        else:
            cur_top_k = max(30, int((2 * n - 1) * 0.3 * (1 - 0.7 * (progress - 0.4) / 0.6)))

        g = lp_solve(a, M, row_idx, col_idx, i_vals, n, top_k=cur_top_k)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1.0 + 1e-10:
            # No improvement possible from this LP
            break

        g = g / g_sum

        # Line search for best step size
        best_t = 0.0
        best_M_trial = M
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_t = t

        if best_t < 0.01:
            break

        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        c = fast_conv(a)
        M = c.max()

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {it} ({elapsed:.0f}s): obj={2*n*best_M:.6f}, g_sum={g_sum:.4f}, t={best_t:.2f}")

    return best_a, 2.0 * n * best_M


def ftrl_optimize(a0, time_limit, verbose=False):
    """FTRL mirror descent with annealed temperature - faster than LP."""
    n = len(a0)
    a = norm_simplex(a0)
    c = fast_conv(a)
    best_M = c.max()
    best_a = a.copy()
    t0 = time.monotonic()

    T_init = best_M * 0.5
    T_min = best_M * 0.01
    eta_base = 2.0

    x = a.copy()
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, t / 30000.0)
        T = T_init * (T_min / T_init) ** progress
        cur_beta = 15.0 + 50.0 * progress
        cur_top_k = max(3, int(20 * (1 - 0.8 * progress) + 3))

        # Nesterov
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Subgradient from top peaks
        c_y = fast_conv(y)
        k_max = min(cur_top_k, len(c_y))
        top_idx = np.argsort(c_y)[::-1][:k_max]
        top_vals = c_y[top_idx]
        max_v = top_vals[0]
        if max_v < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            exponents = cur_beta * (top_vals - max_v) / max_v
            exponents = np.maximum(exponents, -50.0)
            w = np.exp(exponents)
            w_sum = w.sum()
            w = w / w_sum if w_sum > 1e-30 else np.ones(k_max) / k_max

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        # Mirror descent step
        eta = eta_base / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new if s_new > 1e-15 else y.copy()

        x = a_new
        a = a_new

        c_a = fast_conv(a)
        M_a = c_a.max()
        if M_a < best_M - 1e-15:
            best_M = M_a
            best_a = a.copy()

        # Periodic spectral smoothing
        if t % 3000 == 0:
            a = spectral_smooth_one(a, strength=0.2)
            c_s = fast_conv(a)
            if c_s.max() < best_M:
                best_M = c_s.max()
                best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): obj={2*n*best_M:.6f}")

    return best_a, 2.0 * n * best_M


def spectral_smooth_one(a, strength=0.3):
    """One step of spectral flattening."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a

    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    a_out = (1.0 - strength * 0.5) * a + strength * 0.5 * a_new
    return norm_simplex(a_out)

def make_ramp(n, power=1.0):
    """Create a ramp-like sequence: a[i] = (i+1)^power."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_ramp_asym(n, pos=0.7):
    """Asymmetric ramp: peaks at position pos."""
    x = np.linspace(0, 1, n)
    a = np.where(x < pos, (x / pos) ** 1.5, ((1 - x) / (1 - pos)) ** 2.5)
    a = np.maximum(a, 0.01)
    return norm_simplex(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_quadratic_peak(n, peak_pos=0.6):
    """Quadratic with peak at peak_pos."""
    x = np.linspace(0, 1, n)
    a = 1.0 - ((x - peak_pos) / max(peak_pos, 1 - peak_pos)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_descent(a0, time_limit):
    """Local coordinate descent: move mass between positions."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_conv(a)
        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue

            for attempt in range(3):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.02, 0.08)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, 2.0 * n * best_M


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining LP, FTRL, spectral, and CD methods."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_obj(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_simplex(p)
            p_score = eval_obj(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Key insight: ramp gives ~4/3 for large n, much better than parent's 1.504
    n_base = 4200
    ramp = make_ramp(n_base, power=1.0)
    ramp_score = eval_obj(ramp)
    print(f"Ramp init: n={n_base}, score={ramp_score:.6f}")
    if ramp_score < best_score:
        best_score = ramp_score
        best_seq = ramp.copy()
        print(f"  *** Ramp beats parent: {ramp_score:.6f} vs {parent_score if 'parent_score' in dir() else 'N/A'}")

    # Create diverse initializations
    inits = []
    labels = []

    # Ramp variants (most promising)
    inits.append(make_ramp(n_base, power=1.0))
    labels.append("ramp_1")
    inits.append(make_ramp(n_base, power=0.7))
    labels.append("ramp_07")
    inits.append(make_ramp(n_base, power=1.3))
    labels.append("ramp_13")
    inits.append(make_ramp(n_base, power=0.5))
    labels.append("ramp_05")
    inits.append(make_ramp_asym(n_base, pos=0.65))
    labels.append("ramp_asym")
    inits.append(make_ramp_asym(n_base, pos=0.75))
    labels.append("ramp_asym2")
    inits.append(make_quadratic_peak(n_base, peak_pos=0.55))
    labels.append("quad_peak")
    inits.append(make_quadratic_peak(n_base, peak_pos=0.65))
    labels.append("quad_peak2")

    # Other shapes
    inits.append(make_cosine(n_base, mode=1))
    labels.append("cos1")
    inits.append(make_bowl(n_base, sharpness=2.0))
    labels.append("bowl2")

    if parent is not None:
        inits.append(interpolate_sequence(parent, n_base) if len(parent) != n_base else parent)
        labels.append("parent")

    # Evaluate all inits
    print("\n=== Phase 0: Evaluate initializations ===")
    for init, lbl in zip(inits, labels):
        sc = eval_obj(init)
        print(f"  {lbl}: {sc:.6f}")
        try_update(init, f"init_{lbl}")

    # === Phase 1: LP optimization on best inits ===
    print(f"\n=== Phase 1: LP Optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = remaining * 0.45

    # Sort inits by score, take top 5
    init_scores = [(eval_obj(init), init, lbl) for init, lbl in zip(inits, labels)]
    init_scores.sort(key=lambda x: x[0])
    top_inits = init_scores[:5]

    t_per_lp = t_phase1 / len(top_inits)
    for sc_init, init, lbl in top_inits:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_avail = min(t_per_lp, remaining * 0.8)
        if t_avail < 20:
            break
        print(f"  LP on '{lbl}' (score={sc_init:.6f}, budget={t_avail:.0f}s)...")
        a_out, obj_out = lp_optimize(init, t_avail, verbose=False)
        try_update(a_out, f"lp_{lbl}")

        # Spectral polish after LP
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sp, _ = spectral_optimize(a_out, min(20.0, remaining * 0.15))
            try_update(a_sp, f"spectral_lp_{lbl}")

    # === Phase 2: FTRL optimization (faster, more iterations) ===
    print(f"\n=== Phase 2: FTRL Optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = remaining * 0.25

    # FTRL on the current best
    best_n = len(best_seq)
    t_per_ftrl = t_phase2 / 3
    for i, lbl in enumerate(["best", "ramp_1", "ramp_asym"]):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        if lbl == "best":
            a_in = norm_simplex(best_seq)
        elif lbl == "ramp_1":
            a_in = make_ramp(best_n, power=1.0)
        else:
            a_in = make_ramp_asym(best_n, pos=0.7)

        t_avail = min(t_per_ftrl, remaining * 0.8)
        if t_avail < 15:
            break
        print(f"  FTRL on '{lbl}' (budget={t_avail:.0f}s)...")
        a_out, _ = ftrl_optimize(a_in, t_avail, verbose=False)
        try_update(a_out, f"ftrl_{lbl}")

        # Spectral polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_sp, _ = spectral_optimize(a_out, min(15.0, remaining * 0.1))
            try_update(a_sp, f"spectral_ftrl_{lbl}")

    # === Phase 3: Further LP on the best found so far ===
    print(f"\n=== Phase 3: LP Polish on Best ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase3 = min(remaining * 0.25, 300.0)
    if t_phase3 > 30:
        best_n = len(best_seq)
        a_in = norm_simplex(best_seq)
        print(f"  LP on best (n={best_n}, budget={t_phase3:.0f}s)...")
        a_out, _ = lp_optimize(a_in, t_phase3, verbose=True)
        try_update(a_out, "lp_polish_best")

        # FTRL after LP
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_f, _ = ftrl_optimize(a_out, min(60.0, remaining * 0.3))
            try_update(a_f, "ftrl_after_lp_polish")

    # === Phase 4: Coordinate descent final polish ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase4 = min(remaining * 0.15, 120.0)
    if t_phase4 > 10:
        best_n = len(best_seq)
        a_in = norm_simplex(best_seq)
        print(f"  CD (n={best_n}, budget={t_phase4:.0f}s)...")
        a_out, _ = coordinate_descent(a_in, t_phase4)
        try_update(a_out, "cd_final")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Length Variation ===")
    remaining = time_budget - (time.monotonic() - t_global)
    best_n = len(best_seq)

    # Try longer
    for delta in [500, 1000]:
        new_n = best_n + delta
        if new_n > 7000 or remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        # Quick FTRL on longer
        t_long = min(60.0, remaining * 0.2)
        if t_long > 15:
            a_out, _ = ftrl_optimize(a_long, t_long)
            try_update(a_out, f"ftrl_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)

    # Try shorter
    for delta in [500, 1000]:
        new_n = best_n - delta
        if new_n < 2000 or remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_short = interpolate_sequence(best_seq, new_n)
        try_update(a_short, f"interp_short_n{new_n}")

        t_short = min(40.0, remaining * 0.2)
        if t_short > 15:
            a_out, _ = ftrl_optimize(a_short, t_short)
            try_update(a_out, f"ftrl_short_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Final intensive optimization ===
    print(f"\n=== Phase 6: Final Optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        best_n = len(best_seq)
        a_in = norm_simplex(best_seq)

        # Mix of LP and FTRL
        t_final = remaining * 0.5
        if t_final > 20:
            a_out, _ = ftrl_optimize(a_in, t_final, verbose=True)
            try_update(a_out, "final_ftrl")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out, _ = lp_optimize(norm_simplex(best_seq), min(remaining * 0.5, 150.0))
            try_update(a_out, "final_lp")

        # Spectral + CD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sp, _ = spectral_optimize(norm_simplex(best_seq), min(20.0, remaining * 0.3))
            try_update(a_sp, "final_spectral")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_cd, _ = coordinate_descent(norm_simplex(best_seq), min(30.0, remaining * 0.3))
                try_update(a_cd, "final_cd")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_ramp(4000, power=1.0)
        best_score = eval_obj(best_seq)
        print(f"Fallback ramp: score={best_score:.6f}")

    return [float(x) for x in best_seq]
