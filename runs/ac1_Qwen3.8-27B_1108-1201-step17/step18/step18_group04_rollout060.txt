# Fast convolution using FFT
def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = next_power_of_two(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    return 1 << (n - 1).bit_length()

    if s < 1e-15:

    x = np.linspace(0, 1, n, endpoint=False)
    
    def make_seq(coeffs):
        # coeffs[0] is the DC offset (fixed), rest are cosine amplitudes
        a = np.ones(n) / n
        for k in range(1, K + 1):
            a += coeffs[k-1] * np.cos(2 * np.pi * k * x)
        a = np.maximum(a, 0)
        return a / a.sum()
    
    def objective(coeffs):
        a = make_seq(coeffs)
        c = fast_conv(a)
        return 2 * n * c.max()
    
    # Initial guess: all zeros
    x0 = np.zeros(K)
    
    # Use scipy to minimize

    result = minimize(objective, x0, method='L-BFGS-B',

def _np2(n):

def _conv(a):
    n = len(a)
    sz = 2 * n - 1
    fs = _np2(sz)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:sz]

def _norm(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones_like(a) / len(a)

def _J(a):

def _smax(c, t=0.01):
    m = c.max()
    return m + t * np.log(np.sum(np.exp((c - m) / t)))

def _cosine_opt(n, K, tlim=30.0):

    def build(c):
        a = 1.0 / n + c @ cos_basis
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    
    def obj(c):
        return 2.0 * n * _smax(_conv(build(c)), 0.003)
    
    t0 = time.monotonic()
    res = scipy_minimize(obj, np.zeros(K), method='L-BFGS-B',
                         options={'maxiter': 200, 'ftol': 1e-15})

            return 2.0 * n * _smax(_conv(build(c)), 0.001)

                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s = a_t.sum()
                    if s < 1e-15:
                        continue
                    a_t = a_t / s
                    Jt = _J(a_t)
                    if Jt < best_J - 1e-14:
                        a = a_t
                        best_J = Jt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 150:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    f = rng.uniform(0.01, 0.08)
                    amt = a[i] * f
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    if _J(a) < best_J:
                        best_J = _J(a)
                        best_a = a.copy()
                        stall = 0
                stall = 0

    return best_a, best_J


def _interp(a, new_n):
    on = len(a)
    if on == new_n:
        return a.copy()
    an = _norm(a)
    ox = np.linspace(0, 1, on)
    nx = np.linspace(0, 1, new_n)
    ai = np.interp(nx, ox, an)
    ai = np.maximum(ai, 0.0)
    s = ai.sum()
    return ai / s if s > 1e-15 else np.ones(new_n) / new_n


def _try_upd(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    al = [float(v) for v in a]
    try:
        sc = evaluate_sequence(al)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best['score']:
        _best['score'] = sc
        _best['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def _make_beta(n, alpha, beta):
    x = (np.arange(n) + 0.5) / n
    a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    return _norm(a)
