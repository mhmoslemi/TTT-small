def interp_search(a0, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a0)
    if old_n == new_n:
        return a0.copy()
    a_norm = a0 / a0.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_decreasing(n, power=0.5):

def make_increasing(n, power=0.5):

def make_mixed(n, pos1=0.2, pos2=0.65, width=0.1, ratio=1.0):

def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_bell(n, center=0.5, width=0.15):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.exp(-((x - center) / width) ** 2))


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_two_peak(n, p1=0.25, p2=0.75, w1=0.08, w2=0.1):

    return norm_sum1(a)


def make_edge_weighted(n, power=2.0):

    a = x ** power + (1 - x) ** power

    n_best = len(_best_tracker['seq'])
    print("\n=== Phase 5: Multi-n ===")
    candidate_ns = []
    for delta in [-500, -1000, -1500, 500, 1000, 2000, 3000]:
        new_n = n_best + delta
        if 1500 <= new_n <= 10000:
            candidate_ns.append(new_n)
    for new_n in candidate_ns[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print("  n=%d" % new_n)
        a_interp = interp_search(_best_tracker['seq'], new_n)
        try_update(a_interp, "intp%d" % new_n)
        t5a = min(remaining * 0.25, 120.0)
        if t5a > 10:
            a_out, _ = amd_optimize(a_interp, t5a, top_k_init=30,
                                    temp_init=0.003, c0_init=3.0,
                                    restart_every=4000, seed=777 + new_n % 1000,
                                    var_weight=0.12)
            try_update(a_out, "amd%d" % new_n)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t5b = min(remaining * 0.15, 80.0)
            if t5b > 8:
                a_out, _ = peak_surgery(a_interp, t5b, seed=555 + new_n % 1000)
                try_update(a_out, "surge%d" % new_n)

    # Phase 6: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print("\n=== Phase 6: Final Polish (n=%d) ===" % n_best)
    if remaining > 20:
        t6a = min(remaining * 0.35, 180.0)
        if t6a > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = amd_optimize(a_in, t6a, top_k_init=60,
                                    temp_init=0.001, c0_init=2.0,
                                    restart_every=6000, seed=2024, var_weight=0.05)
            try_update(a_out, "amd_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6b = min(remaining * 0.40, 150.0)
            if t6b > 8:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = peak_surgery(a_in, t6b, seed=31415)
                try_update(a_out, "surge_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6c = min(remaining * 0.50, 120.0)
            if t6c > 10:
                a_in = norm_sum1(_best_tracker['seq'])
                a_out, _ = simulated_annealing(a_in, t6c, seed=7777)
                try_update(a_out, "sa_final")

    # Final
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print("\nFINAL: score=%.6f, n=%d, t=%.0fs" %
              (_best_tracker['score'], len(_best_tracker['seq']), time.monotonic() - t_global))
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print("Fallback: score=%.6f" % _best_tracker['score'])

    return [float(v) for v in _best_tracker['seq']]
