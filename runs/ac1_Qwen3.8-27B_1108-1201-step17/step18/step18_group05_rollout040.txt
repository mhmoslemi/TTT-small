That's 2n-1 elements. ✓

And grad = 2 * c.

Great, this is efficient with FFT.

Now, for the optimization:
- Objective: minimize f(a) = 2*n * smoothed_max(conv(a,a)) subject to a >= 0, sum(a) = 1
- Smoothed max: tau * log(sum exp(b_m/tau)) where b = conv(a,a)
- Gradient: 2 * c where c = cross-correlation as computed above
- Projected gradient step with backtracking

The key hyperparameters are tau (smoothing) and eta (step size). As tau decreases, the smoothing becomes sharper and the gradient better approximates the true subgradient, but the function becomes less smooth (requiring smaller steps).

Strategy: Start with large tau (very smooth, large steps) and anneal to small tau (sharp, fine-tuning).

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    global _best_seq, _best_score

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        K_sel = min(K, 2*n-1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)

            continue

        if b_sum < 1e-15:

            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)

                best_alpha = alpha

        s_a = a.sum()
        if s_a < 1e-15:

def smooth_gradient_optimize(a0, time_limit, tau_start=50.0, tau_end=1.0, verbose=True):
    """Smoothed projected gradient with tau annealing and Nesterov."""
    global _best_seq, _best_score
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 0.5
    mom = 0.9  # momentum
    v = np.zeros(n)
    a_nest = a.copy()
    stall = 0
    
    n_iters = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        # Exponential annealing of tau
        tau = tau_start * (tau_end / tau_start) ** progress
        
        ac = fast_autocorr(a_nest)
        g = smooth_gradient(a_nest, tau)
        
        # Project gradient onto tangent space
        g_tan = g - g.mean()
        
        # Momentum update
        v = mom * v - eta * g_tan
        a_trial = a_nest + v
        a_new = project_simplex(a_trial)
        a_new = norm_sum1(a_new)
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            # Increase eta slightly
            eta = min(eta * 1.01, 2.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.8
                v *= 0.5
            if stall > 300:
                # Reset and add perturbation
                mom = 0.5
                eta = 0.3
                pert = np.random.normal(0, 0.005, n) * a_new
                a_pert = np.maximum(a_new + pert, 0.0)
                a_new = a_pert / a_pert.sum()
                stall = 150
        
        a_nest = a_new
        n_iters += 1
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={n_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2f}, eta={eta:.4f}")
    
    return best_a, best_m

def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, verbose=False):
    """Accelerated Mirror Descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = np.maximum(x + beta_t * (x - x_prev), 1e-15)
        y = y / y.sum()
        
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 3000:
            # Perturbation
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.03, n) * best_a
            x = np.maximum(best_a + pert, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            stall_count = 0
    
    return best_a, best_m
