Let me think about the spectral method more carefully.

The DFT of a is A = F*a (where F is the DFT matrix).
|A|^2 is the power spectrum.
The circular autocorrelation R = F^{-1} * |A|^2 (up to normalization).

The linear convolution C = a * a (length 2n-1).
The DFT of C (zero-padded to length >= 2n-1) is |A_padded|^2.

So if we make |A_padded(f)|^2 flat, the linear convolution will be flat.

But A_padded is the DFT of the zero-padded sequence, which is the DTFT of a sampled at discrete frequencies.

The Gerchberg-Saxton approach:
1. Start with an initial sequence a.
2. Compute its FFT (with zero-padding to length >= 2n-1).
3. Modify the magnitude to be (closer to) flat.
4. Take the inverse FFT.
5. Project back onto the constraint set (non-negative, length n, correct sum).
6. Repeat.

This is essentially what the spectral_flatten function in the previous code does. The question is how to do it better.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Diverse asymmetric initializations (power-law, exponential, polynomial ramps)
2. Zero-padded spectral flattening (Gerchberg-Saxton on linear convolution spectrum)

4. Simulated annealing restarts to escape local minima
5. Multi-length exploration and interpolation
Key insight: optimal sequences must be asymmetric; the convolution peak
(of the LINEAR convolution, not circular autocorrelation) can be reduced
below the Cauchy-Schwarz bound by exploiting non-palindromic structure.

import numpy as np
import time


def fast_linconv(a):
    """Compute linear convolution a*a using zero-padded FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective 2*n*max(conv)/(sum)^2 safely."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_linconv(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def init_power_law(n, power):
    """a[i] = (i+1)^power, normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)
    return normalize(x ** power)


def init_exp_law(n, rate):
    """a[i] = exp(rate * i/n), normalized."""
    x = np.linspace(0, 1, n)
    return normalize(np.exp(rate * x))


def init_poly(n, coeffs):
    """a[i] = poly(i/n) with given coefficients [c0, c1, c2, ...], normalized."""
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for k, c in enumerate(coeffs):
        a += c * x ** k
    return normalize(a)


def init_bump(n, center, width, asym):
    """Asymmetric Gaussian bump."""
    x = np.linspace(0, 1, n)
    left_width = width * (1.0 + asym)
    right_width = width * (1.0 - asym)
    a = np.where(x < center,
                 np.exp(-0.5 * ((x - center) / left_width) ** 2),
                 np.exp(-0.5 * ((x - center) / right_width) ** 2))
    return normalize(a)


def spectral_flatten(a0, time_limit, alpha_start=0.5, alpha_end=0.8,
                     iters=3000, verbose=False, rng_seed=42):
    """
    Gerchberg-Saxton on zero-padded spectrum.
    Compresses power spectrum dynamic range iteratively.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_score(a)
    best_m = fast_linconv(a).max() / (a.sum() ** 2)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()
    rng = np.random.default_rng(rng_seed)

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive alpha: start aggressive, become conservative
        frac = min(1.0, it / max(1, iters))
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Compress dynamic range: P_target = P_mean * (P/P_mean)^(1-alpha)
        ratio = P / P_mean
        P_target = P_mean * np.power(ratio, 1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Accept if better, otherwise blend
        conv_new = fast_linconv(a_new)
        m_new = conv_new.max() / (a_new.sum() ** 2)
        score_new = 2.0 * n * m_new

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            best_score = score_new
            a = a_new
        else:
            # Damped step
            blend = 0.2 + 0.3 * frac
            a_blend = (1.0 - blend) * a + blend * a_new
            a_blend = np.maximum(a_blend, 0.0)
            sb = a_blend.sum()
            if sb > 1e-15:
                a_blend = a_blend / sb
                m_blend = fast_linconv(a_blend).max() / (a_blend.sum() ** 2)
                if m_blend < best_m:
                    best_m = m_blend
                    best_a = a_blend.copy()
                    best_score = 2.0 * n * m_blend
                a = a_blend

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpecFlatten it={it} ({elapsed:.0f}s): score={2*n*best_m:.6f}, alpha={alpha:.3f}")

    return best_a, best_score


def gradient_refine(a0, time_limit, n_top=50, cool_rate=0.999,
                    lr0=0.05, verbose=False, rng_seed=42):
    """
    Gradient descent on max convolution peak using top-k weighted subgradient.
    Uses mirror descent (exponential update) for non-negativity.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_linconv(a).max() / (a.sum() ** 2)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    lr = lr0
    stall = 0
    restart_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_linconv(a)
        # Find top-k peaks (exclude k=0 and k=2n-2 which are just a[0]^2 and a[-1]^2)
        valid_range = slice(max(1, n // 10), min(len(conv) - 1, 2 * n - 2 - n // 10))
        conv_valid = np.concatenate([np.zeros(max(1, n // 10)),
                                     conv[valid_range],
                                     np.zeros(max(1, n // 10) + 1)])

        # Softmax weighting over top peaks
        top_k = min(n_top, len(conv_valid))
        log_conv = np.log(np.maximum(conv_valid, 1e-30))
        exp_w = np.exp(log_conv - log_conv.max())
        w = exp_w / exp_w.sum()

        # Compute subgradient: for each peak k with weight w_k,
        # d(conv[k])/d(a[j]) = a[k-j] (if valid) * 2 (or *1 if k=2j)
        grad = np.zeros(n)
        # Only process top peaks with significant weight
        active_idx = np.argsort(conv_valid)[::-1][:top_k]
        for ki in active_idx:
            ki = int(ki)
            wi = w[ki]
            if wi < 1e-8:
                continue
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            pair = ki - idx
            contrib = 2.0 * a[pair]
            # Fix the diagonal: where j == ki-j, i.e., 2j == ki
            if ki % 2 == 0:
                jdiag = ki // 2
                if j_lo <= jdiag <= j_hi:
                    pos = jdiag - j_lo
                    contrib[pos] = a[jdiag]  # single count
            grad[idx] += wi * contrib

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        # Mirror descent step (exponential update)
        log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        conv_new = fast_linconv(a_new)
        m_new = conv_new.max() / (a_new.sum() ** 2)

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            a = a_new  # still accept for exploration
            stall += 1
            # Try to recover
            if stall > 200:
                a = best_a.copy()

        # Decay learning rate
        lr *= cool_rate

        # Restart if stuck
        if stall > 3000:
            stall = 0
            restart_count += 1
            strength = min(0.2, 0.02 * restart_count)
            pert = rng.normal(0, strength, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            lr = lr0 * (0.5 ** restart_count)
            # Vary n_top
            n_top = max(10, min(200, n_top + rng.integers(-20, 30)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradRefine t={t} ({elapsed:.0f}s): score={2*n*best_m:.6f}, lr={lr:.6f}, stall={stall}")

    return best_a, 2.0 * n * best_m


def peak_targeted_descent(a0, time_limit, verbose=False, rng_seed=42):
    """
    Directly target the top peaks of the convolution.
    For each peak k, find which (i, k-i) pair contributes most,
    and redistribute mass from that position to a "safe" position.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_linconv(a).max() / (a.sum() ** 2)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        conv = fast_linconv(a)
        M = conv.max()
        m_curr = M / (a.sum() ** 2)

        # Get top 15 peaks
        top_k = np.argsort(conv)[::-1][:15]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find destination: minimize impact on top peaks
            costs = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                lo2 = max(0, k2i - n + 1)
                hi2 = min(n - 1, k2i)
                if lo2 <= hi2:
                    v2 = np.arange(lo2, hi2 + 1)
                    costs[v2] += a[k2i - v2]

            # Pick from low-cost, low-value positions
            combined = costs + 5.0 * a
            sorted_idx = np.argsort(combined)
            best_dst = sorted_idx[0]
            if best_dst == src:
                best_dst = sorted_idx[1] if len(sorted_idx) > 1 else int(rng.integers(n))
                if best_dst == src:
                    continue

            # Try several fractions
            found = False
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_linconv(a_trial).max() / (st ** 2)
                if m_trial < m_curr - 1e-14:
                    a = a_trial
                    if m_trial < best_m:
                        best_m = m_trial
                        best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Small random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_linconv(a).max() / (a.sum() ** 2)
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Larger perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.02, 0.1)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    return best_a, 2.0 * n * best_m


def interpolate_seq(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    return normalize(a_new)


def combine_sequences(a1, a2, alpha):
    """Blend two normalized sequences."""
    a = alpha * a1 + (1.0 - alpha) * a2
    return normalize(a)


def propose_candidate(budget_s=1800.0):
    """Main search function: finds best sequence within time budget."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_score(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = eval_score(normalize(parent))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = init_power_law(n_base, 0.5)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Spectral flattening from diverse inits ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - (elapsed - t_global)
    print(f"\n=== Phase 1: Spectral flattening (budget={remaining:.0f}s) ===")

    inits = []
    labels = []

    if parent is not None:
        inits.append(normalize(parent))
        labels.append("parent")

    # Asymmetric ramps - key insight: must be asymmetric
    inits.append(init_power_law(n0, 0.3))
    labels.append("pow03")
    inits.append(init_power_law(n0, 0.5))
    labels.append("pow05")
    inits.append(init_power_law(n0, 0.7))
    labels.append("pow07")
    inits.append(init_exp_law(n0, 1.5))
    labels.append("exp15")
    inits.append(init_exp_law(n0, 3.0))
    labels.append("exp30")
    inits.append(init_poly(n0, [1.0, 0.5, 0.3]))
    labels.append("poly_1_05_03")
    inits.append(init_poly(n0, [0.5, 1.0, 0.5]))
    labels.append("poly_05_1_05")
    inits.append(init_bump(n0, 0.7, 0.3, 0.5))
    labels.append("bump_70_03_05")
    inits.append(init_bump(n0, 0.6, 0.4, -0.3))
    labels.append("bump_60_04_neg03")

    t_per_init = max(10.0, remaining * 0.35 / len(inits))
    for idx, (init_a, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed - t_global > time_budget - 60:
            break
        t_avail = min(t_per_init, time_budget - 60 - (elapsed - t_global))
        if t_avail < 8:
            break
        print(f"  SpecFlatten '{lbl}' (t={t_avail:.0f}s)...")
        a_out, sc = spectral_flatten(init_a, t_avail, alpha_start=0.4, alpha_end=0.7,
                                     iters=2000, verbose=False, rng_seed=42 + idx)
        try_update(a_out, f"spec_{lbl}")

    # === Phase 2: Gradient refinement on best ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - (elapsed - t_global)
    print(f"\n=== Phase 2: Gradient refinement (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 300.0)
    if t2 > 30:
        print(f"  Gradient on best (t={t2:.0f}s)...")
        a_out, sc = gradient_refine(normalize(best_seq), t2, n_top=60,
                                    cool_rate=0.9995, lr0=0.03,
                                    verbose=True, rng_seed=42)
        try_update(a_out, "grad_best")

    # === Phase 3: Spectral + Gradient combo on top candidates ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - (elapsed - t_global)
    print(f"\n=== Phase 3: Combined spectral+gradient (budget={remaining:.0f}s) ===")

    # Run spectral on a few different inits, then gradient on results
    combo_inits = [
        (init_power_law(n0, 0.4), "combo_pow04"),
        (init_exp_law(n0, 2.0), "combo_exp20"),
        (init_poly(n0, [0.8, 0.6, 0.4, 0.2]), "combo_poly4"),
    ]
    t_per_combo = max(20.0, remaining * 0.25 / len(combo_inits))
    for init_a, lbl in combo_inits:
        elapsed = time.monotonic() - t_global
        if elapsed - t_global > time_budget - 60:
            break
        t_avail = min(t_per_combo, time_budget - 60 - (elapsed - t_global))
        if t_avail < 30:
            break
        # Spectral first
        t_spec = t_avail * 0.5
        t_grad = t_avail * 0.4
        a_out, _ = spectral_flatten(init_a, t_spec, alpha_start=0.5, alpha_end=0.8,
                                    iters=1000, verbose=False, rng_seed=99)
        a_out2, _ = gradient_refine(a_out, t_grad, n_top=40,
                                    cool_rate=0.999, lr0=0.02,
                                    verbose=False, rng_seed=77)
        try_update(a_out, f"spec_{lbl}")
        try_update(a_out2, f"spec+grad_{lbl}")

    # === Phase 4: Peak-targeted descent ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - (elapsed - t_global)
    print(f"\n=== Phase 4: Peak-targeted descent (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 20:
        print(f"  Peak descent (t={t4:.0f}s)...")
        a_out, sc = peak_targeted_descent(normalize(best_seq), t4,
                                          verbose=False, rng_seed=42)
        try_update(a_out, "peakdescent")

    # === Phase 5: Multi-length exploration ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - (elapsed - t_global)
    print(f"\n=== Phase 5: Multi-length (budget={remaining:.0f}s) ===")

    n_best = len(best_seq)
    lengths_to_try = []
    for delta in [-1000, -500, 500, 1000, 2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))

    for new_n in lengths_to_try[:3]:
        elapsed = time.monotonic() - t_global
        if elapsed - t_global > time_budget - 80:
            break
        remaining = time_budget - (elapsed - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_seq(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")

        t5 = min(remaining * 0.4, 120.0)
        if t5 > 30:
            a_out, _ = spectral_flatten(a_interp, t5 * 0.5, alpha_start=0.4,
                                        alpha_end=0.7, iters=800, verbose=False, rng_seed=13)
            try_update(a_out, f"spec_n{new_n}")
            t5b = t5 * 0.4
            if t5b > 15:
                a_out2, _ = gradient_refine(a_out, t5b, n_top=50,
                                            cool_rate=0.999, lr0=0.02,
                                            verbose=False, rng_seed=13)
                try_update(a_out2, f"spec+grad_n{new_n}")

    # === Phase 6: Final intensive polish ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - (elapsed - t_global)
    print(f"\n=== Phase 6: Final polish (budget={remaining:.0f}s) ===")
    n_best = len(best_seq)

    if remaining > 30:
        # Alternate between gradient and peak descent
        t6_pieces = 4
        t_piece = min(remaining * 0.9 / t6_pieces, 60.0)

        for piece in range(t6_pieces):
            elapsed = time.monotonic() - t_global
            if elapsed - t_global > time_budget - 20:
                break
            t_avail = min(t_piece, time_budget - 20 - (elapsed - t_global))
            if t_avail < 10:
                break

            if piece % 2 == 0:
                # Gradient with very slow cooling
                print(f"  Polish grad piece {piece} (t={t_avail:.0f}s)...")
                a_out, _ = gradient_refine(normalize(best_seq), t_avail,
                                           n_top=80, cool_rate=0.9998, lr0=0.01,
                                           verbose=False, rng_seed=100 + piece)
                try_update(a_out, f"polish_grad_{piece}")
            else:
                # Peak-targeted
                print(f"  Polish peak piece {piece} (t={t_avail:.0f}s)...")
                a_out, _ = peak_targeted_descent(normalize(best_seq), t_avail,
                                                 verbose=False, rng_seed=200 + piece)
                try_update(a_out, f"polish_peak_{piece}")

    # === Phase 7: Very fine gradient with minimal steps ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - (elapsed - t_global)
    if remaining > 20:
        t7 = min(remaining * 0.8, 60.0)
        print(f"\n=== Phase 7: Ultra-fine gradient (t={t7:.0f}s) ===")
        a_out, _ = gradient_refine(normalize(best_seq), t7,
                                   n_top=20, cool_rate=0.9999, lr0=0.005,
                                   verbose=True, rng_seed=999)
        try_update(a_out, "ultra_fine")

    # Final validation via actual evaluate_sequence
    if best_seq is not None:
        final_check = eval_score(best_seq)
        if np.isfinite(final_check):
            best_score = final_check
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = init_power_law(n_fb, 0.5)
        best_score = eval_score(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
