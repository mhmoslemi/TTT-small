def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def softmax_grad_fn(x, ac, beta):
    n = len(x)
    max_ac = ac.max()
    exp_ac = np.exp(beta * (ac - max_ac))
    w = exp_ac / exp_ac.sum()
    top_k = min(80, len(ac))
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_w = w[top_idx]
    if top_w.sum() < 1e-15:
        top_w = np.ones(top_k) / top_k
    grad = np.zeros(n)
    for wi, k in zip(top_w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * x[k_int - idx]
    return grad

            noise = rng.normal(0, 0.02 / (1.0 + 0.3 * div), n_best) * best_seq
            a_start = best_seq + noise
            a_start = np.maximum(a_start, 0.0)
            s_start = a_start.sum()
            if s_start < 1e-15:
                a_start = best_seq.copy()
            else:
                a_start = a_start / s_start
            a_out, _ = lp_fw(a_start, t5, solver,
                             initial_K=100 + 50 * div, max_K=1500 + 300 * div,
                             add_per_iter=20 + 10 * div, verbose=False)
            best_seq, best_score = try_update_global(best_seq, best_score, a_out, "div_lp" + str(div), t_global)

    # Phase 6: Try larger n
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Larger n exploration ===")
    longer_candidates = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_candidates = [l for l in longer_candidates if l <= 10000]
    for new_n in longer_candidates[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_seq(best_seq, new_n)
        t6 = min(remaining * 0.12, 120.0)
        if t6 > 30:
            a_out, _ = lp_fw(a_long, t6, solver,
                             initial_K=100, max_K=min(1500, new_n),
                             add_per_iter=15, verbose=False)
            best_seq, best_score = try_update_global(best_seq, best_score, a_out, "long_n" + str(new_n), t_global)

    # Phase 7: Final LP-FW polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 12000:
        t7 = min(remaining * 0.3, 300.0)
        if t7 > 30:
            a_out, _ = lp_fw(best_seq, t7, solver,
                             initial_K=300, max_K=2500,
                             add_per_iter=30, verbose=True)
            best_seq, best_score = try_update_global(best_seq, best_score, "lp_final" , t_global)
            # Hmm need to fix this - passing a_out not literal
            best_seq, best_score = try_update_global(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final coord polish (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 10:
            a_out, _ = coordinate_polish(best_seq, t8, verbose=True)
            best_seq, best_score = try_update_global(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + str(round(best_score, 8)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_local(best_seq)
        print("Fallback: score=" + str(round(best_score, 8)) + ", n=" + str(len(best_seq)))

    return [float(x) for x in best_seq]
