def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def _norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _lse_val_grad(a, tau):
    """Compute LSE smoothed value and gradient w.r.t. a."""
    n = len(a)
    ac = _fast_autocorr(a)
    ac_s = ac / tau
    ac_shift = ac_s.max()
    ac_s -= ac_shift
    exp_vals = np.exp(ac_s)
    p = exp_vals / exp_vals.sum()
    val = tau * (np.log(exp_vals.sum()) + ac_shift)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    Af = np.fft.rfft(a, fft_size)
    Pf = np.fft.rfft(p, fft_size)

    # t2[m] = sum_j a[j]*p[m-j] = conv(a,p)[m]
    conv_ap = np.fft.irfft(Af * Pf, fft_size)
    t2 = conv_ap[0:n]

    # t1[m] = sum_j a[j]*p[j+m] = conv(a, p_flip)[2n-2-m]
    pf = p[::-1]
    Pff = np.fft.rfft(pf, fft_size)
    conv_apf = np.fft.irfft(Af * Pff, fft_size)
    t1 = conv_apf[n - 2:2 * n - 1][::-1]

    grad = 2.0 * (t1 + t2)
    return val, grad, ac, p


def _softmax_from_x(x):
    x_sh = x - x.max()
    a = np.exp(x_sh)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def _lbfgs_smoothed(a0, time_limit, tau_start=None, n_phases=12, max_iter_per_phase=600, verbose=False):
    """L-BFGS-B on LSE smoothed objective with softmax param."""
    n = len(a0)
    a = _norm_sum1(a0)
    ac_init = _fast_autocorr(a)
    best_a = a.copy()
    best_m = ac_init.max()
    t0 = time.monotonic()

    if tau_start is None:
        tau = max(ac_init.max() * 0.4, 1e-6)
    else:
        tau = tau_start
    tau_min = max(ac_init.max() * 0.003, 1e-10)

    x = np.log(np.maximum(a, 1e-30))
    x -= x.max()

    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break

        tau_cur = tau

        def _obj(x_v):
            a_v = _softmax_from_x(x_v)
            ac_v = _fast_autocorr(a_v)
            ac_s = ac_v / tau_cur
            ac_s -= ac_s.max()
            return tau_cur * (np.log(np.sum(np.exp(ac_s))) + ac_s.max())

        def _grad(x_v):
            a_v = _softmax_from_x(x_v)
            _, g_a, _, _ = _lse_val_grad(a_v, tau_cur)
            dot = np.dot(g_a, a_v)
            return a_v * (g_a - dot)

        mi = min(max_iter_per_phase, 400)
        try:
            res = sp_minimize(_obj, x, jac=_grad, method='L-BFGS-B',
                              options={'maxiter': mi, 'maxfun': mi * 3,
                                       'ftol': 1e-14, 'gtol': 1e-10})
            x = np.asarray(res.x, dtype=np.float64)
        except Exception:
            pass

        a = _softmax_from_x(x)
        ac = _fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()

        tau = max(tau * 0.6, tau_min)

        if verbose and (phase % 3 == 0 or phase == n_phases - 1):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LBFGS ph=" + str(phase) + " t=" + str(round(elapsed)) + "s J=" + str(round(J, 6)) + " tau=" + str(format(tau, '.2e')))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LBFGS done t=" + str(round(elapsed)) + "s J=" + str(round(J, 6)))
    return best_a, best_m


def _lp_fw_ls(a0, time_limit, solver, initial_K=1000, max_K=5000, add_per_iter=80, verbose=True):
    """LP Frank-Wolfe with line search."""
    n = len(a0)
    y = _norm_sum1(a0)
    ac_init = _fast_autocorr(y)
    best_a = y.copy()
    best_m = ac_init.max()
    t0 = time.monotonic()

    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0

    for t in range(1, 999999):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix efficiently
        coeffs_matrix = np.zeros((K, n))
        for jj in range(K):
            k = constraint_list[jj]
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if K > 300:
                constraint_set = set(int(k) for k in constraint_list[:max(300, K // 3)])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if K > 300:
                constraint_set = set(int(k) for k in constraint_list[:max(300, K // 3)])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size -1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_obj = M_curr
        alphas = [2.0 / (t + 2.0), 0.1, 0.2, 0.3, 0.5, 0.05, 0.15, 0.4]
        for alpha in alphas:
            alpha = min(max(alpha, 0.01), 1.0)
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            M_trial = _fast_autocorr(y_trial).max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_alpha = alpha

        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = y.copy()
        else:
            y = y / sy

        iter_count += 1

        if verbose and t % 40 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW-LS it=" + str(t) + " t=" + str(round(elapsed)) + "s J=" + str(round(J, 6)) + " |S|=" + str(len(constraint_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW-LS done t=" + str(round(elapsed)) + "s J=" + str(round(J, 6)) + " it=" + str(iter_count))
    return best_a, best_m

def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _smooth_adam(a0, time_limit, tau_decay=0.9998, lr=0.7, verbose=False):
    """Multiplicative Adam with LSE smoothing (simpler version)."""

    v_hat = np.zeros(n)

    beta1 = 0.9

        if time.monotonic() - t0 > time_limit:
            break

        _, g, ac, _ = _lse_val_grad(a, tau)

        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)

        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2

        step = lr / (np.sqrt(v_c) + 1e-8)

        log_a = np.log(np.maximum(a, 1e-30)) - step * m_c
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)

        s = a_new.sum()
        a = a_new / s

        M = ac.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        tau = max(tau * tau_decay, tau_min)

        if stall > 2000:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a

            x = np.maximum(x, 1e-10)
            a = x / x.sum()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Adam t=" + str(round(elapsed)) + "s J=" + str(round(J, 6)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    Adam done t=" + str(round(elapsed)) + "s J=" + str(round(J, 6)))
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator. Budget: 1800s total."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = _get_solver()
    print("Total time budget: " + str(round(time_budget)) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = _norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + " score=" + str(round(p_score, 6)))
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = _norm_sum1(best_seq)
        best_score = _evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + " score=" + str(round(best_score, 6)))

    n0 = len(best_seq)

    # ==== Phase 1: LP-FW (main workhorse) ====
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: LP-FW (n=" + str(n0) + ", rem=" + str(round(remaining)) + "s) ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 60:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_fw_ls(a_in, t1, solver, initial_K=1000, max_K=5000, add_per_iter=100, verbose=True)
        sc = _evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  *** NEW BEST [lpfw1]: " + str(round(best_score, 6)) + " (n=" + str(len(a_out)) + ")")

    # ==== Phase 2: L-BFGS-B refinement ====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: L-BFGS-B (n=" + str(n_best) + ", rem=" + str(round(remaining)) + "s) ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 30:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lbfgs_smoothed(a_in, t2, n_phases=12, max_iter_per_phase=500, verbose=True)
        sc = _evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  *** NEW BEST [lbfgs1]: " + str(round(best_score, 6)) + " (n=" + str(len(a_out)) + ")")

    # ==== Phase 3: LP-FW round 2 ====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW r2 (n=" + str(n_best) + ", rem=" + str(round(remaining)) + "s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 40 and n_best <= 6000:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_fw_ls(a_in, t3, solver, initial_K=1500, max_K=6000, add_per_iter=120, verbose=True)
        sc = _evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  *** NEW BEST [lpfw2]: " + str(round(best_score, 6)) + " (n=" + str(len(a_out)) + ")")

    # ==== Phase 4: Smoothed Adam ====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Adam (n=" + str(n_best) + ", rem=" + str(round(remaining)) + "s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _smooth_adam(a_in, t4, tau_decay=0.9999, lr=0.5, verbose=False)
        sc = _evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  *** NEW BEST [adam]: " + str(round(best_score, 6)) + " (n=" + str(len(a_out)) + ")")

    # ==== Phase 5: Coordinate descent ====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Coord (n=" + str(n_best) + ", rem=" + str(round(remaining)) + "s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _coord_descent(a_in, t5, rng_seed=271)
        sc = _evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  *** NEW BEST [coord]: " + str(round(best_score, 6)) + " (n=" + str(len(a_out)) + ")")

    # ==== Phase 6: Try different n ====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-n (rem=" + str(round(remaining)) + "s) ===")

    nearby = []
    for delta in [500, -500, 1000, -1000, 300, 800]:
        new_n = n_best + delta
        if 2500 <= new_n <= 7000:
            nearby.append(new_n)

    seen = set()
    unique_nearby = []
    for ln in nearby:
        if ln not in seen:
            seen.add(ln)
            unique_nearby.append(ln)

    for new_n in unique_nearby[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("  Trying n=" + str(new_n) + " (rem=" + str(round(remaining)) + "s)")
        a_long = _interpolate_seq(best_seq, new_n)

        t6a = min(remaining * 0.35, 250.0)
        if t6a > 25:
            a_out, _ = _lbfgs_smoothed(a_long, t6a, n_phases=8, max_iter_per_phase=400, verbose=False)
            sc = _evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print("  *** NEW BEST [multin]: " + str(round(best_score, 6)) + " (n=" + str(new_n) + ")")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100 and new_n <= 5500:
            t6b = min(remaining * 0.40, 250.0)
            if t6b > 30:
                a_out, _ = _lp_fw_ls(_norm_sum1(best_seq), t6b, solver, initial_K=800,
                                      max_K=4000, add_per_iter=80, verbose=False)
                sc = _evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print("  *** NEW BEST [multin_lpfw]: " + str(round(best_score, 6)) + " (n=" + str(new_n) + ")")

    # ==== Phase 7: Final polish ====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final polish (n=" + str(n_best) + ", rem=" + str(round(remaining)) + "s) ===")
    if remaining > 40 and n_best <= 5500:
        t7 = min(remaining * 0.50, 200.0)
        if t7 > 30:
            a_in = _norm_sum1(best_seq)
            a_out, _ = _lp_fw_ls(a_in, t7, solver, initial_K=2000, max_K=6000,
                                  add_per_iter=150, verbose=False)
            sc = _evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print("  *** NEW BEST [final_lpfw]: " + str(round(best_score, 6)))

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.60, 100.0)
            if t7b > 10:
                a_in = _norm_sum1(best_seq)
                a_out, _ = _coord_descent(a_in, t7b, rng_seed=999)
                sc = _evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print("  *** NEW BEST [final_coord]: " + str(round(best_score, 6)))

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + str(round(best_score, 6)) + " n=" + str(len(best_seq)) + " time=" + str(round(elapsed)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = _norm_sum1(best_seq)
        best_score = _evaluate_safe(best_seq)
        print("Fallback: score=" + str(round(best_score, 6)))

    return [float(x) for x in best_seq]
