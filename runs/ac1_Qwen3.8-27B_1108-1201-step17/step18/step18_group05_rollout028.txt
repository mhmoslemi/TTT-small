```python
def spectral_anneal(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Zero-pad factor
    pad_n = 2 * n  # Pad to 2n for linear convolution
    
    it = 0
    alpha = 0.7  # Compression parameter (0 < alpha < 1)
    
    while time.monotonic() - t0 < time_limit:
        # Time to frequency
        a_padded = np.zeros(pad_n)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        power = np.abs(A) ** 2
        
        # Frequency domain: compress power spectrum
        # We want to reduce the peak of IFFT(power)
        # Strategy: apply a power-law compression
        p_nonzero = power[power > 1e-30]
        p_mean = p_nonzero.mean()
        # Compress: new_power = p_mean * (power/p_mean)^alpha
        # This reduces dynamic range
        ratio = power / p_mean
        ratio = np.maximum(ratio, 1e-30)
        new_power = p_mean * ratio ** alpha
        
        # Preserve DC component (to maintain sum)
        new_power[0] = power[0]
        
        # Also preserve the Hermitian symmetry
        # A[-k] = conj(A[k])
        # power is already real and symmetric, so new_power should be too
        # (it is, since we applied a real function to it)
        
        # RECON: compute new A with same phase, new magnitude
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        
        # Frequency to time
        a_new = np.fft.ifft(A_new).real
        a_new = a_new[:n]
        
        # Time domain: project onto non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            # Revert
            a_new = a.copy()
            break
```

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_T_structure(n):
    if n in _T_structure_cache:
        return _T_structure_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        length = j_end - j_start
        rows.extend([k] * length)
        cols.extend(range(j_start, j_end))
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    _T_structure_cache[n] = (rows, cols)
    return (rows, cols)


def solve_lp_higths(a, M):
    """Solve: max sum(b) s.t. conv_matrix(a) @ b <= M, b >= 0 using HiGHS."""
    n = len(a)
    rows, cols = get_T_structure(n)
    vals = a[rows - cols]
    A_mat = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None

def spectral_anneal(a0, time_limit, verbose=True, rng_seed=100):
    """Gerchberg-Saxton style: alternate time-domain (non-neg) and freq-domain (flattening)."""
    global _best_ever, _best_ever_score
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    pad_n = 2 * n
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        # Time to frequency (with zero-padding for linear convolution)
        a_padded = np.zeros(pad_n)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        power = np.abs(A) ** 2

        # Frequency domain: adaptive compression
        # Use a non-uniform compression that targets the peak
        p_nonzero = power[power > 1e-30]
        if len(p_nonzero) == 0:
            break
        p_median = np.median(p_nonzero)

        # Compute current autocorrelation peak
        ac = fast_autocorr(a)
        peak_frac = ac.max() / ac.sum()

        # Adaptive alpha: stronger compression when peak is large
        alpha = 0.3 + 0.4 * min(1.0, peak_frac * n / 5.0)
        alpha = min(alpha, 0.95)

        # Apply compression
        ratio = power / max(p_median, 1e-30)
        ratio = np.maximum(ratio, 1e-30)
        new_power = p_median * ratio ** alpha

        # Preserve DC
        new_power[0] = power[0]

        # Ensure Hermitian symmetry (power is already real and symmetric)
        # Reconstruct A
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)

        # Frequency to time
        a_new = np.fft.ifft(A_new).real
        a_new = a_new[:n]

        # Time domain projection
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            stall += 1
            if stall > 50:
                break
            continue
        a_new = a_new / s

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            sc = 2.0 * n * best_m
            if sc < _best_ever_score:
                _best_ever_score = sc
        else:
            # Try a different strategy: reduce peaks directly in freq domain
            stall += 1
            if stall > 30:
                # Perturbation
                pert = rng.normal(0, 0.01, n) * a
                a_pert = np.maximum(a + pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                    stall = 0
                else:
                    a = a_new
                    stall = 20
            else:
                a = a_new

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral iter={it} ({elapsed:.0f}s): J={J:.6f}, alpha={alpha:.3f}")

    return best_a, best_m

            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            sc = 2.0 * n * best_m
            if sc < _best_ever_score:
                _best_ever_score = sc
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-10)
            x = x_r / x_r.sum()
            x_prev = x.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Aggressive coordinate descent focusing on peak-contributing elements."""
    global _best_ever, _best_ever_score
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:15]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-contribution destinations
            dest_sens = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                ii2 = np.arange(s2, e2 + 1)
                dest_sens[ii2] += a[k2i - ii2]

            candidates = np.argsort(dest_sens)[:10]
            found = False
            for dst in candidates:
                if dst == src or found:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        sc = 2.0 * n * best_m
                        if sc < _best_ever_score:
                            _best_ever_score = sc
                        break

        if not improved:
            stall += 1
            # Random move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    sc = 2.0 * n * best_m
                    if sc < _best_ever_score:
                        _best_ever_score = sc

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    return norm_sum1(a)


def make_comb(n, p=2, duty=0.5):
    """Create a comb-like sequence."""
    a = np.zeros(n)
    for i in range(0, n, p):
        width = max(1, int(p * duty))
        a[i:i+width] = 1.0
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
