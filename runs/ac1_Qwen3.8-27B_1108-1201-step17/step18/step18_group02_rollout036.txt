Now, about the constraint matrix A:

This is the gradient of c[k] w.r.t. a at point y. Specifically:
c[k] = sum_i a[i]*a[k-i]

The linearization at y: c[k] ≈ c[k](y) + sum_m (dc[k]/da[m])(y) * (a[m] - y[m])
= c[k](y) + sum_m (y[k-m] + y[m-k]) * (a[m] - y[m])

The constraint c[k] <= M becomes (in the FW subproblem):
sum_m (y[k-m] + y[m-k]) * g[m] <= M - c[k](y) + sum_m (y[k-m] + y[m-k]) * y[m]
= M - c[k](y) + c[k](y)
= M

Wait, that doesn't look right. Let me redo:

The FW subproblem is: min_d sum_m y_grad[m]*d[m] s.t. sum(d) = 0 (or sum(g) = 1 where g = y + d), g >= 0

But the previous code formulates it as: min M s.t. sum(g) = 1, g >= 0, A@g <= M

where A[jj, i] = the gradient of c[k_j] w.r.t. a[i] at point y.

This is a valid FW variant: instead of minimizing the linearized objective, we minimize M such that the linearized constraints are satisfied. This is the "LP-type" Frank-Wolfe.

The issue is that A@g <= M is a linear approximation of c[k](g) <= M, which is only valid locally. If g is far from y, the approximation can be poor.

To fix this, the **quadratic FW** would use:
A@g + 0.5 * g^T Q g <= M

where Q is the Hessian of c[k] (which is constant for a given k). But this makes the subproblem a QP, which is more expensive.

A middle ground: **FW with line search**. After computing g (the LP solution), do a line search:
min_alpha c_max((1-alpha)*y + alpha*g)

This ensures we're making progress on the actual objective, not just the linearization.

            best_a = a.copy()
        
        tau = max(tau * 0.65, tau_min)
        
        if phase % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LBFGS phase={phase} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2e}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print(f"    LBFGS done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def _lp_fw_line_search(a0, time_limit, solver, initial_K=800, max_K=5000, add_per_iter=80):
    """LP Frank-Wolfe with quadratic line search on the actual objective."""
    n = len(a0)
    y = _norm_sum1(a0)
    ac_init = _fast_autocorr(y)
    best_a = y.copy()
    best_m = ac_init.max()
    t0 = time.monotonic()
    
    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = _fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Retry with fewer constraints
            if K > 200:
                constraint_set = set(int(k) for k in constraint_list[:max(200, K // 3)])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if K > 200:
                constraint_set = set(int(k) for k in constraint_list[:max(200, K // 3)])
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Line search: find best alpha in [0, 1]
        # Try a few values with backtracking
        best_alpha = 0.0
        best_obj = M_curr
        alphas = [2.0 / (t + 2.0), 0.1, 0.2, 0.3, 0.5, 0.05, 0.15]
        alphas = [min(max(a, 0.01), 1.0) for a in alphas]
        # Add the standard FW step
        alphas = list(set([round(a, 4) for a in alphas]))
        
        for alpha in alphas:
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            M_trial = _fast_autocorr(y_trial).max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_alpha = alpha
        
        if best_alpha == 0.0:
            best_alpha = 2.0 / (t + 2.0)
        
        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = y.copy()  # shouldn't happen
        else:
            y = y / sy
        
        iter_count += 1
        
        if t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW-LS iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print(f"    LP-FW-LS done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m

def _coord_descent(a0, time_limit, rng_seed=42):
    """Coordinate descent with optimal transfer search."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 150
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        M_curr = ac.max()
        
        n_top = min(12, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination
            scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]
            
            if dst == src:
                continue
            
            found = False
            for frac in [0.01, 0.02, 0.04, 0.08, 0.12, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = _fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and stall > 30:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = _fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = _fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if stall % 40 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _subgradient_polyak(a0, time_limit, n_iter=2000):
    """Subgradient method with Polyak step size for the max of quadratics."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    z = 1.0 / n  # lower bound on max autocorrelation (average)
    
    for t in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = _fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
        
        # Polyak step size
        gap = M_curr - z
        if gap < 1e-12:
            break
        step = gap / (2.0 * M_curr + 1e-10)
        
        # Subgradient: gradient of c[k*] at a, where k* = argmax
        k_star = int(np.argmax(ac))
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad[idx] += a[k_star - idx]
            # Symmetric part
            js2 = max(0, k_star - (n - 1))
            je2 = min(n - 1, k_star)
            if js2 <= je2:
                idx2 = np.arange(js2, je2 + 1)
                grad[idx2] += a[idx2 - (k_star - 2 * (n - 1))] if False else 0
        
        # Simpler: use the full gradient properly
        # grad[m] = a[k*-m]*1[0<=k*-m<n] + a[m-k*]*1[0<=m-k*<n]
        for m in range(n):
            if 0 <= k_star - m < n:
                grad[m] += a[k_star - m]
            if 0 <= m - k_star < n:
                grad[m] += a[m - k_star]
        
        # Projected subgradient step on the simplex
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check if we need to add noise to escape
        if (t + 1) % 500 == 0 and t > 100:
            noise = rng.normal(0, 0.01, n) * best_a
            a_new = a_new + noise
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
        
        a = a_new
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print(f"    Subgrad done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def _propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = _get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = _norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = _norm_sum1(best_seq)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    n0 = len(best_seq)
    
    # === Phase 1: L-BFGS-B with multiple temperature schedules ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L-BFGS-B (n={n0}, remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:

        # Multiple runs with different tau schedules

                break

            a_out, _ = _lbfgs_smoothed(a_in, t_run, tau_start=tau_init,
                                         n_phases=12, max_iter_per_phase=600)
            check_and_update(a_out, f"lbfgs_{run_idx}_n{n0}")
    
    # === Phase 2: LP-FW with line search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW LS (n={n_best}, remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60 and n_best <= 6000:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_fw_line_search(a_in, t2, solver, initial_K=1000,
                                       max_K=5000, add_per_iter=100)
        check_and_update(a_out, f"lpfwls_n{n_best}")
    
    # === Phase 3: L-BFGS-B refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

        check_and_update(a_out, f"subgrad_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

            check_and_update(a_out, f"lbfgs_n{new_n}")
