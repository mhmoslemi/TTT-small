    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:

def _fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _norm(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _lse_grad(a, tau):
    n = len(a)
    c = _fast_conv(a)
    c_max = c.max()
    p = np.exp((c - c_max) / tau)
    p /= p.sum()
    fft_size = 1 << (3 * n - 3).bit_length()
    P = np.fft.rfft(p.astype(np.float64), fft_size)
    A_rev = np.fft.rfft(a[::-1].copy().astype(np.float64), fft_size)
    conv_r = np.fft.irfft(P * A_rev, fft_size)
    grad = 2.0 * conv_r[n - 1: n - 1 + n]
    return grad


def _smooth_phase(a0, time_limit, tau_decay=0.9995, lr=0.7, verbose=True):
    n = len(a0)
    a = _norm(a0)
    c = _fast_conv(a)
    best_a = a.copy()
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    tau = max(c.max() * 0.4, 1e-6)
    tau_min = max(c.max() * 0.003, 1e-10)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    beta1, beta2 = 0.9, 0.999

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        c = _fast_conv(a)
        M = c.max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        tau = max(tau * tau_decay, tau_min)
        g = _lse_grad(a, tau)
        m_hat = beta1 * m_hat + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * (g * g)
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        step = lr / (np.sqrt(v_c) + 1e-8)
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        max_log = log_a.max()
        a = np.exp(log_a - max_log)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        if stall > 1500:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            a = x / x.sum()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
        if verbose and t % 800 == 0:
            print(f"    Smooth t={t}: M={best_m:.8f}, score={2*n*best_m:.6f}, tau={tau:.2e}")
    return best_a, best_m


def _lp_fw(a0, time_limit, solver, max_K=5000, verbose=True):
    n = len(a0)
    a = _norm(a0)
    c_init = _fast_conv(a)
    best_a = a.copy()
    best_m = c_init.max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()

    init_K = min(1500, 2 * n - 1)
    top_idx = np.argsort(c_init)[::-1][:init_K]
    constraint_set = set(int(k) for k in top_idx)

    iter_count = 0
    fail_count = 0
    g_var_cache = None

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        c_y = _fast_conv(y)
        M_y = c_y.max()
        if M_y < best_m - 1e-14:
            best_m = M_y
            best_a = y.copy()
        c_x = _fast_conv(x)
        M_x = c_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_y - c_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeff = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n, ki + 1)
            if js < je:
                idx = np.arange(js, je)
                coeff[jj, idx] = y[ki - idx]

        gv = cp.Variable(n, nonneg=True)
        Mv = cp.Variable()
        constraints = [cp.sum(gv) == 1, coeff @ gv <= Mv]
        prob = cp.Problem(cp.Minimize(Mv), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            fail_count += 1
            if fail_count > 5 and K > init_K:
                constraint_set = set(int(constraint_list[i]) for i in np.argsort([M_y - c_y[k] for k in constraint_list])[:init_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            fail_count += 1
            continue

        fail_count = 0
        g_val = np.maximum(np.asarray(gv.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs

        # Add violated constraints
        c_ys = _fast_conv(g_dir)
        viol = np.where(c_ys > M_y + 1e-10)[0]
        if len(viol) > 0:
            viol_amt = c_ys[viol] - M_y
            add_n = min(40, len(viol))
            top_v = viol[np.argsort(viol_amt)[::-1][:add_n]]
            for k in top_v:
                constraint_set.add(int(k))

        # Step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / sy if sy > 1e-15 else y.copy()

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            x_new = x_new / sx if sx > 1e-15 else y_new.copy()

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            el = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({el:.0f}s): score={2*n*best_m:.6f}, K={len(constraint_set)}")

    return best_a, best_m


def _coord_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = _norm(a0)
    c = _fast_conv(a)
    best_a = a.copy()
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 300:
            break
        c = _fast_conv(a)
        M_curr = c.max()
        top_k = np.argsort(c)[::-1][:10]
        improved = False

        for k in top_k[:6]:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination
            scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            cands = np.where(~in_peak)[0]
            if len(cands) == 0:
                cands = np.arange(n)
            dst = int(cands[np.argmin(scores[cands])])
            if dst == src:
                continue

            for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_tr = a.copy()
                a_tr[src] -= amt
                a_tr[dst] += amt
                a_tr = np.maximum(a_tr, 0.0)
                st = a_tr.sum()
                if st < 1e-15:
                    continue
                a_tr = a_tr / st
                Mt = _fast_conv(a_tr).max()
                if Mt < best_m - 1e-16:
                    a = a_tr
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    a[i] -= a[i] * frac
                    a[j] += a[i] * frac
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    Mn = _fast_conv(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        stall = 0
        if verbose and stall > 0 and stall % 100 == 0:
            print(f"    Coord ({time.monotonic()-t0:.0f}s): score={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Smooth2 (n={n_best}) ===")
        a_out, _ = _smooth_phase(_norm(best_seq), t3, tau_decay=0.9997, lr=0.5, verbose=True)
        _update(a_out, f"smooth2_n{n_best}")

    # Phase 4: Second LP-FW round
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 40 and n_best <= 6000:
        print(f"\n=== Phase 4: LP-FW2 (n={n_best}) ===")
        a_out, _ = _lp_fw(_norm(best_seq), t4, solver, max_K=6000, verbose=True)
        _update(a_out, f"lp2_n{n_best}")

    # Phase 5: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Coord polish (n={n_best}) ===")
        a_out, _ = _coord_polish(_norm(best_seq), t5, verbose=True)
        _update(a_out, f"coord_n{n_best}")

    # Phase 6: Try nearby lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby = []
    for d in [-400, -200, 200, 400]:
        nn = n_best + d
        if 3000 <= nn <= 6500 and nn != n_best:
            nearby.append(nn)
    seen = set()
    unique = []
    for ln in nearby:
        if ln not in seen:
            seen.add(ln)
            unique.append(ln)

    for new_n in unique[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 200:
            break
        print(f"\n=== Phase 6: n={new_n} ===")
        a_interp = _interpolate(best_seq, new_n)
        ta = min(remaining * 0.25, 150.0)
        if ta > 20:
            a_out, _ = _smooth_phase(a_interp, ta, tau_decay=0.9995, lr=0.6, verbose=False)
            _update(a_out, f"smooth_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 150 and new_n <= 5500:
            tb = min(remaining * 0.35, 250.0)
            if tb > 40:
                a_out, _ = _lp_fw(_norm(best_seq), tb, solver, max_K=4000, verbose=False)
                _update(a_out, f"lp_n{new_n}")

    # Phase 7: Final LP on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60 and n_best <= 5500:
        t7 = min(remaining * 0.40, 250.0)
        if t7 > 30:
            print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
            a_out, _ = _lp_fw(_norm(best_seq), t7, solver, max_K=5000, verbose=True)
            _update(a_out, f"lp_final_n{n_best}")

    # Phase 8: Final coord
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.60, 100.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final coord (n={n_best}) ===")
            a_out, _ = _coord_polish(_norm(best_seq), t8, verbose=False)
            _update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        el = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={el:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = _norm(best_seq)
        best_score = _eval_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
