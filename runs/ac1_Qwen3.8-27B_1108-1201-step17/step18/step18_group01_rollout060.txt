"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Lp-norm smoothed mirror descent on the simplex (FFT-based gradients)
2. p-annealing from smooth (p=2) to sharp (p large)
3. Symmetric optimization variant (a[i] = a[n-1-i])
4. Multi-length exploration
5. Coordinate descent and sparsification post-processing

import numpy as np
import time

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def compute_lp_grad(a, p, fft_size=None, A_fft=None):
    """
    Compute the gradient direction for ||conv(a,a)||_p.
    Returns (grad_projected, c) where grad is projected onto simplex tangent.
    """
    n = len(a)
    c = fast_autocorr(a)
    c_clip = np.maximum(c, 1e-30)
    w = c_clip ** (p - 1)
    
    if fft_size is None:
        fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    h = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    
    # Project onto simplex tangent (mirror descent natural gradient)
    grad = h - np.sum(h * a)
    return grad, c

def symmetric_mirror_descent(a0, time_limit, p_start=2.0, p_end=500.0, 
                              eta_start=0.05, eta_end=0.001,
                              verbose=True, seed=42):
    """
    Mirror descent on the simplex with Lp-smoothed objective.
    Optionally enforces symmetry a[i] = a[n-1-i].
    """
    n = len(a0)
    a = norm_sum1(a0)
    # Enforce symmetry if n is odd (center element is free)
    a = 0.5 * (a + a[::-1])
    a = norm_sum1(a)
    
    best_a = a.copy()
    best_c = fast_autocorr(a)
    best_max = best_c.max()
    
    t0 = time.monotonic()
    eta = eta_start
    p = p_start
    patience = 0
    max_patience = 500
    fft_size = 1 << (3 * n - 1).bit_length()
    iter_count = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        grad, c = compute_lp_grad(a, p, fft_size)
        cur_max = c.max()
        
        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Reset with perturbation
            a = best_a.copy()
            idx = rng.integers(n)
            a[idx] *= 0.9
            a = norm_sum1(a)
            a = 0.5 * (a + a[::-1])
            a = norm_sum1(a)
            patience = 0
            eta = eta_start * 0.5
        
        # Compute step size (use gradient magnitude for normalization)
        g_norm = np.sqrt(np.sum(grad ** 2))
        if g_norm < 1e-20:
            break
        
        # Mirror descent update
        a_new = a * np.exp(-eta * grad / g_norm)
        a_new = np.maximum(a_new, 0.0)
        # Enforce symmetry
        a_new = 0.5 * (a_new + a_new[::-1])
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check if update improved
        c_new = fast_autocorr(a_new)
        max_new = c_new.max()
        
        if max_new <= cur_max + 1e-14:
            a = a_new
        else:
            # Rejected, reduce step size
            eta *= 0.7
        
        # Anneal p
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        p = p_start * (p_end / p_start) ** progress
        
        # Slowly decrease eta
        eta = max(eta_end, eta_start * (eta_end / eta_start) ** (0.3 * progress))
        
        iter_count += 1
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_max
            print(f"      SMD iter={iter_count} ({elapsed:.0f}s): score={score:.6f}, p={p:.1f}, eta={eta:.5f}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        score = 2.0 * n * best_max
        print(f"    SMD done ({elapsed:.0f}s): score={score:.6f}, iters={iter_count}")
    return best_a

def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass from peak-contributing positions to low-impact ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        top_k = np.argsort(ac)[::-1][:8]
        improved = False
        
        for k in top_k[:4]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination
            peak_contrib = np.zeros(n)
            for k2 in top_k[:4]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]
            
            dst = np.argmin(peak_contrib)
            if dst == src:
                continue
            
            for frac in [0.01, 0.02, 0.04, 0.08]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        score = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): score={score:.6f}")
    return best_a

def sparsify(a, n_zeros=50, verbose=False):
    """Try zeroing out small entries to reduce peak."""
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    # Sort by value, try zeroing smallest
    order = np.argsort(a)
    a_test = a.copy()
    for i in range(min(n_zeros, n // 10)):
        a_test[order[i]] = 0.0
        s = a_test.sum()
        if s < 1e-15:
            break
        a_test = a_test / s
        m = fast_autocorr(a_test).max()
        if m < best_m:
            best_m = m
            best_a = a_test.copy()
        else:
            # Keep zeroed but try next

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Lp-norm smoothed mirror descent on the simplex (FFT-based cross-correlation gradients)
2. p-annealing from smooth (p=2) to sharp (p large) for progressive peak focusing
3. Symmetric optimization variant (a[i] = a[n-1-i])
4. Multi-length exploration with parametric initializations
5. Coordinate descent and simulated annealing post-processing
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_lp_grad(a, p, fft_size=None):
    """
    Compute the gradient direction for ||conv(a,a)||_p, projected onto simplex tangent.
    Uses numerically stable log-space computation for the weights.
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n), c
    c_norm = np.maximum(c / c_max, 1e-30)

    # Weights w[k] proportional to c[k]^(p-1), computed in log-space for stability
    log_w = (p - 1) * np.log(c_norm)
    log_w_max = log_w.max()
    w = np.exp(log_w - log_w_max)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum

    # Cross-correlation: h[i] = sum_k w[k] * a[k-i] for i=0,...,n-1
    if fft_size is None:
        fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    h = np.fft.irfft(W * np.conj(A), fft_size)[:n]

    # Project onto simplex tangent (natural gradient for mirror descent)
    grad = h - np.sum(h * a)
    return grad, c


def symmetric_mirror_descent(a0, time_limit, p_start=2.0, p_end=200.0,
                             eta_start=0.05, eta_end=0.001,
                             verbose=True, seed=42):

    Mirror descent on the simplex with Lp-smoothed objective.
    Enforces symmetry a[i] = a[n-1-i].
    """
    n = len(a0)
    a = norm_sum1(a0)
    a = 0.5 * (a + a[::-1])
    a = norm_sum1(a)

    best_a = a.copy()
    best_max = fast_autocorr(a).max()

    t0 = time.monotonic()
    eta = eta_start
    p = p_start
    patience = 0
    max_patience = 600
    fft_size = 1 << (3 * n - 1).bit_length()
    iter_count = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        grad, c = compute_lp_grad(a, p, fft_size)
        cur_max = c.max()

        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            # Reset with perturbation to escape local minimum
            a = best_a.copy()
            n_pert = max(5, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= rng.uniform(0.5, 1.3, n_pert)
            a = 0.5 * (a + a[::-1])
            a = norm_sum1(a)
            patience = 0
            eta = max(eta_start * 0.5, 0.01)

        g_norm = np.sqrt(np.sum(grad ** 2))
        if g_norm < 1e-20:
            break

        # Mirror descent update with clipping for numerical stability
        log_update = -eta * grad / g_norm
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0.0)
        a_new = 0.5 * (a_new + a_new[::-1])
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        c_new = fast_autocorr(a_new)
        max_new = c_new.max()

        if max_new <= cur_max + 1e-14:
            a = a_new
        else:
            eta *= 0.7

        # Anneal p logarithmically over time
        progress = min(1.0, (time.monotonic() - t0) / max(time_limit, 1e-9))
        p = p_start * (p_end / p_start) ** progress

        # Slowly decrease base eta
        eta_floor = max(eta_end, eta_start * (eta_end / eta_start) ** (0.5 * progress))
        if eta < eta_floor:

        print(f"    SMD done ({elapsed:.0f}s): score={score:.6f}, iters={iter_count}")
