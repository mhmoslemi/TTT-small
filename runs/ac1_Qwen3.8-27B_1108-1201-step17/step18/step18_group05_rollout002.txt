```python
def compute_smoothed_grad(a, tau, top_k=None):
    """Compute gradient of smoothed max: (1/tau)*log(sum exp(tau*(a*a)_k))"""
    n = len(a)
    ac = fast_autocorr(a)
    if top_k is not None and top_k < len(ac):
        # Only use top k for speed (approximation)
        sorted_idx = np.argsort(ac)[::-1][:top_k]
        ac_sel = ac[sorted_idx]
        p = np.exp(tau * (ac_sel - ac_sel.max()))
        p = p / p.sum()
        # Compute weighted gradient
        grad = np.zeros(n)
        for pi, ki in zip(p, sorted_idx):
            ki = int(ki)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            grad[idx] += pi * a[ki - idx]
    else:
        p = np.exp(tau * (ac - ac.max()))
        p = p / p.sum()

        # This is a cross-correlation of p and a
        grad = sp_correlate(p, a, mode='valid')
    return grad, ac

    tau = 10.0
    tau_max = 500.0

    eta = 1.0
    eta_min = 0.01

    max_stall = 2000
    iter_count = 0

        t_val = iter_count + 1
        beta_prev = (1 + momentum) / 2  # or (1+m)/(1+2m)
        y = a + (beta_prev - 1) * (a - y) if iter_count > 0 else a
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
```

```python
def pairwise_fw_step(a, grad, alpha):
    """One pairwise FW step: move mass from high-grad coord to low-grad coord."""
    n = len(a)
    # Find top and bottom coordinates
    top_i = np.zeros(n)*(-1)  # initialize
    top_50 = np.argsort(grad)[::-1][:min(50, n)]
    bot_50 = np.argsort(grad)[:min(50, n)]
    
    best_improve = -np.inf
    best_pair = None
    best_amt = 0
    
    for i in top_50:
        if a[i] < 1e-10:
            continue
        for j in bot_50:
            if i == j:
                continue
            improve = (grad[j] - grad[i]) * a[i]  # directional derivative * amount available
            # Actually the improvement per unit moved is (grad_j - grad_i)
            # We want to move from i to j if grad_i > grad_j
            # The improvement is proportional to (grad_i - grad_j) * amt
            diff = grad[i] - grad[j]
            if diff > 0:
                # Try different amounts
                for frac in [0.01, 0.005, 0.002, 0.05]:
                    amt = a[i] * frac
                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    # Quick check (full autocorr is expensive, use approximation)
                    # For now, just accept if diff is large
                    if diff * amt > best_improve:
                        best_improve = diff * amt
                        best_pair = (i, j)
                        best_amt = amt
    
    if best_pair is None:
        return a
    
    i, j = best_pair
    a[j] += best_amt
    a[i] -= best_amt
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a
```

```python
"""

import numpy as np

from scipy.signal import correlate as sp_correlate


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

    except Exception:

def compute_smoothed_grad(a, tau, top_k=None):
    """Compute gradient of smoothed max objective and return autocorr."""
    n = len(a)
    ac = fast_autocorr(a)
    if top_k is not None and top_k < len(ac):
        sorted_idx = np.argsort(ac)[::-1][:top_k]
        ac_sel = ac[sorted_idx]
        shift = ac_sel.max()
        p = np.exp(tau * (ac_sel - shift))
        p = p / p.sum()
        grad = np.zeros(n)
        for ii in range(len(sorted_idx)):
            pi = p[ii]
            ki = int(sorted_idx[ii])
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            grad[idx] += pi * a[ki - idx]
    else:
        p = np.exp(tau * (ac - ac.max()))
        p = p / p.sum()
        grad = sp_correlate(p, a, mode='valid')
    return grad, ac


def compute_subgradient(a, ac, top_k=20):
    """Compute subgradient at top peaks of autocorrelation."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** 4
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for ii in range(k_max):
        wi = w[ii]
        ki = int(top_idx[ii])
        s = max(0, ki - n + 1)
        e = min(n - 1, ki)
        idx = np.arange(s, e + 1)
        grad[idx] += wi * a[ki - idx]
    return grad


def nesterov_smoothed(a0, time_limit, verbose=True, rng_seed=0):
    """Nesterov-accelerated projected gradient on smoothed max."""
    n = len(a0)
    a = norm_sum1(a0)
    y = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    tau = 15.0
    tau_max = 600.0
    eta = 0.8
    momentum = 0.9
    stall = 0
    max_stall = 3000
    it = 0
    top_k = min(100, 2 * n - 1)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Nesterov extrapolation
        if it > 0:
            beta_n = (1.0 + momentum) / 2.0
            y = a + (beta_n - 1.0) * (a - y)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a.copy()

        grad, ac = compute_smoothed_grad(y, tau, top_k=top_k)
        grad_tan = grad - grad.mean()
        grad_norm = np.sqrt(np.sum(grad_tan ** 2))

            break

        alpha = eta
        f_curr = ac.max()
        accepted = False

            m_trial = fast_autocorr(a_trial).max()

                a_new = a_trial
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            a_new = project_simplex(a - alpha * grad_tan)

        a = a_new
        m_new = fast_autocorr(a).max()

        if m_new < best_m - 1e-16:
            best_m = m_new

            stall = 0

                eta *= 0.85

                a_p = np.maximum(a + pert, 0)

                if sp_ > 1e-15:
                    a_p = a_p / sp_
                m_p = fast_autocorr(a_p).max()
                if m_p < best_m:
                    best_m = m_p
                    best_a = a_p.copy()
                    a = a_p
                    y = a_p.copy()
                    stall = 0

                    a = a_new

            print(f"    NG iter={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

def peak_coord_descent(a0, time_limit, verbose=True, rng_seed=200):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    t6_total = min(remaining * 0.10, 180.0)
    trial_lengths = []
    for delta in [400, 800, -400, 1200, -800]:
        new_n = n_best + delta
        if 1000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best][:2]

    for new_n in trial_lengths:
        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_avail = min(remaining * 0.45, 120.0)
        if t_avail < 20:
            break
        print(f"  Trying n={new_n} (t={t_avail:.0f}s)...")
        a_long = interpolate_sequence(best_seq, new_n)
        a_out, _ = nesterov_smoothed(a_long, t_avail * 0.6, verbose=False, rng_seed=new_n % 1000)
        try_update(a_out, f"n{new_n}_ng")

        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 5000:
            t_avail2 = min(remaining * 0.4, 80.0)
            if t_avail2 > 15:
                a_in2 = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out2, _ = peak_coord_descent(a_in2, t_avail2, verbose=False, rng_seed=new_n % 2000)
                try_update(a_out2, f"n{new_n}_pcd")

    # === Phase 7: Final coordinated optimization (10% time) ===
    remaining = BUDGET - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final optimization (n={n_best}) ===")
    if remaining > 40:
        # Nesterov with sharp tau
        t7a = min(remaining * 0.4, 100.0)
        if t7a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_smoothed(a_in, t7a, verbose=False, rng_seed=999)
            try_update(a_out, "final_ng")

        # Pairwise FW polish
        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.35, 80.0)
            if t7b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = pairwise_fw(a_in, t7b, verbose=False, rng_seed=777)
                try_update(a_out, "final_pfw")

        # Coordinate polish
        remaining = BUDGET - (time.monotonic() - t_global)
        if remaining > 20:
            t7c = min(remaining * 0.3, 60.0)
            if t7c > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = peak_coord_descent(a_in, t7c, verbose=False, rng_seed=888)
                try_update(a_out, "final_pcd")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
