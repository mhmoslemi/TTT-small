    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, c, top_k=30, beta=50.0):
    """Compute subgradient using top peaks with softmax weighting.
    dC_k/da_j = 2*a[k-j] when 0 <= k-j < n

def spectral_equalize(a, strength=0.3):
    """Flatten the spectrum of a to reduce autocorrelation peaks."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    # Compress dynamic range of magnitude
    mag_clipped = np.maximum(mag, 1e-30)
    log_mag = np.log(mag_clipped)
    log_mag_mean = log_mag.mean()
    # Toward uniform log-magnitude
    log_mag_new = (1 - strength) * log_mag + strength * log_mag_mean
    mag_new = np.exp(log_mag_new)

    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return a_new


def peak_shaving(a0, time_budget, verbose=False):
    """Local search: reduce the top peak by moving mass from its contributors."""
    n = len(a0)
    a = norm_simplex(a0.copy())
    c = fast_conv(a)
    best_m = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_budget and stall < 200:
        c = fast_conv(a)
        
        # Find top peaks
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_budget:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            # Try moving mass from the highest contributor
            for _ in range(3):
                src = idx[np.argmax(contrib)]
                src_val = a[src]
                if src_val < 1e-10:
                    break
                
                # Move to a random position that doesn't contribute to this peak
                dst = int(rng.integers(n))
                if dst == src or (i_start <= dst < i_end and (k_int - dst) >= i_start and (k_int - dst) < i_end):
                    continue
                
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                    elif M_trial < c[k] + 0.01 * c[k]:
                        # Accept small increases to escape local optima
                        if rng.random() < 0.1:
                            a = a_trial
                            improved = True
                            break
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.05, 0.15)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PS ({elapsed:.0f}s): M={best_m:.10f}, stall={stall}")

    return best_a, best_m

    """Step function: 1 on left frac, ratio on right."""
    a = np.ones(n)
    split = int(n * frac)
    a[split:] = ratio
    return norm_simplex(a)


def make_init_hump(n, width=0.3, pos=0.5):
    """Single Gaussian hump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos) ** 2) / (2 * width ** 2)) + 0.1
    return norm_simplex(a)


def make_init_smooth(n, freq=3, amp=0.5):
    """Smooth periodic perturbation on uniform."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) + amp * np.sin(2 * np.pi * freq * x)
    a = np.maximum(a, 0.05)
    return norm_simplex(a)


def make_init_quadratic(n, asym=0.0):
    """Quadratic with optional asymmetry."""
    x = np.linspace(0, 1, n)
    a = (x - 0.5 + asym) ** 2 + 0.1
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_power(n, p=1.0):
    """Power function: x^p on [0,1]."""
    x = np.linspace(0.01, 1.0, n)
    a = x ** p
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def combined_optimize(a0, time_budget, verbose=False):
    """Run mirror descent followed by peak shaving."""
    n = len(a0)
    t0 = time.monotonic()
    
    # Stage 1: Mirror descent (70% of time)
    t_md = time_budget * 0.7
    a1, m1 = mirror_descent(a0, t_md, verbose=verbose)
    
    # Spectral equalization
    a_sp = a1.copy()
    for _ in range(5):
        if time.monotonic() - t0 > time_budget * 0.8:
            break
        a_sp = spectral_equalize(a_sp, strength=0.4)
        c_sp = fast_conv(a_sp)
        if c_sp.max() < m1:
            m1 = c_sp.max()
            a1 = a_sp.copy()
    
    # Stage 2: Peak shaving (30% of time)
    t_ps = time_budget - (time.monotonic() - t0)
    if t_ps > 5:
        a2, m2 = peak_shaving(a1, t_ps, verbose=verbose)
        if m2 < m1:
            return a2, m2
    
    return a1, m1


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_init_ramp(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    print(f"Starting: n={len(best_seq)}, score={best_score:.6f}, budget={time_budget:.0f}s")

    # === Phase 1: Multi-start mirror descent with diverse initializations ===
    print(f"\n=== Phase 1: Multi-start MD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.5, 700.0)

    if t_phase1 > 100:
        n_base = len(best_seq)
        # Try the parent length and a couple others
        lengths_to_try = [n_base]
        if n_base != 3500:
            lengths_to_try.append(3500)
        if n_base != 5000:
            lengths_to_try.append(5000)
        if n_base != 7000:
            lengths_to_try.append(7000)

        # Limit to 3 lengths for time management
        lengths_to_try = lengths_to_try[:3]

        inits_per_length = {}
        for n_try in lengths_to_try:
            inits = []
            labels = []
            
            # Key insight: ramps and steps are better than bowls!
            inits.append(make_init_ramp(n_try, direction=1))
            labels.append("ramp_up")
            
            inits.append(make_init_ramp(n_try, direction=-1))
            labels.append("ramp_dn")
            
            inits.append(make_init_step(n_try, frac=0.6, ratio=2.0))
            labels.append("step_06_2x")
            
            inits.append(make_init_step(n_try, frac=0.4, ratio=2.5))
            labels.append("step_04_25x")
            
            inits.append(make_init_hump(n_try, width=0.4, pos=0.3))
            labels.append("hump_03")
            
            inits.append(make_init_power(n_try, p=1.5))
            labels.append("pow15")
            
            inits.append(make_init_quadratic(n_try, asym=0.1))
            labels.append("quad_asym")
            
            # Add parent interpolated
            if parent is not None:
                inits.append(interpolate_sequence(parent, n_try))
                labels.append("parent_interp")
            
            inits_per_length[n_try] = (inits, labels)

        total_inits = sum(len(v[0]) for v in inits_per_length.values())
        t_per = t_phase1 / max(1, total_inits)
        print(f"  {total_inits} initializations across {len(lengths_to_try)} lengths, ~{t_per:.0f}s each")

        for n_try in lengths_to_try:
            inits, labels = inits_per_length[n_try]
            for idx, (init, lbl) in enumerate(zip(inits, labels)):
                elapsed_so_far = time.monotonic() - t_global
                remaining_now = time_budget - 60 - elapsed_so_far
                if remaining_now < 20:
                    break
                t_avail = min(t_per, remaining_now * 0.9)
                if t_avail < 10:
                    break
                print(f"    MD n={n_try} '{lbl}' (budget={t_avail:.0f}s)...")
                a_out, _ = mirror_descent(init, t_avail, verbose=False)
                try_update(a_out, f"md_{lbl}_{n_try}")
                # Quick spectral + peak shaving
                remaining_now = time_budget - 60 - (time.monotonic() - t_global)
                if remaining_now > 15:
                    a_sp = spectral_equalize(a_out, strength=0.5)
                    try_update(a_sp, f"spec_{lbl}_{n_try}")
                    if remaining_now > 10:
                        a_ps, _ = peak_shaving(a_out, min(10.0, remaining_now * 0.4))
                        try_update(a_ps, f"ps_{lbl}_{n_try}")

    # === Phase 2: Intensive refinement on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Intensive refinement (n={n_best}) ===")
    t2 = min(remaining * 0.25, 300.0)
    if t2 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = combined_optimize(a_in, t2, verbose=False)
        try_update(a_out, f"combined_n{n_best}")

        # Try a perturbed version
        rng2 = np.random.default_rng(123)
        perturbed = norm_simplex(a_in + rng2.normal(0, 0.02, n_best))
        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.15, 150.0)
        if t2b > 20:
            a_out2, _ = combined_optimize(perturbed, t2b, verbose=False)
            try_update(a_out2, f"perturbed_n{n_best}")

    # === Phase 3: Longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Longer sequences ===")
    for target_n in [n_best + 1000, n_best + 2000]:
        if target_n > 9000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={target_n}...")
        a_long = interpolate_sequence(best_seq, target_n)
        try_update(a_long, f"interp_n{target_n}")
        
        # Also try new ramps at this length
        a_ramp = make_init_ramp(target_n)
        try_update(a_ramp, f"ramp_n{target_n}")
        
        t3 = min(remaining * 0.2, 200.0)
        if t3 > 30:
            a_out, _ = combined_optimize(a_ramp, t3, verbose=False)
            try_update(a_out, f"combined_ramp_n{target_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out2, _ = combined_optimize(a_long, min(remaining * 0.3, 150.0), verbose=False)
                try_update(a_out2, f"combined_interp_n{target_n}")

    # === Phase 4: Shorter sequences (sometimes shorter is better) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 4: Shorter sequences ===")
        for target_n in [max(2000, n_best - 1000), max(1500, n_best - 2000)]:
            if target_n < 1000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            a_short = interpolate_sequence(best_seq, target_n)
            try_update(a_short, f"short_n{target_n}")
            t4 = min(remaining * 0.2, 120.0)
            if t4 > 20:
                a_out, _ = combined_optimize(a_short, t4, verbose=False)
                try_update(a_out, f"combined_short_n{target_n}")

    # === Phase 5: Final intensive mirror descent with very slow annealing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final slow MD (n={n_best}) ===")
    t5 = min(remaining * 0.4, 300.0)
    if t5 > 30:
        a_in = norm_simplex(best_seq)
        # Very slow annealing for final polish
        a_out, _ = mirror_descent(a_in, t5, verbose=True)
        try_update(a_out, f"md_slow_n{n_best}")
        
        # Peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_ps, _ = peak_shaving(a_out, min(remaining * 0.5, 100.0), verbose=True)
            try_update(a_ps, f"ps_slow_n{n_best}")

    # === Phase 6: Final spectral + coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 6: Final spectral polish ===")
        a_in = norm_simplex(best_seq)
        a_cur = a_in.copy()
        for strength in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            if time.monotonic() - t_global > time_budget + t_global - 60:
                break
            a_cur = spectral_equalize(a_cur, strength=strength)
            try_update(a_cur, f"spec_polish_{strength}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_init_ramp(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
