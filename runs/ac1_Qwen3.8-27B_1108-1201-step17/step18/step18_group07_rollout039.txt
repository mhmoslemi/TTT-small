def compute_smooth_gradient(a, c, beta, top_k=None):

    c_max = c.max()

        top_idx = np.argsort(c)[::-1][:top_k]

        top_idx = np.arange(len(c))
        c_top = c

def mirror_descent_smooth(a0, time_limit, verbose=False):
    """Mirror descent with smooth approximation and adaptive parameters."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Parameters
    beta_start = 20.0
    beta_end = 200.0
    top_k_start = 30
    top_k_end = 5
    total_iters = 200000
    lr_start = 0.5
    lr_end = 0.05
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, t / total_iters)
        beta = beta_start + (beta_end - beta_start) * progress
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        lr = lr_start + (lr_end - lr_start) * progress
        
        # Compute convolution and gradient
        c = fast_conv(a)
        M = c.max()
        
        # Update best
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
        
        smooth_val, grad = compute_smooth_gradient(a, c, beta, top_k=cur_top_k)
        
        # Mirror descent step (multiplicative update on simplex)
        # a_new[i] ∝ a[i] * exp(-lr * grad[i])
        exponent = -lr * grad
        exponent = np.clip(exponent, -50.0, 50.0)
        a_new = a * np.exp(exponent)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        
        # Periodic spectral smoothing
        if t % 2000 == 0:
            a = spectral_flatten(a, power=0.3)
            c_check = fast_conv(a)
            if c_check.max() < best_m:
                best_m = c_check.max()
                best_a = a.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): J={2*n*best_m:.8f}, beta={beta:.1f}")
    
    return best_a, best_m


def spectral_flatten(a, power=0.3):
    """Compress dynamic range of |A| in frequency domain."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - power)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a

def coordinate_descent_smoothed(a0, time_limit, verbose=False):

                    a_trial = a.copy()

                break

        else:
            stall += 1

def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return normalize(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return normalize(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.03
    return normalize(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return normalize(a)


def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_gradient_topk(a, c, beta, top_k=20):
    """Compute gradient using top-k peaks with softmax weighting.
    
    Returns (smooth_value, gradient_vector).
    """
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()
    
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -60.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return max_v, grad

def spectral_flatten_blend(a, power=0.3):
    """Compress dynamic range of |A| and blend with original."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - power)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    blend = 0.4 * power
    a_out = (1.0 - blend) * a + blend * a_new
    return normalize(a_out)

def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05

def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.03
    return normalize(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return normalize(a)


def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            remaining_now = time_budget - 40 - (time.monotonic() - t_global)
            if remaining_now > 15:
                t_sp = min(t_per_cycle * 0.8, remaining_now * 0.3)
                a_sp, _ = spectral_optimize_adaptive(normalize(current), t_sp)
                current = a_sp
                try_update(current, f"final_cycle{cycle}_spec")
            
            # CD step (only in last cycle)
            if cycle == 4:
                remaining_now = time_budget - 40 - (time.monotonic() - t_global)
                if remaining_now > 15:
                    t_cd = min(remaining_now * 0.6, 60.0)
                    a_cd, _ = refined_coordinate_descent(normalize(current), t_cd)
                    try_update(a_cd, "final_cd")
                    current = a_cd
        
        # One more spectral pass on the very best
        remaining_now = time_budget - 40 - (time.monotonic() - t_global)
        if remaining_now > 10:
            a_sp, _ = spectral_optimize_adaptive(normalize(best_seq), remaining_now * 0.7)
            try_update(a_sp, "final_spectral_best")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3500
        best_seq = make_init_bowl(n_fb, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.8f}")
    
    return [float(x) for x in best_seq]
