"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive mirror descent with softmax/exact subgradient
2. LP-based refinement (key new component)
3. Coordinate descent polish
4. Multi-length and multi-initialization search
"""

import numpy as np
import time
from scipy.optimize import linprog

_best_tracker = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_obj(a):
    """J = 2n * max(conv(a,a)) / (sum a)^2, scale invariant so normalize first."""
    a = norm_sum1(a)
    n = len(a)
    M = fast_conv(a).max()
    return 2.0 * n * M


def compute_subgrad_softmax(a, conv, top_k=30, temperature=0.005):
    n = len(a)
    conv_max = conv.max()
    # Use top peaks with softmax weights
    n_pk = min(top_k, len(conv))
    peak_indices = np.argsort(conv)[::-1][:n_pk]
    weights = np.exp((conv[peak_indices] - conv_max) / temperature)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(n_pk) / n_pk
    else:
        weights = weights / w_sum
    grad = np.zeros(n)
    for wi, k in zip(weights, peak_indices):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad


def compute_subgrad_exact(a, conv):
    """Exact subgradient at the single peak."""
    n = len(a)
    k_int = int(np.argmax(conv))
    lo = max(0, k_int - n + 1)
    hi = min(n - 1, k_int)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros(n)
    grad[idx] = 2.0 * a[k_int - idx]
    return grad


def amd_optimize(a0, time_limit, top_k=30, temp_init=0.005, c0=3.0,
                 restart_every=4000, seed=42, use_exact=False):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    temperature = temp_init
    stall_count = 0
    restart_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv_y = fast_conv(y)
        if use_exact:
            g = compute_subgrad_exact(y, conv_y)
        else:
            g = compute_subgrad_softmax(y, conv_y, top_k=top_k, temperature=temperature)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.08, 0.02 + 0.008 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            top_k = min(top_k + 5, 80)
            temperature = max(temperature - 0.001, 0.001)
            c0 = 3.0 * (1.0 + 0.05 * min(restart_count, 10))
            restart_every = max(3000, restart_every + 500)
            if use_exact and restart_count > 3:
                use_exact = False  # switch to softmax after a few restarts

    return best_x, best_m


def solve_lp_direction(a, M, n_constraints=80):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for top-k peaks, b >= 0.
    Returns b or None if infeasible.
    """
    n = len(a)
    conv = fast_conv(a)
    top_k = np.argsort(conv)[::-1][:n_constraints]
    
    # Build constraint matrix
    K = len(top_k)
    A = np.zeros((K, n))
    for ii in range(K):
        k = int(top_k[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[ii, j] = a[k - j]
    
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            b = np.maximum(result.x, 0)
            if b.sum() > 1e-15:
                return b
    except Exception:
        pass
    return None
