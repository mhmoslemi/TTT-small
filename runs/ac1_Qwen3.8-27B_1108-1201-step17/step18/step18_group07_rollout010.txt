(1/1) ∫_0^1 R(τ)dτ = ∫_0^1 ∫_0^{1-τ} f(x)f(x+τ)dx dτ = ∫_0^1 ∫_0^{1-x} f(x)f(y)dy dx
= (1/2)(∫f)^2 - (1/2)∫f^2 = (1/2)(1 - ∫f^2)

So the average of R is (1 - ∫f^2)/2. The max of R is at least this average. And R(0) = ∫f^2.

Score ≈ 2 * max(R) (in the continuous limit, with appropriate normalization).

To minimize max(R), we want ∫f^2 close to 1 (so the average is close to 0) and R(0) close to 1. But R(0) = ∫f^2, and if ∫f^2 is close to 1, the average is close to 0. So max(R) ≈ R(0) ≈ 1. Score ≈ 2.

Alternatively, if ∫f^2 = 1/n (discrete uniform), average of R ≈ (1-1/n)/2 ≈ 1/2, max(R) >= 1/2. Score ≈ 2 * n * (1/n) = 2 (discrete).

Hmm, I think for the discrete case with sum=1:
- Average of conv values = 1/(2n-1)
- max(conv) >= 1/(2n-1)
- Score >= 2n/(2n-1) ≈ 1

So the theoretical lower bound is 1 (for large n), achieved by a sequence with perfectly flat convolution. This is like a "Facebook" or "all-ones" sequence complement.

For non-negative sequences, the question is how close to flat we can get.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid spectral-MWU optimizer with:
1. Aggressive log-magnitude compression in frequency domain
2. Multiplicative weight updates (MWU) with adaptive annealing

import numpy as np

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_max(a):
    """Compute only the max of conv(a,a) via FFT (saves memory)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    return float(conv.max())


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * float(c.max())

def spectral_compress(a, alpha=0.5, preserve_phase=True):
    """Compress the magnitude spectrum of a.
    
    alpha < 1 compresses dynamic range (flattens spectrum).
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_max = mag.max()
    if mag_max < 1e-30:
        return a.copy()
    
    # Normalize magnitude to [0, 1], then apply power compression
    mag_norm = mag / mag_max
    mag_compressed = mag_norm ** alpha
    mag_new = mag_max * mag_compressed
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_phase_randomize(a, frac=0.3):
    """Randomize a fraction of the phase spectrum to escape local minima."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    rng = np.random.default_rng(int(time.time() * 1000) % 100000)
    
    n_bins = len(A)
    n_rand = max(1, int(n_bins * frac))
    indices = rng.choice(n_bins, size=n_rand, replace=False)
    
    A_new = A.copy()
    for idx in indices:
        A_new[idx] = np.abs(A_new[idx]) * np.exp(1j * rng.uniform(0, 2 * np.pi))
    
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def mwu_step(a, c, top_k=20, beta=50.0, eta=0.1):
    """One Multiplicative Weight Update step.
    
    a_new[i] ∝ a[i] * exp(-eta * g[i])
    where g[i] = sum over top peaks k of w_k * 2 * a[k-i]
    """
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx]
    
    # Softmax weighting
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    # MWU update
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def hybrid_optimize(a0, time_limit, verbose=False):
    """Hybrid spectral + MWU optimizer with adaptive parameters."""
    n = len(a0)
    a = norm_simplex(a0).copy()
    best_a = a.copy()
    best_c = fast_conv(a)
    best_m = float(best_c.max())
    t0 = time.monotonic()

    alpha_init = 0.8  # spectral compression (will decrease)
    alpha_min = 0.2

    eta_decay = 0.9999
    
    alpha = alpha_init
    top_k = top_k_init
    beta = beta_init
    eta = eta_init
    spec_every = 5  # do spectral every 5 MWU steps
    step_count = 0
    last_improve_step = 0
    stall_count = 0
    max_stall = 500
    
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        step_count += 1
        
        if step_count % spec_every == 0:
            # Spectral compression step
            a_spec = spectral_compress(a, alpha=alpha)
            c_spec = fast_conv(a_spec)
            m_spec = float(c_spec.max())
            
            if m_spec < best_m * (1 - 1e-6):
                best_m = m_spec
                best_a = a_spec.copy()
                last_improve_step = step_count
                a = a_spec.copy()
                # Slightly increase compression
                alpha = max(alpha_min, alpha * 0.95)
                spec_every = max(3, spec_every - 1)
            else:
                # Blend with best
                a = norm_simplex(0.5 * a + 0.5 * best_a)
                alpha = min(0.9, alpha * 1.1)
        else:
            # MWU step
            c = fast_conv(a)
            a_new = mwu_step(a, c, top_k=top_k, beta=beta, eta=eta)
            c_new = fast_conv(a_new)
            m_new = float(c_new.max())
            
            if m_new < best_m * (1 - 1e-7):
                best_m = m_new
                best_a = a_new.copy()
                last_improve_step = step_count
                a = a_new.copy()
                # Decrease eta slightly
                eta *= 0.999
                # Increase beta (sharper peak selection)
                beta = min(200.0, beta * 1.002)
                # Decrease top_k (focus on fewer peaks)
                top_k = max(top_k_min, int(top_k * 0.999))
            else:
                a = a_new  # still take the step but don't update best
                eta *= 0.98
        
        # Stall detection
        if step_count - last_improve_step > max_stall:
            # Try to escape: phase randomization or spectral with different params
            if rng.random() < 0.5:
                a_esc = spectral_phase_randomize(a, frac=0.2)
            else:
                a_esc = spectral_compress(best_a, alpha=max(0.1, alpha - 0.3))
            c_esc = fast_conv(a_esc)
            m_esc = float(c_esc.max())
            if m_esc < best_m:
                best_m = m_esc
                best_a = a_esc.copy()
                a = a_esc.copy()
            else:
                a = norm_simplex(0.7 * best_a + 0.3 * a_esc)
            last_improve_step = step_count
            max_stall = int(max_stall * 0.7)
            stall_count += 1
    
    return best_a.copy(), best_m

def coordinate_polish(a0, time_limit, verbose=False):
    """Local coordinate descent: try swapping/moving mass between pairs."""
    n = len(a0)
    a = norm_simplex(a0).copy()
    best_a = a.copy()
    best_m = fast_conv_max(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M = float(c.max())
        
        # Find top peaks
        n_top = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            
            # Find positions contributing to this peak
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            # Try reducing the top contributor and adding to a low position
            src_ranking = np.argsort(contrib)[::-1][:5]
            
            for src_r in src_ranking:
                src = int(idx[src_r])
                src_val = a[src]
                if src_val < 1e-10:
                    continue
                
                # Try moving to a few candidate destinations
                for _ in range(3):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    for frac in [0.02, 0.05, 0.1]:
                        amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] = max(0, a_trial[src] - amt)
                        a_trial[dst] += amt
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_conv_max(a_trial)
                        if M_trial < best_m * (1 - 1e-8):
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv_max(a)
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a.copy(), best_m


def make_init_cosine(n, mode=1):

def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):

def make_init_ramp(n, power=1.0):

def make_init_sinc(n, bw=5.0):

def make_init_random_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 15):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    if remaining > 10:
        a_fine, _ = hybrid_optimize(a_fine, remaining, verbose=verbose)
    
    return a_fine

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_max(a):
    """Compute only the max of conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    return float(conv.max())


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

def spectral_compress(a, alpha=0.5):
    """Compress the magnitude spectrum: mag_new = mag_max * (mag/mag_max)^alpha."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_max = mag.max()
    if mag_max < 1e-30:
        return a.copy()
    mag_norm = mag / mag_max
    mag_compressed = mag_norm ** alpha
    mag_new = mag_max * mag_compressed
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def mwu_step(a, c, top_k=20, beta=50.0, eta=0.1):
    """One Multiplicative Weight Update step using top-k peak soft-gradient."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx]
    max_v = top_vals[0]
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a_fine, _ = hybrid_optimize(a_fine, remaining)
    return a_fine


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with hybrid spectral-MWU approach."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    parent_data = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_data = p.copy()
            p_score = evaluate_safe(parent_data)
            print(f"Parent: n={len(parent_data)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent_data)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # ============================================================
    # Phase 1: Hybrid optimize on diverse inits at base length
    # ============================================================
    print(f"\n=== Phase 1: Hybrid Optimize (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.38, 580.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent_data is not None:
            inits.append(norm_simplex(parent_data))
            labels.append("parent")

        inits.append(make_init_cosine(n0, mode=1))
        labels.append("cos1")
        inits.append(make_init_bowl(n0, sharpness=3.0))
        labels.append("bowl3")
        inits.append(make_init_bowl(n0, sharpness=5.0))
        labels.append("bowl5")
        inits.append(make_init_dual_peak(n0, 0.2, 0.8, 0.15))
        labels.append("dualpk")
        inits.append(make_init_sinc(n0, bw=8.0))
        labels.append("sinc8")
        inits.append(make_init_random_smooth(n0, seed=77))
        labels.append("rand1")
        inits.append(make_init_rectified_gaussian(n0, sigma=0.25, seed=55))
        labels.append("rectg")

        # Deduplicate
        seen_set = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 20)], 4))
            if sig not in seen_set:
                seen_set.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        print(f"  Running hybrid on {len(inits)} initializations...")
        t_per = t_phase1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            remaining_now = time_budget - 60 - elapsed_so_far
            if remaining_now < 15:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 10:
                break
            print(f"    Hybrid '{lbl}' (budget={t_avail:.0f}s)...")

            a_hyb, _ = hybrid_optimize(init, t_avail * 0.65)
            a_sc = float(evaluate_safe(a_hyb))
            if np.isfinite(a_sc) and a_sc < best_score:
                best_score = a_sc
                best_seq = a_hyb.copy()
                print(f"  *** NEW BEST [hyb_{lbl}]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

            # Spectral polish
            remaining_now = time_budget - 60 - (time.monotonic() - t_global)
            if remaining_now > 10:
                a_sp, _ = spectral_only_optimize(a_hyb, min(t_avail * 0.35, 30.0))
                a_sc2 = float(evaluate_safe(a_sp))
                if np.isfinite(a_sc2) and a_sc2 < best_score:
                    best_score = a_sc2
                    best_seq = a_sp.copy()
                    print(f"  *** NEW BEST [spec_{lbl}]: {a_sc2:.6f} (t={time.monotonic()-t_global:.0f}s)")

        # Extended hybrid on best
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t_ext = min(remaining * 0.25, 180.0)
            print(f"  Extended hybrid (budget={t_ext:.0f}s)...")
            a_ext, _ = hybrid_optimize(norm_simplex(best_seq), t_ext)
            a_sc = float(evaluate_safe(a_ext))
            if np.isfinite(a_sc) and a_sc < best_score:
                best_score = a_sc
                best_seq = a_ext.copy()
                print(f"  *** NEW BEST [hyb_ext]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # ============================================================
    # Phase 2: Coarse-to-fine at different lengths
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Length Exploration ===")

    # Try longer
    for target_n in [n0 + 1000, n0 + 2000]:
        if target_n > 9000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t_cf = min(remaining * 0.15, 150.0)
        print(f"  Coarse-to-fine n={target_n} (budget={t_cf:.0f}s)...")
        a_cf = coarse_to_fine(norm_simplex(best_seq), target_n, t_cf)
        a_sc = float(evaluate_safe(a_cf))
        if np.isfinite(a_sc) and a_sc < best_score:
            best_score = a_sc
            best_seq = a_cf.copy()
            print(f"  *** NEW BEST [cf_n{target_n}]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Try shorter
    for target_n in [max(2500, n0 - 1000), max(2000, n0 - 2000)]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t_cf = min(remaining * 0.12, 120.0)
        print(f"  Coarse-to-fine n={target_n} (budget={t_cf:.0f}s)...")
        a_cf = coarse_to_fine(norm_simplex(best_seq), target_n, t_cf)
        a_sc = float(evaluate_safe(a_cf))
        if np.isfinite(a_sc) and a_sc < best_score:
            best_score = a_sc
            best_seq = a_cf.copy()
            print(f"  *** NEW BEST [cf_n{target_n}]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # ============================================================
    # Phase 3: Intensive spectral on best
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Intensive Spectral (n={n_best}) ===")
    t3 = min(remaining * 0.22, 250.0)
    if t3 > 30:
        a_sp, _ = spectral_only_optimize(norm_simplex(best_seq), t3)
        a_sc = float(evaluate_safe(a_sp))
        if np.isfinite(a_sc) and a_sc < best_score:
            best_score = a_sc
            best_seq = a_sp.copy()
            print(f"  *** NEW BEST [spec_int]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")
        # Hybrid polish after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.12, 120.0)
        if t3b > 15:
            a_hyb, _ = hybrid_optimize(norm_simplex(best_seq), t3b)
            a_sc = float(evaluate_safe(a_hyb))
            if np.isfinite(a_sc) and a_sc < best_score:
                best_score = a_sc
                best_seq = a_hyb.copy()
                print(f"  *** NEW BEST [hyb_after_spec]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # ============================================================
    # Phase 4: Coordinate descent polish
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate Descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 120.0)
    if t4 > 10:
        a_cd, _ = coordinate_polish(norm_simplex(best_seq), t4)
        a_sc = float(evaluate_safe(a_cd))
        if np.isfinite(a_sc) and a_sc < best_score:
            best_score = a_sc
            best_seq = a_cd.copy()
            print(f"  *** NEW BEST [cd]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # ============================================================
    # Phase 5: Final intensive hybrid
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final Hybrid (n={n_best}) ===")
    t5 = min(remaining * 0.35, 300.0)
    if t5 > 20:
        a_fin, _ = hybrid_optimize(norm_simplex(best_seq), t5, verbose=True)
        a_sc = float(evaluate_safe(a_fin))
        if np.isfinite(a_sc) and a_sc < best_score:
            best_score = a_sc
            best_seq = a_fin.copy()
            print(f"  *** NEW BEST [hyb_final]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")
        # Final spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_sp2, _ = spectral_only_optimize(norm_simplex(best_seq), min(40.0, remaining * 0.3))
            a_sc = float(evaluate_safe(a_sp2))
            if np.isfinite(a_sc) and a_sc < best_score:
                best_score = a_sc
                best_seq = a_sp2.copy()
                print(f"  *** NEW BEST [spec_final]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")
        # Final coordinate
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_cd2, _ = coordinate_polish(norm_simplex(best_seq), min(30.0, remaining * 0.5))
            a_sc = float(evaluate_safe(a_cd2))
            if np.isfinite(a_sc) and a_sc < best_score:
                best_score = a_sc
                best_seq = a_cd2.copy()
                print(f"  *** NEW BEST [cd_final]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # ============================================================
    # Phase 6: Quick random restarts
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 6: Random Restarts ===")
        rng = np.random.default_rng(2024)
        for restart in range(6):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 12:
                break
            t_r = min(remaining * 0.08, 35.0)
            pert = rng.normal(0, 0.015, n_best)
            a_pert = norm_simplex(np.maximum(best_seq + pert, 0))
            a_r, _ = hybrid_optimize(a_pert, t_r)
            a_sc = float(evaluate_safe(a_r))
            if np.isfinite(a_sc) and a_sc < best_score:
                best_score = a_sc
                best_seq = a_r.copy()
                print(f"  *** NEW BEST [restart{restart}]: {a_sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
