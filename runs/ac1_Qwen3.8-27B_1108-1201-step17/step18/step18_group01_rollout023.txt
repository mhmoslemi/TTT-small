        violation = np.maximum(c - M, 0)

        # Reducing g[j] by delta reduces c[k] by a[k-j] * delta for each k.

    top_idx = np.argsort(b)[::-1][:top_k]
    changes = np.zeros(len(top_idx))

        # Contribution from dst  
        if 0 <= k_int - dst < n:
            changes[idx] += delta * a[k_int - dst]
    # The new max is approximately max(b[top_idx] + changes)
    new_vals = b[top_idx] + changes
    return new_vals.max()

import numpy as np

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Compute linear convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):

def compute_full_gradient(a, ac, temperature):
    """
    Compute full gradient of T*log(sum exp(b[k]/T)) using FFT-based cross-correlation.
    grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i in [0, n-1])
    This equals 2 * (cross-correlation of p and a)[i]
    """
    n = len(a)
    ac_shifted = ac - ac.max()
    exp_vals = np.exp(ac_shifted / temperature)
    p = exp_vals / exp_vals.sum()
    
    # cross_corr[i] = sum_k p[k] * a[k-i] for valid indices
    # This is conv(p, reverse(a)) at appropriate indices
    a_rev = a[::-1]
    total_size = len(p) + n - 1
    fft_size = 1 << (total_size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(P * Ar, fft_size)
    
    # conv(p, a_rev)[i] = sum_j p[j] * a_rev[i-j] = sum_j p[j] * a[n-1-(i-j)]
    # We want cross_corr[i] = sum_k p[k] * a[k-i]
    # Let's verify: conv(p, a_rev)[m] = sum_j p[j] * a_rev[m-j]
    # a_rev[m-j] = a[n-1-(m-j)] = a[n-1-m+j]
    # So conv(p, a_rev)[m] = sum_j p[j] * a[n-1-m+j]
    # Let k = n-1-m+j, so j = k-(n-1-m) = k+m-n+1
    # = sum_k p[k+m-n+1] * a[k]
    # We want sum_k p[k] * a[k-i], so set m-n+1 = -i, i.e., m = n-1-i
    # cross_corr[i] = conv(p, a_rev)[n-1-i]
    
    grad = np.zeros(n)
    for i in range(n):
        m = n - 1 - i
        if 0 <= m < len(conv_result):
            grad[i] = 2.0 * conv_result[m]
    
    return grad


def compute_topk_gradient(a, ac, K, beta):
    """Compute gradient using top K lags with exponential falloff weights."""
    n = len(a)
    sorted_idx = np.argsort(ac)[::-1][:K]
    sorted_vals = ac[sorted_idx]
    
    w = np.exp(-beta * (sorted_vals.max() - sorted_vals))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(K) / K
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, sorted_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    return grad


def mgd_optimize(a0, time_limit, top_k, eta0, use_full_grad, verbose=True):
    """
    Multiplicative Gradient Descent with adaptive temperature and step size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = eta0
    patience = 0
    max_patience = 500
    cur_top_k = top_k
    iter_count = 0
    temperature = best_m * 0.15

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        # Adaptive temperature: decrease over iterations
        temperature = max(best_m * 0.02, best_m * 0.15 * (0.999 ** iter_count))

        if use_full_grad and n <= 7000:
            grad = compute_full_gradient(a, ac, temperature)
        else:
            grad = compute_topk_gradient(a, ac, cur_top_k, 5.0)
        
        # Center the gradient (remove mean for stability on simplex)
        grad = grad - grad.mean()
        
        gmax = np.abs(grad).max()
        if gmax < 1e-16:
            break
        grad = grad / gmax

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s

        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()

        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.9997
            if eta > 0.08:
                eta = 0.08
        else:
            eta *= 0.6
            if eta < 0.001:
                cur_top_k = max(5, cur_top_k // 2)
                eta = 0.03
                patience = max(0, patience - 200)

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}, T={temperature:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def peak_leveling(a0, time_limit, verbose=True):
    """
    Approximate LP direction: find g such that conv(a, g) is as flat as possible
    under the constraint max conv(a, g) <= M.
    Uses iterative water-filling.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac_a = fast_autocorr(a)
    M = ac_a.max()
    
    best_g = a.copy()
    best_obj = M
    t0 = time.monotonic()
    
    g = np.ones(n) / n
    alpha = 0.3
    
    for iteration in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_cross_conv(a, g)
        c_max = c.max()
        
        if c_max < best_obj * 0.999:
            best_obj = c_max
            best_g = g.copy()
        
        # Compute violation
        violation = np.maximum(c - M * 0.95, 0)
        
        if violation.max() < 1e-12:
            break
        
        # Compute pressure on g: reducing g[j] by delta reduces c[k] by a[k-j]*delta
        pressure = np.zeros(n)
        viol_idx = np.where(violation > 1e-10)[0]
        
        # Limit to top violators for efficiency
        if len(viol_idx) > 50:
            top_viol = viol_idx[np.argsort(violation[viol_idx])[::-1][:50]]
        else:
            top_viol = viol_idx
        
        for k in top_viol:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                pressure[idx] += violation[ki] * a[ki - idx]
        
        max_p = pressure.max()
        if max_p < 1e-15:
            break
        
        # Update g
        g = g * np.exp(-alpha * pressure / max_p)
        g = np.maximum(g, 1e-15)
        g = g / g.sum()
        
        # Slowly decrease alpha
        alpha *= 0.995
        if alpha < 0.01:
            alpha = 0.01

    # Now blend towards the leveled direction
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        a_blend = (1 - t) * a + t * best_g
        a_blend = np.maximum(a_blend, 0)
        s = a_blend.sum()
        if s < 1e-15:
            continue
        a_blend = a_blend / s
        m_blend = fast_autocorr(a_blend).max()

            best_m = m_blend
            best_a = a_blend
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Peak-leveling done ({elapsed:.0f}s): J={J:.6f}")

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                    break
        
        if not improved:
            stall += 1
            # Random escape with multiple positions
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            a = a / a.sum()
            m_new = fast_autocorr(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = norm_sum1(p)
            p_score = evaluate_safe(p_norm)
            if np.isfinite(p_score):
                best_seq = p_norm
                best_score = p_score
                print(f"Parent: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.maximum(np.sin(np.pi * x) ** 2, 0.01)
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Parametric + antisymmetric init ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Parametric + Antisym (n={n0}, remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 10:
        cur_score = 2.0 * n0 * fast_autocorr(best_seq).max()
        a_p, s_p = parametric_init(n0, best_seq, cur_score)
        sc_p = evaluate_safe(a_p)
        if np.isfinite(sc_p) and sc_p < best_score:
            best_score = sc_p
            best_seq = a_p.copy()
            print(f"  Parametric improved to {sc_p:.6f}")
        
        # Antisymmetric perturbation
        a_as = antisymmetric_perturb(best_seq, verbose=False)
        sc_as = evaluate_safe(a_as)
        if np.isfinite(sc_as) and sc_as < best_score:
            best_score = sc_as
            best_seq = a_as.copy()
            print(f"  Antisym improved to {sc_as:.6f}")
    
    # === Phase 2: MGD with full gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: MGD full-grad (n={n_best}, time={min(remaining*0.30, 400):.0f}s) ===")
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 30:
        a_out = mgd_optimize(norm_sum1(best_seq), t2, top_k=50, eta0=0.03, use_full_grad=True, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST MGD: {sc:.6f}")
    
    # === Phase 3: Peak leveling ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak leveling (n={n_best}, time={min(remaining*0.15, 200):.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out = peak_leveling(norm_sum1(best_seq), t3, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST Peak-leveling: {sc:.6f}")
        # Follow up with MGD on the leveled result
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 30:
            t3b = min(remaining2 * 0.15, 150.0)
            if t3b > 20:
                a_out2 = mgd_optimize(norm_sum1(a_out), t3b, top_k=40, eta0=0.02, use_full_grad=True, verbose=False)
                sc2 = evaluate_safe(a_out2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2

                    print(f"  *** NEW BEST MGD-after-leveling: {sc2:.6f}")

    # === Phase 6: Multi-length exploration ===

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 6500:
            t6b = min(remaining * 0.20, 80.0)
            if t6b > 15:

        if t7 > 15:

            sc = evaluate_safe(a_out)

                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final MGD: {sc:.6f}")
    
    # === Phase 8: Final pair swap ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.40, 100.0)
        if t8 > 10:

            if np.isfinite(sc) and sc < best_score:

                best_seq = a_out.copy()

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
