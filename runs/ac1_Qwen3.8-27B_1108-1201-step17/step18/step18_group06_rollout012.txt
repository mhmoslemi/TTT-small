```python
"""
Search for nonnegative step-function heights minimizing

    2*n*max(conv(a,a)) / (sum(a))^2

with the supplied evaluate_sequence().

The key improvement is to stop using mildly increasing ramps with a positive
left endpoint.  A ramp that starts near zero and increases roughly linearly
has a much smaller maximum full-convolution diagonal.  This program nevertheless
runs a real search:
  - scans power ramps and modulated ramps at several lengths,
  - resamples the best shape to candidate lengths,
  - validates against the actual evaluator and falls back to a guaranteed
    low-score linear ramp.
All helpers are top-level and no lambda functions are used.
"""

import numpy as np
import time


def _safe_array(a):
    """Convert to a 1D finite float64 array clipped to [0, 1000]."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim == 0:
        arr = arr.reshape(1)
    elif arr.ndim != 1:
        arr = arr.reshape(-1)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    return np.clip(arr, 0.0, 1000.0)


def _norm_sum1(a):
    """Nonnegative, finite, clipped, normalized to sum 1."""
    arr = _safe_array(a)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    s = float(arr.sum(dtype=np.float64))
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _full_convolution_fft(a):
    """Full linear convolution a * a using FFT.  Returns length 2*n - 1."""
    arr = np.asarray(a, dtype=np.float64)
    n = int(arr.size)
    if n == 0:
        return np.zeros(1, dtype=np.float64)

    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1

    F = np.fft.rfft(arr, size)
    P = F * F

    # Enforce the theoretically real endpoints to reduce roundoff artifacts.
    P = np.array(P, dtype=np.complex128, copy=False)
    P[0] = complex(float(P[0].real), 0.0)
    if size > 1:
        P[-1] = complex(float(P[-1].real), 0.0)

    c = np.fft.irfft(P, size)
    c = np.asarray(c[:need], dtype=np.float64)
    return c


def _conv_max(a):
    """Maximum value of full convolution a * a, nonnegative."""
    c = _full_convolution_fft(a)
    if c.size == 0:
        return 0.0
    m = float(np.max(c))
    if m < 0.0:
        # Small negative roundoff only.
        return 0.0
    return m


def _fast_score(a):
    """Fast FFT-based surrogate for evaluate_sequence()."""
    arr = _safe_array(a)
    n = int(arr.size)
    if n == 0:
        return float("inf")
    if n > 30000:
        return float("inf")

    s = float(arr.sum(dtype=np.float64))
    if s < 0.01:
        return float("inf")

    m = _conv_max(arr)
    return float(2.0 * n * m / (s * s))


def exact_score(a):
    """Use the official evaluator when available."""
    try:
        arr = _safe_array(a)
        if arr.size == 0:
            return float("inf")
        s = float(arr.sum(dtype=np.float64))
        if s < 0.01:
            return float("inf")
        if arr.size > 30000:
            return float("inf")
        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:
        return float("inf")


def consider_candidate(a, best_seq, best_score, label):
    """Update best sequence/score if the candidate is better by FFT score."""
    try:
        arr = _safe_array(a)
        if arr.size == 0:
            return best_seq, best_score
        sc = _fast_score(arr)
        if np.isfinite(sc) and sc < best_score:
            best_seq = np.array(arr, dtype=np.float64, copy=True)
            best_score = float(sc)
            print(f"[{label}] internal score={best_score:.10f} n={arr.size}")
    except Exception:
        pass
    return best_seq, best_score


def make_base(n, p, kind=1):
    """
    Basic increasing power ramps, returned normalized to sum 1.

    kind:
      0: linspace(0, 1, n)
      1: i / n, i = 1..n   (positive, right endpoint 1)
      2: i / (n+1), i = 1..n   (slightly scaled down)
    """
    n = int(n)
    if n <= 0:
        return None
    if n == 1:
        return np.ones(1, dtype=np.float64)

    if kind == 0:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    elif kind == 2:
        x = np.arange(1, n + 1, dtype=np.float64) / float(n + 1)
    else:
        x = np.arange(1, n + 1, dtype=np.float64) / float(n)

    if p == 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        a = np.power(x, float(p))

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    return _norm_sum1(a)


def make_modulated(n, p, c1, c2):
    """
    Modulated power ramp:
        a_i = (i/n)^p * (1 + c1*(i/n) + c2*(i/n)^2)
    normalized to sum 1.  Returns None if the quadratic factor becomes
    nonpositive anywhere.
    """
    n = int(n)
    if n <= 0:
        return None
    if n == 1:
        return np.ones(1, dtype=np.float64)

    x = np.arange(1, n + 1, dtype=np.float64) / float(n)
    q = 1.0 + float(c1) * x + float(c2) * x * x
    if float(np.min(q)) <= 1e-8:
        return None

    a = np.power(x, float(p)) * q
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    return _norm_sum1(a)


def resample_sum1(a, new_n):
    """Linearly re-interpolate a normalized shape to another length."""
    arr = _norm_sum1(a)
    new_n = int(new_n)
    if new_n <= 0:
        return np.ones(1, dtype=np.float64)
    old_n = int(arr.size)
    if old_n == new_n:
        return arr
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if old_n == 1:
        b = np.full(new_n, float(arr[0]), dtype=np.float64)
        return _norm_sum1(b)

    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    b = np.interp(new_x, old_x, arr)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    return _norm_sum1(b)


def propose_candidate(budget_s=1800.0):
    """Main search entry point."""
    try:
        total_time = float(budget_s)
    except Exception:
        total_time = 1800.0

    t_start = time.monotonic()
    if not np.isfinite(total_time) or total_time <= 60.0:
        total_time = 1800.0

    # Keep a reserve for final exact validation.
    hard_deadline = t_start + max(120.0, total_time - 5.0)
    search_deadline = t_start + max(60.0, total_time - 180.0)

    best_seq = None
    best_score = float("inf")

    # 1. Try to inherit the previous known construction, but only if not huge.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = _safe_array(parent)
        if parent.size > 0 and parent.size <= 30000:
            best_seq, best_score = consider_candidate(parent, best_seq, best_score, "parent")
    except Exception:
        pass

    # 2. Guaranteed strong baseline: zero-start / positive-start linear ramps.
    base_lengths = (4200, 5000, 6000)
    base_powers = (1.0, 0.95, 1.05, 0.9, 1.1)
    base_kinds = (1, 0, 2)

    for n in base_lengths:
        if time.monotonic() > search_deadline:
            break
        for kind in base_kinds:
            if time.monotonic() > search_deadline:
                break
            for p in base_powers:
                a = make_base(n, p, kind)
                if a is not None:
                    best_seq, best_score = consider_candidate(
                        a, best_seq, best_score, f"base_k{kind}_p{p}_n{n}"
                    )

    # 3. Small low-dimensional modulated-ramp scan around the winner.
    if time.monotonic() < search_deadline:
        mod_powers = (0.95, 1.0, 1.05)
        c1_values = (-0.50, -0.25, -0.10, 0.00, 0.10, 0.25, 0.50, 1.00, 2.00)
        c2_values = (-0.30, -0.10, 0.00, 0.10, 0.30, 0.60)
        mod_lengths = (4200, 6000)

        for n in mod_lengths:
            if time.monotonic() > search_deadline:
                break
            for p in mod_powers:
                if time.monotonic() > search_deadline:
                    break
                for c1 in c1_values:
                    if time.monotonic() > search_deadline:
                        break
                    for c2 in c2_values:
                        if time.monotonic() > search_deadline:
                            break
                        a = make_modulated(n, p, c1, c2)
                        if a is not None:
                            best_seq, best_score = consider_candidate(
                                a,
                                best_seq,
                                best_score,
                                f"mod_n{n}_p{p}_c1{c1}_c2{c2}",
                            )

    # 4. If nothing was found, install the guaranteed linear ramp.
    if best_seq is None:
        best_seq = make_base(4200, 1.0, 1)
        best_score = _fast_score(best_seq)

    # 5. Try resizing the best shape.
    if time.monotonic() < search_deadline:
        for new_n in (3500, 4200, 5000, 6000):
            if time.monotonic() > search_deadline:
                break
            a = resample_sum1(best_seq, new_n)
            if a is not None:
                best_seq, best_score = consider_candidate(
                    a, best_seq, best_score, f"resample_{new_n}"
                )

    # 6. Choose the final artifact exactly when possible, with a safe fallback.
    arr = _safe_array(best_seq)
    if arr.size == 0 or float(arr.sum(dtype=np.float64)) < 0.01:
        arr = make_base(4200, 1.0, 1)

    # Limit final length for the official direct-convolution evaluator.
    if arr.size > 8000:
        arr = resample_sum1(arr, 6000)

    final_score = exact_score(arr)

    fallback = make_base(4200, 1.0, 1)
    fallback_score = exact_score(fallback)

    use_fallback = False
    if not np.isfinite(final_score):
        use_fallback = True
    elif np.isfinite(fallback_score) and fallback_score < final_score:
        use_fallback = True

    if use_fallback:
        arr = fallback
        final_score = fallback_score if np.isfinite(fallback_score) else _fast_score(arr)

    # Optional fallback of a slightly longer known-good linear ramp if time remains.
    if time.monotonic() < hard_deadline - 5.0:
        for cand_n in (6000, 5000):
            cand = make_base(cand_n, 1.0, 1)
            cs = exact_score(cand)
            if np.isfinite(cs) and cs < final_score:
                arr = cand
                final_score = cs

    arr = _norm_sum1(arr)
    if arr.size == 0 or float(arr.sum(dtype=np.float64)) < 0.01:
        arr = make_base(4200, 1.0, 1)

    out = []
    for v in arr:
        try:
            vf = float(v)
        except Exception:
            vf = 0.0
        if not np.isfinite(vf):
            vf = 0.0
        if vf < 0.0:
            vf = 0.0
        if vf > 1000.0:
            vf = 1000.0
        out.append(vf)

    total = 0.0
    for v in out:
        total += v
    if len(out) == 0 or total < 0.01:
        out = [float(x) for x in make_base(4200, 1.0, 1)]

    print(f"FINAL returned length={len(out)} score={final_score:.10f}")
    return out
```
