def get_solver():

    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def score_from_ac(a, ac):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * ac.max() / (s * s))

def softmax_grad(a, beta, n_top=None):
    """Compute gradient of smoothed max of autocorrelation."""
    n = len(a)
    size = 2 * n - 1
    ac = fast_autocorr(a)
    
    if n_top is not None and n_top < size:
        # Only use top lags for numerical stability
        top_idx = np.argsort(ac)[::-1][:n_top]
        ac_use = ac[top_idx].copy()
    else:
        top_idx = np.arange(size)
        ac_use = ac.copy()
    
    # Softmax weights
    ac_shifted = ac_use - ac_use.max()
    w = np.exp(beta * ac_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(ac_use)) / len(ac_use)
    else:
        w = w / w_sum
    
    # Gradient: sum_k w_k * 2 * S_k(a)
    # S_k(a)_i = a_{k-i} (with a_j = 0 outside [0, n-1])
    # So S_k(a) is a shifted and reversed version of a
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # grad += wi * (a reversed and shifted by k)

        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad[idx] += wi * 2.0 * a[k_int - idx]

def get_solver():
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(200, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.01])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except:
            continue
    return cp.ECOS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def score_from_ac(n, ac):
    return float(2.0 * n * ac.max())

def lse_mirror_descent(a0, time_limit, beta0=10.0, beta1=300.0,
                       lr0=0.05, n_top=150, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Bernoulli schedule for beta
        frac = min(1.0, t / (time_limit * 3.0))
        beta = beta0 * (beta1 / beta0) ** frac
        
        # Compute autocorrelation and gradient
        ac = fast_autocorr(a)
        top_idx = np.argsort(ac)[::-1][:n_top]
        ac_top = ac[top_idx]
        
        # Softmax weights
        w = np.exp(beta * (ac_top - ac_top.max()))
        w /= w.sum()
        
        # Gradient of smoothed max
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n - 1, ki)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                grad[idx] += 2.0 * wi * a[ki - idx]
        
        # Mirror descent step with backtracking
        lr = lr0 / (1.0 + 0.05 * np.sqrt(t))
        improved = False
        for decay in [1.0, 0.5, 0.25, 0.1]:
            a_trial = a * np.exp(-lr * decay * grad)
            a_trial = np.maximum(a_trial, 1e-30)
            a_trial /= a_trial.sum()
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m - 1e-15:
                a = a_trial
                best_m = m_trial
                best_a = a_trial.copy()
                stall = 0
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > max_stall:
                # Random restart with perturbation
                a = norm_sum1(best_a + 0.02 * rng.standard_normal(n) * best_a)
                stall = 0
        
        if verbose and t % 1000 == 0:
            J = 2.0 * n * best_m
            print(f"    LSE-MD t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, beta={beta:.0f}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss
