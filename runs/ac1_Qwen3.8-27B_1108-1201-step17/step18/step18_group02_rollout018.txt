def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def subgradient_at_lags(a, lags, weights):
    """Compute weighted subgradient of max_k (a*a)[k] at given lags."""
    n = len(a)
    grad = np.zeros(n)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    for k_int, w in zip(lags, weights):
        k_int = int(k_int)
        if k_int < 0 or k_int >= size:
            continue
        # grad contribution: 2 * w * a[k - j] for valid j
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            # a[k - idx] where k-idx is in [0, n-1]
            src_idx = k_int - idx
            valid = (src_idx >= 0) & (src_idx < n)
            if valid.any():
                grad[idx[valid]] += 2.0 * w * a[src_idx[valid]]
    return grad


def top_k_lags(ac, K):
    """Get top K lags from autocorrelation with weights."""
    n_lags = len(ac)
    K = min(K, n_lags)
    top_idx = np.argsort(ac)[::-1][:K]
    vals = ac[top_idx]
    total = vals.sum()
    if total < 1e-30:
        weights = np.ones(K) / K
    else:
        weights = vals / total
        # Exaggerate the top
        weights = weights ** 2
        weights = weights / weights.sum()
    return top_idx, weights


def multiplicative_descent(a0, time_limit, K=20, lr0=0.5, lr_decay=0.99995,
                           min_lr=0.01, verbose=True, rng_seed=42):
    """Fast multiplicative descent with exact subgradient and top-K smoothing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    lr = lr0
    stall = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Get top-K lags and compute subgradient
        lags, weights = top_k_lags(ac, K)
        grad = subgradient_at_lags(a, lags, weights)

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a = a_new / a_new.sum()

        lr = max(lr * lr_decay, min_lr)
        iters += 1

        if stall > 3000:
            # Restart with perturbation
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            a = x / x.sum()
            lr = lr0
            stall = 0
            iters = 0

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MulDesc iter={iters} ({elapsed:.0f}s): J={J:.6f}, lr={lr:.4f}, stall={stall}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    MulDesc done ({elapsed:.0f}s): J={J:.6f}, iters={iters}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=400, max_K=3000,
                   add_per_iter=30, verbose=True):
    """LP Frank-Wolfe with adaptive constraint management."""
    n = len(a0)
    a = norm_sum1(a0)
    ac_init = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac_init.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    lp_fail_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K_c = len(constraint_list)
        if K_c == 0:
            break

        # Build constraint matrix
        coeffs_matrix = np.zeros((K_c, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 5 and len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 10:
                constraint_set = set(int(k) for k in constraint_list[:max(200, initial_K // 2)])
            continue

        lp_fail_count = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Strongly safeguarded FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        x_new = (t - 1.0) / (t + 2.0) * x + y_new if t > 1 else y_new.copy()
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_descent(a0, time_limit, rng_seed=42, verbose=False):
    """Coordinate descent: move mass from hot positions to cold positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(12, len(ac))]
        improved = False

        for k in top_k[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find cold destination
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            dst = candidates[np.argmin(scores[candidates])]
            if dst == src:
                continue

            for frac in [0.003, 0.008, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        else:
            stall = 0

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def comb_init(n, n_teeth, width_frac=0.3):
    """Create a comb-like initialization with n_teeth peaks."""
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    positions = np.linspace(0, 1, n_teeth, endpoint=False) + 0.5 / n_teeth
    for pos in positions:
        a += np.exp(-((x - pos) ** 2) / (2.0 * (width_frac / n_teeth) ** 2))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def wavelet_init(n, scale=4):
    """Create a wavelet-like initialization."""
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for s in range(1, scale + 1):
        for k in range(s):
            pos = (k + 0.5) / s
            a += np.exp(-((x - pos) ** 2) / (2.0 * (0.1 / s) ** 2))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def parametric_init(n):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    for freq in [1, 2, 3, 4]:
        for phase in [0.0, 0.25, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.01)
            candidates.append(a / a.sum())

    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr - 0.5) ** p
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())

    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.08, 0.15, 0.25]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            a = np.maximum(a, 0.005)
            candidates.append(a / a.sum())

    candidates.append(np.ones(n) / n)

    # Comb-like
    for nt in [3, 5, 7, 11, 15]:
        candidates.append(comb_init(n, nt, 0.25))

    # Wavelet
    candidates.append(wavelet_init(n, 3))
    candidates.append(wavelet_init(n, 5))

    return candidates

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n0 = len(best_seq)

    # === Phase 1: Try multiple initializations at parent length ===
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 50.0)
    if t1 > 10:
        candidates = parametric_init(n0)
        for i, cand in enumerate(candidates):
            if time.monotonic() - t_global > time_budget - t1:
                break
            check_and_update(cand, f"param{i}")

    # Blend parent with uniform and comb
    rng_blend = np.random.default_rng(77)
    for w in [0.5, 0.7, 0.85]:
        comb_c = comb_init(n0, 5, 0.3)
        cand = w * norm_sum1(best_seq) + (1 - w) * comb_c
        check_and_update(cand, f"blend_w{w}")

    n_best = len(best_seq)

    # === Phase 2: Multiplicative descent (fast) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mult Descent (n={n_best}) ===")
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multiplicative_descent(a_in, t2, K=15, lr0=0.6,
                                          lr_decay=0.99997, min_lr=0.02,
                                          verbose=True, rng_seed=42)
        check_and_update(a_out, f"muldesc_n{n_best}")

        # Second run with different params
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t2b = min(remaining * 0.15, 200.0)
            if t2b > 20:
                a_out2, _ = multiplicative_descent(a_in, t2b, K=30, lr0=0.3,
                                                   lr_decay=0.99999, min_lr=0.01,
                                                   verbose=False, rng_seed=123)
                check_and_update(a_out2, f"muldesc2_n{n_best}")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.25, 450.0)
    if t3 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, initial_K=400,
                                  max_K=3000, add_per_iter=30, verbose=True)
        check_and_update(a_out, f"lpfw_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord Descent (n={n_best}) ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent(a_in, t4, rng_seed=99, verbose=False)
        check_and_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length exploration ===")

    nearby_lengths = []
    for delta in [300, 500, -300, -500, 700, -700]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000:
            nearby_lengths.append(new_n)

    seen = set()
    unique_lengths = []
    for ln in nearby_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    for new_n in unique_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n  Length {new_n}:")
        a_long = interpolate_sequence(best_seq, new_n)
        check_and_update(a_long, f"interp_n{new_n}")

        # Quick multiplicative descent
        t5a = min(remaining * 0.25, 150.0)
        if t5a > 20:
            a_out, _ = multiplicative_descent(a_long, t5a, K=15, lr0=0.4,
                                              lr_decay=0.99998, min_lr=0.02,
                                              verbose=False, rng_seed=new_n % 1000)
            check_and_update(a_out, f"muldesc_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100 and new_n <= 5500:
            t5b = min(remaining * 0.30, 180.0)
            if t5b > 20:
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5b, solver,
                                          initial_K=300, max_K=2500,
                                          add_per_iter=25, verbose=False)
                check_and_update(a_out, f"lpfw_n{new_n}")

    # === Phase 6: Blend sequences from different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 6: Blending ===")
        # Interpolate current best to a few lengths and blend
        for target_n in [len(best_seq) - 200, len(best_seq) + 200, len(best_seq) + 400]:
            if target_n < 1500 or target_n > 7000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            a_interp = interpolate_sequence(best_seq, target_n)
            for w in [0.5, 0.7, 0.8]:
                comb_c = comb_init(target_n, max(3, target_n // 700), 0.25)
                blend = w * a_interp + (1 - w) * comb_c
                check_and_update(blend, f"blend_n{target_n}_w{w}")
            # Quick descent on blend
            t6 = min(remaining * 0.3, 100.0)
            if t6 > 15:
                a_out, _ = multiplicative_descent(a_interp, t6, K=15, lr0=0.3,
                                                  lr_decay=0.99999, min_lr=0.01,
                                                  verbose=False, rng_seed=target_n % 500)
                check_and_update(a_out, f"muldesc_blend_n{target_n}")

    # === Phase 7: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 60 and n_best <= 5500:
        t7 = min(remaining * 0.35, 250.0)
        if t7 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=500,
                                      max_K=4000, add_per_iter=40, verbose=True)
            check_and_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 8: Final multiplicative polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Polish (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.40, 200.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = multiplicative_descent(a_in, t8, K=10, lr0=0.15,
                                              lr_decay=0.99999, min_lr=0.005,
                                              verbose=True, rng_seed=777)
            check_and_update(a_out, f"polish_n{n_best}")

    # === Phase 9: Final coord descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.50, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent(a_in, t9, rng_seed=314, verbose=False)
            check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = comb_init(n_fb, 7, 0.3)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
