        g = np.zeros(n)

        hi = min(n - 1, k_star)

            hi2 = min(n - 1, k2)

        # The projection of a point p onto {x: sum(x)=1, x>=0} is:
        # x_i = max(p_i - tau, 0) where tau is chosen so sum(x) = 1
        p = a - 0.01 * g  # move in direction -g
        # Project p onto simplex
        mu = np.sort(p)[::-1]
        cumulative = np.cumsum(mu)
        rho = np.where(mu * np.arange(1, n+1) - (cumulative - 1) > 0)[0]
        if len(rho) == 0:
            a_new = np.ones(n) / n
        else:
            rho_max = rho[-1]
            tau = (cumulative[rho_max] - 1) / (rho_max + 1)
            a_new = np.maximum(p - tau, 0)
        
        F_new = 2.0 * n * fast_conv(a_new).max()
        if F_new < best_F - 1e-15:
            best_F = F_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_F

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with multi-stage smooth-max temperature annealing  
2. Exact subgradient descent with Euclidean projection onto simplex
3. Active-peak-aware coordinate descent
4. Diverse multi-initialization and multi-length search

import numpy as np

def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(1)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    na, nb = len(a), len(b)
    if na == 0 or nb == 0:
        return np.zeros(1)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def norm_sum1(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_max(conv, t):
    """Compute smooth max value and softmax weights."""
    c = np.asarray(conv, dtype=np.float64) / t
    cmax = c.max()
    lse = cmax + np.log(np.sum(np.exp(c - cmax)))
    val = t * lse
    w = np.exp(c - lse)
    return val, w

def conv_grad(a, w):
    """dM/da where M = sum_k w[k]*conv[k]. Result has same length as a."""
    n = len(a)
    ar = a[::-1]
    c = fast_conv_cross(ar, w)
    return c[n-1:2*n-1]

def _lbf_obj(x, temp):
    n = len(x)
    a = np.maximum(x, 0.0)
    S = a.sum()
    if S < 1e-15:
        return 1e10
    conv = fast_conv(a)
    Mv, _ = smooth_max(conv, temp)
    return 2.0 * n * Mv / (S * S)

def _lbf_grad(x, temp):
    n = len(x)
    a = np.maximum(x, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.zeros(n)
    conv = fast_conv(a)
    Mv, w = smooth_max(conv, temp)
    dM = conv_grad(a, w)
    return (2.0 * n) / (S * S) * (dM - 2.0 * Mv / S)

def project_simplex(v):
    """Project v onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1.0)
    rho = np.where(cond > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1)
    return np.maximum(v - theta, 0.0)
