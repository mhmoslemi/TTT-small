"""

2. FFT-accelerated mirror descent with adaptive softmax temperature

import numpy as np
import time
import cvxpy as cp


def _ffts(next_pow_two_val):
    return 1 << (int(next_pow_two_val) - 1).bit_length()


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fs = _ffts(size)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def _cross_corr_wy(w, y):
    """Compute g[i] = sum_k w[k]*y[k-i] for i=0..n-1 using FFT."""
    n = len(y)
    nw = len(w)
    fs = _ffts(nw + n - 1)
    W = np.fft.rfft(w, fs)
    Y = np.fft.rfft(y, fs)
    cross = np.fft.irfft(W * np.conj(Y), fs)
    return cross[:n]


def _norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def _score_m(a):
    """Return max of autocorrelation for normalized a (sum=1)."""
    ac = _fast_autocorr(a)
    return float(ac.max())


def _full_score(a):
    """Compute J = 2n * max(ac) / sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = _fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))

def _interp_len(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def _parametric_init(n, t_budget, best_a_prev, best_s_prev):
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    t0 = time.monotonic()
    best_a = _norm1(best_a_prev) if best_a_prev is not None else np.ones(n) / n
    best_s = best_s_prev if np.isfinite(best_s_prev) else _full_score(best_a)

    def _try(a, label):
        nonlocal best_a, best_s
        a = _norm1(a)
        s = _full_score(a)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"    Init {label}: J={s:.6f} (t={elapsed:.0f}s)")

    # Anti-symmetric linear ramp (key insight)
    for alpha in [0.3, 0.5, 0.7, 0.9, 1.0, 1.2, 1.5]:
        a = 1.0 + alpha * (2.0 * i / max(n - 1, 1) - 1.0)
        a = np.maximum(a, 0.01)
        _try(a, f"ramp_a{alpha}")

    # U-shaped (cosine)
    for alpha in [0.3, 0.5, 0.7, 0.9, 1.0]:
        a = 1.0 + alpha * np.cos(2.0 * np.pi * x)
        a = np.maximum(a, 0.01)
        _try(a, f"cos_a{alpha}")

    # Higher-order anti-symmetric: x - 0.5 powered
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (np.abs(x - 0.5) / 0.5) ** p + 0.05
        _try(a, f"U_p{p}")

    # Asymmetric power laws
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        _try((i + 1.0) ** p, f"pw{p}")
        _try((n - i) ** p, f"ipw{p}")

    # Sine squared
    for freq in [1, 2, 3]:
        for ph in [0.0, 0.25, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01)
            _try(a, f"sn{freq}_{ph}")

    # Exponential
    for r in [1, 2, 3, 5]:
        _try(np.exp(-r * x), f"er{r}")
        _try(np.exp(r * x), f"enr{r}")

    if time.monotonic() - t0 < t_budget * 0.7:
        # Blend forms
        n_blend = min(6, n)
        base1 = best_a.copy()
        for alpha in [0.5, 0.8, 1.0]:
            a2 = np.maximum(1.0 + alpha * (2.0 * i / max(n - 1, 1) - 1.0), 0.01)
            a2 = _norm1(a2)
            for w in [0.3, 0.5, 0.7]:
                _try(w * base1 + (1.0 - w) * a2, f"bl{w}_a{alpha}")

    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.0f}s): best J={best_s:.6f}")

def _mirror_descent(a0, t_budget, beta_start=10.0, beta_end=300.0, c0=4.0,
                    sf_every=400, sf_strength=0.25, verbose=True):
    """FFT-accelerated mirror descent with adaptive temperature and spectral resets."""
    n = len(a0)
    a = _norm1(a0.copy())
    best_a = a.copy()
    best_m = _score_m(a)
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    total_est = 8000

    for t in range(1, total_est + 1):
        if time.monotonic() - t0 > t_budget:
            break

        progress = min(t / total_est, 1.0)
        beta = beta_start + (beta_end - beta_start) * (progress ** 1.3)

        # Nesterov
        bn = (t - 1.0) / (t + 1.5)
        y = x + bn * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        # Compute convolution
        ac = _fast_autocorr(y)

        # Softmax weights
        ac_shift = ac - ac.max()
        w = np.exp(beta * ac_shift)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(ac)) / len(ac)
        else:
            w = w / w_sum

        # Gradient via FFT cross-correlation: g[i] = sum_k w[k] * y[k-i]
        g = _cross_corr_wy(w, y)
        g = np.maximum(g, 0.0)

        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s

        x_prev = x
        x = a_new

        M_new = _score_m(x)
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()

        # Spectral water-filling reset
        if t % sf_every == 0:
            a_sf = _spectral_wf(best_a, n_iters=8, strength=sf_strength)
            sf_m = _score_m(a_sf)
            if sf_m < best_m:
                best_m = sf_m
                best_a = a_sf.copy()
                x = best_a.copy()
                x_prev = x.copy()
            else:
                # Blend towards spectrally flattened
                blend = 0.2
                x = _norm1((1.0 - blend) * best_a + blend * a_sf)
                x_prev = x.copy()

        if verbose and t % 400 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    MD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def _spectral_wf(a, n_iters=10, strength=0.3):
    """Spectral water-filling: flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = _norm1(a.copy())
    size = 2 * n - 1
    fs = _ffts(size)

    for it in range(n_iters):
        A = np.fft.rfft(a, fs)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        # Water-filling: cap peaks and floor valleys
        P_new = np.where(P > P_mean * (1.0 + strength), P_mean * (1.0 + strength), P)
        P_new = np.where(P < P_mean * (1.0 - strength * 0.5), P_mean * (1.0 - strength * 0.5), P_new)

        # Also try a softer version: geometric mean blend
        P_smooth = np.sqrt(P * P_new)

        phase = np.ones_like(A)
        mask = np.abs(A) > 1e-30
        phase[mask] = A[mask] / np.abs(A[mask])

        A_new = np.sqrt(P_smooth) * phase
        a_new = np.fft.irfft(A_new, fs)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s

    return a


def _lp_frank_wolfe(a0, t_budget, solver, init_K=300, max_K=2500, add_per_iter=40, verbose=True):

    n = len(a0)
    a = _norm1(a0.copy())
    best_a = a.copy()
    best_m = _score_m(a)
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    ac_y = _fast_autocorr(y)
    M_curr = ac_y.max()

    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > t_budget:
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = _fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        cons = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), cons)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed += 1
            if failed > 8:
                constraint_list = sorted(constraint_set)[:min(init_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        size = 2 * n - 1
        fs = _ffts(size)
        Yf = np.fft.rfft(y, fs)
        Gf = np.fft.rfft(g_dir, fs)
        conv_ys = np.fft.irfft(Yf * Gf, fs)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            bt = (t - 1.0) / (t + 2.0)
            x_new = y_new + bt * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def _targeted_cd(a0, t_budget, verbose=True):

    n = len(a0)
    a = _norm1(a0.copy())
    best_a = a.copy()
    best_m = _score_m(a)
    t0 = time.monotonic()
    stall = 0
    max_stall = 5000

        ac = _fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0

        # Top peaks
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

            if time.monotonic() - t0 > t_budget:
                break

            i_start = max(0, k - n + 1)

            n_pairs = min(6, len(idx))
            pair_order = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_order:
                if time.monotonic() - t0 > t_budget:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Risk landscape
                risk = np.zeros(n)
                for k2 in top_k[:10]:
                    k2 = int(k2)
                    j_start = max(0, k2 - n + 1)
                    j_end = min(n - 1, k2)
                    if j_start <= j_end:
                        jj = np.arange(j_start, j_end + 1)
                        risk[jj] += ac[k2] * a[k2 - jj]
                        partner = k2 - jj
                        risk[partner] += ac[k2] * a[jj]

                risk[src] = np.inf
                dst = int(np.argmin(risk))

                for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = _score_m(a_trial)
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbations
            n_moves = int(rng.integers(2, 10))
            for mm in range(n_moves):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii != jj and a[ii] > 1e-10:
                    frac = float(rng.uniform(0.005, 0.04))
                    amt = a[ii] * frac
                    a[ii] -= amt
                    a[jj] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        M_new = _score_m(a)
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    TCD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def _get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            xv = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(xv)), [cp.sum(xv) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    print("  Solver: ECOS (fallback)")
    return cp.ECOS


def _quick_md(a0, t_budget):
    """Quick mirror descent for length exploration."""
    n = len(a0)
    return _mirror_descent(a0, t_budget, beta_start=15.0, beta_end=150.0,
                           c0=3.5, sf_every=200, sf_strength=0.2, verbose=False)


def propose_candidate(budget_s=1800.0):
    """Main orchestrator: multi-phase search for optimal sequence."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Total time budget: {time_budget:.0f}s")

    solver = _get_solver()

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = _norm1(p)
            p_score = _eval_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0.0, 1.0, n_base)
        i = np.arange(n_base, dtype=np.float64)
        best_seq = _norm1(1.0 + 0.7 * (2.0 * i / (n_base - 1) - 1.0))
        best_score = _full_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    def _check(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _full_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    n_use = n0
    print(f"\n=== Phase 1: Parametric init (n={n_use}) ===")
    if t1 > 5:
        a_p, s_p = _parametric_init(n_use, t1, best_seq, best_score)
        _check(a_p, "parametric")

    # === Phase 2: Main mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Mirror descent (n={n_best}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 20:
        a_in = _norm1(best_seq)
        a_out, _ = _mirror_descent(a_in, t2, beta_start=10.0, beta_end=300.0,
                                   c0=4.0, sf_every=400, sf_strength=0.25, verbose=True)
        _check(a_out, "md_main")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP Frank-Wolfe (n={n_best}) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 30 and n_best <= 6000:
        a_in = _norm1(best_seq)
        a_out, _ = _lp_frank_wolfe(a_in, t3, solver, init_K=300, max_K=2500,
                                   add_per_iter=40, verbose=True)
        _check(a_out, "lp_main")

    # === Phase 4: Targeted coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Targeted CD (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        a_in = _norm1(best_seq)
        a_out, _ = _targeted_cd(a_in, t4, verbose=True)
        _check(a_out, "tcd_main")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ===")

    length_cands = []
    for delta in [-500, -300, 300, 500, -1000, 1000]:
        nn = n_best + delta
        if 2000 <= nn <= 7000:
            length_cands.append(nn)
    for rn in [3000, 4000, 5000, 6000]:
        if rn != n_best and 2000 <= rn <= 7000:
            length_cands.append(rn)
    length_cands = sorted(set(length_cands))

    tried = 0
    for nn in length_cands:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60 or tried >= 4:
            break
        tried += 1
        print(f"\n  Trying n={nn} (rem={remaining:.0f}s)")
        a_interp = _interp_len(best_seq, nn)
        t5a = min(remaining * 0.35, 120.0)
        if t5a > 10:
            a_out, _ = _quick_md(a_interp, t5a)
            _check(a_out, f"md_n{nn}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and nn <= 5500:
            t5b = min(remaining * 0.4, 100.0)
            if t5b > 20:
                a_in = _norm1(a_interp) if best_seq is None else _norm1(
                    _interp_len(best_seq, nn))
                a_out, _ = _lp_frank_wolfe(a_in, t5b, solver, init_K=250,
                                           max_K=2000, add_per_iter=35, verbose=False)
                _check(a_out, f"lp_n{nn}")

    # === Phase 6: Final mirror descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final MD polish (n={n_best}) ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 15:
        a_in = _norm1(best_seq)
        a_out, _ = _mirror_descent(a_in, t6, beta_start=50.0, beta_end=500.0,
                                   c0=5.0, sf_every=300, sf_strength=0.15, verbose=True)
        _check(a_out, "md_final")

    # === Phase 7: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    t7 = min(remaining * 0.30, 200.0)
    if t7 > 20 and n_best <= 6000:
        a_in = _norm1(best_seq)
        a_out, _ = _lp_frank_wolfe(a_in, t7, solver, init_K=400, max_K=3000,
                                   add_per_iter=50, verbose=True)
        _check(a_out, "lp_final")

    # === Phase 8: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final CD (n={len(best_seq)}) ===")
    t8 = min(remaining * 0.5, 120.0)
    if t8 > 10:
        a_in = _norm1(best_seq)
        a_out, _ = _targeted_cd(a_in, t8, verbose=True)
        _check(a_out, "tcd_final")

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = _norm1(np.sin(np.pi * x) ** 2)
        best_score = _eval_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(v) for v in best_seq]
