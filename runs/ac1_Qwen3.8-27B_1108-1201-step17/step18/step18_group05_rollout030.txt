_global_solver = None

        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(rho_arr):
        return np.maximum(v, 0.0)
    rho = np.nonzero(rho_arr)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):

def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def compute_gradient_fft(a, ac, top_k=80, beta=5.0, use_softmax=False, tau=50.0):
    """Compute gradient using FFT-based correlation (much faster than loop)."""
    n = len(a)
    ac_len = len(ac)
    
    if use_softmax:
        ac_shifted = tau * (ac - ac.max())
        p = np.exp(ac_shifted)
        p = p / p.sum()
    else:
        k_max = min(top_k, ac_len)
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        p = np.zeros(ac_len)
        p[top_idx] = w
    
    # g[j] = sum_k p[k] * a[k-j] for valid k-j
    # This is correlate(p, a, 'valid')
    g = sp_correlate(p, a, mode='valid')[:n]
    return 2.0 * g


def amd_optimize(a0, time_limit, top_k=80, beta=5.0, c0=5.0,
                 restart_every=5000, adaptive_beta=True, use_softmax=False,
                 tau_init=30.0, tau_final=200.0, verbose=False, rng_seed=42):
    """Nesterov-accelerated multiplicative descent with FFT-based gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_tau = tau_init
    max_iters = 10000000

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 2.0)
        beta_t = min(beta_t, 0.85)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Adaptive temperature
        progress = min(1.0, t / (restart_every * 3))
        cur_tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(y)
        g = compute_gradient_fft(y, ac, top_k=cur_top_k, beta=cur_beta,
                                  use_softmax=use_softmax, tau=cur_tau)

        # Polyak learning rate
        eta = gamma / (np.sqrt(t) + 1.0)

        # Multiplicative update (mirror descent with negative entropy)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
            gamma = min(gamma * 1.005, 15.0)
        else:
            stall_count += 1

        if adaptive_beta and stall_count > restart_every // 3:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 5, 200)
            gamma = max(gamma * 0.95, 1.0)

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 250)
            cur_beta = max(cur_beta - 0.5, 2.0)
            use_softmax = not use_softmax
            gamma = c0 * 0.8

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}, tau={cur_tau:.1f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200,
                    top_K_max=None, use_all=False, threshold=0.85):
    """LP-Frank-Wolfe with threshold-based K and geometric line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Threshold-based K selection
        if use_all:
            K_sel = 2 * n - 1
            binding_idx = np.arange(2 * n - 1)
        else:
            if threshold > 0:
                thresh_val = threshold * M_curr
                binding_idx = np.where(ac >= thresh_val)[0]
                if len(binding_idx) > top_K_max:
                    binding_idx = np.argsort(ac)[::-1][:top_K_max]
            else:
                K_sel = min(K, 2 * n - 1)
                binding_idx = np.argsort(ac)[::-1][:K_sel]

        K_sel = len(binding_idx)
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = min(K + 50, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Geometric line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        alphas = [best_alpha, min(1.0, best_alpha * 1.3), min(1.0, best_alpha * 1.7),
                  min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2, 0.1]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1
        K = min(K + 40, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=False, tau_init=20.0, tau_final=300.0,
                              eta_init=0.5, rng_seed=100):
    """Smoothed projected gradient with adaptive temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    max_stall = 400
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 600.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            if stall > 80:
                eta *= 0.75
            if stall > 180:
                pert = rng.normal(0, 0.005, n) * a
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                    stall = 0
                    eta = eta_init * 0.5
                else:
                    a = a_new
            else:
                a = a_new

        total_iters += 1

        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def peak_shave_optimize(a0, time_limit, verbose=False, rng_seed=300, n_top=20):
    """Peak shaving: identify top autocorrelation peaks and redistribute mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity map
        sens = np.zeros(n)
        for k in top_peaks[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_peaks[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destinations
            low_sens_mask = sens < np.percentile(sens, 20)
            candidates = np.where(low_sens_mask)[0]
            if len(candidates) == 0:
                candidates = np.argsort(sens)[:10]
            else:
                candidates = candidates[np.argsort(sens[candidates])[:8]]

            found = False
            for dst in candidates:
                if dst == src or found:
                    continue
                for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 20 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def double_smooth_optimize(a0, time_limit, verbose=False, rng_seed=200):
    """Two-scale smoothed optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = 1.0
    stall = 0
    for phase_tau in [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]:
        if time.monotonic() - t0 > time_limit:
            break
        phase_start = time.monotonic()
        phase_limit = min(time_limit - (phase_start - t0), time_limit / 7.0)
        stall = 0

        while time.monotonic() - phase_start < phase_limit and stall < 250:
            ac = fast_autocorr(a)
            ac_max = ac.max()
            ac_shifted = phase_tau * (ac - ac_max)
            exp_ac = np.exp(ac_shifted)
            p = exp_ac / exp_ac.sum()

            p_cum = np.concatenate([[0.0], np.cumsum(p)])
            k_lo = np.arange(n)
            k_hi = np.minimum(k_lo + n, 2 * n - 1)
            S = p_cum[k_hi] - p_cum[k_lo]
            cross = sp_correlate(p, a, mode='valid')
            g = a * S + cross
            g_tangent = g - g.mean()

            a_new = project_simplex(a - eta * g_tangent)
            m_new = fast_autocorr(a_new).max()

            if m_new < best_m - 1e-15:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                eta = min(eta * 1.015, 3.0)
            else:
                a = a_new
                stall += 1
                if stall > 40:
                    eta *= 0.8
                if stall > 120:
                    pert = rng.normal(0, 0.003, n) * a
                    a_p = np.maximum(a + pert, 0.0)
                    a = a_p / a_p.sum()
                    stall = 60

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    DoubleSmooth tau={phase_tau} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with refined multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_u_shape(n_base, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init AMD ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.18, 280.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.5))
        labels.append("u3.5")
        inits.append(make_symmetric_u(n0, sharpness=2.5, inner=0.5))
        labels.append("sym_u")
        inits.append(make_bimodal(n0, width=0.18))
        labels.append("bimodal")
        inits.append(make_cosine(n0))
        labels.append("cosine")
        inits.append(spectral_init(n0, n_modes=40, seed=42))
        labels.append("spectral")

        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, 3.0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break

            # Alternate between top-k and softmax modes
            use_sm = (idx % 3 == 1)
            print(f"  AMD '{lbl}' softmax={use_sm} (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=80, beta=5.0, c0=5.0,
                                     restart_every=4000, adaptive_beta=True,
                                     use_softmax=use_sm, tau_init=25.0, tau_final=180.0,
                                     verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1200)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all, threshold=0.85)
        try_update(a_out, "lp_main")

    # === Phase 3: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smoothed gradient (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t3, verbose=True,
                                             tau_init=15.0, tau_final=200.0,
                                             eta_init=0.6, rng_seed=100)
        try_update(a_out, "smooth1")

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak shaving (n={n_best}) ===")
    t4 = min(remaining * 0.10, 170.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shave_optimize(a_in, t4, verbose=True, rng_seed=300, n_top=25)
        try_update(a_out, "peakshave")

    # === Phase 5: Double-scale smoothed ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Double-scale smooth (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = double_smooth_optimize(a_in, t5, verbose=True, rng_seed=200)
        try_update(a_out, "dsmooth")

    # === Phase 6: AMD round 2 (focused on best) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: AMD r2 (n={n_best}) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t6, top_k=100, beta=4.0, c0=4.0,
                                 restart_every=3000, adaptive_beta=True,
                                 use_softmax=False, tau_init=40.0, tau_final=250.0,
                                 verbose=False, rng_seed=600)
        try_update(a_out, "amd_r2")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (base n={n_best}) ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 1000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]
    trial_lengths = trial_lengths[:3]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 150.0)
        if t7a > 20:
            a_out, _ = amd_optimize(a_long, t7a, top_k=80, beta=5.0, c0=4.5,
                                     restart_every=4000, adaptive_beta=True,
                                     use_softmax=False, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 4500:
            t7b = min(remaining * 0.35, 120.0)
            if t7b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                use_all = (new_n <= 1200)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                           top_K_init=min(300, 2*new_n-1),
                                           top_K_max=min(2500, 2*new_n-1),
                                           use_all=use_all, threshold=0.85)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final optimization on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final optimization (n={n_best}) ===")
    if remaining > 50:
        t8 = min(remaining * 0.3, 120.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = smooth_gradient_optimize(a_in, t8, verbose=False,
                                                 tau_init=50.0, tau_final=400.0,
                                                 eta_init=0.3, rng_seed=500)
            try_update(a_out, "smooth_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and n_best <= 5500:
            t8b = min(remaining * 0.4, 130.0)
            if t8b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False,
                                           top_K_init=min(500, 2 * n_best - 1),
                                           top_K_max=min(5000, 2 * n_best - 1),
                                           use_all=(n_best <= 1200), threshold=0.82)
                try_update(a_out, "lp_final")

    # === Phase 9: Final AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final AMD polish (n={n_best}) ===")
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t9, top_k=120, beta=3.5, c0=3.0,
                                     restart_every=3000, adaptive_beta=True,
                                     use_softmax=True, tau_init=100.0, tau_final=500.0,
                                     verbose=False, rng_seed=789)
            try_update(a_out, "amd_final")

    # === Phase 10: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_shave_optimize(a_in, t10, verbose=False, rng_seed=999, n_top=30)
            try_update(a_out, "peakshave_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_u_shape(n_fb, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
