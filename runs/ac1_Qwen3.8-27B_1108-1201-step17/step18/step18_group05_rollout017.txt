def fast_autocorr(a):
    """Compute full linear convolution (autocorrelation) of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if any(x < 0 or not np.isfinite(x) for x in a_list):
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient_for_peaks(a, ac, top_k=50, beta=5.0):
    """Compute a soft subgradient focusing on top autocorrelation peaks."""
    n = len(a)
    k_sel = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_sel]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_sel) / k_sel
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=5000,
                 verbose=False, rng_seed=42):
    """Away-Method Dual (mirror descent) optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_gradient_for_peaks(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 5, 200)

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 0.5, 2.0)

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      AMD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200,
                   top_K_max=None, use_all=False, equalize=False):
    """
    LP-based Frank-Wolfe with adaptive constraint generation.
    If equalize=True, also tries to equalize top peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        elif equalize:
            # Select constraints that are close to the max (within 5% of peak)
            threshold = 0.95 * M_curr
            close_idx = np.where(ac >= threshold)[0]
            if len(close_idx) >= K_sel:
                binding_idx = close_idx[:K_sel]
            else:
                binding_idx = np.argsort(ac)[::-1][:K_sel]
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix: T_sel[i, j] = a[binding_idx[i] - j]
        K_act = len(binding_idx)
        T_sel = np.zeros((K_act, n))
        for ii in range(K_act):
            ki = int(binding_idx[ii])
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr, cp.sum(b_var) >= 1e-10]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            if no_improve > 20:
                K = min(K + 100, top_K_max)
                if K >= top_K_max:
                    break
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            no_improve += 1
            continue
        g_dir = b_val / b_sum

        # Backtracking line search with Armijo condition
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        alphas = [best_alpha]
        for multiplier in [1.5, 2.0, 2.5, 3.0]:
            al = min(1.0, best_alpha * multiplier)
            if al > 0.01:
                alphas.append(al)
        alphas.extend([0.5, 0.3, 0.2, 0.1])

        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m - 1e-18:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Perturb to escape
                rng = np.random.default_rng(iter_count * 1000)
                pert = rng.normal(0, 0.002, n) * a
                a_p = np.maximum(a + pert, 0.0)
                if a_p.sum() > 1e-15:
                    a = a_p / a_p.sum()
                no_improve = 10

        iter_count += 1

        if iter_count % 5 == 0:
            K = min(K + 20, top_K_max)

        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, no_improve={no_improve}")

    return best_a, best_m


def peak_equalization(a0, time_limit, verbose=True, rng_seed=300):
    """
    Try to equalize the top autocorrelation peaks by moving mass
    from positions contributing to high peaks to positions contributing to low peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_peaks = min(20, len(ac))
        top_idx = np.argsort(ac)[::-1][:n_peaks]
        top_vals = ac[top_idx]
        mean_top = top_vals.mean()
        std_top = top_vals.std()

        if std_top < 1e-12:
            break

        # Find which positions of a contribute most to the highest peak
        k_max = int(top_idx[0])
        j_start = max(0, k_max - n + 1)
        j_end = min(n, k_max + 1)
        idx = np.arange(j_start, j_end)
        contrib = a[idx] * a[k_max - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            break

        # Find positions with low contribution to any top peak
        in_peak_mask = np.zeros(n, dtype=bool)
        for ki in top_idx:
            ki_i = int(ki)
            s = max(0, ki_i - n + 1)
            e = min(n, ki_i + 1)
            in_peak_mask[s:e] = True

        low_contrib_positions = np.where(~in_peak_mask)[0]
        if len(low_contrib_positions) == 0:
            low_contrib_positions = np.arange(n)

        # Try moving mass from highest contributor to random low-contrib position
        src = idx[np.argmax(contrib)]
        src_weight = contrib[src] / total_contrib

        if a[src] < 1e-12:
            stall += 1
            continue

        # Try multiple destination and fraction combinations
        n_trials = min(10, len(low_contrib_positions))
        trial_positions = rng.choice(low_contrib_positions, n_trials, replace=False)
        found = False

        for dst in trial_positions:
            if dst == src:
                continue
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                # Also check if top peaks are more equalized
                ac_trial = fast_autocorr(a_trial)
                top_trial = np.sort(ac_trial)[::-1][:n_peaks]
                std_trial = top_trial.std()

                if m_trial < best_m - 1e-15:
                    best_m = m_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    stall = 0
                    found = True
                    break
                elif std_trial < std_top * 0.9 and m_trial <= best_m + 1e-10:
                    # Accept if it equalizes peaks even if max doesn't improve
                    a = a_trial
                    stall = 0
                    found = True
                    break
            if found:
                break

        if not found:
            stall += 1
            if stall > 50:
                # Global perturbation
                idx_pert = rng.choice(n, size=min(n // 10, 50), replace=False)
                for ip in idx_pert:
                    if a[ip] > 1e-10:
                        amt = a[ip] * rng.uniform(0.01, 0.05)
                        a[ip] -= amt
                        a[rng.integers(n)] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stall = 30

        if verbose and (stall % 50 == 0 or found):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      PeakEq ({elapsed:.0f}s): J={J:.6f}, stall={stall}, std_top={std_top:.2e}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Aggressive coordinate descent with sensitivity-based move selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 80
    n_top = 15
    round_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity: how much does each position of a contribute to peak?
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for ki in top_k[:8]:
            ki_i = int(ki)
            s = max(0, ki_i - n + 1)
            e = min(n, ki_i + 1)
            idx = np.arange(s, e)
            in_peak[idx] = True
            sens[idx] += a[ki_i - idx]

        improved = False

        for ki in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki_i = int(ki)
            s = max(0, ki_i - n + 1)
            e = min(n, ki_i + 1)
            idx = np.arange(s, e)
            contrib = a[idx] * a[ki_i - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destinations
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            n_cands = min(5, len(candidates))
            top_cands = candidates[np.argsort(cand_sens)[:n_cands]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        round_count += 1
        if verbose and round_count % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=True, tau_init=10.0, tau_final=200.0,
                             eta_init=0.5, rng_seed=100):
    """Smoothed projected gradient with temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    max_stall = 400
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 400.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        if ac_max < 1e-15:
            break
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        # Gradient computation
        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross

        g_tangent = g - g.mean()
        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.015, 2.0)
        else:
            a = a_new
            stall += 1
            if stall > 80:
                eta *= 0.75
            if stall > 200:
                pert = rng.normal(0, 0.004, n) * a
                a_p = np.maximum(a + pert, 0.0)
                a = a_p / a_p.sum()
                m_p = fast_autocorr(a).max()
                if m_p < best_m:
                    best_m = m_p
                    best_a = a.copy()
                stall = 100
                eta = max(eta, 0.1)

        total_iters += 1

        if verbose and total_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def enter_the_battery(a0, time_limit, solver, verbose=True):
    """
    Enter-the-battery style optimization.
    Maintains active set of binding constraints.
    At each step, solves LP to find best direction reducing max violation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Initial active set: top 50 peaks
    active_set = set()
    ac = fast_autocorr(a)
    for k in np.argsort(ac)[::-1][:50]:
        active_set.add(int(k))

    iter_count = 0

    while time.monotonic() - t0 < time_limit and len(active_set) < 2 * n - 1:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Update active set: add peaks that are at or near the max
        for k in range(2 * n - 1):
            if ac[k] >= M_curr * 0.99:
                active_set.add(int(k))
        # Remove peaks that are well below the max (if set is too large)
        if len(active_set) > 500:
            sorted_active = sorted(ac[k] for k in active_set)
            threshold = sorted_active[len(sorted_active) // 2]
            active_set = {k for k in active_set if ac[k] >= threshold}

        if len(active_set) == 0:
            break

        # Build LP: maximize sum(b) s.t. b >= 0, (a*b)[k] <= M_curr for k in active
        K_act = len(active_set)
        active_list = sorted(active_set)
        T_sel = np.zeros((K_act, n))
        for ii, ki in enumerate(active_list):
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr, cp.sum(b_var) >= 1e-10]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_m_trial = M_curr
        for alpha in [best_alpha, min(1.0, best_alpha * 2), 0.5, 0.3, 0.2]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      ETB iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |active|={len(active_set)}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with improved time allocation."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = 1.0 + 2.0 * x ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init AMD screening ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.18, 250.0)

    if t_phase1 > 50:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.2))
        labels.append("u1.2")
        inits.append(make_u_shape(n0, sharpness=2.5))
        labels.append("u2.5")
        inits.append(make_u_shape(n0, sharpness=4.0))
        labels.append("u4.0")
        inits.append(make_bimodal(n0, width=0.12, pos1=0.1, pos2=0.9))
        labels.append("bimodal")
        inits.append(make_cosine(n0, order=1))
        labels.append("cos1")
        inits.append(make_raised_cosine(n0, beta=1.5))
        labels.append("rc1.5")
        inits.append(spectral_init(n0, n_modes=40, seed=7))
        labels.append("spec40")

        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, 2.0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 40)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                    restart_every=3000, rng_seed=42 + idx * 7)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: Intensive LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1200)
        top_K_init = min(400, 2 * n_best - 1)
        top_K_max = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max,
                                   use_all=use_all, equalize=False)
        try_update(a_out, "lp_main")

    # === Phase 3: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak equalization (n={n_best}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_equalization(a_in, t3, verbose=True, rng_seed=300)
        try_update(a_out, "peq")

    # === Phase 4: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Smoothed gradient (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t4, verbose=True,
                                             tau_init=10.0, tau_final=150.0,
                                             eta_init=0.6, rng_seed=100)
        try_update(a_out, "smooth")

    # === Phase 5: LP-FW with equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW equalize (n={n_best}) ===")
    t5 = min(remaining * 0.13, 180.0)
    if t5 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1200)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max,
                                   use_all=use_all, equalize=True)
        try_update(a_out, "lp_eq")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate polish (n={n_best}) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord1")

    # === Phase 7: Enter-the-battery ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Enter-the-battery (n={n_best}) ===")
    t7 = min(remaining * 0.12, 150.0)
    if t7 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = enter_the_battery(a_in, t7, solver, verbose=True)
        try_update(a_out, "etb")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 2000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.4, 120.0)
        if t8a > 15:
            a_out, _ = amd_optimize(a_long, t8a, top_k=60, beta=6.0, c0=4.0,
                                    restart_every=4000, rng_seed=new_n % 997)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 5000:
            t8b = min(remaining * 0.4, 100.0)
            if t8b > 15:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                use_all = (new_n <= 1200)
                top_K = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False,
                                           top_K_init=min(300, 2 * new_n - 1),
                                           top_K_max=top_K, use_all=use_all)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final intensive LP + coordinate ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final optimization (n={n_best}) ===")
    if remaining > 30:
        t9a = min(remaining * 0.4, 100.0)
        if t9a > 15 and n_best <= 6000:
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 1200)
            top_K_init = min(600, 2 * n_best - 1)
            top_K_max = min(5000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t9a, solver, verbose=False,
                                       top_K_init=top_K_init, top_K_max=top_K_max,
                                       use_all=use_all)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        t9b = min(remaining * 0.3, 60.0)
        if t9b > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9b, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = 1.0 + 2.0 * x ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
