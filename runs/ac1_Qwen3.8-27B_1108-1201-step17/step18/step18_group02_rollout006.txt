"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. High-speed LP Frank-Wolfe with adaptive constraint management (scipy linprog/HiGHS)
2. Subgradient smoothing with top-k max approximation
3. Targeted peak-shaving coordinate descent
4. Multi-length exploration with interpolation
5. Aggressive final polish

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    a_list = [float(x) for x in a]

def lp_solve_fpw(y, constraint_indices, n):
    """Solve the LP: min M s.t. sum(g)=1, g>=0, A_k(g) <= M for k in constraint_indices."""
    K = len(constraint_indices)
    if K == 0:
        return None
    
    # Build constraint matrix (sparse)
    # Row for constraint k: A[k,j] = y[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    
    for ii, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(ii)
            cols.append(j)
            vals.append(y[k - j])
    
    # A_ub: K x (n+1) matrix
    # For each constraint: sum_j A[k,j]*g_j - M <= 0
    # So the M column has -1 for each row
    A_ub_data = vals + [-1.0] * K
    A_ub_rows = rows + [ii for ii in range(K)]
    A_ub_cols = cols + [n for _ in range(K)]  # M is variable n
    
    A_ub = csr_matrix((A_ub_data, (A_ub_rows, A_ub_cols)), shape=(K, n + 1))
    b_ub = np.zeros(K)
    
    # A_eq: sum(g) = 1, M free → [1, 1, ..., 1, 0] * [g_0, ..., g_{n-1}, M] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: g_i >= 0, M >= 0
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    g = np.maximum(result.x[:n], 0.0)
    gs = g.sum()
    if gs < 1e-15:
        return None
    g = g / gs
    M_val = result.x[n]
    return g, M_val

def fpw_optimize(a0, time_limit, max_K=2000, verbose=True):
    """LP Frank-Wolfe with adaptive constraints and HiGHS solver."""
    n = len(a0)
    y = norm_sum1(a0).copy()

    best_y = y.copy()

    for k in random_idx:
        constraints.add(int(k))
    constraints = set(list(constraints)[:max_K])
    
    iter_count = 0
    stagnation = 0

            best_y = y.copy()
            stagnation = 0
        else:
            stagnation += 1

                break
            continue

        if violations.any():
            viol_idx = np.where(violations)[0]

            top_viol = viol_idx[np.argsort(viol_M)[::-1][:50]]
            for k in top_viol:
                constraints.add(int(k))

        # FW gap: f(y) - min_{g in simplex} grad f(y) · g
        # Approximate: M_curr - M_lp (not exact but useful)

        if gap < 1e-12:
            gap = 1e-12
        
        # Backtracking line search
        alpha = 2.0 / (iter_count + 2.0)
        for _ in range(20):
            y_trial = (1 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            s_t = y_trial.sum()
            if s_t > 1e-15:
                y_trial = y_trial / s_t
            M_trial = fast_autocorr(y_trial).max()
            if M_trial < M_curr + 1e-15:
                break
            alpha *= 0.5
        else:
            alpha = 1e-12
        
        y_new = (1 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        y = y_new
        
        # Polyak averaging
        x_avg = (1.0 / (iter_count + 1.0)) * y + (iter_count / (iter_count + 1.0)) * x_avg
        
        # Check averaged solution
        ac_avg = fast_autocorr(x_avg)
        if ac_avg.max() < best_M:
            best_M = ac_avg.max()
            best_y = x_avg.copy()

            y = best_y + pert
            y = np.maximum(y, 1e-10)
            y = y / y.sum()
            x_avg = y.copy()
            stagnation = 0
            # Reset constraints
            ac = fast_autocorr(y)
            top_idx = np.argsort(ac)[::-1][:init_K]
            constraints = set(int(k) for k in top_idx)
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    FPW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraints)}, alpha={alpha:.4f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    if verbose:
        print(f"    FPW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_y, best_M

        a_new = a_new / a_new.sum()
        a = a_new
        
        tau_seq *= 0.999
        step += 1
        
        if verbose and step % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    SubGrad step={step} ({elapsed:.0f}s): J={J:.6f}, tau={tau_seq:.2e}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    if verbose:
        print(f"    SubGrad done ({elapsed:.0f}s): J={J:.6f}, steps={step}")
    return best_a, best_M


def coord_shave(a0, time_limit, verbose=True):
    """Targeted peak shaving: move mass from high-contributing to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_M - 1e-15:
            best_M = M_curr
            best_a = a.copy()
        
        # Find top peaks
        n_peaks = min(10, 2 * n - 1)
        top_peaks = np.argsort(ac)[::-1][:n_peaks]
        
        improved = False
        
        for k in top_peaks:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination: minimize increase in all top peaks
            # For each candidate destination d, compute the increase in each top peak
            # Increase in peak k2 from moving delta from src to dst:
            # (a*a)[k2] changes by: a[dst]*a[k2-dst] + a[dst]*a[src... 
            # This is complex. Approximate: look for dst that contributes least to top peaks.
            
            scores = np.zeros(n)
            for k2 in top_peaks[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            
            # Exclude positions already in the peak (they'd increase the peak)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:4]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            dst = candidates[np.argmin(scores[candidates])]
            if dst == src:
                continue
            
            # Try transferring
            found = False
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-16:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if not found:
                # Small random perturbation
                j = int(rng.integers(n))
                if j != src:
                    amt = src_val * 0.005
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0) / a.sum()
        
        if not improved:
            # Global perturbation
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-10:
                amt = a[i1] * rng.uniform(0.001, 0.01)
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
        
        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    CoordShave iter={iter_count} ({elapsed:.0f}s): J={J:.6f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    if verbose:
        print(f"    CoordShave done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_M


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
