"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-parametric Gaussian-bump initialization with L-BFGS-B
2. Nesterov-accelerated projected subgradient with exact max and adaptive step
3. Frequency-domain spectral equalization (iterative flatness via FFT)
4. LP Frank-Wolfe with warm starts and adaptive constraint management
5. Multi-resolution coordinate descent polish with peak-shifting
6. Multi-length and multi-restart exploration
"""

import numpy as np
import time
import math
from scipy.optimize import minimize as sp_minimize

# ============== Top-level helper functions ==============

def fast_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_nonneg(a):
    """Clamp to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_fast(a):
    """Fast evaluation: 2*n*max(conv)/(sum)^2 assuming sum=1, a>=0."""
    ac = fast_conv(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * len(a) * ac.max() / (s * s)


def proj_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    tmp = sv + np.arange(1, n + 1)
    cond = np.where(1.0 / np.arange(1, n + 1) * (tmp.cumsum() - 1.0) > 0)[0]
    if len(cond) == 0:
        return np.ones(n) / n
    idx = cond[-1]
    mu = (sv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - mu, 0.0)


def subgrad_of_max(a):
    """Compute subgradient of max_k conv(a,a)[k] at point a.
    Returns (grad, k_star) where grad is subgradient w.r.t. a."""
    n = len(a)
    ac = fast_conv(a)
    k_star = int(np.argmax(ac))
    # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0
    grad = np.zeros(n)
    i_start = max(0, k_star - n + 1)
    i_end = min(n, k_star + 1)
    if i_start < i_end:
        idx = np.arange(i_start, i_end)
        grad[idx] = 2.0 * a[k_star - idx]
    return grad, ac, k_star


def nesterov_subgrad(a0, time_limit, lr_init=0.5, lr_decay=0.9999,
                     verbose=True, seed=42):
    """Nesterov-accelerated projected subgradient on simplex."""
    n = len(a0)
    a = norm_nonneg(a0)
    a_prev = a.copy()
    best_a = a.copy()
    best_val = eval_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    lr = lr_init
    t_iter = 1
    stall = 0
    stall_patience = 500
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        momentum = (t_iter - 1.0) / (t_iter + 1.0)
        y = a + momentum * (a - a_prev)
        y = proj_simplex(y)
        
        # Compute subgradient at y
        grad, ac, k_star = subgrad_of_max(y)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step: eta = lr / sqrt(t_iter)
        eta = lr / math.sqrt(max(t_iter, 1))
        
        # Try update with backtracking (simplified: just clip)
        a_trial = proj_simplex(y - eta * grad)
        val_trial = eval_fast(a_trial)
        
        if val_trial < best_val - 1e-15:
            best_val = val_trial
            best_a = a_trial.copy()
            stall = 0
        elif val_trial < best_val * 1.001:
            stall += 1
        else:
            # Step too large, reduce
            eta = eta * 0.5
            a_trial = proj_simplex(y - eta * grad)
            val_trial = eval_fast(a_trial)
            if val_trial < best_val:
                best_val = val_trial
                best_a = a_trial.copy()
                stall = 0
            else:
                stall += 1
        
        a_prev = a
        a = a_trial
        t_iter += 1
        lr = max(lr * lr_decay, lr_init * 0.01)
        
        # Random restart if stalled
        if stall > stall_patience:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            a = proj_simplex(best_a + pert)
            a_prev = a.copy()
            lr = lr_init * 0.5
            t_iter = 1
        
        if verbose and t_iter % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Nesterov t={t_iter} ({elapsed:.0f}s): J={best_val:.6f}, k*={k_star}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"      Nesterov done ({elapsed:.0f}s): J={best_val:.6f}")
    return best_a, best_val


def freq_equalize(a0, time_limit, n_iters=None, verbose=True):
    """Frequency-domain spectral equalization to flatten convolution."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_val = eval_fast(a)
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    if n_iters is None:
        n_iters = int(time_limit * 50)  # rough estimate
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current convolution
        ac = fast_conv(a)
        target = (a.sum() ** 2) / (2 * n - 1)
        
        # Weight: emphasize reducing peaks
        w = np.ones(size)
        peaks = ac > target * 1.5
        w[peaks] = target / np.maximum(ac[peaks], 1e-15)
        w = np.clip(w, 0.1, 2.0)
        
        # In frequency domain: adjust to reduce peak lags
        A = np.fft.rfft(a, fft_size)
        # The convolution in freq domain is A^2
        # We want to "de-emphasize" contributions to peak lags
        # Simple approach: multiply by a gentle window derived from w
        W = np.fft.rfft(w, fft_size)
        A_adj = A * np.sqrt(np.maximum(W / W[W > 0].mean(), 0.1) if (W > 0).any() else 1.0)
        
        a_raw = np.fft.irfft(A_adj, fft_size)[:n]
        a_new = norm_nonneg(a_raw)
        
        # Blend to avoid oscillation
        blend = min(0.3 + it * 0.005, 0.8)
        a = blend * a_new + (1 - blend) * a
        a = norm_nonneg(a)
        
        val = eval_fast(a)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a.copy()
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FreqEq it={it} ({elapsed:.0f}s): J={best_val:.6f}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"      FreqEq done ({elapsed:.0f}s): J={best_val:.6f}")
    return best_a, best_val


def lp_frank_wolfe(a0, time_limit, initial_K=400, max_K=3000,
                   add_per_iter=30, verbose=True):
    """LP Frank-Wolfe with warm starts."""
    try:
        import cvxpy as cp
    except ImportError:
        if verbose:
            print("      cvxpy not available, skipping LP-FW")
        return a0, eval_fast(a0)
    
    # Find a solver
    solver = None
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    if verbose:
        print(f"      LP solver: {solver}")
    
    n = len(a0)
    y = norm_nonneg(a0)
    x = y.copy()
    best_a = y.copy()
    best_m = fast_conv(y).max()
    t0 = time.monotonic()
    
    ac_init = fast_conv(y)
    init_idx = np.argsort(ac_init)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    t_iter = 0
    lp_fails = 0
    
    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[j]) for j in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix A @ g <= M
        # Row k: sum_i y[k-i]*g[i] <= M for valid i
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_s = max(0, k_int - n + 1)
            i_e = min(n, k_int + 1)
            if i_s < i_e:
                idx = np.arange(i_s, i_e)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        cons = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), cons)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fails += 1
            if lp_fails > 8 and len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fails += 1
            if lp_fails > 15:
                constraint_set = set(int(k) for k in constraint_list[:max(200, initial_K//2)])
            continue
        
        lp_fails = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        
        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_yg > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # FW update
        alpha = 2.0 / (t_iter + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        else:
            y_new = y.copy()
        
        # Accelerated FW (FISTA-like)
        if t_iter == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t_iter - 1.0) / (t_iter + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new / sx
            else:
                x_new = y_new.copy()
        
        y = y_new
        x = x_new
        
        # Check x too
        ac_x = fast_conv(x)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()
        
        if verbose and t_iter % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW it={t_iter} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"      LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={t_iter}")
    return best_a, J


def coord_descent(a0, time_limit, verbose=True, seed=42):
    """Peak-shifting coordinate descent."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_val = eval_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))
        
        # Find the pair (i, k_star-i) with largest product
        i_s = max(0, k_star - n + 1)
        i_e = min(n, k_star + 1)
        if i_s >= i_e:
            stall += 1
            break
        
        idx = np.arange(i_s, i_e)
        products = a[idx] * a[k_star - idx]
        src = idx[np.argmax(products)]
        src_val = a[src]
        if src_val < 1e-12:
            stall += 1
            break
        
        # Find best destination: minimize impact on top peaks
        top_k = np.argsort(ac)[::-1][:min(10, len(ac))]
        
        scores = np.zeros(n)
        for k2 in top_k[:6]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n, k2i + 1)
            if s2 < e2:
                idx2 = np.arange(s2, e2)
                scores[idx2] += a[k2i - idx2]
        
        # Try to move small amount to a low-score position
        in_hot = np.zeros(n, dtype=bool)
        for k2 in top_k[:5]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n, k2i + 1)
            if s2 < e2:
                in_hot[s2:e2] = True
        
        candidates = np.where(~in_hot)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        
        cand_scores = scores[candidates]
        dst = candidates[np.argmin(cand_scores)]
        
        improved = False
        for frac in [0.01, 0.03, 0.06, 0.1, 0.15, 0.25]:
            amt = src_val * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] = np.maximum(a_trial[src] - amt, 0.0)
            a_trial[dst] += amt
            a_trial = norm_nonneg(a_trial)
            val = eval_fast(a_trial)
            if val < best_val - 1e-16:
                a = a_trial
                best_val = val
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i_rand = int(rng.integers(n))
            j_rand = int(rng.integers(n))
            if i_rand != j_rand and a[i_rand] > 1e-10:
                frac = rng.uniform(0.001, 0.03)
                amt = a[i_rand] * frac
                a[i_rand] -= amt
                a[j_rand] += amt
                a = norm_nonneg(a)
                val = eval_fast(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall > 0 and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Coord ({elapsed:.0f}s): J={best_val:.6f}, stall={stall}")
    
    if verbose:
        print(f"      Coord done: J={best_val:.6f}")
    return best_a, best_val


def parametric_init_gaussian(n, n_bumps=20, rng_seed=0):
    """Initialize with Gaussian bumps and optimize parameters."""
    rng = np.random.default_rng(rng_seed)
    positions = sorted(rng.uniform(0.05, 0.95, n_bumps).tolist())
    widths = rng.uniform(0.03, 0.15, n_bumps)
    amplitudes = rng.uniform(0.5, 2.0, n_bumps)
    
    x = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for pos, w, amp in zip(positions, widths, amplitudes):
        a += amp * np.exp(-((x - pos) ** 2) / (2 * w * w))
    a = np.maximum(a, 1e-6)
    return norm_nonneg(a)


def gauss_amp_fn(params, n, positions, widths):
    """Objective function for L-BFGS-B on amplitudes."""
    amps = np.abs(params)  # ensure positive
    x = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for j, (pos, w) in enumerate(zip(positions, widths)):
        a += amps[j] * np.exp(-((x - pos) ** 2) / (2 * w * w))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return eval_fast(a)


def multi_param_optimize(a0, n, time_limit, verbose=True, seed=0):
    """Optimize Gaussian bump amplitudes with L-BFGS-B."""
    rng = np.random.default_rng(seed)
    n_bumps = min(n // 150, 30)
    positions = sorted(rng.uniform(0.05, 0.95, n_bumps).tolist())
    widths = rng.uniform(0.03, 0.12, n_bumps)
    
    # Initial amplitudes from a0
    x = np.linspace(0.0, 1.0, n)
    init_amps = np.abs(rng.normal(1.0, 0.3, n_bumps))
    
    t0 = time.monotonic()
    try:
        def objective(params):
            if time.monotonic() - t0 > time_limit:
                return np.inf
            return gauss_amp_fn(params, n, positions, widths)
        
        result = sp_minimize(objective, init_amps, method='L-BFGS-B',
                            options={'maxiter': 5000, 'maxfun': 10000},
                            timeout=time_limit)
    except Exception:
        if verbose:
            print("      L-BFGS-B failed, using initial")
        return norm_nonneg(a0)
    
    if verbose and time.monotonic() - t0 < time_limit:
        print(f"      L-BFGS-B: {result.fun:.6f}, success={result.success}")
    
    # Reconstruct
    amps = np.abs(result.x)
    a = np.zeros(n)
    for j, (pos, w) in enumerate(zip(positions, widths)):
        a += amps[j] * np.exp(-((x - pos) ** 2) / (2 * w * w))
    a = np.maximum(a, 1e-10)
    return norm_nonneg(a)


def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = norm_nonneg(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return norm_nonneg(f_interp)


def evaluate_safe(a):

            print(f"  *** NEW BEST [{label}]: J={val:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Parametric init + L-BFGS-B ===
    print(f"\n=== Phase 1: Parametric + L-BFGS-B (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    
    if t1 > 10:
        # Try a few different bump configurations
        for trial in range(3):
            if time.monotonic() - t_global > time_budget - t1:
                break
            t_sub = (t1 - (time.monotonic() - t_global)) / (3 - trial)
            if t_sub < 5:
                break
            a_trial = multi_param_optimize(best_seq, n0, t_sub, verbose=(trial==0),
                                           seed=trial * 100 + 7)
            check_and_update(a_trial, f"param_{trial}")
    
    # Also try simple parametric families
    x = np.linspace(0, 1, n0)
    i_arr = np.arange(n0, dtype=np.float64)
    n_best = len(best_seq)
    param_families = []
    for freq in [1, 2, 3]:
        param_families.append(np.sin(np.pi * freq * x) ** 2 + 0.01)
    for p in [0.5, 1.0, 2.0]:
        param_families.append((i_arr + 1) ** p * (n0 - i_arr + 1) ** p + 0.01)
    for c in [0.3, 0.5, 0.7]:
        for s in [0.1, 0.2]:
            param_families.append(np.exp(-((x - c)**2) / (2*s*s)) + 0.01)
    
    for pf in param_families:
        check_and_update(pf, "parametric")
    
    # === Phase 2: Nesterov subgradient ===
    print(f"\n=== Phase 2: Nesterov subgradient (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    
    if t2 > 30:
        a_in = norm_nonneg(best_seq)
        a_out, _ = nesterov_subgrad(a_in, t2, lr_init=0.3, verbose=True, seed=7)
        check_and_update(a_out, "nesterov_1")
        
        # Second run with different params
        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.10, 150.0)
        if t2b > 20:
            a_in2 = norm_nonneg(best_seq)
            a_out2, _ = nesterov_subgrad(a_in2, t2b, lr_init=0.1,
                                         lr_decay=0.99995, verbose=False, seed=99)
            check_and_update(a_out2, "nesterov_2")
    
    # === Phase 3: Frequency equalization ===
    print(f"\n=== Phase 3: Frequency equalization (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    
    if t3 > 15 and n_best <= 6000:
        a_in = norm_nonneg(best_seq)
        a_out, _ = freq_equalize(a_in, t3, verbose=True)
        check_and_update(a_out, "freq_eq_1")
        
        # Follow up with Nesterov after frequency step
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.08, 100.0)
        if t3b > 15:
            a_out2, _ = nesterov_subgrad(norm_nonneg(best_seq), t3b,
                                         lr_init=0.2, verbose=False, seed=200)
            check_and_update(a_out2, "post_freq")
    
    # === Phase 4: LP Frank-Wolfe ===
    print(f"\n=== Phase 4: LP Frank-Wolfe (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 400.0)
    
    if t4 > 30 and n_best <= 5500:
        a_in = norm_nonneg(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, initial_K=500, max_K=4000,
                                   add_per_iter=40, verbose=True)
        check_and_update(a_out, "lp_fw_1")
    
    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate descent (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    
    if t5 > 15:
        a_in = norm_nonneg(best_seq)
        a_out, _ = coord_descent(a_in, t5, verbose=False, seed=42)
        check_and_update(a_out, "coord_1")
    
    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    candidate_lengths = []
    for delta in [300, -300, 600, -600, 1000, -1000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000:
            candidate_lengths.append(new_n)
    
    seen = set()
    unique_lengths = []
    for ln in candidate_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)
    
    for new_n in unique_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  --- n={new_n} ---")
        a_interp = interpolate_sequence(best_seq, new_n)
        check_and_update(a_interp, f"interp_{new_n}")
        
        # Quick Nesterov
        t6a = min(remaining * 0.3, 120.0)
        if t6a > 15:
            a_out, _ = nesterov_subgrad(norm_nonneg(best_seq), t6a,
                                        lr_init=0.3, verbose=False, seed=new_n % 1000)
            check_and_update(a_out, f"nest_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 5000:
            t6b = min(remaining * 0.4, 200.0)
            if t6b > 20:
                a_out2, _ = lp_frank_wolfe(norm_nonneg(best_seq), t6b,
                                            initial_K=300, max_K=2500,
                                            add_per_iter=30, verbose=False)
                check_and_update(a_out2, f"lp_n{new_n}")
    
    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    
    if remaining > 30:
        # Nesterov with very fine tuning
        t7a = min(remaining * 0.4, 150.0)
        if t7a > 15:
            a_out, _ = nesterov_subgrad(norm_nonneg(best_seq), t7a,
                                        lr_init=0.05, lr_decay=0.99999,
                                        verbose=False, seed=777)
            check_and_update(a_out, "final_nest")
        
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.4, 100.0)
        if t7b > 10:
            a_out2, _ = coord_descent(norm_nonneg(best_seq), t7b,
                                      verbose=False, seed=1234)
            check_and_update(a_out2, "final_coord")
        
        # One more LP-FW if time permits
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and n_final <= 5000:
            t7c = min(remaining * 0.6, 200.0)
            if t7c > 30:
                a_out3, _ = lp_frank_wolfe(norm_nonneg(best_seq), t7c,
                                            initial_K=600, max_K=5000,
                                            add_per_iter=50, verbose=True)
                check_and_update(a_out3, "final_lp")
    
    # Final validation with the actual evaluate_sequence
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        else:
            # Fallback
            n_fb = 3200
            x = np.linspace(0, 1, n_fb)
            best_seq = norm_nonneg(np.sin(np.pi * x) ** 2)
            best_score = eval_fast(best_seq)
    
    elapsed = time.monotonic() - t_global
    print(f"\nFINAL: J={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    
    return [float(x) for x in best_seq]
