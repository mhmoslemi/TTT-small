def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t0 = time.monotonic()

    for schedule in ['const', 'decay', 'adaptive']:
        if time.monotonic() - t0 > time_limit:
            break
        t_phase = time.monotonic() - t0
        phase_time = time_limit - t_phase

        if schedule == 'const':
            betas = [30.0, 50.0, 100.0, 200.0]
            gammas = [3.0, 5.0, 8.0]
        elif schedule == 'decay':
            betas = [100.0]
            gammas = [5.0]
        else:
            betas = [50.0]
            gammas = [5.0]

        for beta in betas:
            for gamma in gammas:
                if time.monotonic() - t0 > time_limit:
                    break
                x = best_a.copy()
                x_prev = best_a.copy()
                max_iters = 20000
                iters_per_call = max(100, max_iters // (len(betas) * len(gammas)))
                stall = 0

                for t in range(1, iters_per_call + 1):
                    if time.monotonic() - t0 > time_limit:
                        break

                    beta_t = (t - 1.0) / (t + 2.0)
                    y = x + beta_t * (x - x_prev)
                    y = np.maximum(y, 1e-15)
                    s_y = y.sum()
                    if s_y < 1e-15:
                        y = x.copy()
                    else:
                        y = y / s_y

                    ac = fast_autocorr(y)
                    ac_max = ac.max()
                    ac_shifted = ac - ac_max
                    exp_vals = np.exp(beta * ac_shifted)
                    w_sum = exp_vals.sum()
                    if w_sum < 1e-30:
                        break
                    w = exp_vals / w_sum

                    # Gradient computation via FFT
                    fft_len2 = 1 << (3 * n - 3).bit_length()
                    W_ext = np.zeros(fft_len2)
                    W_ext[:len(w)] = w
                    A_ext = np.zeros(fft_len2)
                    A_ext[:n] = y
                    Wf2 = np.fft.rfft(W_ext)
                    Af2 = np.fft.rfft(A_ext)
                    grad_raw = np.fft.irfft(Wf2 * np.conj(Af2), fft_len2)[:n]
                    grad = 2.0 * grad_raw

                    # Adaptive step: reduce if objective increases
                    eta = gamma / (np.sqrt(t) + 1.0)
                    for _ in range(5):
                        log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
                        max_log = log_a.max()
                        a_new = np.exp(log_a - max_log)
                        a_new = np.maximum(a_new, 1e-30)
                        s_new = a_new.sum()
                        if s_new < 1e-15:
                            break
                        a_new = a_new / s_new
                        M_new = fast_autocorr(a_new).max()
                        if M_new <= ac_max + 1e-10 or eta < 1e-6:
                            break
                        eta *= 0.5

                    x_prev = x
                    x = a_new

                    M_new = fast_autocorr(x).max()
                    if M_new < best_m - 1e-14:
                        best_m = M_new
                        best_a = x.copy()
                        stall = 0
                    else:
                        stall += 1

                    if stall > 3000:
                        break

    return best_a, best_m


def lp_fw_improved(a0, time_limit, solver, initial_K=400, max_K=3000,
                   add_per_iter=40, verbose=True):
    """
    Improved LP-Frank-Wolfe with:
    - Larger initial active set
    - Aggressive constraint addition
    - Periodic full-constraint verification
    - Backtracking line search
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = sorted(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 1500
    lp_chars = {}  # Cache LP characteristics for warm starting

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: no improve for " + str(no_improve) + " iters, stopping at t=" + str(t))
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = sorted(int(active_set[i]) for i in keep)

        K = len(active_set)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_set):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for violated constraints and add them
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.append(int(k))
            active_set = sorted(set(active_set))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, default_alpha * 0.5, default_alpha * 1.5,
                  1.0 / (t + 1), 0.1, 0.3, 0.5]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + repr(round(elapsed, 1)) + "s): J=" + repr(J) +
                  ", |S|=" + str(len(active_set)))

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True):
    """
    Targeted peak shaving: identify the top convolution peaks and try to
    reduce them by moving mass from contributing positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_val = ac.max()
        # Get top peak indices
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue

            # Find the position contributing most to this peak
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find a good destination: minimize impact on all top peaks
            dst_scores = np.zeros(n)
            for k2 in top_k:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    dst_scores[idx2] += a[k2i - idx2]
                    dst_scores[k2i - idx2] += a[idx2]

            # Prefer positions not in any peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            # Pick the candidate with lowest score
            cand_scores = dst_scores[candidates]
            # Add some randomness to explore
            if rng.random() < 0.2:
                dst = int(candidates[rng.integers(len(candidates))])
            else:
                dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            # Try multiple step sizes
            for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        # Random perturbations when stalled
        if not improved:
            stall += 1
            if stall % 100 == 0:
                n_shifts = rng.integers(1, 5)
                for _ in range(n_shifts):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave (" + repr(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def multi_start_refine(a0, n, time_limit, n_starts=5, verbose=True):
    """
    Try multiple random perturbations of a0 and refine each with a short
    optimization burst. Keep the best result.
    """
    t0 = time.monotonic()
    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()
    rng = np.random.default_rng(2024)

    for s in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        time_per_start = remaining / (n_starts - s)

        # Create perturbation
        noise_level = 0.05 / (1.0 + 0.3 * s)
        pert = rng.normal(0, noise_level, n) * a0
        a_pert = a0 + pert
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()

        # Short mirror descent burst
        x = a_pert.copy()
        x_prev = a_pert.copy()
        max_iters = min(3000, int(time_per_start * 100))

        for t in range(1, max_iters + 1):
            if time.monotonic() - t0 > time_limit:
                break
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            s_y = y.sum()
            if s_y < 1e-15:
                y = x.copy()
            else:
                y = y / s_y

            ac = fast_autocorr(y)
            ac_max = ac.max()
            ac_shifted = ac - ac_max
            beta_val = 50.0
            exp_vals = np.exp(beta_val * ac_shifted)
            w = exp_vals / exp_vals.sum()

            fft_len2 = 1 << (3 * n - 3).bit_length()
            W_ext = np.zeros(fft_len2)
            W_ext[:len(w)] = w
            A_ext = np.zeros(fft_len2)
            A_ext[:n] = y
            Wf2 = np.fft.rfft(W_ext)
            Af2 = np.fft.rfft(A_ext)
            grad_raw = np.fft.irfft(Wf2 * np.conj(Af2), fft_len2)[:n]
            grad = 2.0 * grad_raw

            eta = 5.0 / (np.sqrt(t) + 1.0)
            log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new

            x_prev = x
            x = a_new

            M_new = fast_autocorr(x).max()
            if M_new < best_m:
                best_m = M_new
                best_a = x.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    MultiStart s=" + str(s) + " (" + repr(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining:
    1. Parametric initialization with multi-n exploration
    2. Softmax mirror descent (novel method)
    3. Improved LP-Frank-Wolfe
    4. Peak shaving
    5. Multi-start refinement
    6. Progressive scaling to larger n
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Total time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Try to use the parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    # If no parent or parent is bad, use parametric init
    if best_seq is None or best_score > 1.8:
        n_init = 5000
        a_param, val_param = parametric_init(n_init, 60.0)
        if a_param is not None:
            sc = evaluate_safe(a_param)
            if np.isfinite(sc) and sc < best_score:
                best_seq = a_param
                best_score = sc
                print("Parametric best: n=" + str(n_init) + ", score=" + repr(sc))
    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1((np.arange(n_base) + 1.0) ** 1.5)
        best_score = evaluate_safe(best_seq)
        print("Fallback init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    # Strategy: work at the parent's length first, then try larger
    n_current = len(best_seq)
    
    if n_current < 5000:
        # Scale up to at least 5000
        n_current = 5000
        best_seq = interpolate_sequence(best_seq, n_current)
        best_score = evaluate_safe(best_seq)
        print("Scaled to n=" + str(n_current) + ", score=" + repr(best_score))

    if n_current > 8000:
        # Scale down to 8000 for initial optimization (faster LPs)
        n_work = 8000
        a_work = interpolate_sequence(best_seq, n_work)
        sc = evaluate_safe(a_work)
        if np.isfinite(sc):
            print("Working at n=" + str(n_work) + ", score=" + repr(sc))
        best_seq_work = a_work
    else:
        n_work = n_current
        best_seq_work = best_seq.copy()

    # Phase 1: Softmax mirror descent (new method)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Softmax Mirror Descent (n=" + str(n_work) + ") ===")
    t1 = min(remaining * 0.15, 180.0)
    if t1 > 30:
        a_out, _ = softmax_mirror_descent(norm_sum1(best_seq_work), t1,
                                          beta_init=80.0, beta_min=10.0,
                                          c0=5.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "softmd", t_global)
        if len(best_seq) != n_work:
            n_work = len(best_seq)

    # Phase 2: LP-Frank-Wolfe (improved)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: LP-FW Improved (n=" + str(n_work) + ") ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60 and n_work <= 9000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t2, solver,
                                  initial_K=500, max_K=3000,
                                  add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw", t_global)

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Peak Shaving (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakshave", t_global)

    # Phase 4: Second round of LP-FW with different parameters
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW Round 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40 and n_best <= 9000:
        # Use a different initial active set
        ac = fast_autocorr(best_seq)
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t4, solver,
                                  initial_K=800, max_K=4000,
                                  add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # Phase 5: Try larger n values
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    larger_ns = []
    for delta in [500, 1000, 2000, 3000]:
        candidate_n = n_best + delta
        if candidate_n > n_best and candidate_n <= 12000:
            larger_ns.append(candidate_n)

    for new_n in larger_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n=== Phase 5: Scale to n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)
        sc_long = evaluate_safe(a_long)
        print("  Interpolated score: " + repr(sc_long))

        # Short mirror descent on the longer sequence
        t5a = min(remaining * 0.25, 150.0)
        if t5a > 30:
            a_out, _ = softmax_mirror_descent(a_long, t5a, beta_init=60.0,
                                               beta_min=15.0, c0=4.0, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "softmd_n" + str(new_n), t_global)

        # LP-FW on the longer sequence (if time and size allow)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100 and new_n <= 8000:
            t5b = min(remaining * 0.35, 200.0)
            if t5b > 40:
                a_out, _ = lp_fw_improved(norm_sum1(best_seq), t5b, solver,
                                          initial_K=400, max_K=2500,
                                          add_per_iter=30, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # Phase 6: Multi-start refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-Start Refinement (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 30:
        a_out, _ = multi_start_refine(norm_sum1(best_seq), n_best, t6, n_starts=5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "multistart", t_global)

    # Phase 7: Final LP-FW push
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 30 and n_best <= 9000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t7, solver,
                                  initial_K=600, max_K=3500,
                                  add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 8: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: Final Peak Shaving ===")
    t8 = min(remaining * 0.30, 150.0)
    if t8 > 15:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakshave_final", t_global)

    # Final mirror descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t9 = min(remaining * 0.40, 100.0)
        if t9 > 15:
            a_out, _ = softmax_mirror_descent(norm_sum1(best_seq), t9,
                                              beta_init=100.0, beta_min=20.0,
                                              c0=3.0, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "softmd_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) +
              ", time=" + repr(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        best_seq = norm_sum1((np.arange(n_fb) + 1.0) ** 1.5)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
