def spectral_shaping(a0, time_limit, lr_init=0.3, beta_init=3.0, verbose=True):
    """

    and time-domain non-negativity/simplex projection.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    lr = lr_init
    stall = 0
    max_stall = 500
    beta = beta_init
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Compute autocorrelation
        ac = fast_convolve(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        
        # Identify peaks in autocorrelation
        # Want to reduce high autocorrelation values
        ac_max = M_curr
        if ac_max < 1e-15:
            break
        
        # Soft-max weighting of autocorrelation peaks
        beta_t = beta + t * 0.005
        w = np.exp(beta_t * (ac / ac_max - 1.0))
        w = w / w.sum()
        
        # Gradient in frequency domain:
        # d/ac[m] -> corresponds to modifying |A|^2
        # We want to reduce high ac values, so subtract weighted contribution
        # In freq domain, reducing ac peaks means flattening |A|^2
        
        # Compute the "unwanted" spectrum component
        # The autocorrelation is the IFFT of |A|^2
        # We want ac to be flat, so we want |A|^2 to be flat
        P_target = np.ones_like(P) * P.mean()
        
        # Gradient step: move P towards P_target, weighted by how much each freq contributes to peaks
        # Simple approach: scale P to be more flat
        P_ratio = P / P_target
        # Damp high values, boost low values
        P_new = P_target * np.sign(P) * (1.0 + lr * np.log(P_ratio))
        P_new = np.maximum(P_new, 0.0)

        mag = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)

        x = x_new
        iter_count += 1
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={K}, no_impr={no_improve}")
        
        # Early termination if stuck
        if no_improve > 2000 and t > 500:
            break
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m

def coord_descent(a0, time_limit, verbose=True):
    """Adaptive batch coordinate descent with lookahead scoring."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_convolve(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        
        # Focus on top lags
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Find top contributing pair
            n_pairs = min(5, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]
            
            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-14:
                    continue
                
                # Find best destination: position that minimizes impact on top lags
                # Score: how much would adding mass at position j increase top lags
                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        # If we put mass at j, it contributes to lag k2 via a[j]*a[k2-j]
                        partner = k2i - j_idx_for_score(idx2, k2i)
                        # Simplified: score[j] += a[j_partner] * ac[k2] for valid partners
                        valid_p = (partner >= 0) & (partner < n)
                        scores[idx2[valid_p]] += a[partner[valid_p]] * ac[k2i]
                
                # Avoid positions already in high-contribution regions
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2+1] = True
                
                cand_mask = ~in_peak
                if cand_mask.sum() < 20:
                    cand_mask = np.ones(n, dtype=bool)
                cand_mask[src] = False
                
                cand_scores = scores.copy()
                cand_scores[~cand_mask] = 1e10
                dst = np.argmin(cand_scores)
                
                if a[dst] > 0.8 * src_val:
                    continue
                
                # Try transfer
                found = False
                for frac in [0.05, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < M_curr - 1e-12:
                        a = a_trial
                        M_curr = M_trial
                        if M_curr < best_m - 1e-12:
                            best_m = M_curr
                            best_a = a.copy()
                        improved = True
                        found = True
                        break
                if not found:
                    break
        
        if not improved:
            stall += 1
            if stall > max_stall:
                break
            if stall % 20 == 0:
                # Random perturbation
                for _ in range(rng.integers(5, 20)):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.01, 0.05))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDescent ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def j_idx_for_score(idx, k):
    """Helper for coordinate descent scoring."""
    return k - idx

def mirror_descent_opt(a0, time_limit, top_k=50, beta=3.0, c0=3.0, restart_every=2000, verbose=True):
    """Mirror descent with soft gradient (exponential family)."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_k = top_k
    cur_beta = beta
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_convolve(y)
        
        # Soft gradient: weighted sum of lag gradients
        k_max = min(cur_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                grad[idx] += 2.0 * wi * y[j_idx]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            pert_type = int(rng.integers(3))
            if pert_type == 0:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.5, 2.5))
                pert_seq = normalize((i_arr + 1.0) ** p)
            elif pert_type == 1:
                x_arr = np.linspace(0, 1, n)
                c = float(rng.uniform(0.2, 0.8))
                sig = float(rng.uniform(0.1, 0.4))
                pert_seq = normalize(np.exp(-((x_arr - c) ** 2) / (2 * sig * sig)))
            else:
                pert = rng.normal(0, 0.05, n) * best_a
                pert_seq = normalize(best_a + pert)
            
            mix_w = float(rng.uniform(0.1, 0.4))
            x = normalize((1 - mix_w) * best_a + mix_w * pert_seq)
            x_prev = x.copy()
            cur_k = min(cur_k + 20, 200)
            cur_beta = max(cur_beta - 0.5, 1.5)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={t} ({elapsed:.0f}s): J={J:.6f}, k={cur_k}, beta={cur_beta:.1f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def parametric_init(n, best_seq, time_limit=10.0):
    """Quick parametric search for good initialization."""
    t0 = time.monotonic()
    best_a = normalize(best_seq)
    best_m = fast_convolve(best_a).max()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    candidates = []
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    # Beta-like
    for alpha in [0.5, 1.0, 1.5]:
        for beta in [0.5, 1.0, 1.5]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
    # Gaussian
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    # Bimodal
    for c1 in [0.15, 0.25]:
        for c2 in [0.75, 0.85]:
            for sigma in [0.05, 0.1, 0.15]:
                candidates.append(np.exp(-((x - c1) ** 2) / (2 * sigma * sigma)) +
                                 np.exp(-((x - c2) ** 2) / (2 * sigma * sigma)))
    # Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [10, 20, 50]:
            candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))
    # Exponential
    for rate in [1, 2, 3, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * x))
    
    for cand in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a_test = normalize(cand)
        m_test = fast_convolve(a_test).max()
        if m_test < best_m:
            best_m = m_test
            best_a = a_test.copy()
    
    # Blend with parent
    a_parent = normalize(best_seq)
    for w in [0.2, 0.4, 0.6, 0.8]:
        a_blend = normalize(w * a_parent + (1 - w) * best_a)
        m_blend = fast_convolve(a_blend).max()
        if m_blend < best_m:
            best_m = m_blend
            best_a = a_blend.copy()
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = _get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 5000
        i_arr = np.arange(n_base, dtype=np.float64)
        best_seq = normalize((i_arr + 1.0) ** 0.7)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = compute_score(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Parametric search ===
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        a_param, _ = parametric_init(n0, best_seq, time_limit=t1)
        check_update(a_param, "parametric")
    
    # === Phase 2: Spectral shaping ===
    print(f"\n=== Phase 2: Spectral shaping (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_in = normalize(best_seq)
        a_out, _ = spectral_shaping(a_in, t2, verbose=True)
        check_update(a_out, f"spectral_n{n0}")
    
    # === Phase 3: FW main optimization ===
    print(f"\n=== Phase 3: FW cutting plane (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.30, 450.0)
    if t3 > 30 and n_best <= 7000:
        a_in = normalize(best_seq)
        a_out, _ = fw_optimize(a_in, t3, solver, K_init=400, K_max=3000, verbose=True)
        check_update(a_out, f"fw_main_n{n_best}")
    
    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror descent (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_in = normalize(best_seq)
        a_out, _ = mirror_descent_opt(a_in, t4, top_k=60, beta=3.5, c0=4.0,
                                       restart_every=3000, verbose=False)
        check_update(a_out, f"md_n{n_best}")
    
    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        a_in = normalize(best_seq)
        a_out, _ = coord_descent(a_in, t5, verbose=True)
        check_update(a_out, f"coord_n{n_best}")
    
    # === Phase 6: Spectral refine ===
    print(f"\n=== Phase 6: Spectral refine (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 10:
        a_in = normalize(best_seq)
        a_out, _ = spectral_shaping(a_in, t6, lr_init=0.1, beta_init=5.0, verbose=False)
        check_update(a_out, f"spectral2_n{n_best}")
    
    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    length_candidates = []
    for delta in [-500, -250, 250, 500, -1000, 1000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 8000:
            length_candidates.append(new_n)
    
    for rn in [3000, 4000, 5000, 5500, 6000]:
        if rn != n_best and 2000 <= rn <= 8000:
            length_candidates.append(rn)
    
    length_candidates = sorted(set(length_candidates))
    
    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        
        # Quick spectral shaping
        t7a = min(remaining * 0.35, 100.0)
        if t7a > 10:
            a_out, _ = spectral_shaping(a_long, t7a, lr_init=0.2, beta_init=4.0, verbose=False)
            check_update(a_out, f"spectral_n{new_n}")
        
        # FW if time permits
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 6000:
            t7b = min(remaining * 0.5, 150.0)
            if t7b > 20:
                a_in = normalize(best_seq) if len(best_seq) == new_n else a_long
                a_out, _ = fw_optimize(a_in, t7b, solver, K_init=300, K_max=2000, verbose=False)
                check_update(a_out, f"fw_n{new_n}")
    
    # === Phase 8: Final FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 7000:
        t8 = min(remaining * 0.40, 200.0)
        if t8 > 20:
            a_in = normalize(best_seq)
            a_out, _ = fw_optimize(a_in, t8, solver, K_init=500, K_max=3500, verbose=True)
            check_update(a_out, f"fw_final_n{n_best}")
    
    # === Phase 9: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final coord descent ===")
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.50, 100.0)
        if t9 > 10:
            a_in = normalize(best_seq)
            a_out, _ = coord_descent(a_in, t9, verbose=True)
            check_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 10: Final MD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 5:
            a_in = normalize(best_seq)
            a_out, _ = mirror_descent_opt(a_in, t10, top_k=100, beta=3.0, c0=5.0,
                                           restart_every=2000, verbose=False)
            check_update(a_out, f"md_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = np.sin(np.pi * np.linspace(0, 1, n_fb)) ** 2
        best_seq = normalize(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
