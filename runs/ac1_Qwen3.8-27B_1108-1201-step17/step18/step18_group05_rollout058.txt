def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_structured_subgradient(a, ac, top_k=80, beta=4.0):
    """Compute subgradient using weighted combination of top peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def improved_amd(a0, time_limit, top_k=80, beta=5.0, c0=5.0,

    x_prev = a.copy()
    gamma = c0
    stall_count = 0

    cur_beta = beta

    for t in range(1, 80000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.8)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_structured_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()

            best_a = x.copy()
            stall_count = 0

            stall_count += 1

            x = x_r
            x_prev = x_r.copy()

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=300, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive K and backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            no_improve += 1
            if no_improve > 5:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            if no_improve > 5:
                break
            K = min(K + 50, top_K_max)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        alphas = [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0),
                  min(1.0, best_alpha * 0.7), 0.5, 0.3, 0.15, 0.1]
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        if iter_count % 8 == 0:
            K = min(K + 40, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def structured_coord_descent(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent that exploits the subgradient structure directly."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_peaks = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-13:
                continue

            # Find destinations: minimize contribution to all top peaks
            dst_scores = np.zeros(n)
            for kp in top_k[:8]:
                kpi = int(kp)
                for j in range(max(0, kpi - n + 1), min(n, kpi + 1)):
                    partner = a[kpi - j]
                    if partner > 0:
                        dst_scores[j] += partner
            dst_scores[src] = np.inf

            n_cands = min(5, n - 1)
            cands = np.argsort(dst_scores)[:n_cands]

            found = False
            for dst in cands:
                if dst == src or found:
                    continue
                for frac in [0.02, 0.01, 0.05, 0.005, 0.03, 0.08, 0.12]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(min(stall, 10)):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    StructCD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=True, tau_init=15.0, tau_final=200.0,

    """Smoothed projected gradient with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    max_stall = 800
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit * 0.8))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        a_new = project_simplex(a - eta * g_tangent)
        m_new = fast_autocorr(a_new).max()

        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.015, 3.0)
        else:
            a = a_new
            stall += 1
            if stall > 100:
                eta *= 0.75
            if stall > 300:
                pert = rng.normal(0, 0.004, n) * a
                a_p = a + pert
                a_p = np.maximum(a_p, 0.0)
                a = a_p / a_p.sum()
                stall = 150

        total_iters += 1

        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_m


def multi_scale_optimize(a_initial, time_budget, solver, verbose=True):
    """
    Multi-scale optimization: coarse to fine with interpolation.
    Key insight: optimize on smaller problems first (faster LPs, better convergence),
    then refine at larger scales.
    """
    t_global = time.monotonic()
    global_best = a_initial.copy()
    global_best_score = evaluate_safe(global_best)
    n_initial = len(global_best)

    def elapsed():
        return time.monotonic() - t_global

    def remaining():
        return time_budget - elapsed()

    # Define scale progression
    # Start coarse, end at original size or slightly larger
    n_coarse = max(1000, n_initial // 3)
    n_medium = max(2000, n_initial // 2)
    n_final = n_initial

    scales = []
    if n_coarse < n_initial:
        scales.append(n_coarse)
    if n_medium < n_initial and n_medium != n_coarse:
        scales.append(n_medium)
    scales.append(n_final)
    # Add one slightly larger
    n_large = n_initial + 800
    if n_large <= 7000:
        scales.append(n_large)

    # Time allocation: more time for finer scales
    weights = [0.10, 0.15, 0.35, 0.20, 0.20][:len(scales)]
    w_sum = sum(weights)
    weights = [w / w_sum for w in weights]

    for scale_idx, n_scale in enumerate(scales):
        if remaining() < 60:
            break

        t_scale = min(remaining() * 0.95, weights[scale_idx] * time_budget * 1.5)
        if t_scale < 30:
            t_scale = remaining() * 0.5
            if t_scale < 20:
                break

        print(f"\n  Scale {scale_idx}: n={n_scale}, time={t_scale:.0f}s")

        # Interpolate to this scale
        a_scale = interpolate_sequence(global_best, n_scale)
        s_cur = evaluate_safe(a_scale)
        print(f"    Interpolated score: {s_cur:.6f}")

        # Phase A: AMD on this scale
        t_amd = t_scale * 0.40
        if t_amd > 15:
            a_out, _ = improved_amd(a_scale, t_amd, top_k=60, beta=5.0, c0=5.0,
                                     restart_every=4000, verbose=False,
                                     rng_seed=42 + scale_idx * 10)
            sc = evaluate_safe(a_out)
            if sc < global_best_score:
                global_best = a_out.copy()
                global_best_score = sc
                print(f"    AMD improved: {sc:.6f}")
            a_scale = global_best if len(global_best) == n_scale else interpolate_sequence(global_best, n_scale)

        # Phase B: LP-FW on this scale
        remaining_scale = t_scale - (elapsed() - (t_scale - t_scale))  # approximate
        t_lp = t_scale * 0.35
        if t_lp > 20 and n_scale <= 5000:
            a_in = interpolate_sequence(global_best, n_scale)
            top_K_init = min(400, 2 * n_scale - 1)
            top_K_max = min(3000, 2 * n_scale - 1)
            a_out, _ = lp_frank_wolfe(a_in, t_lp, solver, verbose=False,
                                       top_K_init=top_K_init, top_K_max=top_K_max)
            sc = evaluate_safe(a_out)
            if sc < global_best_score:
                global_best = a_out.copy()
                global_best_score = sc
                print(f"    LP-FW improved: {sc:.6f}")

        # Phase C: Smoothed gradient
        t_sg = t_scale * 0.15
        if t_sg > 15:
            a_in = interpolate_sequence(global_best, n_scale)
            a_out, _ = smooth_gradient_optimize(a_in, t_sg, verbose=False,
                                                 tau_init=10.0, tau_final=150.0,
                                                 eta_init=0.5, rng_seed=200 + scale_idx)
            sc = evaluate_safe(a_out)
            if sc < global_best_score:
                global_best = a_out.copy()
                global_best_score = sc
                print(f"    SmoothGrad improved: {sc:.6f}")

        # Phase D: Structured coordinate descent
        t_cd = max(0, t_scale * 0.10)
        if t_cd > 10:
            a_in = interpolate_sequence(global_best, n_scale)
            a_out, _ = structured_coord_descent(a_in, t_cd, verbose=False,
                                                 rng_seed=500 + scale_idx)
            sc = evaluate_safe(a_out)
            if sc < global_best_score:
                global_best = a_out.copy()
                global_best_score = sc
                print(f"    StructCD improved: {sc:.6f}")

        print(f"  End scale {scale_idx}: best={global_best_score:.6f} (n={len(global_best)})")

    return global_best, global_best_score


def peak_targeting_refinement(a0, time_limit, verbose=True, rng_seed=777):
    """
    Direct peak-targeting refinement: repeatedly find the worst peak and
    make surgical moves to reduce it using the subgradient structure.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        # Find all peaks above threshold
        threshold = best_m * 0.95
        peak_positions = np.where(ac >= threshold)[0]

        if len(peak_positions) == 0:
            peak_positions = np.argsort(ac)[::-1][:10]

        improved = False
        for k in peak_positions[:12]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]

            if contrib.max() < 1e-25:
                continue

            # Find the worst contributor
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-13:
                continue

            # Score destinations by their contribution to ALL peaks
            dst_penalty = np.zeros(n)
            for kp in np.argsort(ac)[::-1][:20]:
                kpi = int(kp)
                s2 = max(0, kpi - n + 1)
                e2 = min(n - 1, kpi)
                for j in range(s2, e2 + 1):
                    dst_penalty[j] += a[kpi - j]
            dst_penalty[src] = np.inf * 1e30

            # Try top candidates
            n_cands = min(8, n // 10)
            cands = np.argsort(dst_penalty)[:n_cands]

            found = False
            for dst in cands:
                if found:
                    break
                for frac in [0.015, 0.008, 0.03, 0.005, 0.02, 0.05, 0.04, 0.07]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-14:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

        if not improved:
            stall += 1
            # Larger random move to escape
            n_moves = min(stall // 10 + 1, 5)
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        stall = 0
                        break

        if verbose and stall % 100 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakTarget ({el:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator using multi-scale optimization,
    structured subgradient methods, and LP-based Frank-Wolfe.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3000
        x = np.linspace(0, 1, n_base)
        best_seq = make_u_shape(n_base, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Multi-scale optimization ===
    print(f"\n=== Phase 1: Multi-scale optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.50, 800.0)
    if t1 > 100:
        a_out, sc_out = multi_scale_optimize(norm_sum1(best_seq), t1, solver, verbose=True)
        try_update(a_out, "multiscale")

    # === Phase 2: Dedicated LP-FW on best (most powerful method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Dedicated LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 50 and n_best <= 5500:
        a_in = norm_sum1(best_seq)
        # Use larger K for more constraints
        top_K_init = min(600, 2 * n_best - 1)
        top_K_max = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                   top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_dedicated")

    # === Phase 3: Peak targeting refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak targeting (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_targeting_refinement(a_in, t3, verbose=True, rng_seed=777)
        try_update(a_out, "peaktarget")

    # === Phase 4: AMD with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: AMD variant (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = improved_amd(a_in, t4, top_k=120, beta=7.0, c0=6.0,
                                 restart_every=6000, verbose=True, rng_seed=999)
        try_update(a_out, "amd_variant")

    # === Phase 5: Smoothed gradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Smooth gradient (n={n_best}) ===")
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t5, verbose=False,
                                             tau_init=30.0, tau_final=300.0,
                                             eta_init=0.3, rng_seed=500)
        try_update(a_out, "smooth_final")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    trial_lengths = []
    for delta in [-400, 400, -800, 800, 1200, -1200, 1600]:
        new_n = n_best + delta
        if 1500 <= new_n <= 7500:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]
    trial_lengths = trial_lengths[:3]  # Limit to 3

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_interp = interpolate_sequence(best_seq, new_n)

        # AMD on new length
        t6a = min(remaining * 0.4, 150.0)
        if t6a > 20:
            a_out, _ = improved_amd(a_interp, t6a, top_k=80, beta=5.0, c0=5.0,
                                     restart_every=5000, verbose=False,
                                     rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        # LP on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5500:
            t6b = min(remaining * 0.5, 120.0)
            if t6b > 15:
                a_in = interpolate_sequence(best_seq, new_n)
                top_K = min(400, 2 * new_n - 1)
                top_K_max = min(3000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, verbose=False,
                                           top_K_init=top_K, top_K_max=top_K_max)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
        t7 = min(remaining * 0.6, 80.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = structured_coord_descent(a_in, t7, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # === Phase 8: One more LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 5500:
        print(f"\n=== Phase 8: Final LP (n={n_best}, t={remaining:.0f}s) ===")
        t8 = min(remaining * 0.7, 100.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            top_K_init = min(800, 2 * n_best - 1)
            top_K_max = min(6000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=False,
                                       top_K_init=top_K_init, top_K_max=top_K_max)
            try_update(a_out, "lp_ultimate")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
