```python
def autoconv_grad(a, k):
    """Compute gradient of (a*a)[k] with respect to a."""
    n = len(a)
    L = max(0, k - n + 1)
    U = min(n - 1, k)
    g = np.zeros(n)
    for i in range(L, U+1):
        g[i] += a[k - i]       # derivative w.r.t. a[i] from term a[i]*a[k-i]
        g[k - i] += a[i]       # derivative w.r.t. a[k-i] from term a[i]*a[k-i]
    return g
```

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def autoconv_subgrad(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a."""
    L = max(0, k - n + 1)
    U = min(n - 1, k)
    idx = np.arange(L, U + 1)
    g = np.zeros(n)
    g[idx] += a[k - idx]
    k_minus_idx = k - idx
    g[k_minus_idx] += a[idx]
    return g


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = a ** 2
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def peak_chasing_mirror_descent(a0, time_limit, top_k=30, eta_init=0.1,
                                  c_schedule=1.0, restart_every=3000,
                                  rng_seed=42, verbose=False, min_eta=0.001):
    """
    Mirror descent (softmax) with exact subgradients of the peak autoconv.
    Adaptive restarts with perturbation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = eta_init
    stall = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top K active indices
        K = min(top_k, 2 * n - 1)
        # Sort in descending order of autocorrelation values
        # For efficiency with large arrays, use argpartition then sort the top
        if 2 * n - 1 > K * 3:
            part_idx = np.argpartition(ac, -K)[-K:]
            active_idx = part_idx[np.argsort(ac[part_idx])[::-1]]
        else:
            active_idx = np.argsort(ac)[::-1][:K]
        
        active_vals = ac[active_idx]
        
        # Softmax weights (sharp to focus on the peaks)
        if len(active_vals) > 1:
            shifted = active_vals - active_vals.max()
            w = np.exp(shifted * 5.0)  # temperature 0.2
            w = w / w.sum()
        else:
            w = np.ones(1)
        
        # Compute subgradient
        g = np.zeros(n)
        for wi, ki in zip(w, active_idx):
            ki_int = int(ki)
            g += wi * autoconv_subgrad(a, ki_int, n)
        
        # Scale gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Mirror descent step
        # a_new = exp(log(a) - eta * g) / Z
        # Use log-space for numerical stability
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        a = a_new
        iter_count += 1
        
        # Adaptive step size
        if stall > 200:
            eta = max(eta * 0.9, min_eta)
        elif stall == 0:
            eta = min(eta * 1.02, 1.0)
        
        # Periodic restart
        if stall > restart_every:
            # Perturb from best
            noise_scale = 0.02 * (1 + 0.1 * (stall // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0
            eta = eta_init
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    PC-MD restart at t={iter_count} ({elapsed:.0f}s), M={best_M:.8f}")
        
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    PC-MD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_M


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Backtracking line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()
        
        iter_count += 1
        
        if iter_count % 10 == 0:
            K = min(K + 30, top_K_max)
        
        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")
    
    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123, top_k_feats=10):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed (using the quadratic structure of autoconv).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find which lags are "active" (near maximum)
        threshold = M_curr * 0.8
        active_lags = np.where(ac >= threshold)[0]
        # Limit to top few for efficiency
        if len(active_lags) > top_k_feats * 5:
            top_part = np.argpartition(ac, -top_k_feats * 5)[-top_k_feats * 5:]
            active_lags = top_part[np.argsort(ac[top_part])[::-1]]
        
        improved = False
        
        # Try moving mass from high-contribution coordinates to low-contribution ones
        # Compute "sensitivity": how much each coordinate contributes to the peaks
        sens = np.zeros(n)
        for k in active_lags[:top_k_feats]:
            ki = int(k)
            L = max(0, ki - n + 1)
            U = min(n - 1, ki)
            idx = np.arange(L, U + 1)
            sens[idx] += a[ki - idx]
        
        # Sort coordinates by sensitivity (high = important for peaks)
        order = np.argsort(sens)
        
        # Try to move mass from high-sensitivity to low-sensitivity positions
        n_moves = min(100, n // 5)
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a high-sensitivity source
            src_candidates = np.where(sens > sens.mean())[0]
            if len(src_candidates) == 0:
                src_candidates = np.arange(n)
            src = src_candidates[rng.integers(len(src_candidates))]
            
            # Pick a low-sensitivity destination
            dst_candidates = np.where(sens < sens.mean())[0]
            if len(dst_candidates) == 0:
                dst_candidates = np.arange(n)
            dst = dst_candidates[rng.integers(len(dst_candidates))]
            
            if src == dst:
                continue
            
            # Try several fractions
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Add random perturbation to escape
            pert = rng.normal(0, 0.003, n) * a
            a = a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def frequency_domain_init(n, n_peaks=5, seed=0):
    """
    Construct a sequence by designing its DFT to have controlled peaks.
    We want |A(w)|^2 to be as flat as possible.
    """
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Start with a flat magnitude response with small random perturbations
    A = np.ones(fft_size // 2 + 1) * (1.0 / np.sqrt(n))
    for i in range(1, fft_size // 2):
        A[i] += rng.normal(0, 0.05)
    
    # Convert to real sequence
    a = np.fft.irfft(A, fft_size)[:2*n-1]
    # Fold into length n by averaging with reversed
    a_fold = (a[:n] + a[n-1::-1]) / 2 if n < len(a) else a[:n]
    a_fold = np.maximum(a_fold, 0)
    s = a_fold.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_fold / s

_global_solver = None

def get_solver():

    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:

            prob.solve(solver=s, verbose=False)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def autoconv_subgrad(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a (linear convolution)."""
    L = max(0, k - n + 1)
    U = min(n - 1, k)
    if L > U:
        return np.zeros(n)
    idx = np.arange(L, U + 1)
    g = np.zeros(n)
    g[idx] += a[k - idx]
    km_idx = k - idx
    g[km_idx] += a[idx]
    return g


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)

def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def frequency_domain_init(n, seed=0):
    """Construct sequence with flatter DFT magnitude."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.ones(fft_size // 2 + 1) * (1.0 / np.sqrt(n))
    for i in range(1, fft_size // 2):
        A[i] += rng.normal(0, 0.08)
    a = np.fft.irfft(A, fft_size)[:2 * n - 1]
    if n <= len(a):
        a_fold = (a[:n] + a[n - 1::-1]) / 2.0
    else:
        a_fold = a[:n]
    a_fold = np.maximum(a_fold, 0)
    s = a_fold.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_fold / s


def peak_chasing_mirror_descent(a0, time_limit, top_k=30, eta_init=0.05,
                                 restart_every=4000, rng_seed=42, verbose=False,
                                 min_eta=0.001, sharpness=5.0):
    """
    Mirror descent (softmax) with exact subgradients of peak autoconv.
    Adaptive restarts with perturbation and schedule.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = eta_init
    stall = 0
    iter_count = 0
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top K active indices
        K = min(cur_top_k, 2 * n - 1)
        if 2 * n - 1 > K * 3:
            part_idx = np.argpartition(ac, -K)[-K:]
            active_idx = part_idx[np.argsort(ac[part_idx])[::-1]]
        else:
            active_idx = np.argsort(ac)[::-1][:K]

        active_vals = ac[active_idx]

        # Softmax weights with adaptive sharpness
        if len(active_vals) > 1:
            shifted = active_vals - active_vals.max()
            w = np.exp(shifted * sharpness)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(K) / K
            else:
                w = w / w_sum
        else:
            w = np.ones(1)

        # Compute subgradient
        g = np.zeros(n)
        for wi, ki in zip(w, active_idx):
            ki_int = int(ki)
            g += wi * autoconv_subgrad(a, ki_int, n)

        # Scale gradient for numerical stability
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Mirror descent step in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        a = a_new
        iter_count += 1

        # Adaptive step size
        if stall > 500:
            eta = max(eta * 0.95, min_eta)
        elif stall > 200:
            eta = max(eta * 0.98, min_eta)
        elif stall == 0:
            eta = min(eta * 1.01, 0.5)

        # Periodic restart
        if stall > restart_every:
            restart_level = stall // restart_every
            noise_scale = 0.015 * (1.0 + 0.2 * restart_level)
            pert = rng.normal(0, noise_scale, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0
            eta = eta_init * (0.8 + 0.1 * rng.random())

            # Occasionally increase top_k to consider more peaks
            if restart_level % 3 == 0:
                cur_top_k = min(cur_top_k + 5, 80)

            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_M
                print(f"    PC-MD restart iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, K={cur_top_k}")

        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    PC-MD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_M


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_init=200, top_K_max=None):
    """LP-based Frank-Wolfe with adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if top_K_max is None:
        top_K_max = 2 * n - 1
    K = top_K_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_m = base_obj
        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        iter_count += 1

        if iter_count % 10 == 0:
            K = min(K + 30, top_K_max)

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123, n_top=10):
    """
    Sensitivity-guided coordinate descent: move mass from high-peak-contribution
    coordinates to low-contribution ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find active peaks
        threshold = M_curr * 0.7
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > n_top * 3:
            part = np.argpartition(ac, -n_top * 3)[-n_top * 3:]
            active_lags = part[np.argsort(ac[part])[::-1]]

        # Compute sensitivity map
        sens = np.zeros(n)
        for k in active_lags[:n_top]:
            ki = int(k)
            L = max(0, ki - n + 1)
            U = min(n - 1, ki)
            idx = np.arange(L, U + 1)
            sens[idx] += a[ki - idx]

        improved = False

        # Try moving mass from high-sens to low-sens
        n_trials = min(200, n // 3)
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            src_cands = np.where(sens > sens.mean() * 1.5)[0]
            if len(src_cands) == 0:
                src_cands = np.where(sens > sens.max() * 0.1)[0]
            if len(src_cands) == 0:
                break
            src = src_cands[rng.integers(len(src_cands))]

            dst_cands = np.where(sens < sens.mean() * 0.5)[0]
            if len(dst_cands) == 0:
                dst_cands = np.where(sens < sens.max() * 0.3)[0]
            if len(dst_cands) == 0:
                break
            dst = dst_cands[rng.integers(len(dst_cands))]

            if src == dst or a[src] < 1e-12:
                continue

            for frac in [0.003, 0.007, 0.015, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            pert = rng.normal(0, 0.002, n) * a
            a = a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def alternating_lp_mirror(a0, time_limit, solver, verbose=True,
                           mirror_time_frac=0.6, top_K=300):
    """
    Alternate between mirror descent (for speed) and LP-FW (for structured improvement).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed

        # Mirror descent phase
        t_mirror = min(remaining * mirror_time_frac, 60.0)
        if t_mirror < 5:
            break
        a_out, m_out = peak_chasing_mirror_descent(
            a, t_mirror, top_k=30, eta_init=0.03,
            restart_every=3000, rng_seed=int(elapsed * 7) % 10000, verbose=False)
        a = a_out
        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        # LP phase
        remaining = time_limit - (time.monotonic() - t0)
        t_lp = min(remaining * (1.0 - mirror_time_frac), 30.0)
        if t_lp < 5:
            break
        a_out, m_out = lp_frank_wolfe(
            a, t_lp, solver, verbose=False,
            top_K_init=min(top_K, 2 * n - 1),
            top_K_max=min(2000, 2 * n - 1))
        a = a_out
        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AltLP-MD ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Intensive peak-chasing mirror descent ===
    print(f"\n=== Phase 1: Peak-chasing MD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_chasing_mirror_descent(
            a_in, t1, top_k=30, eta_init=0.05,
            restart_every=4000, rng_seed=42, verbose=True)
        try_update(a_out, "pcmd_main")

    # === Phase 2: Alternating LP + Mirror ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Alt LP+MD (n={n_best}) ===")
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 60 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = alternating_lp_mirror(
            a_in, t2, solver, verbose=True,
            mirror_time_frac=0.55, top_K=min(500, 2 * n_best - 1))
        try_update(a_out, "alt_lp_md")

    # === Phase 3: Multi-start with diverse inits ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Multi-start (n={n0}) ===")
    t3_total = min(remaining * 0.20, 300.0)
    if t3_total > 60:
        inits = [
            (make_u_shape(n0, 1.0), "u1.0"),
            (make_u_shape(n0, 3.0), "u3.0"),
            (make_u_shape(n0, 5.0), "u5.0"),
            (make_bimodal(n0, 0.08), "bimodal_tight"),
            (make_bimodal(n0, 0.25), "bimodal_wide"),
            (make_cosine(n0), "cosine"),
            (spectral_init(n0, 40, seed=7), "spectral"),
            (frequency_domain_init(n0, seed=3), "freq_init"),
            (frequency_domain_init(n0, seed=99), "freq_init2"),
        ]
        if parent is not None:
            p_norm = norm_sum1(parent)
            inits.append((norm_sum1(0.6 * p_norm + 0.4 * make_u_shape(n0, 1.5)), "blend_u"))
            inits.append((norm_sum1(0.6 * p_norm + 0.4 * make_bimodal(n0, 0.1)), "blend_b"))
            inits.append((norm_sum1(0.5 * p_norm + 0.5 * make_cosine(n0)), "blend_cos"))

        t_per = max(12.0, t3_total / len(inits))
        for idx, (init, lbl) in enumerate(inits):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 80:
                break
            t_avail = min(t_per, time_budget - 80 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  MS '{lbl}' (t={(t_avail):.0f}s)...")
            a_out, _ = peak_chasing_mirror_descent(
                init, t_avail, top_k=25, eta_init=0.06,
                restart_every=2000, rng_seed=200 + idx, verbose=False)
            try_update(a_out, f"ms_{lbl}")

    # === Phase 4: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length exploration (base n={n_best}) ===")

    trial_lengths = []
    for delta in [200, 400, -200, 600, -400, 1000, -600]:
        new_n = n_best + delta
        if 1500 <= new_n <= 9000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t4a = min(remaining * 0.25, 180.0)
        if t4a > 20:
            a_out, _ = peak_chasing_mirror_descent(
                a_long, t4a, top_k=35, eta_init=0.04,
                restart_every=3000, rng_seed=new_n % 10000, verbose=False)
            try_update(a_out, f"pcmd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5000:
            t4b = min(remaining * 0.25, 100.0)
            if t4b > 15:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out, _ = lp_frank_wolfe(
                    a_in, t4b, solver, verbose=False,
                    top_K_init=min(300, 2 * new_n - 1),
                    top_K_max=min(2000, 2 * new_n - 1))
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 5: Final intensive optimization on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final optimization (n={n_best}) ===")
    if remaining > 40:
        t5 = min(remaining * 0.35, 200.0)
        if t5 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_chasing_mirror_descent(
                a_in, t5, top_k=50, eta_init=0.02,
                restart_every=6000, rng_seed=9999, verbose=True,
                sharpness=8.0)
            try_update(a_out, "final_pcmd")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 6000:
            t5b = min(remaining * 0.35, 150.0)
            if t5b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(
                    a_in, t5b, solver, verbose=False,
                    top_K_init=min(600, 2 * n_best - 1),
                    top_K_max=min(5000, 2 * n_best - 1))
                try_update(a_out, "final_lp")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 6: Coordinate polish (n={n_best}) ===")
        t6 = min(remaining * 0.5, 100.0)
        if t6 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_polish(
                a_in, t6, verbose=False, rng_seed=789, n_top=12)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
