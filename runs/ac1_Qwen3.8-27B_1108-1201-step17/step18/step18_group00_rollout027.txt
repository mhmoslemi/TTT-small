def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v, target_sum=1.0):
    """Project v onto {x : x>=0, sum(x)=target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.maximum(v, 0.0) / max(v.sum(), 1e-15)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_objective(a, n):
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw_optimize(a0, time_limit, solver, initial_K=500, max_K=3000,
                   add_per_iter=40, verbose=True):
    """LP-based Frank-Wolfe with adaptive constraint management and backtracking line search."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac_y = fast_autocorr(y)
    init_frac = 0.95
    active_set = set(int(k) for k in np.where(ac_y >= best_m * init_frac)[0])
    if len(active_set) < initial_K:
        active_set = set(int(k) for k in np.argsort(ac_y, kind='stable')[::-1][:initial_K])

    lp_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    no_improve = 0
    max_no_improve = 500
    iter_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, constraint_list, n)

        constraints = [cp.sum(lp_var) == 1, coeffs @ lp_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(constraint_list[i]) for i in range(min(initial_K, len(constraint_list))))
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(lp_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt, kind='stable')[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Backtracking line search
        if t == 1:
            y = g_dir.copy()
        else:
            best_alpha = 2.0 / (t + 2.0)
            best_trial_M = fast_autocorr(y).max()

            # Try aggressive alpha first
            alphas = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, best_alpha, 0.05, 0.02, 0.01]
            alphas = [a for a in alphas if 0 < a <= 1.0]
            for alpha in alphas:
                trial = y + alpha * (g_dir - y)
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                M_trial = fast_autocorr(trial).max()
                if M_trial < best_trial_M - 1e-14:
                    best_alpha = alpha
                    best_trial_M = M_trial

            y = y + best_alpha * (g_dir - y)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        iter_count += 1
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", iters=" + str(iter_count))
    return best_a, best_m


def nesterov_subgradient(a0, time_limit, top_k=80, beta_start=2.0, beta_end=80.0,
                         c0=5.0, verbose=True):
    """Nesterov-accelerated smoothed subgradient with adaptive temperature."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()

    total_iters = 50000
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature
        frac = min(t / total_iters, 1.0)
        beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov momentum
        beta_t = (1.0 - np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Softmax weights over top lags
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac, kind='stable')[::-1][:k_max]
        top_vals = ac[top_idx]
        shifted = top_vals - M_curr
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        # Compute gradient: 2 * sum_k w_k * (Q_k * y)
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-10:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * y[k_int - idx]

        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)

        # Update
        x_new = y - eta * grad
        x_new = proj_simplex(x_new, 1.0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-13:
            best_m = M_new
            best_a = x.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    NSD iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", beta=" + str(round(beta, 1)))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    NSD done (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
    return best_a, best_m


def hard_subgradient_polish(a0, time_limit, top_k=50, c0=3.0, restart_every=5000, verbose=True):
    """Subgradient method using hard (non-smoothed) subgradient with restarts."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (1.0 - np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Hard subgradient: uniform weights over lags at (or near) the peak
        thresh = M_curr * 0.99
        active = np.where(ac >= thresh)[0]
        k_max = min(len(active), top_k)
        if k_max > 0:
            active_sub = active[np.argsort(ac[active], kind='stable')[::-1][:k_max]]
            w = np.ones(len(active_sub)) / len(active_sub)
        else:
            active_sub = np.array([int(np.argmax(ac))])
            w = np.array([1.0])

        grad = np.zeros(n)
        for wi, k in zip(w, active_sub):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * y[k_int - idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        x_new = y - eta * grad
        x_new = proj_simplex(x_new, 1.0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-13:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r > 1e-15:
                x_r = x_r / s_r
            else:
                x_r = best_a.copy()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    HSD iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print("    HSD done (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=True):
    """Targeted coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(40, len(ac))
        top_k = np.argsort(ac, kind='stable')[::-1][:n_top]

        improved = False

        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score all positions by their contribution to top lags
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            # Find lowest-scoring position as destination
            candidates = np.argsort(scores)[:max(50, n // 10)]
            dst = int(candidates[0])
            if dst == src:
                dst = int(candidates[1]) if len(candidates) > 1 else src

            if dst == src:
                continue

            found = False
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 50:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 30 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + ", stall=" + str(stall))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print("Total time budget: " + repr(round(time_budget, 1)) + "s, solver=" + str(solver))

    best_seq = None
    best_score = float('inf')

    # Try to use parent as initialization
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric search
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Parametric search (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 5:
        a_param, _ = parametric_search(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "parametric", t_global)

    # Phase 2: LP-FW main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-FW (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.28, 400.0)
    if t2 > 30:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t2, solver,
                                  initial_K=500, max_K=3000,
                                  add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(n_best), t_global)

    # Phase 3: Nesterov subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov subgradient (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 20:
        a_out, _ = nesterov_subgradient(norm_sum1(best_seq), t3, top_k=80,
                                        beta_start=2.0, beta_end=60.0, c0=5.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nsd_n" + str(n_best), t_global)

    # Phase 4: Hard subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Hard subgradient (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_out, _ = hard_subgradient_polish(norm_sum1(best_seq), t4, top_k=50,
                                           c0=3.0, restart_every=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "hsd_n" + str(n_best), t_global)

    # Phase 5: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Coordinate polish (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 10:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t5, rng_seed=77, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(n_best), t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length exploration ===")

    # Try a few different lengths
    candidate_lengths = []
    for delta in [300, 600, -300, -600, 1000]:
        nl = n_best + delta
        if 2000 <= nl <= 9000:
            candidate_lengths.append(nl)

    # Remove duplicates and sort by proximity to current
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))

    for new_n in candidate_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n  --- Trying n=" + str(new_n) + " ---")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick LP-FW
        t6a = min(remaining * 0.25, 180.0)
        if t6a > 20 and new_n <= 7000:
            a_out, _ = lp_fw_optimize(a_long, t6a, solver,
                                      initial_K=400, max_K=2500,
                                      add_per_iter=30, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

        # Subgradient polish
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.15, 100.0)
        if t6b > 15:
            a_out, _ = nesterov_subgradient(norm_sum1(best_seq), t6b, top_k=60,
                                            beta_start=3.0, beta_end=50.0, c0=4.0, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nsd_n" + str(new_n), t_global)

    # Phase 7: Final LP-FW at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP-FW (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 7000:
        t7 = min(remaining * 0.35, 250.0)
        if t7 > 30:
            a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t7, solver,
                                      initial_K=600, max_K=3500,
                                      add_per_iter=50, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final_n" + str(n_best), t_global)

    # Phase 8: Final subgradient + coord
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t8a = min(remaining * 0.5, 150.0)
        if t8a > 15:
            a_out, _ = nesterov_subgradient(norm_sum1(best_seq), t8a, top_k=80,
                                            beta_start=5.0, beta_end=100.0, c0=4.0, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nsd_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8b = min(remaining * 0.6, 100.0)
        if t8b > 10:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t8b, rng_seed=314, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
