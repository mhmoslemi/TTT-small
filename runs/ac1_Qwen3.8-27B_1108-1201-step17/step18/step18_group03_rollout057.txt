    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def pnorm_value(c, p):
    if p >= 1e10:
        return c.max()
    cp = np.maximum(c, 1e-30) ** p
    return float(np.sum(cp) ** (1.0 / p))


def pnorm_gradient_fft(a, c, p, fft_size):
    n = len(a)
    Fp = pnorm_value(c, p)
    if Fp < 1e-30:
        return np.zeros(n), Fp
    if p >= 1e10:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        c_clipped = np.maximum(c, 1e-30)
        w = p * c_clipped ** (p - 1) / Fp ** (p - 1)
    w_full = np.zeros(fft_size)
    w_full[:len(w)] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    W = np.fft.rfft(w_full)
    A = np.fft.rfft(a_full)
    grad = 2.0 * np.fft.irfft(W * np.conj(A), fft_size)[:n]
    return grad, Fp


def pn_homotopy(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    p_schedule = [2, 2, 4, 4, 8, 8, 16, 16, 32, 32, 64, 64, 128, 128, 256, 256, 512, 512, 1024, 1024]
    iters_per_p = 400
    total_iters = 0
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        lr_base = 8.0 if p <= 4 else 4.0 if p <= 16 else 2.0
        for t in range(1, iters_per_p + 1):
            if time.monotonic() - t0 > time_limit:
                break
            beta = (t - 1.0) / (t + 1.5)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
            c = fast_autocorr(y)
            grad, Fp = pnorm_gradient_fft(y, c, p, fft_size)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-20:
                break
            lr = lr_base / (1.0 + 0.05 * np.sqrt(t))
            log_step = -lr * grad / max(grad_norm, 1e-15)
            log_step = log_step - log_step.max()
            a_new = y * np.exp(log_step)
            a_new = np.maximum(a_new, 1e-15)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / sn
            x_prev = x.copy()
            x = a_new
            M_new = fast_autocorr(x).max()
            if M_new < best_m - 1e-14:
                best_m = M_new
                best_a = x.copy()
            total_iters += 1
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    p-norm homotopy ({elapsed:.0f}s): J={J:.6f}, iters={total_iters}")
    return best_a, best_m


def spectral_shaping(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    lr = 0.3
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_size)
        c = fast_autocorr(a)
        c_peak = c.max()
        c_mean = c.mean()
        excess = np.maximum(c - c_mean, 0)
        excess_norm = excess / max(c_peak, 1e-15)
        E = np.fft.rfft(np.concatenate([excess, np.zeros(fft_size - len(excess))]), fft_size)
        phase = np.angle(A)
        amp = np.abs(A)
        amp_target = amp.copy()
        amp_reduction = np.abs(E) / max(np.max(np.abs(E)), 1e-15)
        amp_target = amp * (1.0 - lr * 0.1 * amp_reduction)
        amp_target = np.maximum(amp_target, 1e-10)
        A_new = amp_target * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.02, 1.0)
        else:
            stall += 1
            if stall > 500:
                a = norm_sum1(best_a + rng.normal(0, 0.005, n) * best_a)
                stall = 0
                lr = max(lr * 0.8, 0.05)
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpectralShape ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    SpectralShape done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=2500, add_per_iter=30, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    iter_count = 0
    failed_solves = 0
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        K = len(constraint_list)
        if K == 0:
            break
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 8:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        y = y_new
        x = x_new
        iter_count += 1
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_descent(a0, time_limit, rng_seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    max_stall = 400
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
        improved = False
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        peak_k = int(top_k[0])
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js <= je:
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[peak_k - idx]
            n_moves = min(5, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_moves]
            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * ac[k2i] * 0.5
                        scores[k2i - idx2] += a[idx2] * ac[k2i] * 0.5
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True
                candidate_mask = ~in_peak
                if candidate_mask.sum() < 20:
                    candidate_mask = np.ones(n, dtype=bool)
                candidate_mask[src] = False
                cand_scores = scores.copy()
                cand_scores[~candidate_mask] = np.inf
                if np.all(np.isinf(cand_scores)):
                    cand_scores = scores
                    cand_scores[src] = np.inf
                dst = int(np.argmin(cand_scores))
                if dst == src:
                    continue
                for frac in [0.02, 0.04, 0.06, 0.1, 0.15]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
        if not improved:
            stall += 1
            if stall > max_stall:
                break
            if stall % 20 == 0:
                n_moves = int(rng.integers(2, 10))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.003, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
        if verbose and stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    return best_a, best_m


def parametric_init(n, best_init, best_score):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = norm_sum1(best_init)
    best_val = 2.0 * n * fast_autocorr(best_a).max()
    candidates = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            for sigma in [0.08, 0.12, 0.18]:
                candidates.append(np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2)))
    for rate in [1, 2, 3, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * x))
    for freq in [1, 2]:
        for ph in [0.0, 0.25, 0.5]:
            candidates.append(np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01))
    for cand in candidates:
        cand_n = norm_sum1(cand)
        ac = fast_autocorr(cand_n)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = cand_n.copy()
    return best_a, best_val
