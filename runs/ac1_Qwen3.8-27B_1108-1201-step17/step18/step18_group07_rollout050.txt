"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gradient descent with annealed log-sum-exp smoothing (exact O(n log n) gradient via FFT)
2. Adam optimizer with non-negativity projection
3. Periodic spectral flattening (frequency-domain amplitude compression)
4. Multi-start with diverse initializations
5. Local peak-reduction search
6. Length exploration
"""

import numpy as np
import time

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    """Normalize to probability simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_lse_gradient(a, c, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*c)) wrt a.
    
    Returns gradient of the smoothed objective (before the 1/beta factor).
    Actually returns d/d a_i of [sum_k exp(beta*c_k) / sum_m exp(beta*c_m)] * beta * a_{k-i}
    which is the gradient of log(sum exp(beta*c_k)) wrt a_i.
    """
    n = len(a)
    # Numerically stable log-sum-exp
    max_c = c.max()
    exp_bc = np.exp(beta * (c - max_c))
    w = exp_bc / exp_bc.sum()
    
    # Gradient: grad_i = beta * sum_j w[j+i] * a[j] for valid j
    # Computed via cross-correlation using FFT
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)
    grad = beta * cross[:n]
    return grad

def lse_objective_value(a, beta):
    """Compute the smoothed objective value."""
    c = fast_conv(a)
    max_c = c.max()
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * (c - max_c)))) + max_c
    # This equals (1/beta) * log(sum exp(beta*c_k)) (shifted for stability)

def adam_lse_optimize(a0, time_limit, beta_start=1.0, beta_end=200.0, 
                      lr=0.01, verbose=False):
    """Adam optimizer with annealed log-sum-exp smoothing."""
    n = len(a0)
    a = norm_simplex(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Adam state
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    
    # Beta annealing
    n_anneal = max(1, int(time_limit * 2))  # anneal over this many iterations
    
    while True:
        if time.monotonic() - t0 > time_limit:
            break
        t += 1
        
        # Anneal beta
        progress = min(1.0, t / n_anneal)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient
        c = fast_conv(a)
        grad = compute_lse_gradient(a, c, beta)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam gradient descent with annealed log-sum-exp smoothing (exact O(n log n) FFT gradient)
2. Natural gradient (Fisher metric) on the probability simplex

4. Multi-start with diverse initializations (bowl, cosine, parent, random)
5. Nesterov-accelerated gradient steps
6. Local peak-targeted mass redistribution
7. Length exploration around parent length

import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_lse_gradient(a, c, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*c_k)) w.r.t. a.
    
    grad_i = sum_j w[j+i] * a[j] where w_k = exp(beta*c_k)/sum(exp(beta*c))
    Computed via FFT-based cross-correlation in O(n log n).
    """
    n = len(a)
    max_c = c.max()
    exp_bc = np.exp(beta * (c - max_c))
    w = exp_bc / exp_bc.sum()
    
    # Cross-correlation: cross[i] = sum_j a[j] * w[j+i] for i=0,...,n-1
    # Need FFT size >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)
    return cross[:n]


def compute_topk_gradient(a, c, top_k, beta=50.0):
    """Compute gradient using only top-k peaks with softmax weighting."""
    n = len(a)
    k_eff = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_eff]
    top_vals = c[top_idx]
    
    max_v = top_vals[0]
    if max_v < 1e-30:
        w_top = np.ones(k_eff) / k_eff
    else:
        exponents = beta * (top_vals - max_v) / max_v
        exponents = np.maximum(exponents, -50.0)
        w_top = np.exp(exponents)
        s = w_top.sum()
        if s < 1e-30:
            w_top = np.ones(k_eff) / k_eff
        else:
            w_top = w_top / s
    
    # Build sparse weight vector
    w = np.zeros(len(c))
    w[top_idx] = w_top
    
    # Cross-correlation
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)
    return cross[:n]

            lr = max(0.001, lr * 0.8)
    
    return best_a, best_m


def spectral_flatten(a, alpha=0.3, iterations=50, time_limit=30.0):
    """Iterative spectral flattening with non-negativity repair."""
    n = len(a)
    a = norm_simplex(a.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (3 * n - 2).bit_length()
    
    for it in range(iterations):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Target: flatten the power spectrum
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Compressed flattening factor
        ratio = P / (P_mean + 1e-30)
        # Soft clipping to avoid extreme corrections
        ratio_clipped = np.clip(ratio, 0.1, 10.0)
        factor = (1.0 / ratio_clipped) ** (alpha / 2.0)
        
        A_new = np.abs(A) * factor * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Non-negativity: use softplus-like repair
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Blend
        a_trial = (1.0 - alpha * 0.5) * a + alpha * 0.5 * a_new
        a_trial = norm_simplex(a_trial)
        
        c = fast_conv(a_trial)
        M = c.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a_trial.copy()
            a = a_trial
            # Increase alpha slightly
            alpha = min(0.6, alpha * 1.02)
        else:
            # Revert, decrease alpha
            alpha = max(0.1, alpha * 0.85)
            if alpha < 0.1:
                break
    
    return best_a, best_m


def peak_reduction_local(a0, time_limit, verbose=False):
    """Local search: directly reduce top peaks by moving mass."""
    n = len(a0)
    a = norm_simplex(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 1000:
        c = fast_conv(a)
        
        # Find top peaks
        n_top = min(10, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            # Find source (highest contributor)
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try moving mass to positions that DON'T contribute to this peak
            candidates = [i for i in range(n) if i < i_start or i >= i_end]
            if not candidates:
                continue
            
            # Pick a random candidate from the less-contributing positions
            for attempt in range(8):
                dst = int(rng.choice(candidates))
                for frac in [0.02, 0.05, 0.08, 0.12]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_dual_peak(n, p1=0.25, p2=0.75, w=0.12):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - p1) ** 2) / (2 * w ** 2)) + np.exp(-((x - p2) ** 2) / (2 * w ** 2)) + 0.05
    return norm_simplex(a)


def make_init_smooth_random(n, seed=0, n_modes=15):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for f in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.01)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate using the global evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def polish_pipeline(a, time_budget, n):
    """Apply a full polishing pipeline to a sequence."""
    best_a = norm_simplex(a.copy())
    best_m = fast_conv(best_a).max()
    original_score = 2.0 * n * best_m
    t0 = time.monotonic()
    
    # Step 1: Quick spectral flatten
    remaining = time_budget - (time.monotonic() - t0)
    t_spec = min(remaining * 0.2, 20.0)
    if t_spec > 3:
        a_sp, m_sp = spectral_flatten(best_a, alpha=0.25, iterations=50, time_limit=t_spec)
        if m_sp < best_m:
            best_a = a_sp
            best_m = m_sp
    
    # Step 2: Adam LSE optimization
    remaining = time_budget - (time.monotonic() - t0)
    t_lse = min(remaining * 0.5, 60.0)
    if t_lse > 5:
        a_lse, m_lse = adam_lse_optimize(best_a, t_lse, beta_start=1.0, beta_end=200.0,
                                          lr=0.005, verbose=False)
        if m_lse < best_m:
            best_a = a_lse
            best_m = m_lse
    
    # Step 3: Local peak reduction
    remaining = time_budget - (time.monotonic() - t0)
    t_local = min(remaining * 0.3, 20.0)
    if t_local > 3:
        a_loc, m_loc = peak_reduction_local(best_a, t_local)
        if m_loc < best_m:
            best_a = a_loc
            best_m = m_loc
    
    return best_a


def propose_candidate(budget_s=1800.0):

    t_global = time.monotonic()

    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = norm_simplex(parent)
            p_score = evaluate_safe(p_norm)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = p_norm.copy()
            best_score = p_score
    except Exception:
        pass

    print(f"\n=== Phase 1: Adam LSE multi-start (n={n0}) ===")
