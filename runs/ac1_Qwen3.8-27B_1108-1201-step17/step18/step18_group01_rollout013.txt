import numpy as np

def fast_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return result

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # Sort in descending order
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad(a, p):
    """
    Compute gradient of soft-max smoothed max conv.
    g[i] = sum_k p[k] * a[k-i] for valid indices.
    
    Using FFT: g[i] = conv(a, p_rev)[2n-2-i] where p_rev[k] = p[2n-2-k]
    """
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (3 * n - 2 - 1).bit_length()  # need length >= (n-1) + (2n-2) + 1 = 3n-2
    A = np.fft.rfft(a, fft_size)
    p_rev = p[::-1]  # reverse p
    P = np.fft.rfft(p_rev, fft_size)
    conv_result = np.fft.irfft(A * P, fft_size)[:3*n-2]
    # g[i] = conv(a, p_rev)[2n-2-i] for i=0,...,n-1
    # conv_result has length 3n-2, indices 0 to 3n-3
    # We need indices 2n-2, 2n-3, ..., n-2 (which is n values)
    g = conv_result[2*n-2 : n-2 : -1]  # This gives n values
    # Wait, let me recheck.
    # conv(a, p_rev)[m] = sum_j a[j] * p_rev[m-j]
    # We want g[i] = sum_j a[j] * p[i+j]
    # p[i+j] = p_rev[2n-2-i-j]

    # For i=0: index 2n-2
    # For i=n-1: index 2n-2-(n-1) = n-1
    # So g[i] = conv_result[2n-2-i]
    g = conv_result[2n-2 - np.arange(n)]

def mgd_optimize(a0, time_limit, top_k=40, eta0=0.04, verbose=False):
    """Multiplicative Gradient Descent with multi-lag subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    eta = eta0
    patience = 0
    max_patience = 400
    cur_top_k = top_k
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, fft_size)
        c = np.fft.irfft(A * A, fft_size)[:conv_len]
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        k_max = min(cur_top_k, conv_len)
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = np.maximum(c[top_idx], 1e-30)
        w = top_vals ** 2.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * a[ki - idx]
        
        grad = grad - grad.mean()
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        A_new = np.fft.rfft(a_new, fft_size)
        c_new = np.fft.irfft(A_new * A_new, fft_size)[:conv_len]
        M_new = c_new.max()
        
        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.9995
        else:
            eta *= 0.5
            if eta < 0.0005:
                cur_top_k = max(5, cur_top_k // 2)
                eta = 0.02
                patience = max(0, patience - 100)
        
        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}, tk={cur_top_k}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a

                M_trial = c_t.max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                A_new = np.fft.rfft(a, fft_size)
                c_new = np.fft.irfft(A_new * A_new, fft_size)[:conv_len]
                if c_new.max() < best_m:

        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def block_search(a0, time_limit, block_size=20, verbose=False):
    """Block-level local search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    stall = 0
    max_stall = 400
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        A = np.fft.rfft(a, fft_size)
        c = np.fft.irfft(A * A, fft_size)[:conv_len]
        M = c.max()
        improved = False
        
        top_lags = np.argsort(c)[::-1][:20]
        
        for attempt in range(40):
            if time.monotonic() - t0 > time_limit:
                break
            
            if attempt < 15:
                k_pick = int(top_lags[attempt % len(top_lags)])
                js = max(0, k_pick - n + 1)
                je = min(n - 1, k_pick)
                center = (js + je) // 2
                start = max(0, min(n - block_size, center - block_size // 2 + int(rng.integers(-10, 10))))
            else:
                start = int(rng.integers(0, max(1, n - block_size + 1)))
            
            end = min(n, start + block_size)
            bs = end - start
            if bs < 2:
                continue
            
            block = a[start:end].copy()
            mean_block = block.mean() + 1e-8
            scale = rng.uniform(0.01, 0.06)
            pert = rng.normal(0, scale * mean_block, bs)
            new_block = np.maximum(block + pert, 0.0)
            
            a_temp = a.copy()
            a_temp[start:end] = new_block
            a_temp = np.maximum(a_temp, 0.0)
            s_t = a_temp.sum()
            if s_t < 1e-15:
                continue
            a_temp = a_temp / s_t
            A_t = np.fft.rfft(a_temp, fft_size)
            c_t = np.fft.irfft(A_t * A_t, fft_size)[:conv_len]
            M_temp = c_t.max()
            
            if M_temp < M - 1e-14:
                a = a_temp
                M = M_temp
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                improved = True
                if attempt >= 25:
                    break
        
        if not improved:
            stall += 1
            # Random escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                A_new = np.fft.rfft(a, fft_size)
                c_new = np.fft.irfft(A_new * A_new, fft_size)[:conv_len]
                if c_new.max() < best_m:
                    best_m = c_new.max()
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Block done ({elapsed:.0f}s): J={J:.6f}")
    return best_a
