def normalize_simplex(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.reshape(-1)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    arr = arr / s
    arr = np.clip(arr, 0.0, 1000.0)
    s2 = float(arr.sum())

def corr_y_x(y, x):

"""
Hybrid search for the evaluation objective.

The code first checks the actual evaluator on simple candidates.  For the
written evaluator with non-negative entries, the objective has a trivial
lower bound of 2, achieved by a uniform sequence; in that regime it returns
one quickly.  If the available evaluator or parent construction indicates a
different landscape, it refines the best known non-negative sequence using:
1) an LP-dual / multiplicative-weights alternating method inspired by the
   previous state of the art;
2) a black-box hill-climbing pool using Fourier, spectral, smoothing,
   transfer, and crossover mutations;
3) tie-broken evaluations against the supplied evaluate_sequence oracle.
"""

import numpy as np
import time


MAX_SEARCH_LEN = 8000
TARGET_PREFERENCE_LEN = 4200
PREFERRED_SCORE_EPS = 1e-12


def safe_next_fast_len(x):
    try:
        x = int(x)
    except Exception:
        x = 1
    if x < 1:
        x = 1
    try:
        return int(np.fft.next_fast_len(x))
    except Exception:
        p = 1
        while p < x:
            p <<= 1
        return p


def normalize_simplex(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.reshape(-1)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    arr = arr / s
    arr = np.clip(arr, 0.0, 1000.0)
    s2 = float(arr.sum())
    if s2 < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s2


def make_uniform(n):
    n = int(max(1, n))
    return np.full(n, 1.0 / float(n), dtype=np.float64)

def interp_to_length(a, m):
    arr = normalize_simplex(a)
    m = int(max(1, m))
    if arr.size == m:
        return arr.copy()
    if arr.size == 1:
        return np.full(m, arr.sum() / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, m, dtype=np.float64)
    b = np.interp(x_new, x_old, arr)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s < 1e-15:
        return make_uniform(m)
    return b / s


def fast_convolution(a):
    arr = np.asarray(a, dtype=np.float64).reshape(-1)
    n = arr.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = 2 * n - 1
    L = safe_next_fast_len(size)
    A = np.fft.rfft(arr, L)
    out = np.fft.irfft(A * A, L)[:size]
    return np.maximum(out, 0.0)


def fast_convolution2(a, b):
    arr = np.asarray(a, dtype=np.float64).reshape(-1)
    brr = np.asarray(b, dtype=np.float64).reshape(-1)
    if arr.size == 0 or brr.size == 0:
        return np.empty(0, dtype=np.float64)
    size = arr.size + brr.size - 1
    L = safe_next_fast_len(size)
    A = np.fft.rfft(arr, L)
    B = np.fft.rfft(brr, L)
    out = np.fft.irfft(A * B, L)[:size]
    return np.maximum(out, 0.0)


def corr_y_x(y, x):
    """Return d[j] = sum_i y[i+j]*x[i] for j=0..len(x)-1."""
    y_arr = np.asarray(y, dtype=np.float64).reshape(-1)
    x_arr = np.asarray(x, dtype=np.float64).reshape(-1)
    if y_arr.size == 0 or x_arr.size == 0:
        return np.empty(0, dtype=np.float64)
    if y_arr.size < x_arr.size:
        y_arr = np.pad(y_arr, (0, x_arr.size - y_arr.size), mode="constant")
    L = safe_next_fast_len(y_arr.size + x_arr.size - 1)
    Y = np.fft.rfft(y_arr, L)
    X = np.fft.rfft(x_arr, L)
    out = np.fft.irfft(Y * np.conj(X), L)[:x_arr.size]
    return np.maximum(out, 0.0)


def internal_score(a):
    arr = normalize_simplex(a)
    n = arr.size
    if n == 0:
        return float("inf")
    c = fast_convolution(arr)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(c.max()))


def safe_evaluate(a):

def better_candidate(a, score, best_a, best_score):

        return False

    lb = int(len(best_a))

        return True
    if lb == TARGET_PREFERENCE_LEN and la != TARGET_PREFERENCE_LEN:

    if la < lb:

    return False


def init_cosine(n, mode=1):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.cos(np.pi * float(mode) * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0) + 1e-6
    return normalize_simplex(a)


def init_bowl(n, sharpness=2.0):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (4.0 * (x - 0.5) ** 2) ** float(sharpness) + 0.05
    return normalize_simplex(a)


def init_dual(n, p1=0.25, p2=0.75, width=0.12):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-((x - p1) ** 2) / (2.0 * width * width))
    a += np.exp(-((x - p2) ** 2) / (2.0 * width * width))
    a += 0.05
    return normalize_simplex(a)


def init_random_smooth(n, seed=0):
    n = int(max(2, n))
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.full(n, 0.5, dtype=np.float64)
    for f in range(1, 25):
        amp = float(rng.normal(0.0, 0.12 / np.sqrt(float(f))))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a += amp * np.sin(2.0 * np.pi * float(f) * x + ph)
    a = np.maximum(a, 1e-5)
    return normalize_simplex(a)


def blend_with_uniform(a, alpha):
    a = normalize_simplex(a)
    alpha = float(np.clip(alpha, 0.0, 1.0))
    uniform = np.ones_like(a) / float(a.size)
    return normalize_simplex((1.0 - alpha) * a + alpha * uniform)


def fourier_perturb(a, rng, amp=0.1, terms=10):
    a = normalize_simplex(a)
    n = a.size
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    pattern = np.zeros(n, dtype=np.float64)
    amp = float(np.clip(amp, 1e-6, 2.0))
    terms = int(max(1, terms))
    for _ in range(terms):
        f = int(rng.integers(1, 32))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        cf = amp / np.sqrt(float(terms))
        pattern += cf * float(rng.standard_normal()) * np.sin(2.0 * np.pi * f * x + ph)
        pattern += cf * float(rng.standard_normal()) * np.cos(2.0 * np.pi * f * x + ph)
    pattern = np.clip(pattern, -4.0, 4.0)
    trial = a * np.exp(pattern)
    trial = np.maximum(trial, 0.0)
    s = float(trial.sum())
    if s < 1e-15:
        return a.copy()
    return trial / s


def neighbor_smooth(a, rng):
    a = normalize_simplex(a)
    n = a.size
    if n < 3:
        return a.copy()
    center = int(rng.integers(n))
    w = int(rng.integers(1, min(64, max(2, n // 3))))
    lo = max(0, center - w)
    hi = min(n, center + w + 1)
    if hi <= lo:
        return a.copy()
    avg = float(a[lo:hi].mean())
    frac = float(rng.uniform(0.25, 0.9))
    trial = a.copy()
    trial[lo:hi] = (1.0 - frac) * trial[lo:hi] + frac * avg
    return normalize_simplex(trial)


def pair_transfer(a, rng):
    a = normalize_simplex(a)
    n = a.size
    if n < 2:
        return a.copy()
    src = int(rng.integers(n))
    dst = int(rng.integers(n))
    if src == dst:
        dst = (dst + 1) % n
    amt = float(a[src]) * float(rng.uniform(0.01, 0.25))
    trial = a.copy()
    trial[src] -= amt
    trial[dst] += amt
    return normalize_simplex(trial)


def crossover_candidates(a, b, rng):
    a = normalize_simplex(a)
    b = normalize_simplex(b)
    if len(a) != len(b):
        b = interp_to_length(b, len(a))
    lam = float(rng.uniform(0.25, 0.75))
    trial = lam * a + (1.0 - lam) * b
    return normalize_simplex(trial)


def spectral_compress(a, gamma=0.6):
    a = normalize_simplex(a)
    n = a.size
    if n < 4:
        return a.copy()
    L = safe_next_fast_len(2 * n - 1)
    A = np.fft.rfft(a, L)
    mag = np.abs(A)
    ph = np.angle(A)
    mmean = float(mag.mean())
    if mmean < 1e-30:
        return a.copy()
    gamma = float(np.clip(gamma, 0.05, 0.95))
    ratio = mag / (mmean + 1e-30)
    mag_new = mmean * np.power(np.maximum(ratio, 1e-30), gamma)
    A_new = mag_new * np.exp(1j * ph)
    b = np.fft.irfft(A_new, L)[:n]
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s < 1e-15:
        return a.copy()
    b = b / s
    mix = 0.35 + 0.45 * gamma
    return normalize_simplex((1.0 - mix) * a + mix * b)

                best_a = a_out.copy()
                best_score = sc_out
                print(f"new best after LP-dual: n={len(best_a)}, score={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        t_bb = min(remaining * 0.45, 650.0)
        if t_bb > 30.0:
            print(f"running black-box pool refinement for {t_bb:.0f}s")
            a_out, sc_out = blackbox_refine(
                best_a,
                best_score,
                time.monotonic() + t_bb,
                max_evals=12000,
            )
            if better_candidate(a_out, sc_out, best_a, best_score):
                best_a = a_out.copy()
                best_score = sc_out
                print(f"new best after black-box: n={len(best_a)}, score={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        t_lp2 = min(remaining * 0.55, 320.0)
        if t_lp2 > 30.0:
            print(f"running second LP-dual pass for {t_lp2:.0f}s")
            a_out, sc_out = lp_dual_refine(
                best_a,
                best_score,
                time.monotonic() + t_lp2,
                max_actual_evals=450,
                max_iters=9000,
            )
            if better_candidate(a_out, sc_out, best_a, best_score):
                best_a = a_out.copy()
                best_score = sc_out
                print(f"new best after final LP-dual: n={len(best_a)}, score={best_score:.8f}")

    final_seq = normalize_simplex(best_a)
    if len(final_seq) == 0:
        final_seq = make_uniform(TARGET_PREFERENCE_LEN)

    final_score = safe_evaluate(final_seq)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL returned: n={len(final_seq)}, last score={best_score:.8f}")
    return [float(x) for x in final_seq.tolist()]
