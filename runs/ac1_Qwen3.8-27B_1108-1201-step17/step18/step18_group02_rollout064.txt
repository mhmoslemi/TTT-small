"""
Minimizes 2*n*max_conv(a,a)/(sum(a))^2 using:
1. LP Frank-Wolfe with away-steps, line search, margin-based constraint management
2. Subgradient multiplicative polish exploiting the active lag set
3. Diverse parametric initialization (spectral, comb, chirp, edge-concentrated)
4. Multi-length exploration via linear interpolation
All helpers are top-level (no closures, no lambdas) per the constraints.
"""

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                print("  LP solver: " + str(s))
                return s
        except Exception:
            continue
    print("  LP solver: ECOS (fallback)")
    return cp.ECOS


def build_lp_matrix(constraint_list, n, y):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_frank_wolfe(a0, time_limit, solver, K_init=200, K_max=2000, verbose=True):
    n = len(a0)
    y = norm_sum1(a0).copy()
    x_avg = y.copy()
    ac = fast_autocorr(y)
    best_a = y.copy()
    best_m = ac.max()
    t0 = time.monotonic()

    top_idx = np.argsort(ac)[::-1][:K_init]
    constraint_set = set(int(k) for k in top_idx)
    it = 0
    fails = 0

    while time.monotonic() - t0 < time_limit:
        ac_y = fast_autocorr(y)
        M = ac_y.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = y.copy()
        ac_x = fast_autocorr(x_avg)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x_avg.copy()

        cl = sorted(constraint_set)
        if len(cl) > K_max:
            margins = np.array([M - ac_y[k] for k in cl])
            keep = np.argsort(margins)[:K_max]
            constraint_set = set(int(cl[i]) for i in keep)
            cl = sorted(constraint_set)

        K = len(cl)
        if K < 10:
            break

        C = build_lp_matrix(cl, n, y)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        con = [cp.sum(g_var) == 1, C @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), con)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            fails += 1
            if fails > 3:
                constraint_set = set(int(k) for k in np.argsort(ac_y)[::-1][:K_init])
                fails = 0
            continue

        if prob.status not in ("optimal", "optimal_inaccurate"):
            fails += 1
            if fails > 5 and K > K_init:
                constraint_set = set(int(k) for k in cl[:K_init])
                fails = 0
            continue

        fails = 0
        g = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g.sum()
        if gs < 1e-15:
            continue
        g = g / gs

        size = 2 * n - 1

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g, fft_size)

        viol = np.where(conv_yg > M + 1e-8)[0]

        e_s[s_idx] = 1.0
        d_std = g - y
        d_away = g - e_s
        gap_std = float(y @ d_std)
        gap_away = float(y @ d_away)
        use_away = (y[s_idx] > 1e-5 and gap_away < gap_std - 1e-10)
        direction = d_away if use_away else d_std

        best_alpha = 2.0 / (it + 2.0)
        best_M = M
        alphas = [1.0, 0.7, 0.5, 0.3, 0.15, best_alpha, 0.05]
        for alpha in alphas:

                continue

            if yt.min() < -1e-12:
                continue

            yt = yt / st
            Mt = fast_autocorr(yt).max()
            if Mt < best_M:
                best_M = Mt
                best_alpha = alpha

        y_new = y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta = (it + 1.0) / (it + 3.0)
        x_avg = y_new + beta * (y_new - y)
        x_avg = np.maximum(x_avg, 0.0)
        sx = x_avg.sum()
        if sx < 1e-15:
            x_avg = y_new.copy()
        else:
            x_avg = x_avg / sx

        y = y_new
        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print("    LPFW iter=%d (%.0fs): J=%.6f |S|=%d" % (it, elapsed, 2.0 * n * best_m, len(constraint_set)))

    if verbose:
        print("    LPFW done (%.0fs): J=%.6f it=%d" % (time.monotonic() - t0, 2.0 * n * best_m, it))
    return best_a, best_m


def gradient_polish(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    stall = 0

            break
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break

        thresh = M - max(M * 0.005, 1e-10)
        active_lags = np.where(ac >= thresh)[0]
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(ac))])

        grad = np.zeros(n)
        for k in active_lags:
            ki = int(k)
            js1 = max(0, ki - n + 1)
            je1 = min(n - 1, ki)
            if js1 <= je1:
                grad[js1:je1 + 1] += a[ki - np.arange(js1, je1 + 1)]
            js2 = max(0, ki)
            je2 = min(n - 1, ki + n - 1)
            if js2 <= je2:
                grad[js2:je2 + 1] += a[np.arange(js2, je2 + 1) - ki]
        grad = grad - grad.mean()

        log_a = np.log(np.maximum(a, 1e-30)) - step * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        if fast_autocorr(a_new).max() < M - 1e-15:
            a = a_new
            step = min(step * 1.02, 1.0)
        else:
            step *= 0.7

        if verbose and it % 500 == 0:
            print("    GradPolish it=%d (%.0fs): J=%.6f" % (it, time.monotonic() - t0, 2.0 * n * best_m))
    return best_a, best_m


def make_initializations(n):
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    out = []
    for freq in [1, 2, 3, 4]:
        a = np.sin(np.pi * freq * x) ** 2
        a = np.maximum(a, 1e-6)
        out.append(a / a.sum())
    a = np.cos(np.pi * (x - 0.5)) ** 2
    out.append(a / a.sum())
    out.append(np.ones(n) / n)
    for sigma in [0.1, 0.2, 0.3]:
        c = n // 2
        a = np.exp(-((i - c) ** 2) / (2 * (sigma * n) ** 2))
        out.append(a / a.sum())
    for alpha in [0.5, 1.0, 2.0]:
        a = np.minimum(i + 1.0, (n - 1) - i + 1.0) ** alpha
        a = np.maximum(a, 1e-6)
        out.append(a / a.sum())
    for period in [2, 3, 4, 5]:
        a = np.where(np.arange(n) % period < period // 2, 1.0, 0.1)
        out.append(a / a.sum())
    for rate in [2, 5, 10]:
        a = (1.0 + np.cos(2 * np.pi * rate * x * x)) / 2.0
        a = np.maximum(a, 0.05)
        out.append(a / a.sum())
    return out


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    si = fi.sum()
    return fi / si if si > 1e-15 else np.ones(new_n) / new_n


def record_best(best_seq, best_score, a, label, t_start):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    an = a / s
    sc = evaluate_safe(an)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = an.copy()
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.0fs)" % (label, sc, len(an), time.monotonic() - t_start))
    return best_seq, best_score
