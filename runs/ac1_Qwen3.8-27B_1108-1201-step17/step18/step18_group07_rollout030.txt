from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lp_step(a, M):

    n = len(a)
    # Build constraint matrix A_ub (shape: (2n-1, n))
    # A_ub[k, j] = a[j] if 0 <= k-j <= n-1, else 0
    # i.e., A_ub[k, j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k)

    rows = []
    cols = []
    vals = []
    
    for j in range(n):

        row_start = j
        row_end = j + n  # exclusive
        if a[j] > 0:
            rows.extend(range(row_start, row_end))
            cols.extend([j] * n)
            vals.extend([a[j]] * n)
    
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)

    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_conv_matrix_sparse(a):
    """Build sparse matrix T(a) where (T(a)*g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        aj = a[j]
        if aj > 1e-15:
            r = np.arange(j, j + n)
            rows.append(r)
            cols.append(np.full(n, j, dtype=np.int32))
            vals.append(np.full(n, aj))
    if not rows:
        return sparse.csr_matrix((2 * n - 1, n), dtype=np.float64)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_step(a, M, verbose=False):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)[k] <= M for all k. Returns g or None."""
    n = len(a)
    A_ub = build_conv_matrix_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
                return g
    except Exception as e:
        if verbose:
            print(f"    LP error: {e}")
    return None

def smooth_grad_fft(a, c, beta):
    """Compute gradient of (2n/beta)*log(sum(exp(beta*c))) using FFT cross-correlation."""

    L = 1 << (3 * n - 1).bit_length()

    exponents = np.minimum(beta * (c - max_c), 50.0)

    A = np.fft.rfft(a, L)
    R = np.fft.irfft(W * np.conj(A), L)[:n]
    
    grad = 4.0 * n * R
    return grad

def spectral_smooth_step(a, alpha=0.3):
    """One step of spectral flattening."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - alpha)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return norm_simplex(a_new)

    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    step = 0.3
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_candidate = spectral_smooth_step(a, step)
        c = fast_conv(a_candidate)
        M = c.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a_candidate.copy()
            a = a_candidate
            step = min(0.5, step * 1.02)
        else:
            a_candidate = norm_simplex(0.7 * a + 0.3 * best_a)
            step = max(0.05, step * 0.8)
            if step < 0.05:
                break
    
    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            for attempt in range(3):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_simplex(a)


def make_init_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_tukey(n, r=0.3):
    """Tukey (tapered cosine) window."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    m = int(r * n / 2)
    if m > 0:
        for i in range(m):
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / m - 1)))
            a[n - 1 - i] = a[i]
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
