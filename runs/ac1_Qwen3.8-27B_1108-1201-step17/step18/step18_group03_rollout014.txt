"""
Minimizes J = 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive softmax mirror descent (accelerated) as main optimizer
2. LP-based Frank-Wolfe with cutting planes for precise local optimization
3. Multi-length exploration and parametric initialization

import numpy as np
import time
import cvxpy as cp

_solvers_tried = None

def _get_solver():
    global _solvers_tried
    if _solvers_tried is not None:
        return _solvers_tried
    _solvers_tried = None
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solvers_tried = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    print("  Solver fallback: ECOS")
    return cp.ECOS


def _fft_autocorr(a):
    """Compute full linear autocorrelation (self-convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _norm_sum1(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _score_normalized(a):
    """Compute J assuming a is already normalized (sum=1, non-negative)."""
    n = len(a)
    ac = _fft_autocorr(a)
    return 2.0 * n * ac.max()


def _score_any(a):
    """Compute J for any input (will be clipped and normalized by evaluator)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = _fft_autocorr(a)
    return float(2.0 * len(a) * ac.max() / (s * s))


def _mirror_descent(a0, time_limit, beta_start, beta_end, c_lr, momentum, restart_every):
    """
    Accelerated mirror descent (multiplicative weights) on the simplex.
    Minimizes max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    
    Parameters:
    - a0: initial sequence (will be normalized)
    - time_limit: seconds to run
    - beta_start, beta_end: softmax temperature schedule (increasing)
    - c_lr: learning rate constant (eta_t = c_lr / sqrt(t))
    - momentum: momentum coefficient for acceleration
    - restart_every: restart after this many stalled iterations
    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fft_autocorr(a).max()
    t0 = time.monotonic()
    
    x_prev = a.copy()
    x = a.copy()
    fft_size = 1 << (3 * n - 3).bit_length()  # For gradient computation
    stall = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Nesterov extrapolation
        if t > 1:
            mu = (t - 1.0) / (t + 1.0)
            y = x + mu * (x - x_prev)
        else:
            y = x.copy()
        
        # Project to simplex (clip negative, renormalize)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation
        ac = _fft_autocorr(y)
        M = ac.max()
        
        # Check improvement
        if M < best_m - 1e-12:
            best_m = M
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > restart_every:
            stall = 0
            # Perturb and restart from best
            rng = np.random.default_rng(t)
            noise = rng.normal(0, 0.01, n) * best_a
            x = _norm_sum1(best_a + noise)
            x_prev = x.copy()
            continue
        
        # Adaptive beta
        progress = min(1.0, t / max(1, restart_every * 5))
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute softmax weights on autocorrelation
        # Clip for numerical stability
        ac_clipped = np.minimum(ac, 100.0 / beta)  # Prevent overflow
        exp_beta = np.exp(beta * (ac_clipped - ac.max()))
        p = exp_beta / exp_beta.sum()
        
        # Compute gradient: grad[i] = sum_k p[k] * y[k-i] (valid indices)
        # Using FFT: grad[i] = conv(y, p_rev)[2n-2-i]
        p_rev = p[::-1]
        Y = np.fft.rfft(y, fft_size)
        P = np.fft.rfft(p_rev, fft_size)
        conv_full = np.fft.irfft(Y * P, fft_size)[:3*n-2]
        grad = conv_full[2*n-2 : n-2 : -1]  # n elements, reversed
        grad = np.maximum(grad, 0.0)  # Gradient should be non-negative for non-negative a
        
        # Learning rate
        eta = c_lr / (np.sqrt(t) + 1.0)
        
        # Multiplicative weights update
        log_x = np.log(np.maximum(y, 1e-30)) - eta * grad
        log_x -= log_x.max()  # Numerical stability
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = x
        x = x_new
        
        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print(f"    MD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def _lp_fw(a0, time_limit, solver, init_K, max_K):
    """
    LP-based Frank-Wolfe with cutting planes and Nesterov acceleration.
    """
    n = len(a0)
    x = _norm_sum1(a0)
    best_a = x.copy()
    best_m = _fft_autocorr(x).max()
    t0 = time.monotonic()
    
    # Initialize constraint set with top autocorrelation lags
    ac = _fft_autocorr(x)
    init_idx = np.argsort(ac)[::-1][:min(init_K, 2*n-1)]
    constraint_set = set(int(k) for k in init_idx)
    
    y = x.copy()  # Nesterov's y
    x_prev = x.copy()
    t = 0
    failed = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Update constraint set size
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            ac_y = _fft_autocorr(y)
            M_y = ac_y.max()
            margins = np.array([M_y - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build LP
        # Variables: g (n vars) and M (1 var)
        # Minimize M
        # Subject to: (y*g)[k] <= M for k in constraint_list
        #              sum(g) = 1
        #              g >= 0
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        # Build constraint matrix efficiently
        # (y*g)[k] = sum_i y[i] * g[k-i] for valid i
        # Coefficient of g[j] in constraint k: y[k-j] if 0 <= k-j < n
        
        constraints = [cp.sum(g_var) == 1]
        
        # For efficiency, build the matrix in chunks if needed
        A = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            # g[j] appears with coefficient y[k-j] if 0 <= k-j < n
            # So j ranges from max(0, k-n+1) to min(k, n-1)
            j_start = max(0, k_int - n + 1)
            j_end = min(k_int, n - 1)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                A[jj, js] = y[k_int - js]
        
        constraints.append(A @ g_var <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed += 1
            if failed > 3:
                # Reduce constraints
                constraint_list = sorted(constraint_set)[:min(init_K // 2, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.asarray(g_var.value, dtype=np.float64)
        g_val = np.maximum(g_val, 0.0)
        sg = g_val.sum()
        if sg < 1e-15:
            continue
        g_dir = g_val / sg
        
        # Find violated constraints (add to set)
        Yf = np.fft.rfft(y, 1 << (2*n-2).bit_length())
        Gf = np.fft.rfft(g_dir, 1 << (2*n-2).bit_length())
        conv_yg = np.fft.irfft(Yf * Gf, 1 << (2*n-2).bit_length())[:2*n-1]
        
        M_curr = _fft_autocorr(y).max()
        violated = np.where(conv_yg > M_curr + 1e-7)[0]
        for k in violated[:50]:
            constraint_set.add(int(k))
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Update x with Nesterov
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Track best
        M_x = _fft_autocorr(x).max()
        if M_x < best_m - 1e-12:
            best_m = M_x
            best_a = x.copy()
        M_y = _fft_autocorr(y).max()
        if M_y < best_m - 1e-12:
            best_m = M_y
            best_a = y.copy()
        
        if t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def _parametric_search(n, time_limit, best_so_far):
    """Try parametric families to find good initial points."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_so_far.copy()
    best_val = _score_normalized(best_so_far)
    
    candidates = []
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)
    
    # Beta-like
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
    
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.5]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    
    # Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20]:
            candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))
    
    # Sinusoidal
    for freq in [1, 2, 3]:
        for ph in [0.0, 0.25, 0.5]:
            candidates.append(np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01))
    
    # Exponential
    for rate in [1, 2, 5, 10]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * x))
    
    for cand in candidates:
        if time.monotonic() - t0 > time_limit:
            break
        a_n = _norm_sum1(cand)
        val = _score_normalized(a_n)
