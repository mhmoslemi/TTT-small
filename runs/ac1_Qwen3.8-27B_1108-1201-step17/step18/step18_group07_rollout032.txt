import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely evaluate the sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def subgradient_smoothed(a, c, beta, top_k=60):

    """
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argpartition(c, -k_max)[-k_max:]
    top_c = c[top_idx]
    
    max_tc = top_c.max()
    if max_tc < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exp_vals = np.exp(np.maximum((top_c - max_tc) / beta, -80.0))
        w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k_int = int(k)

        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad


def subgradient_top(a, c, top_k=20, hard=False):
    """Subgradient using top-k peaks (hard max approximation)."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx]
    
    max_v = top_vals[0]
    if hard:
        w = np.zeros(k_max)
        w[0] = 1.0
    else:
        if max_v < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            raw_w = np.exp(50.0 * (top_vals - max_v) / max_v)
            raw_w = np.maximum(raw_w, 1e-30)
            w = raw_w / raw_w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-14:
            continue

        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad


def spectral_project(a, n_iters=20, blend=0.5):
    """Iterative spectral projection: flatten |A|^2 while maintaining a>=0, sum=1."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        mag_sq = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Compress dynamic range of power spectrum
        mag_sq_mean = mag_sq.mean()
        if mag_sq_mean < 1e-30:
            break
        ratio = mag_sq / (mag_sq_mean + 1e-30)
        # Geometric mean compression
        log_ratio = np.log(ratio + 1e-30)
        log_new = np.maximum(log_ratio * (1.0 - blend), -30.0)
        mag_sq_new = mag_sq_mean * np.exp(log_new)
        
        # Reconstruct
        mag_new = np.sqrt(mag_sq_new)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        a = a_new
    
    return norm_simplex(a)


def ada_grad_optimize(a0, time_limit, verbose=False):
    """AdaGrad with Nesterov momentum on smoothed max-conv objective."""
    n = len(a0)
    a = norm_simplex(a0.copy())
    v = a.copy()  # Nesterov velocity
    adagrad_state = 1e-8 * np.ones(n)  # Accumulated squared gradients
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    best_obj = 2.0 * n * best_m
    t0 = time.monotonic()
    
    # AdaGrad base learning rate
    lr = 0.05
    mu = 0.9  # Nesterov momentum
    
    # Temperature schedule
    beta_init = best_m * 0.5
    beta_min = best_m * 0.005
    total_iters_est = 80000
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 20000 == 0 and time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(1.0, it / total_iters_est)
        beta = beta_init * (beta_min / beta_init) ** progress
        beta = max(beta, best_m * 0.003)
        
        # Nesterov step
        v = a + mu * (a - v) if it > 1 else a.copy()
        v = np.maximum(v, 1e-15)
        sv = v.sum()
        if sv < 1e-15:
            v = a.copy()
        else:
            v = v / sv
        
        # Compute gradient
        c = fast_conv(v)
        cur_max = c.max()
        
        # Use log-sum-exp smoothing for the gradient
        g = subgradient_smoothed(v, c, beta=beta, top_k=80)
        
        # Also add a small component from hard max for sharpness
        if progress > 0.3:
            g_hard = subgradient_top(v, c, top_k=5, hard=True)
            g = 0.7 * g + 0.3 * g_hard
        
        # AdaGrad update
        adagrad_state += g ** 2
        step = lr / np.sqrt(adagrad_state)
        
        a_new = v - step * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = v.copy()
        else:
            a_new = a_new / s_new
        
        # Update Nesterov velocity
        v = a_new
        a = a_new
        
        # Track best
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a.copy()
            best_obj = 2.0 * n * best_m
        
        # Periodic spectral projection
        if it % 3000 == 0:
            a_spec = spectral_project(a, n_iters=10, blend=0.4)
            c_spec = fast_conv(a_spec)
            if c_spec.max() < best_m:
                best_m = c_spec.max()
                best_a = a_spec.copy()
                a = a_spec.copy()
                v = a.copy()
            
            # Decay learning rate periodically
            if it % 15000 == 0:
                lr *= 0.7
                mu = min(mu + 0.02, 0.98)
        
        # Adaptive restart: if no improvement for a while, perturb
        if it % 10000 == 0:
            c_check = fast_conv(a)
            if c_check.max() > best_m * 1.01:
                # Revert to best and perturb
                rng = np.random.default_rng(it)
                a = norm_simplex(best_a + rng.normal(0, 0.005, n))
                v = a.copy()
                adagrad_state = 1e-8 * np.ones(n)

            elapsed = time.monotonic() - t0
            print(f"      AdaGrad it={it} ({elapsed:.0f}s): obj={2*n*best_m:.6f}, beta={beta:.6f}")
    
    return best_a, best_m


def coordinate_descent_smart(a0, time_limit, verbose=False):
    """Smart coordinate descent: targeted mass transfer to reduce peaks."""
    n = len(a0)
    a = norm_simplex(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        cur_max = c.max()
        
        # Find top peaks
        n_top = min(15, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            if contrib.max() < 1e-10:
                continue
            
            # Try reducing the largest contributor
            src_pos = idx[np.argmax(contrib)]
            src_val = a[src_pos]
            if src_val < 1e-10:
                continue
            
            # Try moving mass to positions that DON'T contribute to this peak
            # or to positions that reduce other peaks
            n_tries = 8
            for _ in range(n_tries):
                # Choose destination: prefer positions far from the peak's support
                dst = int(rng.integers(n))
                if dst == src_pos:
                    continue
                
                # Check if dst contributes to this peak
                if i_start <= dst <= i_end and (k_int - dst) >= 0 and (k_int - dst) < n:
                    # This would still contribute; try anyway with small amount
                    fracs = [0.005, 0.01]
                else:
                    fracs = [0.005, 0.01, 0.03, 0.05, 0.1]
                
                for frac in fracs:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src_pos] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape local minimum
            if stall % 10 == 0:
                n_pert = max(1, n // 500)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m


def make_init_bowl(n, sharpness=2.0, asymmetry=1.0):

    right = (1.0 - x / asymmetry) ** sharpness
    a = np.maximum(left + right, 0.0) + 0.05

def make_init_multi_freq(n, seed=42):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed subgradient with AdaGrad optimizer and Nesterov momentum
2. Adaptive temperature annealing for smoothing
3. Iterative spectral projection (Gerchberg-Saxton style)

import numpy as np
import time


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely evaluate the sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def subgradient_smoothed(a, c, beta, top_k=60):
    """Subgradient of log-sum-exp smoothed max of convolution."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argpartition(c, -k_max)[-k_max:]
    top_c = c[top_idx]
    
    max_tc = top_c.max()
    if max_tc < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exp_vals = np.exp(np.maximum((top_c - max_tc) / beta, -80.0))
        w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad


def subgradient_top(a, c, top_k=20, hard=False):
    """Subgradient using top-k peaks."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx]
    
    max_v = top_vals[0]
    if hard:
        w = np.zeros(k_max)
        w[0] = 1.0
    else:
        if max_v < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            raw_w = np.exp(50.0 * (top_vals - max_v) / max_v)
            raw_w = np.maximum(raw_w, 1e-30)
            w = raw_w / raw_w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k_int = int(k)

            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad


def spectral_project(a, n_iters=20, blend=0.5):
    """Iterative spectral projection: flatten |A|^2 while maintaining a>=0, sum=1."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        mag_sq = np.abs(A) ** 2
        phase = np.angle(A)
        
        mag_sq_mean = mag_sq.mean()
        if mag_sq_mean < 1e-30:
            break
        ratio = mag_sq / (mag_sq_mean + 1e-30)
        log_ratio = np.log(ratio + 1e-30)
        log_new = np.maximum(log_ratio * (1.0 - blend), -30.0)
        mag_sq_new = mag_sq_mean * np.exp(log_new)
        
        mag_new = np.sqrt(mag_sq_new)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    return norm_simplex(a)
