    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0
    Returns the optimal g (normalized to sum 1) or None."""
    n = len(f)

    cols = []

            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

        s = g.sum()
        if s > 1e-10:

    return None

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    active_mask = conv >= threshold

        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * wi * a[k - j]

    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    m = fast_conv(a).max()
    return 2.0 * n * m / (s * s)

def update_best(a, label):
    if a is None:
        return
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        return
    try:
        sc = evaluate_sequence([float(v) for v in a])
    except:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

def peak_cost_map(a, conv, K=10):
    """For each position j, compute its total contribution to the top K peaks."""
    n = len(a)
    top_k = np.argsort(conv)[::-1][:K]
    costs = np.zeros(n)
    for k in top_k:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            costs[j] += a[j] * a[k - j]
    return costs
