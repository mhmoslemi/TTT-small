def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_search(n, time_limit, best_init, best_val):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = norm_sum1(best_init)
    cur_best = best_val

    def try_update(a, label):
        nonlocal cur_best, best_a
        a_n = norm_sum1(a)
        ac = fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < cur_best:
            cur_best = val
            best_a = a_n.copy()
            elapsed = time.monotonic() - t0
            print(f"    Param [{label}]: J={val:.6f} (t={elapsed:.1f}s)")

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        try_update((i_arr + 1.0) ** p, f"pow{p}")
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        try_update((n - i_arr) ** p, f"ipow{p}")

    # Beta-like
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        for beta in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            try_update(a, f"be{alpha}_{beta}")

    # U-shaped (high at both ends)
    for c in [0.5]:
        for s in [0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
            a = np.exp(-((x - c) ** 2) / (2.0 * s * s))
            try_update(a, f"gc{s}")
            # Inverted gaussian (U-shape)
            a2 = 1.0 - 0.5 * np.exp(-((x - c) ** 2) / (2.0 * s * s))
            try_update(a2, f"ug{s}")

    # Bimodal
    for c1 in [0.15, 0.2, 0.25, 0.3]:
        for c2 in [0.7, 0.75, 0.8, 0.85]:
            for sigma in [0.05, 0.08, 0.1, 0.12, 0.15, 0.2]:
                a = np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma))
                try_update(a, f"bi{c1}_{c2}_{sigma}")

    # Sinusoidal
    for freq in [1, 2, 3, 4, 5]:
        for ph in [0.0, 0.2, 0.4, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.001)
            try_update(a, f"sn{freq}_{ph}")

    # Exponential
    for rate in [0.5, 1, 2, 3, 5, 8, 12]:
        try_update(np.exp(-rate * x), f"er{rate}")
        try_update(np.exp(rate * x), f"enr{rate}")
        # Double exponential
        a = np.exp(-rate * x) + np.exp(-rate * (1 - x))
        try_update(a, f"de{rate}")

    # Piecewise linear (trapezoid)
    for r in [0.1, 0.2, 0.3, 0.4]:
        a = np.clip(np.minimum(x / r, (1 - x) / r), 0, 1)
        try_update(a, f"trap{r}")

    # Combine with best
    best_norm = best_init / best_init.sum()
    for w in [0.1, 0.2, 0.3, 0.5]:
        for p in [0.5, 1.0, 1.5, 2.0]:
            a_lin = norm_sum1((i_arr + 1.0) ** p)
            a_blend = w * best_norm + (1 - w) * a_lin
            try_update(a_blend, f"bl{w}_{p}")

    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.1f}s): best J={cur_best:.6f}")
    return best_a, cur_best


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs_matrix = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs_matrix[jj, idx] = y[k_int - idx]
    return coeffs_matrix


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=4000, add_per_iter=60, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0
    no_improve_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve_count = 0

        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
            no_improve_count = 0

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k + 1] for k in range(len(constraint_list))])
            # Actually need to index correctly
            margins = np.array([M_curr - ac[constraint_list[j]] for j in range(len(constraint_list))])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = build_constraint_matrix(y, constraint_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 8:
                constraint_list = sorted(constraint_set)[:min(initial_K // 2, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            failed_solves += 1
            if failed_solves > 5:
                constraint_set = set(list(constraint_set)[:min(initial_K, len(constraint_set))])
                failed_solves = 0
            continue
        failed_solves = 0

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            n_add = min(add_per_iter, len(violations))
            top_v = violations[np.argsort(viol_amt)[::-1][:n_add]]
            for k in top_v:
                constraint_set.add(int(k))

        # Also add top peaks of current y if not already in set
        top_peaks = np.argsort(ac)[::-1][:20]
        for k in top_peaks:
            constraint_set.add(int(k))

        # Nesterov acceleration with adaptive step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        # Check if x_new is better, use it as y if so (restart momentum)
        M_xnew = fast_autocorr(x_new).max()
        M_ynew = fast_autocorr(y_new).max()

        if M_xnew < M_ynew:
            # x_new is better, use it and reset y
            y = x_new
            x = x_new.copy()
        else:
            y = y_new
            x = x_new

        iter_count += 1

        # Track no-improvement for potential restart
        if M_curr >= best_m - 1e-12:
            no_improve_count += 1
        else:
            no_improve_count = 0

        # Adaptive: if stuck, shrink constraint set and restart
        if no_improve_count > 500 and iter_count > 100:
            top_keep = np.argsort(ac)[::-1][:initial_K // 2]
            constraint_set = set(int(k) for k in top_keep)
            no_improve_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LP-CW restart iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|->{len(constraint_set)}")

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-CW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def subgradient_optimize(a0, time_limit, verbose=False):
    """Subgradient method for minimizing max conv(a,a) with projection onto simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    eta = 0.1
    patience = 0
    max_patience = 5000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            # Perturb and restart
            pert = rng.normal(0, 0.01, n) * best_a
            a = norm_sum1(best_a + pert)
            patience = 0
            eta = max(eta * 0.5, 0.001)

        # Find peak lag(s)
        top_k = np.argsort(ac)[::-1][:5]
        k_star = int(top_k[0])

        # Subgradient: d(conv(a,a)[k])/da[i] = 2*a[k-i] (for valid i)
        grad = np.zeros(n)
        for k in top_k[:3]:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            weights = ac[ki] / sum(ac[top_k[:3]])
            grad[idx] += 2.0 * weights * a[ki - idx]

        # Projected subgradient step
        lr = eta / (1.0 + 0.1 * np.sqrt(t))
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a = a_new

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Subgrad t={t} ({elapsed:.0f}s): J={J:.6f}, patience={patience}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    Subgrad done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=42, verbose=True):

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    max_stall = 800
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0

        improved = False
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            # Find top contributing pairs
            n_pairs = min(6, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[pi])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score destinations: want positions that don't contribute to top lags
                scores = np.zeros(n)
                for k2 in top_k[:12]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * ac[k2i]
                        scores[k2i - idx2] += a[idx2] * ac[k2i]

                # Also penalize positions near the peak lag
                scores[src] = np.inf

                # Find best destination
                valid = scores < np.inf
                if valid.sum() < 5:
                    valid = np.ones(n, dtype=bool)
                    valid[src] = False
                dst = int(np.argmin(scores[valid]))

                if a[dst] > 0.7 * src_val:
                    continue

                # Binary search on transfer fraction
                lo, hi = 0.0, 0.4
                best_frac = 0.0
                best_trial_m = M_curr
                for _ in range(8):
                    mid = (lo + hi) / 2
                    if mid < 1e-6:
                        break
                    a_trial = a.copy()
                    amt = src_val * mid
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        break
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_frac = mid
                        hi = mid
                    else:
                        lo = mid

                if best_frac > 0.005:
                    amt = src_val * best_frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    a = a / st
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m - 1e-16:
                        best_m = M_new
                        best_a = a.copy()
                        improved = True

        if not improved:
            stall += 1
            if stall > max_stall:
                break

            # Random perturbation
            if stall % 20 == 0:
                n_moves = int(rng.integers(5, 20))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.005, 0.03))
                        a[i] -= a[i] * frac
                        a[j] += a[i] * frac * (1.0 / (1.0 - frac))
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break

            if verbose and stall % 200 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    CoordPolish ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def fine_polish(a0, time_limit, verbose=True):
    """Very fine-grained polishing: try small transfers to many destinations."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        # Focus on the single worst lag
        k_star = int(np.argmax(ac))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]

        # Target the single largest contributing pair
        pi = int(np.argmax(contrib))
        src = int(idx[pi])
        src_val = a[src]
        if src_val < 1e-12:
            break

        # Try transferring to a random set of destinations
        n_tries = 50
        tried = set()
        improved = False
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            dst = int(rng.integers(n))
            if dst == src or dst in tried:
                continue
            tried.add(dst)
            if a[dst] > 0.9 * src_val:
                continue

            # Try a small transfer
            for frac in [0.01, 0.02, 0.03, 0.05]:
                amt = src_val * frac
                if amt < 1e-14:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try random small perturbations
            for _ in range(10):
                if time.monotonic() - t0 > time_limit:
                    break
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 == i2 or a[i1] < 1e-10:
                    continue
                frac = float(rng.uniform(0.001, 0.01))
                a[i1] -= a[i1] * frac
                a[i2] += a[i1] * frac / (1 - frac) if frac < 1 else 0
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m - 1e-16:
                        best_m = M_new
                        best_a = a.copy()
                        improved = True
                        break

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    FinePolish done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m
