def fast_autocorr(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_gradient(c, beta, top_k=200):
    """
    Compute gradient of soft-max approximation to max(c).
    Returns (grad_contribution, weights) where grad is with respect to the 
    convolution values c, and weights are the softmax probabilities.
    """
    # Use top-k for numerical stability and efficiency
    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    top_vals_max = top_vals.max()
    exp_vals = np.exp(beta * (top_vals - top_vals_max))
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = exp_vals / exp_sum
    return top_idx, weights


def compute_conv_grad(a, c, beta, top_k=200):
    """
    Compute the exact gradient of the soft-max of the full convolution a*a.
    f(a) = (1/beta) * log(sum_k exp(beta * (a*a)[k]))
    
    grad[j] = sum_k p_k * (a[k-j] * I(0<=k-j<n) + a[j-k] * I(0<=j-k<n))
    """
    n = len(a)
    top_idx, weights = softmax_gradient(c, beta, top_k)
    
    grad = np.zeros(n)
    for k_int, w in zip(top_idx, weights):
        k = int(k_int)
        # First term: a[k-j] valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start1 = max(0, k - n + 1)
        j_end1 = min(n - 1, k)
        if j_start1 <= j_end1:
            js = np.arange(j_start1, j_end1 + 1)
            grad[js] += w * a[k - js]
        
        # Second term: a[j-k] valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
        j_start2 = max(0, k)
        j_end2 = min(n - 1, k + n - 1)
        if j_start2 <= j_end2:
            js = np.arange(j_start2, j_end2 + 1)
            grad[js] += w * a[js - k]
    
    return grad


def simplex_project(z):
    """Project z onto the simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(z)
    if n == 0:
        return z
    if n == 1:
        return np.array([max(z[0], 0.0)])  # Actually should be [1.0] for sum=1
        # Wait, if z[0] < 0, we need p[0] = 1? No, the constraint is p >= 0 and sum(p) = 1.
        # So p[0] must be 1.
        return np.array([1.0])
    
    mu = np.sort(z)[::-1]
    cv = np.cumsum(mu)
    rho = np.where(mu > (cv - 1.0) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        # All z are very negative, return uniform
        return np.ones(n) / n
    rho = rho[-1]
    tau = (cv[rho] - 1.0) / (rho + 1.0)
    p = np.maximum(z - tau, 0.0)
    # Ensure sum is exactly 1
    p_sum = p.sum()
    if p_sum > 0:
        p = p / p_sum
    else:
        p = np.ones(n) / n
    return p


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def apgd_optimize(a0, time_limit, beta_start=10.0, beta_end=100.0, 
                  top_k=150, lr_init=0.1, momentum=True, verbose=True):
    """
    Accelerated Projected Gradient Descent with soft-max penalty.
    Minimizes max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_c = fast_autocorr(a)
    best_m = best_c.max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    y = a.copy()  # momentum position
    
    # Initial step size
    # Lipschitz constant estimate: L ~ 2 * max(a) * (2n-1) roughly
    # Better: use the norm of the convolution operator
    # ||a*a|| / ||a|| type bound
    c0 = fast_autocorr(a)
    L = 4.0 * c0.max() * n  # rough upper bound on Lipschitz constant
    lr = lr_init / np.sqrt(max(L, 1.0))
    
    stall = 0
    max_stall = 500
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta: start low, increase
        progress = min(1.0, t / 10000.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Momentum (Nesterov)
        if momentum and t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-30:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = x.copy()
        
        # Compute gradient at y
        c_y = fast_autocorr(y)
        grad = compute_conv_grad(y, c_y, beta, top_k)
        
        # Line search (backtracking)
        f_y = (1.0 / beta) * np.log(np.sum(np.exp(beta * (c_y - c_y.mean())))) 
        # Actually, let's just use the raw max for checking improvement
        lr_t = lr
        improved = False
        for ls in range(20):
            a_trial = y - lr_t * grad
            a_trial = simplex_project(a_trial)
            c_trial = fast_autocorr(a_trial)
            m_trial = c_trial.max()
            if m_trial < c_y.max() - 1e-15:
                improved = True
                break
            lr_t *= 0.5
        
        if not improved:
            # Try smaller step
            a_trial = y - lr_t * grad
            a_trial = simplex_project(a_trial)
        
        x_prev = x
        x = a_trial
        
        # Check improvement
        c_x = fast_autocorr(x)
        m_x = c_x.max()
        if m_x < best_m - 1e-15:
            best_m = m_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive learning rate
        if stall > 100:
            lr *= 0.8
            top_k = min(top_k + 20, 500)
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    APGD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, lr={lr_t:.6f}, stall={stall}")
    
    return best_a, best_m


def lp_cutting_plane(a0, time_limit, solver, initial_K=300, max_K=2500, 
                     add_per_iter=40, verbose=True):
    """LP-based cutting plane Frank-Wolfe for the non-convex max problem."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix A (K x n) where A[j,k] = y[k-j] or y[j-k]
        # (y*g)[k] = sum_i y[i]*g[k-i]
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP: min M s.t. A*g <= M, sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce constraints on failure
            constraint_set = set(constraint_list[:init_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        iter_count += 1
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, rng_seed=42, verbose=True):
    """Coordinate descent: try moving mass from high-autocorrelation contributors."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        M_curr = c.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
        
        improved = False
        # Focus on top lags
        n_top = min(30, len(c))
        top_k = np.argsort(c)[::-1][:n_top]
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Find the pair with max contribution
            max_i = np.argmax(contrib)
            src = idx[max_i]
            dst_idx = k_int - src
            
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try moving some mass from src to a "safe" location
            # Safe = doesn't increase any top lag
            # Simply try random targets
            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                dst = int(rng.integers(n))
                if dst == src or dst == dst_idx:
                    continue
                if a[dst] > 0.3 * src_val:
                    continue
                
                for frac in [0.05, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    c_t = fast_autocorr(a_trial)
                    if c_t.max() < best_m - 1e-16:
                        a = a_trial
                        best_m = c_t.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > max_stall:
                break
            # Random perturbation
            if stall % 20 == 0:
                n_moves = int(rng.integers(2, 8))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.01, 0.05))
                        a[i] -= a[i] * frac
                        a[j] += a[i] * frac
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
    
    return best_a, best_m
