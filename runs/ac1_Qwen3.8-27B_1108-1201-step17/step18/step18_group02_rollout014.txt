def subgradient_of_max_autocorr(a, b, active_lags):
    """Compute subgradient of max(b[k]) w.r.t. a using the shifted-flip structure.
    Gradient of b[k] w.r.t. a[i] = 2*a[k-i] if 0<=k-i<n, else 0.
    Subgradient of max is average over active lags.

    n = len(a)
    g = np.zeros(n)
    for k in active_lags:
        k_int = int(k)
        for i in range(n):
            diff = k_int - i
            if 0 <= diff < n:
                g[i] += 2.0 * a[diff]
    g /= max(1, len(active_lags))
    return g

def subgradient_polish(a0, time_limit, verbose=True):
    """Exact subgradient descent using the shifted-flip gradient structure."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adaptive step size
    step0 = 0.1
    decay = 0.9995
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            break

        # Find active lags (within tolerance of max)
        tol = max(0.02 * M_curr, 1e-10)
        active_lags = np.where(ac >= M_curr - tol)[0]

        # Limit active set size for efficiency
        if len(active_lags) > 200:
            # Keep the top 200
            top_active = ac[active_lags]
            order = np.argsort(top_active)[::-1][:200]
            active_lags = active_lags[order]

        # Compute subgradient using shifted-flip structure
        g = np.zeros(n)
        for k in active_lags:
            k_int = int(k)
            # g[i] += 2*a[k-i] for valid i
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                g[idx] += 2.0 * a[k_int - idx]
        g /= max(1, len(active_lags))

        # Project gradient to tangent space of simplex (sum = 0)
        g_t = g - np.mean(g)

        # Adaptive step size with backtracking
        step = step0 * (decay ** t)
        a_new = a - step * g_t
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < M_curr:
            a = a_new
        else:
            # Backtrack
            step *= 0.5
            step0 = max(step, 0.001)
            a_new = a - step * g_t
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                M_new = fast_autocorr(a_new).max()
                if M_new < M_curr:
                    a = a_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad t={t} ({elapsed:.0f}s): J={J:.6f}, step={step:.4f}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    SubGrad done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

                j = int(rng.integers(n))
                attempts += 1
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, rng=None):
    """Generate diverse parametric initial sequences."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []

    # Sin^2 family with various frequencies and phases
    for freq in [1, 2, 3, 4]:
        for phase in [0.0, 0.3, 0.7]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.005)
            candidates.append(a / a.sum())

    # Asymmetric power law
    for p in [0.5, 1.0, 1.5, 2.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())

    # Two-bump asymmetric
    for c1, c2, s1, s2 in [(0.3, 0.7, 0.1, 0.15), (0.25, 0.65, 0.12, 0.1), (0.4, 0.8, 0.08, 0.12)]:
        a = np.exp(-((x - c1) ** 2) / (2 * s1 ** 2)) + 0.7 * np.exp(-((x - c2) ** 2) / (2 * s2 ** 2))
        candidates.append(a / a.sum())

    # Random smooth (via random Fourier features)
    for trial in range(5):
        n_freq = 20
        freqs = rng.uniform(0.5, 10.0, n_freq)
        phases = rng.uniform(0, 2 * np.pi, n_freq)
        amps = rng.uniform(0.5, 1.0, n_freq)
        a = np.ones(n)
        for f, ph, am in zip(freqs, phases, amps):
            a += am * np.cos(2 * np.pi * f * x + ph)
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())

    # Cosine taper (asymmetric)
    for skew in [0.3, 0.5, 0.7]:
        a = np.cos(np.pi * (x - skew) * (1 - abs(x - skew))) ** 2
        a = np.maximum(a, 0.005)
        candidates.append(a / a.sum())

    return candidates


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick parametric search ===
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 5:
        rng_init = np.random.default_rng(99)
        candidates = parametric_init(n0, rng_init)
        curr_best_m = fast_autocorr(best_seq).max()
        for i, cand in enumerate(candidates):
            if time.monotonic() - t_global > time_budget - t1:
                break
            ac = fast_autocorr(cand)
            val = 2.0 * n0 * ac.max()
            if val < 2.0 * n0 * curr_best_m:
                check_and_update(cand, f"param{i}")

    n_best = len(best_seq)

    # === Phase 2: Main LP-FW with away steps ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW-Away (n={n_best}) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_away(a_in, t2, solver, initial_K=800, max_K=9000,
                             add_per_iter=30, verbose=True)
        check_and_update(a_out, f"lpfw_away_n{n_best}")

    # === Phase 3: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Subgradient polish (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_polish(a_in, t3, verbose=True)
        check_and_update(a_out, f"subgrad_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_polish(a_in, t4, rng_seed=111, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = [n_best + 300, n_best - 300, n_best + 500, n_best - 500]
    nearby_lengths = [ln for ln in nearby_lengths if 3000 <= ln <= 8000]

    seen = set()
    unique_lengths = []
    for ln in nearby_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)

    for new_n in unique_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 200:
            break
        print(f"\n=== Phase 5: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        # LP-FW on new length
        t5 = min(remaining * 0.35, 300.0)
        if t5 > 60:
            a_out, _ = lp_fw_away(a_long, t5, solver, initial_K=600, max_K=6000,
                                 add_per_iter=25, verbose=False)
            check_and_update(a_out, f"lpfw_n{new_n}")

    # === Phase 6: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP-FW (n={n_best}) ===")
    if remaining > 120 and n_best <= 6000:
        t6 = min(remaining * 0.50, 350.0)
        if t6 > 60:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_fw_away(a_in, t6, solver, initial_K=1500, max_K=10000,
                                 add_per_iter=40, verbose=True)
            check_and_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 7: Final subgradient + coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 40:
        t7a = min(remaining * 0.4, 150.0)
        if t7a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_polish(a_in, t7a, verbose=False)
            check_and_update(a_out, f"subgrad_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.5, 100.0)
            if t7b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coord_polish(a_in, t7b, rng_seed=271, verbose=False)
                check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
