import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v):
    """Project onto probability simplex (Duchi et al. 2012)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = rho_candidates[np.argmax(cond)]
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):

    try:

def isotonic_regression(v, direction='increasing'):
    """Isotonic regression via Pool Adjacent Violators Algorithm."""
    n = len(v)
    if direction == 'decreasing':
        v = -v
    blocks = [[v[0], 1]]
    for i in range(1, n):
        blocks.append([v[i], 1])
        while len(blocks) > 1 and blocks[-2][0] / blocks[-2][1] > blocks[-1][0] / blocks[-1][1]:
            m2, c2 = blocks.pop()
            m1, c1 = blocks.pop()
            c = c1 + c2
            s = m1 + m2
            blocks.append([s, c])
    result = np.zeros(n)
    pos = 0
    for s, c in blocks:
        result[pos:pos + c] = s / c
        pos += c
    if direction == 'decreasing':
        return -result
    return result


def make_ramp(n, power=1.0):
    """Create a power ramp: a[i] = (i+1)^power, normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_tapered_ramp(n, power=1.0, taper=0.3):
    """Ramp with softened top to reduce edge peak."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    # Taper the top
    top_start = int(n * (1.0 - taper))
    if top_start < n:
        x_top = np.arange(top_start, n, dtype=np.float64)
        x_ref = top_start
        a[top_start:] = a[top_start] * np.exp(-0.5 * ((x_top - x_ref) / (taper * n * 0.5)) ** 2)
    return norm_simplex(np.maximum(a, 0.0))


def make_exp_ramp(n, rate=1.0):
    """Exponential ramp: a[i] = exp(rate * i / n), normalized."""
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return norm_simplex(a)


def solve_active_lp(a, M, m=50):
    """Solve reduced LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for top m peaks of conv(a,a)."""
    n = len(a)
    c = fast_conv(a)
    top_idx = np.argsort(c)[::-1][:m]

    # Build constraint matrix C (m x n): C[j, l] = a[top_idx[j] - l] if valid
    C = np.zeros((m, n))
    for j, k in enumerate(top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            l = np.arange(i_start, i_end)
            C[j, l] = a[k_int - l]

    # Solve: max 1^T b s.t. C @ b <= M*1, b >= 0
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = M * np.ones(m)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.success:
            b = np.maximum(res.x, 0.0)
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    return None


def active_lp_optimize(a0, time_limit, m_start=20, m_end=80, verbose=False):
    """Iterative active-set LP optimization."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()

        M = c.max()

        if b is None:

        if b is None:

        # Line search
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * b
            a_trial = norm_simplex(a_trial)
            obj_trial = obj_value(a_trial)
            if obj_trial < best_obj - 1e-15:
                a = a_trial
                best_obj = obj_trial

                improved = True
                break

        if not improved:
            # Try isotonic blend to break symmetry

                    improved = True

        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={iter_count} (t={elapsed:.0f}s): obj={best_obj:.8f}")

def compute_pnorm_grad(a, c, p):
    """Compute gradient of sum(c^p) w.r.t. a using FFT."""
    n = len(a)
    d = c ** (p - 1.0)  # length 2n-1
    d_rev = d[::-1]  # length 2n-1

    # grad[i] = p * sum_j a[j] * d[i+j] = p * conv(a, d_rev)[2n-2-i]
    # Compute conv(a, d_rev) via FFT
    fft_size = 1 << (n + len(d_rev) - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    D = np.fft.rfft(d_rev, fft_size)
    conv = np.fft.irfft(A * D, fft_size)[:n + len(d_rev) - 1]

    # grad[i] = p * conv[2n-2-i] for i=0,...,n-1
    # So grad = p * conv[n-1 : 2n-1][::-1]
    grad = p * conv[n - 1:2 * n - 1][::-1]
    return grad


def pnorm_prox_grad(a0, time_limit, p_start=2.0, p_end=32.0, verbose=False):
    """Proximal gradient for p-norm of convolution with simplex projection."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    best_obj = 2.0 * n * best_m
    t0 = time.monotonic()

    lr = 1.0 / (2.0 * n * p_start)  # Initial learning rate
    lr_min = 1e-10
    max_iter = 50000

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (max_iter * 0.9))
        p = p_start * (p_end / p_start) ** progress

        c = fast_conv(a)
        grad = compute_pnorm_grad(a, c, p)

        # Gradient step with backtracking
        step_lr = lr
        improved = False
        for _ in range(10):
            a_trial = proj_simplex(a - step_lr * grad)
            c_trial = fast_conv(a_trial)
            M_trial = c_trial.max()
            obj_trial = 2.0 * n * M_trial
            if obj_trial < best_obj:
                a = a_trial
                best_obj = obj_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                break
            step_lr *= 0.5
            if step_lr < lr_min:
                break

        if improved:
            lr = min(lr * 1.05, 1.0 / (2.0 * n * p))
        else:
            lr *= 0.7

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      pnorm it={it} (t={elapsed:.0f}s): p={p:.1f}, obj={best_obj:.8f}")

def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)

        # Find top peaks
        n_top = min(10, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]

        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            max_contrib_idx = np.argmax(contrib)
            src = idx[max_contrib_idx]
            src_val = a[src]
            if src_val < 1e-10:

                dst = int(rng.integers(n))

                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] = max(0.0, a_trial[src] - amt)
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:

                    if M_new < best_m:

                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def spectral_shape(a, amount=0.2):
    """Flatten the power spectrum slightly."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - amount)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return norm_simplex((1 - amount * 0.5) * a + amount * 0.5 * a_new)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
