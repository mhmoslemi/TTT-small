"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. U-shaped parametric initialization (two power-law blocks)
2. Soft-max accelerated mirror descent (main workhorse, FFT-based)
3. Exact-max gradient refinement (top-K peak tracking)
4. LP-Frank-Wolfe polishing (cvxpy)
5. Multi-length exploration
"""

import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_soft_gradient(a, w, fft_size):
    """
    Compute gradient of soft-max objective.
    grad[i] ≈ 2 * (w cross-correlate a)[i]
    """
    n = len(a)
    # Zero-pad a to fft_size
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    # Zero-pad w to fft_size
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    # Cross-correlation via FFT
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)
    # Circular cross-correlation: IFFT(W * conj(A)) gives (w ⊛ a)
    c = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * c


def u_shaped_init(n, alpha=1.0, beta=1.0, gap_frac=0.3):
    """Create a U-shaped initialization with two power-law blocks."""
    a = np.zeros(n)
    gap = int(n * gap_frac)
    p = (n - gap) // 2
    q = n - gap - p
    # Left block: power law increasing
    i_left = np.arange(p, dtype=np.float64)
    a[:p] = (i_left + 1.0) ** alpha
    # Right block: power law increasing (mirror)
    i_right = np.arange(q, dtype=np.float64)
    a[n-q:] = (i_right + 1.0) ** beta
    return norm_sum1(a)


def softmax_weights(conv, beta):
    """Compute soft-max weights over convolution values."""
    conv_clipped = np.clip(conv, -1e10, 1e10)
    max_conv = conv_clipped.max()
    exp_vals = np.exp(beta * (conv_clipped - max_conv))
    Z = exp_vals.sum()
    if Z < 1e-30:
        return np.ones_like(conv) / len(conv)
    return exp_vals / Z


def softmax_mirror_descent(a0, time_limit, beta_start=1.0, beta_end=10.0,
                           c0=1.0, verbose=True, max_iters=500000):
    """
    Soft-max accelerated mirror descent for minimizing max autocorrelation.
    """
    n = len(a0)
    size = 2 * n - 1
    fft_size_ac = 1 << (size - 1).bit_length()
    # For cross-correlation, need L >= (2n-1) + n - 1 = 3n-2
    fft_size_cc = 1 << (3 * n - 3).bit_length()
    
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    eta = c0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            mu = (t - 1.0) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation of y
        ac_y = fast_autocorr(y)
        
        # Adaptive beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 5000.0)
        
        # Compute soft-max weights
        w = softmax_weights(ac_y, beta)
        
        # Compute gradient
        grad = compute_soft_gradient(y, w, fft_size_cc)
        
        # Mirror descent step
        eta_t = eta / (1.0 + np.sqrt(t))
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            if verbose and t % 1000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SMD iter={t} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * M_x
            print(f"    SMD iter={t} ({elapsed:.0f}s): J_cur={J:.6f}, J_best={2.0*n*best_m:.6f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    SMD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def topk_mirror_descent(a0, time_limit, top_k=50, c0=1.0, verbose=True, max_iters=500000):
    """
    Mirror descent using only the top-K peaks (more precise than soft-max).
    """
    n = len(a0)
    size = 2 * n - 1
    fft_size_ac = 1 << (size - 1).bit_length()
    fft_size_cc = 1 << (3 * n - 3).bit_length()
    
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    eta = c0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            mu = (t - 1.0) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        
        # Find top-K peaks
        top_idx = np.argsort(ac_y)[::-1][:top_k]
        top_vals = ac_y[top_idx]
        # Weight by peak height (normalized)
        w_full = np.zeros(len(ac_y))
        w_full[top_idx] = top_vals / top_vals.sum()
        
        # Compute gradient from top-K peaks
        grad = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            wi = w_full[k_int]
            # d(conv[k])/d(a[i]) = a[k-i] (if valid) + a[k-i] (if i != k-i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                j_idx = k_int - idx
                # Factor of 2 except when i == k-i (i.e., k = 2i)
                factors = np.where(idx == j_idx, 1.0, 2.0)
                grad[idx] += wi * y[j_idx] * factors
        
        # Mirror descent step
        eta_t = eta / (1.0 + np.sqrt(t))
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            if verbose and t % 500 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    TopK-MD iter={t} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * M_x
            print(f"    TopK-MD iter={t} ({elapsed:.0f}s): J_cur={J:.6f}, J_best={2.0*n*best_m:.6f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    TopK-MD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=2000,
                   add_per_iter=30, verbose=True):
    """LP-based Frank-Wolfe with cutting planes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    failed_solves = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 3:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m
