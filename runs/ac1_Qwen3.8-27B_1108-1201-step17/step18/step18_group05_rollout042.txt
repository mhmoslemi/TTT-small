"""
Search for low peak linear self-convolution non-negative heights.

The objective is scale-invariant:
    score = 2*n*max(convolve(a,a))/(sum(a)**2)

Writing a as a probability vector, the problem is to minimize the largest
coefficient of P(x)^2, where P(x)=sum_i a_i x^i.  The main local optimizer
minimizes a smooth LogSumExp approximation to the maximum coefficient using
exponential-gradient steps in log-softmax coordinates.  A few L-BFGS runs are
then used to polish the continuous surrogate, followed by optional length
migration and final polishing.
"""

import numpy as np
import time
from scipy import optimize as scipy_optimize


_LB_DEADLINE = 0.0
_LB_LATEST = None


class _LbfgsTimeout(Exception):
    pass


def _lb_latest_callback(x):
    global _LB_LATEST
    if x is not None:
        _LB_LATEST = np.asarray(x, dtype=np.float64).copy()
    if time.monotonic() > _LB_DEADLINE:
        raise _LbfgsTimeout()


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_positive(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 1e-15:
        arr = np.ones_like(arr, dtype=np.float64)
        s = float(arr.sum())
    if s <= 0.0:
        arr = np.ones_like(arr, dtype=np.float64)
        s = float(arr.sum())
    arr = arr / s
    return arr


def softmax_from_z(z):
    zz = np.asarray(z, dtype=np.float64).ravel()
    if zz.size == 0:
        return None
    zz = np.nan_to_num(zz, nan=-100.0, posinf=50.0, neginf=-50.0)
    m = float(np.max(zz))
    if not np.isfinite(m):
        m = 0.0
    e = np.exp(zz - m)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return e / s


def conv_full(a, N):
    n = len(a)
    L = 2 * n - 1
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)[:L]
    C = np.asarray(C, dtype=np.float64)
    C = np.nan_to_num(C, nan=0.0, posinf=0.0, neginf=0.0)
    C = np.where(C > 0.0, C, 0.0)
    return C


def score_fast(a):
    arr = normalize_positive(a)
    if arr is None:
        return float("inf")
    n = len(arr)
    N = next_pow2(2 * n - 1)
    C = conv_full(arr, N)
    return float(2.0 * n * float(C.max()))


def make_default_sequence(n=5200):
    x = np.linspace(0.0, 1.0, n)
    a = 0.55 * np.exp(-1.35 * x) + 0.45 * np.maximum(1.0 + 0.75 * np.cos(np.pi * x), 1e-8)
    return normalize_positive(a)


def make_shape_exponential(n=5200):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-1.8 * x) + 0.08
    return normalize_positive(a)


def make_shape_cosine(n=5200):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.85 * np.cos(np.pi * x) + 0.08
    return normalize_positive(np.maximum(a, 1e-8))


def make_shape_random(n=5200, seed=17):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = 0.75 * np.ones(n, dtype=np.float64)
    for f in range(1, 31):
        amp = 0.16 / np.sqrt(f)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.sin(2.0 * np.pi * f * x + ph)
    return normalize_positive(np.maximum(a, 1e-8))


def interpolate_sequence(a, new_n):
    arr = normalize_positive(a)
    if arr is None:
        return np.ones(int(new_n))
    old_n = len(arr)
    if old_n == int(new_n):
        return arr.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, int(new_n))
    b = np.interp(new_x, old_x, arr)
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    b = np.maximum(b, 0.0)
    return normalize_positive(b)


def _load_parent_sequence():
    try:
        seq = height_sequence_1
    except NameError:
        return None
    except Exception:
        return None
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if float(arr.sum()) < 1e-12:
        return None
    return arr


def run_sumexp(
    a0,
    time_limit,
    tau_start=3.0,
    tau_end=0.10,
    eta_init=0.15,
    eta_max=1.0,
    eta_min=1e-4,
    rng=None,
    verbose=True,
):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = int(a0.size)
    if n <= 0:
        return np.ones(1)

    if rng is None:
        rng = np.random.default_rng(7)

    L = 2 * n - 1
    Lf = float(L)
    logL = np.log(Lf)
    N = next_pow2(L)

    a = normalize_positive(a0)
    if a is None:
        a = np.ones(n, dtype=np.float64)
    z = np.log(np.maximum(a, 1e-30))
    z -= float(np.mean(z))
    a2 = softmax_from_z(z)
    if a2 is None:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a2

    if time_limit <= 1e-6:
        return a

    t0 = time.monotonic()
    P = np.fft.rfft(a, N)
    C = conv_full(a, N)
    best_cmax = float(C.max())
    best_a = a.copy()
    last_imp = t0
    eta = float(eta_init)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        comp = min(1.0, (time.monotonic() - t0) / max(1e-9, float(time_limit)))
        tau = tau_start * ((tau_end / tau_start) ** comp)
        if tau <= 0:
            tau = max(1e-3, tau_end * 0.5)

        P = np.fft.rfft(a, N)
        C = conv_full(a, N)
        cmax = float(C.max())

        if cmax < best_cmax - 1e-16:
            best_cmax = cmax
            best_a = a.copy()
            last_imp = time.monotonic()

        Cn = C * Lf
        mCn = float(Cn.max())

        zrel = (Cn - mCn) / tau
        zrel = np.nan_to_num(zrel, nan=-100.0, posinf=0.0, neginf=-100.0)
        zmax = float(zrel.max())
        if zmax > 0.0:
            zrel = zrel - zmax
        e = np.exp(zrel)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 1e-300:
            w = np.zeros_like(Cn, dtype=np.float64)
            w[int(np.argmax(Cn))] = 1.0
            val_cur = mCn
        else:
            w = e / s
            val_cur = mCn + tau * (float(np.log(s)) - logL)

        W = np.fft.rfft(w, N)
        corr = np.fft.irfft(np.conj(P) * W, N)[:n]
        corr = np.nan_to_num(corr, nan=0.0, posinf=0.0, neginf=0.0)
        corr = np.where(corr > 0.0, corr, 0.0)

        g = 2.0 * Lf * corr
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        g = g - float(np.mean(g))
        scale = float(np.sqrt(np.mean(g * g)))
        if not np.isfinite(scale) or scale < 1e-14:
            z_base = np.log(np.maximum(best_a, 1e-30))
            z_base -= float(np.mean(z_base))
            z = z_base + rng.normal(0.0, 0.02, n)
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            if a is None:
                a = best_a.copy()
            continue

        d = g / scale
        d = np.clip(d, -3.0, 3.0)
        d = d - float(np.mean(d))
        sd = float(np.sqrt(np.mean(d * d)))
        if not np.isfinite(sd) or sd < 1e-14:
            continue
        d = d / sd

        best_alpha = 0.0
        best_val = val_cur
        best_trial_cmax = cmax
        best_z = None
        best_p = None
        allow_cmax = cmax + max(1e-14, cmax * 1e-5)

        alphas = [eta, 0.5 * eta, 1.8 * eta]
        for alpha in alphas:
            if alpha <= 0.0 or not np.isfinite(alpha):
                continue
            if time.monotonic() - t0 > time_limit:
                break
            zt = z - alpha * d
            p = softmax_from_z(zt)
            if p is None:
                continue

            P2 = np.fft.rfft(p, N)
            C2 = np.fft.irfft(P2 * P2, N)[:L]
            C2 = np.nan_to_num(C2, nan=0.0, posinf=0.0, neginf=0.0)
            C2 = np.where(C2 > 0.0, C2, 0.0)
            cmax2 = float(C2.max())
            Cn2 = C2 * Lf
            mCn2 = float(Cn2.max())

            zrel2 = (Cn2 - mCn2) / tau
            zrel2 = np.nan_to_num(zrel2, nan=-100.0, posinf=0.0, neginf=-100.0)
            zm2 = float(zrel2.max())
            if zm2 > 0.0:
                zrel2 = zrel2 - zm2
            s2 = float(np.exp(zrel2).sum())
            if np.isfinite(s2) and s2 > 0.0:
                val2 = mCn2 + tau * (float(np.log(s2)) - logL)
            else:
                val2 = mCn2

            if (
                cmax2 < best_trial_cmax - 1e-16
            ) or (
                val2 < best_val - 1e-14 and cmax2 <= allow_cmax
            ):
                best_trial_cmax = cmax2
                best_val = val2
                best_alpha = alpha
                best_z = zt
                best_p = p
                if cmax2 < best_cmax - 1e-15:
                    best_cmax = cmax2
                    best_a = p.copy()
                    last_imp = time.monotonic()
                if cmax2 < cmax * 0.999:
                    break

        if best_alpha != 0.0:
            eta = min(eta * 1.12, float(eta_max))
            z = best_z.copy()
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            if a is None:
                a = best_a.copy()
                z = np.log(np.maximum(a, 1e-30))
                z -= float(np.mean(z))
        else:
            eta = max(eta * 0.80, float(eta_min))

        if time.monotonic() - last_imp > 12.0:
            z_base = np.log(np.maximum(best_a, 1e-30))
            z_base -= float(np.mean(z_base))
            noise = 0.018 + 0.035 * min(2.5, (time.monotonic() - last_imp) / 20.0)
            z = z_base + rng.normal(0.0, noise, n)
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            if a is None:
                a = best_a.copy()
            eta = max(eta, 0.06)
            last_imp = time.monotonic()

        if verbose and (it % 250 == 0 or time.monotonic() - last_imp > 30.0):
            print(
                "    sumexp {0:4d} t={1:5.0f}s score={2:.6f} tau={3:.3f} eta={4:.4f}".format(
                    it,
                    time.monotonic() - t0,
                    2.0 * n * best_cmax,
                    tau,
                    eta,
                )
            )

    return best_a


def smooth_obj(z, tau, N, L):
    if time.monotonic() > _LB_DEADLINE:
        raise _LbfgsTimeout()

    p = softmax_from_z(z)
    if p is None:
        return 1e100

    P = np.fft.rfft(p, N)
    C = np.fft.irfft(P * P, N)[:int(L)]
    C = np.nan_to_num(C, nan=0.0, posinf=0.0, neginf=0.0)
    C = np.where(C > 0.0, C, 0.0)

    Cn = C * float(L)
    mCn = float(Cn.max())
    zrel = (Cn - mCn) / float(tau)
    zrel = np.nan_to_num(zrel, nan=-100.0, posinf=0.0, neginf=-100.0)
    zm = float(zrel.max())
    if zm > 0.0:
        zrel = zrel - zm
    s = float(np.exp(zrel).sum())
    if not np.isfinite(s) or s <= 0.0:
        return float(mCn)
    val = mCn + float(tau) * (float(np.log(s)) - np.log(float(L)))
    if not np.isfinite(val):
        return 1e100
    return float(val)


def smooth_grad(z, tau, N, L):
    if time.monotonic() > _LB_DEADLINE:
        raise _LbfgsTimeout()

    p = softmax_from_z(z)
    if p is None:
        return np.zeros_like(z, dtype=np.float64)

    P = np.fft.rfft(p, N)
    C = np.fft.irfft(P * P, N)[:int(L)]
    C = np.nan_to_num(C, nan=0.0, posinf=0.0, neginf=0.0)
    C = np.where(C > 0.0, C, 0.0)

    Cn = C * float(L)
    mCn = float(Cn.max())
    zrel = (Cn - mCn) / float(tau)
    zrel = np.nan_to_num(zrel, nan=-100.0, posinf=0.0, neginf=-100.0)
    zm = float(zrel.max())
    if zm > 0.0:
        zrel = zrel - zm
    e = np.exp(zrel)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros_like(p, dtype=np.float64)

    v = e / s
    W = np.fft.rfft(v, N)
    corr = np.fft.irfft(np.conj(P) * W, N)[:len(p)]
    corr = np.nan_to_num(corr, nan=0.0, posinf=0.0, neginf=0.0)
    corr = np.where(corr > 0.0, corr, 0.0)

    jac_a = 2.0 * float(L) * corr
    dot = float(np.dot(p, jac_a))
    jac_z = jac_a - p * dot
    jac_z = np.nan_to_num(jac_z, nan=0.0, posinf=0.0, neginf=0.0)
    jac_z = jac_z - float(np.mean(jac_z))
    return jac_z


def run_lbfgs_smooth(a0, tau, time_allow, maxiter=350):
    global _LB_DEADLINE, _LB_LATEST

    arr = normalize_positive(a0)
    if arr is None:
        arr = np.ones(1)
    n = len(arr)
    L = 2 * n - 1
    N = next_pow2(L)

    z0 = np.log(np.maximum(arr, 1e-30))
    z0 -= float(np.mean(z0))
    a0s = softmax_from_z(z0)
    if a0s is None:
        return arr

    _LB_DEADLINE = time.monotonic() + max(1e-3, float(time_allow) * 0.96)
    _LB_LATEST = z0.copy()

    try:
        res = scipy_optimize.minimize(
            smooth_obj,
            z0,
            jac=smooth_grad,
            args=(tau, N, L),
            method="L-BFGS-B",
            options={
                "maxiter": maxiter,
                "ftol": 1e-12,
                "gtol": 1e-7,
                "maxcor": 20,
                "maxls": 20,
            },
            callback=_lb_latest_callback,
        )
        x = res.x if res is not None and getattr(res, "x", None) is not None else _LB_LATEST
    except _LbfgsTimeout:
        x = _LB_LATEST
    except Exception as exc:
        print("    lbfgs err tau={0:.4f}: {1}".format(tau, exc))
        x = _LB_LATEST

    if x is None:
        x = z0
    p = softmax_from_z(x)
    if p is None:
        return arr
    return p


def propose_candidate():

        best = make_default_sequence(5200)

    if best is None:

    n = len(best)
    print(

    )

    rng = np.random.default_rng(2024)

    # If no good parent exists, spend some time on a few independent starts.
    if parent is None or best_score > 1.55:
        remaining = max(0.0, deadline - time.monotonic())
        if remaining > 280:
            inits = [
                make_default_sequence(n),
                make_shape_exponential(n),
                make_shape_cosine(n),
                make_shape_random(n, seed=31),
            ]
            used = 0.0
            for idx, init in enumerate(inits):
                rem = max(0.0, deadline - time.monotonic())
                if rem < 80:
                    break
                alloc = min(250.0, rem - 100.0)
                if alloc < 35:
                    break
                print("  no-good-parent start {0} alloc={1:.0f}s".format(idx, alloc))
                a_new = run_sumexp(
                    init,
                    alloc,
                    tau_start=2.5,
                    tau_end=0.12,
                    eta_init=0.18,
                    eta_max=1.0,
                    rng=rng,
                    verbose=False,
                )
                sc = score_fast(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best = normalize_positive(a_new)
                    best_score = sc
                    print("   new best {0:.6f}".format(best_score))
                used += (time.monotonic() - t_start)

    # Main soft maximization optimizer on the current best sequence.
    rem = max(0.0, deadline - time.monotonic())
    t_main = min(rem * 0.48, 820.0)

        print("  main sumexp allowable={0:.0f}s n={1}".format(t_main, len(best)))

            tau_start=3.0,

            eta_max=1.0,
            rng=rng,
            verbose=True,
        )
        sc = score_fast(a_new)
        if np.isfinite(sc) and sc < best_score:
            best = normalize_positive(a_new)
            best_score = sc
            print("  main sumexp improved to {0:.6f}".format(best_score))
        else:
            print("  main sumexp no exact improvement; best remains {0:.6f}".format(best_score))

    # L-BFGS polishing of the smooth surrogate.  Multiple temperatures are used.
    taus_setup = [0.70, 0.34, 0.18, 0.10, 0.06]
    for tau in taus_setup:
        rem = max(0.0, deadline - time.monotonic())
        if rem < 75:
            break
        alloc = min(120.0, rem - 70.0)
        if alloc < 25:
            break
        print("  lbfgs tau={0:.3f} alloc={1:.0f}s".format(tau, alloc))
        a_new = run_lbfgs_smooth(best, tau, alloc, maxiter=350)
        sc = score_fast(a_new)
        if np.isfinite(sc) and sc < best_score:
            best = normalize_positive(a_new)
            best_score = sc
            print("   lbfgs new best {0:.6f}".format(best_score))

    # Final short continuous polishing with tighter temperature.
    rem = max(0.0, deadline - time.monotonic())
    if rem > 110:
        t_pol = min(rem - 75.0, 170.0)
        if t_pol > 30:
            print("  final sumexp polish alloc={0:.0f}s".format(t_pol))
            a_new = run_sumexp(

                t_pol,
                tau_start=0.23,
                tau_end=0.05,
                eta_init=0.07,
                eta_max=0.5,
                rng=rng,
                verbose=True,
            )
            sc = score_fast(a_new)
            if np.isfinite(sc) and sc < best_score:
                best = normalize_positive(a_new)
                best_score = sc
                print("  final polish improved to {0:.6f}".format(best_score))

    # Optional length migration.  Larger n can lower the limiting smooth
    # objective, but only if enough time remains to re-optimize.
    rem = max(0.0, deadline - time.monotonic())
    if rem > 220 and len(best) <= 8000:
        cand = []
        base_n = len(best)
        for delta in [1000, 1800, 600, -800]:
            nn = base_n + delta
            if 2200 <= nn <= 8000 and nn != base_n:
                cand.append(nn)
        cand = list(dict.fromkeys(cand))
        for nn in cand[:2]:

            rem = max(0.0, deadline - time.monotonic())
            if rem > 70:

                    if np.isfinite(sc) and sc < best_score:
                        best = normalize_positive(a_new)
                        best_score = sc
                        print("  length {0} lbfgs improved to {1:.6f}".format(nn, best_score))

    # Very last tiny polish, if anything is left.
    rem = max(0.0, deadline - time.monotonic())
    if rem > 55:
        t_last = min(rem - 30.0, 60.0)
        if t_last > 15:
            a_new = run_sumexp(
                best,
                t_last,
                tau_start=0.12,
                tau_end=0.04,
                eta_init=0.04,
                eta_max=0.3,
                rng=rng,
                verbose=False,
            )
            sc = score_fast(a_new)
            if np.isfinite(sc) and sc < best_score:
                best = normalize_positive(a_new)
                best_score = sc

    best = normalize_positive(best)
    out = [float(max(0.0, min(1000.0, x))) for x in best]
    ssum = sum(out)
    if ssum < 0.01:
        m = len(out)
        if m <= 0:
            out = [1.0]
        else:
            out = [1.0 / float(m)] * m
            ssum = 1.0

    exact = float("inf")
    try:
        exact = float(evaluate_sequence(out))
    except Exception:
        exact = float("inf")

    if np.isfinite(exact) and exact < best_score:
        best_score = exact

    print(
        "FINAL score={0:.6f} n={1} time={2:.0f}s exact_call={3:.6f}".format(
            best_score, len(out), time.monotonic() - t_start, exact
        )
    )
    return out
