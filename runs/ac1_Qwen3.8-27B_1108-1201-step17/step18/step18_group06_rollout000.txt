    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def peak_gradient(a, ac, n_active=5):
    """
    Compute subgradient using the top n_active peaks.
    For peak k: d(conv[k])/d(a[i]) = 2*a[k-i] for i in valid range (simplified)
    """
    n = len(a)
    # Find top peaks
    top_idx = np.argsort(ac)[::-1][:n_active]
    # Weight by prominence (how much above the median)
    ac_median = np.median(ac)
    weights = np.zeros(n_active)
    for i, k in enumerate(top_idx):
        ki = int(k)
        weights[i] = max(0, ac[ki] - ac_median)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(n_active) / n_active
    else:
        weights = weights / wsum
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        ki = int(k)
        w = weights[i]
        if w < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        # grad[idx] += 2 * w * a[pair_idx]
        # But we need to be careful: d(conv[k])/d(a[i]) = a[k-i] + a[k-i] (if i != k-i)
        # For simplicity, use 2*a[k-i] for all (the factor of 2 at the diagonal is minor)
        grad[idx] += 2.0 * w * a[pair_idx]
    return grad

def grad_descent_focused(a0, time_limit, n_active=3, eta0=0.05, verbose=False, rng_seed=42):
    """
    Focused peak-chasing gradient descent with multiplicative updates.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    eta = eta0
    x_mom = a.copy()  # momentum
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        g = peak_gradient(a, ac, n_active=n_active)
        
        # Multiplicative update (mirror descent on simplex)
        # a_new = a * exp(-eta * g) / sum
        log_a = np.log(np.maximum(a, 1e-30)) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Momentum blend
        x_mom = 0.7 * x_mom + 0.3 * a_new
        x_mom = np.maximum(x_mom, 1e-30)
        x_mom = x_mom / x_mom.sum()
        
        a = x_mom
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Decrease learning rate when stalled
            if stall % 100 == 0:
                eta *= 0.9
                # Random perturbation
                if stall > 500:
                    pert = rng.normal(0, 0.05, n) * a
                    a = a + pert
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()
                    best_m = min(best_m, fast_autocorr(a).max())
                    if fast_autocorr(a).max() < best_m:
                        best_m = fast_autocorr(a).max()
                        best_a = a.copy()
                        stall = 0
        
        # Adapt n_active: start focused, gradually widen
        n_active = min(3 + t // 5000, 20)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  FOCUS t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, eta={eta:.4f}, peaks={n_active}")
    
    return best_a, best_m

def spectral_flatten(a0, time_limit, alpha0=0.3, verbose=False):
    """Gerchberg-Saxton flat spectrum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    it_count = 0
    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, fft_size)
        A2 = A * A
        P = np.abs(A2)
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        alpha = alpha0 + (1.0-alpha0) * min(1.0, it_count / 3000.0)
        P_target = P_mean * (P / np.maximum(P_mean, 1e-30)) ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)
        
        phase = np.angle(A2)
        ratio = np.sqrt(P_target / np.maximum(P, 1e-30))
        ratio = np.clip(ratio, 0.75, 1.3)
        A2_new = A2 * ratio
        A_new = np.sqrt(A2_new)  # This is problematic for complex numbers
        # Actually, we can't take sqrt of complex easily. Use phase-preserving approach:
        A_new = A * np.clip(np.sqrt(P_target / np.maximum(np.abs(A)**2, 1e-30)), 0.8, 1.2)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.2
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()
        
        it_count += 1
        if verbose and it_count % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  SPECTRAL it={it_count} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def lp_projection(f, time_limit, verbose=False):
    """
    LP-based projection from SOTA:
    Maximize sum(b) subject to b >= 0 and ||f*b||_inf <= M
    where M = ||f*f||_inf
    """
    n = len(f)
    M = fast_autocorr(f).max()
    
    # Build constraint matrix A_con: (2n-1) x n
    # A_con[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., (f*b)[k] = sum_j f[j]*b[k-j]... wait, that's not right.
    # conv(f, b)[k] = sum_j f[j] * b[k-j]
    # For this to be valid, we need 0 <= j < n and 0 <= k-j < n
    # So j is in [max(0, k-n+1), min(k, n-1)]
    # A_con[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1), else 0
    
    # Actually, let me use scipy's fftconvolve to check, but for the LP we need the matrix.
    # The matrix is 2n-1 x n. For n=4200, that's 8399 x 4200.
    # We can use a sparse representation or just compute it.
    
    # Actually, for n=4200, the dense matrix would be 8399*4200*8 bytes ≈ 283 MB.
    # That's a lot. Let me use scipy sparse if available, or just do a smaller version.
    
    # Alternative: use a smaller effective n for the LP
    # Or use scipy.optimize.linprog with a callback
    
    # For now, let's try with the full size but use the 'highs' solver
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp
    
    # Build the matrix in sparse form
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A_con = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: maximize sum(b) s.t. A_con @ b <= M, b >= 0
    # = minimize -sum(b) s.t. A_con @ b <= M, b >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A_con, b_ub=np.full(2 * n - 1, M), bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        b_lp = res.x
        b_lp = np.maximum(b_lp, 0)
        s = b_lp.sum()
        if s < 1e-10:
            return f.copy()
        b_ls = b_lp / s
        return b_ls
    return f.copy()

def make_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)

def make_sine_ramp(n, alpha=0.5):
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * x + 0.1 * np.sin(2 * np.pi * x)
    a = np.maximum(a, 0.1)
    return norm_sum1(a)

def make_log_ramp(n):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.log(x)
    return norm_sum1(a)
