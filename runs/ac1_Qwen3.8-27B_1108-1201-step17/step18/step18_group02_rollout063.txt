"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-method ensemble:
1. Adaptive p-norm smoothed multiplicative (Fourier Transform) optimizer
2. Frequency-domain peak-shaving (spectral flattening)
3. Targeted coordinate descent with pair-transfer optimization
4. Multi-seed exploration with periodic restarts
5. Multi-length interpolation and re-optimization
"""

import numpy as np
import time
import math

def fast_conv_self(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def pnorm_gradient(a, c, p):
    """Compute gradient of (sum c^p)^(1/p) w.r.t. a.
    
    Returns unnormalized gradient (scale doesn't matter for FT updates).
    """
    n = len(a)
    # w[k] = c[k]^(p-1) for k = 0..2n-2
    w = np.maximum(c, 0.0) ** (p - 1)
    # grad[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = conv(a_flip, w)[i + n - 1]
    a_flip = a[::-1].copy()
    size = n + (2 * n - 1) - 1  # 3n - 2
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Wf = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(Af * Wf, fft_size)[:size]
    return conv[n - 1:n - 1 + n]

def ft_update(a, grad, lr):
    """Multiplicative (Fourier Transform) update."""
    log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def spectral_flatten(a, alpha=0.5, iters=3):
    """Frequency-domain peak shaving."""
    n = len(a)
    a = a.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for _ in range(iters):
        F = np.fft.rfft(a, fft_size)
        P = F.real**2 + F.imag**2  # |F|^2
        P_mean = P.mean()
        # Flatten: reduce peaks, boost troughs
        P_new = alpha * P + (1 - alpha) * P_mean
        P_new = np.maximum(P_new, 1e-30)
        # Reconstruct with same phase
        magnitude = np.sqrt(P_new)
        phase = F / np.maximum(np.abs(F), 1e-30)
        F_new = magnitude * phase
        a_new = np.fft.irfft(F_new, fft_size)[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = a_new = np.maximum(a_new, 0.0)
            s = a.sum()
        a = a / s
    
    return a

def coord_descent_targeted(a, c, top_k=10, n_trials=50):
    """Targeted coordinate descent: move mass to reduce top peaks."""
    n = len(a)
    a = a.copy()
    best_m = c.max()
    best_a = a.copy()
    rng = np.random.default_rng(12345)
    
    for trial in range(n_trials):
        # Find top peaks
        top_indices = np.argsort(c)[::-1][:top_k]
        
        for k in top_indices[:5]:
            k_int = int(k)
            # Positions that contribute to c[k_int]: pairs (i, k_int-i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            # Find the pair with maximum contribution
            js = np.arange(i_start, i_end + 1)
            contrib = a[js] * a[k_int - js]
            max_pos = js[np.argmax(contrib)]
            
            if a[max_pos] < 1e-10:
                continue
            
            # Find a good destination: position that minimizes impact on top peaks
            # Simple heuristic: position far from all top peak contributions
            scores = np.zeros(n)
            for k2 in top_indices[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            
            # Pick destination with lowest score
            dst = np.argmin(scores)
            if dst == max_pos:
                dst = (dst + 1) % n
            
            # Try to move some mass
            frac = rng.uniform(0.01, 0.08)
            amt = a[max_pos] * frac
            a_trial = a.copy()
            a_trial[max_pos] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            
            c_trial = fast_conv_self(a_trial)
            m_trial = c_trial.max()
            
            if m_trial < best_m:
                best_m = m_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                c = c_trial
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def generate_inits(n, parent=None, rng_seed=0):
    """Generate diverse initial sequences of length n."""
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    inits = []
    rng = np.random.default_rng(rng_seed)
    
    # Sine-squared variants
    for freq in [1, 1.5, 2, 2.5, 3]:
        for phase in [0, 0.2, 0.4]:
            a = np.maximum(np.sin(np.pi * freq * x + phase * np.pi) ** 2, 0.01)
            inits.append(a / a.sum())
    
    # Cosine taper
    a = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(a / a.sum())
    
    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.15, 0.25, 0.4]:
            a = np.exp(-((x - c) ** 2) / (2 * sigma ** 2))
            inits.append(a / a.sum())
    
    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = (i + 1) ** p
        inits.append(a / a.sum())
        a = (n - i) ** p
        inits.append(a / a.sum())
    
    # Alternating (high-low pattern)
    for ratio in [2, 3, 5]:
        a = np.ones(n)
        a[::2] = ratio
        inits.append(a / a.sum())
    
    # If parent exists, add perturbed versions
    if parent is not None:
        p_parent = interpolate_sequence(parent, n)
        inits.append(p_parent.copy())
        for seed in [1, 2, 3]:
            r = np.random.default_rng(seed)
            pert = r.normal(0, 0.05, n) * p_parent
            a = p_parent + pert
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            inits.append(a)
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-method ensemble:
1. Adaptive p-norm smoothed multiplicative (FT) optimizer with log-space stability
2. Frequency-domain spectral flattening (peak-shaving in power spectrum)
3. Targeted coordinate descent with pair-transfer on top autocorrelation peaks
4. Multi-seed exploration with diverse parametric initializations
5. Multi-length interpolation and re-optimization
"""

import numpy as np
import time
import math
from operator import itemgetter


def fast_conv_self(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def pnorm_grad(a, c, p):
    """Compute gradient of (sum c^p)^(1/p) w.r.t a, using normalized c for stability."""
    n = len(a)
    max_c = c.max()
    if max_c < 1e-30:
        return np.zeros(n)
    c_norm = c / max_c
    w = c_norm ** (p - 1.0)
    # grad[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = conv(a_flip, w)[i + n - 1]
    a_flip = a[::-1].copy()
    w_len = len(w)
    size = n + w_len - 1
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Wf = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(Af * Wf, fft_size)[:size]
    return conv[n - 1:n - 1 + n]


def ft_update(a, grad, lr):
    """Multiplicative Fourier Transform update (preserves non-negativity)."""
    log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_flatten(a, alpha, n_iters):
    """Frequency-domain peak shaving: flatten the power spectrum."""
    n = len(a)
    a = a.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for _ in range(n_iters):
        F = np.fft.rfft(a, fft_size)
        mag_sq = F.real ** 2 + F.imag ** 2
        mag_mean = mag_sq.mean()
        if mag_mean < 1e-30:
            break
        mag_sq_new = alpha * mag_sq + (1.0 - alpha) * mag_mean
        mag_sq_new = np.maximum(mag_sq_new, 1e-30)
        new_mag = np.sqrt(mag_sq_new)
        # Preserve phase
        abs_F = np.abs(F)
        phase = np.where(abs_F > 1e-30, F / abs_F, 1.0)
        F_new = new_mag * phase
        a_new = np.fft.irfft(F_new, fft_size)[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.maximum(a_new, 0.0)
            s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
    return a


def coord_descent_tc(a, c, top_k, n_trials, rng):
    """Targeted coordinate descent: move mass to reduce top autocorrelation peaks."""
    n = len(a)
    a = a.copy()
    best_m = c.max()
    best_a = a.copy()

    for _ in range(n_trials):
        top_idx = np.argsort(c)[::-1][:top_k]

        for k_idx in top_idx[:6]:
            k_int = int(k_idx)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            js = np.arange(i_start, i_end + 1)
            contrib = a[js] * a[k_int - js]
            if contrib.max() < 1e-20:
                continue

            src = js[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue

            # Score destinations: lower score = less impact on top peaks
            scores = np.zeros(n)
            for k2 in top_idx[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            dst = int(np.argmin(scores))
            if dst == src:
                dst = (dst + 1) % n

            frac = rng.uniform(0.005, 0.06)
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv_self(a_trial).max()

            if m_trial < best_m:
                best_m = m_trial
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_m


def smooth_pnorm_opt(a0, time_limit, p_start, p_end, lr, verbose, rng_seed=42):
    """Adaptive p-norm smoothed FT optimizer."""
    n = len(a0)
    a = norm_sum1(a0)
    c = fast_conv_self(a)
    best_a = a.copy()
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    p = float(p_start)
    restart_every = 1500

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M = c.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        grad = pnorm_grad(a, c, p)
        grad = np.maximum(grad, 0.0)

        effective_lr = lr / (1.0 + 0.05 * p)
        a = ft_update(a, grad, effective_lr)

        # Periodic spectral flattening
        if t % 80 == 0:
            a = spectral_flatten(a, alpha=0.6, n_iters=2)

        # Grow p geometrically
        p = min(p * 1.002, p_end)

        # Recompute c
        if t % 3 == 0 or t < 10:
            c = fast_conv_self(a)

        # Restart if stalled
        if stall > restart_every:
            stall = 0
            p = p_start
            pert = rng.normal(0, 0.03, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            c = fast_conv_self(a)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            c = fast_conv_self(a)
            best_m = min(best_m, c.max())
            print("      pnorm t={} ({:.0f}s): J={:.6f}, p={:.1f}, stall={}".format(
                t, elapsed, 2.0 * n * best_m, p, stall))

    elapsed = time.monotonic() - t0
    c = fast_conv_self(best_a)
    best_m = c.max()
    if verbose:
        print("      pnorm done ({:.0f}s): J={:.6f}".format(elapsed, 2.0 * n * best_m))
    return best_a, best_m


def combined_opt(a0, time_limit, verbose, rng_seed=777):
    """Combined optimizer: alternates between pnorm, spectral, and coord methods."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv_self(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    cycle_idx = 0
    cycle_len = max(30.0, time_limit * 0.08)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        phase = cycle_idx % 3
        burst = min(cycle_len, remaining * 0.4)

        if phase == 0:
            # P-norm smoothed FT
            if burst > 5:
                a_out, m_out = smooth_pnorm_opt(a, burst, p_start=6.0, p_end=40.0,
                                                lr=0.7, verbose=False, rng_seed=rng_seed + cycle_idx)
                if m_out < best_m:
                    best_m = m_out
                    best_a = a_out.copy()
                a = a_out

        elif phase == 1:
            # Spectral flattening with perturbation
            steps = max(10, int(burst / 0.3))
            for _ in range(steps):
                if time.monotonic() - t0 > time_limit - 5:
                    break
                alpha = rng.uniform(0.4, 0.85)
                a_new = spectral_flatten(a, alpha=alpha, n_iters=3)
                m_new = fast_conv_self(a_new).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    a = a_new
                # Small perturbation to escape
                if _ % 10 == 0:
                    pert = rng.normal(0, 0.008, n) * a
                    a = a + pert
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()

        else:
            # Coordinate descent
            c_curr = fast_conv_self(a)
            trials = max(20, int(burst / 0.5))
            a_new, m_new = coord_descent_tc(a, c_curr, top_k=12, n_trials=trials, rng=rng)
            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()
            a = a_new

        cycle_idx += 1

        if verbose and cycle_idx % 3 == 0:
            elapsed = time.monotonic() - t0
            print("    Combined ({:.0f}s): J={:.6f}, cycle={}".format(
                elapsed, 2.0 * n * best_m, cycle_idx))

    c_final = fast_conv_self(best_a)
    best_m = c_final.max()
    if verbose:
        elapsed = time.monotonic() - t0
        print("    Combined done ({:.0f}s): J={:.6f}".format(elapsed, 2.0 * n * best_m))
    return best_a, best_m


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return norm_sum1(f)
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def gen_inits(n, parent, rng_seed):
    """Generate diverse initial sequences of length n."""
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    inits = []

    # Sine-squared family
    for freq in [1.0, 1.5, 2.0, 2.5, 3.0]:
        for phase in [0.0, 0.15, 0.3]:
            a = np.maximum(np.sin(np.pi * freq * x + phase * np.pi) ** 2, 0.01)
            inits.append(a / a.sum())

    # Cosine taper
    a = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(a / a.sum())

    # Gaussian bumps
    for c in [0.25, 0.4, 0.5, 0.6, 0.75]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            inits.append(a / a.sum())

    # Power law
    for p in [0.3, 0.7, 1.0, 1.5, 2.0]:
        a = (i + 1.0) ** p
        inits.append(a / a.sum())
        a = (n - i) ** p
        inits.append(a / a.sum())

    # Alternating
    for ratio in [2.0, 3.0, 5.0]:
        a = np.ones(n)
        a[::2] = ratio
        inits.append(a / a.sum())

    # Bilinear / tent
    a = np.minimum(x, 1.0 - x) * 2.0
    inits.append(a / a.sum())

    # If parent exists, add its interpolation and perturbations
    if parent is not None:
        p_par = interpolate_seq(parent, n)
        inits.append(p_par)
        for seed in [1, 2, 3, 4, 5]:
            r = np.random.default_rng(seed * 31)
            pert = r.normal(0, 0.04, n) * p_par
            a = p_par + pert
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            inits.append(a)
        # Smoothed parent
        from numpy import convolve as npconv
        kernel = np.ones(21) / 21.0
        a_smooth = npconv(p_par, kernel, mode='same')
        a_smooth = np.maximum(a_smooth, 0.0)
        a_smooth = a_smooth / a_smooth.sum()
        inits.append(a_smooth)

    return inits


def quick_score(a):
    """Quick objective computation assuming sum(a)=1."""
    n = len(a)
    c = fast_conv_self(a)
    return 2.0 * n * c.max()
