```python
"""
Searches for a non-negative coefficient sequence minimizing

    2*n*max(conv(a,a))/(sum(a)**2)

using a hybrid of:
  - resampling to attractive power-of-two lengths,
  - a small-grid global shape search,
  - top-lag mirrored multiplicative gradient descent,
  - full-lag variance equalization,
  - smoothed Frank-Wolfe style directions,
  - exact coordinate moves for final polishing.

The search is time-bounded and always returns a valid list of floats.
"""

import numpy as np
import time


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    a = np.maximum(a, 0.0)
    n = max(1, int(a.size))
    if floor > 0:
        a = np.maximum(a, floor)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def fft_conv_same(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    out_len = 2 * n - 1
    fsize = next_pow2(out_len)
    A = np.fft.rfft(a, fsize)
    C = np.fft.irfft(A * A, fsize)[:out_len]
    C = np.nan_to_num(C, nan=0.0, posinf=0.0, neginf=0.0)
    np.maximum(C, 0.0, out=C)
    return C


def internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf")
    p = a / s
    C = fft_conv_same(p)
    return float(2 * p.size * float(C.max()))


def resample(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    old_n = int(a.size)
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([], dtype=np.float64)
    if old_n <= 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    if old_n == new_n:
        return normalize_positive(a)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    p = a / s
    old_x = (np.arange(old_n) + 0.5) / float(old_n)
    new_x = (np.arange(new_n) + 0.5) / float(new_n)
    f = np.interp(new_x, old_x, p)
    f = np.maximum(f, 0.0)
    fs = float(f.sum())
    if fs <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f / fs


def make_beta_density(n, alpha, beta):
    n = int(n)
    x = (np.arange(n) + 0.5) / float(n)
    f = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return f / s


def make_exp_density(n, lam):
    n = int(n)
    x = (np.arange(n) + 0.5) / float(n)
    z = np.clip(lam * x, -50.0, 50.0)
    f = np.exp(z)
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return f / s


def random_smooth(n, seed=0, modes=60):
    n = int(n)
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) / float(n)
    f = np.ones(n, dtype=np.float64) * 0.8
    for m in range(1, int(modes) + 1):
        amp = 0.22 / np.sqrt(float(m))
        phase1 = rng.uniform(0.0, 2.0 * np.pi)
        phase2 = rng.uniform(0.0, 2.0 * np.pi)
        f += amp * rng.normal(loc=0.0, scale=1.0) * np.sin(2.0 * np.pi * m * x + phase1)
        f += 0.5 * amp * rng.normal(loc=0.0, scale=1.0) * np.cos(2.0 * np.pi * m * x + phase2)
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return f / s


def select_top_weights(C, K, beta):
    C = np.asarray(C, dtype=np.float64)
    L = int(C.size)
    K = int(K)
    if K <= 0 or L == 0:
        return np.array([], dtype=np.int64), np.array([], dtype=np.float64)
    K = min(K, L)
    if K >= L:
        return np.arange(L, dtype=np.int64), np.ones(L, dtype=np.float64) / L

    idx = np.argpartition(C, L - K)[L - K:]
    vals = C[idx]
    maxv = float(vals.max())
    if maxv <= 1e-300:
        w = np.ones(K, dtype=np.float64) / K
        return idx.astype(np.int64), w

    ratio = np.maximum(vals, 0.0) / maxv
    w = ratio ** float(beta)
    s = float(w.sum())
    if s <= 1e-300:
        w = np.ones(K, dtype=np.float64) / K
    else:
        w = w / s
    return idx.astype(np.int64), w.astype(np.float64)


def correlate_top(a, idx, w):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    g = np.zeros(n, dtype=np.float64)
    if idx.size == 0:
        return g
    for kk, ww in zip(idx, w):
        if ww == 0.0:
            continue
        k = int(kk)
        if k < 0 or k >= 2 * n - 1:
            continue
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        if s >= e:
            continue
        lo = k - e + 1
        hi = k - s + 1
        g[s:e] += float(ww) * a[lo:hi][::-1]
    return g


def correlate_dense(a, q):
    a = np.asarray(a, dtype=np.float64)
    q = np.asarray(q, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * q[0])], dtype=np.float64)
    L = 2 * n - 1
    q_rev = q[::-1]
    total = n + L - 1
    fsize = next_pow2(total)
    A = np.fft.rfft(a, fsize)
    Q = np.fft.rfft(q_rev, fsize)
    conv = np.fft.irfft(A * Q, fsize)[:total]
    return conv[n - 1:2 * n - 1][::-1].copy()


def log_step(a, g_t, alpha, scale, floor):
    f = -float(alpha) * (g_t / scale)
    f = np.clip(f, -40.0, 40.0)
    b = a * np.exp(f)
    b = np.maximum(b, floor)
    s = float(b.sum())
    if s <= 1e-300:
        return a.copy()
    return b / s


def make_simplex_direction(g, gamma):
    n = int(g.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    gt = g - g.mean()
    scale = float(np.percentile(np.abs(gt), 90))
    if scale <= 1e-300:
        scale = 1.0
    z = -float(gamma) * (gt / scale)
    z = z - z.max()
    d = np.exp(z)
    d = 0.995 * d + 0.005 * (1.0 / n)
    s = float(d.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return d / s


def optimizer_mirror(a0, time_limit, K=120, beta=8.0,
                     alphas=(0.05, 0.2, 0.5, 1.0, 2.0, 5.0),
                     seed=0, verbose=False, soft_all=False, soft_temp_ratio=0.04):
    n = int(np.asarray(a0, dtype=np.float64).size)
    if n <= 0:
        return np.array([], dtype=np.float64), float("inf"), np.array([], dtype=np.float64)
    time_limit = float(time_limit)
    if time_limit <= 0.0:
        a = normalize_positive(a0)
        C = fft_conv_same(a)
        M = float(C.max())
        return a.copy(), M, a

    floor = 1e-12 / max(1, n)
    a = normalize_positive(a0, floor=floor)
    C = fft_conv_same(a)
    M = float(C.max())
    best_a = a.copy()
    best_M = M

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    cur_K = int(K)
    cur_beta = float(beta)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        L = 2 * n - 1
        if soft_all:
            maxC = float(C.max())
            temp = max(maxC * float(soft_temp_ratio), 1e-300)
            q = np.exp((C - maxC) / temp)
            qs = float(q.sum())
            if qs <= 1e-300:
                q = np.ones(L, dtype=np.float64) / L
            else:
                q = q / qs
            g = correlate_dense(a, q)
        else:
            idx, w = select_top_weights(C, min(cur_K, L - 1 if L > 1 else 1), cur_beta)
            if idx.size == 0:
                idx = np.arange(L, dtype=np.int64)
                w = np.ones(L, dtype=np.float64) / L
            g = correlate_top(a, idx, w)

        g_t = g - g.mean()
        scale = float(np.percentile(np.abs(g_t), 90))
        if scale <= 1e-300:
            scale = 1.0

        best_trial_M = float("inf")
        best_trial_a = None
        best_trial_C = None

        for alpha in alphas:
            b = log_step(a, g_t, float(alpha), scale, floor)
            Cb = fft_conv_same(b)
            Mb = float(Cb.max())
            if not np.isfinite(Mb):
                continue
            if Mb < best_trial_M:
                best_trial_M = Mb
                best_trial_a = b
                best_trial_C = Cb

        if best_trial_a is None:
            break

        allow = 1.0
        if stall > 30:
            allow = 1.0 + 1e-5
        if stall > 120:
            allow = 1.0 + 1e-4

        if best_trial_M <= M * allow:
            a = best_trial_a
            C = best_trial_C
            M = best_trial_M
            if M < best_M:
                best_M = M
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 0 and stall % 250 == 0:
            cur_beta = min(cur_beta * 1.25, 60.0)
            cur_K = min(cur_K + 20, max(200, n // 8))

        if stall > 900:
            a = best_a + rng.normal(loc=0.0, scale=0.02, size=n) * best_a
            a = normalize_positive(a, floor=floor)
            C = fft_conv_same(a)
            M = float(C.max())
            cur_beta = float(beta)
            cur_K = int(K)
            stall = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      mirror n={n} it={it} t={elapsed:.1f}s score={2.0*n*best_M:.6f} best_trial={2.0*n*best_trial_M:.6f}")

    return best_a, best_M, a

def optimizer_fw(a0, time_limit, K=120, beta=8.0,
                 gammas=(5.0, 20.0, 80.0, 300.0, 1000.0),
                 alphas=(0.01, 0.02, 0.05, 0.1, 0.2, 0.5),
                 seed=0, verbose=False):
    n = int(np.asarray(a0, dtype=np.float64).size)
    if n <= 0:
        return np.array([], dtype=np.float64), float("inf"), np.array([], dtype=np.float64)
    time_limit = float(time_limit)
    if time_limit <= 0.0:
        a = normalize_positive(a0)
        C = fft_conv_same(a)
        M = float(C.max())
        return a.copy(), M, a

    floor = 1e-12 / max(1, n)
    a = normalize_positive(a0, floor=floor)
    C = fft_conv_same(a)
    M = float(C.max())
    best_a = a.copy()
    best_M = M

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    cur_K = int(K)
    cur_beta = float(beta)
    it = 0
    L = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        idx, w = select_top_weights(C, min(cur_K, L - 1 if L > 1 else 1), cur_beta)
        if idx.size == 0:
            idx = np.arange(L, dtype=np.int64)
            w = np.ones(L, dtype=np.float64) / L
        g = correlate_top(a, idx, w)

        best_trial_M = float("inf")
        best_trial_a = None
        best_trial_C = None

        for gamma in gammas:
            d = make_simplex_direction(g, gamma)
            for alpha in alphas:
                b = (1.0 - float(alpha)) * a + float(alpha) * d
                b = np.maximum(b, floor)
                s = float(b.sum())
                if s <= 1e-300:
                    continue
                b = b / s
                Cb = fft_conv_same(b)
                Mb = float(Cb.max())
                if not np.isfinite(Mb):
                    continue
                if Mb < best_trial_M:
                    best_trial_M = Mb
                    best_trial_a = b
                    best_trial_C = Cb

        if best_trial_a is None:
            break

        allow = 1.0
        if stall > 30:
            allow = 1.0 + 1e-5
        if stall > 120:
            allow = 1.0 + 1e-4

        if best_trial_M <= M * allow:
            a = best_trial_a
            C = best_trial_C
            M = best_trial_M
            if M < best_M:
                best_M = M
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 0 and stall % 100 == 0:
            cur_K = min(cur_K + 15, max(200, n // 8))
            cur_beta = min(cur_beta * 1.1, 40.0)

        if stall > 800:
            a = best_a + rng.normal(loc=0.0, scale=0.02, size=n) * best_a
            a = normalize_positive(a, floor=floor)
            C = fft_conv_same(a)
            M = float(C.max())
            cur_K = int(K)
            cur_beta = float(beta)
            stall = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FW n={n} it={it} t={elapsed:.1f}s score={2.0*n*best_M:.6f}")

    return best_a, best_M, a


def apply_move_inplace(a, C, i, j, delta):
    if i == j or delta == 0.0:
        return
    n = int(a.size)
    L = int(C.size)
    d = float(delta)

    si = i
    ei = i + n
    if ei > L:
        ei = L
    if si < ei:
        cnt = ei - si
        C[si:ei] -= 2.0 * d * a[si - i:si - i + cnt]

    sj = j
    ej = j + n
    if ej > L:
        ej = L
    if sj < ej:
        cnt = ej - sj
        C[sj:ej] += 2.0 * d * a[sj - j:sj - j + cnt]

    if 2 * j < L:
        C[2 * j] += d * d
    if i + j < L:
        C[i + j] -= 2.0 * d * d
    if 2 * i < L:
        C[2 * i] += d * d

    a[i] -= d
    a[j] += d


def coordinate_polish(a0, time_limit, K=100, beta=8.0, seed=1, verbose=False):
    n = int(np.asarray(a0, dtype=np.float64).size)
    if n <= 0:
        return np.array([], dtype=np.float64), float("inf")
    time_limit = float(time_limit)
    if time_limit <= 0.0:
        a = normalize_positive(a0)
        C = fft_conv_same(a)
        return a.copy(), float(C.max())

    floor = 1e-12 / max(1, n)
    a = normalize_positive(a0, floor=floor)
    C = fft_conv_same(a)
    M = float(C.max())
    best_a = a.copy()
    best_M = M

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    L = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it % 200 == 0 or stall > 30:
            a = np.maximum(a, floor)
            a = a / a.sum()
            C = fft_conv_same(a)
            M = float(C.max())
            if M < best_M:
                best_M = M
                best_a = a.copy()

        idx, w = select_top_weights(C, min(K, L - 1 if L > 1 else 1), beta)
        if idx.size == 0:
            break

        qdict = {}
        for kk, ww in zip(idx, w):
            qdict[int(kk)] = float(ww)

        g = correlate_top(a, idx, w)
        d = g - g.mean()

        ndonor = min(15, n)
        nrecip = min(15, n)
        donors = np.argsort(d, kind="mergesort")[::-1][:ndonor]
        recipients = np.argsort(d, kind="mergesort")[:nrecip]

        cand_pairs = []
        for ii in donors:
            i = int(ii)
            ai = float(a[i])
            if ai <= floor:
                continue
            for jj in recipients:
                j = int(jj)
                if i == j:
                    continue
                s = float(g[j] - g[i])
                if s >= 0.0:
                    continue
                qii = qdict.get(2 * i, 0.0)
                qij = qdict.get(i + j, 0.0)
                qjj = qdict.get(2 * j, 0.0)
                quad = qii - 2.0 * qij + qjj

                if quad > 1e-300:
                    delta = -s / quad
                elif quad < -1e-300:
                    delta = ai * 0.05
                else:
                    delta = ai * 0.5

                if delta <= 0.0:
                    continue
                max_delta = ai * 0.8
                if delta > max_delta:
                    delta = max_delta

                score = 2.0 * s * delta + quad * delta * delta
                if score < 0.0:
                    cand_pairs.append((score, i, j, float(delta)))

        cand_pairs.sort()
        improved = False

        for score, i, j, base_delta in cand_pairs[:3]:
            deltas = []
            raw_vals = (
                base_delta,
                0.5 * base_delta,
                0.2 * base_delta,
                0.1 * base_delta,
                0.05 * a[i],
                0.2 * a[i],
                0.5 * a[i],
            )
            for val in raw_vals:
                val = float(val)
                if val <= 0.0:
                    continue
                max_delta = a[i] * 0.8
                if val > max_delta:
                    val = max_delta
                dup = False
                for od in deltas:
                    if abs(val - od) <= 1e-12 * max(1.0, abs(od)):
                        dup = True
                        break
                if not dup:
                    deltas.append(val)

            best_val = M
            best_delta = None
            for delta in deltas[:6]:
                apply_move_inplace(a, C, i, j, delta)
                Mnew = float(C.max())
                if Mnew < best_val:
                    best_val = Mnew
                    best_delta = delta
                apply_move_inplace(a, C, j, i, delta)

            if best_delta is not None and best_val < M:
                apply_move_inplace(a, C, i, j, best_delta)
                M = best_val
                improved = True
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                    stall = 0
                break

        if not improved:
            stall += 1
            if len(donors) > 0 and len(recipients) > 0:
                i = int(donors[0])
                j = int(recipients[0])
                if i != j and a[i] > floor:
                    delta = float(rng.uniform(0.005, 0.05)) * float(a[i])
                    for delta_try in (delta, delta * 0.3, delta * 0.1):
                        apply_move_inplace(a, C, i, j, delta_try)
                        Mnew = float(C.max())
                        if Mnew < M:
                            M = Mnew
                            improved = True
                            if M < best_M:
                                best_M = M
                                best_a = a.copy()
                                stall = 0
                            break
                        apply_move_inplace(a, C, j, i, delta_try)

            if not improved and stall > 80:
                a = best_a + rng.normal(loc=0.0, scale=0.015, size=n) * best_a
                a = normalize_positive(a, floor=floor)
                C = fft_conv_same(a)
                M = float(C.max())
                stall = 0

        elif verbose and it % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"      coord n={n} it={it} t={elapsed:.1f}s score={2.0*n*best_M:.6f}")

    return best_a, best_M


def consider_candidate(seq, best_seq, best_score, label=""):
    if seq is None:
        return best_seq, best_score
    a = normalize_positive(seq)
    if a.size == 0:
        return best_seq, best_score
    sc = internal_score(a)
    if np.isfinite(sc) and sc < best_score - 1e-15:
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={a.size})")
        return a, sc
    return best_seq, best_score


def keep_top_candidates(cand_list, keep=2):
    top = []
    for sc, a, lbl in cand_list:
        if not np.isfinite(sc):
            continue
        if len(top) < keep:
            top.append([sc, a, lbl])
        else:
            worst = 0
            for i in range(1, len(top)):
                if top[i][0] > top[worst][0]:
                    worst = i
            if sc < top[worst][0]:
                top[worst] = [sc, a, lbl]
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            if top[j][0] < top[i][0]:
                top[i], top[j] = top[j], top[i]
    return top


def run_small_search(time_limit, n, parent, seed=27):
    n = int(n)
    time_limit = float(time_limit)
    if time_limit <= 1.0:
        init = make_beta_density(n, 2.0, 1.0)
        if parent is not None:
            init = resample(parent, n)
        a_out, M_, _ = optimizer_mirror(init, time_limit, K=80, beta=8.0, seed=seed)
        return a_out

    t0 = time.monotonic()
    inits = []
    if parent is not None:
        inits.append(resample(parent, n))
    inits.append(make_beta_density(n, 2.0, 1.0))
    inits.append(make_beta_density(n, 3.0, 1.2))
    inits.append(make_beta_density(n, 1.0, 3.0))
    inits.append(make_beta_density(n, 0.6, 2.5))
    inits.append(make_beta_density(n, 2.5, 0.6))
    inits.append(make_exp_density(n, 1.5))
    inits.append(make_exp_density(n, -1.5))
    inits.append(random_smooth(n, seed=seed, modes=55))
    inits.append(random_smooth(n, seed=seed + 1, modes=55))

    best = None
    best_score = float("inf")

    for i, init in enumerate(inits):
        rem = time_limit - (time.monotonic() - t0)
        if rem <= 2.0:
            break
        t_this = min(8.0, rem)
        a_out, M_, _ = optimizer_mirror(
            init, t_this,
            K=80,
            beta=8.0,
            seed=seed + i,
            verbose=False,
            alphas=(0.1, 0.3, 1.0, 3.0)
        )
        sc = internal_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a_out

    return best


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    budget = max(120.0, float(budget_s) - 35.0)
    t_end = t_start + budget

    best_seq = None
    best_score = float("inf")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and np.isfinite(p).all():
            p = np.clip(p, 0.0, 1000.0)
            if p.sum() > 1e-300:
                parent = normalize_positive(p)
                best_seq, best_score = consider_candidate(parent, best_seq, best_score, "parent")
    except Exception:
        pass

    if best_seq is None:
        parent = make_beta_density(8192, 2.0, 1.0)
        best_seq, best_score = consider_candidate(parent, best_seq, best_score, "fallback_parent")

    print(f"Initial parent: n={best_seq.size}, score={best_score:.6f}")

    small_best = None
    rem = t_end - time.monotonic()
    if rem > 160.0:
        small_time = min(95.0, rem * 0.06)
        n_small = 2048
        print(f"Small-grid search: n={n_small}, time={small_time:.0f}s")
        small_best = run_small_search(small_time, n_small, parent, seed=101)
        if small_best is not None:
            best_seq, best_score = consider_candidate(small_best, best_seq, best_score, f"small_n{small_best.size}")

    lengths = []

        lengths.append(int(parent.size))

    uniq_lengths = []
    for L in lengths:
        if L >= 2048 and L <= 32768 and L not in uniq_lengths:
            uniq_lengths.append(L)

    cand_list = []
    rem = t_end - time.monotonic()

    for si, (base_score, a_start, label) in enumerate(starts):
        rem = t_end - time.monotonic()
        if rem < 120.0:
            break
        n = int(a_start.size)
        if si == 0:

        else:

        if t_mirror < 20.0:
            continue

        K = min(300, max(90, n // 18))
        beta = 10.0 if n > 8000 else 8.0
        print(f"Main mirror start {label}: n={n}, time={t_mirror:.0f}s, base={base_score:.6f}")
        a1, M1, a1f = optimizer_mirror(
            a_start, t_mirror,
            K=K, beta=beta,
            seed=200 + si * 37,
            verbose=(si == 0),
            alphas=(0.1, 0.3, 1.0, 2.5, 6.0)
        )
        best_seq, best_score = consider_candidate(a1, best_seq, best_score, label + ":mirror_best")
        best_seq, best_score = consider_candidate(a1f, best_seq, best_score, label + ":mirror_final")

    rem = t_end - time.monotonic()
    if rem > 180.0 and best_seq is not None:
        nv = int(best_seq.size)
        t_var = min(rem * 0.17, 260.0)
        print(f"Variance equalization: n={nv}, time={t_var:.0f}s")
        a2, M2, a2f = optimizer_variance(best_seq, t_var, seed=300, verbose=False)
        best_seq, best_score = consider_candidate(a2, best_seq, best_score, "variance_best")
        best_seq, best_score = consider_candidate(a2f, best_seq, best_score, "variance_final")

    rem = t_end - time.monotonic()
    if rem > 140.0 and best_seq is not None:
        nf = int(best_seq.size)
        t_fw = min(rem * 0.13, 190.0)
        print(f"Smooth FW: n={nf}, time={t_fw:.0f}s")
        a3, M3, a3f = optimizer_fw(
            best_seq, t_fw,
            K=min(250, max(90, nf // 18)),
            beta=10.0,
            gammas=(5.0, 30.0, 150.0, 800.0),
            alphas=(0.01, 0.02, 0.05, 0.12, 0.25),
            seed=400,
            verbose=False
        )
        best_seq, best_score = consider_candidate(a3, best_seq, best_score, "FW_best")
        best_seq, best_score = consider_candidate(a3f, best_seq, best_score, "FW_final")

    rem = t_end - time.monotonic()
    if rem > 120.0 and best_seq is not None:
        nc = int(best_seq.size)
        t_coord = min(rem * 0.30, 360.0)
        print(f"Coordinate polish: n={nc}, time={t_coord:.0f}s")
        a4, M4 = coordinate_polish(
            best_seq, t_coord,
            K=min(160, max(80, nc // 20)),
            beta=9.0,
            seed=500,
            verbose=False
        )
        best_seq, best_score = consider_candidate(a4, best_seq, best_score, "coord")

    rem = t_end - time.monotonic()
    if rem > 90.0 and best_seq is not None:
        print("Final length sweep and short polish...")
        final_lengths = []
        for L in (8192, 16384, 32768):
            if L not in final_lengths:
                final_lengths.append(L)
        current_len = int(best_seq.size)
        final_lengths = [L for L in final_lengths if L != current_len]

        for L in final_lengths:
            rem = t_end - time.monotonic()
            if rem < 70.0:
                break
            c = resample(best_seq, L)
            sc = internal_score(c)
            c_label = f"final_L{L}"
            if np.isfinite(sc) and sc < best_score + 0.0025:
                best_seq, best_score = consider_candidate(c, best_seq, best_score, c_label)
                rem = t_end - time.monotonic()
                t_opt = min(rem * 0.25, 70.0 if L <= 16384 else 50.0)
                if t_opt > 15.0:
                    a_opt, M_opt, a_optf = optimizer_mirror(
                        c, t_opt,
                        K=min(250, max(80, L // 20)),
                        beta=11.0,
                        seed=800 + L,
                        verbose=False,
                        alphas=(0.1, 0.3, 1.0, 2.5)
                    )
                    best_seq, best_score = consider_candidate(a_opt, best_seq, best_score, c_label + ":mirror")
                    best_seq, best_score = consider_candidate(a_optf, best_seq, best_score, c_label + ":mirror_f")
                rem = t_end - time.monotonic()
                if rem > 35.0:
                    a_co, M_co = coordinate_polish(
                        best_seq, min(rem * 0.2, 45.0),
                        K=min(140, max(70, L // 22)),
                        beta=10.0,
                        seed=900 + L,
                        verbose=False
                    )
                    best_seq, best_score = consider_candidate(a_co, best_seq, best_score, c_label + ":coord")

    rem = t_end - time.monotonic()
    if rem > 30.0 and best_seq is not None:
        nt = int(best_seq.size)
        a5, M5, a5f = optimizer_mirror(
            best_seq, min(rem * 0.5, 80.0),
            K=min(250, max(80, nt // 20)),
            beta=12.0,
            seed=1000,
            verbose=False,
            alphas=(0.1, 0.3, 1.0, 2.5)
        )
        best_seq, best_score = consider_candidate(a5, best_seq, best_score, "final_mirror")
        best_seq, best_score = consider_candidate(a5f, best_seq, best_score, "final_mirror_f")

    if best_seq is None:
        best_seq = make_beta_density(8192, 2.0, 1.0)
        best_score = internal_score(best_seq)

    best_arr = np.asarray(best_seq, dtype=np.float64)
    if best_arr.size == 0 or not np.isfinite(best_arr).all():
        best_arr = make_beta_density(8192, 2.0, 1.0)
    best_arr = normalize_positive(best_arr)

    final_list = [float(x) for x in best_arr]
    try:
        ev = float(evaluate_sequence(final_list))
        if np.isfinite(ev):
            print(f"Final evaluate_sequence: {ev:.6f} (n={len(final_list)})")
    except Exception:
        pass

    fs = float(internal_score(best_arr))
    print(f"FINAL internal score: {fs:.6f}, n={len(final_list)}, time={time.monotonic() - t_start:.1f}s")
    return final_list
```
