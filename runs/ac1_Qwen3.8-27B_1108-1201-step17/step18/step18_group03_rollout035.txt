g[i] = sum_{m=i}^{i+n-1} a[m-i] * u[m]

This is (a * u'[i])[0] where u' is reversed... I'm overcomplicating this.

The simplest way: g[i] = dot(a, u[i:i+n]) is a correlation that can be computed as:
g = np.convolve(a, u[::-1], mode='valid')[::-1] ... no.

Actually, let's think of it this way:
g[i] = sum_j a[j] * u[i+j]

If I reverse a: a_rev[j] = a[n-1-j], then:
g[i] = sum_j a[n-1-(n-1-j)] * u[i+j] = sum_j a_rev[n-1-j] * u[i+j]

Let k = n-1-j: g[i] = sum_k a_rev[k] * u[i+n-1-k] = (a_rev * u)[i+n-1] (this IS a convolution!)

So: g[i] = (a_rev * u)[i+n-1] where u = w (the softmax weights).

Using FFT:
N = next_power_of_2(2*n - 1)
A_rev = FFT(a_rev, N)
U = FFT(w, N)
conv_full = IFFT(A_rev * U)
g[i] = conv_full[i+n-1] for i = 0, ..., n-1
So g = conv_full[n-1 : 2*n-1]

This is O(n log n)!

Great, so the algorithm is:
1. r = full_conv(a, a) [via FFT, O(n log n)]
2. w = softmax(beta * r) [O(n)]
3. a_rev = a[::-1]
4. g = IFFT(FFT(a_rev) * FFT(w))[n-1:2n-1] [O(n log n)]
5. a_new = a * exp(-eta * g), normalize [O(n)]

Each iteration is O(n log n)! This is very fast.

- Too large: oscillation, negative values after exp
- Too small: slow convergence

For the beta annealing:

- Gradually increase to sharpen the focus on peaks

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE (Log-Sum-Exp) Mirror Descent with FFT-based gradient computation
2. Chirp and windowed-signal based initialization
3. LP-Frank-Wolfe cutting plane for fine-tuning
4. Adaptive coordinate descent for final polish
Key innovation: O(n log n) gradient via spectral computation of softmax-weighted autocorrelation.

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    """Compute full autocorrelation (convolution of a with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_gradient(a, beta, N):
    """
    Compute the gradient of the log-sum-exp of autocorrelation.
    F(a) = (1/beta) * log(sum_k exp(beta * r_k))
    grad F[i] = sum_j softmax(beta*r)[i+j] * a[j]
    
    Parameters:
    a: input sequence, length n
    beta: sharpness parameter
    N: FFT size (power of 2, >= 2n-1)
    
    Returns: gradient array of length n
    """
    n = len(a)
    r = fast_autocorr(a)  # length 2n-1
    
    # Softmax (numerically stable)
    br = beta * r
    br_shifted = br - br.max()
    w = np.exp(br_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    
    # Compute g[i] = sum_j w[i+j] * a[j] for i = 0,...,n-1
    # Using: g[i] = (a_rev * w)[i+n-1] where a_rev = a[::-1]
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(A_rev * W, N)
    g = conv_full[n - 1:2 * n - 1]
    
    return g, w, r


def lse_mirror_descent(a0, time_limit, beta_start=1.0, beta_end=50.0, 
                       eta0=1.0, n_betas=10, iters_per_beta=5000,
                       verbose=True, seed=42):
    """
    Log-Sum-Exp Mirror Descent for minimizing max autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    N = 1 << (2 * n - 2).bit_length()
    
    # Beta schedule
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t0 = time.monotonic()
        for t in range(1, iters_per_beta + 1):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute gradient
            g, w, r = lse_gradient(a, beta, N)
            
            # Step size: eta_t = eta0 / sqrt(t + 1)
            eta = eta0 / (np.sqrt(t) + 1.0)
            
            # Mirror descent step: a_new = a * exp(-eta * g)
            log_a_new = np.log(np.maximum(a, 1e-30)) - eta * g
            log_a_new -= log_a_new.max()  # numerical stability
            a_new = np.exp(log_a_new)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            # Check improvement
            r_new = fast_autocorr(a_new)
            M_new = r_new.max()
            
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
            
            a = a_new
            
            if verbose and t % 1000 == 0 and time.monotonic() - beta_t0 > 5:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LSE-MD beta={beta:.1f} t={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")
        
        # Periodic perturbation to escape local minima
        if bi < len(betas) - 1:
            pert = rng.normal(0, 0.01 * best_m / best_m, n)  # small perturbation
            a_pert = best_a + pert * best_a
            a = norm_sum1(a_pert)
            
    # Final check
    r_best = fast_autocorr(best_a)
    best_m = r_best.max()
    
    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"    LSE-MD done ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def lse_md_with_restarts(a0, time_limit, n_restarts=5, verbose=True):
    """LSE-MD with multiple restarts from different perturbations."""
    n = len(a0)
    best_a = norm_sum1(a0).copy()
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    per_restart_time = time_limit / n_restarts
    
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 30:
            break
        
        # Perturb the best solution for this restart
        if r == 0:
            a_init = best_a.copy()
        else:
            # Different types of perturbation
            pert_type = r % 4
            if pert_type == 0:
                # Gaussian perturbation
                noise = rng.normal(0, 0.05, n) * best_a
                a_init = norm_sum1(best_a + noise)
            elif pert_type == 1:
                # Chirp blend
                x = np.arange(n, dtype=np.float64)
                chirp = np.exp(-1j * 0.5 * x / n)
                a_chirp = np.abs(best_a * chirp)
                a_init = norm_sum1(0.7 * best_a + 0.3 * a_chirp)
            elif pert_type == 2:
                # Shift and blend
                shift = int(rng.integers(1, max(2, n // 10)))
                shifted = np.roll(best_a, shift)
                a_init = norm_sum1(0.6 * best_a + 0.4 * shifted)
            else:
                # Inverse frequency modulation
                A = np.fft.rfft(best_a, 1 << (2*n-2).bit_length())
                # Slightly modify the spectrum
                phase = rng.normal(0, 0.1, len(A))
                A_mod = A * np.exp(1j * phase)
                a_mod = np.fft.irfft(A_mod, 1 << (2*n-2).bit_length())[:n]
                a_init = norm_sum1(0.5 * best_a + 0.5 * np.abs(a_mod))
        
        remaining = time_limit - (time.monotonic() - t0)
        t_alloc = min(per_restart_time, remaining - 20)
        if t_alloc < 10:
            break
        
        a_out, m_out = lse_mirror_descent(
            a_init, t_alloc,
            beta_start=2.0 + r * 2.0,
            beta_end=30.0 + r * 10.0,
            eta0=0.8 + 0.1 * r,
            n_betas=8,
            iters_per_beta=3000,
            verbose=False,
            seed=100 + r
        )
        
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  *** LSE-MD restart {r}: NEW BEST J={J:.6f} (t={elapsed:.0f}s)")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=2500, 
                   add_per_iter=40, verbose=True):
    """LP-based Frank-Wolfe with cutting planes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    failed_solves = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()
        
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 5:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-CW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Aggressive coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    stall = 0
    max_stall = 300
    N = 1 << (2 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        r = fast_autocorr(a)
        M_curr = r.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
        
        improved = False
        n_top = min(30, len(r))
        top_k = np.argsort(r)[::-1][:n_top]
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            n_pairs = min(6, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]
            
            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Find best destination
                scores = np.zeros(n)
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * r[k2i]
                        scores[k2i - idx2] += a[idx2] * r[k2i]
                
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True
                
                candidate_mask = ~in_peak
                if candidate_mask.sum() < 10:
                    candidate_mask = np.ones(n, dtype=bool)
                candidate_mask[src] = False
                
                cand_scores = scores.copy()
                cand_scores[~candidate_mask] = np.inf
                dst = np.argmin(cand_scores)
                
                if dst == src or a[dst] > 0.5 * src_val:
                    continue
                
                found = False
                for frac in [0.02, 0.04, 0.07, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                
                if not found:
                    break
        
        if not improved:
            stall += 1
            if stall > max_stall:
                break
            if stall % 5 == 0:
                n_moves = int(rng.integers(2, 8))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.005, 0.02))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
            if verbose and stall % 50 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    CoordDescent ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def parametric_init(n, best_seq, best_score):
    """Generate diverse parametric initializations."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = norm_sum1(best_seq)
    best_val = best_score
    
    results = []
    
    def try_update(a, label):
        nonlocal best_a, best_val
        a_n = norm_sum1(a)
        ac = fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        results.append((val, a_n, label))
        if val < best_val:
            best_val = val
            best_a = a_n.copy()
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        try_update((i_arr + 1.0) ** p, f"pow_{p}")
        try_update((n - i_arr) ** p, f"ipow_{p}")
    
    # Chirp-like (real part of complex chirp, made non-negative)
    for rate in [1, 2, 3, 5]:
        chirp = np.exp(-1j * rate * np.pi * x * x)
        try_update(np.abs(chirp), f"chirp_{rate}")  # always 1, skip
        try_update(np.maximum(np.real(chirp), 0), f"chirpR_{rate}")
        try_update(np.maximum(np.cos(rate * np.pi * x * x), 0), f"chirpC_{rate}")
    
    # Windowed versions
    for alpha in [0.5, 1.0, 1.5]:
        for beta_w in [0.5, 1.0, 1.5]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta_w
            try_update(a, f"win_{alpha}_{beta_w}")
    
    # Sine waves
    for freq in range(1, 20):
        for ph in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(freq * np.pi * x + ph * np.pi), 0) ** 2
            if a.sum() > 0:
                try_update(a, f"sin{freq}_{ph:0.2f}")
    
    # Gaussian
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.1, 0.2, 0.3, 0.4]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            try_update(a, f"gauss_c{c:0.1f}_s{sigma:0.1f}")
    
    # Bimodal
    for c1, c2 in [(0.2, 0.8), (0.25, 0.75), (0.3, 0.7)]:
        for sigma in [0.05, 0.1, 0.15]:
            a = np.exp(-((x - c1)**2)/(2*sigma**2)) + np.exp(-((x - c2)**2)/(2*sigma**2))
            try_update(a, f"bimo_{sigma:0.2f}")
    
    # Exponential
    for rate in [1, 2, 3, 5, 10]:
        try_update(np.exp(-rate * x), f"expd_{rate}")
        try_update(np.exp(rate * x), f"expi_{rate}")
    
    # Cosine-tapered
    for alpha in [0.5, 1.0, 2.0]:
        a = np.cos(np.pi * x - np.pi/2) ** alpha
        try_update(np.maximum(a, 0), f"costap_{alpha}")

    try_update(0.54 - 0.46 * np.cos(2 * np.pi * x), "hann")
    try_update(0.54 - 0.46 * np.cos(4 * np.pi * x), "hann2")
    
    # Sinc-like
    for bw in [1, 2, 3]:
        a = np.sinc(bw * (x - 0.5))
        try_update(np.maximum(a, 0), f"sinc_{bw}")
    
    results.sort(key=lambda x: x[0])
    return best_a, best_val, results[:10]


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]

    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_lse_gradient(a, beta, N):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta * r_k)) wrt a.
    grad[i] = sum_j w[i+j] * a[j] where w = softmax(beta * r)
    Computed via FFT in O(n log n).
    Returns (gradient, autocorrelation, max_val)
    """
    n = len(a)
    r = fast_autocorr(a)
    M_curr = r.max()

    br = beta * r
    br_shifted = br - br.max()
    w = np.exp(br_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1, dtype=np.float64) / (2 * n - 1)
    else:
        w = w / w_sum

    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(A_rev * W, N)
    g = conv_full[n - 1:2 * n - 1]

    return g, r, M_curr


def lse_mirror_descent(a0, time_limit, beta_start=3.0, beta_end=80.0,
                       eta0=0.8, n_betas=12, iters_per_beta=5000,
                       verbose=False, seed=42):
    """LSE Mirror Descent with beta annealing."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    N = 1 << (3 * n - 2).bit_length()
    betas = np.geomspace(beta_start, beta_end, n_betas)
    total_iters = 0

    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break

        for t in range(1, iters_per_beta + 1):
            if time.monotonic() - t0 > time_limit:
                break

            g, r, M_curr = compute_lse_gradient(a, beta, N)

            eta = eta0 / (np.sqrt(t) + 1.0)
            log_a_new = np.log(np.maximum(a, 1e-30)) - eta * g
            max_log = log_a_new.max()
            if max_log > 700:
                max_log = 700
            a_new = np.exp(log_a_new - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            r_new = fast_autocorr(a_new)
            M_new = r_new.max()

            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()

            a = a_new
            total_iters += 1

            if verbose and t % 2000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LSE-MD b={beta:.1f} t={t} ({elapsed:.0f}s): J={J:.6f}")

        # Perturb to escape local minima between beta stages
        if bi < len(betas) - 1:
            noise_scale = 0.02 * (1.0 + 0.1 * bi)
            perturb = rng.normal(0, noise_scale, n) * best_a
            a = norm_sum1(best_a + perturb)

    r_best = fast_autocorr(best_a)
    best_m = r_best.max()
    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"    LSE-MD done ({elapsed:.0f}s): J={J:.6f}, iters={total_iters}")
    return best_a, best_m


def lse_md_multi_restart(a0, time_limit, n_restarts=4, verbose=True):
    """LSE-MD with diverse restarts."""
    n = len(a0)
    best_a = norm_sum1(a0).copy()
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)

    per_time = time_limit / n_restarts

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        t_alloc = min(per_time * (1.5 - 0.1 * r), remaining - 15)
        if t_alloc < 10:
            break

        if r == 0:
            a_init = best_a.copy()
        else:
            pt = r % 5
            if pt == 1:
                noise = rng.normal(0, 0.04, n) * best_a
                a_init = norm_sum1(best_a + noise)
            elif pt == 2:
                x = np.arange(n, dtype=np.float64) / n
                chirp_phase = rng.uniform(0.5, 3.0)
                mod = np.cos(chirp_phase * np.pi * (x - 0.5) ** 2)
                a_mod = np.abs(best_a * mod)
                a_init = norm_sum1(0.6 * best_a + 0.4 * a_mod)
            elif pt == 3:
                shift = int(rng.integers(2, max(3, n // 15)))
                shifted = np.roll(best_a, shift)
                a_init = norm_sum1(0.5 * best_a + 0.5 * shifted)
            elif pt == 4:
                A = np.fft.rfft(best_a, 1 << (2 * n - 2).bit_length())
                phase = rng.normal(0, 0.15, len(A))
                A_mod = A * np.exp(1j * phase)
                a_mod = np.fft.irfft(A_mod, 1 << (2 * n - 2).bit_length())[:n]
                a_init = norm_sum1(0.4 * best_a + 0.6 * np.abs(a_mod))
            else:
                a_init = norm_sum1(best_a + rng.normal(0, 0.03, n) * best_a)

        beta_s = 2.0 + 1.5 * r
        beta_e = 40.0 + 20.0 * r
        eta_v = 0.7 + 0.1 * r

        a_out, m_out = lse_mirror_descent(
            a_init, t_alloc,
            beta_start=beta_s, beta_end=beta_e,
            eta0=eta_v, n_betas=10, iters_per_beta=4000,
            verbose=False, seed=500 + r
        )

        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  *** LSE-MD restart {r}: NEW BEST J={J:.6f} (t={elapsed:.0f}s)")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=2500,
                   add_per_iter=50, verbose=True):
    """LP-based Frank-Wolfe with cutting planes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints_list = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints_list)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 5:
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-CW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coord_descent_polish(a0, time_limit, verbose=True):
    """Aggressive coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit:
        r = fast_autocorr(a)
        M_curr = r.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        improved = False
        n_top = min(25, len(r))
        top_k = np.argsort(r)[::-1][:n_top]

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            n_pairs = min(5, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * r[k2i]
                        scores[k2i - idx2] += a[idx2] * r[k2i]

                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True

                candidate_mask = ~in_peak
                if candidate_mask.sum() < 10:
                    candidate_mask = np.ones(n, dtype=bool)
                candidate_mask[src] = False

                cand_scores = scores.copy()
                cand_scores[~candidate_mask] = np.inf
                dst = int(np.argmin(cand_scores))

                if dst == src or a[dst] > 0.5 * src_val:
                    continue

                found = False
                for frac in [0.02, 0.04, 0.06, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

                if not found:
                    break

        if not improved:
            stall += 1
            if stall > max_stall:
                break
            if stall % 5 == 0:
                n_moves = int(rng.integers(2, 8))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.003, 0.015))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
            if verbose and stall % 50 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    CoordDescent ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best
