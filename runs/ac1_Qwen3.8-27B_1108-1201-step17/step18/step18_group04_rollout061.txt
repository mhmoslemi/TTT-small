def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad_active(a, conv, n_active=50, min_gap=None):
    """Compute subgradient using active set of near-maximal peaks.
    
    The subgradient of max_k conv(k) at a point where multiple peaks are
    near-equal is a convex combination of the individual gradients.
    """
    n = len(a)
    conv_max = conv.max()
    
    if min_gap is None:
        min_gap = 1e-4 * conv_max
    
    threshold = conv_max - min_gap
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Weight by how close to max (exponential weighting)
    gaps = conv_max - conv[active]
    w = np.exp(-gaps / (conv_max * 1e-3))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    
    # Cap active set size
    if len(active) > n_active:
        order = np.argsort(w)[::-1][:n_active]
        active = active[order]
        w = w[order]
        w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad


def peak_shave(a, n_shave=10, n_dst=20, time_limit=10.0):
    """Direct peak shaving: identify the highest peaks and redistribute mass
    from the contributing elements to the least-affected positions."""
    n = len(a)
    a = a.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    for _ in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv(a)
        # Find top peaks
        top_peaks = np.argsort(conv)[::-1][:n_shave]
        peak_vals = conv[top_peaks]
        
        if peak_vals[0] / peak_vals[min(n_shave-1, len(peak_vals)-1)] < 1.01:
            # Peaks are already well-equalized
            break
        
        # For the highest peak, find its contributors
        k_max = int(top_peaks[0])
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k_max - idx]
        
        # Find the biggest contributor
        src_rel = np.argmax(contrib)
        src_i = int(idx[src_rel])
        
        # Find a destination that minimally increases other peaks
        # Heuristic: positions where the total convolution contribution is lowest
        total_contrib = np.zeros(n)
        for k in top_peaks[:5]:
            k_i = int(k)
            lo2 = max(0, k_i - n + 1)
            hi2 = min(n - 1, k_i)
            idx2 = np.arange(lo2, hi2 + 1)
            total_contrib[idx2] += a[idx2] * a[k_i - idx2]
        
        # Mask out the source
        total_contrib[src_i] = np.inf
        
        # Find positions with lowest total contribution
        dst_candidates = np.argsort(total_contrib)[:n_dst]
        
        improved = False
        for dst_i in dst_candidates:
            dst_i = int(dst_i)
            if dst_i == src_i:
                continue
            # Try moving a small fraction
            for frac in [0.02, 0.05, 0.1, 0.15]:
                amt = a[src_i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src_i] -= amt
                a_trial[dst_i] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, fast_conv(best_a).max()


def amd_optimize_v2(a0, time_limit, top_k=30, temp=0.002, c0=3.0,
                    restart_every=5000, seed=42, verbose=False,
                    use_peak_shave=True):
    """Improved AMD with active set tracking and peak shaving."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_conv = fast_conv(x)
    best_m = best_conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall_count = 0
    restart_count = 0
    peak_shave_interval = 5000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        conv_y = fast_conv(y)
        g = compute_subgrad_active(y, conv_y, n_active=top_k, min_gap=temp * conv_y.max())
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Peak shaving periodically
        if use_peak_shave and t % peak_shave_interval == 0:
            a_shaved, m_shaved = peak_shave(x, time_limit=min(5.0, time_limit - (time.monotonic() - t0)))
            if m_shaved < best_m:
                best_m = m_shaved
                best_x = a_shaved.copy()
                x = a_shaved.copy()
                x_prev = a_shaved.copy()
                stall_count = 0
        
        # Restart logic
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            # Adaptive perturbation
            strength = min(0.08, 0.02 + 0.01 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
            # Adapt parameters
            top_k = min(top_k + 3, 60)
            c0 = 3.0 * (1.0 + 0.03 * min(restart_count, 15))
            restart_every = max(4000, restart_every + 1000)
            
            if verbose:
                J = 2.0 * n * best_m
                print(f"    AMD restart {restart_count} t={t}: J={J:.6f}")
    
    return best_x, best_m


def newton_refine(a0, time_limit, seed=42):
    """Newton-style refinement: solve for the direction that equalizes active peaks.
    
    At the optimum, the gradient of the max should be balanced. We use a
    second-order approach: find the direction d on the simplex that minimizes
    the maximum of the linearized convolution values.
    """
    n = len(a0)
    a = norm_sum1(a0)
    t0 = time.monotonic()
    best_a = a.copy()
    best_m = fast_conv(a).max()

    for iteration in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv(a)
        conv_max = conv.max()

        threshold = conv_max * 0.999
        active_indices = np.where(conv >= threshold)[0]
        
        if len(active_indices) == 0:
            active_indices = np.array([np.argmax(conv)])
        
        # Solve: minimize max_k (conv + 2*a_k^T * d)(k) subject to sum(d)=0, a+d >= 0
        # Linearized: (a*a)(k) + 2*sum_i a_i*d_{k-i} + 2*sum_i d_i*a_{k-i} = conv(k) + 2*(a*d)(k) + 2*(d*a)(k)
        # Since a*d = d*a (convolution is commutative), this is conv(k) + 4*(a*d)(k)
        # Wait no: (a+d)*(a+d)(k) = (a*a)(k) + (a*d)(k) + (d*a)(k) + (d*d)(k)
        # = conv(k) + 2*(a*d)(k) + (d*d)(k)
        # Linearized: conv(k) + 2*(a*d)(k)
        
        # We want to find d such that conv(k) + 2*(a*d)(k) <= target for all active k
        # And minimize target. Also sum(d) = 0 (to maintain normalization) and a+d >= 0.
        
        # This is a linear program:
        # minimize target
        # subject to: 2*(a*d)(k) - target <= -conv(k) for all k in active
        #             sum(d) = 0
        #             -a <= d <= inf
        
        # Too expensive for large n. Instead, use an iterative approach:
        # Gradient of the max at the active set, projected onto the tangent of the simplex.
        
        # Compute the "effective gradient" from active peaks
        grad = np.zeros(n)
        n_active = len(active_indices)
        # Weigh each active peak by how much it exceeds the mean active value
        active_vals = conv[active_indices]
        mean_active = active_vals.mean()
        spread = active_vals.max() - active_vals.min()
        if spread < 1e-15:
            # All active peaks are equal - we're at a local min or plateau
            # Try a random perturbation
            d = rng.normal(0, 0.01, n)
            d -= d.mean()
            d = np.where(a + np.abs(d) > 0, d, 0)
        else:
            # Weight peaks that are higher more heavily
            w = (active_vals - mean_active + spread * 0.1)
            w = np.maximum(w, 0.01)
            w = w / w.sum()
            
            for wi, k in zip(w, active_indices):
                k_i = int(k)
                lo = max(0, k_i - n + 1)
                hi = min(n - 1, k_i)
                idx = np.arange(lo, hi + 1)
                grad[idx] += 2.0 * wi * a[k_i - idx]
        
        # Project gradient onto simplex tangent (remove mean)
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean
        
        # Scale to ensure feasibility
        step_size = 1.0
        for _ in range(10):
            d = -step_size * grad_tangent
            a_trial = a + d
            if a_trial.min() >= 0:
                break
            step_size *= 0.5
        
        a_trial = a - step_size * grad_tangent
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = a_trial / a_trial.sum()
        
        M_trial = fast_conv(a_trial).max()
        if M_trial < best_m - 1e-15:
            a = a_trial
            best_m = M_trial
            best_a = a_trial.copy()
        else:
            # Try smaller step

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_sinc_family(n, bw, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - center) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_power_asym(n, alpha, beta):
    """Asymmetric power law: a_i ∝ (i+1)^alpha * (n-i)^beta"""
    x = np.arange(n, dtype=np.float64)
    a = (x + 1) ** alpha * (n - x) ** beta
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_gaussian_mix(n, centers, widths, amps=None):
    x = np.linspace(0, 1, n)
    if amps is None:
        amps = [1.0] * len(centers)
    a = np.zeros(n)
    for c, w, amp in zip(centers, widths, amps):
        a += amp * np.exp(-((x - c) / w) ** 2)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)
