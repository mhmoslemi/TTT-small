import numpy as np
import time
from math import log, sqrt, pi, exp, cos, sin

_best_tracker = {'seq': None, 'score': float('inf')}

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgrad(a, conv, top_k=40, temperature=0.005):
    n = len(a)
    conv_max = conv.max()
    # Use a wider band for smoother gradient
    threshold = conv_max - temperature * 10.0
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def amd_optimize(a0, time_limit, top_k_init=30, temp_init=0.004, c0_init=3.0,

def spectral_flatten(a, time_limit, alpha=0.5, n_iter_max=500, seed=42, verbose=False):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = norm_sum1(a).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    fft_size = 1 << (4 * n - 1).bit_length()  # Pad to avoid wrap-around
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A)**2
        
        # Spectral flattening: compress dynamic range
        P_mean = P.mean()
        P_flat = P_mean + alpha * (P - P_mean)
        P_flat = np.maximum(P_flat, 1e-20)
        
        # Reconstruct with original phases
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = np.sqrt(P_flat) * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M = fast_conv(a_new).max()
        if M < best_m:
            best_m = M
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            alpha = max(0.05, alpha * 0.95)
            a = best_a.copy()
    
    return best_a, best_m

def coord_descent_polish(a0, time_limit, seed=42, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Find which source indices contribute most to the peak
        n_top_src = max(8, min(30, n // 100))
        top_src = idx[np.argsort(contrib)[::-1][:n_top_src]]
        
        # Compute cost (contribution to all significant peaks) for each position
        n_top_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_top_peaks]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]
        
        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            candidates = np.argsort(costs + 1e-10 * a)[:max(15, n // 25)]
            candidates = [int(c) for c in candidates if int(c) != src_i]
            for dst_i in candidates[:12]:
                found = False
                for frac in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.08)

                a[i] -= amt
                a[j] += amt

                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:

                    best_a = a.copy()
                    stall = 0

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss
