def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def mgd_smooth(a0, time_limit, beta=10.0, eta0=0.03, verbose=False):
    """
    MGD with softmax-weighted all-lag subgradient.
    beta controls concentration on top lags.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    patience = 0
    max_patience = 400

    iter_count = 0

        M = ac.max()

            best_a = a.copy()

            patience += 1

            a = norm_sum1(a)
            patience = 0
            eta = max(eta, eta0 * 0.3)

        ac_max_shifted = ac_shifted.max()

        ac_norm = ac_shifted / ac_max_shifted

            w = w / w_sum

        if active.sum() < 3:
            active = np.argsort(w)[::-1][:10]
            active = np.zeros_like(w, dtype=bool)

def coord_descent(a0, time_limit, verbose=False):

    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1

        n_pass_improvements = 0

            if time.monotonic() - t0 > time_limit:
                break

                # Try adding some mass here
                lo, hi = 0.0, a.max() * 0.5
            else:
                lo, hi = 0.0, x * 2.5 + 0.001
            
            if hi <= lo * 1.01:
                hi = lo + 0.001

            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)

            a_trial1[i] = a1
            a_trial1 = a_trial1 / a_trial1.sum()
            f1 = fast_autocorr(a_trial1).max()

                    a_trial1[i] = a1

                    a1 = a2
                    f1 = f2

                    a_trial2[i] = a2

def block_search(a0, time_limit, block_size=20, verbose=False):
    """
    Block-level local search with targeted perturbations.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False
        
        # Targeted perturbations near peak-contributing positions
        top_lags = np.argsort(ac)[::-1][:8]
        
        for attempt in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            if attempt < 15 and len(top_lags) > 0:
                k_pick = int(top_lags[attempt % len(top_lags)])
                js = max(0, k_pick - n + 1)
                je = min(n - 1, k_pick)
                center = (js + je) // 2
                start = max(0, min(n - block_size, center - block_size // 2 + int(rng.integers(-10, 10))))
            else:
                start = int(rng.integers(0, max(1, n - block_size + 1)))
            
            end = min(n, start + block_size)
            bs = end - start
            if bs < 2:
                continue
            
            block = a[start:end].copy()
            mean_block = block.mean() + 1e-8
            
            # Different perturbation strategies
            strategy = attempt % 3
            if strategy == 0:
                # Gaussian perturbation
                scale = rng.uniform(0.03, 0.1)
                pert = rng.normal(0, scale * mean_block, bs)
                new_block = np.maximum(block + pert, 0.0)
            elif strategy == 1:
                # Scale block
                factor = rng.uniform(0.7, 1.3)
                new_block = block * factor
            else:
                # Swap within block
                new_block = block.copy()
                p = rng.permutation(bs)
                new_block = block[p]
            
            if np.allclose(new_block, block, atol=1e-10):
                continue
            
            a_temp = a.copy()
            a_temp[start:end] = new_block
            a_temp = np.maximum(a_temp, 0.0)
            s_t = a_temp.sum()
            if s_t < 1e-15:
                continue
            a_temp = a_temp / s_t
            M_temp = fast_autocorr(a_temp).max()
            
            if M_temp < M - 1e-14:
                a = a_temp
                M = M_temp
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                improved = True
        
        if not improved:
            stall += 1
            # Random escape: move mass between random positions
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                frac = rng.uniform(0.02, 0.1)
                amt = a.max() * frac
                a[i] = max(0, a[i] - amt)
                a[j] += amt
                a = norm_sum1(a)
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Block done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, best_a, best_score):
    """Try parametric families for initialization."""
    x = np.linspace(0, 1, n)

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:

        families.append(np.exp(rate * (x - 1.0)))
    # Cosine windows
    for beta_p in [0.5, 1.0, 2.0, 5.0, 10.0]:
        families.append(0.54 + 0.46 * np.cos(2 * np.pi * x))
    # Triangle
    families.append(np.maximum(0, 1 - 2 * np.abs(x - 0.5)))
    # Uniform (baseline)
    families.append(np.ones(n))
    
    best_local = best_a.copy()
    best_local_score = best_score
    
    for fam in families:
        fam_norm = norm_sum1(fam)
        ac = fast_autocorr(fam_norm)
        val = 2.0 * n * ac.max()
        if val < best_local_score:
            best_local_score = val
            best_local = fam_norm.copy()
    
    return best_local, best_local_score
