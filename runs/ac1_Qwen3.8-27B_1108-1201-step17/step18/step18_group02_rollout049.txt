def fast_autocorr(a):
    """Compute full linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize so sum = 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_weighted_gradient(a, ac, tau, top_k=None):
    """
    Compute gradient of smoothed peak objective w.r.t. a.
    Uses log-sum-exp weighting, optionally sharpened to top_k peaks.
    """
    n = len(a)
    size = len(ac)
    ac_scaled = ac / tau
    ac_scaled -= ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()
    
    if top_k is not None and top_k < size:
        # Sharpen: boost weight on top peaks
        threshold = np.sort(p)[::-1][top_k] * (1 + 1.0 / top_k)
        boost = np.where(p >= threshold, 2.0, 0.3)
        p = p * boost
        p /= p.sum()
    
    # grad[i] = 2 * sum_j p[i+j] * a[j] for valid j
    # = 2 * (reverse(a) * p)[i + n - 1]
    a_flip = a[::-1].astype(np.float64)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)


def peak_shaving_step(a, ac, n_shave=10, n_cands=5, verbose=False):
    """
    Direct peak-shaving: for each top peak, find elements that contribute most
    and transfer mass to elements with minimal impact on all top peaks.
    Returns improved (a, M) or original if no improvement.
    """
    n = len(a)
    M_curr = ac.max()
    a_work = a.copy()
    best_improvement = 0
    best_a = a.copy()
    
    # Identify top peaks
    top_peaks = np.argsort(ac)[::-1][:n_shave]
    
    # Build sensitivity matrix: sens[i] = sum over top peaks of contribution of a[i]
    # For peak at lag k: grad wrt a[i] = a[k-i] (if valid)
    sens = np.zeros(n)
    peak_contribs = {}
    for k in top_peaks:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        contrib = a_work[idx] * a_work[k_int - idx]
        peak_contribs[k_int] = (idx, contrib)
        for i in idx:
            sens[i] += a_work[k_int - i]
    
    if sens.max() < 1e-20:
        return a, M_curr
    
    # For each element with high sensitivity, try moving mass to low-sensitivity elements
    low_sens_idx = np.argsort(sens)[:n_cands * 10]
    low_sens_idx = low_sens_idx[sens[low_sens_idx] < sens.max() * 0.3]
    
    if len(low_sens_idx) == 0:
        low_sens_idx = np.where(sens < sens.max() * 0.3)[0]
    
    if len(low_sens_idx) == 0:
        return a, M_curr
    
    # Try transferring from highest-sensitivity elements
    high_sens_idx = np.argsort(sens)[::-1][:min(20, n)]
    
    for src in high_sens_idx:
        if a_work[src] < 1e-10:
            continue
        for frac in [0.01, 0.02, 0.05]:
            amt = a_work[src] * frac
            best_dst = None
            best_M = M_curr
            for dst_k in np.random.default_rng(int(src)).choice(low_sens_idx, size=min(5, len(low_sens_idx)), replace=False):
                dst = int(dst_k)
                if dst == src:
                    continue
                trial = a_work.copy()
                trial[src] -= amt
                trial[dst] += amt
                trial = np.maximum(trial, 0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                ac_t = fast_autocorr(trial)
                M_t = ac_t.max()
                if M_t < best_M:
                    best_M = M_t
                    best_dst = dst
                    best_trial = trial
            if best_dst is not None and best_M < best_improvement or (best_improvement == 0 and best_M < M_curr):
                a_work = best_trial
                best_improvement = best_M
                M_curr = best_M
                # Update sensitivity
                sens = np.zeros(n)
                top_peaks = np.argsort(best_improvement if isinstance(best_improvement, np.ndarray) else fast_autocorr(a_work))[::-1][:n_shave]
                for k in top_peaks:
                    k_int = int(k)
                    i_start = max(0, k_int - n + 1)
                    i_end = min(n - 1, k_int)
                    if i_start > i_end:
                        continue
                    idx = np.arange(i_start, i_end + 1)
                    for i in idx:
                        sens[i] += a_work[k_int - i]
    
    if best_improvement > 0:
        return a_work, M_curr
    return a, M_curr


def optimized_adam(a0, time_limit, tau_init=None, top_k=None, verbose=True, seed=42):
    """
    Adaptive log-sum-exp Adam with optional peak sharpening.
    More aggressive cooling and better restarts.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    if tau_init is None:
        tau = max(ac.max() * 0.3, 1e-8)
    else:
        tau = tau_init
    tau_min = max(ac.max() * 0.001, 1e-12)
    tau_decay = 0.9995
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    lr = 1.0
    beta1 = 0.9
    beta2 = 0.999
    stall = 0
    max_stall = 1500
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        
        # Adaptive top_k: start with all peaks, gradually focus
        eff_top_k = None
        if top_k is not None:
            # Gradually increase focus
            progress = min(1.0, t / 3000.0)
            eff_top_k = max(1, int(top_k * (1.0 - 0.8 * progress)))
        
        g = lse_weighted_gradient(a, ac, tau, eff_top_k)
        
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        # Multiplicative update (log-space)
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        max_log = log_a.max()
        a = np.exp(log_a - max_log)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        # Periodic peak shaving
        if t % 200 == 0 and stall > 200:
            a, M_shave = peak_shaving_step(a, fast_autocorr(a), n_shave=8, n_cands=5)
            if M_shave < best_m:
                best_m = M_shave
                best_a = a.copy()
                stall = 0
        
        # Restart logic
        if stall > max_stall:
            stall = 0
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            # Simulated annealing: accept worse with small probability
            if rng.random() < 0.3:
                # Perturb from best
                noise = rng.normal(0, 0.05 * best_m, n)
                a = best_a + noise
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                # More aggressive restart
                a = best_a.copy()
                # Randomly zero out some fraction and redistribute
                n_zero = n // 10
                zero_idx = rng.choice(n, size=n_zero, replace=False)
                mass = a[zero_idx].sum()
                a[zero_idx] = 0
                a += mass / n
                a = a / a.sum()
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      Adam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}, stall={stall}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"      Adam done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m

def generate_structures(n, parent=None):
    """Generate diverse structural initializations."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    structures = []
    
    # Standard windows
    structures.append(('hann', np.sin(np.pi * x) ** 2))
    structures.append(('hamming', 0.54 - 0.46 * np.cos(2 * np.pi * x)))
    structures.append(('blackman', 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)))
    structures.append(('uniform', np.ones(n)))
    
    # Chirps (linearly varying frequency)
    for freqs in [4, 8, 16, 32]:
        a = np.abs(np.sin(np.pi * freqs * x + 2 * np.pi * 3 * x ** 2)) ** 2
        structures.append((f'chirp{freqs}', a))
    
    # Multi-lobe
    for nlobes in [2, 3, 4, 6]:
        a = np.abs(np.sin(np.pi * nlobes * x)) ** 2
        structures.append((f'lobe{nlobes}', a))
    
    # Asymmetric structures
    for asym in [0.3, 0.5, 0.7]:
        a = np.exp(-((x - asym) ** 2) / (2 * 0.15 ** 2))
        structures.append((f'asym{asym}', a))
    
    # Two-bump (asymmetric split)
    for center in [0.3, 0.4, 0.6, 0.7]:
        a = np.exp(-((x - center) ** 2) / (2 * 0.2 ** 2)) + 0.5 * np.exp(-((x - (1 - center)) ** 2) / (2 * 0.3 ** 2))
        structures.append((f'twobump{center}', a))
    
    # Power law (asymmetric)
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        structures.append((f'pow{p}', a))
        a = (n - i_arr) ** p
        structures.append((f'powR{p}', a))
    
    # Cosine with phase
    for phase in [0.1, 0.3, 0.5]:
        a = np.maximum(np.cos(2 * np.pi * (x - phase)), 0) ** 2
        structures.append((f'cosph{phase}', a))
    
    # Comb-like with taper
    for period in [4, 6, 8]:
        comb = (1 + np.cos(2 * np.pi * x * n // period)) / 2
        taper = np.sin(np.pi * x) ** 0.5
        a = comb * taper
        structures.append((f'comb{period}', a))
    
    # Add parent if available
    if parent is not None and len(parent) == n:
        structures.append(('parent', parent.copy()))
    elif parent is not None:
        # Interpolate parent to this length
        old_n = len(parent)
        p_norm = parent / parent.sum()
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, n)
        p_interp = np.interp(new_x, old_x, p_norm)
        structures.append(('parent_interp', np.maximum(p_interp, 0)))
    
    # Normalize all
    result = {}
    for name, s in structures:
        s = np.maximum(s, 0)
        ss = s.sum()
        if ss > 1e-10:
            result[name] = s / ss
    return result


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            candidates = np.where(sens < sens[src] * 0.5)[0]
            if len(candidates) == 0:
                candidates = np.where(sens < sens[src])[0]
            if len(candidates) == 0:
                candidates = np.argsort(sens)[:n // 4]
            candidates = candidates[candidates != src]
            
            if len(candidates) == 0:
                continue
            
            # Try a few fractions and pick best
            best_frac = None
            best_trial_m = M_curr
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                # Try a few random destinations
                n_try = min(3, len(candidates))
                cands = rng.choice(candidates, size=n_try, replace=False)
                for dst in cands:
                    trial = a.copy()
                    trial[src] -= amt
                    trial[int(dst)] += amt
                    trial = np.maximum(trial, 0)
                    s_t = trial.sum()
                    if s_t < 1e-15:
                        continue
                    trial = trial / s_t
                    M_t = fast_autocorr(trial).max()
                    if M_t < best_trial_m:
                        best_trial_m = M_t
                        best_frac = (frac, int(dst))
            
            if best_frac is not None and best_trial_m < M_curr - 1e-16:
                frac, dst = best_frac
                amt = a[src] * frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_curr = fast_autocorr(a).max()
                if M_curr < best_m:
                    best_m = M_curr
                    best_a = a.copy()
                    improved = True
                    stall = 0
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Polish done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-structure, multi-length, multi-phase approach."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = parent / parent.sum()
            p_score = evaluate_safe(p_norm)
            if np.isfinite(p_score):
                best_seq = p_norm.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            improvement = best_score - sc if np.isfinite(best_score) else 0
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s, -{improvement:.6f})")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Quick structural scan at parent length ===
    print(f"\n=== Phase 1: Structural scan (n={n_parent}) ===")
    t_phase_start = time.monotonic()
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    
    if t1 > 15:
        structures = generate_structures(n_parent, parent)
        # Quick eval all
        scored = []
        for name, s in structures.items():
            ac = fast_autocorr(s)
            score = 2.0 * n_parent * ac.max()
            scored.append((score, name, s))
        scored.sort(key=lambda x: x[0])
        
        # Evaluate top few with actual evaluate_sequence
        for score, name, s in scored[:5]:
            check_and_update(s, f"struct_{name}")
        
        elapsed = time.monotonic() - t_phase_start
        print(f"  Phase 1 done ({elapsed:.0f}s), best score so far: {best_score:.6f}")
    
    # === Phase 2: Multi-phase Adam on best ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multi-phase Adam (n={n_best}) ===")
    t2 = min(remaining * 0.30, 350.0)
    
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_peak_adam(a_in, t2, verbose=True, seed=42)
        check_and_update(a_out, f"multi_adam_n{n_best}")
    
    # === Phase 3: Fine polish ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Fine polish (n={n_best}) ===")
    t3 = min(remaining * 0.12, 150.0)
    
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = finer_polish(a_in, t3, verbose=True)
        check_and_update(a_out, f"polish_n{n_best}")
    
    # === Phase 4: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length exploration ===")
    
    # Try several nearby lengths
    length_options = []
    for delta in [300, -300, 600, -600, 1000, -1000, 1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000:
            length_options.append(new_n)
    
    # Also try some standard lengths
    for std_n in [3000, 4000, 5000, 6000]:
        if std_n != n_best:
            length_options.append(std_n)
    
    # Deduplicate and sort by proximity to best
    seen = set()
    unique_lengths = []
    for ln in sorted(length_options, key=lambda x: abs(x - n_best)):
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)
    
    for new_n in unique_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        
        print(f"\n  Trying n={new_n} (remaining: {remaining:.0f}s)")
        t_phase = min(remaining * 0.2, 200.0)
        if t_phase < 30:
            continue
        
        a_interp = interpolate_to_length(best_seq, new_n)
        
        # Quick multi-phase Adam
        t_adam = t_phase * 0.7
        a_out, _ = multi_peak_adam(a_interp, t_adam, verbose=False, seed=new_n)
        check_and_update(a_out, f"adam_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            # Quick polish
            a_out2, _ = finer_polish(norm_sum1(best_seq), min(remaining * 0.3, 60.0), verbose=False)
            check_and_update(a_out2, f"polish_n{new_n}")
    
    # === Phase 5: Aggressive final push on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final push (n={n_best}) ===")
    
    if remaining > 60:
        # Run Adam again with different seed and sharper focus
        t5a = min(remaining * 0.4, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = optimized_adam(a_in, t5a, tau_init=max(fast_autocorr(a_in).max() * 0.02, 1e-10),
                                   top_k=max(3, n_best // 200), verbose=True, seed=999)
        check_and_update(a_out, f"final_adam_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
    
    if remaining > 30:
        # Final polish
        t5b = min(remaining * 0.5, 100.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = finer_polish(a_in, t5b, verbose=True)
        check_and_update(a_out, f"final_polish_n{n_best}")
    
    # === Phase 6: One more Adam from a different angle ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    if remaining > 40:
        # Blend best with a fresh structure, then optimize
        x = np.linspace(0, 1, n_best)
        fresh = np.abs(np.sin(2 * np.pi * 5 * x + np.pi * 0.3)) ** 2
        fresh = norm_sum1(fresh)
        blend = 0.6 * norm_sum1(best_seq) + 0.4 * fresh
        blend = norm_sum1(blend)
        
        t6 = min(remaining * 0.5, 150.0)
        a_out, _ = multi_peak_adam(blend, t6, verbose=False, seed=7777)
        check_and_update(a_out, "blend_adam")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
