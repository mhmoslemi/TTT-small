Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative weight (mirror descent) optimization
2. LP-Frank-Wolfe with cutting planes for refinement
3. Adaptive peak-targeting with multiple lag tracking
4. Periodic restarts with parametric perturbations

import numpy as np
import time
import cvxpy as cp

def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def eval_score(a):
    """Compute J = 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return 2.0 * n * c.max() / (s * s)

def multiplicative_weight_optimize(a0, time_limit, top_k=50, verbose=True):
    """Main optimization using multiplicative weight updates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    rng = np.random.default_rng(123)
    eta = 2.0  # initial learning rate
    eta_decay = 0.9995
    stall = 0
    max_stall = 5000
    
    for t in range(1, 10_000_000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        M = c.max()
        score = 2.0 * n * M  # since sum(a) = 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            # Restart with perturbation
            pert = rng.normal(0, 0.02, n) * best_a
            a = norm_sum1(best_a + pert)
            x_prev = a.copy()
            stall = 0
            eta = max(eta, 1.0)
            continue
        
        # Find top K peak lags
        k_max = min(top_k, len(c) - 1)
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = c[top_idx]
        
        # Weights: softmax of top values
        beta_w = 3.0
        w = top_vals ** beta_w
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.ones(k_max) / k_max
        
        # Compute pressure P[i] = sum_j w_j * 2 * a[k_j - i]
        P = np.zeros(n)
        for jj in range(k_max):
            k = int(top_idx[jj])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                P[idx] += w[jj] * 2.0 * a[k - idx]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * P)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Nesterov-like momentum
        if t > 1:
            beta = (t - 1.0) / (t + 1.5)
            a_new = a_new + beta * (a_new - x_prev)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
        
        x_prev = a
        a = a_new
        
        # Decay learning rate
        if t % 1000 == 0:
            eta *= eta_decay
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={t} ({elapsed:.0f}s): J={score:.6f}, best={best_score:.6f}, eta={eta:.4f}")
    
    return best_a, best_score
