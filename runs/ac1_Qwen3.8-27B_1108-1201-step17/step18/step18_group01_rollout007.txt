Alternative: **Direct optimization of the max using smoothing**.

Instead of minimizing max_k c(k), minimize a smooth approximation:
f(T) = (1/T) * log(sum_k exp(T*c(k)))

For large T, this approximates the max. The gradient is:
df/da[i] = (1/T) * sum_k exp(T*c(k)) * 2*a[k-i] / sum_k exp(T*c(k))
= 2 * sum_k w_k * a[k-i] where w_k = exp(T*c(k)) / sum_j exp(T*c(j))

This gives a smooth subgradient that we can use for gradient descent. As T increases, it converges to the true subgradient of the max.

This is the **log-sum-exp smoothing** approach. Let me implement this.

Algorithm:
1. Normalize a to sum=1, a >= 0
2. Compute c = conv(a,a)
3. Compute weights w_k = exp(T*(c(k) - max_c)) / sum exp(T*(c(j) - max_c)) (softmax)
4. Compute gradient: grad[i] = 2 * sum_k w_k * a[k-i] (for valid indices)
5. Update: a_new = a * exp(-eta * grad) (mirror descent)
6. Normalize: a_new = a_new / sum(a_new)
7. Adapt T and eta

The key advantage of the log-sum-exp approach is that it gives a smooth gradient that accounts for all peaks, not just the top few. Peaks that are close to the maximum get high weight, so the gradient is "informative."

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent (multiplicative updates on simplex)
2. LP-based direction finding (Lagrangian relaxation of peak constraints)
3. Block-level local search with targeted perturbations
4. Coordinate descent polish
5. Multi-length exploration via interpolation
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def obj_value(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 without calling evaluate_sequence."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * M / (s * s)

For real signals: sum_m w[i+m] * a[m]

This can be computed as: ifft(fft(w) * conj(fft(a))) gives the cross-correlation... let me check.

Actually, the cross-correlation R[i] = sum_m w[m+i]*a[m] can be computed as:
R = ifft(fft(w, N) * conj(fft(a, N)))[:n]

where N >= 2n-1 for linear correlation.

Wait, let me be more careful. The standard formula:
- Cross-correlation: (f ⊛ g)[n] = sum_k f[k+n] * conj(g[k])
- In frequency: FFT(f ⊛ g) = conj(FFT(g)) * FFT(f)

So: R = ifft(FFT(w) * conj(FFT(a)))

And R[i] = sum_k w[k+i] * a[k] (for real signals, no conjugate needed)

Yes! So grad = 2 * R[:n] where R = ifft(fft(w) * conj(fft(a))).

Let me use this for O(N log N) computation instead of O(n^2).

Now let me also think about the LP approach. The LP is:
maximize: 1^T g

Where A[k, i] = a[k-i] if 0 <= k-i <= n-1, else 0, for the K peak lags k.

g_new = g_opt / sum(g_opt) (normalize to sum=1)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent with FFT-based gradient
2. LP-based direction finding for peak reduction
3. Block-level local search
4. Multi-length exploration
"""

import numpy as np
import time
import math
from scipy.optimize import linprog

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def obj_quick(a):
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * M / (s * s)

def lse_smoothed_grad(a, T, fft_size):
    """O(N log N) gradient using log-sum-exp smoothing."""
    n = len(a)
    ac = fast_autocorr(a)
    
    # Softmax weights (numerically stable)
    ac_s = ac - ac.max()
    exp_ac = np.exp(T * ac_s)
    w = exp_ac / exp_ac.sum()
    
    # Gradient via cross-correlation: grad[i] = 2 * sum_m w[i+m] * a[m]
    # = 2 * (w cross-corr a)[i]
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    R = np.fft.irfft(W * np.conj(A), fft_size)
    grad = 2.0 * R[:n]
    
    return grad, w, ac

def mirror_descent_optimize(a0, time_limit, T0, T1, verbose=False):
    """LSE-smoothed multiplicative gradient descent on simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    fft_size = 1 << (4 * n - 1).bit_length()
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 0.1
    patience = 0
    max_patience = 500
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T0 + (T1 - T0) * progress  # Anneal temperature
        
        grad, w, ac = lse_smoothed_grad(a, T, fft_size)
        M = ac.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        # Center the gradient (to preserve sum invariance)
        grad = grad - grad.mean()
        
        # Scale gradient
        gmax = np.abs(grad).max()
        if gmax < 1e-18:
            break
        grad = grad / gmax
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Accept/reject
        M_new = fast_autocorr(a_new).max()
        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.5
            if eta < 0.001:
                # Escape: add random perturbation
                rng = np.random.default_rng(iters)
                pert = rng.normal(0, 0.01, n)
                a = np.maximum(a + pert, 0)
                a = a / a.sum()
                eta = 0.05
                patience = max(0, patience - 200)
        
        iters += 1
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"      MD iter={iters} ({elapsed:.0f}s): J={J:.6f}, T={T:.1f}, eta={eta:.4f}")
    
    if verbose:
        J = 2.0 * n * best_M
        print(f"      MD done: J={J:.6f}, iters={iters}")
    return best_a

def lp_direction(a, M, K, time_budget=5.0):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for K peak lags, g >= 0."""
    n = len(a)
    ac = fast_autocorr(a)
    # Get K peak lags
    top_k = np.argsort(ac)[::-1][:K]
    top_k = top_k[top_k < 2*n-1]
    K_actual = len(top_k)
    
    if K_actual == 0:
        return None
    
    # Build constraint matrix A (K_actual x n)
    # A[k_idx, i] = a[k - i] if 0 <= k - i <= n-1, else 0
    A = np.zeros((K_actual, n))
    for idx, k in enumerate(top_k):
        k_int = int(k)
        # a[k-i] valid when 0 <= k-i <= n-1, i.e., k-n+1 <= i <= k
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            A[idx, i_start:i_end+1] = a[k_int - i_end:k_int - i_start:-1]
    
    # max sum(g) s.t. A@g <= M*ones, g >= 0
    # In linprog form: min -sum(g) s.t. A@g <= M
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(K_actual, M), bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        g = res.x
        if g.sum() > 1e-10:
            return g / g.sum()
    return None
