def lp_flattening_direction(a, M, top_k_constraints=None):
    """Solve LP to find direction b that flattens the convolution.
    
    max sum(b) s.t. conv(a,b)[k] <= M for all k (or top k), b >= 0.
    """
    n = len(a)
    # Build constraint matrix
    # For each k in 0..2n-2: conv(a,b)[k] = sum_{i} a[i]*b[k-i] <= M
    if top_k_constraints is None:

        c = fast_conv(a)
        # Take top 50 peaks and 50 random other positions
        top_idx = np.argsort(c)[::-1][:50]
        remaining = np.setdiff1d(np.arange(len(c)), top_idx)
        rng = np.random.default_rng(42)
        sample_idx = rng.choice(remaining, size=min(50, len(remaining)), replace=False)
        constraint_idx = np.concatenate([top_idx, sample_idx])
    else:
        constraint_idx = top_k_constraints
    
    m = len(constraint_idx)
    # Build sparse matrix
    # Row constraint_idx[j]: conv(a,b)[k] for k = constraint_idx[j]
    rows = []
    cols = []
    vals = []
    for j, k in enumerate(constraint_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):

def spectral_smooth(a, alpha=0.3):
    """Compress dynamic range of magnitude spectrum."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - alpha)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return (1 - 0.5 * alpha) * a + 0.5 * alpha * a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based flattening direction finding (solving max sum(b) s.t. conv(a,b)<=M, b>=0)
2. Log-sum-exp smoothed subgradient descent with annealing
3. Spectral amplitude compression
4. Multi-initialization and length variation strategy
5. Iterative LP + FTRL hybrid refinement
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

_USE_LP = True


def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lp_flattening_direction(a, M, top_k=150, rng_seed=42):
    """Solve LP to find direction b that flattens the convolution.
    
    max sum(b) s.t. conv(a,b)[k] <= M for selected constraints, b >= 0.
    Returns (b_direction, success_flag).
    """
    n = len(a)
    c = fast_conv(a)
    num_c = len(c)
    
    # Select constraints: top peaks + random samples
    k_sel = min(top_k, num_c)
    top_idx = np.argsort(c)[::-1][:k_sel]
    if k_sel < 300:
        remaining = np.setdiff1d(np.arange(num_c), top_idx)
        if len(remaining) > 0:
            rng = np.random.default_rng(rng_seed)
            n_sample = min(100, len(remaining))
            sample_idx = rng.choice(remaining, size=n_sample, replace=False)
            top_idx = np.concatenate([top_idx, sample_idx])
    
    m = len(top_idx)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for j, k in enumerate(top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            if a[i] > 1e-18:
                rows.append(j)
                cols.append(k_int - i)
                vals.append(a[i])
    
    if len(rows) == 0:
        return None, False
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(m, n))
    
    # Solve: maximize sum(b) => minimize -sum(b)
    c_obj = np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(-c_obj, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
    except Exception:
        return None, False
    
    if res.status not in (0, 3):  # 0=optimal, 3=time limit reached (still use solution)
        return None, False
    
    if res.x is None:
        return None, False
    
    b_dir = np.maximum(res.x, 0.0)
    s = b_dir.sum()
    if s < 1e-15:
        return None, False
    
    return b_dir / s, True


def ftrl_optimize(a0, time_limit, verbose=False):
    """FTRL mirror descent with annealed log-sum-exp smoothing."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, t / 40000.0)
        beta = 15.0 + 70.0 * progress
        top_k = max(5, int(25 * (1.0 - 0.7 * progress) + 5))
        eta = 0.8 / (np.sqrt(t) + 1.0)
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        c = fast_conv(y)
        
        # Top-k weighted subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx].copy()
        max_v = top_vals[0]
        if max_v < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            exponents = beta * (top_vals - max_v) / max_v
            exponents = np.clip(exponents, -50.0, 0.0)
            w = np.exp(exponents)
            ws = w.sum()
            if ws < 1e-30:
                w = np.ones(top_k) / top_k
            else:
                w = w / ws
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror descent update (entropy-regularized)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x = a_new
        a = a_new
        
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): J_max={best_m:.10f}")
    
    return best_a, best_m


def spectral_smooth(a, alpha=0.25):
    """Compress dynamic range of magnitude spectrum (one step)."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - alpha)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return (1.0 - 0.5 * alpha) * a + 0.5 * alpha * a_new


def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening with adaptive step size."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    step = 0.25
    stall = 0
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 80:
            break
        
        a = spectral_smooth(a, alpha=step)
        c = fast_conv(a)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            step = min(0.45, step * 1.03)
            stall = 0
        else:
            step = max(0.05, step * 0.85)
            stall += 1
            if step < 0.06:
                # Blend back toward best
                a = norm_simplex(0.7 * a + 0.3 * best_a)
    
    return best_a, best_m


def lp_iterate(a0, time_limit, max_iters=150, verbose=False):
    """LP-based iterative refinement with line search."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        # Solve LP for direction
        b_dir, ok = lp_flattening_direction(a, M, top_k=200, rng_seed=it)
        if not ok or b_dir is None:
            break
        
        # Line search: try multiple step sizes
        best_t = 0.0
        best_m_try = M
        for t in [0.2, 0.1, 0.15, 0.3, 0.05, 0.4, 0.5]:
            a_try = norm_simplex((1.0 - t) * a + t * b_dir)
            m_try = fast_conv(a_try).max()
            if m_try < best_m_try:
                best_m_try = m_try
                best_t = t
        
        if best_t > 0:
            a = norm_simplex((1.0 - best_t) * a + best_t * b_dir)
            if best_m_try < best_m:
                best_m = best_m_try
                best_a = a.copy()
        else:
            # Stuck: try a random perturbation and re-solve
            a = norm_simplex(a + rng.normal(0, 0.01, n))
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it} ({elapsed:.0f}s): M={best_m:.10f}, t={best_t:.3f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            ftrl_time = min(2.0, remaining * 0.3)
            a_ftrl, m_ftrl = ftrl_optimize(a, ftrl_time)
            if m_ftrl < best_m:
                best_m = m_ftrl
                best_a = a_ftrl.copy()
            a = a_ftrl
        
        it += 1
        
        # Periodic spectral smoothing
        if it % 40 == 0 and remaining > 3:
            a = spectral_smooth(a, alpha=0.15)
            c = fast_conv(a)
            if c.max() < best_m:
                best_m = c.max()
                best_a = a.copy()
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining LP, FTRL, and spectral methods."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_safe(parent)
            best_seq = parent.copy()
            best_score = parent_score
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.6f}")
    except Exception:
        pass
    
    # Fallback initialization
    if best_seq is None:
        n_base = 4200
        best_seq = make_init_bowl(n_base, sharpness=2.0, base=0.1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}, budget={time_budget:.0f}s")
    
    # === Phase 1: LP-based optimization from parent (most promising) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LP Iteration from parent (n={n0}, {t1:.0f}s) ===")
        a_in = norm_simplex(best_seq)
        a_out, m_out = lp_iterate(a_in, t1, max_iters=80, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Phase 1 improved: {sc:.6f}")
        else:
            print(f"  Phase 1: no improvement (best={sc:.6f} vs {best_score:.6f})")
    
    # FTRL polish after Phase 1
    remaining = time_budget - (time.monotonic() - t_global)
    t1b = min(remaining * 0.10, 120.0)
    if t1b > 15:
        print(f"\n=== Phase 1b: FTRL polish ({t1b:.0f}s) ===")
        a_in = norm_simplex(best_seq)
        a_out, _ = ftrl_optimize(a_in, t1b)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** FTRL polish improved: {sc:.6f}")
    
    # === Phase 2: Hybrid LP+FTRL on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Hybrid LP+FTRL (n={len(best_seq)}, {t2:.0f}s) ===")
        a_in = norm_simplex(best_seq)
        a_out, _ = hybrid_optimize(a_in, t2)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Hybrid improved: {sc:.6f}")
    
    # === Phase 3: LP from diverse initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Diverse LP inits ({t3:.0f}s) ===")
        n0 = len(best_seq)
        
        init_list = []
        label_list = []
        
        for sharp in [1.5, 2.5, 3.5, 5.0]:
            init_list.append(make_init_bowl(n0, sharpness=sharp, base=0.05))
            label_list.append(f"bowl_{sharp}")
        
        init_list.append(make_init_cosine(n0, mode=1))
        label_list.append("cos1")
        
        init_list.append(make_init_dual_peak(n0, 0.15, 0.85, 0.1))
        label_list.append("dual_15_85")
        
        init_list.append(make_init_dual_peak(n0, 0.1, 0.9, 0.08))
        label_list.append("dual_10_90")
        
        for seed in [1, 2, 3]:
            init_list.append(make_init_random_smooth(n0, seed=seed))
            label_list.append(f"rand_{seed}")
        
        # Also try the parent perturbed
        rng_p = np.random.default_rng(55)
        init_list.append(norm_simplex(parent + rng_p.normal(0, 0.02, n0)))
        label_list.append("parent_pert")
        
        t_per = t3 / len(init_list)
        for idx, (init, lbl) in enumerate(zip(init_list, label_list)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            t_avail = min(t_per, remaining - 5)
            if t_avail < 10:
                break
            
            a_out, _ = lp_iterate(init, t_avail, max_iters=25, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** {lbl} improved: {sc:.6f}")
            else:
                print(f"    {lbl}: {sc:.6f} (no improvement)")
    
    # === Phase 4: Spectral optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Spectral opt ({t4:.0f}s) ===")
        a_in = norm_simplex(best_seq)
        a_out, _ = spectral_optimize(a_in, t4)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Spectral improved: {sc:.6f}")
        
        # FTRL after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out2, _ = ftrl_optimize(a_out, min(remaining * 0.5, 80.0))
            sc2 = evaluate_safe(a_out2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_out2.copy()
                print(f"  *** Post-spectral FTRL improved: {sc2:.6f}")
    
    # === Phase 5: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        print(f"\n=== Phase 5: Length variation ({t5:.0f}s) ===")
        n_best = len(best_seq)
        
        for delta in [300, 600, -300, -600]:
            new_n = n_best + delta
            if new_n < 2000 or new_n > 8000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 25:
                break
            
            a_interp = interpolate_sequence(best_seq, new_n)
            sc = evaluate_safe(a_interp)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_interp.copy()
                print(f"  *** Interp n={new_n}: {sc:.6f}")
            
            # Quick LP on interpolated
            t_avail = min(t5 * 0.25, remaining - 10)
            if t_avail > 10:
                a_out, _ = lp_iterate(a_interp, t_avail, max_iters=15, verbose=False)
                sc2 = evaluate_safe(a_out)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_out.copy()
                    print(f"  *** LP n={new_n}: {sc2:.6f}")
    
    # === Phase 6: Final intensive optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.40, 300.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Final intensive ({t6:.0f}s) ===")
        a_in = norm_simplex(best_seq)
        
        # Hybrid for the bulk
        t6a = t6 * 0.6
        a_out, _ = hybrid_optimize(a_in, t6a)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Final hybrid: {sc:.6f}")
        
        # FTRL for the rest
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.5, 100.0)
        if t6b > 10:
            a_in2 = norm_simplex(best_seq)
            a_out2, _ = ftrl_optimize(a_in2, t6b, verbose=True)
            sc2 = evaluate_safe(a_out2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_out2.copy()
                print(f"  *** Final FTRL: {sc2:.6f}")
    
    # === Phase 7: Very final LP push ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 7: Final LP push ({remaining:.0f}s) ===")
        a_in = norm_simplex(best_seq)
        t7 = min(remaining - 10, 80.0)
        if t7 > 5:
            a_out, _ = lp_iterate(a_in, t7, max_iters=20, verbose=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** Final LP: {sc:.6f}")
    
    # Final validation
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL RESULT: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"\nFALLBACK: score={best_score:.6f}, n={n_fb}")
    
    return [float(x) for x in best_seq]
