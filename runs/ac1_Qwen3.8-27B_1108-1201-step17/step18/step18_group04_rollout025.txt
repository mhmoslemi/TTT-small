        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])

        # (how much each a[i] contributes to the peak)
        # Then find a direction that reduces all peaks simultaneously
        
        # Strategy: for each peak k, the contribution of a[i] is a[k-i]
        # We want to reduce the total: sum_k w_k * sum_i a[i]*a[k-i]
        # which is the same as the smoothed objective

        contributions = np.zeros(n)
        for k in peaks:

            hi = min(n - 1, k_int)
            idx = np.arange(lo, hi + 1)
            contributions[idx] += a[k_int - idx]
        
        # Find the top contributors to the peaks
        top_contrib = np.argsort(contributions)[::-1][:max(10, n // 100)]
        
        # Try moving mass from high-contributing positions to low-contributing ones
        low_contrib = np.argsort(contributions)[:max(10, n // 100)]
        
        improved = False
        for src in top_contrib[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            for dst in low_contrib[:10]:

                if dst_i == src_i:

                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial /= s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:

                break
        
        if not improved:

                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
    
    return best_a, best_m

def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

    x_c = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm_sum1(x_c ** (-power))

def make_exponential(n, rate=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm_sum1(np.exp(-rate * x))

def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.1 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)
