    beta = 1.0

        top_vals = c[top_idx]

        # Gradient in a-space
        grad_a = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)

            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx

        # If a = softmax(b), then d/db = (a_i * grad_a_i) (approximately, for the log-sum-exp parametrization)

        prev_grad = velocity

        eta_t = eta / (1.0 + 0.1 * t)

        b_new = b - eta_t * velocity

import numpy as np
import time

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def evaluate_safe(a):
    """Safely evaluate the sequence using the provided evaluate_sequence."""
    try:

def subgradient_multi_peak(a, c, n_peaks=30, beta=50.0):
    """Compute subgradient using multiple peaks with temperature-based weighting."""
    n = len(a)
    k_max = min(n_peaks, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()
    
    max_v = top_vals[0]
    min_v = top_vals[-1]
    range_v = max_v - min_v
    
    if range_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        # Exponential weighting based on peak prominence
        normalized = (top_vals - min_v) / range_v
        w = np.exp(beta * (normalized - 1.0))
        w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)

            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def spectral_step(a, strength=0.3):
    """One step of spectral peak shaving."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Compress dynamic range of amplitude
    mag_max = mag.max()
    mag_min = mag.min()
    if mag_max < 1e-30:
        return a
    
    # Normalize and apply power compression
    mag_norm = mag / mag_max
    mag_new = mag_max * mag_norm ** (1.0 + strength)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    # Blend
    a_out = (1.0 - strength * 0.5) * a + strength * 0.5 * a_new
    return norm_simplex(a_out)

def adam_logspace(a0, time_limit, verbose=False):
    """Adam optimizer in log-space."""
    n = len(a0)
    a = norm_simplex(a0)
    b = np.log(a + 1e-15)  # log-space parameter
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        g_a = subgradient_multi_peak(a, c, n_peaks=20, beta=40.0)
        
        # Natural gradient in log-space
        g_b = a * g_a
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g_b
        v = beta2 * v + (1 - beta2) * g_b ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to simplex
        a_new = np.exp(b - b.max())
        a_new = a_new / a_new.sum()
        
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and reduce lr
            b = np.log(a + 1e-15)
            lr = max(0.0001, lr * 0.95)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0

def hill_climb(a0, time_limit, verbose=False):
    """Hill climbing with diverse move types."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    step_size = 0.05
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            break
        
        # Choose move type
        move_type = rng.integers(4)
        a_trial = a.copy()
        
        if move_type == 0:
            # Single position nudge
            i = int(rng.integers(n))
            delta = rng.normal(0, step_size / np.sqrt(n))
            a_trial[i] += delta
        elif move_type == 1:
            # Pair move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = rng.uniform(-step_size, step_size) / n
                a_trial[i] -= amt
                a_trial[j] += amt
        elif move_type == 2:
            # Block nudge
            width = int(rng.integers(5, 50))
            i = int(rng.integers(max(1, n - width)))
            delta = rng.normal(0, step_size / np.sqrt(width))
            a_trial[i:i+width] += delta
        else:
            # Reshape: multiply by a smooth function
            x = np.linspace(0, 1, n)
            f = 1 + step_size * np.sin(2 * np.pi * rng.uniform(1, 10) * x + rng.uniform(0, 2*np.pi))
            a_trial = a * f
        
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        
        M_trial = fast_conv(a_trial).max()
        if M_trial < best_m * (1 - 1e-10):
            best_m = M_trial
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            step_size = min(0.1, step_size * 1.01)
        else:
            no_improve += 1
            step_size = max(0.001, step_size * 0.98)
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv(a):

    n = len(a)

def norm_simplex(a):

def refined_subgradient(a, c, threshold_frac=0.97, beta=80.0, max_peaks=50):
    """Compute subgradient using all peaks above threshold with exponential weighting."""
    n = len(a)
    max_c = c.max()

        w_sum = w.sum()
        if w_sum < 1e-30:

        else:
            w = w / w_sum

        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)

        idx = np.arange(i_start, i_end)

def spectral_step_simple(a, strength=0.3):
    """One step of spectral peak shaving (compress amplitude dynamic range)."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    mag_max = mag.max()
    if mag_max < 1e-30:
        return a

    mag_norm = mag / mag_max
    mag_new = mag_max * mag_norm ** (1.0 + strength)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s

    a_out = (1.0 - strength * 0.5) * a + strength * 0.5 * a_new
    return norm_simplex(a_out)


def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening with adaptive strength."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    strength = 0.25
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        a_new = spectral_step_simple(a, strength=strength)
        c = fast_conv(a_new)
        M = c.max()

        if M < best_m * (1 - 1e-10):
            best_m = M
            best_a = a_new.copy()
            a = a_new
            strength = min(0.6, strength * 1.03)
        else:
            strength = max(0.05, strength * 0.85)
            if it > 200:
                a = norm_simplex(0.7 * a + 0.3 * best_a)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Spectral it={it} ({elapsed:.0f}s): score={2*n*best_m:.7f}")

    return best_a, best_m


def adam_logspace(a0, time_limit, verbose=False):
    """Adam optimizer in log-space with adaptive per-coordinate learning rates."""
    n = len(a0)
    a = norm_simplex(a0)
    b = np.log(a + 1e-15)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    lr = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(a)
        g_a = refined_subgradient(a, c, threshold_frac=0.97, beta=60.0)
        g_b = a * g_a

        m = beta1 * m + (1 - beta1) * g_b
        v = beta2 * v + (1 - beta2) * g_b ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        b_new = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.exp(b_new - b_new.max())
        a_new = a_new / a_new.sum()

        c_new = fast_conv(a_new)
        M_new = c_new.max()

        if M_new < best_m * (1 - 1e-10):
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            b = b_new
            no_improve = 0
        else:
            no_improve += 1
            lr = max(1e-5, lr * 0.97)
            if no_improve > 3000:
                # Restart from best with perturbation
                rng = np.random.default_rng(t)
                a = norm_simplex(best_a + rng.normal(0, 0.02, n))
                b = np.log(a + 1e-15)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.005
                no_improve = 0

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={t} ({elapsed:.0f}s): score={2*n*best_m:.7f}, lr={lr:.6f}")

    return best_a, best_m


def quadratic_peak_reduction(a0, k_target, max_iter=200):
    """Reduce a specific peak conv(a,a)[k_target] via quadratic projection."""
    n = len(a0)
    a = norm_simplex(a0)
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > 5.0:
            break

        c = fast_conv(a)
        val = c[k_target]
        grad = single_lag_gradient(a, k_target)

        # Project: a - mu * grad, choose mu to reduce val
        # val' ≈ val - 2*mu*(grad . a) + mu^2*(grad . Q grad)
        # Simplification: use line search
        best_mu = 0.0
        best_val = val
        for mu_frac in [0.01, 0.03, 0.1, 0.3, 1.0, 3.0]:
            mu = mu_frac * val / (np.dot(grad, a) + 1e-30)
            if mu <= 0:
                mu = 0.01 * mu_frac
            a_trial = a - mu * grad
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            c_trial = fast_conv(a_trial)
            # Check that we don't increase other peaks too much
            if c_trial[k_target] < best_val and c_trial.max() < best_val * 2:
                best_val = c_trial[k_target]
                best_mu = mu
                best_a_trial = a_trial.copy()

        if best_mu > 0:
            a = best_a_trial
        else:
            break

    return a


def iterated_local_search(a0, time_limit, verbose=False):
    """ILS: perturb + re-optimize cycle."""
    n = len(a0)
    best_a = norm_simplex(a0)
    best_m = fast_conv(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        # Perturb
        pert_scale = max(0.005, 0.03 * (1 - 0.5 * min(1.0, iter_count / 20.0)))
        perturbed = norm_simplex(best_a + rng.normal(0, pert_scale, n))

        # Short re-optimization
        t_avail = time_limit - (time.monotonic() - t0)
        t_opt = min(t_avail * 0.3, 30.0)
        if t_opt < 3:
            break

        a_out, _ = ftrl_refined(perturbed, t_opt, verbose=False)
        c = fast_conv(a_out)
        M = c.max()

        if M < best_m * (1 - 1e-10):
            best_m = M
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      ILS iter={iter_count} ({elapsed:.0f}s): score={2*n*best_m:.7f}")

    return best_a, best_m


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return norm_simplex(a)


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.3, pos2=0.7, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.7f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.7f}")

    # === Phase 1: Intensive FTRL from parent (40% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FTRL refined from parent (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = ftrl_refined(a_in, t1, verbose=True, thr_frac=0.97, beta_w=80.0)
        try_update(a_out, "ftrl_refined_parent")

        # Try with different parameters
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t1b = min(remaining * 0.2, 200.0)
            a_out2, _ = ftrl_refined(a_in, t1b, verbose=False, thr_frac=0.95, beta_w=50.0)
            try_update(a_out2, "ftrl_refined_parent_v2")

    # === Phase 2: Adam in log-space (20% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam log-space (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.2, 250.0)
    if t2 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = adam_logspace(a_in, t2, verbose=True)
        try_update(a_out, "adam_logspace")

    # === Phase 3: Spectral optimization (10% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.1, 150.0)
    if t3 > 15:
        a_in = norm_simplex(best_seq)
        a_out, _ = spectral_optimize(a_in, t3, verbose=False)
        try_update(a_out, "spectral")

    # === Phase 4: Iterated Local Search (15% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: ILS (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = iterated_local_search(a_in, t4, verbose=True)
        try_update(a_out, "ils")

    # === Phase 5: Multi-length exploration (10% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length variation ===")
    for delta in [-500, -200, 200, 500]:
        new_n = n0 + delta
        if new_n < 2000 or new_n > 7000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        a_interp = interpolate_sequence(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")

        t5 = min(remaining * 0.1, 80.0)
        if t5 > 15:
            a_out, _ = ftrl_refined(a_interp, t5, verbose=False)
            try_update(a_out, f"ftrl_n{new_n}")

    # === Phase 6: Quadratic peak reduction on worst peaks ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Quadratic peak reduction ===")
    if remaining > 20:
        a_in = norm_simplex(best_seq)
        c = fast_conv(a_in)
        # Get top 5 peaks
        top_peaks = np.argsort(c)[::-1][:5]
        a_work = a_in.copy()
        for k_pk in top_peaks:
            a_work = quadratic_peak_reduction(a_work, k_pk, max_iter=100)
            c_check = fast_conv(a_work)
            if c_check.max() < best_score / (2 * len(a_work)):
                try_update(a_work, f"quad_peak_{int(k_pk)}")

    # === Phase 7: Final combined polish (remaining budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 30:
        a_in = norm_simplex(best_seq)
        # FTRL
        t7a = min(remaining * 0.4, 200.0)
        if t7a > 15:
            a_out, _ = ftrl_refined(a_in, t7a, verbose=False, thr_frac=0.98, beta_w=100.0)
            try_update(a_out, "ftrl_final")

        # Spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = spectral_optimize(norm_simplex(best_seq), min(remaining * 0.3, 60.0), verbose=False)
            try_update(a_out, "spectral_final")

        # Adam
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = adam_logspace(norm_simplex(best_seq), min(remaining * 0.3, 60.0), verbose=False)
            try_update(a_out, "adam_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_init_cosine(4000, 1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.7f}")

    return [float(x) for x in best_seq]
