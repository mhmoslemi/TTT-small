"""

2. Exact subgradient mirror descent with Nesterov acceleration
3. Multi-length search with shape interpolation

import numpy as np
import time


def fast_autocorr(a):
    """Compute full convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_linear_ramp(n):
    """a[i] = (i+1)/S, the asymptotically optimal shape among power laws."""
    x = np.arange(1, n + 1, dtype=np.float64)
    return norm_sum1(x)


def make_truncated_ramp(n, trunc_frac=0.5):
    """Ramp up to trunc_frac*n, then flat."""
    i = np.arange(n, dtype=np.float64)
    cap = trunc_frac * n
    a = np.minimum(i + 1, cap)
    return norm_sum1(a)

def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgrad(a, ac, n_top=5):
    """Exact subgradient using top-n convolution peaks."""
    n = len(a)
    top_k = np.argsort(ac)[::-1][:n_top]
    grad = np.zeros(n)
    for k in top_k:
        ki = int(k)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * a[pair_idx]
    return grad

                improved = False
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

                if not improved:
                    # Random exploration
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if not improved:
            stall += 1

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      PEAK t={stall} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def parametric_search(n, time_limit):
    """Quick search over parametric shapes for a given n."""
    t0 = time.monotonic()
    best_a = None
    best_score = float('inf')

    candidates = []
    # Linear ramp (best theoretical)
    candidates.append((make_linear_ramp(n), "lin"))
    # Truncated ramps
    for tf in [0.3, 0.5, 0.7]:
        candidates.append((make_truncated_ramp(n, tf), f"trunc_{tf}"))
    # Power ramps
    for p in [0.8, 0.9, 1.0, 1.1, 1.2, 1.5]:
        candidates.append((make_power_ramp(n, p), f"pow_{p}"))
    # Mixed ramps
    for al in [0.1, 0.3, 0.5, 1.0]:
        candidates.append((make_mixed_ramp(n, al), f"mix_{al}"))

    for a, label in candidates:
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    if best_a is not None:
        print(f"    Parametric n={n}: best J={best_score:.6f}")

    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = norm_sum1(parent)
    except Exception:
        pass

    # Phase 1: Parametric shape search at multiple lengths
    print(f"\n=== Phase 1: Parametric search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_p1 = min(remaining * 0.10, 120.0)

    if t_p1 > 20:
        lengths = [3000, 5000, 7000, 10000, 15000]
        if parent is not None:
            n_par = len(parent)
            # Add parent length and nearby
            lengths.append(n_par)
            lengths.append(max(2000, n_par - 1000))
            lengths.append(n_par + 1000)
        lengths = sorted(set(l for l in lengths if 2000 <= l <= 20000))

        t_per_len = t_p1 / len(lengths)
        for n try_len in lengths:
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per_len, time_budget - 60 - elapsed_so_far)
            if t_avail < 5:
                break
            print(f"  Parametric n={n} (t={t_avail:.0f}s)...")
            a_loc, sc_loc = parametric_search(n, t_avail)
            if a_loc is not None:
                try_update(a_loc, f"param_n{n}")

    # Phase 2: Gradient optimization on best from Phase 1 at multiple lengths
    print(f"\n=== Phase 2: Gradient optimization (multi-length) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else 5000

    # Determine lengths to optimize
    grad_lengths = set()
    grad_lengths.add(n_best)
    grad_lengths.add(5000)
    grad_lengths.add(7000)
    grad_lengths.add(10000)
    if parent is not None:
        grad_lengths.add(len(parent))
    grad_lengths = sorted(set(l for l in grad_lengths if 2000 <= l <= 20000))

    t_per_grad = min(remaining * 0.50 / len(grad_lengths), 300.0)

    for n_g in grad_lengths:
        elapsed_so_far = time.monotonic() - t_global
        if elapsed_so_far > time_budget - 60:
            break
        t_avail = min(t_per_grad, time_budget - 60 - elapsed_so_far)
        if t_avail < 15:
            break

        print(f"  GRAD n={n_g} (t={t_avail:.0f}s)...")

        # Initialize from best known shape, interpolated
        if best_seq is not None:
            a_init = interpolate_seq(best_seq, n_g)
        else:
            a_init = make_linear_ramp(n_g)

        # Run gradient optimization
        a_out, _ = gradient_optimize(a_init, t_avail * 0.7, n_top=5,
                                      c0=8.0, verbose=False, rng_seed=42)
        try_update(a_out, f"grad_n{n_g}")

        # Quick polish with peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        t_p2b = min(remaining * 0.05, 30.0)
        if t_p2b > 10:
            a_out2, _ = local_peak_shaving(a_init, t_p2b, verbose=False, rng_seed=7)
            try_update(a_out2, f"peak_n{n_g}")

    # Phase 3: Aggressive refinement on the best sequence
    print(f"\n=== Phase 3: Aggressive refinement ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else 5000

    if remaining > 60:
        # Multiple gradient runs with different parameters
        configs = [
            (10, 6.0, 2000, 100),   # n_top, c0, restart, seed
            (3, 10.0, 5000, 200),
            (7, 5.0, 3000, 300),
            (5, 12.0, 4000, 400),
            (12, 4.0, 6000, 500),
        ]

        t_per_conf = min(remaining * 0.60 / len(configs), 200.0)
        for idx, (nt, c, rst, s) in enumerate(configs):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per_conf, time_budget - 60 - elapsed_so_far)
            if t_avail < 15:
                break
            print(f"  REFINEMENT cfg{idx} (n_top={nt}, c0={c}, t={t_avail:.0f}s)...")
            a_in = norm_sum1(best_seq)
            a_out, _ = gradient_optimize(a_in, t_avail, n_top=nt, c0=c,
                                          verbose=False, rng_seed=s, restart_every=rst)
            try_update(a_out, f"refine_cfg{idx}_n{n_best}")

    # Phase 4: Peak shaving on best
    print(f"\n=== Phase 4: Peak shaving ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else 5000

    if remaining > 30:
        t_p4 = min(remaining * 0.30, 150.0)
        print(f"  PEAK (t={t_p4:.0f}s)...")
        a_in = norm_sum1(best_seq)
        a_out, _ = local_peak_shaving(a_in, t_p4, verbose=True, rng_seed=99)
        try_update(a_out, f"peak_final_n{n_best}")

    # Phase 5: Final gradient polish
    print(f"\n=== Phase 5: Final polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t_p5 = min(remaining * 0.50, 200.0)
        print(f"  FINAL GRAD (t={t_p5:.0f}s)...")
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_optimize(a_in, t_p5, n_top=3, c0=15.0,
                                      verbose=True, rng_seed=777, restart_every=8000)
        try_update(a_out, f"final_grad_n{len(a_out)}")

    # Phase 6: Try slightly different lengths around best
    print(f"\n=== Phase 6: Length fine-tuning ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else 5000

    if remaining > 40:
        for delta in [-500, -200, 200, 500, 1000]:
            new_n = n_best + delta
            if new_n < 2000 or new_n > 20000:
                continue
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 30:
                break
            t_avail = min(remaining * 0.20, 80.0)
            if t_avail < 10:
                break
            a_int = interpolate_seq(best_seq, new_n)
            try_update(a_int, f"interp_n{new_n}")
            a_out, _ = gradient_optimize(a_int, t_avail * 0.8, n_top=5, c0=8.0,
                                          verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"grad_n{new_n}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = make_linear_ramp(5000)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
