def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a = np.maximum(a_arr, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_gradient(a, ac, tau):
    """Compute gradient of LSE-smoothed max autocorrelation."""
    n = len(a)
    ac_scaled = ac / tau
    ac_scaled -= ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()
    a_flip = a[::-1].astype(np.float64)
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)


def smooth_adam(a0, time_limit, tau_decay=0.9998, lr=0.7, beta1=0.9, beta2=0.999,
                restart_every=2000, verbose=True):
    """Multiplicative Adam with LSE smoothing and adaptive tau."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    tau = max(ac.max() * 0.5, 1e-6)
    tau_min = max(ac.max() * 0.003, 1e-10)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        tau = max(tau * tau_decay, tau_min)
        g = lse_gradient(a, ac, tau)
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        step = lr / (np.sqrt(v_c) + 1e-8)
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new
        if stall > restart_every:
            stall = 0
            # Multiple perturbation strategies
            strat = t % 3
            if strat == 0:
                pert = rng.normal(0, 0.02, n) * best_a
                x = best_a + pert
            elif strat == 1:
                # Move mass from peak to trough
                peak_pos = np.argmax(a)
                trough_pos = np.argmin(a)
                amt = a[peak_pos] * 0.1
                x = a.copy()
                x[peak_pos] -= amt
                x[trough_pos] += amt
            else:
                # Random swap
                i, j = rng.integers(0, n, 2)
                x = a.copy()
                amt = a[i] * 0.05
                x[i] -= amt
                x[j] += amt
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SmoothAdam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothAdam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tau={tau:.2e}")
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    SmoothAdam done ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m


def lp_frank_wolfe_v3(a0, time_limit, solver, initial_K=800, max_K=5000,

    constraint_set = set(int(k) for k in init_idx)
    iter_count = 0
    lp_fail_count = 0

            break
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall_count = 0
        elif M_curr < last_best_m - 1e-10:
            stall_count = 0
        else:
            stall_count += 1
        
        # Re-anchor if stalled

            stall_count = 0

            x = best_a.copy()
            # Reset constraint set to top peaks of best
            ac_best = fast_autocorr(best_a)
            top_k = np.argsort(ac_best)[::-1][:initial_K]
            constraint_set = set(int(k) for k in top_k)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP-FW3 re-anchor at t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
        
        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
        
        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            # Prune: keep constraints with smallest margin
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 10 and len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in np.argsort(ac_curr)[::-1][:initial_K])
            continue

        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # FW step with d2t (diminishing to target) step size
        alpha = min(0.3, 2.0 / (t % (reanchor_every * 2) + 2.0))
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Accelerated FW (FTRL-style)
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        x = x_new
        iter_count += 1

            elapsed = time.monotonic() - t0

            for k2 in top_k[:5]:

                        a_best_try = a_trial.copy()
                        M_best_try = M_trial
                    else:
                        hi = mid
                if lo > 0:
                    a = a_best_try
                    best_m = min(best_m, M_best_try)
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

def interpolate_sequence(f, new_n):
    """Linearly interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_init(n, freq=1, rng_seed=0):
    """Generate a sequence with approximately flat spectrum."""
    rng = np.random.default_rng(rng_seed)
    # Start with random phases, set magnitude to be flat
    fft_size = 1 << (2 * n - 1).bit_length()
    mag = np.ones(fft_size // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    X = mag * np.exp(1j * phase)
    # Enforce Hermitian symmetry
    X_full = np.zeros(fft_size, dtype=complex)
    X_full[:fft_size // 2 + 1] = X
    for i in range(1, fft_size // 2):
        X_full[fft_size - i] = np.conj(X[i])
    a = np.fft.irfft(X_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def parametric_init(n):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    for freq in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.01)
            candidates.append(a / a.sum())
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(a / a.sum())
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())
    candidates.append(np.ones(n) / n)
    # Spectral inits
    for seed in [0, 1, 2]:
        candidates.append(spectral_init(n, rng_seed=seed))
    return candidates

            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 10 and len(constraint_list) > initial_K:
                ac_curr = fast_autocorr(y)
                constraint_set = set(int(k) for k in np.argsort(ac_curr)[::-1][:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 15:
                constraint_set = set(int(k) for k in constraint_list[:max(400, initial_K // 2)])
            continue

        lp_fail_count = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Adaptive step size with line search
        best_alpha = 0.1
        best_obj = M_curr
        for candidate_alpha in [0.05, 0.1, 0.15, 0.2, 0.3]:
            y_trial = (1.0 - candidate_alpha) * y + candidate_alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            st = y_trial.sum()
            if st < 1e-15:
                continue
            y_trial = y_trial / st
            obj_trial = fast_autocorr(y_trial).max()
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_alpha = candidate_alpha

        alpha = best_alpha
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 40 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW3 iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW3 done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def peak_shaving_coord(a0, time_limit, rng_seed=42, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(8, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:4]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:4]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:4]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            n_try = min(4, len(candidates))
            dst_order = candidates[np.argsort(cand_scores)[:n_try]]

            found = False
            for dst in dst_order:
                if dst == src:
                    continue
                lo, hi = 0.0, src_val * 0.4
                a_best_try = None
                M_best_try = best_m
                for _ in range(7):
                    mid = (lo + hi) / 2
                    a_trial = a.copy()
                    a_trial[src] -= mid
                    a_trial[dst] += mid
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        break
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M_best_try:
                        M_best_try = M_trial
                        a_best_try = a_trial.copy()
                        lo = mid
                    else:
                        hi = mid
                if a_best_try is not None and lo > 0:
                    a = a_best_try
                    best_m = M_best_try
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                j = int(rng.integers(n))
                if j != src:
                    for frac in [0.002, 0.008, 0.02]:
                        amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                if found:
                    break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_flat_init(n, rng_seed=0):
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    mag = np.ones(fft_size // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    X = mag * np.exp(1j * phase)
    X_full = np.zeros(fft_size, dtype=complex)
    X_full[:fft_size // 2 + 1] = X
    for i in range(1, fft_size // 2):
        X_full[fft_size - i] = np.conj(X[i])
    a = np.fft.irfft(X_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def generate_candidates(n):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    for freq in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.01)
            candidates.append(a / a.sum())
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(a / a.sum())
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.15, 0.25]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())
    candidates.append(np.ones(n) / n)
    for seed in [0, 1, 2, 3]:
        candidates.append(spectral_flat_init(n, rng_seed=seed))
    return candidates


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick multi-start Adam ===
    print(f"\n=== Phase 1: Multi-start Adam ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t1 = min(remaining * 0.06, 100.0)
    if t1 > 20:
        candidates = generate_candidates(n_cur)
        # Include parent and blends
        if parent is not None:
            candidates.insert(0, norm_sum1(parent))
        # Sample up to 8 diverse candidates
        rng_pick = np.random.default_rng(77)
        idx = list(range(min(8, len(candidates))))
        for i in idx:
            if time.monotonic() - t_global > time_budget - t1:
                break
            t_sub = min(15.0, t1 / max(1, len(idx) - i))
            a_out, _ = smooth_adam(candidates[i], t_sub, seed=100 + i)
            check_and_update(a_out, f"adam_start_{i}")

    n_best = len(best_seq)

    # === Phase 2: Main LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW3 (n={n_best}) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 60 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v3(a_in, t2, solver, initial_K=1000,
                                      max_K=6000, add_per_iter=100,
                                      reanchor_every=80, verbose=True)
        check_and_update(a_out, f"lpfw3_n{n_best}")

    # === Phase 3: Peak shaving coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak Shaving (n={n_best}) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving_coord(a_in, t3, rng_seed=555)

    # === Phase 4: Final Adam polish ===

    remaining = time_budget - (time.monotonic() - t_global)
