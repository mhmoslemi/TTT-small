def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_to(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_J(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def build_constraint_matrix(y, active_lags, n):
    K = len(active_lags)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(active_lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def solve_lp_direction(y, active_lags, n, solver):
    K = len(active_lags)
    if K == 0:
        return None, float('inf')
    coeffs = build_constraint_matrix(y, active_lags, n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1, coeffs @ g <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, float('inf')
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, float('inf')
    g_dir = g_val / g_sum
    return g_dir, float(M.value) if M.value is not None else float('inf')

def fast_subgradient_nesterov(a0, time_limit, top_k=80, beta=5.0, c0=4.0, verbose=False):
    n = len(a0)
    x = normalize(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    cur_top = top_k
    cur_beta = beta

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        mu = (t - 1.0) / (t + 1.0)
        y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / ws
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * y[k_int - idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        log_a = log_a - log_a.max()
        a_new = np.maximum(np.exp(log_a), 1e-30)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn
        x_prev = x
        x = a_new

        Mn = fast_autocorr(x).max()
        if Mn < best_M - 1e-14:
            best_M = Mn
            best_x = x.copy()

    if verbose:
        print("    SubGrad done (" + str(round(time.monotonic() - t0, 1)) + "s): J=" + str(round(2.0 * n * best_M, 8)))

def coordinate_polish(a0, time_limit, rng_seed=42, verbose=False):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 600
    step_fracs = [0.003, 0.005, 0.01, 0.02, 0.04]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            for _ in range(3):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                found = False
                for frac in step_fracs:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    at = a.copy()
                    at[src] -= amt
                    at[dst] += amt
                    at = np.maximum(at, 0.0)
                    st = at.sum()
                    if st < 1e-15:
                        continue
                    at = at / st
                    Mt = fast_autocorr(at).max()
                    if Mt < best_M - 1e-16:
                        a = at
                        best_M = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.002, 0.01)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    Mn = fast_autocorr(a).max()
                    if Mn < best_M:
                        best_M = Mn
                        best_a = a.copy()
                        stall = 0

    if verbose:
        print("    Coord done (" + str(round(time.monotonic() - t0, 1)) + "s): J=" + str(round(2.0 * n * best_M, 8)))
    return best_a, best_M
