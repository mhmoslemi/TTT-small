import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a, b=None):
    """Compute full convolution of a with b (or with itself)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_conv(a, a)
    return 2.0 * n * c.max() / (s * s)

def solve_lp(a, n):
    """Solve the LP: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0
    
    Returns (g, t) or (None, None) if infeasible.
    """
    # Variables: g[0], ..., g[n-1], t  (total n+1 variables)
    # Minimize: [0, 0, ..., 0, 1] @ vars
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: (a*g)[k] - t <= 0 for k = 0, ..., 2n-2
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0 (for j = 0,...,n-1)
    # A_ub[k, n] = -1
    rows = []
    cols = []
    data = []
    
    for k in range(2 * n - 1):
        # Determine valid range of i (where i = k - j, j = 0,...,n-1)
        # Need 0 <= i < n, i.e., 0 <= k-j < n, i.e., k-n < j <= k
        j_lo = max(0, k - n + 1)  # Wait, k - (n-1) <= j, i.e., j >= k-n+1
        j_hi = min(n - 1, k)
        # a[i] = a[k-j], so A[k, j] = a[k-j]
        for j in range(j_lo, j_hi + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                data.append(a[i])
        # Coefficient of t
        rows.append(k)
        cols.append(n)  # index of t variable
        data.append(-1.0)
    
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2 * n - 1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[j] >= 0, t >= 0
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    g = result.x[:n]
    t = result.x[n]
    return g, t

def norm_sum1(a):
    """Normalize a to sum to 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_power_ramp(n, p=0.8):
    """Power ramp: a[j] ~ (j+1)^p, normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)

    return norm_sum1(a)

    x = np.linspace(eps, 1.0, n)

    return norm_sum1(a)

    if n_right > 0:

    A_ub = csr_matrix((data, (rows, cols)), shape=(len(top_k), n + 1))
    b_ub = np.zeros(len(top_k))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={"time_limit": time_limit})
        if result.status == 0:
            g = np.maximum(result.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            t_val = float(result.x[n])
            return g, t_val
    except Exception:
        pass
    return None, None


def mirror_descent_step(a, ac, eta, n_peaks=20):
    """One step of mirror descent (multiplicative update) targeting top peaks."""
    n = len(a)
    top_k = np.argsort(ac)[::-1][:n_peaks]
    # Compute weighted subgradient
    sg = np.zeros(n)
    for k in top_k:
        ki = int(k)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        # d/d(a[j]) conv(a,a)[k] = 2*a[k-j]
        for j in idx:
            i = ki - j
            if 0 <= i < n:
                sg[j] += 2.0 * a[i]

    # Multiplicative update
    a_new = a * np.exp(-eta * sg)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def lp_optimize(a0, time_limit, n_active=50, verbose=False, rng_seed=42):
    """Main LP-based optimization: alternating LP + line search."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    it = 0

    print(f"    LP opt start: n={n}, obj={best_obj:.6f}")

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Determine LP time budget
        lp_time = min(remaining * 0.3, 30.0)
        if it < 5:
            # Full LP for first few iterations
            g, t_val = solve_lp_direction(a, n, time_limit=lp_time)
        else:
            # Small LP (cutting plane) for speed
            g, t_val = solve_lp_small(a, n, n_active=n_active, time_limit=lp_time)
            if g is None:
                g, t_val = solve_lp_direction(a, n, time_limit=lp_time)

        it += 1

        if g is None:
            # Fallback: mirror descent step
            ac = fast_conv(a, a)
            eta = 0.01 / (it + 1)
            a_trial = mirror_descent_step(a, ac, eta, n_peaks=30)
            obj_trial = objective_value(a_trial)
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial
                stall = 0
            else:
                stall += 1
            continue

        # Line search: try different alphas
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_trial = (1 - alpha) * a + alpha * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            obj_trial = objective_value(a_trial)
            if obj_trial < best_obj - 1e-10:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                stall = 0
                break
            elif obj_trial < objective_value(a):
                # Accept even if not new best (progress)
                a = a_trial
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
                stall = 0
                improved = True
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 5:
                pert = rng.normal(0, 0.02, n) * a
                a_p = a + pert
                a_p = np.maximum(a_p, 1e-10)
                a_p = a_p / a_p.sum()
                obj_p = objective_value(a_p)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_p.copy()
                a = a_p
                stall = 0

        # Periodically adapt n_active
        if it % 20 == 0:
            n_active = min(n_active + 10, 200)

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP it={it} ({elapsed:.0f}s): obj={best_obj:.6f}, stall={stall}")

    return best_a, best_obj


def spectral_flatten(a0, time_limit, verbose=False):
    """Gerchberg-Saxton spectral flattening."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        progress = min(it / 200.0, 1.0)
        alpha = 0.3 + 0.7 * progress

        P_target = P_mean * (P / np.maximum(P_mean, 1e-30)) ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target / np.maximum(P, 1e-30))
        mag_new = np.clip(mag_new, 0.5, 2.0)

        A_new = A * mag_new
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = objective_value(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            blend = 0.3
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            obj_curr = objective_value(a)
            if obj_curr < best_obj:
                best_obj = obj_curr
                best_a = a.copy()

    return best_a, best_obj
