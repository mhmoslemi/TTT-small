```python
    """Solve LP to find steepest descent direction for max convolution."""
    n = len(a)
    m = len(active_peaks)
    # Variables: d[0..n-1], theta (index n)
    # Minimize: theta
    # Constraints:
    #   v_k^T d <= theta for k in active → v_k^T d - theta <= 0 → v_k^T d + [-1]*theta <= 0
```

```python
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def grad_conv_k(a, k):
    """Gradient of conv_k(a) = sum_i a[i]*a[k-i] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        j = k - i
        if i == j:
            g[i] += a[j]
        else:
            g[i] += a[j]
            g[j] += a[i]
    return g
```

Wait, this loop is O(n) per peak. For n=5000 and 3 peaks, that's 15000 iterations in a Python loop. That's slow! Let me vectorize:

```python
def grad_conv_k(a, k):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    paired = k - idx
    # g[idx] += a[paired]
    g[idx] += a[paired]
    # For paired indices that are different from idx:
    mask = paired != idx
    g[paired[mask]] += a[idx[mask]]
    return g
```

This is O(n) in vectorized numpy, much faster than a Python loop.
