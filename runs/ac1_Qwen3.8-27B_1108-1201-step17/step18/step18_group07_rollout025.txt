"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain spectral flattening (Gerchberg-Saxton style)
2. Adam gradient descent with adaptive log-sum-exp smoothing
3. Peak-targeted local search with mass redistribution
4. Symmetry exploitation and multi-length search
5. Simulated annealing for escape from local minima
"""

import numpy as np
import time

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_symmetric(a):
    """Enforce symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    a = 0.5 * (a + a[::-1])
    return np.maximum(a, 0.0)

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_flatten(a, exponent=0.3, n_iters=100, time_limit=None, t0=None, verbose=False):
    """Iterative spectral flattening (GS-style)."""
    n = len(a)
    a = norm_simplex(a.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    if t0 is None:
        t0 = time.monotonic()
    
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: compress dynamic range
        mag_mean = mag.mean() + 1e-30
        ratio = mag / mag_mean
        mag_new = mag_mean * np.maximum(ratio, 1e-10) ** (1.0 - exponent)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = norm_simplex(a_new)
        
        c = fast_conv(a_new)
        M = c.max()
        
        if M < best_m:
            best_m = M
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and reduce exponent
            a = norm_simplex(best_a)
            if exponent < 0.05:
                break
            exponent *= 0.9
    
    return best_a, best_m

def grad_lse(a, c, T, top_k=50):
    """Compute gradient of log-sum-exp approximation to max(c).
    
    Returns gradient of LSE(c, T) w.r.t. a.
    """
    n = len(a)
    # Use top-k peaks for efficiency
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx]
    
    # Softmax weights
    max_v = top_vals[0]
    exponents = (top_vals - max_v) / max(T, 1e-30)
    exponents = np.maximum(exponents, -50.0)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    # Gradient: sum_k w_k * d(c[k])/da
    # d(c[k])/da[i] = a[k-i] + a[i] (for valid indices, with edge corrections)
    # Actually: d(c[k])/da[i] = a[k-i] if 0<=k-i<n, plus a[i] if... 
    # Let me be more careful.
    # c[k] = sum_j a[j]*a[k-j] where 0<=j<n and 0<=k-j<n
    # d(c[k])/da[i] = a[k-i] (from j=i term) + a[k-i] (from k-j=i term) 
    # Wait: from j=i: derivative is a[k-i] (if 0<=k-i<n)
    # From k-j=i, i.e., j=k-i: derivative is a[j] = a[k-i] (if 0<=k-i<n)
    # But we need to be careful: if i = k-i (i.e., 2i=k), we only count it once? No,
    # in the sum c[k] = sum_j a[j]*a[k-j], the term j=i gives a[i]*a[k-i], and the
    # term j=k-i gives a[k-i]*a[i]. These are the same term only if i = k-i.
    # So d(c[k])/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if i != k-i
    #                    = 2*a[k-i] if i != k-i (still 2 because both j=i and j=k-i give contributions)
    # Actually no. Let me redo:
    # c[k] = sum_j a[j] * a[k-j]
    # d/d(a[i]) = sum_j [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}]
    #           = a[k-i] (from first term, if 0<=k-i<n) + a[k-i] (from second term, if 0<=k-i<n)
    #           = 2*a[k-i] (if 0<=k-i<n)
    #           = a[k-i] (if only one of i or k-i is in range) -- this happens at edges
    # Actually both i and k-i need to be in [0, n-1] for the term to exist.
    # If i is in [0,n-1] (which it is) and k-i is in [0,n-1], then contribution is 2*a[k-i].
    # If k-i is not in [0,n-1], then there's no term involving a[i] in c[k].
    
    # Wait, I need to be more careful. c[k] = sum_j a[j]*a[k-j] where j ranges over
    # values such that both j and k-j are in [0, n-1].
    # So j ranges from max(0, k-(n-1)) to min(n-1, k).
    # d(c[k])/da[i] is non-zero only if i is in this range, AND k-i is in [0, n-1].
    # If i is in the valid range for c[k], then k-i is automatically in [0, n-1]
    # (because the valid range is defined so that both are in bounds).
    # So d(c[k])/da[i] = 2*a[k-i] for valid i, and 0 otherwise.
    
    # Actually wait, I realize there might be an issue at the "center" where i = k-i.
    # Let's see: c[k] contains the term a[i]*a[k-i]. If i = k-i (i.e., k = 2i),
    # this term is a[i]^2. Its derivative w.r.t. a[i] is 2*a[i].
    # In the formula d(c[k])/da[i] = a[k-i] + a[k-i] = 2*a[i], this gives 2*a[i]. ✓
    # So the formula d(c[k])/da[i] = 2*a[k-i] works in all cases.
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-15:
            continue
        k_int = int(k)
        # Valid i: max(0, k_int - (n-1)) <= i <= min(n-1, k_int)
        i_start = max(0, k_int - (n - 1))
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad

def adam_optimize(a0, time_limit=300.0, verbose=False):
    """Adam optimization with annealed log-sum-exp."""
    n = len(a0)
    a = norm_simplex(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Adam parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Annealing: temperature for LSE
    T_init = best_m * 10.0
    T_final = best_m * 0.1
    top_k_init = 100
    top_k_final = 5
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, t / 20000.0)
        T = T_init * (T_final / T_init) ** progress
        cur_top_k = max(top_k_final, int(top_k_init * (1 - progress) + top_k_final))
        
        # Compute LSE and gradient
        c = fast_conv(a)
        g = grad_lse(a, c, T, top_k=cur_top_k)
        
        # Project gradient to simplex (remove mean component)
        g_proj = g - g.mean()
        
        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * g_proj
        v_hat = beta2 * v_hat + (1 - beta2) * g_proj ** 2
        m_hat_corr = m_hat / (1 - beta1 ** t)
        v_hat_corr = v_hat / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat_corr / (np.sqrt(v_hat_corr) + epsilon)
        
        # Project to simplex
        a_new = norm_simplex(a_new)
        
        # Evaluate
        c_new = fast_conv(a_new)
        M_new = c_new.max()
        
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Reduce learning rate if stalled
        if stall > max_stall:
            lr *= 0.5
            stall = 0
            if lr < 1e-6:
                break
        
        # Periodic spectral flattening
        if t % 2000 == 0:
            a_spec, M_spec = spectral_flatten(a, exponent=0.2, n_iters=50, t0=t0, time_limit=time_limit)
            if M_spec < best_m:
                best_m = M_spec
                best_a = a_spec.copy()
                a = a_spec.copy()
                lr = max(lr, 0.005)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, T/T_init={T/T_init:.3f}, lr={lr:.2e}")
    
    return best_a, best_m

def peak_cancellation(a, time_limit=100.0, verbose=False):
    """Local search: target the largest peaks and redistribute mass."""
    n = len(a)
    a = norm_simplex(a.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        
        # Find top peaks
        n_top = min(5, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - (n - 1))
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Find positions with highest contribution to this peak
            top_contrib = np.argsort(contrib)[::-1][:min(10, len(contrib))]
            
            for tc in top_contrib:
                src = int(idx[tc])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Try moving to various destinations
                # Prefer destinations that don't contribute to any top peak
                n_candidates = min(20, n)
                candidates = rng.choice(n, size=n_candidates, replace=False)
                
                for dst in candidates:
                    if dst == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = norm_simplex(a_trial)
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = norm_simplex(a)
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

def make_init_rand(n, seed=42):

        inits.append(make_init_dual_peak(n_best, 0.2, 0.8, 0.12))
        labels.append("dualpeak")
        inits.append(make_init_rand(n_best, seed=1))
        labels.append("rand1")
        inits.append(make_init_rand(n_best, seed=2))
        labels.append("rand2")
        
        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n_best//20)], 6))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        
        t_per_init = t2 / len(unique_inits)
        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)

            t_avail = min(t_per_init, remaining)
            if t_avail < 15:
                break
            print(f"    Opt '{lbl}' (budget={t_avail:.0f}s)...")
            
            # Step 1: Spectral flattening
            t_spec = t_avail * 0.3
            a_spec, _ = spectral_flatten(init, exponent=0.35, n_iters=200, t0=time.monotonic(), time_limit=t_spec)
            try_update(a_spec, f"spec_{lbl}")
            
            # Step 2: Adam optimization
            remaining = time_budget - (time.monotonic() - t_global)
            t_adam = min(remaining * 0.5, t_avail * 0.45)
            if t_adam > 10:
                a_adam, _ = adam_optimize(a_spec, time_limit=t_adam)
                try_update(a_adam, f"adam_{lbl}")

            remaining = time_budget - (time.monotonic() - t_global)
            t_peak = min(remaining * 0.3, t_avail * 0.25)
            if t_peak > 5:
                a_peak, _ = peak_cancellation(a_adam, time_limit=t_peak)
                try_update(a_peak, f"peak_{lbl}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    t3 = min(remaining * 0.30, 400.0)

        remaining = time_budget - (time.monotonic() - t_global)
        t3c = min(remaining * 0.5, t3 * 0.25)
        if t3c > 10:
            a_spec, _ = spectral_flatten(a_out, exponent=0.15, n_iters=100, t0=time.monotonic(), time_limit=t3c*0.5)
            try_update(a_spec, "spec_refine1")
            a_out2, _ = adam_optimize(a_spec, time_limit=t3c*0.5)
            try_update(a_out2, "adam_refine2")
        
        # Round 4: Final peak cancellation
        remaining = time_budget - (time.monotonic() - t_global)
        t3d = min(remaining * 0.5, t3 * 0.10)
        if t3d > 5:
            a_out, _ = peak_cancellation(a_out2, time_limit=t3d)

    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length fine-tuning ===")
    if remaining > 60:
        for delta in [100, 200, -100, -200, 500]:

            if new_n < 1000 or new_n > 12000:

            if remaining < 30:

            t_avail = min(remaining * 0.5, 80.0)
            if t_avail < 15:
                break
            a_interp = interpolate_sequence(best_seq, new_n)
            try_update(a_interp, f"fine_n{new_n}")
            a_spec, _ = spectral_flatten(a_interp, exponent=0.2, n_iters=50, t0=time.monotonic(), time_limit=t_avail*0.5)
            try_update(a_spec, f"spec_fine_n{new_n}")
            a_adam, _ = adam_optimize(a_spec, time_limit=t_avail*0.4)
            try_update(a_adam, f"adam_fine_n{new_n}")
    
    # === Phase 5: Final polish ===

    n_best = len(best_seq)

    if remaining > 20:

            a_out2, _ = adam_optimize(a_out, time_limit=t5_adam)
            try_update(a_out2, "adam_final")
        
        # Final spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 5:
            a_spec, _ = spectral_flatten(a_out2, exponent=0.1, n_iters=50, t0=time.monotonic(), time_limit=remaining*0.5)
            try_update(a_spec, "spec_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")

        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
