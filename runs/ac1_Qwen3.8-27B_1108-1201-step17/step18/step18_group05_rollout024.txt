Let me think about the golden section line search more carefully:

We have a and g (the FW direction, normalized to sum 1). We want to find alpha in [0,1] that minimizes:
f(alpha) = max_k conv((1-alpha)*a + alpha*g)_k

Note that conv((1-alpha)*a + alpha*g) = (1-alpha)^2 * conv(a) + 2*alpha*(1-alpha) * cross(a,g) + alpha^2 * conv(g)

where cross(a,g) = conv(a,g) (not autocorrelation but cross-correlation).

Wait, actually conv is bilinear: conv(a+b, a+b) = conv(a,a) + conv(a,b) + conv(b,a) + conv(b,b).

And conv(a,b) = conv(b,a) (for real sequences, the linear convolution is commutative).

So conv((1-α)a + αg, (1-α)a + αg) = (1-α)² conv(a) + 2α(1-α) conv(a,g) + α² conv(g).

This is a quadratic in α for each lag k. The max of quadratics is a convex function of α.

For the line search, I can:
1. Precompute conv(a), conv(a,g), conv(g)
2. For a given α, compute the combined conv and find its max
3. Use golden section search over α ∈ [0, 1]

The conv(a,g) can be computed via FFT as well.

Actually, for efficiency, I'll compute:
- ca = fast_autocorr(a) -- this is conv(a, a)
- cag = fast_crosscorr(a, g) -- this is conv(a, g)  
- cg = fast_autocorr(g) -- this is conv(g, g)

Then for any α:
conv_α = (1-α)² * ca + 2α(1-α) * cag + α² * cg

And max(conv_α) is what we want to minimize.

This is very efficient because we only need 3 convolutions total, and then the line search is just arithmetic.

```python
def fast_crosscorr(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]
```

Then the golden section search:

```python
def golden_section_search(ca, cag, cg, a_lo=0.0, a_hi=1.0, tol=1e-10):
    """Find alpha in [a_lo, a_hi] minimizing max(ca*(1-alpha)^2 + 2*cag*alpha*(1-alpha) + cg*alpha^2)."""
    GR = (np.sqrt(5) + 1) / 2
    inv_GR = 1.0 / GR
    inv_GR2 = 1.0 / (GR * GR)
    
    a = a_lo + inv_GR2 * (a_hi - a_lo)
    b = a_lo + inv_GR * (a_hi - a_lo)
    
    fa = _eval_max(ca, cag, cg, a)
    fb = _eval_max(ca, cag, cg, b)
    
    while abs(a_hi - a_lo) > tol:
        if fa < fb:
            a_hi = b
            b = a
            fb = fa
            a = a_lo + inv_GR2 * (a_hi - a_lo)
            fa = _eval_max(ca, cag, cg, a)
        else:
            a_lo = a
            a = b
            fa = fb
            b = a_lo + inv_GR * (a_hi - a_lo)
            fb = _eval_max(ca, cag, cg, b)
    
    return (a_lo + a_hi) / 2

def _eval_max(ca, cag, cg, alpha):
    conv_alpha = (1-alpha)**2 * ca + 2*alpha*(1-alpha) * cag + alpha**2 * cg
    return conv_alpha.max()
```

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_crosscorr(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _eval_conv_max(ca, cag, cg, alpha):
    """Evaluate max of convolution at given alpha."""
    one_minus_a = 1.0 - alpha
    conv_alpha = one_minus_a * one_minus_a * ca + 2.0 * alpha * one_minus_a * cag + alpha * alpha * cg
    return conv_alpha.max()


def golden_section_search(ca, cag, cg, a_lo=0.0, a_hi=1.0, max_iter=60):
    """Golden section search for optimal alpha minimizing max conv."""
    GR = (np.sqrt(5.0) + 1.0) / 2.0
    inv_GR = 1.0 / GR
    inv_GR2 = 1.0 / (GR * GR)

    a = a_lo + inv_GR2 * (a_hi - a_lo)
    b = a_lo + inv_GR * (a_hi - a_lo)
    fa = _eval_conv_max(ca, cag, cg, a)
    fb = _eval_conv_max(ca, cag, cg, b)

    for _ in range(max_iter):
        if a_hi - a_lo < 1e-12:
            break
        if fa < fb:
            a_hi = b
            b = a
            fb = fa
            a = a_lo + inv_GR2 * (a_hi - a_lo)
            fa = _eval_conv_max(ca, cag, cg, a)
        else:
            a_lo = a
            a = b
            fa = fb
            b = a_lo + inv_GR * (a_hi - a_lo)
            fb = _eval_conv_max(ca, cag, cg, b)

    best_alpha = (a_lo + a_hi) / 2.0
    return best_alpha


def lp_frank_wolfe_enhanced(a0, time_limit, solver, K_init=2000, K_max=5000,
                             verbose=True, rng_seed=42):
    """
    Enhanced LP-Based Frank-Wolfe with:
    - Golden section line search using precomputed cross-correlations
    - Dynamic constraint set adaptation
    - Restart with perturbation on stall
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    K = min(K_init, 2 * n - 1)
    iter_count = 0
    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < 1e-16:
            break

        # Select top K lags for constraints
        K_sel = min(K, 2 * n - 1)
        if K_sel < 2 * n - 1:
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        else:
            binding_idx = np.arange(2 * n - 1)

        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        # Solve LP: max sum(b) s.t. T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 200, K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Golden section line search
        ca = fast_autocorr(a)
        cag = fast_crosscorr(a, g_dir)
        cg = fast_autocorr(g_dir)

        alpha_opt = golden_section_search(ca, cag, cg, 0.0, 1.0, max_iter=50)

        # Also check alpha = 1.0 and a few values near the optimum
        candidates = [alpha_opt, min(1.0, alpha_opt + 0.1), max(0.0, alpha_opt - 0.1), 1.0]
        best_alpha = alpha_opt
        best_trial_m = _eval_conv_max(ca, cag, cg, alpha_opt)
        for al in candidates:
            if 0.0 <= al <= 1.0:
                m_trial = _eval_conv_max(ca, cag, cg, al)
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_alpha = al

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        iter_count += 1

        # Adapt K
        if iter_count % 20 == 0:
            K = min(K + 100, K_max)

        # Restart with perturbation
        if stall > max_stall:
            pert = rng.normal(0, 0.02, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-15)
            a = a_r / a_r.sum()
            stall = 0
            K = max(500, min(K - 500, K_max))

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}, M={best_m:.2e}")

    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=60, beta=6.0, c0=5.0, restart_every=4000,

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x_prev = a.copy()

    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x = a_new

        M_new = fast_autocorr(x).max()

            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)
            cur_top_k = min(cur_top_k + 5, len(ac))

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x = x_r / x_r.sum()
            x_prev = x.copy()
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 1.0, 2.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish_enhanced(a0, time_limit, rng_seed=123, verbose=False):
    """Enhanced coordinate descent with FFT-based evaluation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity map
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved_this_round = False

        # Try moves at the most expensive positions
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destinations
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 3:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:5]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.02, 0.01, 0.03, 0.05, 0.08, 0.12]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved_this_round = True
                        found = True
                        break

        # Random exploration if stuck
        if not improved_this_round:
            stall += 1
            for _ in range(5):
                if stall > 20:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))

                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved_this_round = True
                    break
            if not improved_this_round:
                a = best_a.copy()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m
