"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated softmax subgradient for max-convolution
2. Nesterov-accelerated mirror descent on the simplex
3. Adaptive temperature and restart strategy

import numpy as np
import time

_best_tracker = {'seq': None, 'score': float('inf')}


def fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def subgrad_spatial(a, conv, n_active=60, temp=0.003):
    """Softmax subgradient of max(conv) in spatial domain."""
    n = len(a)
    order = np.argsort(conv)[::-1]
    top = order[:n_active]
    cmax = conv.max()
    lw = (conv[top] - cmax) / temp
    lw -= lw.max()
    w = np.exp(lw)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(top)) / len(top)
    else:
        w = w / ws
    grad = np.zeros(n)
    for wi, k in zip(w, top):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[ki - idx]
    return grad


def subgrad_fft(a, conv, temp=0.003):
    """FFT-based subgradient: faster when many peaks are active."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    cmax = conv.max()
    thresh = cmax - temp * 12.0
    mask = conv >= thresh
    if mask.sum() < 20:
        top_idx = np.argsort(conv)[::-1][:20]
        mask = np.zeros(size, dtype=bool)
        mask[top_idx] = True
    ca = conv[mask]
    lw = (ca - cmax) / temp
    lw -= lw.max()
    w = np.exp(lw)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(mask.sum()) / mask.sum()
    else:
        w = w / ws
    pi = np.zeros(fft_size)
    pi[mask] = w
    A_hat = np.fft.rfft(a, fft_size)
    Pi_hat = np.fft.rfft(pi, fft_size)
    G_hat = 2.0 * np.conj(A_hat) * Pi_hat
    G_hat += np.conj(G_hat[::-1]) * 0  # ensure even symmetry handled by irfft
    g = np.fft.irfft(G_hat, fft_size)[:n]
    return g


def amd_optimize(a0, time_limit, seed=42, verbose=False):

        a_new = np.maximum(a_new, 1e-30)

            best_x = x.copy()

def coord_polish(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent: move mass from high-cost to low-cost positions."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fft_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        conv = fft_conv(a)
        n_top = min(25, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_top]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        significant = a > 1e-12
        n_sig = significant.sum()
        if n_sig < 2:
            break

        src_idx = np.where(significant)[0]
        src_i = int(src_idx[np.argmax(costs[src_idx])])
        dst_mask = ~significant
        if dst_mask.sum() < 10:
            dst_mask = np.ones(n, dtype=bool)
            dst_mask[src_i] = False
        dst_candidates = np.where(dst_mask)[0]
        dst_i = int(dst_candidates[np.argmin(costs[dst_candidates])])

        improved = False
        for frac in [0.08, 0.15, 0.25, 0.4, 0.6, 0.8]:
            amt = a[src_i] * frac
            a_t = a.copy()
            a_t[src_i] -= amt
            a_t[dst_i] += amt
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st < 1e-15:
                continue
            a_t = a_t / st
            Mt = fft_conv(a_t).max()
            if Mt < best_m - 1e-14:
                a = a_t
                best_m = Mt
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if a[src_i] > 1e-10:
                frac = rng.uniform(0.01, 0.06)
                amt = a[src_i] * frac
                a[src_i] -= amt
                j = int(rng.integers(n))

                    a[j] += amt

                a = a / a.sum()
                Mn = fft_conv(a).max()
                if Mn < best_m:
                    best_m = Mn
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    ai = np.interp(nx, ox, a_norm)
    ai = np.maximum(ai, 0.0)
    ss = ai.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return ai / ss


def init_unif(n):

def init_dec(n, pw=0.5):
    x = np.arange(n, dtype=np.float64)

def init_inc(n, pw=0.5):

def init_bell(n, c=0.5, w=0.2):

def init_ushape(n, s=2.0):
    x = np.linspace(0, 1, n)
    return norm1(((x - 0.5) / 0.3) ** 2 * s + 0.1)

def init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    return norm1(np.maximum(a, 0.0))

def init_cos(n, m=1):
    x = np.linspace(0, 1, n)
    return norm1(np.maximum(np.cos(np.pi * m * (x - 0.5)) ** 2, 0.0))

def init_2peak(n, p1=0.2, p2=0.8, w=0.08):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - p1) / w) ** 2) + np.exp(-((x - p2) / (w * 1.3)) ** 2)
    return norm1(a)

def init_3peak(n):

         np.exp(-((x - 0.85) / 0.06) ** 2))
    return norm1(a)

def init_rnd(n, seed=0, nf=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, nf + 1):
        a += rng.normal(0, 0.12 / np.sqrt(f)) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    return norm1(np.maximum(a, 0.001))

def init_powctr(n, pw=0.5):
    xc = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm1(xc ** (-pw))

def init_exp(n, r=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm1(np.exp(-r * x) + np.exp(-r * (1 - x)))
