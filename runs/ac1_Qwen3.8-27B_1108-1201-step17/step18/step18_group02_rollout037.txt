**Algorithm: Smoothed Projected Gradient with Multiplicative Updates**

1. Initialize a (non-negative, sum=1)
2. For each iteration:

   b. Compute LSE gradient: grad = 2 * cross_correlation(a, p) where p = softmax(C/τ)

   e. Update τ (decrease for sharper approximation of max)
   f. Track best solution

This is essentially the "mirror descent" or "exponentiated gradient" method on the simplex, using the LSE-smoothed max as the objective.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponentiated gradient (mirror descent) with log-sum-exp smoothing
2. Frank-Wolfe with cutting planes for tight constraints
3. Diverse initializations including error-reduction (input-output) method
4. Multi-scale exploration and local polish
"""

import numpy as np
import time
import cvxpy as cp
import math

# --- Helper functions ---

def fast_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_nonneg(a):
    """Project onto non-negative simplex (sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def lse_objective(C, tau):
    """Compute log-sum-exp of C/tau times tau."""
    C_scaled = C / tau
    C_scaled -= C_scaled.max()
    return tau * np.log(np.sum(np.exp(C_scaled)))

def lse_smooth_gradient(a, C, tau):
    """
    Compute gradient of tau*log(sum exp(C[m]/tau)) w.r.t. a.
    Using the identity: dC[m]/da[j] = 2*a[m-j] (zero-padded)
    So grad_a[j] = 2 * sum_m p[m] * a[m-j] = 2 * cross_correlation(a, p)[j]
    where p[m] = exp(C[m]/tau) / sum_k exp(C[k]/tau)
    """
    n = len(a)
    C_scaled = C / tau
    C_scaled -= C_scaled.max()
    exp_C = np.exp(C_scaled)
    p = exp_C / exp_C.sum()
    
    # grad_a[j] = 2 * sum_k a[k] * p[j+k]
    # = 2 * (flip(a) * p)[n-1+j]
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A_flip = np.fft.rfft(a[::-1], fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A_flip * P, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)

def exp_gradient_descent(a0, time_limit, tau_init=None, tau_decay=0.9997,
                         lr_init=0.5, lr_decay=0.9999, verbose=True, tau_min=None):
    """
    Exponentiated gradient (mirror descent) on the simplex with LSE smoothing.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    C = fast_convolve(a)
    
    if tau_init is None:
        tau_init = max(C.max() * 0.5, 1e-6)
    if tau_min is None:
        tau_min = max(C.max() * 0.01, 1e-10)
    
    best_a = a.copy()
    best_m = C.max()
    best_obj = 2.0 * n * best_m
    t0 = time.monotonic()
    
    tau = tau_init
    lr = lr_init
    stall = 0
    stall_limit = 3000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        C = fast_convolve(a)
        M = C.max()
        
        if M < best_m - 1e-12:
            best_m = M
            best_a = a.copy()
            best_obj = 2.0 * n * M
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        lr = max(lr_init * (lr_decay ** t), 1e-6)
        
        grad = lse_smooth_gradient(a, C, tau)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a = normalize_nonneg(a_new)
        
        if stall > stall_limit:
            stall = 0
            stall_limit = int(stall_limit * 1.5)
            # Perturbation
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.05, n) * best_a
            a = normalize_nonneg(best_a + pert)
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EGD t={t} ({elapsed:.0f}s): obj={2*n*M:.6f}, tau={tau:.2e}, lr={lr:.4f}")
    
    elapsed = time.monotonic() - t0
    print(f"    EGD done ({elapsed:.0f}s): best_obj={best_obj:.6f}")
    return best_a, best_obj

def error_reduction(a0, time_limit, n_iter=500, verbose=True):
    """
    Error reduction (input-output) algorithm for flat spectrum design.
    Alternates between:
    - Time domain: project onto non-negative simplex
    - Frequency domain: modify spectrum to flatten C
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    best_obj = 2.0 * n * best_m
    t0 = time.monotonic()
    
    for t in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current spectrum
        A = np.fft.rfft(a, fft_size)
        
        # Current C
        C = np.fft.irfft(A * A, fft_size)[:size]
        M = C.max()
        
        if M < best_m - 1e-12:
            best_m = M
            best_a = a.copy()
            best_obj = 2.0 * n * M
        
        # Target: flatten C
        # We want |A_new|^2 = fft(C_target)
        # C_target = 1/size * ones(size) would give flat C
        # But fft of constant is a delta, so |A_target| would be constant
        # Instead, let's use a "soft" target: reduce peaks in C
        C_target = np.ones(size) / size
        # Blend with current to avoid oscillation
        C_modified = 0.5 * C / C.sum() + 0.5 * C_target
        
        # Get target spectrum magnitude
        C_mod_fft = np.fft.rfft(C_modified, fft_size)
        # |A_new|^2 should be proportional to C_mod_fft
        # But C_mod_fft might be complex - take magnitude
        target_mag = np.sqrt(np.abs(C_mod_fft))
        
        # Set new magnitude, keep phase
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = target_mag * phase
        
        # Transform back
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a = normalize_nonneg(a_new)
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            C_check = fast_convolve(a)
            print(f"    ER t={t} ({elapsed:.0f}s): obj={2*n*C_check.max():.6f}")
    
    elapsed = time.monotonic() - t0
    print(f"    ER done ({elapsed:.0f}s): best_obj={best_obj:.6f}")
    return best_a, best_obj
