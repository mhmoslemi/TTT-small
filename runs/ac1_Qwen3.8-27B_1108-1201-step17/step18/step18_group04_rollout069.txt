def fast_conv(a):

    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgrad(a, conv, top_k=30, temperature=0.003):

    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:

def mw_best_response(f, n, iterations=800, lr=0.15, tol=1e-5):
    y = np.ones(2 * n - 1) / (2 * n - 1)
    f_rev = f[::-1].copy()
    k_arr = np.arange(2 * n - 1)

    for it in range(iterations):
        dual_full = np.convolve(f_rev, y, mode='full')
        dual_vals = dual_full[n - 1:n - 1 + n]

        targets = 1.0 / np.maximum(f, 1e-12)
        violations = np.maximum(targets - dual_vals, 0.0)
        max_viol = violations.max()
        if max_viol < tol:
            break

        prefix = np.concatenate([[0.0], np.cumsum(violations)])
        j_lo = np.maximum(0, k_arr - n + 1)
        j_hi = np.minimum(n - 1, k_arr)
        update = (prefix[j_hi + 1] - prefix[j_lo]) / n

        y += lr * max_viol * update
        y = np.maximum(y, 1e-15)

    dual_full = np.convolve(f_rev, y, mode='full')
    dual_vals = dual_full[n - 1:n - 1 + n]
    b = np.maximum(0.0, 1.0 - dual_vals)
    s = b.sum()
    if s < 1e-15:
        return f.copy()
    return b / s


def lp_best_response(f, n, timeout=20):
    try:
        diagonals = []
        offsets = []
        for d in range(n):
            length = n - d
            if length <= 0:
                break
            diagonals.append(np.full(length, f[d]))
            offsets.append(d)
        A = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
        c_obj = -np.ones(n)
        b_ub = np.ones(2 * n - 1)
        bounds_list = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds_list,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return f.copy()

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def make_random_smooth(n, seed=0, n_freq=8):
