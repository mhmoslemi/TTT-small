import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_gradient_multi_lag(a, lags, weights):
    """
    Compute the weighted subgradient: sum_l weights[l] * 2*a[k_l - i] for valid i.
    This gives d(max_l c[k_l])/da[i] approximately.
    """
    n = len(a)
    grad = np.zeros(n)
    for wi, k in zip(weights, lags):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    return grad

def block_search(a0, time_limit, block_size, verbose):
    """
    Block-level local search: perturb small blocks and keep improvements.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    stall = 0
    max_stall = 300
    n_blocks = max(1, (n - block_size) // block_size)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False

        # Focus perturbations near peak-contributing regions
        top_lags = np.argsort(ac)[::-1][:10]

        for attempt in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            if attempt < 10 and len(top_lags) > 0:
                # Targeted: perturb near a peak-contributing position
                k_pick = int(top_lags[attempt % len(top_lags)])
                js = max(0, k_pick - n + 1)
                je = min(n - 1, k_pick)
                center = (js + je) // 2
                start = max(0, min(n - block_size, center - block_size // 2 + int(rng.integers(-5, 5))))
            else:
                start = int(rng.integers(0, max(1, n - block_size + 1)))

            end = min(n, start + block_size)
            bs = end - start
            if bs < 2:
                continue

            block = a[start:end].copy()
            mean_block = block.mean() + 1e-8
            scale = rng.uniform(0.02, 0.08)
            pert = rng.normal(0, scale * mean_block, bs)
            new_block = np.maximum(block + pert, 0.0)

            if abs(new_block.sum() - block.sum()) < 1e-15:
                continue

            a_temp = a.copy()
            a_temp[start:end] = new_block
            a_temp = np.maximum(a_temp, 0.0)
            s_t = a_temp.sum()
            if s_t < 1e-15:
                continue
            a_temp = a_temp / s_t
            M_temp = fast_autocorr(a_temp).max()

            if M_temp < M - 1e-14:
                a = a_temp
                M = M_temp
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                improved = True
                if attempt >= 20:
                    break

        if not improved:
            stall += 1
            # Random escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Block done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def coord_polish(a0, time_limit, verbose):
    """
    Coordinate descent: move mass from peak-contributing positions to low-impact ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    stall = 0
    max_stall = 250

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(12, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score: total peak contribution at each position
            peak_contrib = np.zeros(n)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_scores = peak_contrib[candidates]
            dst = candidates[np.argmin(cand_scores)]
            if dst == src:
                continue

            for frac in [0.015, 0.03, 0.05, 0.08]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def parametric_init(n, best_a, best_score):
    """Try parametric families for initialization."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    results = []

    families = []
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        families.append((i_arr + 1.0) ** p)
        families.append((n - i_arr).clip(0.5) ** p)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            families.append(np.exp(-((x - c) ** 2) / (2.0 * sigma ** 2)))
    for freq in [1, 2, 3]:
        families.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))
    for rate in [1, 3, 10]:
        families.append(np.exp(-rate * x))
        families.append(np.exp(rate * (x - 1.0)))

    best_local = best_a.copy()
    best_local_score = best_score

    for fam in families:
        fam_norm = norm_sum1(fam)
        ac = fast_autocorr(fam_norm)
        val = 2.0 * n * ac.max()
        if val < best_local_score:
            best_local_score = val
            best_local = fam_norm.copy()

    return best_local, best_local_score
