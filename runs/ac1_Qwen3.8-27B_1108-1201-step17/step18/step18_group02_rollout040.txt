"""
Search for low-peak self-convolution step-function heights.

The program keeps a global best sequence according to the supplied
evaluate_sequence().  It uses:
  1. A small parametric initialization pool to infer the effective objective.
  2. Log-sum-exp multiplicative Adam on the convolution peak.
  3. Exact O(n) incremental coordinate transfers for local polishing.
  4. Optional cutting-plane Frank-Wolfe LP direction from scipy, if available.
  5. Length interpolation and random actual-evaluator polish.

The output is returned as a Python list of non-negative floats.
"""

except Exception:
    _LINPROG = None

PROP_BUDGET = 1740.0
PROP_BEST_A = None
PROP_BEST_SCORE = float("inf")
PROP_LAST_RECORD = 0.0
PROP_RECORD_INTERVAL = 3.0
PROP_RECORD_COUNT = 0
PROP_MAX_RECORDS = 400


def init_proposal_state(budget_s):
    global PROP_START, PROP_BUDGET, PROP_BEST_A, PROP_BEST_SCORE
    global PROP_LAST_RECORD, PROP_RECORD_COUNT

    now = time.monotonic()
    PROP_START = now
    try:

    PROP_BEST_SCORE = float("inf")
    PROP_LAST_RECORD = now
    PROP_RECORD_COUNT = 0


def remaining_time():
    if PROP_START <= 0.0:
        return 0.0
    return max(0.0, PROP_BUDGET - (time.monotonic() - PROP_START))


def normalize_nonneg(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.reshape(-1)
    if arr.size == 0:
        return np.array([1.0], dtype=np.float64)

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def linear_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.zeros(1, dtype=np.float64)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[:size]
    c = np.maximum(c, 0.0)
    c = np.where(np.isfinite(c), c, 0.0)
    return c


def make_mask(n, mode):
    if mode == 0:
        return None
    L = 2 * int(n) - 1
    mask = np.ones(L, dtype=bool)
    mask[int(n) - 1] = False
    return mask


def masked_max(c, mask):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    if mask is None:
        return float(np.max(c))
    vals = c[mask]
    if vals.size == 0:
        return 0.0
    return float(np.max(vals))


def safe_evaluate(a):
    arr = normalize_nonneg(a)
    if arr.size == 0:

def lse_proba(c, mask, tau):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    e = np.zeros(L, dtype=np.float64)
    if L == 0:
        return e
    if tau <= 0.0 or not np.isfinite(tau):
        tau = 1e-12

    if mask is None:
        m = float(np.max(c))
        z = (c - m) / tau
        np.maximum(z, -745.0, out=z)
        ex = np.exp(z)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0.0:
            idx = int(np.argmax(c))
            e[idx] = 1.0
            return e
        return ex / s

    vals = c[mask]
    if vals.size == 0:
        return e

    m = float(np.max(vals))
    z = (vals - m) / tau
    np.maximum(z, -745.0, out=z)
    ev = np.exp(z)
    s = float(ev.sum())
    if not np.isfinite(s) or s <= 0.0:
        idx = np.where(mask)[0]
        i = int(idx[int(np.argmax(vals))])
        e[i] = 1.0
        return e
    e[mask] = ev / s
    return e


def grad_from_p(a, p):
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = int(a.size)
    L = int(p.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)

    fft_size = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a[::-1], fft_size)
    fp = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(fa * fp, fft_size)
    g = 2.0 * conv[n - 1:n - 1 + n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    return g


def smooth_noise(n, rng, amp=0.35):
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    out = np.zeros(n, dtype=np.float64)
    scale = max(amp / 2.5, 1e-6)
    for freq in (1, 2, 3, 5, 8):
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp_local = float(rng.normal(0.0, scale))
        out += amp_local * np.cos(2.0 * np.pi * freq * (x - 0.5) + phase)
    return np.exp(np.clip(out, -6.0, 6.0))


def blended_restart(best_a, n, rng, alpha):
    best_a = normalize_nonneg(best_a)
    noise = smooth_noise(int(n), rng, 0.38)
    x = (1.0 - float(alpha)) * best_a + float(alpha) * noise
    return normalize_nonneg(x)


def smooth_lse_search(a0, time_limit, mask_mode,
                      tau_decay=0.999, lr=0.40,
                      beta1=0.90, beta2=0.999,
                      restart_every=1500, seed=1, label="lse"):
    if time_limit <= 0.0 or remaining_time() < 5.0:
        return normalize_nonneg(a0)

    n = int(len(a0))
    a = normalize_nonneg(a0)
    if n == 0:
        return a

    mask = make_mask(n, mask_mode)
    c = linear_conv(a)
    m0 = masked_max(c, mask)
    if not np.isfinite(m0) or m0 <= 0.0:
        m0 = max(1e-12, float(c.max()) if c.size else 1e-12)

    best_m = m0
    best_a = a.copy()

    tau = max(1e-12, 0.30 * m0)
    tau_min = max(1e-12, 0.001 * m0)

    log_a = np.log(np.maximum(a, 1e-300))
    m_hat = np.zeros(n, dtype=np.float64)
    v_hat = np.zeros(n, dtype=np.float64)
    bc1 = 0.0
    bc2 = 0.0

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    last_rec = t0
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit and remaining_time() > 5.0:
        it += 1

        p = lse_proba(c, mask, tau)
        g = grad_from_p(a, p)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        bc1 += (1.0 - beta1) * (1.0 - bc1)
        bc2 += (1.0 - beta2) * (1.0 - bc2)

        step = lr / (np.sqrt(v_hat / max(bc2, 1e-12)) + 1e-8)
        step = np.maximum(step, 0.0)
        step = np.minimum(step, 5.0)

        log_a = log_a - step * g
        if not np.all(np.isfinite(log_a)):
            log_a = np.log(np.maximum(best_a, 1e-300))

        maxlog = float(log_a.max())
        a = np.exp(log_a - maxlog)
        a = np.maximum(a, 1e-300)
        sa = float(a.sum())
        if not np.isfinite(sa) or sa < 1e-300:
            a = best_a.copy()
            log_a = np.log(np.maximum(a, 1e-300))
        else:
            a = a / sa

        c = linear_conv(a)
        m = masked_max(c, mask)
        if np.isfinite(m) and m < best_m - 1e-12:
            best_m = m
            best_a = a.copy()
            stall = 0
            now = time.monotonic()
            if now - last_rec > PROP_RECORD_INTERVAL:
                record_candidate(a, label + "_impr")
                last_rec = now
        else:
            stall += 1

        tau = max(tau * tau_decay, tau_min)

        if it % 1000 == 0:
            now = time.monotonic()
            if now - last_rec > 2.0 * PROP_RECORD_INTERVAL:
                record_candidate(best_a, label + "_check")
                last_rec = now

        if stall > restart_every:
            stall = 0
            a = blended_restart(best_a, n, rng, 0.22)
            log_a = np.log(np.maximum(a, 1e-300))
            m_hat = np.zeros(n, dtype=np.float64)
            v_hat = np.zeros(n, dtype=np.float64)
            bc1 = 0.0
            bc2 = 0.0
            c = linear_conv(a)
            m = masked_max(c, mask)
            if np.isfinite(m) and m < best_m:
                best_m = m
                best_a = a.copy()
            tau = max(1e-12, 0.30 * max(m, m0 * 1e-3))

    record_candidate(best_a, label + "_final", force=True)
    return best_a


def apply_transfer_delta_c(c, a, src, dst, delta, sign):
    n = len(a)
    L = len(c)
    src = int(src)
    dst = int(dst)
    if src == dst or delta == 0.0:
        return

    if sign > 0:
        c[dst:dst + n] += 2.0 * delta * a
        k = 2 * dst
        if k < L:
            c[k] += delta * delta

        c[src:src + n] -= 2.0 * delta * a
        k = 2 * src
        if k < L:
            c[k] += delta * delta

        k = src + dst
        if k < L:
            c[k] -= 2.0 * delta * delta
    else:
        c[dst:dst + n] -= 2.0 * delta * a
        k = 2 * dst
        if k < L:
            c[k] -= delta * delta

        c[src:src + n] += 2.0 * delta * a
        k = 2 * src
        if k < L:
            c[k] -= delta * delta

        k = src + dst
        if k < L:
            c[k] += 2.0 * delta * delta


def top_indices(c, mask, k):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.array([], dtype=int)

    k = int(k)
    if mask is None:
        idx = np.argsort(c)[::-1]
        return idx[:max(0, min(k, idx.size))].astype(int, copy=False)

    mat = np.where(mask)[0]
    if mat.size == 0:
        return np.array([], dtype=int)
    vals = c[mat]
    order = np.argsort(vals)[::-1]
    idx = mat[order[:max(0, min(k, order.size))]]
    return idx.astype(int, copy=False)

def cross_conv(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = int(a.size)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    c = np.fft.irfft(fa * fb, fft_size)[:size]
    c = np.where(np.isfinite(c), c, 0.0)
    return c


def lp_frank_wolfe(a0, time_limit, mask_mode,
                   initial_K=100, max_K=300,
                   add_per_iter=20, label="lp"):
    if _LINPROG is None:
        return normalize_nonneg(a0)
    if time_limit <= 0.0 or remaining_time() < 5.0:
        return normalize_nonneg(a0)

    a = normalize_nonneg(a0)
    n = int(a.size)
    if n <= 1:
        return a

    mask = make_mask(n, mask_mode)
    y = a.copy()
    x = a.copy()
    c = linear_conv(y)
    best_m = masked_max(c, mask)
    best_a = y.copy()

    constraints = [int(k) for k in top_indices(c, mask, initial_K)]
    fail_count = 0
    no_improve = 0
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and remaining_time() > 10.0:
        it += 1
        if len(constraints) > max_K:
            constraints = [int(k) for k in top_indices(c, mask, max_K)]

        if not constraints:
            break

        m_curr = masked_max(c, mask)
        M_target = max(1e-15, m_curr) * 1.000001 + 1e-12

        rows = []
        for kk in constraints:
            kk = int(kk)
            row = np.zeros(n, dtype=np.float64)
            start = max(0, kk - n + 1)
            end = min(n - 1, kk)
            if start <= end:
                idx = np.arange(start, end + 1)
                row[idx] = y[kk - idx]
            rows.append(row)

        if not rows:
            break

        A = np.vstack(rows)
        b = np.full(len(rows), M_target, dtype=np.float64)
        A = np.vstack([A, np.ones(n, dtype=np.float64)])
        b = np.concatenate((b, np.array([5.0], dtype=np.float64)))

        c_obj = -np.ones(n, dtype=np.float64)
        res = None
        for meth in ("highs", "interior-point"):
            try:
                res = _LINPROG(
                    c_obj, A_ub=A, b_ub=b,
                    bounds=(0.0, 0.1),
                    method=meth,
                    options={"presolve": True, "time_limit": 3.0}
                )
                break
            except Exception:
                try:
                    res = _LINPROG(
                        c_obj, A_ub=A, b_ub=b,
                        bounds=(0.0, 0.1),
                        method=meth
                    )
                    break
                except Exception:
                    res = None

        if res is None or getattr(res, "x", None) is None or not getattr(res, "success", False):
            fail_count += 1
            if fail_count > 10:
                break
            continue

        g_val = np.asarray(res.x, dtype=np.float64)
        g_val = np.where(np.isfinite(g_val), g_val, 0.0)
        g_val = np.maximum(g_val, 0.0)
        gs = float(g_val.sum())
        if gs < 1e-14:
            fail_count += 1
            if fail_count > 12:
                break
            continue

        fail_count = 0
        g_dir = g_val / gs
        cross = cross_conv(y, g_dir)

        if mask is None:
            viol = np.where(cross > M_target + 1e-10)[0]
        else:
            viol = np.where(mask & (cross > M_target + 1e-10))[0]

        if viol.size > 0:
            order = np.argsort(cross[viol])[::-1][:max(1, int(add_per_iter))]
            for v in viol[order]:
                constraints.append(int(v))

        constraints = list(set(constraints))
        if len(constraints) > max_K:
            constraints = [int(k) for k in top_indices(c, mask, max_K)]

        alpha = 2.0 / (float(it) + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y = normalize_nonneg(y_new)
        x = 0.5 * x + 0.5 * y
        x = normalize_nonneg(x)

        c = linear_conv(y)
        m = masked_max(c, mask)
        if np.isfinite(m) and m < best_m - 1e-12:
            best_m = m
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 3 == 0:
            record_candidate(y, label)

        if no_improve > 35:
            break

    record_candidate(best_a, label + "_final", force=True)
    return best_a


def interpolate_sequence(f, new_n):
    f = normalize_nonneg(f)
    old_n = int(f.size)
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([1.0], dtype=np.float64)
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, float(f[0]), dtype=np.float64)

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    return normalize_nonneg(g)


def parametric_candidates(n, parent=None):
    n = int(n)
    if n <= 0:
        return [np.array([1.0], dtype=np.float64)]

    out = []

    p = None
    if parent is not None:
        p = normalize_nonneg(parent)
        if p.size != n:
            p = interpolate_sequence(parent, n)
        if p is not None and np.isfinite(p).all():
            out.append(p.copy())
            out.append(normalize_nonneg(0.7 * p + 0.3 * np.ones(n) / float(n)))

    x = np.linspace(0.0, 1.0, n)
    t = np.linspace(-1.0, 1.0, n)
    t2 = t * t

    raw = []
    raw.append(np.ones(n, dtype=np.float64))

    for power in (0.25, 0.5, 1.0, 2.0, 4.0):
        raw.append(np.maximum(1.0 - t2, 1e-8) ** power)

    for power in (2, 4, 8, 16):
        raw.append(np.abs(np.cos(np.pi * (x - 0.5))) ** power)

    for sigma in (0.12, 0.20, 0.32):
        raw.append(np.exp(-((x - 0.5) ** 2) / (2.0 * sigma * sigma)))

    raw.append(np.exp(-((x - 0.25) ** 2) / (2.0 * 0.15 * 0.15)) +
               np.exp(-((x - 0.75) ** 2) / (2.0 * 0.15 * 0.15)))

    for freq in (1, 2, 3):
        raw.append(1.0 + 0.5 * np.sin(2.0 * np.pi * freq * x))

    raw.append(np.maximum(np.sin(np.pi * x) ** 2, 1e-8))
    raw.append(np.maximum(np.sin(2.0 * np.pi * x) ** 2, 1e-8))
    raw.append(np.maximum(np.sin(3.0 * np.pi * x) ** 2, 1e-8))

    if p is not None:
        raw.append(0.8 * p + 0.2 * np.ones(n) / float(n))
        raw.append(0.5 * p + 0.5 * np.ones(n) / float(n))

    for r in raw:
        r = np.asarray(r, dtype=np.float64)
        if r.size != n:
            continue
        r = np.where(np.isfinite(r), r, 0.0)
        r = np.maximum(r, 1e-10)
        s = float(r.sum())
        if s > 0.0:
            out.append(r / s)

    if not out:
        out.append(np.ones(n, dtype=np.float64) / float(n))

    return out


def choose_mask_from_inference(inference):
    usable = []
    for actual, full, off in inference:
        if np.isfinite(actual) and full is not None and off is not None:
            if full > 0.0 and off > 0.0:
                usable.append((float(actual), float(full), float(off)))

    if not usable:
        return 0

    for a, f, o in usable:
        if a < 1.985:
            return 1

    err_full = 0.0
    err_off = 0.0
    for a, f, o in usable:
        rf = a / f
        ro = a / o
        err_full += (rf - 1.0) * (rf - 1.0)
        err_off += (ro - 1.0) * (ro - 1.0)

    err_full /= float(len(usable))
    err_off /= float(len(usable))
    return 1 if err_off < err_full else 0


def random_actual_polish(a0, time_limit, label="rand"):
    if time_limit <= 10.0 or remaining_time() < 10.0:
        return
    a = normalize_nonneg(a0)
    n = int(a.size)
    if n <= 1:
        return

    rng = np.random.default_rng(98765)
    t0 = time.monotonic()
    max_evals = int(max(5, min(35, time_limit / 8.0)))

    for _ in range(max_evals):
        if time.monotonic() - t0 > time_limit or remaining_time() < 10.0:
            break

        top = np.argsort(a)[::-1][:max(1, min(80, n))]
        src = int(top[rng.integers(top.size)])
        low = np.argsort(a)[:max(1, min(80, n))]
        dst = int(low[rng.integers(low.size)])
        if src == dst or a[src] <= 1e-10:
            continue

        delta = float(a[src]) * float(rng.uniform(0.003, 0.025))
        trial = a.copy()
        trial[src] -= delta
        trial[dst] += delta
        trial = normalize_nonneg(trial)

        score = safe_evaluate(trial)
        if np.isfinite(score) and score < PROP_BEST_SCORE - 1e-12:
            PROP_BEST_SCORE = score
            PROP_BEST_A = trial.copy()
            print(f"NEW BEST [{label}] score={score:.9f} n={n} t={time.monotonic() - PROP_START:.1f}")

    for amp in (0.08, 0.20, 0.35):
        trial = normalize_nonneg((1.0 - amp) * a + amp * smooth_noise(n, rng, 0.45))
        score = safe_evaluate(trial)
        if np.isfinite(score) and score < PROP_BEST_SCORE - 1e-12:
            PROP_BEST_SCORE = score
            PROP_BEST_A = trial.copy()
            print(f"NEW BEST [{label}_noise] score={score:.9f} n={n} t={time.monotonic() - PROP_START:.1f}")
