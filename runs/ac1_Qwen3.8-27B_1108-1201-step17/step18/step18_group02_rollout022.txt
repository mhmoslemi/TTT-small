def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (cssv - 1.0) / np.arange(1, n + 1, dtype=np.float64) < u
    rho = np.where(cond)[0]
    if len(rho) == 0:
        rho = np.array([0])
    theta = (cssv[rho[0]] - 1.0) / (rho[0] + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_grad(a, tau):
    n = len(a)
    ac = fast_autocorr(a)
    ac_s = ac / tau
    ac_s -= ac_s.max()
    exp_v = np.exp(ac_s)
    p = exp_v / exp_v.sum()
    # grad[j] = 2 * sum_i a[i] * p[i+j]
    size = 2 * n - 1
    fft_size = 1 << (size + n - 2).bit_length()
    af = a[::-1].astype(np.float64)
    Af = np.fft.rfft(af, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_ap = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_ap[n - 1:n - 1 + n]
    return grad, p, ac

def multiplicative_subgrad(a0, time_limit, verbose=True):
    """Multiplicative subgradient with random lag sampling and adaptive rate."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    eta0 = 0.5 / n
    decay = 0.999
    eta = eta0
    
    for it in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_now = ac.max()
        
        if M_now < best_m - 1e-14:
            best_m = M_now
            best_a = a.copy()
        
        # Select active lags (top few)
        top_k = min(5, len(ac))
        active = np.argsort(ac)[::-1][:top_k]
        
        # Uniform mix over active lags
        lam = np.ones(top_k) / top_k
        
        # Build subgradient: g[j] = 2 * sum_k lam[k_idx] * a[k - j] (with bounds)
        g = np.zeros(n)
        for idx, k_val in enumerate(active):
            k_int = int(k_val)
            for j in range(n):
                ij = k_int - j
                if 0 <= ij < n:
                    g[j] += 2.0 * lam[idx] * a[ij]
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        a = a_new
        
        eta *= decay
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad it={it} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.2e}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    SubGrad done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def fista_run(a0, time_limit, verbose=True):
    """FISTA on smoothed objective with backtracking step."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    ac0 = fast_autocorr(a)
    M0 = ac0.max()
    
    tau = M0 * 0.25
    tau_min = M0 * 0.005
    tau_decay = 0.999
    
    best_a = a.copy()
    best_m = M0
    t0 = time.monotonic()
    
    x = a.copy()
    z = a.copy()
    t_mom = 1.0
    step = 0.4
    
    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad_z, p_z, ac_z = lse_grad(z, tau)
        M_z = ac_z.max()
        
        if M_z < best_m - 1e-14:
            best_m = M_z
            best_a = z.copy()
        
        # Gradient step
        x_new = z - step * grad_z
        x_new = project_simplex(x_new)
        
        # Backtracking check
        obj_new, _, _ = lse_grad(x_new, tau)
        obj_cur, _, _ = lse_grad(x, tau)
        # Quick check: use max of ac as proxy
        ac_new = fast_autocorr(x_new)
        if ac_new.max() > fast_autocorr(x).max() + 1e-12:
            step *= 0.5
            x_new = z - step * grad_z
            x_new = project_simplex(x_new)
        else:
            step = min(step * 1.1, 1.0)
        
        x = x_new
        
        # Nesterov
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z = x + (t_mom - 1.0) / t_new * (x - z)
        z = project_simplex(z)
        t_mom = t_new
        
        tau = max(tau * tau_decay, tau_min)
        
        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA it={it} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2e}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    FISTA done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def lp_fw_improved(a0, time_limit, solver, verbose=True):

            margins = np.array([M_y - ac_y[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_list = [constraint_list[i] for i in keep]
            constraint_set = set(constraint_list)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix: row j is the constraint conv(y,g)[constraint_list[j]] <= M
        # conv(y,g)[k] = sum_i y[i]*g[k-i]

        coeffs = np.zeros((K, n))
