    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    ratio = mag / (mag_mean + 1e-30)
    # Compress dynamic range
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np

def fast_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Normalize to probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_internal(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safely evaluate the sequence."""

def compute_subgradient(a, c, top_k=40, beta=40.0):
    """Compute weighted subgradient using top peaks with softmax weighting."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = c[top_idx].copy()
    exponents = beta * (top_vals - top_vals[0]) / top_vals[0]
    exponents = np.maximum(exponents, -50.0)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max

        w = w / w_sum
    grad = np.zeros(n)

        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

        if time.monotonic() - t0 > time_limit:
            break
        if stagnation > max_stagnation:
            break
        
        progress = min(1.0, t / 80000.0)
        beta_cur = beta_init + (beta_final - beta_init) * progress
        top_k_cur = max(top_k_final, int(top_k_init * (1 - 0.85 * progress) + top_k_final))
        lr_cur = max(0.005, lr_init * (lr_decay ** t))
        
        # Nesterov extrapolation
        mu = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mu = min(mu, 1.0) * momentum
        y = x + mu * (x - prev_a)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        c = fast_conv(y)
        g = compute_subgradient(y, c, top_k=top_k_cur, beta=beta_cur)
        
        # Mirror descent step in log-space
        log_y = np.log(np.maximum(y, 1e-30))
        log_new = log_y - lr_cur * g
        # Light regularization toward uniform for stability
        reg_strength = 0.005 * progress
        log_new = (1.0 - reg_strength) * log_new + reg_strength * (-np.log(n))
        
        max_log = log_new.max()
        a_new = np.exp(log_new - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        prev_a = x.copy()
        x = a_new
        a = a_new
        lr = max(0.003, lr * lr_decay)
        
        # Track best
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Periodic spectral smoothing
        if t % spectral_interval == 0 and t > 0:
            a_sp = spectral_flatten(a, strength=0.15 + 0.1 * (1 - progress))
            c_sp = fast_conv(a_sp)
            if c_sp.max() < best_m:
                best_m = c_sp.max()
                best_a = a_sp.copy()
                a = a_sp
                x = a_sp.copy()
                prev_a = a_sp.copy()
                stagnation = 0
        
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NMD t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, lr={lr:.4f}, beta={beta_cur:.0f}")
    
    return best_a, best_m


def spectral_flatten(a, strength=0.3):
    """Compress the dynamic range of the power spectrum."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - strength)
    mag_new = np.maximum(mag_new, 1e-30)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    a_out = (1.0 - strength * 0.5) * a + strength * 0.5 * a_new
    return norm_simplex(a_out)


def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening with adaptive step."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    step = 0.25
    no_improve = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 50:
            break
        
        a_trial = spectral_flatten(a, strength=step)
        c = fast_conv(a_trial)
        M = c.max()
        
        if M < best_m - 1e-12:
            best_m = M
            best_a = a_trial.copy()
            a = a_trial
            step = min(0.6, step * 1.03)
            no_improve = 0
        else:
            step = max(0.03, step * 0.85)
            no_improve += 1
    
    return best_a, best_m


def hill_climb(a0, time_limit, verbose=False):
    """Stochastic hill climbing: try random perturbations, keep improvements."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random perturbation: move mass between two positions
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        frac = rng.uniform(0.002, 0.05)
        amt = a[i] * frac
        if amt < 1e-12:
            continue
        
        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        M_trial = fast_conv(a_trial).max()
        
        if M_trial < best_m - 1e-12:
            a = a_trial
            best_m = M_trial
            best_a = a_trial.copy()
    
    return best_a, best_m


def make_init_gaussian(n, sigma_frac=0.25):
    x = np.linspace(0, 1, n)
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_init_bowl(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.05
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.03
    return norm_simplex(a)


def make_init_ramp(n, power=0.5):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_init_powerlaw(n, alpha=1.5, center=0.5):
    x = np.linspace(0.01, 0.99, n)
    dist = np.abs(x - center)
    a = 1.0 / (dist + 0.05) ** alpha
    return norm_simplex(a)


def make_init_trapezoid(n, edge_frac=0.3, edge_height=1.0):
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    n_edge = int(n * edge_frac)
    if n_edge > 0:
        ramp = np.linspace(edge_height, 0.3, n_edge)
        a[:n_edge] = ramp
        a[-n_edge:] = ramp[::-1]
    return norm_simplex(a)


def make_init_random_smooth(n, seed=42, n_fourier=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.8
    for f in range(1, n_fourier + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass from peak-contributing positions."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 400
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        n_top = min(15, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            # Try the top 3 contributors
            top_contrib = np.argsort(contrib)[::-1][:3]
            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-10:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    # Try multiple destinations
                    for d in range(3):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_m - 1e-12:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search: Nesterov mirror descent + spectral flattening + multi-start
    + coordinate descent + length variation to minimize the autocorrelation objective.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
    except Exception:
        pass
    
    if parent is not None:
        p_score = evaluate_safe(parent / parent.sum())
        best_seq = norm_simplex(parent)
        best_score = p_score
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        n_base = 4200
        best_seq = make_init_bowl(n_base, sharpness=1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Intensive Nesterov MD on parent (30% budget) ===
    print(f"\n=== Phase 1: Nesterov MD on parent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.28, 420.0)
    if t1 > 60:
        a_out, _ = nesterov_mirror_descent(norm_simplex(best_seq), t1, verbose=True)
        try_update(a_out, "nmd_parent")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_sp, _ = spectral_optimize(a_out, min(45.0, remaining * 0.15))
            try_update(a_sp, "spec_parent")
    
    # === Phase 2: Multi-start exploration (25% budget) ===
    print(f"\n=== Phase 2: Multi-start ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2_total = min(remaining * 0.25, 350.0)
    if t2_total > 60:
        inits = []
        labels = []
        
        inits.append(make_init_bowl(n0, sharpness=1.0))
        labels.append("bowl1")
        inits.append(make_init_bowl(n0, sharpness=2.0))
        labels.append("bowl2")
        inits.append(make_init_bowl(n0, sharpness=3.0))
        labels.append("bowl3")
        inits.append(make_init_gaussian(n0, sigma_frac=0.2))
        labels.append("gauss02")
        inits.append(make_init_gaussian(n0, sigma_frac=0.35))
        labels.append("gauss035")
        inits.append(make_init_cosine(n0, mode=1))
        labels.append("cos1")
        inits.append(make_init_dual_peak(n0, 0.15, 0.85, 0.12))
        labels.append("dual15")
        inits.append(make_init_dual_peak(n0, 0.25, 0.75, 0.08))
        labels.append("dual25")
        inits.append(make_init_trapezoid(n0, 0.3, 1.0))
        labels.append("trap")
        inits.append(make_init_powerlaw(n0, alpha=1.0, center=0.5))
        labels.append("powlaw")
        inits.append(make_init_random_smooth(n0, seed=7, n_fourier=10))
        labels.append("rand7")
        inits.append(make_init_random_smooth(n0, seed=13, n_fourier=6))
        labels.append("rand13")
        inits.append(make_init_sinc(n0, bw=4.0))
        labels.append("sinc4")
        inits.append(make_init_ramp(n0, power=0.3))
        labels.append("ramp03")
        
        # Deduplicate
        seen_set = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 15)], 5))
            if sig not in seen_set:
                seen_set.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        
        inits = unique_inits
        labels = unique_labels
        n_inits = len(inits)
        print(f"  {n_inits} inits, budget={t2_total:.0f}s")
        
        t_per = t2_total / n_inits
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now < 10:
                break
            t_avail = min(t_per, remaining_now * 0.9)
            if t_avail < 8:
                break
            print(f"    NMD '{lbl}' ({t_avail:.0f}s)...")
            a_out, _ = nesterov_mirror_descent(init, t_avail, verbose=False)
            try_update(a_out, f"nmd_{lbl}")
            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now > 10:
                a_sp, _ = spectral_optimize(a_out, min(12.0, remaining_now * 0.25))
                try_update(a_sp, f"spec_{lbl}")
    
    # === Phase 3: Coordinate descent + hill climbing (15% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)

    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:

        try_update(a_cd, "cd")
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.08, 80.0)
        if t3b > 10:
            a_hc, _ = hill_climb(norm_simplex(best_seq), t3b, verbose=False)
            try_update(a_hc, "hillclimb")
    
    # === Phase 4: Length variation (15% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length variation (base n={n_best}) ===")
    
    for delta in [300, 700, 1200]:
        new_n = n_best + delta
        if new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_{new_n}")
        t4 = min(remaining * 0.18, 130.0)
        if t4 > 20:
            a_out, _ = nesterov_mirror_descent(a_long, t4, verbose=False)
            try_update(a_out, f"nmd_{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_sp, _ = spectral_optimize(a_out, min(20.0, remaining * 0.2))
                try_update(a_sp, f"spec_{new_n}")
    
    for delta in [300, 700]:
        new_n = n_best - delta
        if new_n < 2000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  n={new_n} (shorter)...")
        a_short = interpolate_sequence(best_seq, new_n)
        try_update(a_short, f"interp_{new_n}")
        t4s = min(remaining * 0.15, 100.0)
        if t4s > 20:
            a_out, _ = nesterov_mirror_descent(a_short, t4s, verbose=False)
            try_update(a_out, f"nmd_{new_n}")
    
    # === Phase 5: Final intensive polish (17% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final polish (n={n_best}) ===")
    
    t5 = min(remaining * 0.45, 280.0)
    if t5 > 20:
        # Start from best and also from a perturbed version
        a_in = norm_simplex(best_seq)
        a_out, _ = nesterov_mirror_descent(a_in, t5 * 0.6, verbose=True)
        try_update(a_out, "final_nmd")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t5b = min(remaining * 0.2, 60.0)
            a_sp, _ = spectral_optimize(a_out, t5b, verbose=False)
            try_update(a_sp, "final_spec")
    
    # Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t5c = min(remaining * 0.5, 60.0)
        a_out, _ = coordinate_descent_polish(norm_simplex(best_seq), t5c, verbose=False)
        try_update(a_out, "final_cd")
    
    # Final hill climb
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t5d = min(remaining * 0.6, 40.0)
        a_out, _ = hill_climb(norm_simplex(best_seq), t5d, verbose=False)
        try_update(a_out, "final_hc")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_init_bowl(n_fb, sharpness=1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
