"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-subgradient multiplicative gradient descent (MGD) with adaptive temperature
2. L-BFGS-B polishing via softmax parameterization with temperature annealing
3. Multi-length exploration via interpolation and re-optimization
4. Greedy coordinate descent for final polish

import numpy as np
import time
from scipy.optimize import minimize


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def soft_max_value(c, tau):
    """Compute smooth approximation to max(c)."""
    c_max = np.max(c)
    return c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))


def compute_soft_gradient(a, tau):
    """
    Compute gradient of soft_max(conv(a,a), tau) with respect to a.
    Returns (grad, c_max) where grad[i] = 2 * sum_k p[k] * a[k-i].
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = np.max(c)
    # Softmax weights
    log_p = (c - c_max) / tau
    log_p -= np.max(log_p)
    p = np.exp(log_p)
    p = p / p.sum()

    # grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i in [0, n-1]
    # = 2 * (p cross_correlate a)[i]
    # Using FFT: (p cross_correlate a) = ifft(conj(FFT(p)) * FFT(a))
    fft_len = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_len)
    A = np.fft.rfft(a, fft_len)
    cross = np.fft.irfft(np.conj(P) * A, fft_len)
    grad = 2.0 * cross[:n]
    return grad, c_max


def mgd_soft(a0, time_limit, tau_start, tau_end, eta0=0.03, verbose=False):
    """
    Multiplicative Gradient Descent with soft-subgradient and adaptive temperature.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_cmax = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta0
    patience = 0
    max_patience = 200
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Adaptive temperature: start high (diffuse), end low (focused)
        tau = tau_start * (tau_end / tau_start) ** frac

        c = fast_autocorr(a)
        c_max = c.max()
        
        if c_max < best_cmax - 1e-15:
            best_cmax = c_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Compute soft gradient
        grad, _ = compute_soft_gradient(a, tau)
        
        # Center the gradient (remove mean for stability on simplex)
        grad_centered = grad - grad.mean()
        gmax = np.abs(grad_centered).max()
        if gmax < 1e-15:
            break
        grad_norm = grad_centered / gmax

        # Multiplicative update
        a_new = a * np.exp(-eta * grad_norm)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Check improvement
        c_new = fast_autocorr(a_new)
        c_max_new = c_new.max()
        
        if c_max_new <= c_max + 1e-12:
            a = a_new
            eta *= 0.999
        else:
            eta *= 0.5
            if eta < 0.001:
                eta = 0.02
                patience = max(0, patience - 50)

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            J = 2.0 * n * best_cmax
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.4f}, eta={eta:.5f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * len(a) * best_cmax
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def lbfgs_soft_polish(a0, tau_start, tau_end, max_iter=2000, verbose=False):
    """
    L-BFGS-B polish using softmax parameterization and soft-max temperature annealing.
    """
    n = len(a0)
    a = norm_sum1(a0)
    
    # Convert to log-space (with offset for zeros)
    a_safe = np.maximum(a, 1e-15)
    z0 = np.log(a_safe)
    z0 -= z0.max()  # center for numerical stability
    
    best_a = a.copy()
    best_cmax = fast_autocorr(a).max()
    
    # Annealing: split into phases
    n_phases = 4
    taus = np.linspace(tau_start, tau_end, n_phases + 1)
    
    z = z0.copy()
    for phase in range(n_phases):
        tau = 0.5 * (taus[phase] + taus[phase + 1])
        
        def objective(zv):
            az = np.exp(zv - np.max(zv))
            az = az / az.sum()
            c = fast_autocorr(az)
            val = soft_max_value(c, tau)
            return 2.0 * n * val
        
        def gradient(zv):
            az = np.exp(zv - np.max(zv))
            az = az / az.sum()
            a_local = az
            grad_a, _ = compute_soft_gradient(a_local, tau)
            # Softmax Jacobian: dz/dz_i = a_i*(e_i - a)
            # So grad_z = a * (grad_a - dot(a, grad_a))
            dot = np.dot(a_local, grad_a)
            grad_z = a_local * (grad_a - dot)
            return 2.0 * n * grad_z
        
        try:
            result = minimize(
                objective, z, jac=gradient,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            z = result.x
            a_local = np.exp(z - np.max(z))
            a_local = a_local / a_local.sum()
            cmax_local = fast_autocorr(a_local).max()
            if cmax_local < best_cmax:
                best_cmax = cmax_local
                best_a = a_local.copy()
                if verbose:
                    print(f"    L-BFGS-B phase {phase+1}: J={2.0*n*best_cmax:.6f}, tau={tau:.4f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS-B phase {phase+1} error: {e}")
    
    return best_a

        # Shuffle for diversity
        if len(peak_indices) > 0:
            shuffled = rng.permutation(len(peak_indices))
            peak_indices = peak_indices[shuffled]
        
        # Try perturbations on peak-contributing coordinates
        for idx in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            if a[int(idx)] < 1e-10:
                continue
            
            src = int(idx)
            # Find a good destination: low contribution to peaks
            dst_candidates = []
            for k2 in np.argsort(c)[::-1][:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    if d != src:
                        dst_candidates.append(d)
            
            # Pick destination with minimum a value among non-peak
            if len(dst_candidates) == 0:
                continue
            dst_arr = np.array(dst_candidates)
            # Score: how much a[dst] contributes to peaks
            scores = np.zeros(len(dst_arr))
            for i, d in enumerate(dst_arr):
                for k2 in np.argsort(c)[::-1][:3]:
                    k2i = int(k2)
                    if 0 <= k2i - d < n:
                        scores[i] += a[k2i - d]
            dst = int(dst_arr[np.argmin(scores)])
            
            for frac in step_sizes:
                amt = a[src] * frac
                if amt < 1e-12:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                cmax_t = fast_autocorr(a_trial).max()
                if cmax_t < best_cmax - 1e-14:
                    a = a_trial
                    best_cmax = cmax_t
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    cmax_new = fast_autocorr(a).max()
                    if cmax_new < best_cmax:
                        best_cmax = cmax_new
                        best_a = a.copy()
                        stall = 0
                        break
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_cmax
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n):
    """Generate several parametric initializations and return the best."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Sine squared
    candidates.append(np.sin(np.pi * x) ** 2)
    # Cosine bump
    candidates.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # Triangle
    candidates.append(1.0 - np.abs(2 * x - 1))
    # Power laws
    for p in [0.5, 1.0, 2.0]:
        candidates.append((i_arr + 1.0) ** (-p))
        candidates.append((n - i_arr).clip(1.0) ** (-p))
    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.15, 0.25, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sigma ** 2)))
    # Exponential
    for rate in [2, 5, 10]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))
    
    best = None
    best_score = float('inf')
    for cand in candidates:
        cand = norm_sum1(cand)
        cmax = fast_autocorr(cand).max()
        score = 2.0 * n * cmax
        if score < best_score:
            best_score = score
            best = cand.copy()
    
    return best, best_score

    best_seq = None
    best_score = float('inf')

    n_parent = 5200

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = norm_sum1(p)
            p_score = evaluate_safe(p_norm)
            if np.isfinite(p_score):
                best_seq = p_norm
                best_score = p_score
                n_parent = len(p)
                print(f"Parent loaded: n={n_parent}, score={p_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n0 = 5000
        a_p, s_p = parametric_init(n0)
        best_seq = a_p
        best_score = s_p
        print(f"Parametric init: n={n0}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Quick parametric search ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_p, s_p = parametric_init(n0)
        if s_p < best_score:
            best_score = s_p
            best_seq = a_p.copy()
            print(f"Parametric improved: {best_score:.6f}")
    
    # === Phase 2: MGD with soft subgradient (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 30:

        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  ** MGD improved: {sc:.6f}")
    
    # === Phase 3: L-BFGS-B polish ===
    remaining = time_budget - (time.monotonic() - t_global)

                                  max_iter=500, verbose=True)

            best_score = sc
            best_seq = a_out.copy()
            print(f"  ** L-BFGS-B improved: {sc:.6f}")
    
    # === Phase 4: MGD round 2 with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:

    t5 = min(remaining * 0.15, 200.0)
