def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

            y = x + beta_new * (x - y_prev)
            y = np.maximum(y, 1e-15)
            s_y = y.sum()
            if s_y > 1e-15:

            y_prev = x.copy()
            beta = beta_new

            y = x.copy()
            y_prev = x.copy()

        M_curr = ac.max()

        # Check current x against active set constraints
        ac_x = fast_autocorr(x)
        for k in active_set:
            if ac_x[k] > M_curr + 1e-10:
                # Already in active set, just update M_curr

```python
def compute_smooth_grad_a(a, w, n):
    """g_a[j] = 2 * sum_m a[m] * w[m+j]"""
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A_rev = np.fft.rfft(a[::-1], fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(A_rev * W, fft_size)
    h = conv[n-1:2*n-1]  # h[j] = sum_m a[m] * w[m+j] for j=0,...,n-1
    return 2.0 * h
```
