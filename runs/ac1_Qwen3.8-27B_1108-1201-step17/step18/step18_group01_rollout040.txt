def max_mode(ac, mode):
    n=(len(ac)+1)//2

def normalize_nonneg(a):

    if arr.ndim != 1:

def autocorr_fft(a):
    n=int(a.size)
    if n==0: return np.empty(0,dtype=np.float64)
    size=next_pow2_int(2*n-1)
    A=np.fft.rfft(a.astype(np.float64, copy=False), size)
    ac=np.fft.irfft(A*A, size)[:2*n-1]
    ac = np.maximum(ac, 0.0)
    return ac
```

def soft_objective_and_grad(a, mode, beta):
    ac=autocorr_fft(a)
    n=a.size
    M=max_mode(ac, mode)
    mask=get_mask(n, mode)
    if beta <= 0 or not np.any(mask):
        grad=np.zeros(n,dtype=np.float64)
        return M, grad, ac, M
    x=beta*ac
    x=np.where(mask, x, -1e300)
    xmax=np.max(x)
    if not np.isfinite(xmax):
       return M, np.zeros(n), ac, M
    p=np.exp(x-xmax)
    psum=p.sum()
    if psum <=0:
       L=0; p=np.zeros_like(p)
    else:
       L=float(xmax+math.log(psum)); p/=psum
    P2=p[::-1]
    size=next_pow2_int(len(P2)+n-1)

    grad=2.0*h[n-1:2*n-1][::-1]
    # Clean tiny imaginary none
    grad=np.where(np.isfinite(grad), grad, 0.0)

def make_basis(n, seed=123):

       b=b - b.mean()
       norm=np.linalg.norm(b)
       if norm>1e-8: outs.append(b/norm)
    if not outs: outs=[np.ones(n)]

def interpolate_to_length(a, new_n):
    a=np.asarray(a,dtype=np.float64).ravel()
    old_n=a.size

def make_basis(n, seed=123):
   rng=np.random.default_rng(seed)
   x=np.linspace(0,1,n)
   basis=[]
   basis.append(2*x-1)
   basis.append((2*x-1)**2 - 1.0/3.0)
   for f in (1,2,3,5):
      basis.append(np.cos(2*np.pi*f*x))
      basis.append(np.sin(2*np.pi*f*x))
   for c in (0.2,0.4,0.6,0.8):
      sigma=float(rng.uniform(0.03,0.15))
      cc=float(rng.uniform(0.1,0.9))
      basis.append(np.exp(-((x-cc)**2)/(2.0*sigma*sigma)))
   cleaned=[]
   for b in basis:
      b=np.asarray(b,dtype=np.float64)
      b=b - float(np.mean(b))
      norm=float(np.linalg.norm(b))
      if norm >1e-10:
         cleaned.append(b/norm)
   if not cleaned:
      cleaned=[np.ones(n,dtype=np.float64)]
   return np.stack(cleaned, axis=1)

"""
Robust search for non-negative step-function heights.

The program combines:
- direct black-box evaluation of sequences returned by the environment,
- fast FFT-based autocorrelation surrogates for several possible peak
  selections (full, center-excluded, positive-lag side),
- smooth log-sum-exp subgradient optimization on the simplex,
- active-lag mass-transfer polish,
- low-dimensional smooth basis deformation,
- random direct-polish,
- multi-length interpolation and re-optimization.

It tracks the best sequence according to the actual evaluator whenever the
evaluator is available and finite, and falls back to the surrogate if not.
"""

import numpy as np
import time
import math

_MASK_CACHE = {}
_DIRECT_OK = False


def next_pow2_int(m):
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_nonneg(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return np.array([1.0], dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-15:
        return np.ones(arr.shape, dtype=np.float64) / float(arr.size)
    arr = arr / s
    arr = np.minimum(arr, 1000.0)
    return arr


def interpolate_to_length(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    old_n = a.size
    new_n = int(max(1, new_n))
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if old_n == new_n:
        return normalize_nonneg(a)
    if old_n == 1:
        return np.full(new_n, 1.0 / float(new_n), dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    vals = np.interp(new_x, old_x, a)
    return normalize_nonneg(vals)


def autocorr_fft(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = arr.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = next_pow2_int(2 * n - 1)
    A = np.fft.rfft(arr, size)
    ac = np.fft.irfft(A * A, size)[:2 * n - 1]
    ac = np.maximum(ac, 0.0)
    return ac


def get_mask(n, mode):
    n = int(n)
    mod = str(mode)
    key = (n, mod)
    mask = _MASK_CACHE.get(key)
    if mask is None:
        N = 2 * n - 1
        mask = np.ones(N, dtype=bool)
        if mod == "off":
            if n > 1:
                mask[n - 1] = False
        elif mod == "side":
            if n > 1:
                mask[:n - 1] = False
        _MASK_CACHE[key] = mask
        if len(_MASK_CACHE) > 80:
            _MASK_CACHE.clear()
            _MASK_CACHE[key] = mask
    return mask


def max_mode(ac, mode):
    ac = np.asarray(ac, dtype=np.float64)
    if ac.size == 0:
        return 0.0
    n = (ac.size + 1) // 2
    mod = str(mode)
    if mod == "full":
        return float(np.max(ac))
    if mod == "off":
        if n <= 1:
            return 0.0
        left = float(np.max(ac[:n - 1])) if n - 1 > 0 else 0.0
        right = float(np.max(ac[n:])) if n < ac.size else 0.0
        return max(left, right)
    if mod == "side":
        if n >= ac.size:
            return 0.0
        return float(np.max(ac[n:]))
    return float(np.max(ac))


def surrogate_score_modes(a, modes):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = arr.size
    if n == 0:
        return [float("inf")] * len(modes)
    s = float(arr.sum())
    if s <= 1e-15:
        return [float("inf")] * len(modes)
    ac = autocorr_fft(arr)
    inv_s2 = 1.0 / (s * s)
    out = []
    for m in modes:
        M = max_mode(ac, m)
        out.append(float(2.0 * n * M * inv_s2))
    return out


def soft_value_and_hard(a, mode, beta):
    arr = np.asarray(a, dtype=np.float64).ravel()
    ac = autocorr_fft(arr)
    n = arr.size
    M = max_mode(ac, mode)
    if n == 0 or beta <= 0.0:
        return float(M), float(M)
    mask = get_mask(n, mode)
    if not np.any(mask):
        return 0.0, float(M)
    x = beta * ac
    x = np.where(mask, x, -1e300)
    xmax = float(np.max(x))
    if not math.isfinite(xmax):
        return 0.0, float(M)
    p = np.exp(x - xmax)
    total = float(p.sum())
    if total <= 0.0 or not math.isfinite(total):
        return 0.0, float(M)
    L = float(xmax + math.log(total))
    return L, float(M)


def soft_objective_and_grad(a, mode, beta):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = arr.size
    ac = autocorr_fft(arr)
    M = max_mode(ac, mode)
    if n == 0 or beta <= 0.0:
        return float(M), np.zeros(n, dtype=np.float64), ac, float(M)
    mask = get_mask(n, mode)
    if not np.any(mask):
        return float(M), np.zeros(n, dtype=np.float64), ac, float(M)
    x = beta * ac
    x = np.where(mask, x, -1e300)
    xmax = float(np.max(x))
    if not math.isfinite(xmax):
        return float(M), np.zeros(n, dtype=np.float64), ac, float(M)
    p = np.exp(x - xmax)
    psum = float(p.sum())
    if psum <= 0.0 or not math.isfinite(psum):
        L = 0.0
        p = np.zeros_like(p)
    else:
        L = float(xmax + math.log(psum))
        p = p / psum

    P2 = p[::-1]
    size = next_pow2_int(len(P2) + n - 1)
    Fp = np.fft.rfft(P2, size)
    Fa = np.fft.rfft(arr, size)
    h = np.fft.irfft(Fp * Fa, size)
    grad = 2.0 * h[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(L), grad, ac, float(M)


def lag_grad(a, top_idx, weights):
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    for ww, k in zip(weights, top_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1, dtype=np.int64)
        grad[idx] += 2.0 * float(ww) * a[k - idx]
    return grad

    best_mode = "full"

    for i, m in enumerate(modes):

def make_basis(n, seed=123):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    basis = []
    basis.append(2.0 * x - 1.0)
    basis.append((2.0 * x - 1.0) ** 2 - 1.0 / 3.0)
    for f in (1, 2, 3, 5):
        basis.append(np.cos(2.0 * np.pi * f * x))
        basis.append(np.sin(2.0 * np.pi * f * x))
    for c in (0.2, 0.4, 0.6, 0.8):
        sigma = float(rng.uniform(0.03, 0.15))
        cc = float(rng.uniform(0.1, 0.9))
        basis.append(np.exp(-((x - cc) ** 2) / (2.0 * sigma * sigma)))

    cleaned = []
    for b in basis:
        b = np.asarray(b, dtype=np.float64)
        b = b - float(np.mean(b))
        norm = float(np.linalg.norm(b))
        if norm > 1e-10:
            cleaned.append(b / norm)
    if not cleaned:
        cleaned = [np.ones(n, dtype=np.float64)]
    return np.stack(cleaned, axis=1)

            best_cand_surr = local_surr

            for delta in (0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.2, -0.2):
                new_coeff = float(coeffs[idx]) + float(delta)
                if abs(new_coeff) > 0.9:
                    continue
                new_log = logf + (new_coeff - coeffs[idx]) * ph
                factor = np.exp(np.clip(new_log, -3.0, 3.0))
                cand = base * factor
                cand = normalize_nonneg(cand)
                ac2 = autocorr_fft(cand)
                M2 = max_mode(ac2, mode)
                surr2 = 2.0 * cand.size * M2
                if surr2 < best_cand_surr - 1e-12:
                    best_cand_surr = surr2
                    best_cand = cand
                    best_delta = float(delta)

            if best_cand is None:
                continue

            coeffs[idx] += best_delta
            logf = logf + best_delta * ph
            search = best_cand
            local_surr = best_cand_surr
            coord_seen += 1

            if _DIRECT_OK:
                if coord_seen % 6 == 0:
                    sc = direct_score(search)
                    if math.isfinite(sc) and sc < best_score - 1e-15:
                        best_score = sc
                        best_seq = search.copy()
                        round_improved = True
            else:
                if local_surr < best_score - 1e-12:
                    best_score = local_surr
                    best_seq = search.copy()
                    round_improved = True

        if round_improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if no_improve_rounds == 2:
                coeffs = coeffs * 0.5
                logf = basis @ coeffs
                factor = np.exp(np.clip(logf, -3.0, 3.0))
                search = normalize_nonneg(base * factor)
                ac2 = autocorr_fft(search)
                M2 = max_mode(ac2, mode)
                local_surr = 2.0 * search.size * M2
                if not _DIRECT_OK and local_surr < best_score - 1e-12:
                    best_score = local_surr
                    best_seq = search.copy()
            if no_improve_rounds >= 6:
                break

    if _DIRECT_OK:
        sc = direct_score(search)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_seq = search.copy()
    else:
        if local_surr < best_score - 1e-12:
            best_score = local_surr
            best_seq = search.copy()

    return best_seq, best_score


def blackbox_polish(a0, current_score, deadline, mode, time_limit):
    t0 = time.monotonic()
    limit = min(deadline, t0 + float(time_limit))
    a = normalize_nonneg(a0)
    n = a.size

    ac = autocorr_fft(a)
    M = max_mode(ac, mode)
    surrogate = 2.0 * n * M

    best_seq = a.copy()
    best_score = float(current_score) if math.isfinite(current_score) else float("inf")
    if best_score == float("inf") and not _DIRECT_OK:
        best_score = surrogate

    rng = np.random.default_rng(101)
    no_improve = 0
    iters = 0

    while time.monotonic() < limit and iters < 5000:
        iters += 1
        r = float(rng.random())
        cand = a.copy()

        if r < 0.35:
            max_bs = max(3, min(n, 100))
            bs = int(rng.integers(2, max_bs + 1))
            hi = max(1, n - bs + 1)
            start = int(rng.integers(0, hi))
            factor = float(rng.uniform(0.4, 2.0))
            cand[start:start + bs] *= factor
        elif r < 0.65:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = float(rng.uniform(0.001, 0.02)) * max(float(a[i]), 1e-8)
                cand[i] -= amt
                cand[j] += amt
        elif r < 0.85:
            noise = float(rng.uniform(0.002, 0.01)) * rng.normal(size=n)
            cand = cand * (1.0 + noise)
        else:
            bs = int(rng.integers(1, max(2, min(n, 50))))
            hi = max(1, n - bs + 1)
            start = int(rng.integers(0, hi))
            cand[start:start + bs] = float(rng.uniform(0.01, 0.2)) * max(float(cand.mean()), 1e-12)

        cand = normalize_nonneg(cand)
        if _DIRECT_OK:
            sc = direct_score(cand)
        else:
            ac2 = autocorr_fft(cand)
            M2 = max_mode(ac2, mode)
            sc = 2.0 * cand.size * M2

        if math.isfinite(sc) and sc < best_score - 1e-12:
            a = cand
            best_seq = cand.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400 and iters > 200:
                break

    if _DIRECT_OK:
        sc = direct_score(a)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_seq = a.copy()

    return best_seq, best_score


def multilength_search(a0, current_score, deadline, mode, time_limit):
    t0 = time.monotonic()
    limit = min(deadline, t0 + float(time_limit))
    best = a0.copy()
    best_score = float(current_score) if math.isfinite(current_score) else float("inf")

    if best_score == float("inf") and not _DIRECT_OK:
        ac = autocorr_fft(best)
        best_score = 2.0 * best.size * max_mode(ac, mode)

    n0 = best.size
    lengths = []
    for off in (-2000, -1000, -500, -200, 0, 200, 500, 1000, 2000, 4000):
        ll = int(n0 + off)
        if 900 <= ll <= 10000:
            lengths.append(ll)
    for ll in (1500, 2500, 4000, 6000, 8000, 10000):
        lengths.append(ll)

    items = []
    for ll in set(lengths):
        items.append((abs(ll - n0), int(ll)))
    items.sort()
    lengths = [ll for _, ll in items]

    pure = 0
    for ll in lengths:
        if time.monotonic() >= limit - 60 or pure >= 6:
            break
        cand = interpolate_to_length(best, ll) if ll != best.size else best.copy()
        if _DIRECT_OK:
            sc = direct_score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best = cand
                best_score = sc
        else:
            ac = autocorr_fft(cand)
            M = max_mode(ac, mode)
            sc = 2.0 * cand.size * M
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best = cand
                best_score = sc
        pure += 1

    opt_set = set()
    opt_set.add(int(best.size))
    for ll in lengths[:6]:
        opt_set.add(int(ll))
    opt_set.add(int(max(900, int(best.size * 0.8))))
    opt_set.add(int(min(10000, int(best.size * 1.3))))

    opt_items = []
    for ll in opt_set:
        if 900 <= ll <= 10000:
            opt_items.append((abs(ll - best.size), int(ll)))
    opt_items.sort()
    opt_lengths = [ll for _, ll in opt_items[:3]]

    for ll in opt_lengths:
        if time.monotonic() >= limit - 80:
            break
        cand = interpolate_to_length(best, ll) if ll != best.size else best.copy()
        if _DIRECT_OK:
            work_sc = direct_score(cand)
        else:
            ac = autocorr_fft(cand)
            M = max_mode(ac, mode)
            work_sc = 2.0 * cand.size * M

        rem = limit - time.monotonic()
        if rem <= 30:
            break
        alloc = min(150.0, max(30.0, rem * 0.35))
        seq, sc, _, _ = soft_mgd(
            cand, work_sc, deadline, mode, alloc,
            sharp_start=15.0, sharp_end=90.0, eta0=0.08, direct_every=40
        )
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best = seq
            best_score = sc
            print(f"  multilength n={ll} improved score={sc:.6f}")

        rem = limit - time.monotonic()
        if rem <= 30:
            break
        alloc2 = min(60.0, max(15.0, rem * 0.5))
        seq2, sc2 = subgrad_mass_polish(best, best_score, deadline, mode, alloc2)
        if math.isfinite(sc2) and sc2 < best_score - 1e-15:
            best = seq2
            best_score = scen2 if False else sc2

    if _DIRECT_OK:
        sc = direct_score(best)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc

    return best, best_score


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    safe_budget = max(60.0, float(budget_s) - 25.0)
    deadline = t0 + min(safe_budget, 1775.0)

    global_best_seq = None
    global_best_score = float("inf")

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.size > 0 and float(p.sum()) > 1e-8:
            if p.size > 12000:
                p = interpolate_to_length(p, 8000)
            p = normalize_nonneg(p)
            sc = direct_score(p)
            global_best_seq = p
            if math.isfinite(sc):
                global_best_score = sc
                print(f"parent cached: n={p.size}, score={sc:.6f}")
            else:
                global_best_score = float("inf")
                print(f"parent cached: n={p.size}, direct score not finite")
    except Exception:
        pass

    if global_best_seq is None:
        param_deadline = min(deadline, t0 + 45.0)
        global_best_seq, global_best_score = parametric_candidates(param_deadline)
        print(f"starting parametric: n={global_best_seq.size}, score={global_best_score:.6f}")

    if global_best_seq is None:
        x = np.linspace(0.0, 1.0, 4000)
        global_best_seq = normalize_nonneg(np.sin(np.pi * x) ** 2)
        global_best_score = direct_score(global_best_seq)

    mode = calibrate_mode(global_best_seq, global_best_score, deadline)

    remaining = deadline - time.monotonic()
    if remaining > 120:
        alloc = min(450.0, max(60.0, remaining * 0.30))
        seq, sc, _, _ = soft_mgd(
            global_best_seq, global_best_score, deadline, mode, alloc,
            sharp_start=5.0, sharp_end=80.0, eta0=0.12, direct_every=30
        )
        if math.isfinite(sc) and sc < global_best_score - 1e-15:
            global_best_seq = seq
            global_best_score = sc
            print(f"main soft_mgd improved score={sc:.6f}")

    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(240.0, max(40.0, remaining * 0.18))
        seq, sc = subgrad_mass_polish(global_best_seq, global_best_score, deadline, mode, alloc)
        if math.isfinite(sc) and sc < global_best_score - 1e-15:
            global_best_seq = seq
            global_best_score = sc
            print(f"mass polish improved score={sc:.6f}")

    remaining = deadline - time.monotonic()
    if remaining > 70:
        alloc = min(180.0, max(30.0, remaining * 0.12))
        seq, sc = basis_polish(global_best_seq, global_best_score, deadline, mode, alloc)
        if math.isfinite(sc) and sc < global_best_score - 1e-15:
            global_best_seq = seq
            global_best_score = sc
            print(f"basis polish improved score={sc:.6f}")

    remaining = deadline - time.monotonic()
    if remaining > 150:
        alloc = min(420.0, max(60.0, remaining * 0.35))
        seq, sc = multilength_search(global_best_seq, global_best_score, deadline, mode, alloc)
        if math.isfinite(sc) and sc < global_best_score - 1e-15:
            global_best_seq = seq
            global_best_score = sc
            print(f"multilength improved score={sc:.6f}")

    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(150.0, max(30.0, remaining * 0.10))
        seq, sc = blackbox_polish(global_best_seq, global_best_score, deadline, mode, alloc)
        if math.isfinite(sc) and sc < global_best_score - 1e-15:
            global_best_seq = seq
            global_best_score = sc
            print(f"blackbox improved score={sc:.6f}")

    remaining = deadline - time.monotonic()
    if remaining > 35:
        alloc = min(250.0, max(25.0, remaining * 0.5))
        seq, sc, _, _ = soft_mgd(
            global_best_seq, global_best_score, deadline, mode, alloc,
            sharp_start=25.0, sharp_end=120.0, eta0=0.05, direct_every=50
        )
        if math.isfinite(sc) and sc < global_best_score - 1e-15:
            global_best_seq = seq
            global_best_score = sc
            print(f"final soft_mgd improved score={sc:.6f}")

    if global_best_seq is None:
        x = np.linspace(0.0, 1.0, 4000)
        global_best_seq = normalize_nonneg(np.sin(np.pi * x) ** 2)
        global_best_score = direct_score(global_best_seq)

    final_seq = np.asarray(global_best_seq, dtype=np.float64).ravel()
    if final_seq.size > 20000:
        final_seq = interpolate_to_length(final_seq, 10000)

    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    final_seq = np.maximum(final_seq, 0.0)
    final_seq = final_seq + 1e-15
    final_seq = final_seq / max(float(final_seq.sum()), 1e-15)
    final_seq = np.minimum(final_seq, 1000.0)

    if _DIRECT_OK:
        fs = direct_score(final_seq)
        if math.isfinite(fs):
            global_best_score = fs

    elapsed = time.monotonic() - t0
    print(f"FINAL: score={global_best_score:.6f}, n={final_seq.size}, elapsed={elapsed:.0f}s")
    return [float(v) for v in final_seq]
