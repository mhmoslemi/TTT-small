```python
"""
Search for non-negative coefficient sequences minimizing
J(a) = 2*n*max(np.convolve(a,a)) / sum(a)^2.

Main ideas:
- FFT-based fast self-convolution and gradient of weighted convolution.
- Mirror/smoothed max descent on the simplex with adaptive line search.
- Multi-length and multi-start exploration, simulated-annealing random polish.
- Final validation against the supplied evaluator and safe fallback.
"""

import numpy as np
import time


def _power_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _conv_self(a):
    """Linear convolution a*a by zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = 2 * n - 1
    N = _power_two(size)
    F = np.fft.rfft(a, N)
    c = np.fft.irfft(F * F, N)
    c = c[:size]
    c = np.where(c > 0.0, c, 0.0)
    return c


def _normalize(a):
    """Non-negative, cap at 1000, sum to one."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _fast_score(a):
    """Internal fast value of the objective."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _conv_self(a / s)
    return float(2.0 * n * float(np.max(c)))


def _maybe_update_best(a, best_score, best_seq):
    if a is None:
        return best_score, best_seq
    sc = _fast_score(a)
    if np.isfinite(sc) and sc < best_score:
        return sc, _normalize(a).copy()
    return best_score, best_seq


def _weights_from_c(c, temp):
    """Log-sum-exp style weights, relative emphasis on current large lags."""
    c = np.asarray(c, dtype=np.float64)
    c = np.where(c > 0.0, c, 0.0)
    m = float(np.max(c))
    L = c.size
    if m <= 0.0:
        w = np.zeros(L, dtype=np.float64)
        if L > 0:
            w[0] = 1.0
        return w
    tau = max(1e-300, m * float(temp))
    z = (c - m) / tau
    w = np.exp(z)
    total = float(np.sum(w))
    if not np.isfinite(total) or total <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        active = c > m * (1.0 - 1e-12)
        if not np.any(active):
            w[int(np.argmax(c))] = 1.0
        else:
            w[active] = 1.0
    return w


def _gradient_from_weights(a, w):
    """
    Gradient of F_w(a)=sum_k w_k (a*a)_k.
    For c_k = sum_i a_i a_{k-i}, d c_k / d a_i = 2 a_{k-i}.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    if n == 0:
        return np.array([], dtype=np.float64)
    if w.size != L:
        tmp = np.zeros(L, dtype=np.float64)
        m = min(L, w.size)
        tmp[:m] = w[:m]
        w = tmp
    h = w[::-1]
    conv_len = n + L - 1
    N = _power_two(conv_len)
    Fa = np.fft.rfft(a, N)
    Fh = np.fft.rfft(h, N)
    conv = np.fft.irfft(Fa * Fh, N)
    seg = conv[n - 1:2 * n - 1]
    return 2.0 * seg[::-1]


def _standardize_direction(g, rng):
    """Produce a non-negative direction with useful dynamic range."""
    g = np.asarray(g, dtype=np.float64)
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    gmin = float(np.min(g))
    u = g - gmin
    pos = u[u > 1e-300]
    if pos.size == 0:
        u = np.abs(rng.normal(0.0, 1.0, u.size))
        u = u - float(np.min(u))
        pos = u[u > 1e-300]
    if pos.size > 0:
        scale = float(np.percentile(pos, 90.0))
        if not np.isfinite(scale) or scale <= 1e-300:
            scale = float(np.max(pos))
        if scale > 1e-300:
            u = u / scale
    if float(np.max(u)) <= 1e-15:
        u = np.abs(rng.normal(0.0, 1.0, u.size))
        u = u - float(np.min(u))
        scale = float(np.max(u))
        if scale > 1e-300:
            u = u / scale
    u = np.maximum(u, 0.0)
    u = np.minimum(u, 8.0)
    return u


def _perturb_log(a, rng, sigma):
    """Multiplicative log-normal perturbation, kept on the simplex."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    loga = np.log(np.maximum(a, 1e-300))
    loga = loga + rng.normal(0.0, sigma, a.size)
    loga = np.nan_to_num(loga, nan=0.0, posinf=50.0, neginf=-50.0)
    m = float(np.max(loga))
    b = np.exp(loga - m)
    s = float(b.sum())
    if not np.isfinite(s) or s < 1e-15:
        b = np.ones(a.size, dtype=np.float64) / a.size
    else:
        b = b / s
    return b


def _interpolate_sequence(a, new_n):
    a = _normalize(a)
    new_n = int(max(1, new_n))
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / new_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    b = np.interp(new_x, old_x, a, left=0.0, right=0.0)
    return _normalize(b)


def _make_starts(n, parent, rng, max_starts=8):
    n = int(max(100, n))
    starts = []

    if parent is not None and parent.size > 0:
        p = _interpolate_sequence(parent, n)
        starts.append(p)
        ss = (p + p[::-1]) * 0.5
        starts.append(_normalize(ss))
        if len(starts) < max_starts:
            starts.append(_perturb_log(p, rng, 0.25))

    while len(starts) < max_starts:
        kind = int(rng.integers(0, 6))
        if kind == 0:
            arr = rng.exponential(1.0, n)
        elif kind == 1:
            arr = rng.beta(2.0, 2.0, n)
        elif kind == 2:
            base = rng.exponential(1.0, n)
            kernel = np.ones(5, dtype=np.float64) / 5.0
            arr = np.convolve(base, kernel, mode='same')
        elif kind == 3:
            x = np.linspace(0.0, 1.0, n)
            m = int(rng.integers(1, 5))
            alpha = float(rng.uniform(0.2, 0.9))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            arr = 1.0 + alpha * np.sin(2.0 * np.pi * m * x + phase)
            arr = np.maximum(arr, 0.0)
        elif kind == 4:
            alpha = float(rng.uniform(0.3, 2.0))
            beta = float(rng.uniform(0.3, 2.0))
            i = np.arange(n, dtype=np.float64)
            arr = (i + 1.0) ** alpha * (n - i) ** beta
            arr = np.maximum(arr, 0.0)
        else:
            x = np.linspace(0.0, 1.0, n)
            m1 = int(rng.integers(1, 4))
            m2 = int(rng.integers(1, 6))
            a1 = float(rng.uniform(0.0, 1.0))
            a2 = float(rng.uniform(0.0, 1.0))
            p1 = float(rng.uniform(0.0, 2.0 * np.pi))
            p2 = float(rng.uniform(0.0, 2.0 * np.pi))
            arr = (1.0
                   + 0.45 * a1 * np.cos(2.0 * np.pi * m1 * x + p1)
                   + 0.45 * a2 * np.cos(2.0 * np.pi * m2 * x + p2))
            arr = np.maximum(arr, 0.01)

        arr = _normalize(arr)
        if arr.size == n and np.all(np.isfinite(arr)):
            starts.append(arr)

    return starts


def _optimize_mirror(a0, time_limit, seed,
                     start_temp=1.5, end_temp=0.08,
                     start_eta=0.08, max_eta=0.6, min_eta=1e-4,
                     momentum=0.5, n_eta=3, accept_prob=0.25):
    if time_limit <= 0:
        a = _normalize(a0)
        return a, _fast_score(a)

    rng = np.random.default_rng(seed)
    a = _normalize(a0)
    n = a.size
    c = _conv_self(a)
    cur_max = float(np.max(c))
    best_a = a.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max

    mo = np.zeros(n, dtype=np.float64)
    eta = start_eta
    no_accept = 0

    t0 = time.monotonic()
    t_end = t0 + float(time_limit)
    last_best = t0
    stall_threshold = max(12.0, 0.08 * float(time_limit))
    iter_count = 0

    while time.monotonic() < t_end:
        elapsed = time.monotonic() - t0
        phase = min(1.0, elapsed / max(1e-9, 0.9 * float(time_limit)))

        if phase < 0.5:
            temp = start_temp + (0.25 - start_temp) * (phase / 0.5)
        else:
            temp = 0.25 + (end_temp - 0.25) * ((phase - 0.5) / 0.5)

        if no_accept > 2000:
            temp = 0.1 + 2.5 * float(rng.random())
        temp = max(1e-6, float(temp))

        c = np.maximum(c, 0.0)
        cur_max = float(np.max(c))
        cur_score = 2.0 * n * cur_max

        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_score = cur_score
            best_a = a.copy()
            last_best = time.monotonic()

        w = _weights_from_c(c, temp)
        g = _gradient_from_weights(a, w)
        u = _standardize_direction(g, rng)

        if momentum > 0.0:
            mo = momentum * mo + (1.0 - momentum) * u
            vmin = float(np.min(mo))
            direction = mo - vmin
            rstd = float(np.sqrt(np.mean(direction * direction)))
            if rstd > 1e-18:
                direction = direction / rstd
            else:
                direction = u
        else:
            direction = u

        direction = np.minimum(direction, 8.0)
        dmax = float(np.max(direction))
        if dmax <= 1e-15:
            direction = np.exp(-np.abs(rng.normal(0.0, 1.0, n)))
            direction = direction - float(np.min(direction))
            dmax = float(np.max(direction))
            if dmax > 1e-15:
                direction = direction / dmax

        trial_best_max = float('inf')
        trial_a = None
        trial_c = None

        for eidx in range(int(max(1, n_eta))):
            e = eta / (2.0 ** eidx)
            if e < min_eta and eidx > 0:
                continue
            logf = -e * direction
            logf = np.maximum(logf, -60.0)
            fac = np.exp(logf)
            ap = a * fac
            sp = float(ap.sum())
            if not np.isfinite(sp) or sp < 1e-12:
                continue
            ap = ap / sp
            cp = _conv_self(ap)
            mp = float(np.max(cp))
            if mp < trial_best_max:
                trial_best_max = mp
                trial_a = ap
                trial_c = cp

        accept = False
        if trial_a is not None and trial_best_max < cur_max - 1e-14:
            accept = True
            eta = min(max_eta, eta * 1.15)
        else:
            tol = max(1e-8, 2e-5 * cur_max)
            if trial_a is not None and trial_best_max < cur_max + tol and rng.random() < accept_prob:
                accept = True
                eta = max(min_eta, eta * 0.9)

        if accept:
            a = trial_a
            c = trial_c
            cur_max = trial_best_max
            no_accept = 0
            if cur_max < best_max - 1e-15:
                best_max = cur_max
                best_score = 2.0 * n * cur_max
                best_a = a.copy()
                last_best = time.monotonic()
        else:
            no_accept += 1
            eta = max(min_eta, eta * 0.8)
            cur_score_check = 2.0 * n * cur_max
            best_score_check = 2.0 * n * best_max
            if (no_accept > 5000 or
                    (cur_score_check > best_score_check * (1.0 + 1e-3) and no_accept > 300) or
                    (time.monotonic() - last_best > 2.0 * stall_threshold) or
                    (time.monotonic() - last_best > stall_threshold and no_accept > 50)):
                sigma = 0.05 + 0.30 * float(rng.random())
                a = _perturb_log(best_a, rng, sigma)
                c = _conv_self(a)
                cur_max = float(np.max(c))
                no_accept = 0
                eta = max(min_eta, start_eta * 0.7)
                mo.fill(0.0)

        iter_count += 1
        if iter_count % 20000 == 0:
            print(f"    mirror seed={seed} best J={best_score:.6f} current J={2.0 * n * cur_max:.6f}")

    return best_a, best_score


def _random_search(a0, time_limit, seed):
    """Fast simulated-annealing multiplicative random search."""
    if time_limit <= 0:
        a = _normalize(a0)
        return a, _fast_score(a)

    rng = np.random.default_rng(seed)
    a = _normalize(a0)
    n = a.size
    c = _conv_self(a)
    cur_max = float(np.max(c))
    best_a = a.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max
    last_best = time.monotonic()
    t0 = time.monotonic()
    t_end = t0 + float(time_limit)
    no_improve = 0
    it = 0
    T_start = 5e-5
    T_end = 1e-6

    while time.monotonic() < t_end:
        elapsed = time.monotonic() - t0
        phase = min(1.0, elapsed / max(1e-9, float(time_limit)))
        sigma = 0.02 + 0.12 * (1.0 - phase) + 0.05 * float(rng.random())
        source = best_a if float(rng.random()) < 0.3 else a
        a2 = _perturb_log(source, rng, sigma)
        c2 = _conv_self(a2)
        m2 = float(np.max(c2))
        s2 = 2.0 * n * m2

        T = T_start + (T_end - T_start) * phase
        accept = False
        if np.isfinite(s2):
            if s2 < 2.0 * n * cur_max - 1e-14:
                accept = True
            else:
                up_tol = T * max(1.0, 2.0 * n * cur_max)
                if s2 < 2.0 * n * cur_max + up_tol and float(rng.random()) < 0.4:
                    accept = True

        if accept:
            a = a2
            c = c2
            cur_max = m2
            no_improve = 0
            if s2 < best_score:
                best_score = s2
                best_max = m2
                best_a = a.copy()
                last_best = time.monotonic()
            else:
                no_improve += 1
        else:
            no_improve += 1

        cur_score = 2.0 * n * cur_max
        if (no_improve > 5000 or
                cur_score > best_score * (1.0 + 5e-4) or
                time.monotonic() - last_best > max(20.0, 0.15 * float(time_limit))):
            a = best_a.copy()
            c = _conv_self(a)
            cur_max = float(np.max(c))
            no_improve = 0

        it += 1
        if it % 10000 == 0:
            print(f"    random seed={seed} best J={best_score:.6f}")

    return best_a, best_score


def _alternative_candidates(a, max_candidates=18):
    a = _normalize(a)
    out = []
    out.append(_normalize((a + a[::-1]) * 0.5))

    n = a.size
    for frac in (0.02, 0.05, 0.10, 0.20):
        m = int(n * frac)
        if n > 2 * m + 20:
            out.append(_normalize(a[m:n - m]))
        if m > 0 and n - m > 100:
            out.append(_normalize(a[m:]))
            out.append(_normalize(a[:n - m]))

    k3 = np.array([0.25, 0.50, 0.25], dtype=np.float64)
    out.append(_normalize(np.convolve(a, k3, mode='full')))

    k5 = np.ones(5, dtype=np.float64) / 5.0
    out.append(_normalize(np.convolve(a, k5, mode='full')))

    if n % 2 == 0 and n // 2 >= 200:
        out.append(_normalize(a.reshape(-1, 2).mean(axis=1)))

    if n > 3000:
        out.append(_interpolate_sequence(a, int(n * 0.9)))
        out.append(_interpolate_sequence(a, int(n * 0.8)))

    return out[:max_candidates]


def _localize_length(n, parent, time_budget, seed):
    n = int(max(200, n))
    if time_budget <= 0:
        return None, float('inf')

    t0 = time.monotonic()
    t_end = t0 + float(time_budget)
    rng = np.random.default_rng(seed)

    starts = _make_starts(n, parent, rng, max_starts=8)
    ts = []
    ta = []

    for arr in starts:
        if arr is None:
            continue
        arr = _normalize(arr)
        if arr.size < 100:
            continue
        sc = _fast_score(arr)
        if not np.isfinite(sc):
            continue
        if len(ts) < 3:
            ts.append(sc)
            ta.append(arr)
        else:
            worst = 0
            for i in range(1, 3):
                if ts[i] > ts[worst]:
                    worst = i
            if sc < ts[worst]:
                ts[worst] = sc
                ta[worst] = arr

    if len(ts) == 0:
        a = np.ones(n, dtype=np.float64) / n
        return a, _fast_score(a)

    order = list(range(len(ts)))
    for i in range(1, len(order)):
        ki = order[i]
        kval = ts[ki]
        j = i - 1
        while j >= 0 and ts[order[j]] > kval:
            order[j + 1] = order[j]
            j -= 1
        order[j + 1] = ki

    local_score = ts[order[0]]
    local_a = ta[order[0]].copy()

    if len(order) >= 2 and time_budget > 120:
        quick_end = t0 + min(40.0, 0.20 * float(time_budget))
        if quick_end < t_end - 30:
            for jj in range(min(2, len(order))):
                now = time.monotonic()
                remaining = quick_end - now
                if jj == 0:
                    alloc = min(20.0, remaining)
                else:
                    alloc = min(20.0, remaining)
                if alloc > 3:
                    b, bsc = _optimize_mirror(
                        ta[order[jj]], alloc, seed + 5 + 7 * jj,
                        start_temp=1.2, end_temp=0.2,
                        start_eta=0.08, max_eta=0.4, n_eta=2
                    )
                    if np.isfinite(bsc) and bsc < local_score:
                        local_score = bsc
                        local_a = b.copy()

    alloc = t_end - time.monotonic() - 2.0
    if alloc > 5:
        if time_budget > 300:
            b, bsc = _optimize_mirror(
                local_a, alloc, seed + 100,
                start_temp=2.0, end_temp=0.06,
                start_eta=0.10, max_eta=0.70, n_eta=3
            )
        elif time_budget > 100:
            b, bsc = _optimize_mirror(
                local_a, alloc, seed + 100,
                start_temp=1.2, end_temp=0.07,
                start_eta=0.09, max_eta=0.50, n_eta=3
            )
        else:
            b, bsc = _optimize_mirror(
                local_a, alloc, seed + 100,
                start_temp=0.6, end_temp=0.05,
                start_eta=0.08, max_eta=0.40, n_eta=2
            )
        if np.isfinite(bsc) and bsc < local_score:
            local_score = bsc
            local_a = b.copy()

    return local_a, local_score


def propose_candidate(budget_s=1800.0):
    """
    Returns the best coefficient sequence found.
    The sequence is non-negative and normalized; scale does not affect the score.
    """
    start_global = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1800.0

    if budget <= 90:
        working_budget = max(5.0, 0.75 * budget)
    else:
        working_budget = min(budget - 60.0, 1740.0)
    if working_budget < 10:
        working_budget = max(1.0, budget * 0.5)

    end_global = start_global + float(working_budget)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
        p = np.maximum(p, 0.0)
        p = np.minimum(p, 1000.0)
        if p.size > 0 and float(p.sum()) > 1e-12 and np.all(np.isfinite(p)):
            parent = _normalize(p)
    except Exception:
        parent = None

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = _fast_score(parent)
    else:
        n_fb = 4000
        x = np.linspace(0.0, 1.0, n_fb)
        best_seq = _normalize(np.sin(np.pi * x) ** 2)
        best_score = _fast_score(best_seq)

    print(f"Initial best: J={best_score:.6f}, n={best_seq.size}")

    n0 = int(best_seq.size)
    if n0 < 2000:
        n0 = 2000
    if n0 > 12000:
        n0 = 12000

    lengths = [n0]
    for L in (int(round(n0 * 1.45)),
              int(round(n0 * 0.75)),
              int(round(n0 * 1.20))):
        if L < 2000:
            L = 2000
        if L > 12000:
            L = 12000
        dup = False
        for e in lengths:
            if abs(e - L) <= max(100, int(0.03 * max(e, L))):
                dup = True
                break
        if not dup:
            lengths.append(L)

    length_caps = (750.0, 350.0, 200.0)
    for idx, L in enumerate(lengths):
        now = time.monotonic()
        remaining = end_global - now
        if remaining < 60:
            break

        if idx == 0:
            alloc = min(length_caps[0], remaining - 300.0)
        elif idx == 1:
            alloc = min(length_caps[1], remaining - 200.0)
        else:
            alloc = min(length_caps[2] if idx < len(length_caps) else 150.0, remaining - 120.0)

        if alloc < 30:
            break

        print(f"Working length L={L}, allocation={alloc:.1f}s")
        b, bsc = _localize_length(L, parent, alloc, seed=1000 + idx)
        if np.isfinite(bsc):
            best_score, best_seq = _maybe_update_best(b, best_score, best_seq)
        print(f"  length {L} done, global best J={best_score:.6f}, n={best_seq.size}")

    for cand in _alternative_candidates(best_seq, 18):
        best_score, best_seq = _maybe_update_best(cand, best_score, best_seq)
    print(f"Alternative variants best J={best_score:.6f}, n={best_seq.size}")

    now = time.monotonic()
    rem = end_global - now
    if rem > 30:
        t_rand = min(rem * 0.40, 90.0)
        if t_rand > 5:
            b, bsc = _random_search(best_seq, t_rand, seed=314)
            best_score, best_seq = _maybe_update_best(b, best_score, best_seq)
            for cand in _alternative_candidates(best_seq, 12):
                best_score, best_seq = _maybe_update_best(cand, best_score, best_seq)

    now = time.monotonic()
    rem = end_global - now
    if rem > 12:
        t_fin = min(rem * 0.75, 90.0)
        if t_fin > 5:
            b, bsc = _optimize_mirror(
                best_seq, t_fin, seed=999,
                start_temp=0.12, end_temp=0.02,
                start_eta=0.05, max_eta=0.25, n_eta=3
            )
            best_score, best_seq = _maybe_update_best(b, best_score, best_seq)
            for cand in _alternative_candidates(best_seq, 12):
                best_score, best_seq = _maybe_update_best(cand, best_score, best_seq)

    # Final exact validation and best-of candidates.
    candidates = [best_seq]
    if parent is not None:
        candidates.append(parent)

    best_final_list = [float(x) for x in best_seq]
    best_exact = float('inf')

    if time.monotonic() - start_global < max(1.0, budget - 5.0):
        for cand in candidates:
            try:
                lst = [float(x) for x in _normalize(cand)]
                es = float(evaluate_sequence(lst))
                if np.isfinite(es) and es < best_exact:
                    best_exact = es
                    best_final_list = lst
            except Exception:
                pass

    if not np.isfinite(best_exact):
        best_final_list = [float(x) for x in best_seq]

    # Hard sanity fallback.
    try:
        arr = np.asarray(best_final_list, dtype=np.float64)
        if arr.size == 0 or float(arr.sum()) < 0.01:
            x = np.linspace(0.0, 1.0, 4000)
            best_final_list = [float(v) for v in (_normalize(np.sin(np.pi * x) ** 2))]
    except Exception:
        x = np.linspace(0.0, 1.0, 4000)
        best_final_list = [float(v) for v in (_normalize(np.sin(np.pi * x) ** 2))]

    return best_final_list
```
