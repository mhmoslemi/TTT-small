def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

```python
def best_line_search(a, b, ac_a, ac_cross, ac_b, n_max_trials=30):
    """Find t in [0,1] that minimizes max_k ((1-t)^2*ac_a[k] + 2t(1-t)*ac_cross[k] + t^2*ac_b[k])"""
    # ac_a = conv(a,a), ac_cross = cross(a,b), ac_b = conv(b,b)
    
    def obj_at_t(t):
        ac_t = (1-t)**2 * ac_a + 2*t*(1-t) * ac_cross + t**2 * ac_b
        return ac_t.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(n_max_trials):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)
    
    t_best = (lo + hi) / 2
    return t_best
```
