def fast_autocorr(a):

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pocs_optimize(a0, time_limit, verbose=False, n_iters_max=500000, rng_seed=0):
    """POCS: alternate between flat-spectrum projection and non-negativity projection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Initialize with flat spectrum and random phases
    fft_size = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    # Make it conjugate symmetric for real output
    A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    # Ensure Hermitian symmetry
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size // 2 + 1] = A_spec
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A_spec[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = a0.copy()
    else:
        a = a / s

    ac = fast_autocorr(a)
    M = ac.max()
    if M < best_m:
        best_m = M
        best_a = a.copy()

    # POCS iterations
    for it in range(n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break

        # Frequency domain: enforce flat magnitude
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        # Smooth toward flat: blend current magnitude with mean
        target_mag = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        alpha_blend = 0.1 + 0.5 * (it / min(it + 1, 100))  # ramp up blending
        new_mag = (1 - alpha_blend) * mag + alpha_blend * target_mag
        # Apply
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Time domain: enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    POCS it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def pocs_with_phases(a0, time_limit, verbose=False, n_trials=5, rng_seed=0):
    """POCS with multiple random phase initializations."""
    n = len(a0)
    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 2).bit_length()

        t_trial = min((time_limit - (time.monotonic() - t0)) / (n_trials - trial), 30.0)

        rng = np.random.default_rng(rng_seed + trial)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        A_spec = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
        A_full = np.zeros(fft_size, dtype=np.complex128)
        A_full[:fft_size // 2 + 1] = A_spec
        for i in range(1, fft_size // 2):
            A_full[fft_size - i] = np.conj(A_spec[i])
        a_init = np.fft.irfft(A_full, fft_size)[:n]
        a_init = np.maximum(a_init, 0.0)
        s = a_init.sum()
        if s < 1e-15:
            continue
        a_init = a_init / s

        a_out, m_out = pocs_optimize(a_init, t_trial, verbose=False, rng_seed=rng_seed + trial)
        if m_out < best_m:

            if verbose:

    return best_a, best_m


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=6000,
                 adaptive_beta=True, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if adaptive_beta and stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)
            cur_top_k = min(cur_top_k + 5, len(ac))

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 1.0, 2.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K_init=200, top_K_max=None, use_all=False):

        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(K, 2 * n - 1)
        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))

            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = a[ki - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K + 50, top_K_max)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        base_obj = M_curr
        best_alpha = 2.0 / (iter_count + 2.0)

        for alpha in [best_alpha, min(1.0, best_alpha * 1.5), min(1.0, best_alpha * 2.0), 0.5, 0.3, 0.2]:
            a_trial = (1.0 - alpha) * a + alpha * g_dir

            if s_t < 1e-15:

            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_new = fast_autocorr(a).max()

            best_a = a.copy()

        if iter_count % 10 == 0:

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}, alpha={best_alpha:.4f}")

    return best_a, best_m


def smooth_gradient_optimize(a0, time_limit, verbose=False, tau_init=20.0, tau_final=200.0,

    max_stall = 500
    total_iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / min(time_limit, 600.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        ac = fast_autocorr(a)
        ac_max = ac.max()
        ac_shifted = tau * (ac - ac_max)
        exp_ac = np.exp(ac_shifted)
        p = exp_ac / exp_ac.sum()

        p_cum = np.concatenate([[0.0], np.cumsum(p)])
        k_lo = np.arange(n)
        k_hi = np.minimum(k_lo + n, 2 * n - 1)
        S = p_cum[k_hi] - p_cum[k_lo]
        cross = sp_correlate(p, a, mode='valid')
        g = a * S + cross
        g_tangent = g - g.mean()

        a_trial = a - eta * g_tangent
        a_new = project_simplex(a_trial)

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            if stall > 100:
                eta *= 0.7
            if stall > 200:
                pert = rng.normal(0, 0.005, n) * a
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_autocorr(a_pert).max()
                if m_pert < best_m:
                    a = a_pert
                    best_m = m_pert
                    best_a = a.copy()
                    stall = 0
                    eta = eta_init * 0.5
                else:
                    a = a_new
            else:
                a = a_new

        total_iters += 1

        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100
    n_top = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved_this_round = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            top_cands = candidates[np.argsort(cand_sens)[:3]]

            found = False
            for dst in top_cands:
                if dst == src or found:
                    continue
                for frac in [0.01, 0.005, 0.02, 0.03, 0.05, 0.1, 0.15]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved_this_round = True
                        found = True
                        break

            if not found and stall > 30:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        if not improved_this_round:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_gradient_hybrid(a0, time_limit, verbose=False, rng_seed=50):
    """Hybrid: alternate between spectral projection and gradient steps."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0
    max_stall = 200
    it = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1

        if it % 5 == 0:
            # Spectral step: push toward flat spectrum
            A = np.fft.rfft(a, fft_size)
            mag = np.abs(A)
            mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            # Gentle blend toward flat
            new_mag = 0.7 * mag + 0.3 * mag_mean
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_spec = np.fft.irfft(A_new, fft_size)[:n]
            a_spec = np.maximum(a_spec, 0.0)
            s = a_spec.sum()
            if s > 1e-15:
                a_spec = a_spec / s
            m_spec = fast_autocorr(a_spec).max()
            if m_spec < best_m:
                best_m = m_spec
                best_a = a_spec.copy()
                a = a_spec
                stall = 0
            else:
                stall += 1
        else:
            # Gradient step
            ac = fast_autocorr(a)
            top_idx = np.argsort(ac)[::-1][:30]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** 4.0
            w = w / w.sum()
            g = np.zeros(n)
            for wi, k in zip(w, top_idx):
                ki = int(k)
                i_start = max(0, ki - n + 1)
                i_end = min(n, ki + 1)
                if i_start >= i_end:
                    continue
                idx = np.arange(i_start, i_end)
                j_idx = ki - idx
                g[idx] += 2.0 * wi * a[j_idx]

            eta = 0.5 / np.sqrt(it + 1)
            log_a = np.log(np.maximum(a, 1e-15)) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()

            m_new = fast_autocorr(a_new).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
            a = a_new

        if stall > 100:
            pert = rng.normal(0, 0.02, n) * best_a
            a_p = np.maximum(best_a + pert, 1e-10)
            a = a_p / a_p.sum()
            stall = 50

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpecGrad it={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = a ** 2
    return norm_sum1(a)


def make_cosine_mod(n, alpha=0.5):
    """Cosine modulation: a[i] = 1 + alpha*cos(2*pi*i/n)"""
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with POCS + AMD + LP-FW + hybrid spectral-gradient."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: POCS Spectral Initialization ===
    print(f"\n=== Phase 1: POCS Spectral Init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.12, 180.0)

    if t_phase1 > 30:
        a_pocs, _ = pocs_with_phases(best_seq, t_phase1, verbose=True, n_trials=4, rng_seed=7)
        try_update(a_pocs, "pocs")

    # === Phase 2: Multi-init AMD ===
    print(f"\n=== Phase 2: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = min(remaining * 0.22, 330.0)

    if t_phase2 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3.0")
        inits.append(make_bimodal(n0, width=0.2))
        labels.append("bimodal")
        inits.append(make_cosine(n0))
        labels.append("cosine")
        inits.append(make_cosine_mod(n0, alpha=0.3))
        labels.append("cosmod")
        inits.append(spectral_init(n0, n_modes=30, seed=42))
        labels.append("spectral")

        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_u_shape(n0, 2.5)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Add POCS result as init
        if best_score < float('inf'):
            inits.append(norm_sum1(best_seq))
            labels.append("best_sofar")

        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t_phase2 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                     restart_every=4000, adaptive_beta=True,
                                     verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 3: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(300, 2 * n_best - 1)
        top_K_max = min(2500, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all)
        try_update(a_out, "lp_main")

    # === Phase 4: Spectral-Gradient Hybrid ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Spectral-Gradient Hybrid (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_gradient_hybrid(a_in, t4, verbose=True, rng_seed=50)
        try_update(a_out, "specgrad")

    # === Phase 5: Smoothed gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Smoothed gradient (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t5, verbose=True,
                                             tau_init=15.0, tau_final=150.0,
                                             eta_init=0.8, rng_seed=100)
        try_update(a_out, "smooth1")

    # === Phase 6: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP-FW r2 (n={n_best}) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True,
                                    top_K_init=top_K_init, top_K_max=top_K_max,
                                    use_all=use_all)
        try_update(a_out, "lp_r2")

    # === Phase 7: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coordinate (n={n_best}) ===")
    t7 = min(remaining * 0.08, 120.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t7, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500, 2000, -2000]:
        new_n = n_best + delta
        if 800 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.3, 150.0)
        if t8a > 20:
            a_out, _ = amd_optimize(a_long, t8a, top_k=60, beta=6.0, c0=4.0,
                                     restart_every=5000, adaptive_beta=True,
                                     verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5000:
            t8b = min(remaining * 0.4, 120.0)
            if t8b > 20:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                use_all = (new_n <= 1500)
                top_K = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False,
                                           top_K_init=min(300, 2 * new_n - 1),
                                           top_K_max=top_K, use_all=use_all)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final optimization on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final optimization (n={n_best}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.3, 120.0)
        if t9 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t9, top_k=80, beta=8.0, c0=6.0,
                                     restart_every=3000, adaptive_beta=True,
                                     verbose=False, rng_seed=999)
            try_update(a_out, "amd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and n_best <= 5000:
            t9b = min(remaining * 0.4, 150.0)
            if t9b > 20:
                a_in = norm_sum1(best_seq)
                use_all = (n_best <= 1500)
                top_K_init = min(600, 2 * n_best - 1)
                top_K_max = min(6000, 2 * n_best - 1)
                a_out, _ = lp_frank_wolfe(a_in, t9b, solver, verbose=False,
                                           top_K_init=top_K_init, top_K_max=top_K_max,
                                           use_all=use_all)
                try_update(a_out, "lp_final")

    # === Phase 10: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 10: Final polish (n={n_best}) ===")
        t10 = min(remaining * 0.7, 80.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t10, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
