import numpy as np
import time
from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgradient(a, ac, top_k=30, temperature=1.0):
    """Compute weighted subgradient using top-k peaks with softmax tail."""
    n = len(a)
    len_ac = len(ac)
    k_max = min(top_k, len_ac)
    
    # Get top k peaks
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Softmax weights over top peaks
    if temperature < 0.01:
        w = np.zeros(k_max)
        w[0] = 1.0
    else:
        log_w = (top_vals - top_vals[0]) / temperature
        exp_w = np.exp(log_w)
        w = exp_w / exp_w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for ii in range(k_max):
        ki = int(top_idx[ii])
        wi = w[ii]
        if wi < 1e-12:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        # d conv[k]/d a[j] = 2*a[k-j] (simplified, valid for most cases)
        grad[idx] += 2.0 * wi * a[pair_idx]
    
    # Subtract mean to make it sum-zero (for normalized updates)
    grad -= grad.sum() / n
    return grad


def nesterov_mirror_descent(a0, time_limit, top_k=40, temp0=2.0, c0=4.0,
                            restart_every=3000, rng_seed=42):
    """Nesterov-accelerated mirror descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_temp = temp0
    restart_count = 0
    best_iter = 0

    for t in range(1, 50000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac_y = fast_autocorr(y)
        g = compute_subgradient(y, ac_y, top_k=top_k, temperature=cur_temp)

        eta = gamma / (np.sqrt(t + 1.0))
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Evaluate
        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            stall_count = 0
            best_iter = t
        else:
            stall_count += 1

        # Adaptive temperature: decrease when stalling
        cur_temp = max(0.5, temp0 * np.exp(-0.0003 * stall_count))

        # Periodic restart
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.12, 0.02 + 0.008 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_temp = max(0.5, temp0 - 0.1 * restart_count)
            gamma = c0 * (1.0 + 0.03 * min(restart_count, 8))

    return best_a, best_m


def active_set_lp_refine(a0, time_limit, n_active=30, blend=0.3, rng_seed=42):
    """LP-based active set refinement using linearized autocorrelation constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    for it in range(2000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        # Identify active set: peaks within 5% of max
        threshold = M * 0.95
        active_k = np.where(ac >= threshold)[0]
        active_k = active_k[np.argsort(ac[active_k])[::-1]][:n_active]

        if len(active_k) == 0:
            active_k = np.array([np.argmax(ac)])

        # Build LP: maximize sum(b) subject to linearized constraints
        # Constraint for peak k: sum_j a[k-j] * b[j] <= c_k
        # where c_k = (M + conv[k](a)) / 2
        m_cons = len(active_k)
        # A_ub: m_cons x n
        A_ub = np.zeros((m_cons, n))
        b_ub = np.zeros(m_cons)

        for ii, ki in enumerate(active_k):
            ki = int(ki)
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            idx = np.arange(j_lo, j_hi + 1)
            coeffs = a[ki - idx]  # a[k-j] for j in idx
            A_ub[ii, idx] = coeffs
            c_k = (M + ac[ki]) / 2.0
            b_ub[ii] = c_k

        # Equality: sum(b) = 1
        A_eq = np.ones((1, n))
        b_eq = np.array([1.0])

        # Objective: minimize -sum(b) = maximize sum(b)
        c_obj = -np.ones(n)

        # Bounds: b >= 0
        bounds = [(0, None)] * n

        # Solve LP
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'time_limit': max(1.0, time_limit - elapsed - 5)})
            if res.success and res.x is not None:
                b_lp = np.maximum(res.x, 0.0)
                s_lp = b_lp.sum()
                if s_lp < 1e-10:
                    b_lp = a.copy()
                else:
                    b_lp = b_lp / s_lp
                # Blend
                a_new = (1.0 - blend) * a + blend * b_lp
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                M_new = fast_autocorr(a_new).max()
                if M_new < best_m - 1e-14:
                    best_m = M_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
                    # Try smaller blend
                    a_new2 = (1.0 - blend * 0.3) * a + blend * 0.3 * b_lp
                    a_new2 = np.maximum(a_new2, 0.0)
                    a_new2 = a_new2 / a_new2.sum()
                    M_new2 = fast_autocorr(a_new2).max()
                    if M_new2 < best_m - 1e-14:
                        best_m = M_new2
                        best_a = a_new2.copy()
                        a = a_new2
                        stall = 0
                    else:
                        a = a_new2  # Still move towards LP solution
            else:
                stall += 1
        except Exception:
            stall += 1

        if stall > 20:
            # Perturb and reset
            pert = rng.normal(0, 0.01, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0

    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, rng_seed=42):
    """Coordinate descent: optimize each coordinate via 1D search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    def try_update_coord(a, j, new_val):
        """Try setting a[j] to new_val and renormalizing."""
        a_new = a.copy()
        a_new[j] = max(0.0, new_val)
        s = a_new.sum()
        if s < 1e-15:
            return a, best_a, best_m
        a_new = a_new / s
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            return a_new, a_new.copy(), M_new
        return a, best_a, best_m

    passes = 0
    while time.monotonic() - t0 < time_limit and passes < 200:
        passes += 1
        improved = False
        # Cycle through coordinates (randomized order each pass)
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            # Try a few candidate values for a[j]
            cur_val = a[j]
            if cur_val < 1e-15:
                # Try small positive values
                candidates = [cur_val * 2, cur_val + 1e-4, 1e-4, 5e-4]
            else:
                candidates = [cur_val * 0.5, cur_val * 0.8, cur_val * 1.2,
                              cur_val * 1.5, cur_val * 0.3, cur_val + 0.001,
                              max(0, cur_val - 0.001)]

            for cv in candidates:
                cv = max(0.0, cv)
                a_trial = a.copy()
                a_trial[j] = cv
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and passes > 10:
            # Try random pair moves
            for _ in range(n // 10):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = max(0, a_trial[i] - amt)
                a_trial[j] += amt
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True

        if not improved and passes > 30:
            break

    return best_a, best_m


def make_power_law_ramp(n, power=0.5):
    """Increasing power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_mild_ramp(n, center_frac=0.3):
    a = 1.0 + center_frac * np.linspace(0, 1, n)
    return norm_sum1(a)


def make_quadratic_ramp(n, alpha=2.0):
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * x ** 2
    return norm_sum1(a)


def make_sigmoid_ramp(n, steepness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * (1.0 + np.tanh(steepness * (x - 0.5)))
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
