"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-factored spectral initialization with diverse parametric families
2. Benders decomposition with full FFT-based violation detection
3. FACA (Frank-Wolfe with Acceleration) with adaptive step sizes
4. Non-convex gradient polish on active constraints
5. Multi-length exploration with interpolation-based warm-starting

Key improvement: Full violation check every iteration (not just top-K),
adaptive cut management, and a dedicated non-convex QP polish phase.

import numpy as np
import time
import cvxpy as cp

_SOLVER = None
_FFT_CACHE = {}


def get_fft_size(n):
    size = 2 * n - 1
    for key in _FFT_CACHE:
        if key >= size:
            return key
    fft_size = 1 << (size - 1).bit_length()
    _FFT_CACHE[fft_size] = True
    return fft_size


def get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _SOLVER = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _SOLVER


def fast_convolve(a, b):
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = get_fft_size(max(n1, n2))
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    return fast_convolve(a, a)


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    try:

    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))


def spectral_init(n, iters=80, seed=42):
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    noise_scale = 0.3
    a = a + noise_scale * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    fft_size = get_fft_size(n)
    for i in range(iters):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        P_dev = P - P_mean
        beta = 8.0 + i * 0.4
        w = np.exp(np.maximum(beta * P_dev / P_mean, -50.0))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(w) / len(w)
        else:
            w = w / w_sum
        grad = np.fft.irfft(2.0 * A * w, fft_size)[:n]
        lr = 0.8 / (1.0 + 0.3 * np.sqrt(i))
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    return a


def parametric_family(n, family_id):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    if family_id == 0:
        p = 0.7
        return norm_sum1((i_arr + 1.0) ** p)
    elif family_id == 1:
        p = 1.5
        return norm_sum1((n - i_arr) ** p)
    elif family_id == 2:
        c, sig = 0.5, 0.25
        return norm_sum1(np.exp(-((x - c) ** 2) / (2.0 * sig * sig)))
    elif family_id == 3:
        c1, c2, sig = 0.25, 0.75, 0.15
        return norm_sum1(np.exp(-((x - c1) ** 2) / (2.0 * sig ** 2)) +
                         np.exp(-((x - c2) ** 2) / (2.0 * sig ** 2)))
    elif family_id == 4:
        return norm_sum1(np.maximum(np.sin(np.pi * x) ** 2, 0.01))
    elif family_id == 5:
        alpha, beta_v = 1.5, 1.5
        return norm_sum1(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                         np.maximum(n - i_arr, 0.5) ** beta_v)
    elif family_id == 6:
        return norm_sum1(np.exp(-2.0 * x))
    elif family_id == 7:
        freq = 2.0
        return norm_sum1(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.001))
    else:
        return norm_sum1(np.ones(n))


def build_constraint_matrix(y, cut_list, n):
    K = len(cut_list)
    C = np.zeros((K, n))
    for jj, k in enumerate(cut_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            C[jj, idx] = y[k_int - idx]
    return C


def solve_cut_lp(y, cut_list, n, solver):
    K = len(cut_list)
    if K == 0:
        return y.copy(), 0.0
    C = build_constraint_matrix(y, cut_list, n)
    C = np.maximum(C, 0.0)
    g = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g) == 1.0, C @ g <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    g_val = np.asarray(g.value, dtype=np.float64)
    g_val = np.maximum(g_val, 0.0)
    gs = g_val.sum()
    if gs < 1e-15:
        return None, None
    g_val = g_val / gs
    M_val = max(0.0, float(M_var.value))
    return g_val, M_val


def benders_optimize(a_init, time_limit, solver, max_cuts=4000, verbose=True):
    n = len(a_init)
    y = norm_sum1(a_init)
    best_a = y.copy()
    best_M = fast_autocorr(y).max()
    t0 = time.monotonic()
    fft_size = get_fft_size(n)

    # Initialize cut set with top lags
    ac_y = fast_autocorr(y)
    init_K = min(500, 2 * n - 1)
    cut_indices = np.argsort(ac_y)[::-1][:init_K]
    cut_set = set(int(k) for k in cut_indices)

    # FACA state
    x = y.copy()
    x_prev = y.copy()
    iter_count = 0
    stall_count = 0
    failed_solves = 0

    for t in range(1, 1000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Compute current M
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()

        if M_curr < best_M - 1e-12:
            best_M = M_curr
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Manage cut set
        cut_list = sorted(cut_set)
        if len(cut_list) > max_cuts:
            margins = np.array([M_curr - ac_y[k] for k in cut_list])
            top_keep = np.argsort(margins)[:max_cuts]
            cut_set = set(int(cut_list[i]) for i in top_keep)
            cut_list = sorted(cut_set)

        if len(cut_list) == 0:
            break

        # Solve LP
        g_val, M_lp = solve_cut_lp(y, cut_list, n, solver)
        if g_val is None:
            failed_solves += 1
            if failed_solves > 8:
                # Reduce cuts and retry
                cut_list = sorted(cut_set)[:min(300, len(cut_set))]
                cut_set = set(cut_list)
                failed_solves = 0
            continue
        failed_solves = 0

        # Full violation check: conv(y, g_val) for all lags
        conv_yg = fast_convolve(y, g_val)
        violations = np.where(conv_yg > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_curr
            # Add all violations (up to a cap)
            n_add = min(len(violations), 200)
            top_v = violations[np.argsort(viol_amt)[::-1][:n_add]]
            for k in top_v:
                cut_set.add(int(k))

        # Also check: conv(g_val, g_val) to find self-violations
        ac_g = fast_autocorr(g_val)
        violations2 = np.where(ac_g > M_curr + 1e-9)[0]
        if len(violations2) > 0:
            viol_amt2 = ac_g[violations2] - M_curr
            n_add2 = min(len(violations2), 100)
            top_v2 = violations2[np.argsort(viol_amt2)[::-1][:n_add2]]
            for k in top_v2:
                cut_set.add(int(k))

        # FACA update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g_val
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        # Check x as well
        ac_x = fast_autocorr(x)
        if ac_x.max() < best_M - 1e-12:
            best_M = ac_x.max()
            best_a = x.copy()
            stall_count = 0

        # Restart if stalled too long
        if stall_count > 50:
            stall_count = 0
            # Small perturbation
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.02, n) * best_a
            y = norm_sum1(best_a + pert)
            x = y.copy()
            cut_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:400])

        if verbose and (t % 20 == 0 or t <= 5):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Benders iter={t} ({elapsed:.0f}s): J={J:.6f}, |cuts|={len(cut_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    if verbose:
        print(f"    Benders done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_M


def gradient_polish(a_init, time_limit, verbose=True):
    """Non-convex gradient polish focusing on the active (max) lags."""
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = get_fft_size(n)
    rng = np.random.default_rng(123)

    lr = 0.5
    stall = 0
    top_k_count = 20

    for t in range(1, 100000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_M - 1e-12:
            best_M = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            # Restart with perturbation
            a = norm_sum1(best_a + rng.normal(0, 0.01, n) * best_a)
            stall = 0
            lr *= 0.7

        # Soft gradient over top lags
        k_max = min(top_k_count, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-20)
        w = top_vals ** 4.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * wi * a[k_int - idx]

        # Mirror descent step
        eta = lr / (1.0 + 0.1 * np.sqrt(t))
        log_a = np.log(np.maximum(a, 1e-15)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        a = a_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    GradPolish t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_M


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def active_set_polish(a_init, time_limit, verbose=True):
    """Solve the reduced problem using only active constraints."""
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Identify active lags
    ac = fast_autocorr(a)
    M_thresh = best_M * 0.95
    active = np.where(ac >= M_thresh)[0]
    active_list = sorted(int(k) for k in active)

    if len(active_list) == 0:
        return best_a, best_M

    # Solve LP with only active constraints (tighter)
    solver = get_solver()
    max_iters = 50
    for it in range(max_iters):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        g_val, M_lp = solve_cut_lp(a, active_list, n, solver)
        if g_val is None:
            break

        # Check for new violations at the new solution
        conv_ag = fast_convolve(a, g_val)
        new_violations = np.where(conv_ag > best_M + 1e-8)[0]
        if len(new_violations) > 0:
            for k in new_violations[:50]:
                active_list.append(int(k))

        # Update
        alpha = 0.5
        a_new = (1.0 - alpha) * a + alpha * g_val
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        a = a_new

        ac_new = fast_autocorr(a)
        if ac_new.max() < best_M - 1e-12:
            best_M = ac_new.max()
            best_a = a.copy()

        # Re-identify active set
        active = np.where(ac_new >= best_M * 0.9)[0]
        active_list = sorted(set(int(k) for k in active))

    return best_a, best_M


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score) and p_score > 0:
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        best_seq = spectral_init(n_base, iters=60, seed=42)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc > 0 and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Multi-start initialization ===
    print(f"\n=== Phase 1: Multi-start (n={len(best_seq)}) ===")
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)

    if t1 > 10:
        for fam in range(8):
            a_fam = parametric_family(n0, fam)
            a_fam = spectral_init(n0, iters=30, seed=100 + fam)
            # Blend with parametric
            for w in [0.5, 0.7]:
                blend = w * a_fam + (1.0 - w) * parametric_family(n0, fam)
                check_and_update(blend, f"init_fam{fam}_w{w}")
            check_and_update(a_fam, f"spectral_fam{fam}")

        # Spectral init with parent blend
        if parent_loaded:
            spec = spectral_init(n0, iters=50, seed=777)
            for w in [0.3, 0.5, 0.7]:
                blend = w * best_seq + (1.0 - w) * spec
                check_and_update(blend, f"blend_parent_spec_{w}")

    # === Phase 2: Main Benders optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Benders (n={n_best}) ===")
    t2 = min(remaining * 0.40, 660.0)

    if t2 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, M_out = benders_optimize(a_in, t2, solver, max_cuts=4000, verbose=True)
        check_and_update(a_out, f"benders_n{n_best}")

    # === Phase 3: Active set polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Active set polish (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)

    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = active_set_polish(a_in, t3, verbose=True)
        check_and_update(a_out, f"activ

e_n{n_best}")

    # === Phase 4: Gradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Gradient polish (n={n_best}) ===")
    t4 = min(remaining * 0.08, 120.0)

    if t4 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_polish(a_in, t4, verbose=False)
        check_and_update(a_out, f"grad_n{n_best}")

    # === Phase 5: Second Benders pass (with improved init) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Benders pass 2 (n={n_best}) ===")
    t5 = min(remaining * 0.20, 330.0)

    if t5 > 30 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = benders_optimize(a_in, t5, solver, max_cuts=5000, verbose=True)
        check_and_update(a_out, f"benders2_n{n_best}")

    # === Phase 6: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-n exploration ===")

    length_candidates = []
    for delta in [500, -500, 1000, -1000, 1500, -1500, 2000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 10000:
            length_candidates.append(new_n)
    for rn in [3000, 4000, 5000, 6000, 7000, 8000]:
        if abs(rn - n_best) > 300 and 2000 <= rn <= 10000:
            length_candidates.append(rn)
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick spectral init for new length
        a_spec = spectral_init(new_n, iters=40, seed=new_n % 1000)
        # Blend
        a_start = 0.5 * a_long + 0.5 * a_spec
        check_and_update(a_start, f"blend_n{new_n}")

        # Short Benders
        t6a = min(remaining * 0.5, 200.0)
        if t6a > 20 and new_n <= 8000:
            a_out, _ = benders_optimize(a_start, t6a, solver, max_cuts=3000, verbose=False)
            check_and_update(a_out, f"benders_n{new_n}")

        # Gradient polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.3, 80.0)
            if t6b > 10:
                a_out, _ = gradient_polish(a_start, t6b, verbose=False)
                check_and_update(a_out, f"grad_n{new_n}")

    # === Phase 7: Final polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 60:
        t7 = min(remaining * 0.4, 200.0)
        if t7 > 20 and n_best <= 8000:
            a_in = norm_sum1(best_seq)
            a_out, _ = benders_optimize(a_in, t7, solver, max_cuts=5000, verbose=True)
            check_and_update(a_out, f"final_benders_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = gradient_polish(a_in, t8, verbose=False)
            check_and_update(a_out, f"final_grad_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.6, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = active_set_polish(a_in, t9, verbose=True)
            check_and_update(a_out, f"final_active_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs > 0:
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = np.sin(np.pi * np.linspace(0, 1, n_fb)) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
