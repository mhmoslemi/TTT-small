def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_objective(a, n):
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def try_update(best_seq, best_score, a, label, t_global):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def smooth_gradient_fft(a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*(a*a)_k)) w.r.t. a via FFT.
    gradient[m] = 2 * sum_k p[k] * a[k-m] where p = softmax(beta * (a*a))"""
    n = len(a)
    ac = fast_autocorr(a)
    beta_ac = beta * ac
    max_val = beta_ac.max()
    p = np.exp(beta_ac - max_val)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(ac)) / len(ac)
    else:
        p = p / p_sum
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Pf = np.fft.rfft(p, fft_size)
    Arf = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(Pf * Arf, fft_size)
    g = conv[n - 1:2 * n - 1]
    return 2.0 * g, ac


def mirror_descent_smooth(a0, time_limit, beta_start=5.0, beta_end=200.0,

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k = int(constraint_list[jj])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2500,
