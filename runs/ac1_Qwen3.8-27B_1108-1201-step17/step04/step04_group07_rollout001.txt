Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_normalized(a):
    """Compute 2n * max_conv for normalized a (sum=1). Returns (obj, conv)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf'), np.zeros(2*n-1)
    a_n = a / s
    conv = fast_convolve(a_n)
    return 2.0 * n * conv.max(), conv

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    # C is (2n-1) x n
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # C[k, j] = a[j] shifted... 
    # Actually: (a*g)[k] = sum_j a[j] * g[k-j] (with g out of range = 0)
    # So C[k, j] = a[j] if 0 <= k-j < n... no
    # (a*g)[k] = sum_{j=0}^{n-1} a[j] * g[k-j]
    # For this to be a matrix-vector product C@g, we need C[k,j] = a[j] if k-j is in [0,n-1], i.e., j <= k < j+n
    # So C[k,j] = a[j] if j <= k <= j+n-1, i.e., k-j in [0, n-1]
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def solve_lp(a, M, verbosity=0):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns (g, dual)."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        g = res.x
        dual = None
        try:
            dual = res.ineqlin.marginals
        except:
            pass
        return g, dual
    return None, None

def line_search(a, d, obj_cur, n, max_funs=16):
    """Simple line search along direction d from a."""
    best_t = 0.0
    best_obj = obj_cur
    funcs = 0
    # Try a range of steps
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
        if funcs >= max_funs:

        s = a_try.sum()
        if s < 1e-15:

        a_try = a_try / s
        obj_try, _ = obj_normalized(a_try)
        funcs += 1
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    return best_t, best_obj

    a = np.maximum(a0.astype(np.float64), 0.0)

            nd = np.linalg.norm(d)

                    s_new = a_new.sum()
                    if s_new > 1e-15:

                    _, conv = obj_normalized(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0

            # Loosen if stuck

            M = conv.max()
            no_improve = 0
