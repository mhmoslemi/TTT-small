coordinate descent polishing, and multi-length/multi-restart exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_active_set(f, peak_indices):
    """
    Compute gradient of sum_{k in peak_indices} conv(f,f)_k w.r.t. f.
    Uses FFT cross-correlation: grad_j = 2 * sum_{m} f_m * s_{j+m}
    where s is the indicator of peak_indices.
    """
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    s = np.zeros(size)
    s[peak_indices] = 1.0
    
    # Cross-correlation: (f ⊛ s)(j) = sum_m f_m * s_{m+j}
    # In FFT: ifft(FFT(f) * conj(FFT(s)))
    F = np.fft.rfft(f, fft_size)
    S = np.fft.rfft(s, fft_size)
    corr = np.fft.irfft(F * np.conj(S), fft_size)[:n]
    
    return 2.0 * corr

def smoothed_max_objective(f, K, temperature=0.1):
    """Compute log-sum-exp approximation of max, plus the actual max."""
    conv = fast_convolve(f)
    n_conv = len(conv)
    
    # Get top K values
    if K < n_conv:
        top_k = np.argpartition(conv, -K)[-K:]
        top_vals = conv[top_k]
    else:
        top_vals = conv
    
    # Log-sum-exp
    t = top_vals / temperature
    t_max = t.max()
    lse = t_max + np.log(np.sum(np.exp(t - t_max)))
    lse *= temperature
    
    float_max = conv.max()
    return float_max, lse, top_k if K < n_conv else np.arange(n_conv)


def sgd_smoothed_max(f0, time_limit, K_init=100, K_min=20, lr_init=0.05,
                     lr_decay=0.9995, temp_init=0.5, temp_decay=1.002,
                     verbose=False, seed=42):
    """
    Projected gradient descent minimizing smoothed max of auto-convolution.
    Uses log-sum-exp of top-K convolution values as smooth proxy for max.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    K = min(K_init, 2 * n - 1)
    temp = temp_init
    lr = lr_init
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute convolution and select peaks
        conv = fast_convolve(f)
        float_max = conv.max()
        
        # Adaptive K: decrease over time to focus on fewer peaks
        K_current = max(K_min, int(K * (0.999 ** (it % 500))))
        K_current = min(K_current, 2 * n - 1)
        
        # Select top K peaks
        if K_current < len(conv):
            peak_idx = np.argpartition(conv, -K_current)[-K_current:]
        else:
            peak_idx = np.arange(len(conv))
        
        # Compute gradient
        grad = compute_gradient_active_set(f, peak_idx)
        
        # Gradient step with simplex projection
        f_new = f - lr * grad
        f_new = project_simplex(f_new)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f}, K={K_current}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        f = f_new
        
        # Update hyperparameters
        lr *= lr_decay
        temp *= temp_decay
        
        # Restart if stuck
        if no_improve > 200:
            scale = 0.003 + 0.005 * min(no_improve // 500, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0
            lr = lr_init * 0.3
            K = max(K_min, K_init // 2)
        
        # Periodically re-randomize K
        if it % 500 == 0:
            K = max(K_min, rng.integers(K_min, K_init + 1))
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, m_tolerance=0.5):
    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.08, 4.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Refine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 17 + 3)
            scale = 0.003 + 0.004 * min(no_improve // 100, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Fast coordinate descent for polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()
            
            for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Triangular
        x = np.arange(n)
        peak = n // 2
        f = np.maximum(0, 1 - np.abs(x - peak) / (n / 2.0))
        s = f.sum()
        return f / s if s > 1e-15 else np.ones(n) / n
    elif mode == 2:
        # Gaussian
        x = np.arange(n)
        sigma = n / 4.0
        f = np.exp(-0.5 * ((x - n / 2.0) / sigma) ** 2)
        s = f.sum()
        return f / s if s > 1e-15 else np.ones(n) / n
    elif mode == 3:
        # Cosine window
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        f = np.maximum(f, 0)
        s = f.sum()
        return f / s if s > 1e-15 else np.ones(n) / n
    elif mode == 4:
        # Random with smoothing
        f = rng.random(n)
        # Simple moving average
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        f = np.convolve(f, kernel, mode='same')
        f = np.maximum(f, 0)
        s = f.sum()
        return f / s if s > 1e-15 else np.ones(n) / n
    else:
        # Bimodal
        x = np.arange(n)
        f = np.exp(-0.5 * ((x - n * 0.3) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - n * 0.7) / (n * 0.1)) ** 2)
        f = np.maximum(f, 0)
        s = f.sum()
        return f / s if s > 1e-15 else np.ones(n) / n

            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 15)
            a_opt, _ = sgd_smoothed_max(f_pert, time_limit=t_alloc,
                                         K_init=n_best // 8, K_min=5,
                                         lr_init=0.01, lr_decay=0.9999)
            try_seq(a_opt, f"perturb_{i}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
