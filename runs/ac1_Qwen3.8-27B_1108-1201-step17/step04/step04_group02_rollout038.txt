"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using ramp-structured initializations
with projected Nesterov subgradient descent, parameterized family exploration,
and coordinate-descent polishing.

Key insight: ramp-like sequences (mass concentrated toward one end) have
asymptotically lower peak convolution than uniform sequences, achieving
scores near 4/3 for large n vs 2 for uniform.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def nesterov_sg(f0, time_limit, lr=0.005, beta=0.9, top_k=32, temp=0.001):
    """Nesterov accelerated projected subgradient descent."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    y = f.copy()  # Nesterov lookahead
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    t_val = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, top_k=top_k, temp=cur_temp)
        # Gradient step in Nesterov point
        x_new = project_simplex(y - cur_lr * sg)
        # Update Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = (t_new / t_val) * x_new - ((t_new - 1.0) / t_val) * f
        y = project_simplex(y)
        t_val = t_new
        f = x_new
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        # Adaptive parameters
        cur_temp *= 1.002
        cur_lr *= 0.99995
        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            t_val = 1.0
            conv = fast_convolve(f)
            cur_lr = lr * 0.7
            cur_temp = temp * 0.5
            no_improve = 0
    return best_f, best_obj


def coord_descent_polish(f0, time_limit, max_sweeps=50):
    """Coordinate descent: optimize each element in turn."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        # Random coordinate order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            # Save old value
            old_val = f[i]
            if old_val < 1e-12:
                continue
            # Compute the "base" convolution contribution from other elements
            # c[k] = c_other[k] + 2*a[i]*a[k-i] + a[i]^2 * delta(2i=k)
            # We want to find a[i] that minimizes max_k c[k] subject to sum constraint
            # Simplified: try a few candidate values
            # The optimal a[i] balances the peaks it affects
            # Quick approach: try reducing a[i] slightly and see effect
            # For each affected convolution index k, c[k] depends on a[i]
            # c[k] = sum_{j != i} a[j]a[k-j] + a[i]*a[k-i] + a[k-i]*a[i] + a[i]^2*delta
            # Let's just try: new_val = old_val * factor for several factors
            best_v = old_val
            best_c = conv.max()
            for factor in [0.0, 0.5, 0.8, 1.2, 1.5]:
                trial = old_val * factor
                f[i] = trial
                # Renormalize
                s2 = f.sum()
                if s2 < 1e-15:
                    f[i] = old_val
                    continue
                f_trial = f / s2
                conv_trial = fast_convolve(f_trial)
                c_max = conv_trial.max()
                if c_max < best_c:
                    best_c = c_max
                    best_v = trial
            if best_v != old_val:
                f[i] = best_v
                s2 = f.sum()
                if s2 > 1e-15:
                    f = f / s2
                    conv = fast_convolve(f)
                    if conv.max() < best_obj - 1e-14:
                        best_obj = conv.max()
                        best_f = f.copy()
            else:
                f[i] = old_val

    return best_f, best_obj


def make_ramp_inits(n, rng):
    """Generate diverse ramp-like initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Linear ramp increasing
    r1 = x + 1.0
    inits.append(r1 / r1.sum())

    # Linear ramp decreasing
    r2 = n - x
    inits.append(r2 / r2.sum())

    # Power ramps
    for p in [0.5, 1.5, 2.0, 3.0, 4.0]:
        rp = (x + 1.0) ** p
        inits.append(rp / rp.sum())
        rp2 = (n - x) ** p
        inits.append(rp2 / rp2.sum())

    # Exponential ramps
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        e1 = np.exp(alpha * x / n)
        inits.append(e1 / e1.sum())
        e2 = np.exp(-alpha * x / n)
        inits.append(e2 / e2.sum())

    # Skewed Gaussians
    for center in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for width in [0.1, 0.15, 0.2, 0.3]:
            g = np.exp(-0.5 * ((x / n - center) / width) ** 2)
            gs = g.sum()
            if gs > 0:
                inits.append(g / gs)

    # Sine-based
    for p in [1, 2, 3]:
        s1 = np.sin(np.pi * (x + 1) / (n + 1)) ** p
        ss = s1.sum()
        if ss > 0:
            inits.append(s1 / ss)
        s2 = np.sin(np.pi * (x + 1) / (n + 1)) ** p * (x / n + 0.1)
        ss2 = s2.sum()
        if ss2 > 0:
            inits.append(np.maximum(s2, 0) / np.maximum(s2, 0).sum())

    # Random smooth
    for _ in range(3):
        freqs = rng.uniform(0.5, 3.0, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.5, 2.0, size=5)
        sig = np.zeros(n)
        for freq, phase, amp in zip(freqs, phases, amps):
            sig += amp * np.sin(2 * np.pi * freq * x / n + phase)
        sig = np.abs(sig) + 0.1
        inits.append(sig / sig.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: exploits ramp-structured initializations with Nesterov
    subgradient descent and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score~{evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    print("=== Phase 1: Parametric Initialization Search ===")

    # Focus on sizes that work well
    sizes = [256, 320, 384, 448, 512]
    if parent is not None:
        pn = len(parent)
        if pn not in sizes:
            sizes.insert(0, pn)
        else:
            sizes.remove(pn)
            sizes.insert(0, pn)

    # For each size, try several initializations
    for n in sizes:
        if time.monotonic() > deadline - 100:
            break

        inits = make_ramp_inits(n, rng)
        # Limit number of initializations per size
        max_inits = min(len(inits), 12)
        # Select diverse subset
        selected = []
        for i in range(0, len(inits), max(1, len(inits) // max_inits)):
            selected.append(inits[i])
        selected = selected[:max_inits]

        # Time budget per initialization
        remaining = deadline - time.monotonic() - 80
        n_total = max(1, len(selected) * len(sizes))
        t_per = max(10, min(remaining / n_total, 40))

        for idx, f_init in enumerate(selected):
            if time.monotonic() > deadline - 60:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 50)
            if t_alloc < 8:
                break

            # Quick evaluation of the initialization itself
            try_seq(f_init, f"init_n{n}_i{idx}")

            # Run Nesterov SG
            a_opt, obj = nesterov_sg(f_init, time_limit=t_alloc,
                                     lr=0.004, beta=0.9, top_k=32, temp=0.001)
            try_seq(a_opt, f"nest_n{n}_i{idx}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep optimization on best candidates ===
    print("\n=== Phase 2: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple Nesterov runs with different hyperparameters
        configs = [
            {"lr": 0.003, "beta": 0.9, "top_k": 64, "temp": 0.0005},
            {"lr": 0.005, "beta": 0.85, "top_k": 32, "temp": 0.001},
            {"lr": 0.002, "beta": 0.95, "top_k": 100, "temp": 0.0003},
            {"lr": 0.008, "beta": 0.8, "top_k": 16, "temp": 0.002},
        ]

        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(rem // (len(configs) - ci), 60)
            if t_alloc < 10:
                t_alloc = 10

            f_init = best_seq.copy()
            if ci > 0:
                f_init = project_simplex(f_init + rng.standard_normal(n_best) * 0.005)

            a_opt, obj = nesterov_sg(f_init, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_nest_{ci}")

        # Coordinate descent polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, obj = coord_descent_polish(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "coord_poly")

    # === Phase 3: Basin hopping ===
    print("\n=== Phase 3: Basin Hopping ===")

    if best_seq is not None:
        n_best = len(best_seq)
        n_hops = 8
        rem = deadline - time.monotonic() - 30
        t_per_hop = max(10, rem // n_hops) if rem > n_hops * 10 else 10

        for hop in range(n_hops):
            if time.monotonic() > deadline - 25:
                break
            # Perturb best
            noise_scale = 0.01 + 0.01 * hop / n_hops
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            a_opt, obj = nesterov_sg(f_pert, time_limit=t_per_hop,
                                     lr=0.004, beta=0.9, top_k=48, temp=0.001)
            try_seq(a_opt, f"hop_{hop}")

            # Also try structural perturbation (blend with a ramp)
            x = np.arange(n_best, dtype=np.float64)
            ramp = (x + 1.0) ** 2
            ramp = ramp / ramp.sum()
            f_blend = project_simplex(0.7 * best_seq + 0.3 * ramp)
            a_opt2, obj2 = nesterov_sg(f_blend, time_limit=t_per_hop // 2,
                                       lr=0.004, beta=0.9, top_k=48, temp=0.001)
            try_seq(a_opt2, f"hop_blend_{hop}")

    # === Phase 4: Try other sizes from best ===
    print("\n=== Phase 4: Cross-size Transfer ===")

    if best_seq is not None:
        n_best = len(best_seq)
        # Interpolate to different sizes
        for target_n in [int(n_best * 0.75), int(n_best * 1.25)]:
            if time.monotonic() > deadline - 40:
                break
            if target_n < 64 or target_n > 1024:
                continue
            x_new = np.arange(target_n)
            x_old = np.linspace(0, 1, n_best)
            x_new_norm = np.linspace(0, 1, target_n)
            interp = np.interp(x_new_norm, x_old, best_seq)
            interp = project_simplex(interp)
            a_opt, obj = nesterov_sg(interp, time_limit=min(30, deadline - time.monotonic() - 20),
                                     lr=0.004, beta=0.9, top_k=32, temp=0.001)
            try_seq(a_opt, f"xsize_{target_n}")

    # === Phase 5: Final Polish ===
    print("\n=== Phase 5: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine Nesterov
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = nesterov_sg(best_seq, time_limit=min(rem, 40),
                                   lr=0.001, beta=0.95, top_k=n_best // 4, temp=0.0001)
            try_seq(a_opt, "polish_fine")

        # Coordinate descent on polished
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = coord_descent_polish(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "polish_coord")

        # Micro perturbations + quick optimize
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            for m in range(3):
                if time.monotonic() > deadline - 4:
                    break
                f_m = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, _ = nesterov_sg(f_m, time_limit=min(rem, 8),
                                       lr=0.001, beta=0.95, top_k=n_best // 3, temp=0.0001)
                try_seq(a_opt, f"micro_{m}")

    # Fallback
    if best_seq is None:
        # Linear ramp fallback
        n = 384
        x = np.arange(n, dtype=np.float64) + 1.0
        best_seq = x / x.sum()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback score: {best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        n = 384
        x = np.arange(n, dtype=np.float64) + 1.0
        best_seq = x / x.sum()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
