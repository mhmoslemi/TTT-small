"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with
multiple sequence lengths, Adam-based subgradient refinement, and
coordinate descent polishing. Explores sequences up to length 2048.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n):
    c = -np.ones(n)
    # Build constraint matrix A_ub where row k is the k-th convolution constraint
    # (f*g)_k = sum_j f_j * g_{k-j} for valid j
    # Row k: sum over j where 0<=k-j<n, i.e., max(0,k-n+1) <= j <= min(n-1,k)
    # A_ub[k, j] = f_{k-j} if j is in range for constraint k
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For each g_j, it appears in constraints k=j, j+1, ..., j+n-1
        # with coefficient f_{k-j} = f_0, f_1, ..., f_{n-1}
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def adam_subgradient(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999, 
                      top_k=30, temp=0.0001, verbose=False):
    """Adam optimizer with subgradient of max convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # Soft-max weights on peaks
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new)
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new  # Still move, but track best
        
        cur_lr *= 0.9998
        cur_temp *= 1.005
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            cur_lr = max(lr * 0.7, 0.001)
            no_improve = 0
            conv = fast_convolve(f)
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            multipliers = [0.3, 0.5, 0.7, 1.3, 1.7, 2.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def block_coordinate_descent(f0, time_limit, block_size=8):
    """Optimize blocks of coordinates simultaneously using small LPs or grid search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            end = min(start + block_size, n)
            block = f[start:end]
            rest_sum = 1.0 - block.sum()
            
            if rest_sum < 1e-10:
                continue
            
            # Try scaling the block
            for scale_factor in [0.5, 0.7, 1.3, 1.5, 2.0]:
                f_trial = f.copy()
                f_trial[start:end] = block * scale_factor
                # Renormalize
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f_trial.copy()
                    improved = True
                    f = f_trial
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe across

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def adam_subgradient(f0, time_limit, lr=0.004, beta1=0.9, beta2=0.999,
                     top_k=40, temp=0.00008, verbose=False):
    """Adam optimizer with soft-max weighted subgradient of max convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t_iter = 0
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new)
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and t_iter % 100 == 0:
                print(f"    Adam iter {t_iter}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9997
        cur_temp *= 1.004

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            t_iter = 0
            cur_lr = max(lr * 0.6, 0.0005)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak

            multipliers = [0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.2]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 4:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, rng):
    """Initialize using spectral methods: design a smooth spectrum."""
    # We want |F(omega)|^2 to be concentrated near omega=0

    shape = shapes[rng.integers(len(shapes))]
    # Random phase
    phase = rng.standard_normal(n)

    F[0] = np.abs(F[0])  # Ensure DC component is positive

    a = np.maximum(a, 0)
    s = a.sum()

                                     lr=0.004, top_k=60, temp=0.0001)
        try_seq(a_opt, "adam_2048")

        # Second Adam run with different params
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            a_opt2, _ = adam_subgradient(a_opt, time_limit=min(rem * 0.4, 100),
                                          lr=0.001, top_k=80, temp=0.00003)
            try_seq(a_opt2, "adam_2048_fine")

    # === Phase 5: Try spectral initialization at n=1024 ===
    print("\n=== Phase 5: Spectral init n=1024 ===")
    rem = deadline - time.monotonic() - 300
    if rem > 60:
        for attempt in range(2):
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            f_spec = spectral_init(1024, rng)
            print(f"  Spectral init {attempt+1}, t={min(rem*0.5, 100):.0f}s")
            a_opt, _ = adam_subgradient(f_spec, time_limit=min(rem * 0.5, 100),
                                         lr=0.004, top_k=50, temp=0.0001)
            try_seq(a_opt, f"spectral_{attempt}")

    # === Phase 6: Deep LP-FW on best (multiple diverse restarts) ===
    print(f"\n=== Phase 6: Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        if n_best > 1200:
            # Too long for LP, use Adam instead
            rem = deadline - time.monotonic() - 150
            if rem > 30:
                a_opt, _ = adam_subgradient(best_seq, time_limit=min(rem * 0.6, 120),
                                             lr=0.002, top_k=max(40, n_best // 8),
                                             temp=0.00004)
                try_seq(a_opt, "deep_adam_long")
        else:
            for p in range(6):
                rem = deadline - time.monotonic() - 150
                if rem < 20:
                    break
                t_alloc = min(rem / max(1, 6 - p), 130)

                if p == 0:
                    f_init = best_seq.copy()
                elif p == 1:
                    noise = rng.standard_normal(n_best) * 0.005
                    f_init = project_simplex(best_seq + noise)
                elif p == 2:
                    f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                    f_init = project_simplex(f_init)
                elif p == 3:
                    f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                    f_init = project_simplex(f_init)
                elif p == 4:
                    w = rng.dirichlet(np.ones(n_best) * 0.3)
                    f_init = project_simplex(best_seq * w / (w.max() + 1e-10))
                else:
                    # Shift-based perturbation
                    shift = rng.integers(1, max(2, n_best // 20))
                    f_init = np.roll(best_seq, shift)
                    f_init = project_simplex(f_init)

                print(f"  Round {p+1}, t={t_alloc:.0f}s")
                a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
                try_seq(a_opt, f"deep_{p}")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: CD Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_cd = min(rem * 0.6, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 8: Final Adam polish ===
    print(f"\n=== Phase 8: Final Adam (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = adam_subgradient(best_seq, time_limit=min(rem, 80),
                                         lr=0.0008, top_k=max(20, n_best // 6),
                                         temp=0.00002)
            try_seq(a_opt, "adam_final")

    # === Phase 9: Micro-optimizations ===
    print(f"\n=== Phase 9: Micro (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        for attempt in range(4):
            rem = deadline - time.monotonic() - 8
            if rem < 6:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            if n_best <= 1200:
                a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem, 15))
            else:
                a_opt, _ = adam_subgradient(f_pert, time_limit=min(rem, 12),
                                             lr=0.001, top_k=max(10, n_best // 10),
                                             temp=0.00002)
            try_seq(a_opt, f"micro_{attempt}")

        # Very final LP-FW if time permits
        rem = deadline - time.monotonic() - 5
        if rem > 10 and len(best_seq) <= 1200:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "very_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
