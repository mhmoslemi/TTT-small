import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, c, T, fft_size):
    """Gradient of (1/T)*log(sum(exp(c/T))) w.r.t. a, using FFT cross-correlation."""
    n = len(a)
    c_scaled = c / T
    c_scaled -= c_scaled.max()
    exp_c = np.exp(c_scaled)
    w = exp_c / exp_c.sum()
    
    # grad[j] = (2/T) * sum_m w[j+m] * a[m]
    # = (2/T) * (w cross-correlate a)[j]
    N = fft_size
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)
    return (2.0 / T) * R[:n]


def project_simplex(a):
    """Project onto probability simplex."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

    n = len(a)
    c0 = fast_convolve(a)
    
    if T0 is None:
        T0 = 0.05 * c0.max()
    
    # FFT size for gradient computation (need >= 3n-2)
    need = 3 * n - 2

    best_obj = 2 * n * c0.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    T = T0
    lr = lr0
    momentum = momentum0
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    best_lr = lr0

    T_end = T_min

        elapsed = time.monotonic() - t0

def spectral_relax(a0, time_limit, iterations=2000, verbose=False):
    """
    Spectral relaxation (Fienup direct method variant):
    Alternate between amplitude constraint in frequency and non-negativity in time.
    """
    a = project_simplex(a0.astype(np.float64))
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_obj = 2 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < iterations:
        it += 1
        
        # Frequency domain step: flatten |A|^2
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A)
        amp[0] = np.sqrt(n)  # Keep DC component fixed (sum = 1 means A[0] = 1)
        # Actually A[0] = sum(a) = 1, so amp[0] = 1
        amp[0] = 1.0
        
        # Target: flat amplitude
        target_amp = np.ones_like(amp)
        target_amp[0] = 1.0
        
        # Blend current and target
        alpha = 0.1  # Relaxation parameter
        new_amp = (1 - alpha) * amp + alpha * target_amp
        new_amp[0] = 1.0
        
        # Keep phase
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        
        # Time domain: take IFFT, keep first n elements, project
        a_new_full = np.fft.irfft(A_new, fft_size)
        a_new = np.maximum(a_new_full[:n], 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        c_new = fast_convolve(a_new)
        obj_new = 2 * n * c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        a = a_new
    
    return best_a.copy(), best_obj


def generate_inits(parent, rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Copy of parent (if same length)
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())

    inits.append(np.ones(n) / n)
    
    # Asymmetric: linear ramp (non-palindromic!)
    for slope in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a = 1.0 + slope * x / (n - 1)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    # Asymmetric: exponential (non-palindromic)
    for beta in [0.5, 1.0, 2.0, 3.0]:
        a = np.exp(-beta * x / n)
        inits.append(a / a.sum())
        a = np.exp(-beta * (n - 1 - x) / n)
        inits.append(a / a.sum())

        a = np.maximum(a, 0.001)

import numpy as np
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, c, T, fft_size):
    """Gradient of (1/T)*log(sum(exp(c/T))) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    c_scaled = c / T
    c_scaled -= c_scaled.max()
    exp_c = np.exp(c_scaled)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = exp_c / s
    N = fft_size
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)
    return (2.0 / T) * R[:n]


def project_simplex(a):
    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def smd_optimize(a0, time_limit, T0=None, T_min=1e-6, momentum0=0.9,
                 lr0=0.008, verbose=False, label=""):
    """Smooth Momentum Descent with adaptive temperature and step size."""
    a = project_simplex(a0.astype(np.float64))
    n = len(a)
    c0 = fast_convolve(a)
    if T0 is None:
        T0 = 0.05 * c0.max()
    need = 3 * n - 2
    fft_size = 1 << (need - 1).bit_length()
    best_obj = 2.0 * n * c0.max()
    best_a = a.copy()
    v = np.zeros(n)
    T = T0
    lr = lr0
    momentum = momentum0
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr_min = 1e-5
    lr_max = 0.05

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        T = T_min + 0.5 * (T0 - T_min) * (1.0 + math.cos(math.pi * progress))
        c = fast_convolve(a)
        g = compute_smooth_grad(a, c, T, fft_size)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()
        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, lr_max)
        else:
            no_improve += 1
            lr = max(lr * 0.98, lr_min)
        a = a_new
        if no_improve > 1500:
            rng_l = np.random.default_rng(it * 13 + 7)
            a = best_a + rng_l.standard_normal(n) * 0.003
            a = project_simplex(a)
            v = np.zeros(n)
            lr = min(lr * 3.0, lr_max)
            no_improve = 0
        if verbose and it % 3000 == 0:
            print(f"    SMD {label} it={it}: obj={best_obj:.8f} T={T:.5f} lr={lr:.5f}")
    return best_a.copy(), best_obj


def spectral_relax(a0, time_limit, alpha=0.15, verbose=False, label=""):
    """Spectral relaxation: alternate amplitude flattening in freq with non-neg in time."""
    a = project_simplex(a0.astype(np.float64))
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A).copy()
        phase = np.angle(A)
        target = np.ones_like(amp)
        new_amp = (1.0 - alpha) * amp + alpha * target
        new_amp[0] = amp[0]
        A_new = new_amp * np.exp(1j * phase)
        a_full = np.fft.irfft(A_new, fft_size)
        a_new = np.maximum(a_full[:n], 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        a = a_new
        if verbose and it % 500 == 0:
            print(f"    SpecRelax {label} it={it}: obj={best_obj:.8f}")
    return best_a.copy(), best_obj


def coord_polish(a0, time_limit, verbose=False):
    """Coordinate descent focusing on peak-contributing coordinates."""
    a = project_simplex(a0.astype(np.float64))
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(a)
        peak_idx = int(c.argmax())
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        contribs = a[js:je+1] * a[peak_idx - je:peak_idx - js + 1][::-1]
        top_k = min(30, len(contribs))
        top_local = np.argsort(contribs)[-top_k:][::-1]
        top_coords = top_local + js
        rng.shuffle(top_coords)
        for j in top_coords:
            old = a[j]
            if old < 1e-12:
                continue
            for factor in [0.4, 0.6, 0.8, 0.9, 1.1, 1.4, 1.8, 0.5, 0.7]:
                a[j] = max(old * factor, 0.0)
                a = project_simplex(a)
                c = fast_convolve(a)
                obj = 2.0 * n * c.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                a = best_a.copy()
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break
        if verbose and no_improve % 10 == 0:
            print(f"    CoordPolish: obj={best_obj:.8f} no_improve={no_improve}")
    return best_a.copy(), best_obj


def generate_inits(parent, rng, n):
    """Generate diverse asymmetric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / max(n - 1, 1)
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
    inits.append(np.ones(n) / n)
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + slope * xn
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    for slope in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 - slope * xn
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    for beta in [0.5, 1.0, 2.0, 4.0]:
        a = np.exp(-beta * xn)
        inits.append(a / a.sum())
        a = np.exp(-beta * (1.0 - xn))
        inits.append(a / a.sum())
    for c in [0.2, 0.3, 0.4, 0.6]:
        a = np.exp(-8.0 * ((xn - c) ** 2))
        a = a + 0.05
        inits.append(a / a.sum())
    for c1, c2, s in [(0.3, 0.7, 0.1), (0.2, 0.8, 0.08), (0.25, 0.75, 0.12)]:
        a = np.exp(-((xn - c1) ** 2) / (2 * s ** 2)) + 0.7 * np.exp(-((xn - c2) ** 2) / (2 * (s * 1.3) ** 2))
        a = a + 0.02
        inits.append(a / a.sum())
    for ph in [0.5, 1.0, 1.5, 2.0]:
        a = np.abs(np.cos(np.pi * xn + ph)) + 0.05
        inits.append(a / a.sum())
    for _ in range(4):
        inits.append(rng.dirichlet(np.ones(n) * 0.4))
    for _ in range(4):
        a = rng.uniform(0.1, 1.0, n) + 0.5 * rng.uniform(0, 1) * xn
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    return inits
