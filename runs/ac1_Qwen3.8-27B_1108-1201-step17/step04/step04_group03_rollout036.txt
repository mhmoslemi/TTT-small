def subspace_gd(a0, time_limit, top_k=20, lr=0.01, verbose=False):

    Gradient descent focusing on the top-k peaks with adaptive weighting.
    Uses the subgradient of the max function (weighted by peak values).
    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_sel = min(top_k, len(c))
        peaks = np.argpartition(c, -k_sel)[-k_sel:]
        pv = c[peaks]
        temp = 0.1 * M
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()
        
        # Compute subgradient: weighted combination of gradients of active peaks
        # grad of c_k w.r.t. a_m = 2 * a[k-m] (if valid)
        # Use FFT for efficiency
        grad = np.zeros(n)
        L = 1
        while L < 3 * n - 1:
            L <<= 1

        w_signal = np.zeros(2 * n - 1)
        w_signal[peaks] = 2.0 * w
        W = np.fft.rfft(w_signal, L)
        A = np.fft.rfft(a, L)
        corr = np.fft.irfft(W * np.conj(A), L)[:n]
        grad = corr
        
        # Momentum update
        momentum = mu * momentum - lr * grad
        a_new = project_simplex(a + momentum)
        
        # Check improvement
        c_new = fast_convolve(a_new)
        if c_new.max() < M:
            a = a_new
        else:

            lr *= 0.8
            a_new = project_simplex(a - lr * 0.5 * momentum)
            a = a_new

            lr = 0.005
            no_improve = 0

def generate_inits(n, rng):
    """Generate diverse initializations exploiting asymmetry."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Exponential decay (front-loaded)
    for gamma in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.exp(-gamma * x / (n - 1))
        inits.append(a / a.sum())
    
    # Power law (front-loaded)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        a = (n - x) ** alpha
        inits.append(np.maximum(a, 0) / np.maximum(a, 0).sum())
    
    # Half-delta (mass in first half only)
    a = np.zeros(n)
    a[:n//2] = 1.0
    inits.append(a / a.sum())
    
    # Quarter-delta
    a = np.zeros(n)
    a[:n//4] = 1.0
    inits.append(a / a.sum())
    
    # Cosine window (front-loaded)
    a = 0.5 * (1 + np.cos(np.pi * x / n))
    inits.append(a / a.sum())
    
    # Triangle peak at position n/4
    a = np.exp(-0.5 * ((x - n*0.25) / (n*0.1)) ** 2)
    inits.append(a / a.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive softmax-approximated
gradient descent (log-sum-exp) with FFT-based gradient computation.
Key innovations: (1) smooth log-sum-exp objective for stable gradients,
(2) O(n log n) gradient via FFT cross-correlation, (3) adaptive beta schedule
transitioning from L2 to L-infinity behavior, (4) diverse asymmetric
initializations exploiting the structure of linear autocorrelation,
(5) backtracking line search for robust step-size selection.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_grad(a, c, beta):
    """
    Gradient of (1/beta)*log(sum_k exp(beta*c_k)) w.r.t. a.
    Uses FFT cross-correlation for O(n log n) computation.
    Returns 2*(w cross-correlate a)[:n] where w = softmax(beta*c).
    """
    n = len(a)
    bc = beta * c
    bc -= bc.max()
    w = np.exp(bc)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(W * np.conj(A), L)[:n]
    return 2.0 * corr

    to the max of the autocorrelation. Adaptive beta increases over time.

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        beta = beta_start * (beta_end / beta_start) ** frac

        c = fast_convolve(a)
        M = c.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()

        else:

        # Backtracking line search
        best_lr = cur_lr
        best_val = M
        lr_try = cur_lr
        for _ in range(8):
            a_try = project_simplex(a - lr_try * grad)
            val_try = fast_convolve(a_try).max()
            if val_try < best_val:
                best_val = val_try
                best_lr = lr_try
            lr_try *= 0.6

        if best_lr > 0:
            a = project_simplex(a - best_lr * grad)
            cur_lr = min(best_lr * 1.2, lr_start)
        else:
            # No improvement possible with this direction, perturb
            rng = np.random.default_rng(it * 7919 + 13)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            cur_lr = lr_start * 0.5

        # Periodic restart
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % (2**31 - 1))
            noise_scale = 0.002 + 0.003 * min(1.0, no_improve / 600.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            cur_lr = lr_start * 0.7
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"      iter={it}, beta={beta:.1f}, M={M:.7f}, best={best_obj:.7f}, lr={cur_lr:.5f}")

    return best_a, best_obj
