"""
Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: FW-LP with adaptive constraints, mirror descent with LSE smoothing,
coordinate descent polishing, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_sparse(f):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.append(k * np.ones(len(js), dtype=int))
            cols.append(js.astype(int))
            vals.append(f[k - js])
    if not rows:
        return csr_matrix((2 * n - 1, n))
    return csr_matrix((np.concatenate(vals), (np.concatenate(rows), np.concatenate(cols))),
                      shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max 1^T g s.t. A f g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_lp_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def mirror_descent(f0, time_limit, beta_start=10.0, beta_end=5000.0):
    """Mirror descent (multiplicative weights) with LSE smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    eta = 1.0
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient: grad[i] = 2 * sum_k w[k] * f[k-i] (cross-correlation)
        # Use FFT
        f_flipped = f[::-1].copy()
        total_len = (2 * n - 1) + n - 1
        fft_size = 1 << (total_len - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f_flipped, fft_size)
        conv_wf = np.fft.irfft(W * F, fft_size)
        grad = 2.0 * conv_wf[n - 1:n - 1 + n]

        # Mirror descent update: f_new = f * exp(-eta * grad) / Z
        log_f = np.where(f > 1e-15, np.log(f), -20.0)
        new_log_f = log_f - eta * grad
        new_log_f -= new_log_f.max()
        f_new = np.exp(new_log_f)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / f_new_sum

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            eta = min(eta * 1.01, 5.0)
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                eta *= 0.95
            if no_improve > 100:
                f = best_f.copy()
                eta = min(2.0, eta)
                beta = min(beta * 1.3, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            eta = 1.0
            beta = min(beta_start * 2, beta_end)
            no_improve = 0

    return best_f, best_obj

    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.5 / n
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    temp = 0.001

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Top-k peaks
        k_eff = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_idx = np.arange(js, je + 1)
            sg[j_idx] += 2.0 * w[idx] * f[k - j_idx]

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.005, 2.0 / n)
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.93
            if no_improve > 200:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 0.3 / n
                no_improve = 0

        temp *= 1.003

    return best_f, best_obj


def cd_polish(f0, time_limit):
    """Coordinate descent via pairwise swaps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        # Try random pairs
        n_trials = min(200, n * 2)
        for _ in range(n_trials):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if f[i] < 1e-10 and f[j] < 1e-10:
                continue

            # Try moving mass from j to i: f[i]+t, f[j]-t
            # Find optimal t via bisection
            t_min = -f[j] * 0.99
            t_max = f[i] * 0.99

            best_t = 0.0
            best_val = conv.max()

            # Sample some t values
            ts = np.linspace(t_min, t_max, 9)
            for t in ts:
                f_test = f.copy()
                f_test[i] += t
                f_test[j] -= t
                if f_test[i] < -1e-14 or f_test[j] < -1e-14:
                    continue
                f_test = np.maximum(f_test, 0)
                st = f_test.sum()
                if st < 1e-10:
                    continue
                val = fast_convolve(f_test).max()
                if val < best_val:
                    best_val = val
                    best_t = t

            if best_t != 0:
                # Refine
                t_span = max(abs(best_t), 1e-4)
                lo = best_t - t_span * 0.3
                hi = best_t + t_span * 0.3
                lo = max(lo, t_min)
                hi = min(hi, t_max)
                for _ in range(12):
                    mid = (lo + hi) / 2
                    f_test = f.copy()
                    f_test[i] += mid
                    f_test[j] -= mid
                    f_test = np.maximum(f_test, 0)
                    val = fast_convolve(f_test).max()
                    # Compare with nearby points
                    f_test2 = f.copy()
                    mid2 = (lo + mid) / 2
                    f_test2[i] += mid2
                    f_test2[j] -= mid2
                    f_test2 = np.maximum(f_test2, 0)
                    val2 = fast_convolve(f_test2).max()
                    if val <= val2:
                        lo = mid
                    else:
                        hi = mid
                    if val < best_val:
                        best_val = val
                        best_t = mid

                if best_val < best_obj - 1e-13:
                    f[i] += best_t
                    f[j] -= best_t
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    conv = fast_convolve(f)
                    obj = conv.max()
                    if obj < best_obj:
                        best_obj = obj
                        best_f = f.copy()
                        improved = True

        if not improved:
            # Try systematic sweep
            any_improve = False
            for i in range(0, n, max(1, n // 100)):
                for j in range(i + 1, min(i + 5, n)):
                    t_min = -f[j] * 0.5
                    t_max = f[i] * 0.5
                    if t_min >= t_max:
                        continue
                    ts = np.linspace(t_min, t_max, 5)
                    for t in ts:
                        f_test = f.copy()
                        f_test[i] += t
                        f_test[j] -= t
                        f_test = np.maximum(f_test, 0)
                        val = fast_convolve(f_test).max()
                        if val < best_obj - 1e-13:
                            f = f_test / f_test.sum()
                            conv = fast_convolve(f)
                            if conv.max() < best_obj:
                                best_obj = conv.max()
                                best_f = f.copy()
                                any_improve = True
                                break
                    if any_improve:
                        break
                if any_improve:
                    break
            if not any_improve:
                break

    return best_f, best_obj


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        a = a[np.isfinite(a)]
        if len(a) == 0:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        return score
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase:
    1. FW-LP global optimization
    2. Mirror descent (LSE) polishing
    3. Subgradient polish
    4. Coordinate descent
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_score = float('inf')
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        score = evaluate_safe(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        p = p[np.isfinite(p)]
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_safe(parent):.8f}")
            update_best(parent)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(448) / 448
        update_best(parent)

    n_parent = len(parent)
    current_best = best_seq if best_seq is not None else parent

    # === Phase 1: FW-LP ===
    remaining = deadline - time.monotonic()
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: FW-LP ({t1:.0f}s) ===")
    a_opt, obj = fw_lp(current_best, time_limit=t1)
    update_best(a_opt)

    # === Phase 2: Mirror Descent ===
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.30
    current_best = best_seq if best_seq is not None else current_best
    print(f"\n=== Phase 2: Mirror Descent ({t2:.0f}s) ===")
    a_opt, obj = mirror_descent(current_best, time_limit=t2,
                                beta_start=8.0, beta_end=4000.0)
    update_best(a_opt)

    # === Phase 3: Subgradient Polish ===
    remaining = deadline - time.monotonic()
    t3 = remaining * 0.20
    current_best = best_seq if best_seq is not None else current_best
    n_best = len(current_best)
    print(f"\n=== Phase 3: Subgradient ({t3:.0f}s) ===")
    a_opt, obj = subgradient_polish(current_best, time_limit=t3,
                                    top_k=max(8, n_best // 32))
    update_best(a_opt)

    # === Phase 4: Coordinate Descent ===
    remaining = deadline - time.monotonic()
    t4 = remaining * 0.25
    current_best = best_seq if best_seq is not None else current_best
    print(f"\n=== Phase 4: CD Polish ({t4:.0f}s) ===")
    a_opt, obj = cd_polish(current_best, time_limit=t4)
    update_best(a_opt)

    # === Phase 5: Alternative lengths ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Alt Lengths ({remaining:.0f}s) ===")
    alt_lengths = [320, 384, 512, 576, 640]
    if n_parent in alt_lengths:
        alt_lengths.remove(n_parent)

    for n_alt in alt_lengths[:3]:
        t_rem = deadline - time.monotonic() - 10
        if t_rem < 30:
            break
        t_alloc = min(80, t_rem / max(1, len(alt_lengths) - alt_lengths.index(n_alt)))
        if t_alloc < 20:
            break

        # Interpolate
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        # Quick optimize
        t_fw = t_alloc * 0.5
        t_md = t_alloc * 0.5
        a_opt, _ = fw_lp(f_interp, time_limit=t_fw)
        update_best(a_opt)
        a_opt, _ = mirror_descent(a_opt, time_limit=t_md,
                                  beta_start=10.0, beta_end=2000.0)
        update_best(a_opt)

    # Final best
    if best_seq is None:
        best_seq = np.ones(448) / 448
        best_score = evaluate_safe(best_seq)

    # Final verification
    final_score = evaluate_safe(best_seq)
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
