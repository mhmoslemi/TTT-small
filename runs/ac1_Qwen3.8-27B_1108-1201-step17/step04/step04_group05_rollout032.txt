"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated smoothed
gradient descent, LP-based Frank-Wolfe, and coordinate descent polishing.
Multi-length search with diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_smoothed(a, p):
    """
    Compute gradient of T*log(sum exp(conv(a,a)/T)) w.r.t. a.
    p = softmax(conv(a,a)/T) (length 2n-1)
    Returns gradient of length n.
    grad_i = 2 * sum_m p_{m+i} * a_m
    """
    n = len(a)
    # r[k] = sum_m p_{m+k} * a_m for k = 0, ..., n-1
    # This is the cross-correlation of p and a
    # Use FFT: r = ifft(conj(fft(a)) * fft(p)) ... 
    # Actually, r[k] = sum_m a_m * p_{m+k}
    # Let's compute via: flip a, convolve with p, extract
    a_flip = a[::-1]
    # conv(a_flip, p)[n-1+k] = sum_j a_flip[j] * p[n-1+k-j] = sum_j a[n-1-j] * p[n-1+k-j]
    # Let m = n-1-j: = sum_m a[m] * p[n-1+k-(n-1-m)] = sum_m a[m] * p[m+k] = r[k] ✓
    fft_size = 1 << (n + 2 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a_flip, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(A * P, fft_size)[:n + 2*n-2-1]
    r = conv_full[n-1:n-1+n]
    return 2.0 * r

def nesterov_smoothed_optimize(a0, time_limit, T0=0.01, T_min=0.0005, verbose=False):
    """
    Nesterov accelerated gradient descent on the simplex with log-sum-exp smoothing.
    Minimizes T * log(sum exp(conv(a,a)/T)) which approximates max(conv(a,a)).
    """
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T0
    t = 1.0  # Nesterov parameter
    y = a.copy()  # Nesterov extrapolated point
    z = a.copy()  # Previous y
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    step_size = 2.0 / (n * T)  # Initial step size (heuristic)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov extrapolation
        beta = (t - 1) / (t + 2)
        y_new = y + beta * (y - z)
        z = y.copy()
        y = project_simplex(y_new)
        
        # Compute smoothing weights
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        exp_sum = np.exp(scaled).sum()
        p = np.exp(scaled) / exp_sum
        
        # Compute gradient
        grad = compute_gradient_smoothed(y, p)
        
        # Gradient step with backtracking
        alpha = step_size / t
        x_new = y - alpha * grad
        x_new = project_simplex(x_new)
        
        # Backtracking line search (simplified: just try and accept if better)
        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        y = x_new
        t = (1 + np.sqrt(1 + 4 * t * t)) / 2
        
        # Decrease temperature
        if it % 50 == 0:
            T = max(T_min, T * 0.95)
        
        # Periodic restart if stuck
        if no_improve > 200:
            y = project_simplex(best_a + np.random.standard_normal(n) * 0.003)
            no_improve = 0
            T = max(T0 * 0.5, T)
        
        if verbose and it % 500 == 0:
            print(f"    Nesterov iter {it}: obj={best_obj:.10f}, T={T:.6f}")

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining
Nesterov-accelerated smoothed gradient descent, LP-based Frank-Wolfe,
coordinate descent with pair transfers, and multi-length search.
Key improvements over previous: Nesterov momentum, adaptive temperature
annealing, pair-swap coordinate descent, and more aggressive exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - threshold_safe(v, theta), 0.0)


def threshold_safe(v, theta):
    return np.full_like(v, theta)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, B=None):
    """Solve linearized LP for Frank-Wolfe direction."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def compute_smoothed_grad(a, p):
    """
    Compute gradient of T*log(sum exp(conv(a,a)/T)) w.r.t. a.
    p = softmax of conv(a,a)/T, length 2n-1.
    grad_i = 2 * sum_m p_{m+i} * a_m
    """
    n = len(a)
    a_flip = a[::-1]
    total_len = n + (2 * n - 1) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a_flip, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(A * P, fft_size)[:total_len]
    r = conv_full[n - 1:n - 1 + n]
    return 2.0 * r


def nesterov_optimize(a0, time_limit, T_init=0.02, T_min=0.0002,
                       verbose=False):
    """Nesterov accelerated projected gradient with log-sum-exp smoothing."""
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    n = len(a)

    T = T_init
    t_param = 1.0
    y = a.copy()
    y_prev = a.copy()
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov extrapolation
        beta = (t_param - 1.0) / (t_param + 2.0)
        y_mom = y + beta * (y - y_prev)
        y_prev = y.copy()
        y = project_simplex(y_mom)

        # Compute smoothing probabilities
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        exp_sum = exp_vals.sum()
        p = exp_vals / exp_sum

        # Gradient
        grad = compute_smoothed_grad(y, p)

        # Step size with backtracking
        step = 2.0 / (t_param * T * n)
        x_cand = project_simplex(y - step * grad)

        # Accept if better on the actual (unsmoothed) objective
        conv_x = fast_convolve(x_cand)
        obj_x = conv_x.max()

        if obj_x < best_obj - 1e-14:
            best_obj = obj_x
            best_a = x_cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = x_cand
        t_param = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0

        # Anneal temperature
        if it % 30 == 0:
            T = max(T_min, T * 0.97)

        # Restart if stuck
        if no_improve > 150:
            y = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            y_prev = y.copy()
            t_param = 1.0
            T = max(T_init * 0.4, T)
            no_improve = 0

        if verbose and it % 300 == 0:
            print(f"    Nesterov iter {it}: obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def linfw_optimize(a0, time_limit, use_reduced=False, reduced_k=25,
                   verbose=False):

    f = a0.astype(np.float64).copy()

        current_obj = conv.max()

            no_improve += 5
            if no_improve > 80:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
            continue

        target = np.maximum(f + g, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        # Golden section line search
        phi = 0.618
        lo, hi = 0.0, 1.0
        c1 = lo + (1 - phi) * (hi - lo)
        c2 = lo + phi * (hi - lo)

        def eval_alpha(al):
            ft = (1.0 - al) * f + al * target
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                return 1e10
            ft = ft / st
            return fast_convolve(ft).max()

        f1 = eval_alpha(c1)
        f2 = eval_alpha(c2)
        for _ in range(12):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = lo + (1 - phi) * (hi - lo)
                f1 = eval_alpha(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + phi * (hi - lo)
                f2 = eval_alpha(c2)

        best_alpha = (lo + hi) / 2.0
        best_val = eval_alpha(best_alpha)

        if best_val < best_obj - 1e-14:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            best_obj = fast_convolve(f).max()

                f = project_simplex(best_f + rng.standard_normal(n) * (0.002 + 0.0001 * (no_improve // 30)))
                no_improve = 0

    return best_f, best_obj

def coordinate_descent(a0, time_limit, rng, n_pairs_per_iter=50):
    """
    Coordinate descent via pair transfers.
    For each pair (i,j), try transferring mass and keep the best.
    """
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Sample random pairs
        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try transferring from i to j (and vice versa)
            # a_i' = a_i - delta, a_j' = a_j + delta
            # delta range: [max(-a_j, -a_i), min(a_i, a_j)] for bidirectional
            # We'll try a few delta values
            for delta in [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a[i] += delta
                a[j] -= delta
                conv_new = fast_convolve(a)
                obj = conv_new.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    a[i] -= delta
                    a[j] += delta

        no_improve += 1
        if no_improve > 200:
            # Reset to best and add small noise
            a = best_a + rng.standard_normal(n) * 0.001
            a = project_simplex(a)
            no_improve = 0

    return best_a, best_obj


def subgrad_optimize(a0, time_limit, lr=0.002, mom=0.92, top_k=40,
                     temp=0.0001):
    """Subgradient descent with multi-peak targeting."""
    f = a0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = np.maximum(f, 0.0)
        f = f / f.sum()
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    # Various Gaussians
    for sigma_frac in [0.12, 0.2, 0.3, 0.45]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.02
    inits.append(c / c.sum())

    # Beta-like
    b = (x + 1) ** 2.5 * (n - x) ** 2.5
    inits.append(b / b.sum())

    # U-shape (edge concentrated)
    edge = 1.0 / (0.3 * np.abs(x - mid) / n + 0.05)
    inits.append(edge / edge.sum())

    # Two bumps
    for frac in [0.25, 0.3]:
        bump1 = np.exp(-0.5 * ((x - frac * (n - 1)) / (0.08 * n)) ** 2)
        bump2 = np.exp(-0.5 * ((x - (1 - frac) * (n - 1)) / (0.08 * n)) ** 2)
        two = bump1 + bump2
        inits.append(two / two.sum())

    # Sawtooth / periodic
    for period in [8, 12, 16]:
        saw = 0.5 + 0.5 * np.sin(2 * np.pi * x / period)
        saw = saw + 0.1 * np.ones(n)
        inits.append(saw / saw.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 800:
            break
        inits = generate_inits(n_try, parent, rng)
        for i, ini in enumerate(inits[:4]):
            conv = fast_convolve(ini)
            proxy = 2 * n_try * conv.max()  # since sum=1
            screened.append((proxy, n_try, ini, f"n{n_try}_i{i}"))

    screened.sort(key=lambda x: x[0])
    # Evaluate top candidates
    for proxy, n_try, ini, label in screened[:12]:
        try_seq(ini, label)

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    print(f"Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: LP Frank-Wolfe ===
    print("\n=== Phase 2: LP Frank-Wolfe ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 600
    t_p2 = min(remaining * 0.4, 500)

    if t_p2 > 30:
        a_opt, _ = linfw_optimize(best_seq, time_limit=t_p2)
        try_seq(a_opt, "linfw")

        # Perturbed restart
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            noise = rng.standard_normal(n_best) * 0.008
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = linfw_optimize(f_pert, time_limit=min(rem * 0.6, 200))
            try_seq(a_opt, "linfw_pert")

    print(f"Phase 2 best: {best_score:.8f}")

    # === Phase 3: Nesterov Smoothed Gradient ===
    print("\n=== Phase 3: Nesterov Gradient ===")
    remaining = deadline - time.monotonic() - 400
    t_p3 = min(remaining * 0.35, 400)

    if t_p3 > 30:
        a_opt, _ = nesterov_optimize(best_seq, time_limit=t_p3 * 0.7,
                                      T_init=0.03, T_min=0.0003)
        try_seq(a_opt, "nesterov")

        rem = deadline - time.monotonic() - 300
        if rem > 30:
            a_opt, _ = nesterov_optimize(best_seq, time_limit=min(rem * 0.5, 150),
                                          T_init=0.01, T_min=0.0001)
            try_seq(a_opt, "nesterov_fine")

    print(f"Phase 3 best: {best_score:.8f}")

    # === Phase 4: Subgradient + Coordinate Descent ===
    print("\n=== Phase 4: Subgrad + CoordDesc ===")
    remaining = deadline - time.monotonic() - 200
    t_p4 = min(remaining * 0.4, 300)

    if t_p4 > 20:
        a_opt, _ = subgrad_optimize(best_seq, time_limit=min(t_p4 * 0.4, 100),
                                     lr=0.002, mom=0.93,
                                     top_k=max(20, n_best // 8), temp=0.00008)
        try_seq(a_opt, "subgrad")

        rem = deadline - time.monotonic() - 150
        if rem > 20:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.5, 80),
                                           rng=rng, n_pairs_per_iter=80)
            try_seq(a_opt, "coorddesc")

    print(f"Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final polish and length exploration ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic() - 50
    t_p5 = min(remaining * 0.6, 250)

    if t_p5 > 20:
        # Final LP polish
        a_opt, _ = linfw_optimize(best_seq, time_limit=min(t_p5 * 0.5, 100))
        try_seq(a_opt, "final_linfw")

        rem = deadline - time.monotonic() - 30
        if rem > 20:
            # Max-sum FW for a different angle
            a_opt, _ = maxsum_fw(best_seq, time_limit=min(rem * 0.5, 60))
            try_seq(a_opt, "final_maxfw")

        # Try nearby lengths
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            for n_try in [n_best - 8, n_best + 8, n_best - 32, n_best + 32]:
                if n_try < 64 or rem < 10:
                    continue
                px = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, n_best), best_seq)
                px = np.maximum(px, 0)
                ps = px.sum()
                if ps > 1e-10:
                    px = px / ps
                    a_opt, _ = linfw_optimize(px, time_limit=min(rem * 0.3, 25))
                    try_seq(a_opt, f"len{n_try}")
                    rem = deadline - time.monotonic() - 15

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
