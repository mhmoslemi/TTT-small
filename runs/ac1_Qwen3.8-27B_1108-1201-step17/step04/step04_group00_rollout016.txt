```python
    a = normalize(a0)  # sum to 1, clip to 0
    n = len(a)
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    v = np.zeros(n)  # momentum buffer
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        # Compute convolution
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
        
        # Find top-K peaks
        k_actual = min(K, 2*n-1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        
        # Softmax weights
        scaled = pv / (pv.max() * 0.1)  # temperature relative to max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx, k in enumerate(peaks):
            if w[idx] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            i_arr = np.arange(js, je + 1)
            sg[i_arr] += 2.0 * w[idx] * a[k - i_arr]
        
        # Momentum
        v = momentum * v + sg
        
        # Multiplicative weights update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * v
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new /= a_new.sum()
        a = a_new
        
        # Adaptive eta
        # (could add backtracking here)
    
    return best_a, best_obj
```

```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute the objective value for a sequence (assumes sum(a)=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return n * conv.max()  # This is n * max_conv, and the full objective is 2 * this


def normalize_nn(a):
    """Normalize to be non-negative with sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

    """Create a geometric sequence of length n with ratio r, normalized to sum 1."""
    if abs(r - 1.0) < 1e-10:

    a = a / a.sum()

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        if g is None:
            return None
    gs = g.sum()
    if gs < 1e-15:

    g = g / gs
    return g
```

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_nn(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def geometric_seq(n, r):
    """Geometric sequence normalized to sum 1."""
    if abs(r - 1.0) < 1e-10:
        return np.ones(n) / n
    a = r ** np.arange(n, dtype=np.float64)
    return normalize_nn(a)


def spectral_init(n, n_iter=20, seed=0):
    """Gerchberg-Saxton style initialization for flat spectrum."""
    rng = np.random.default_rng(seed)
    phases = rng.uniform(0, 2 * np.pi, n)
    a = np.ones(n) / n
    for _ in range(n_iter):
        # Get DFT of current a
        A = np.fft.fft(a)
        # New phases = phases of DFT, new magnitudes = 1 (flat)
        target = np.ones(n) * np.exp(1j * np.angle(A))
        # Inverse DFT
        h = np.fft.ifft(target)
        a = np.abs(h) ** 2
        a = normalize_nn(a)
    return a


def exp_gradient(a0, time_limit, K=30, eta0=0.5, momentum=0.7,
                 temp_frac=0.05, restart_every=300, verbose=False):
    """
    Exponential gradient (multiplicative weights) with momentum.
    Works on probability vectors (sum=1, non-negative).
    """
    n = len(a0)
    a = normalize_nn(a0.astype(np.float64))
    v = np.zeros(n)
    
    conv = fast_convolve(a)
    best_raw = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    best_obj = n * best_raw
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(a)
        raw = conv.max()
        obj = n * raw
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_raw = raw
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"      EG {it}: obj={obj:.8f}")
        else:
            no_improve += 1
        
        # Top-K peaks
        k_act = min(K, 2 * n - 1)
        if k_act >= 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv.copy()
        else:
            idx_k = np.argpartition(conv, -k_act)[-k_act:]
            peaks = idx_k
            pv = conv[idx_k]
        
        # Softmax weights
        pmax = pv.max()
        if pmax < 1e-300:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            temp = temp_frac * pmax
            s = pv / temp
            s -= s.max()
            w = np.exp(s)
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
            else:
                w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for pi in range(len(peaks)):
            if w[pi] < 1e-12:
                continue
            k = int(peaks[pi])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            iarr = np.arange(js, je + 1)
            sg[iarr] += 2.0 * w[pi] * a[k - iarr]
        
        # Momentum update
        v = momentum * v + sg
        vn = np.linalg.norm(v)
        if vn < 1e-300:
            break
        
        # Step size decay
        eta = eta0 * max(0.1, 1.0 / (1.0 + 0.005 * no_improve))
        
        # Multiplicative weights update
        la = np.log(np.maximum(a, 1e-300))
        la -= eta * v / vn
        la -= la.max()
        a_new = np.exp(la)
        sn = a_new.sum()
        if sn < 1e-300:
            a = np.ones(n) / n
            v = np.zeros(n)
            no_improve += 100
        else:
            a = a_new / sn
        
        # Periodic restart
        if no_improve >= restart_every:
            rng = np.random.default_rng(it * 7919 + 104729)
            scale = 0.005 * min(no_improve // restart_every + 1, 5)
            a = normalize_nn(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj


def solve_lp_dir(f, M, n):
    """LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw(a0, time_limit, max_iters=30, verbose=False):
    """LP-based Frank-Wolfe for limited iterations."""
    n = len(a0)
    f = normalize_nn(a0.astype(np.float64))
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_raw = M
    m_exp = 1.0
    
    t0 = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = solve_lp_dir(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 4.0:
                break
            continue
        
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-12:
            break
        
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new
            m_exp = 1.0
            if M < best_raw - 1e-14:
                best_raw = M
                best_f = f.copy()
                if verbose:
                    print(f"      LP-FW {it}: raw={M:.10f}")
    
    return best_f, best_raw * n


def interp_seq(f, new_n):
    """Interpolate to new length."""
    if len(f) == new_n:
        return f.copy()
    ox = np.linspace(0, 1, len(f))
    nx = np.linspace(0, 1, new_n)
    return normalize_nn(np.interp(nx, ox, f))


def eval_and_track(a, best_score, label):
    """Evaluate sequence and return (is_better, score)."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        return False, float('inf')
    score = float(evaluate_sequence(a.tolist()))
    return (score < best_score), score
