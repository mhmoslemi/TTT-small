    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    tau = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - tau, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result * (target_sum / s)
    else:
        result = np.full(n, target_sum / n)
    return result


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step(a, conv, n):
    """One LP-based step with step sizing."""
    M = conv.max()
    peak_idx = int(np.argmax(conv))
    C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.0001, C)
    if not success or g_sum <= 1.0 + 1e-8:
        return a.copy(), False
    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False
    conv_ag = C @ g
    D = conv_ag[peak_idx] - M
    if D < -1e-10:
        t_max_est = min(1.0, -D / max(abs(D) * 0.1, 1e-10))
    else:
        t_max_est = 0.3
    candidates = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    candidates = [t for t in candidates if t <= t_max_est * 1.5] or [0.05]
    best_a = a.copy()
    best_obj = 2.0 * n * M
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            break
    if best_a is not a:
        return best_a, True
    return a.copy(), False


def gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.003):
    """Nesterov-accelerated gradient descent with adaptive peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        conv_y = fast_convolve(y)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * a[k - idxs]
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        lr *= 0.9995
        temp *= 1.003
        mom = min(0.97, mom + 0.0002)
        if no_improve >= 200:
            break
    return best_a, best_obj


def dual_ascent(a0, time_limit, eta_init=0.3, inner_iters=80, lr_init=0.3,
                verbose=False):
    """
    Dual ascent (multiplicative weights + inner QP via projected gradient).
    Solves: min_{a>=0, sum=1} max_k conv(a,a)[k]
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    delta = np.ones(2 * n - 1) / (2 * n - 1)

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    lr = lr_init
    no_improve = 0
    iteration = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Inner loop: solve min_a sum_k delta[k]*conv(a,a)[k] via projected gradient
        for _ in range(inner_iters):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            corr = cross_corr_delta_a(delta, a)
            grad = 2.0 * corr
            a_new = proj_simplex(a - lr * grad)
            a = a_new

        # Evaluate current solution
        conv_aa = fast_convolve(a)
        obj = 2.0 * n * conv_aa.max()
        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Outer step: multiplicative weights update on delta
        max_conv = conv_aa.max()
        if max_conv > 1e-15:
            scaled = conv_aa / max_conv
            # Use a temperature to control concentration
            alpha = min(3.0, eta * 10.0)
            delta = delta * np.exp(alpha * scaled)
            dn = delta.sum()
            if dn > 1e-15:
                delta = delta / dn
            else:
                delta = np.ones(2 * n - 1) / (2 * n - 1)

        # Adaptive parameters
        lr = max(0.05, lr * 0.9995)
        eta = min(5.0, eta * 1.0005)

        iteration += 1
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Dual iter {iteration}: obj={best_obj:.8f}, "
                  f"eta={eta:.3f}, lr={lr:.4f}, t={elapsed:.1f}s")

        # Handle stagnation
        if no_improve >= 200:
            # Perturb
            scale = 0.002 * min(no_improve // 100, 5)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            # Reset delta with some randomness
            delta = np.ones(2 * n - 1)
            conv_chk = fast_convolve(a)
            delta = delta * (1.0 + 0.5 * conv_chk / max(conv_chk.max(), 1e-15))
            delta = delta / delta.sum()
            no_improve = 0
            if no_improve > 1000:
                break

    return best_a, best_obj


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        a_new, improved = lp_step(a, conv, n)
        if improved:
            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        iteration += 1
        if verbose and iteration % 50 == 0:
            print(f"    LP iter {iteration}: obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")
        if no_improve >= max_no_improve:
            scale = 0.003 + 0.004 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 20 * perturb_count)
            if perturb_count > 30:
                break

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent: try adjusting pairs of elements."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        batch_size = max(1, int(300000 / n))
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit - 0.3:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            improved = False
            for delta in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
        if no_improve > 100:
            break
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    alt = np.ones(n)
    alt[::2] = 1.4
    alt[1::2] = 0.6
    inits['alternating'] = alt / alt.sum()
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid dual ascent + LP + gradient + coordinate descent search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = parent.copy()
                print(f"Parent loaded: n={len(p)}, score={parent_score:.6f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick gradient scan across lengths
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = 150.0

    lengths = [512, 768, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'parent_interp', 'uniform', 'sine1', 'random']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:3]

        t_per = max(8.0, (phase1_limit - 5) / (len(lengths) * 3))

        for name in init_names:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 200)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, obj = gradient_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.01, momentum=0.9,
                                           top_k=40, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Grad n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Dual ascent on best candidates (NEW METHOD)
    # =========================================================
    print("=== Phase 2: Dual Ascent ===")
    t_start2 = time.monotonic()

    # Collect top candidates
    top_configs = []
    seen = set()
    if best_sequence is not None:
        top_configs.append((len(best_sequence), best_sequence.copy()))
        seen.add(len(best_sequence))
    for score, n, name, a_opt in scan_results:
        if n not in seen and len(top_configs) < 3:
            top_configs.append((n, a_opt))
            seen.add(n)

    remaining = deadline - time.monotonic() - 400.0
    n_cand = max(1, len(top_configs))
    t_per_da = max(30.0, remaining * 0.7 / n_cand)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_da, deadline - time.monotonic() - 250)
        if t_alloc < 15:
            break
        print(f"  Dual Ascent candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = dual_ascent(a0, time_limit=t_alloc,
                                 eta_init=0.5, inner_iters=60, lr_init=0.3,
                                 verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  DualAscent {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  DualAscent {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP-based deep optimization
    # =========================================================
    print("=== Phase 3: LP Optimization ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 300.0
        if remaining > 50:
            # Multiple LP runs
            for pi in range(3):
                if time.monotonic() > deadline - 200:
                    break
                t_alloc = min(remaining / (3 - pi) * 0.7, deadline - time.monotonic() - 100)
                if t_alloc < 15:
                    break
                if pi == 0:
                    a0 = best_sequence.copy()
                else:
                    scale = 0.002 * (pi + 1)
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()
                a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  LP {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Second round of dual ascent + gradient (different params)
    # =========================================================
    print("=== Phase 4: Second-Order Refinement ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        if remaining > 60:
            # Dual ascent with different hyperparameters
            t_da = min(remaining * 0.4, 120.0)
            if t_da > 15:
                a_opt, obj = dual_ascent(best_sequence, time_limit=t_da,
                                         eta_init=1.0, inner_iters=100, lr_init=0.2,
                                         verbose=False)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  DualAscent2: {score:.6f} *NEW BEST*")

            # Gradient with different params
            remaining = deadline - time.monotonic() - 100.0
            if remaining > 15:
                for gi in range(2):
                    if time.monotonic() > deadline - 80:
                        break
                    t_sg = min(remaining / (2 - gi) * 0.6, 40.0)
                    if t_sg < 8:
                        break
                    cfgs = [
                        {'lr_init': 0.005, 'momentum': 0.93, 'top_k': 60, 'temp_init': 0.001},
                        {'lr_init': 0.008, 'momentum': 0.88, 'top_k': 30, 'temp_init': 0.004},
                    ]
                    cfg = cfgs[gi % len(cfgs)]
                    a_opt, obj = gradient_optimize(best_sequence, time_limit=t_sg, **cfg)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Grad2_{gi}: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Coordinate descent polishing
    # =========================================================
    print("=== Phase 5: Coordinate Descent ===")
    t_start5 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        t_cd = min(remaining * 0.5, 60.0)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

            # Final LP micro-step
            remaining = deadline - time.monotonic() - 15.0
            if remaining > 5:
                a_opt, obj = lp_optimize(best_sequence, time_limit=remaining)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  FinalLP: {score:.6f} *NEW BEST*")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        if parent is not None:
            best_sequence = parent
        else:
            best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
