```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse structured initializations at multiple lengths
2. L-BFGS optimization with smoothed max (log-sum-exp) on softmax parametrization
3. FFT-accelerated gradient computation
4. Animated temperature for the smoothing
5. Fine multiplicative weight subgradient polishing
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smoothed_obj_grad(x, tau, n):
    """
    Compute the smoothed objective and its gradient.
    a = softmax(x), F = 2n * tau * log(sum_k exp(c[k]/tau))
    Returns (value, gradient).
    """
    # Softmax with numerical stability
    x_max = np.max(x)
    a = np.exp(x - x_max)
    a_sum = a.sum()
    a = a / a_sum
    
    # Convolution
    c = fast_convolve(a)
    c_max = np.max(c)
    
    # p = softmax(c / tau)
    c_scaled = (c - c_max) / tau
    p = np.exp(c_scaled)
    p_sum = p.sum()
    p = p / p_sum
    
    # Value: 2n * (c_max + tau * log(sum_k exp((c[k]-c_max)/tau)))
    value = 2.0 * n * (c_max + tau * np.log(p_sum))
    
    # T[j] = sum_k p[k] * (a[k-j] + a[j-k]) via FFT
    L = 1 << (3 * n - 2 + 1).bit_length()  # Need L >= 3n-2
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    T = np.fft.irfft(P * 2.0 * np.real(A), L)[:n]
    
    # S = sum_k p[k] * c[k]
    S = np.dot(p, c)
    
    # Gradient: grad[j] = 2n/tau * a[j] * (T[j] - 2*S)
    grad = (2.0 * n / tau) * a * (T - 2.0 * S)
    
    return value, grad

def lbfgs_smoothed(a_init, tau_init, n_outer=8, time_budget=600, verbose=True):
    """
    Run L-BFGS with animated temperature.
    a_init: initial probability vector (sum=1, non-negative)
    tau_init: initial smoothing temperature
    """
    n = len(a_init)
    # Convert to softmax parametrization: x = log(a) (approximately)
    a_safe = np.maximum(a_init, 1e-30)
    x = np.log(a_safe / a_safe.sum())
    
    tau = tau_init
    t0 = time.monotonic()
    best_val = float('inf')
    best_x = x.copy()
    
    for outer in range(n_outer):
        elapsed = time.monotonic() - t0
        if elapsed > time_budget - 30:
            break
        
        remaining = time_budget - elapsed
        time_per_outer = remaining / (n_outer - outer)
        
        # L-BFGS on x
        result = minimize(
            fun=lambda x: compute_smoothed_obj_grad(x, tau, n)[0],
            x0=x,
            jac=lambda x: compute_smoothed_obj_grad(x, tau, n)[1],
            method='L-BFGS-B',
            options={
                'maxiter': max(50, int(time_per_outer * 100)),  # Rough estimate
                'maxfun': 5000,
                'ftol': 1e-12,
                'gtol': 1e-8,
                'iprint': 0
            }
        )
        
        x = result.x
        val, _ = compute_smoothed_obj_grad(x, tau, n)
        
        if val < best_val:
            best_val = val
            best_x = x.copy()
        
        if verbose:
            a_curr = np.exp(x - np.max(x))
            a_curr /= a_curr.sum()
            true_val = 2.0 * n * np.max(fast_convolve(a_curr))
            print(f"    tau={tau:.6f}, smooth_val={val:.8f}, true_val={true_val:.8f}, iterations={result.nit}")
        
        # Reduce temperature
        tau *= 0.5
        if tau < 1e-10:
            break
    
    # Return the best a
    x = best_x
    a = np.exp(x - np.max(x))
    a /= a.sum()
    return a

def multiplicative_weight_polish(a_init, time_budget, lr_init=0.01, decay=0.999, top_k=20, temp_init=0.0001, verbose=True):
    """
    Multiplicative weight (mirror descent) polish.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-20)
    a = a / a.sum()
    n = len(a)
    
    c = fast_convolve(a)
    best_val = np.max(c)
    best_a = a.copy()
    
    lr = lr_init
    temp = temp_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_budget:
        it += 1
        c = fast_convolve(a)
        M = np.max(c)
        
        # Top-k peaks with softmax weighting
        if top_k >= len(c):
            peaks = np.arange(len(c))
        else:
            peaks = np.argpartition(c, -top_k)[-top_k:]
        
        pv = c[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Gradient: for peak k, grad[j] = 2*a[|k-j|] (with bounds)
        # Equivalent: grad = sum_i w[i] * (a convolved with delta at peak)
        # Use the FFT-based approach
        # grad[j] = 2 * sum_i w[i] * (a[k_i - j] + a[j - k_i])
        # This is 2 * T where T is the same as before but with w instead of p
        
        # Compute gradient via direct method (simpler for top_k peaks)
        grad = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-14:
                continue
            k = int(peaks[ii])
            # a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je+1] += 2.0 * w[ii] * a[k - js:k - je + 1][::-1]
            # a[j-k] for j in [max(0,k), min(n-1,k+n-1)]
            js2 = max(0, k)
            je2 = min(n - 1, k + n - 1)
            if js2 <= je2:
                grad[js2:je2+1] += 2.0 * w[ii] * a[js2 - k:je2 - k + 1]
        
        # Multiplicative update: a <- a * exp(-lr * grad) / sum
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-25)
        a = a / a.sum()
        
        c = fast_convolve(a)
        val = np.max(c)
        
        if val < best_val - 1e-15:
            best_val = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= decay
        temp *= 1.002
        
        if no_improve > 200:
```

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_obj_grad(x, tau, n):
    """
    Compute smoothed objective and gradient.
    a = softmax(x), F = 2n * (c_max + tau * log(sum exp((c-c_max)/tau)))
    Returns (value, gradient).
    """
    x_max = np.max(x)
    a = np.exp(x - x_max)
    a_sum = a.sum()
    a = a / a_sum

    c = fast_convolve(a)
    c_max = np.max(c)

    c_scaled = (c - c_max) / tau
    p = np.exp(c_scaled)
    p_sum = p.sum()
    p = p / p_sum

    value = 2.0 * n * (c_max + tau * np.log(p_sum))

    # T[j] = sum_k p[k] * (a[k-j] + a[j-k]) via FFT
    L = 1 << (3 * n - 1 + 1).bit_length()
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    # V = conv(p, a)
    V = np.fft.irfft(P * A, L)
    # r = reversed p
    r = p[::-1]
    R = np.fft.rfft(r, L)
    # U = conv(a, r)
    U = np.fft.irfft(A * R, L)
    idx = 2 * n - 2 - np.arange(n)
    T = V[:n] + U[idx]

    S = np.dot(p, c)

    grad = (2.0 * n / tau) * a * (T - 2.0 * S)

    return value, grad


def smoothed_value(x, tau, n):
    val, _ = compute_smoothed_obj_grad(x, tau, n)
    return val


def smoothed_grad(x, tau, n):
    _, grad = compute_smoothed_obj_grad(x, tau, n)
    return grad


def lbfgs_smoothed(a_init, tau_init, max_outers=12, time_budget=600, verbose=True):
    """L-BFGS with animated temperature on smoothed max objective."""
    n = len(a_init)
    a_safe = np.maximum(a_init, 1e-30)
    x = np.log(a_safe / a_safe.sum())

    tau = tau_init
    t0 = time.monotonic()
    best_val = float('inf')
    best_x = x.copy()

    for outer in range(max_outers):
        elapsed = time.monotonic() - t0
        if elapsed > time_budget - 15:
            break

        remaining = time_budget - elapsed
        maxfun = max(100, int(remaining * 20))

        result = minimize(
            fun=smoothed_value,
            x0=x,
            jac=smoothed_grad,
            args=(tau, n),
            method='L-BFGS-B',
            options={
                'maxiter': maxfun,
                'maxfun': maxfun,
                'ftol': 1e-14,
                'gtol': 1e-10,
                'iprint': 0
            }
        )

        x = result.x
        val, _ = compute_smoothed_obj_grad(x, tau, n)

        if val < best_val:
            best_val = val
            best_x = x.copy()

        if verbose:
            a_curr = np.exp(x - np.max(x))
            a_curr /= a_curr.sum()
            true_val = 2.0 * n * np.max(fast_convolve(a_curr))
            print(f"    outer={outer}, tau={tau:.6e}, smooth={val:.8f}, true={true_val:.8f}, nit={result.nit}")

        tau *= 0.45
        if tau < 1e-12:
            break

    a = np.exp(best_x - np.max(best_x))
    a /= a.sum()
    return a


def multiplicative_weight_polish(a_init, time_budget, lr_init=0.01, decay=0.999, top_k=20, temp_init=0.0001, verbose=False):
    """Multiplicative weight (mirror descent) polish with peak-focused updates."""
    a = np.maximum(a_init.astype(np.float64), 1e-20)
    a = a / a.sum()
    n = len(a)

    c = fast_convolve(a)
    best_val = np.max(c)
    best_a = a.copy()

    lr = lr_init
    temp = temp_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_budget:
        it += 1
        c = fast_convolve(a)
        M = np.max(c)

        if top_k >= len(c):
            peaks = np.arange(len(c))
        else:
            peaks = np.argpartition(c, -top_k)[-top_k:]

        pv = c[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        grad = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-14:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += 2.0 * w[ii] * a[k - js:k - je + 1][::-1]
            js2 = max(0, k)
            je2 = min(n - 1, k + n - 1)
            if js2 <= je2:
                grad[js2:je2 + 1] += 2.0 * w[ii] * a[js2 - k:je2 - k + 1]

        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-25)
        a = a / a.sum()

        c = fast_convolve(a)
        val = np.max(c)

        if val < best_val - 1e-15:
            best_val = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= decay
        temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng((int(time.monotonic() * 100000) + it) % 1000000)
            a = best_a * np.exp(rng.standard_normal(n) * 0.005)
            a = a / a.sum()
            lr = max(lr_init * 0.6, 0.002)
            temp = temp_init * 3.0
            no_improve = 0

    return best_a


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(w / w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::1] = 0.5
    inits.append(alt / alt.sum())

    per = np.ones(n)
    per[::3] = 1.5
    per[1::3] = 0.5
    inits.append(per / per.sum())

    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    sparse = np.zeros(n)
    sparse[::2] = 1.0
    inits.append(sparse / sparse.sum())

    block = max(1, n // 8)
    two_scale = np.ones(n)
    for i in range(0, n, 2 * block):
        two_scale[i:i + block] = 2.0
    inits.append(two_scale / two_scale.sum())

    fourier = np.abs(np.sum(np.exp(2j * np.pi * k * x / n) for k in range(max(1, n // 10)))) + 0.1
    inits.append(fourier / fourier.sum())

    return inits


class BestTracker:
    def __init__(self):
        self.best_seq = None
        self.best_score = float('inf')

    def update(self, a, label=""):
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
