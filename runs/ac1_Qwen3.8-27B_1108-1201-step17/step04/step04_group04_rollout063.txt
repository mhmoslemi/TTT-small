"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP with bisection
step sizing, exact subgradient polishing, and multi-length exploration.
Key improvements: (1) bisection step sizing replaces fixed candidates,
(2) active-set LP reduces constraint count, (3) exact subgradient for
final polish, (4) multiple length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute the objective for a sequence (assumes sum > 0)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def active_set_lp(a, M, top_k=10, C=None):
    """
    Solve active-set LP: max sum(g) s.t. (a*g)[k] <= M for top_k peaks, g >= 0.
    Returns (g, success).
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    
    # Get current convolution to identify active set
    conv_a = C @ a  # This gives a*a
    # Get the top_k lag indices
    peak_indices = np.argpartition(conv_a, -top_k)[-top_k:]
    
    # Build reduced constraint matrix
    A_red = C[peak_indices, :]  # shape (top_k, n)
    b_red = np.full(top_k, M)
    
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_vec, A_ub=A_red, b_ub=b_red, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, True
    except Exception:
        pass
    return a.copy(), False

    Uses bisection on the step size.
    Returns (best_a, best_obj, found).
    """
    best_a = a.copy()
    best_obj = 2.0 * n * M
    base_sum = a.sum()
    
    # Check if direction improves at t=1
    a_test = a + d
    a_test = np.maximum(a_test, 0.0)

        return best_a, best_obj, False
    a_test = a_test / a_test.sum()
    conv_test = fast_convolve(a_test)
    obj_test = 2.0 * n * conv_test.max()
    
    if obj_test >= best_obj - 1e-14:
        # Even t=1 doesn't help; try smaller
        lo, hi = 0.0, 1.0
    else:
        # t=1 helps; try to go beyond? No, we're constrained to [0,1] for convexity
        # Actually, we can go beyond 1 since we normalize
        # But let's keep it simple: binary search in [0, 1]
        lo, hi = 0.0, 1.0

        s = a_mid.sum()

            hi = mid
            continue
        a_mid = a_mid / s
        conv_mid = fast_convolve(a_mid)

            best_a = a_mid.copy()
            lo = mid
        else:

def gradient_optimize(a0, time_limit, lr=0.01, momentum=0.9, top_k=20, temp=0.005):
    """Nesterov-accelerated gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_temp = temp
    cur_lr = lr
    cur_mom = momentum
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + cur_mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        conv_y = fast_convolve(y)
        
        # Soft gradient
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j = k - np.arange(n)
            mask = (j >= 0) & (j < n)
            grad[mask] += w[idx] * 2.0 * y[j[mask]]
        
        # Update
        v = cur_mom * v - cur_lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            break
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive params
        cur_lr *= 0.999
        cur_temp *= 1.005
        cur_mom = min(0.97, cur_mom + 0.0003)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def exact_gradient_polish(a0, time_limit, step_sizes=None):
    """
    Polish using exact subgradient at the peak.
    At the peak lag k*, the subgradient is 2*a[k*-i] (truncated).
    We take small steps in the negative subgradient direction.
    """
    if step_sizes is None:
        step_sizes = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
    
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), 2.0 * len(a0) * fast_convolve(a0).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Find peak
        peak_idx = int(np.argmax(conv))
        
        # Compute exact subgradient
        j = peak_idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = np.zeros(n)
        grad[mask] = 2.0 * a[j[mask]]
        
        # Try different step sizes
        improved = False
        for step in step_sizes:
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-14:
                a = a_try
                conv = conv_try
                best_obj = obj_try
                best_a = a.copy()
                no_improve = 0
                improved = True
                break
            elif obj_try < 2.0 * n * conv.max() - 1e-14:
                # At least reduces current peak, take it
                a = a_try
                conv = conv_try
                no_improve = 0
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Bohman-like: product of two triangles
    tri1 = np.maximum(0, 1 - np.abs(x - (n-1)/2) / ((n-1)/2 + 1))
    inits['bohman'] = (tri1 * tri1)
    s = inits['bohman'].sum()
    if s > 1e-15:
        inits['bohman'] = inits['bohman'] / s
    else:
        inits['bohman'] = np.ones(n) / n
    
    # Raised cosine
    inits['raised_cos'] = (0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01)
    inits['raised_cos'] = inits['raised_cos'] / inits['raised_cos'].sum()
    
    # Sine
    inits['sine'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine'] = inits['sine'] / inits['sine'].sum()
    
    # Bimodal
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                              np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
                # Also try with noise
                noise = rng.standard_normal(n) * 0.01
                pn = pp + noise
                pn = np.maximum(pn, 0)
                pns = pn.sum()
                if pns > 1e-10:
                    inits['parent_noisy'] = pn / pns
    
    return inits

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def obj_from_conv(n, conv, sum_a):
    """Compute objective given n, convolution, and sum."""
    return 2.0 * n * conv.max() / (sum_a * sum_a)


def active_set_lp(a, M, top_k, C):
    """
    Solve active-set LP: max sum(g) s.t. (row_k of C) @ g <= M for top_k peaks.
    Returns (g, success).
    """
    n = len(a)
    conv_a = C @ a
    # Get peak indices (top_k largest values)
    peak_indices = np.argpartition(conv_a, -top_k)[-top_k:]
    
    A_ub = C[peak_indices, :]
    b_ub = np.full(top_k, M)
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': 30})
        if result.success:
            g = result.x
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return a.copy(), False


def bisection_step(a, d, n, max_iters=12):
    """
    Find best step t in (0, 1.5] via bisection to minimize objective.
    Returns (best_a, best_obj, found_improvement).
    """
    base_sum = a.sum()
    base_conv = fast_convolve(a)
    base_obj = 2.0 * n * base_conv.max()
    best_a = a.copy()
    best_obj = base_obj
    
    # Binary search for best t
    lo = 0.0
    hi = 1.0
    
    for _ in range(max_iters):
        mid = (lo + hi) / 2.0
        if mid < 1e-10:
            break
        a_mid = a + mid * d
        a_mid = np.maximum(a_mid, 0.0)
        s_mid = a_mid.sum()
        if s_mid < 1e-15:
            hi = mid
            continue
        a_mid = a_mid / s_mid
        conv_mid = fast_convolve(a_mid)
        obj_mid = 2.0 * n * conv_mid.max()
        
        if obj_mid < best_obj - 1e-14:
            best_obj = obj_mid
            best_a = a_mid.copy()
            lo = mid
        else:
            hi = mid
    
    if best_obj < base_obj - 1e-14:
        return best_a, best_obj, True
    return a.copy(), base_obj, False


def lp_optimize_active(a0, time_limit, top_k_init=5, top_k_max=30, verbose=False):
    """
    LP-based optimization with active set and bisection step sizing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    C = build_conv_matrix(a)
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    top_k = top_k_init
    perturb_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Periodically rebuild C
        if iteration % 20 == 0:
            C = build_conv_matrix(a)
        
        # Solve active-set LP
        M_target = M * (1.0 + 1e-4)
        g, success = active_set_lp(a, M_target, top_k, C)
        
        if not success:
            no_improve += 5
        else:
            d = g - a
            if np.linalg.norm(d) < 1e-12:
                no_improve += 1
            else:
                a_new, obj_new, improved = bisection_step(a, d, n)
                
                if improved:
                    a = a_new
                    conv = fast_convolve(a)
                    M = conv.max()
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        iteration += 1
        
        if verbose and iteration % 50 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, "
                  f"top_k={top_k}, t={time.monotonic()-t0:.1f}s")
        
        # Adaptive top_k
        if no_improve > 15 and top_k < top_k_max:
            top_k = min(top_k_max, top_k + 2)
        if no_improve > 40 and top_k > top_k_init:
            top_k = max(top_k_init, top_k - 3)
        
        # Stagnation: perturb and restart
        if no_improve >= 40:
            rng = np.random.default_rng(iteration * 17 + perturb_count * 3)
            scale = 0.001 + 0.002 * min(perturb_count, 8)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            C = build_conv_matrix(a)
            no_improve = 0
            perturb_count += 1
            
            if perturb_count > 25:
                break
    
    return best_a, best_obj


def gradient_optimize(a0, time_limit, lr=0.012, momentum=0.92, top_k=30, temp=0.003):
    """Nesterov-accelerated gradient descent with soft-max peak selection."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_temp = temp
    cur_lr = lr
    cur_mom = momentum
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + cur_mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        conv_y = fast_convolve(y)
        
        # Soft-max gradient
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j = k - np.arange(n)
            mask = (j >= 0) & (j < n)
            grad[mask] += w[idx] * 2.0 * y[j[mask]]
        
        # Update
        v = cur_mom * v - cur_lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            break
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        cur_temp *= 1.004
        cur_mom = min(0.97, cur_mom + 0.0002)
        
        if no_improve >= 250:
            break
    
    return best_a, best_obj


def exact_subgradient_polish(a0, time_limit, max_restarts=5):
    """
    Polish using exact subgradient at the argmin peak.
    Cycles through step sizes and uses restarts.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), 2.0 * len(a0) * fast_convolve(a0).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step_sizes = [0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
    restart_count = 0
    
    while time.monotonic() - t0 < time_limit:
        no_improve_local = 0
        
        for _ in range(500):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            peak_idx = int(np.argmax(conv))
            
            # Exact subgradient: 2 * a[peak_idx - i] for valid i
            j = peak_idx - np.arange(n)
            mask = (j >= 0) & (j < n)
            grad = np.zeros(n)
            grad[mask] = 2.0 * a[j[mask]]
            
            improved = False
            for step in step_sizes:
                a_try = a - step * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                
                if obj_try < best_obj - 1e-14:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve_local = 0
                    improved = True
                    break
                elif obj_try < 2.0 * n * conv.max() * 0.999:
                    # Reduces current peak even if not global best
                    a = a_try
                    conv = conv_try
                    no_improve_local = 0
                    improved = True
                    break
            
            if not improved:
                no_improve_local += 1
                if no_improve_local > 50:
                    break
        
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Restart with perturbation if stuck
        restart_count += 1
        if restart_count > max_restarts * 3:
            break
        if no_improve_local > 30:
            rng = np.random.default_rng(restart_count * 41)
            scale = 0.0005 * restart_count
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                break
            conv = fast_convolve(a)
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Bohman-like (product of triangles)
    mid = (n - 1) / 2.0
    tri = np.maximum(0.0, 1.0 - np.abs(x - mid) / (mid + 0.5))
    b = tri * tri
    bs = b.sum()
    inits['bohman'] = b / bs if bs > 1e-15 else np.ones(n) / n
    
    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n)) + 0.01
    inits['raised_cos'] = rc / rc.sum()
    
    # Sine squared
    ss = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine'] = ss / ss.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
            # Noisy version
            noise = rng.standard_normal(n) * 0.008
            pn = inits.get('parent', inits.get('parent_interp', np.ones(n)/n)) + noise
            pn = np.maximum(pn, 0)
            pns = pn.sum()
            if pns > 1e-10:
                inits['parent_noisy'] = pn / pns
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: hybrid active-set LP + gradient + exact subgradient search.
    Multi-length exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            pscore = float(evaluate_sequence(p.tolist()))
            if np.isfinite(pscore):
                best_score = pscore
                best_sequence = p.copy()
                print(f"Parent: n={len(p)}, score={pscore:.6f}")
    except Exception as e:
        print(f"Parent load issue: {e}")
    
    # =========================================================
    # Phase 1: Quick gradient scan (100s)
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t1 = time.monotonic()
    
    lengths = [768, 512, 1024]
    scan_results = []
    
    for n in lengths:
        if time.monotonic() > deadline - 1200:
            break
        inits = generate_initializations(n, parent, rng)
        
        priority = ['parent', 'parent_interp', 'parent_noisy', 'bohman', 'uniform', 'sine']
        init_names = [nm for nm in priority if nm in inits][:3]
        
        t_per = max(10.0, 90.0 / (len(lengths) * 3))
        
        for name in init_names:
            if time.monotonic() > deadline - 1100:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 1000)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            a_opt, obj = gradient_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  n={n} {name}: {score:.6f}")
    
    print(f"Phase 1: {time.monotonic()-t1:.0f}s, best={best_score:.6f}")
    
    # =========================================================
    # Phase 2: Deep LP optimization (750s)
    # =========================================================
    print("=== Phase 2: Active-Set LP ===")
    t2 = time.monotonic()
    
    candidates = []
    if best_sequence is not None:
        candidates.append((len(best_sequence), best_sequence.copy()))
    
    seen_len = set(len(best_sequence)) if best_sequence is not None else set()
    for sc, n, nm, ao in sorted(scan_results):
        if n not in seen_len:
            candidates.append((n, ao.copy()))
            seen_len.add(n)
        if len(candidates) >= 3:
            break
    
    remaining = deadline - time.monotonic() - 500.0
    t_per_lp = max(60.0, remaining * 0.75 / max(1, len(candidates)))
    
    for i, (n, a0) in enumerate(candidates):
        if time.monotonic() > deadline - 450:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 400)
        if t_alloc < 30:
            break
        
        print(f"  LP {i} (n={n}): {t_alloc:.0f}s")

                                         top_k_init=5, top_k_max=25, verbose=True)

    return [float(x) for x in best_sequence]
