    return np.fft.irfft(A * A, fft_size)[:size]

def exp_gradient_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                          temp_decay=0.9997, eta_decay=0.99995,
                          top_k_frac=0.25, verbose=False):
    """Exponentiated gradient / multiplicative weights on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax-weighted subgradient
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
            peek = c[indices]
        else:
            indices = np.arange(len(c))
            peek = c.copy()
        
        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
        
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        
        if it % 1000 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
        
        eta *= eta_decay
        temp = max(temp * temp_decay, 1e-6)
        
        if no_improve > 5000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.5
            temp = max(temp, temp0 * 0.1)
            avg_a = a.copy()
            count = 0
    
    return best_a.copy(), best_obj

def lp_direction(a, M=None):
    """Solve LP to find best direction g."""
    n = len(a)
    if M is None:
        c = fast_convolve(a)
        M = c.max()
    
    # Primal: max sum(g) s.t. A@g <= M, g >= 0
    # A[k, i] = a[k-i] if 0 <= k-i < n
    num_constraints = 2 * n - 1
    c_vec = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix (sparse structure)
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = a[k - j]
    
    b_ub = np.full(num_constraints, M)
    
    result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, 
                     bounds=[(0, None)] * n, method='highs')
    
    if result.success:
        g = result.x
        gsum = g.sum()
        if gsum > 1e-10:
            return g / gsum  # normalized direction
    return None

def lp_refine(a0, time_limit, max_iters=5000, verbose=False):
    """Refine using LP-based direction finding."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    M = best_obj
    
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP
        direction = lp_direction(a, M)
        if direction is None:
            break
        
        # Line search
        improved = False
        for t in [1.0, 0.5, 0.3, 0.1, 0.05, 0.01]:
            a_new = (1 - t) * a + t * direction
            if a_new.min() < -1e-15:
                continue
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = fast_convolve(a_new)
            obj_new = c_new.max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                M = c_new.max()
                c = c_new
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            # Try to restart with perturbation
            if no_improve > 50:
                rng_local = np.random.default_rng(it * 17 + 3)
                a = best_a + rng_local.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                ps = a.sum()
                if ps > 1e-10:
                    a = a / ps
                    c = fast_convolve(a)
                    best_obj = c.max()
                    M = best_obj
                no_improve = 0
            if no_improve > 200:
                break
    
    return best_a.copy(), best_obj

def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate-wise optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Pick a random coordinate
        i = rng.integers(0, n)
        old_val = a[i]
        
        # Try several values
        candidates = [old_val * 0.5, old_val * 0.75, old_val * 0.9,
                     old_val * 1.1, old_val * 1.25, old_val * 1.5,
                     old_val + 0.001, old_val - 0.001, 0.0, 0.005]
        candidates = [max(0, x) for x in candidates if x >= 0]
        
        improved = False
        for new_val in candidates:
            a[i] = new_val
            s2 = a.sum()
            if s2 < 1e-15:
                a[i] = old_val
                continue
            a_norm = a / s2
            c2 = fast_convolve(a_norm)
            obj2 = c2.max()
            if obj2 < best_obj - 1e-14:
                best_obj = obj2
                best_a = a_norm.copy()
                a = a_norm.copy()
                c = c2
                improved = True
                break
            else:
                a[i] = old_val
        
        if not improved:
            no_improve += 1
            if no_improve > 10000:
                break
        else:
            no_improve = 0
    
    return best_a.copy(), best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Exponentiated gradient with adaptive softmax subgradients

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient_topk(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c) using top-k peaks."""
    n = len(a)
    len_c = len(c)
    
    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()
    
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[int(np.argmax(peek))] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * wi * a[k - je:k - js + 1][::-1]
    
    return sg


def exp_gradient_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                          temp_decay=0.9997, eta_decay=0.99995,
                          top_k_frac=0.25, verbose=False):
    """Exponentiated gradient on the simplex with softmax subgradients."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    avg_a = a.copy()
    count = 0
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient_topk(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        
        if it % 2000 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-14:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
        
        eta = max(eta * eta_decay, 1e-6)
        temp = max(temp * temp_decay, 1e-7)
        
        if no_improve > 5000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = max(eta0 * 0.5, 0.1)
            temp = max(temp, temp0 * 0.1)
            avg_a = a.copy()
            count = 0
    
    return best_a.copy(), best_obj


def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = a[k - j]
    
    c_vec = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    
    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs')
        if result.success:
            g = result.x
            gsum = g.sum()
            if gsum > 1e-10:
                return g / gsum
    except Exception:
        pass
    return None


def lp_refine(a0, time_limit, max_iters=5000, verbose=False):
    """Refine using LP-based Frank-Wolfe steps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    M = best_obj
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        
        direction = solve_lp_direction(a, M)
        if direction is None:
            break
        
        # Check if direction is useful (not just a itself)
        dir_diff = np.abs(direction - a).max()
        if dir_diff < 1e-10:
            no_improve += 10
            if no_improve > 100:
                break
            # Perturb to escape
            rng_local = np.random.default_rng(it * 13 + 5)
            a = best_a + rng_local.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
                c = fast_convolve(a)
                M = c.max()
            continue
        
        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            a_new = (1.0 - t) * a + t * direction
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = fast_convolve(a_new)
            obj_new = c_new.max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                c = c_new
                M = obj_new
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Restart with perturbation
                rng_local = np.random.default_rng(it * 17 + 3)
                a = best_a + rng_local.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                ps = a.sum()
                if ps > 1e-10:
                    a = a / ps
                    c = fast_convolve(a)
                    M = c.max()
                    best_obj = M
                no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: obj={best_obj:.10f}, no_improve={no_improve}")
    
    return best_a.copy(), best_obj


def coordinate_descent(a0, time_limit, max_iters=50000, verbose=False):
    """Coordinate-wise optimization with random coordinate selection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    it = 0
    
    factors = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 0.01, 0.05])
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        i = int(rng.integers(0, n))
        old_val = a[i]
        
        improved = False
        for f in factors:
            new_val = old_val * f
            if new_val < 0:
                continue
            a[i] = new_val
            s2 = a.sum()
            if s2 < 1e-15:
                a[i] = old_val
                continue
            a_norm = a / s2
            c2 = fast_convolve(a_norm)
            obj2 = c2.max()
            if obj2 < best_obj - 1e-14:
                best_obj = obj2
                best_a = a_norm.copy()
                a = a_norm.copy()
                c = c2
                improved = True
                break
            else:
                a[i] = old_val
        
        if not improved:
            no_improve += 1
            if no_improve > 20000:
                break
        else:
            no_improve = 0
        
        if verbose and it % 10000 == 0:
            print(f"    CD iter {it}: obj={best_obj:.10f}")
    
    return best_a.copy(), best_obj
