    """Smooth approximation of max using log-sum-exp."""
    x_max = x.max()
    return x_max + np.log(np.sum(np.exp(beta * (x - x_max)))) / beta

import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(theta):
    theta = theta - theta.max()  # for numerical stability
    exp_theta = np.exp(theta)
    return exp_theta / exp_theta.sum()

def obj_from_conv(conv, n):
    return 2.0 * n * conv.max()

def smooth_grad_a(a, beta):
    """Compute smooth gradient of 2n*smooth_max(conv(a,a), beta) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    scaled = beta * (conv - conv_max)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    # g_i = sum_j w_{i+j} * a_j
    g = np.correlate(w, a, mode='valid')
    return 2.0 * n * g, conv, w

    """Compute gradient w.r.t. theta given a = softmax(theta) and g = grad wrt a."""
    ga = np.dot(g, a)
    return a * (g - ga)

    y = theta_init.copy()

"""
Multi-phase optimization using smooth gradient on softmax parameterization.
Phase 1: Quick exploration across lengths with Adam.
Phase 2: Focused FISTA refinement with adaptive β.
Phase 3: Final polish with tiny steps and LP hybrid.
"""

import numpy as np
import time
from scipy.optimize import linprog

# === Helper functions ===

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax_stable(theta):
    theta = theta - theta.max()
    e = np.exp(theta)
    return e / e.sum()

def compute_smooth_weights(conv, beta):
    conv_max = conv.max()
    scaled = beta * (conv - conv_max)
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    return w

def compute_grad_a(a, w):
    """g_i = 2n * sum_j w_{i+j} * a_j"""
    n = len(a)
    g = np.correlate(w, a, mode='valid')
    return 2.0 * n * g

def compute_grad_theta(a, grad_a):
    ga = np.dot(grad_a, a)
    return a * (grad_a - ga)

def evaluate_obj(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def adam_optimize(a_init, time_limit, beta_init=200.0, beta_final=2000.0, 
                  lr_init=0.05, lr_final=0.001):
    """Adam optimization on softmax parameterization with adaptive beta."""
    n = len(a_init)
    # Initialize theta from a_init
    a_init = np.maximum(a_init, 1e-10)
    theta = np.log(a_init / a_init.sum())
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a_init.copy() / a_init.sum()
    best_conv = fast_convolve(best_a)
    best_obj = 2.0 * n * best_conv.max()
    
    t0 = time.monotonic()
    step = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        # Adaptive beta and lr
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_init + (beta_final - beta_init) * progress
        lr = lr_init * (lr_final / lr_init) ** progress
        
        a = softmax_stable(theta)
        conv = fast_convolve(a)
        w = compute_smooth_weights(conv, beta)
        grad_a = compute_grad_a(a, w)
        grad = compute_grad_theta(a, grad_a)
        
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        theta -= theta.mean()
        
        if step % 5000 == 0:
            print(f"    Adam step {step}, obj={best_obj:.8f}, beta={beta:.0f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_obj

def fista_optimize(a_init, time_limit, beta=500.0, lr=0.01):

    n = len(a_init)

    theta = np.log(a_init / a_init.sum())
    y = theta.copy()

    best_a = a_init.copy() / a_init.sum()
    best_conv = fast_convolve(best_a)
    best_obj = 2.0 * n * best_conv.max()
    
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        a_y = softmax_stable(y)
        conv = fast_convolve(a_y)
        w = compute_smooth_weights(conv, beta)
        grad_a = compute_grad_a(a_y, w)
        grad = compute_grad_theta(a_y, grad_a)

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_y.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        y_new -= y_new.mean()

        t_k = t_new
        step += 1

"""
Multi-phase smooth gradient optimization on softmax parameterization for minimizing
2*n*max(auto_conv(a))/(sum(a))^2. Uses Adam/FISTA with adaptive beta scheduling,
top-k peak focusing, and LP polishing. Key improvements: 100x faster per-iteration
than LP (FFT-based gradients), enabling exploration of much larger solution landscapes.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_stable(theta):
    """Numerically stable softmax."""
    theta = theta - theta.max()
    e = np.exp(theta)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def compute_smooth_weights(conv, beta):
    """Compute softmax weights over convolution for smooth max."""
    conv_max = conv.max()
    scaled = beta * (conv - conv_max)
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def compute_grad_a(a, w, factor):
    """Compute gradient of smooth objective w.r.t. a. factor = 4*n."""
    g = np.correlate(w, a, mode='valid')
    return factor * g


def compute_grad_theta(a, grad_a):
    """Chain rule: gradient w.r.t. theta from gradient w.r.t. a."""
    ga = np.dot(grad_a, a)
    return a * (grad_a - ga)


def eval_obj(a):
    """Compute true objective 2n*max(conv) for normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def adam_optimize(a_init, time_limit, beta_init=100.0, beta_final=2000.0,
                  lr_init=0.05, lr_final=0.002, top_k=0, restart_interval=20000):
    """Adam optimization on softmax parameterization with adaptive beta."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    theta = np.log(a_init / a_init.sum())
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a_init.copy() / a_init.sum()
    best_obj_val = eval_obj(best_a)[0]
    
    t0 = time.monotonic()
    step = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_init + (beta_final - beta_init) * progress
        lr = lr_init * (lr_final / lr_init) ** progress
        
        a = softmax_stable(theta)
        conv = fast_convolve(a)
        
        if top_k > 0 and top_k < len(conv):
            top_idx = np.argpartition(conv, -top_k)[-top_k:]
            conv_top = conv[top_idx]
            conv_max = conv_top.max()
            scaled = beta * (conv_top - conv_max)
            scaled -= scaled.max()
            w_top = np.exp(scaled)
            w_top /= w_top.sum()
            w = np.zeros(len(conv))
            w[top_idx] = w_top
        else:
            w = compute_smooth_weights(conv, beta)
        
        factor = 4.0 * n
        grad_a = compute_grad_a(a, w, factor)
        grad = compute_grad_theta(a, grad_a)
        
        obj = 2.0 * n * conv.max()
        if obj < best_obj_val - 1e-14:
            best_obj_val = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adam update
        step += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** step)
        v_hat = v / (1 - b2 ** step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        theta -= theta.mean()
        
        # Restart if stuck
        if no_improve >= restart_interval:
            rng = np.random.default_rng(step)
            scale = 0.001 * min(5, no_improve // restart_interval)
            noise = rng.standard_normal(n) * scale
            theta = np.log(np.maximum(best_a * (1 + noise), 1e-15))
            theta -= theta.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam step {step}, obj={best_obj_val:.8f}, beta={beta:.0f}, t={elapsed:.1f}s")
    
    return best_a, best_obj_val


def fista_optimize(a_init, time_limit, beta=500.0, lr=0.008, top_k=0):

    n = len(a_init)

    theta = np.log(a_init / a_init.sum())
    y = theta.copy()

    best_a = a_init.copy() / a_init.sum()
    best_obj_val = eval_obj(best_a)[0]
    
    t0 = time.monotonic()
    step = 0
    factor = 4.0 * n
    
    while time.monotonic() - t0 < time_limit:
        a_y = softmax_stable(y)
        conv = fast_convolve(a_y)
        
        if top_k > 0 and top_k < len(conv):
            top_idx = np.argpartition(conv, -top_k)[-top_k:]
            conv_top = conv[top_idx]
            conv_max = conv_top.max()
            scaled = beta * (conv_top - conv_max)
            scaled -= scaled.max()
            w_top = np.exp(scaled)
            w_top /= w_top.sum()
            w = np.zeros(len(conv))
            w[top_idx] = w_top
        else:
            w = compute_smooth_weights(conv, beta)
        
        grad_a = compute_grad_a(a_y, w, factor)
        grad = compute_grad_theta(a_y, grad_a)
        
        obj = 2.0 * n * conv.max()
        if obj < best_obj_val - 1e-14:
            best_obj_val = obj
            best_a = a_y.copy()
        
        # FISTA step
        theta_new = y - lr * grad
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = theta_new + ((t_k - 1.0) / t_new) * (theta_new - y)
        
        theta_new -= theta_new.mean()
        y_new -= y_new.mean()
        
        y = y_new
        t_k = t_new
        step += 1
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA step {step}, obj={best_obj_val:.8f}, t={elapsed:.1f}s")
    
    return best_a, best_obj_val


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def lp_polish(a_init, time_limit):
    """LP-based polishing for final improvements."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()

        bounds = [(0.0, None)] * n

        except Exception:
            break

            tightness = min(1.003, tightness * 1.01)
            if tightness > 1.01:
                break
            continue
        
        g = result.x
        g_sum = g.sum()
        
        if g_sum <= 1.0 + 1e-7:
            tightness = min(1.003, tightness * 1.005)
            no_improve += 1
            if no_improve > 20 or tightness > 1.01:
                break
        else:
            d = g - a
            best_t = 0.0
            best_o = best_obj_val
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                a_try = np.maximum(a + t * d, 0.0)
                s = a_try.sum()
                if s < 1e-15:

                a_try = a_try / s
                conv_try = fast_convolve(a_try)
                o_try = 2.0 * n * conv_try.max()
                if o_try < best_o:

                best_a = a.copy()
                no_improve = 0
                tightness = max(0.999, tightness - 0.0002)
            else:
                no_improve += 1
                tightness = min(1.003, tightness * 1.005)
                if no_improve > 20 or tightness > 1.01:
                    break
    
    return best_a, best_obj_val


def resample_seq(seq, new_n):
    """Resample sequence to new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def perturb_sequence(a, rng, scale=0.003):
    """Add structured perturbation."""
    n = len(a)
    noise = rng.standard_normal(n) * scale
    x = np.arange(n)
    smooth = np.sin(np.pi * x / n) * scale * 0.3
    a_new = a + noise + smooth
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    t_start3 = time.monotonic()
    
    if best_state[0] is not None:
        remaining3 = deadline - time.monotonic() - 150.0
        n_best = len(best_state[0])
        
        if remaining3 > 30:
            n_rounds = max(1, int(remaining3 / 60))
            n_rounds = min(n_rounds, 6)
            t_per_round = remaining3 / n_rounds
            
            # Different perturbation strategies
            strategies = [
                ('noise', 0.002),
                ('noise_large', 0.005),
                ('smooth', 0.003),
                ('noise', 0.001),
                ('noise_large', 0.008),
                ('smooth', 0.001),
            ]
            
            for ri in range(n_rounds):
                if time.monotonic() > deadline - 130:
                    break
                t_alloc = min(t_per_round, deadline - time.monotonic() - 120)
                if t_alloc < 10:
                    break
                
                strat = strategies[ri % len(strategies)]
                if strat[0] == 'noise':
                    a_pert = perturb_sequence(best_state[0], rng, scale=strat[1])
                else:
                    n = len(best_state[0])
                    x = np.arange(n)
                    smooth = np.sin(2 * np.pi * x / (n * 3)) * strat[1]
                    a_pert = best_state[0] + smooth
                    a_pert = np.maximum(a_pert, 0.0)
                    a_pert = a_pert / a_pert.sum()
                
                # Alternate between Adam and FISTA
                if ri % 2 == 0:
                    a_opt, obj = adam_optimize(a_pert, t_alloc,
                                               beta_init=1000, beta_final=10000,
                                               lr_init=0.01, lr_final=0.0005,
                                               top_k=15, restart_interval=20000)
                else:
                    a_opt, obj = fista_optimize(a_pert, t_alloc,
                                                beta=3000, lr=0.003, top_k=15)
                
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_state[1]:
                    best_state[1] = score
                    best_state[0] = a_opt.copy()
                    print(f"  P3 round {ri} ({strat[0]}): {score:.8f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_state[1]:.8f}")
    
    # ==================== Phase 4: LP Polish ====================
    print("=== Phase 4: LP Polish ===")
    t_start4 = time.monotonic()
    
    if best_state[0] is not None and time.monotonic() < deadline - 30:
        t_lp = min(120.0, deadline - time.monotonic() - 20)
        if t_lp > 20:
            a_opt, obj = lp_polish(best_state[0], t_lp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_state[1]:
                best_state[1] = score
                best_state[0] = a_opt.copy()
                print(f"  LP polish: {score:.8f}")
            
            # One more Adam round after LP
            if time.monotonic() < deadline - 25:
                t_final = min(40.0, deadline - time.monotonic() - 20)
                if t_final > 10:
                    a_opt, obj = adam_optimize(best_state[0], t_final,
                                               beta_init=5000, beta_final=30000,
                                               lr_init=0.003, lr_final=0.0001,
                                               top_k=5, restart_interval=50000)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_state[1]:
                        best_state[1] = score
                        best_state[0] = a_opt.copy()
                        print(f"  Final Adam: {score:.8f}")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_state[1]:.8f}")
    
    # Fallback
    if best_state[0] is None:
        best_state[0] = np.ones(512) / 512
        best_state[1] = float(evaluate_sequence(best_state[0].tolist()))
    
    # Final check
    final_score = float(evaluate_sequence(best_state[0].tolist()))
    if np.isfinite(final_score):
        best_state[1] = final_score
    
    print(f"\nFINAL: score={best_state[1]:.8f}, n={len(best_state[0])}")
    return [float(x) for x in best_state[0]]
