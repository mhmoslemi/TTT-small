Actually, one more thought: the problem has a specific structure that I can exploit. The convolution is a "blurring" of the product of the sequence with itself. For a smooth sequence, the convolution is also smooth, and the peak is a local property.

For the subgradient method, the key is to target the right peaks. If there's a single dominant peak, we focus on that. If there are multiple peaks of similar height, we need to reduce all of them.

The softmax weighting over peaks (as in the previous code) handles this well. The temperature controls the "focus": low temperature → focus on the single highest peak; high temperature → spread the effort over all peaks.

I'll use an adaptive temperature that starts high (spread) and decreases over time (focus).

Let me also add a "peak detection" step: if the convolution has a very sharp peak (one value much larger than its neighbors), the sequence likely has a "bump" at the corresponding position, and we should try to flatten it.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature subgradient
descent with Nesterov momentum, coordinate descent polishing, and diverse
parameterized initializations. Avoids LP solvers for speed.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=20, temp=0.001):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        # grad of conv[k] wrt a
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        # conv[k] = sum_{j=js}^{je} a[j]*a[k-j]
        # d conv[k]/d a[i] = 2*a[k-i] if k-i in [js,je] and i != k-i
        #                     a[i] if i == k-i (k=2i)
        # Simpler: d conv[k]/d a[i] = 2*a[k-i] for i in [js,je], k-i in [js,je], i != k/2
        #                            = a[i] for i = k/2
        # But since js to je is exactly the range where both j and k-j are valid,
        # we have: for i in [js, je], d conv[k]/d a[i] = 2*a[k-i]
        #   except when k-i == i (i.e., k=2i), where it's a[i].
        
        # Actually the cleanest: d conv[k] / d a[i] = 2 * a[k-i] if 0 <= k-i <= n-1
        #   with the correction that if k-i == i, it's a[i] not 2*a[i].
        # But 0 <= k-i <= n-1 is equivalent to i in [k-(n-1), k] intersect [0, n-1]
        # which is exactly [js, je].
        
        # So for i in [js, je]:
        #   if 2*i == k: grad = a[i]
        #   else: grad = 2*a[k-i]
        
        half_k = k / 2.0
        for i_val in range(js, je + 1):
            if 2 * i_val == k:
                sg[i_val] += w[i] * a[i_val]
            else:
                sg[i_val] += w[i] * 2.0 * a[k - i_val]
    
    return sg
```

```python
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        # 2*a[k-i] for i in idx, with correction at i = k/2
        contrib = 2.0 * a[k - idx]
        # Fix the middle term if k is even
        if k % 2 == 0:
            mid = k // 2
            if js <= mid <= je:
                mid_pos = mid - js
                contrib[mid_pos] = a[mid]  # not 2*a[mid]
        sg[idx] += w[idx_i] * contrib
    
    return sg
```

This is O(N) per peak, vectorized within each peak. Good.

```python
def subgrad_optimize(a0, time_limit, lr=0.005, mom=0.9, top_k=30, 
                     temp_init=0.0005, temp_growth=1.005, restart_every=200):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp_init
    rng = np.random.default_rng(int(abs(hash(a0.tobytes())) % 100000))
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient(a, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_temp *= temp_growth
        if no_improve > 20:
            cur_lr *= 0.95
            no_improve = 0
        
        if it % restart_every == 0 and no_improve > restart_every // 2:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = lr * 0.7
            cur_temp = temp_init * 0.5
            no_improve = 0
    
    return best_a, best_obj
```

```python
def coordinate_descent(a0, time_limit, max_sweeps=100):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            old_val = a[i]
            if old_val < 1e-15:
                # Try a small positive value
                a[i] = 1.0 / (n * 10)
                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = 0.0
                continue
            
            # Find optimal a[i] by golden section search
            # The objective as a function of x = a[i]:
            # f(x) = 2*n*max_conv(x) / (S' + x)^2
            # where S' = sum of other elements
            
            S_prime = s - old_val
            # x = 0: f = 2*n*max(conv without a[i]) / S_prime^2
            # x = old_val: f = 2*n*current_max / s^2 (current objective)
            # x -> large: f -> infinity
            
            # Try a few values
            candidates = [0.0, old_val * 0.1, old_val * 0.3, old_val * 0.5, 
                         old_val * 0.7, old_val, old_val * 1.3, old_val * 1.7]
            
            a[i] = 0.0  # temporarily
            conv_zero = conv - 2.0 * old_val * a  # approximation (not exact)
            # Actually, let's just recompute
            conv_at_zero = fast_convolve(a)  # with a[i] = 0
            max_at_zero = conv_at_zero.max()
            
            best_x = old_val
            best_f = 2.0 * n * conv.max() / (s * s)
            
            for x in candidates:
                a[i] = x
                new_s = S_prime + x
                if new_s < 1e-15:
                    continue
                conv_x = conv_at_zero  # need to recompute
                # Actually conv changes nontrivially, let me just compute
                conv_x = fast_convolve(a)
                f_x = 2.0 * n * conv_x.max() / (new_s * new_s)
                if f_x < best_f:
                    best_f = f_x
                    best_x = x
            
            a[i] = best_x
            s = best_x + S_prime
            
            # Check global improvement
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj
```

Hmm, the coordinate descent as written is O(N^2) per sweep because for each coordinate, we do O(1) FFT convolutions (each O(N log N)). So a sweep is O(N * N log N) = O(N^2 log N). For N=384, that's about 384 * 384 * 9 ≈ 1.3M operations per sweep. That might be a bit slow in Python.

Let me optimize: instead of doing a full FFT for each candidate x, I can compute the change in the convolution analytically.

When we change a[i] from 0 to x:
- conv[k] changes by:
  - For k in [i, i+N-1] intersect [0, 2N-2]:
    - If 2i == k: change = x^2
    - Else: change = 2*x*a[k-i]

So I can precompute the "template" for the change and just add x * template (and x^2 at position 2i).

def coordinate_descent_fast(a0, time_limit, max_sweeps=200):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            S_prime = s - old_val
            
            # Template: contribution to conv[k] from a[i] = 1
            # conv[k] += 2*a[k-i] for k != 2i, conv[2i] += 1
            # (when a[i] goes from 0 to 1)
            # So for a[i] going from 0 to x:
            # conv[k] += 2*x*a[k-i] for k != 2i
            # conv[2i] += x^2
            
            # Let's compute the "base" conv (a[i] = 0)
            a[i] = 0.0
            # We need conv with a[i] = 0
            # conv_base[k] = conv_current[k] - contribution_from_a[i]
            # contribution: 2*old_val*a[k-i] for k != 2i (in range), old_val^2 for k=2i
            
            # Actually, it's easier to just compute conv_base once
            # But that's O(N log N) per coordinate, which defeats the purpose.
            
            # Alternative: compute the change directly.
            # conv_new[k] = conv_old[k] + 2*(new-old)*a[k-i] (for k != 2i)
            #                  + (new^2 - old^2) (for k = 2i)
            # This is O(N) per candidate.
            
            # Let's try a few candidates
            if old_val < 1e-10:
                # Try adding a small value
                for x in [0.5/n, 1.0/n, 2.0/n]:
                    a[i] = x
                    new_s = S_prime + x
                    if new_s < 1e-15:
                        a[i] = 0
                        continue
                    # Compute new conv
                    # conv_new[k] = conv[k] + 2*x*a[k-i] for k in [i, i+n-1], k != 2i
                    # conv_new[2i] = conv[2i] + x^2 (since old was 0)
                    # But wait, conv was computed with a[i] = old_val = ~0.
                    # So conv is approximately the conv with a[i] = 0.
                    # Change: conv_new[k] = conv[k] + 2*x*a[k-i] (k != 2i)
                    #                       + x^2 (k = 2i)
                    # But we need a[k-i] from the "base" sequence (a[i]=0), which is the same
                    # as current a[k-i] (since a[i]=0 doesn't affect a[k-i] for k-i != i).
                    
                    ks = max(0, i)
                    ke = min(2*n-2, i+n-1)
                    k_range = np.arange(ks, ke+1)
                    # Conv values at these positions
                    conv_changed = conv[k_range].copy()
                    for ki in range(len(k_range)):
                        k_val = k_range[ki]
                        if 2*i == k_val:
                            conv_changed[ki] += x*x
                        else:
                            ki_offset = k_val - i
                            if 0 <= ki_offset <= n-1 and ki_offset != i:
                                conv_changed[ki] += 2*x*a[ki_offset]
                            elif 0 <= ki_offset <= n-1:
                                conv_changed[ki] += 2*x*a[ki_offset]
                    
                    # Actually this is getting complicated. Let me just use FFT for simplicity
                    # since N=384 is small.
                    conv_new = fast_convolve(a)
                    f_new = 2.0 * n * conv_new.max() / (new_s * new_s)
                    if f_new < best_obj:
                        best_obj = f_new
                        best_a = a.copy()
                        conv = conv_new
                        s = new_s
                        improved = True
                    else:
                        a[i] = 0.0
                continue
            
            # For non-zero a[i], try reducing it
            for factor in [0.0, 0.3, 0.5, 0.7, 1.0, 1.3]:
                x = old_val * factor
