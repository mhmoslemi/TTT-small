        if w[idx] < 1e-10:
            continue
        k = int(peak_indices[idx])
        weight = 2.0 * w[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # g[i] += weight * a[k-i] for i in [lo, hi]
            # k-i ranges from k-lo to k-hi
            a_seg = a[k - hi:k - lo + 1]
            g[lo:hi + 1] += weight * a_seg
    
    return g


def nesterov_coord_descent(a, lr_init=0.003, momentum=0.9, num_steps=200,
                           top_k=15, temperature=0.002, time_limit=None):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0
    
    while True:
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        if step >= num_steps:
            break
        
        # Nesterov look-ahead
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            break
        y = y / s_y
        
        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, top_k=top_k, temperature=temperature)
        
        v = momentum * v - lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        step += 1
        
        # Adaptive learning rate
        if no_improve % 30 == 0 and no_improve > 0:
            lr *= 0.95
            temperature *= 1.05
            momentum = min(0.98, momentum + 0.002)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def coordinate_descent(a, num_passes=3, time_limit=None):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for p in range(num_passes):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        for i in range(n):
            if time_limit is not None and time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            if old_val < 1e-10:
                continue
            
            # Try a few candidate values for a[i]
            candidates = [0.0, old_val * 0.5, old_val * 0.7, old_val * 1.0,
                         old_val * 1.3, old_val * 1.5]
            
            best_cand = old_val
            best_cand_obj = fast_convolve(a).max()
            
            for cv in candidates:
                a[i] = cv
                s_tmp = a.sum()
                if s_tmp < 1e-15:
                    a[i] = old_val
                    continue
                conv_tmp = fast_convolve(a)
                obj_tmp = conv_tmp.max() / (s_tmp * s_tmp)
                if obj_tmp < best_cand_obj / (s * s) - 1e-15:
                    best_cand_obj = obj_tmp * s_tmp * s_tmp
                    best_cand = cv
            
            a[i] = best_cand
            s = a.sum()
            if s > 1e-15:
                a = a / s
            
            # Check improvement
            obj_now = 2.0 * n * fast_convolve(a).max()
            if obj_now < best_obj - 1e-13:
                best_obj = obj_now
                best_a = a.copy()
    
    return best_a, best_obj


def lp_optimize_improved(a0, time_limit, top_k_lp=20, verbose=False):
    """LP-based optimization with adaptive top-k constraint selection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    tightness = 1.0
    
    # Adaptive top-k: start with fewer constraints, expand as we converge
    current_top_k = min(top_k_lp, int(0.05 * (2 * n - 1)))
    
    while time.monotonic() - t0 < time_limit:
        # Build constraints using top-k peaks
        A_ub, top_indices = build_conv_matrix_top(a, current_top_k)
        
        # Solve LP with slightly tightened bound
        M_lp = M * tightness
        b_ub = np.full(len(top_indices), M_lp)
        
        g, g_sum = solve_lp_fast(a, M_lp, A_ub, b_ub)
        
        if g_sum <= 1.0 + 1e-7:
            # No improvement possible - relax
            tightness = min(1.01, tightness * 1.003)
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-12:
                no_improve += 1
            else:
                # Line search
                best_t = 0.0
                best_t_obj = best_obj
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = np.maximum(a + t * d, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    obj_try = 2.0 * n * fast_convolve(a_try).max()
                    if obj_try < best_t_obj - 1e-14:
                        best_t_obj = obj_try
                        best_t = t
                
                if best_t > 0 and best_t_obj < best_obj - 1e-14:
                    a = np.maximum(a + best_t * d, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    tightness = max(0.997, tightness - 0.0003)
                    
                    # Expand top-k slightly as we converge
                    if best_obj < 1.52:
                        current_top_k = min(top_k_lp, current_top_k + 1)
                else:
                    no_improve += 1
        
        iteration += 1
        
        if verbose and iteration % 30 == 0:
            print(f"    LP it={iteration}, obj={best_obj:.8f}, "
                  f"t={time.monotonic()-t0:.1f}s, top_k={current_top_k}")
        
        if no_improve >= 50:
            # Perturb with structured noise
            rng = np.random.default_rng(iteration * 31 + 17)
            scale = 0.002 + 0.001 * min(no_improve // 50, 10)
            
            # Use frequency-domain perturbation
            A_f = np.fft.rfft(a, 2 * n)
            noise_f = rng.standard_normal(len(A_f)) * scale * 0.5
            noise_f[0] = 0  # Keep DC fixed
            A_f += noise_f
            a_pert = np.fft.irfft(A_f, 2 * n)[:n]
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert / s_pert
            else:
                a = best_a.copy()
            
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            tightness = 1.002
            current_top_k = min(top_k_lp, max(5, current_top_k - 3))
            
            if no_improve > 200:
                break
    
    return best_a, best_obj


def multi_scale_optimize(n_target, time_limit, parent_seq=None):
    """Optimize at small scale, then upscale and refine."""
    # Start at smaller scale
    n_small = max(32, n_target // 8)
    n_mid = max(64, n_target // 4)
    
    t0 = time.monotonic()
    
    # Phase A: Optimize at small scale
    if parent_seq is not None:
        # Downsample parent
        x_old = np.linspace(0, 1, len(parent_seq))
        x_new = np.linspace(0, 1, n_small)
        a_small = np.interp(x_new, x_old, parent_seq)
        a_small = np.maximum(a_small, 0.0)
        s = a_small.sum()
        if s > 1e-15:
            a_small = a_small / s
    else:
        # Chirp-like initialization
        x = np.arange(n_small, dtype=np.float64)
        a_small = 1.0 + 0.3 * np.cos(2.0 * np.pi * x * (x + 1) / (2.0 * n_small))
        a_small = np.maximum(a_small, 0.0)
        a_small = a_small / a_small.sum()
    
    t_alloc_small = min(30.0, time_limit * 0.15)
    a_small_opt, _ = lp_optimize_improved(a_small, time_limit=t_alloc_small, top_k_lp=8)
    
    # Upscale to mid
    x_old = np.linspace(0, 1, len(a_small_opt))
    x_mid = np.linspace(0, 1, n_mid)
    a_mid = np.interp(x_mid, x_old, a_small_opt)
    a_mid = np.maximum(a_mid, 0.0)
    s = a_mid.sum()
    if s > 1e-15:
        a_mid = a_mid / s
    
    t_alloc_mid = min(60.0, time_limit * 0.25)
    a_mid_opt, _ = lp_optimize_improved(a_mid, time_limit=t_alloc_mid, top_k_lp=15)
    
    # Upscale to target
    x_old = np.linspace(0, 1, len(a_mid_opt))
    x_target = np.linspace(0, 1, n_target)
    a_target = np.interp(x_target, x_old, a_mid_opt)
    a_target = np.maximum(a_target, 0.0)
    s = a_target.sum()
    if s > 1e-15:
        a_target = a_target / s
    
    print(f"  Multi-scale: small={fast_convolve(a_small_opt).max()*2*len(a_small_opt):.6f}, "
          f"mid={fast_convolve(a_mid_opt).max()*2*len(a_mid_opt):.6f}, "
          f"target_init={fast_convolve(a_target).max()*2*len(a_target):.6f} "
          f"({time.monotonic()-t0:.1f}s)")
    
    return a_target


def generate_chirp_init(n, freq_mod=1.0):
    """Generate chirp-like initialization with good autocorrelation properties."""
    x = np.arange(n, dtype=np.float64)
    # Quadratic phase chirp
    phase = 2.0 * np.pi * freq_mod * x * x / (2.0 * n)
    a = 1.0 + 0.2 * np.cos(phase)
    a = np.maximum(a, 0.1)
    return a / a.sum()


def generate_oscillation_init(n, num_lobes=3):
    """Generate multi-lobed oscillatory initialization."""
    x = np.arange(n, dtype=np.float64)
    t = x / n
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * num_lobes * t)
    a = np.maximum(a, 0.1)
    return a / a.sum()
