import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient: 2 * cross_correlation(weights, f) of length n."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - j] for j in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 25})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def lse_flat_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0, gamma=0.5):
    """
    LSE optimization with flatness penalty.
    Objective: LSE(conv, beta) + gamma * variance_of_top_peaks
    This encourages not just a small max but also a flat convolution profile.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(8, n // 16)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Standard LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Add flatness penalty: weight the top-k peaks more
        if it % 3 == 0:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
            peak_mean = peak_vals.mean()
            # Extra penalty for peaks above mean
            for k in peaks:
                if conv[k] > peak_mean:
                    w[k] += gamma * (conv[k] - peak_mean) * beta
            w = w / w.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 1.4, beta_end)
                momentum = 0.88
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if it % 800 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-flat it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.5, n_restarts=5):
    """
    Multiplicative weight updates (exponentiated gradient / mirror descent with KL divergence).
    Update: a_new[i] ∝ a[i] * exp(-lr * grad[i])
    This is better suited for the simplex than Euclidean projection.
    """
    f = np.maximum(f0.astype(np.float64), 1e-30).astype(np.float64)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta = 50.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # LSE weights for smooth gradient
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)

        # Multiplicative update
        log_f = np.log(f)
        log_f -= lr * grad / n
        # Clip for numerical stability
        log_f = np.clip(log_f, -50.0, 50.0)
        f_new = np.exp(log_f - log_f.max())
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / f_new_sum
        f_new = np.maximum(f_new, 0.0)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
            if no_improve > 200:
                # Restart with perturbation
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                lr = lr_init
                no_improve = 0

        # Slowly increase beta
        if beta < 2000.0:
            beta *= 1.005

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_f, best_obj


def polyak_subgrad_optimize(f0, time_limit, top_k=10):
    """
    Subgradient descent with Polyak step size on the max-convolution objective.
    Uses multiple top peaks for a smoother gradient estimate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Lower bound estimate: average of convolution
    lb = conv.sum() / len(conv)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)
    momentum = 0.90

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        cur_max = conv.max()

        # Polyak step size
        gap = max(cur_max - lb, 1e-8)
        lr = gap / (2.0 * n)
        lr = min(lr, 6.0 / n)

        # Subgradient from top-k peaks (smoothed)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Weight peaks by their excess over the second-largest
        pv_max = pv.max()
        weights = np.exp((pv - pv_max) * 10.0)
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * weights[i] * f[k - idx]

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Update lower bound
            lb = max(lb, obj_new * 0.999)
        else:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                no_improve = 0

        if it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polyak it={it}, obj={best_obj:.12f}, lb={lb:.8f}, t={elapsed:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_sequence(f_src, n_dst):
    """Interpolate a sequence to a new length."""
    n_src = len(f_src)
    x_old = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, n_dst)
    f_new = np.interp(x_new, x_old, f_src)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(n_dst) / n_dst
    return f_new

    - Polyak subgradient
    - Multi-length exploration

    best_seq = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"\nStarting optimization with n={n_parent}, best={best_tracker[0]:.8f}")

    # === Phase 1: LSE with flatness penalty (main optimizer) ===
    print("\n=== Phase 1: LSE-Flat ===")
    t_alloc = min(500, deadline - time.monotonic() - 700)

    # === Phase 3: FW-LP Refinement ===
    print("\n=== Phase 3: FW-LP ===")

    # === Phase 4: Polyak Subgradient Polish ===
    print("\n=== Phase 4: Polyak Subgradient ===")

    t_alloc = min(200, deadline - time.monotonic() - 200)

    # === Phase 5: Second round of LSE-Flat on best ===
    print("\n=== Phase 5: LSE-Flat Round 2 ===")

    t_alloc = min(200, deadline - time.monotonic() - 150)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")

        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
