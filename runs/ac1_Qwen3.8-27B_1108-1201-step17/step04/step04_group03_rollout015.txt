import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def logsumexp(x, beta):
    """Compute (1/beta)*log(sum(exp(beta*x))) stably."""
    x_max = x.max()
    if x_max > 100:  # overflow protection
        return x_max
    return x_max + np.log(np.sum(np.exp(beta * (x - x_max)))) / beta

def smooth_objective_and_grad(p, beta):
    """
    Compute smooth approximation of max(conv(p,p)) and its gradient.
    p is on the probability simplex (sum=1, p>=0).
    Returns (objective_value, gradient_on_simplex).
    """
    n = len(p)
    conv = fast_convolve(p)
    # Log-sum-exp smoothing
    conv_shifted = conv - conv.max()
    exp_beta = np.exp(beta * conv_shifted)
    Z = exp_beta.sum()
    # Smooth max
    smooth_max = conv.max() + np.log(Z) / beta
    # Softmax weights
    w = exp_beta / Z
    # Gradient of smooth max w.r.t. p[j]:
    # d(smooth_max)/dp[j] = sum_k w_k * d(conv[k])/dp[j]
    # d(conv[k])/dp[j] = 2*p[k-j] (with bounds)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * p[k - idx]
    return smooth_max, grad

def adam_logspace(p0, time_limit, beta_init=5.0, beta_final=200.0, 
                  lr_init=0.01, decay=0.999, verbose=False):
    """
    Adam optimizer in log-space with log-sum-exp smoothing.
    Works with p on simplex, maintains positivity via log parametrization.
    """
    p = np.maximum(p0.astype(np.float64), 1e-30)
    s = p.sum()
    p = p / s
    n = len(p)
    
    # Log-space parameters
    x = np.log(p)
    # Remove mean for scale invariance
    x = x - x.mean()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    beta_cur = beta_init
    lr_cur = lr_init
    
    # Compute initial objective
    p_cur = np.exp(x - np.max(x))
    p_cur = p_cur / p_cur.sum()
    conv = fast_convolve(p_cur)
    best_obj = conv.max()
    best_p = p_cur.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    eps = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Update beta (annealing)
        progress = min(1.0, t / max(1, int(time_limit / 2)))
        beta_cur = beta_init * (beta_final / beta_init) ** progress
        lr_cur = lr_init * (decay ** t)
        
        # Current probability
        p_cur = np.exp(x - np.max(x))
        s = p_cur.sum()
        p_cur = p_cur / s
        
        # Compute gradient in log-space
        smooth_max, grad_simplex = smooth_objective_and_grad(p_cur, beta_cur)
        
        # Convert simplex gradient to log-space gradient
        # dF/dx[i] = p_cur[i] * (dF/dp[i] - sum_j dF/dp[j]*p_cur[j])
        grad_dot_p = np.dot(grad_simplex, p_cur)
        grad_log = p_cur * (grad_simplex - grad_dot_p)
        
        # Also need to account for the normalization (sum p = 1 constraint)
        # Since we work in log-space, the normalization is automatic
        # (exp(x)/sum(exp(x)) is always on simplex)
        
        # Adam update
        m = 0.9 * m + 0.1 * grad_log
        v = 0.99 * v + 0.01 * grad_log ** 2
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.99 ** t)
        
        x -= lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        
        # Remove mean (scale invariance)
        x = x - x.mean()
        
        # Compute actual objective
        p_new = np.exp(x - np.max(x))
        s = p_new.sum()
        p_new = p_new / s
        conv = fast_convolve(p_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            x = np.log(np.maximum(best_p, 1e-15))
            x = x - x.mean()
            x += rng.standard_normal(n) * 0.01
            x = x - x.mean()
            no_improve = 0
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t}, obj={obj:.8f}, best={best_obj:.8f}, elapsed={elapsed:.0f}s")
    
    return best_p, best_obj

def multiplicative_weight_v2(p0, time_limit, lr_init=0.03, top_k=20, 
                            beta_init=10.0, beta_growth=1.003, verbose=False):
    """
    Improved multiplicative weight with log-sum-exp gradient.
    Uses smooth gradient instead of hard peak selection.
    """

    p = p / p.sum()
    n = len(p)
    
    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr_cur = lr_init
    beta_cur = beta_init
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        conv = fast_convolve(p)
        M = conv.max()
        
        # Use log-sum-exp weights for a smoother gradient
        conv_shifted = conv - M
        exp_beta = np.exp(beta_cur * conv_shifted)
        w = exp_beta / exp_beta.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        # Only iterate over significant weights
        significant = w > 1e-10
        for k in np.where(significant)[0]:
            w_k = w[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w_k * p[k - idx]
        
        # Multiplicative update
        p = p * np.exp(-lr_cur * grad)
        p = np.maximum(p, 1e-25)
        p = p / p.sum()
        
        conv = fast_convolve(p)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr_cur *= 0.9995
        beta_cur *= beta_growth
        
        if no_improve > 80:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            p = best_p * np.exp(rng.standard_normal(n) * 0.003)
            p = p / p.sum()
            lr_cur = max(lr_init * 0.5, 0.005)
            no_improve = 0
    
    return best_p, best_obj

def coordinate_descent(p0, time_limit, verbose=False):

    Cyclic coordinate descent on the simplex.
    For each coordinate, try both increasing and decreasing,
    choosing the step that minimizes the objective.
    """
    p = np.maximum(p0.astype(np.float64), 1e-15)
    p = p / p.sum()
    n = len(p)
    
    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    it = 0
    step_size = 0.01
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Cycle through coordinates
        for j in range(n):
            # Try moving mass from j to other coordinates
            # or redistributing
            p_old = p[j]
            if p_old < 1e-10:
                continue
            
            # Try decreasing p[j] by a small amount and distributing
            delta = step_size * p_old
            new_p = p.copy()
            new_p[j] -= delta
            # Distribute to neighbors (or uniformly)
            new_p = new_p + delta / n
            
            conv_test = fast_convolve(new_p)
            if conv_test.max() < best_obj:
                p = new_p
                best_obj = conv_test.max()
                best_p = p.copy()

        step_size *= 0.999

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Random Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    
    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c / c.sum())
    
    # Periodic
    freq = rng.uniform(1, 5)
    per = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
    inits.append(per / per.sum())
    
    # Sine wave
    freq2 = rng.uniform(1, 8)
    sin = 0.5 + 0.5 * np.sin(2 * np.pi * freq2 * x / n)
    inits.append(sin / sin.sum())
    
    # Two peaks
    center1 = n // 3
    center2 = 2 * n // 3
    two = np.exp(-0.5 * ((x - center1) / (n/8))**2) + np.exp(-0.5 * ((x - center2) / (n/8))**2)
    inits.append(two / two.sum())
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(p, beta):
    """
    Compute smooth approximation of max(conv(p,p)) and gradient w.r.t. p.
    p should be on the probability simplex.
    Returns (smooth_max_value, gradient_vector_on_simplex).
    """
    n = len(p)
    conv = fast_convolve(p)
    M = conv.max()
    conv_shifted = conv - M
    exp_beta = np.exp(beta * conv_shifted)
    Z = exp_beta.sum()
    if Z < 1e-300:
        smooth_max = M
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        smooth_max = M + np.log(Z) / beta
        w = exp_beta / Z

    # Gradient: d(smooth_max)/dp[j] = sum_k w[k] * 2*p[k-j]
    grad = np.zeros(n)
    significant = w > 1e-12

        wk = w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * wk * p[k - idx]
    return smooth_max, grad

        lr_cur = lr_init * (decay ** t)

        # Current distribution
        p_cur = np.exp(x - x.max())
        s = p_cur.sum()
        p_cur = p_cur / s

        # Smooth gradient on simplex
        _, grad_simplex = compute_smooth_grad(p_cur, beta_cur)

        # Transform to log-space gradient:
        # dF/dx[i] = p[i] * (dF/dp[i] - sum_j dF/dp[j]*p[j])
        grad_dot_p = np.dot(grad_simplex, p_cur)
        grad_log = p_cur * (grad_simplex - grad_dot_p)

        # Adam update
        m = 0.9 * m + 0.1 * grad_log
        v = 0.99 * v + 0.01 * (grad_log * grad_log)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.99 ** t)

        x = x - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        x = x - x.mean()

        # Evaluate
        p_new = np.exp(x - x.max())
        s = p_new.sum()
        p_new = p_new / s
        conv = fast_convolve(p_new)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve >= restart_every:
            rng = np.random.default_rng((t * 2654435761) % 2**31)
            x = np.log(np.maximum(best_p, 1e-20))
            x = x - x.mean()
            x += rng.standard_normal(n) * (0.005 + 0.005 * min(1.0, no_improve / float(restart_every * 3)))
            x = x - x.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_p, best_obj


def multiplicative_weight_v2(p0, time_limit, lr_init=0.03, beta_init=5.0,
                             beta_growth=1.004, restart_every=70):
    """
    Multiplicative weight (mirror descent) with log-sum-exp smooth gradient.
    Uses smooth peak weighting instead of hard top-k selection.
    """
    p = np.maximum(p0.astype(np.float64), 1e-30)
    p = p / p.sum()
    n = len(p)

    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr_cur = lr_init
    beta_cur = beta_init
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Compute smooth gradient
        _, grad = compute_smooth_grad(p, beta_cur)

        # Multiplicative update
        update = np.exp(-lr_cur * grad)
        p = p * update
        p = np.maximum(p, 1e-25)
        p = p / p.sum()

        conv = fast_convolve(p)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr_cur *= 0.9996
        beta_cur *= beta_growth

        if no_improve >= restart_every:
            rng = np.random.default_rng((t * 40503) % 2**31)
            scale = 0.002 + 0.004 * min(1.0, no_improve / float(restart_every * 2))
            p = best_p * np.exp(rng.standard_normal(n) * scale)
            p = np.maximum(p, 1e-25)
            p = p / p.sum()
            lr_cur = max(lr_init * 0.6, 0.003)
            beta_cur = beta_init * 1.5
            no_improve = 0

    return best_p, best_obj


def palindromic_optimize(p0, time_limit, lr_init=0.02, beta_init=10.0):
    """
    Optimize under the palindromic constraint a[i] = a[n-1-i].
    This reduces the effective dimension and may find better local optima.
    """
    p = np.maximum(p0.astype(np.float64), 1e-30)
    p = p / p.sum()
    n = len(p)

    # Extract half (for even n: first n/2; for odd n: first (n+1)/2)
    half_len = (n + 1) // 2
    half = p[:half_len].copy()

    def expand(h):
        """Expand half to full palindromic sequence."""
        full = np.zeros(n)
        full[:half_len] = h
        full[half_len:] = h[:n - half_len][::-1]
        return full / full.sum()

    half = half / half.sum() if half.sum() > 0 else np.ones(half_len) / half_len
    p = expand(half)

    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()
    best_half = half.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr_cur = lr_init
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Compute gradient on full sequence
        _, grad_full = compute_smooth_grad(p, beta_init)

        # Fold gradient to half
        grad_half = grad_full[:half_len].copy()
        if n - half_len > 0:
            grad_half += grad_full[half_len:][::-1]

        # Simple projected gradient on half (with non-negativity)
        half = half - lr_cur * grad_half
        half = np.maximum(half, 1e-15)
        half = half / half.sum()

        p = expand(half)
        conv = fast_convolve(p)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_p = p.copy()
            best_half = half.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr_cur *= 0.9997

        if no_improve >= 60:
            rng = np.random.default_rng((t * 1234567) % 2**31)
            half = best_half * np.exp(rng.standard_normal(half_len) * 0.005)
            half = np.maximum(half, 1e-15)
            half = half / half.sum()
            lr_cur = max(lr_init * 0.5, 0.005)
            no_improve = 0

    return best_p, best_obj


def iterated_local_search(p0, time_limit, verbose=False):
    """
    Iterated local search: optimize locally, then make structured perturbation.
    Cycles through different perturbation strategies.
    """
    p = np.maximum(p0.astype(np.float64), 1e-15)
    p = p / p.sum()
    n = len(p)

    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()

    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Local optimization (rotates between MW and Adam)
        t_alloc_local = min(remaining * 0.4, 50)
        if t_alloc_local < 3:
            break

        if iteration % 2 == 1:
            p_opt, obj_opt = multiplicative_weight_v2(
                p, time_limit=t_alloc_local,
                lr_init=0.015, beta_init=10.0, beta_growth=1.005,
                restart_every=50
            )
        else:
            p_opt, obj_opt = adam_logspace_opt(
                p, time_limit=t_alloc_local,
                beta_init=20.0, beta_final=200.0, lr_init=0.003,
                restart_every=60
            )

        if obj_opt < best_obj - 1e-13:
            best_obj = obj_opt
            best_p = p_opt.copy()

        # Structured perturbation
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        scale = 0.003 + 0.004 * min(1.0, iteration / 15.0)
        rng = np.random.default_rng(iteration * 7919 + 13)
        strategy = iteration % 4

        if strategy == 0:
            # Global Gaussian
            p = best_p * np.exp(rng.standard_normal(n) * scale)
        elif strategy == 1:
            # Block perturbation
            block_size = max(2, n // 6)
            start = int(rng.integers(0, n - block_size))
            p = best_p.copy()
            p[start:start + block_size] *= np.exp(rng.standard_normal(block_size) * scale * 2)
        elif strategy == 2:
            # Mirror-swap perturbation
            p = best_p.copy()
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            delta = scale * p[i]
            p[i] += delta
            p[j] += delta
            # Remove from center
            center = n // 2
            remove = 2 * delta
            p[center] -= remove
        else:
            # Random two-point swap
            p = best_p.copy()
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = 0.01 * min(p[i], p[j])
                p[i] += delta
                p[j] -= delta

        p = np.maximum(p, 1e-15)
        p = p / p.sum()

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ILS iter={iteration}, obj={best_obj:.8f}, elapsed={elapsed:.0f}s")

    return best_p, best_obj


def fine_polish(p0, time_limit):
    """Very fine multiplicative weight polish with tiny steps."""
    p = np.maximum(p0.astype(np.float64), 1e-20)
    p = p / p.sum()
    n = len(p)

    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()

    t0 = time.monotonic()
    lr = 0.003
    beta = 50.0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        _, grad = compute_smooth_grad(p, beta)
        p = p * np.exp(-lr * grad)
        p = np.maximum(p, 1e-25)
        p = p / p.sum()

        conv = fast_convolve(p)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_p = p.copy()

        lr *= 0.9999
        beta *= 1.001

    return best_p, best_obj


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Concentrated Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Spread Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 3.0))

    # Edge-weighted (favors boundaries)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge / edge.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c / c.sum())

    # Periodic pattern
    freq = 3.0
    per = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
    inits.append(per / per.sum())

    # Sine
    freq2 = 5.0
    sin_pat = 0.5 + 0.5 * np.sin(2 * np.pi * freq2 * x / n)
    inits.append(sin_pat / sin_pat.sum())

    # Two peaks (edges)
    two = np.exp(-0.5 * (x / (n / 6.0)) ** 2) + np.exp(-0.5 * ((x - (n - 1)) / (n / 6.0)) ** 2)
    inits.append(two / two.sum())

    # Three peaks
    c1, c2, c3 = n // 4, n // 2, 3 * n // 4
    three = (np.exp(-0.5 * ((x - c1) / (n / 10.0)) ** 2) +
             np.exp(-0.5 * ((x - c2) / (n / 10.0)) ** 2) +
             np.exp(-0.5 * ((x - c3) / (n / 10.0)) ** 2))
    inits.append(three / three.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Multi-length exploration with diverse inits (~100s)
    Phase 2: Log-space Adam with adaptive temperature (~400s)
    Phase 3: Improved MW + Palindromic variant (~400s)
    Phase 4: Iterated local search (~550s)
    Phase 5: Final fine polish (~150s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0).astype(np.float64)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            pscore = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={pscore:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"No parent available: {e}")

    # === Phase 1: Multi-length Exploration (~100s) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t_phase1 = time.monotonic() + 90

    # Determine lengths to try
    lengths_to_try = [384]
    if parent is not None:
        np_ = len(parent)
        lengths_to_try = [np_]
        if np_ >= 300:
            lengths_to_try.append(max(256, np_ - 64))
        if np_ <= 500:
            lengths_to_try.append(min(640, np_ + 64))

    for n_len in lengths_to_try:
        if time.monotonic() > t_phase1 - 5:
            break
        rem = t_phase1 - time.monotonic()
        if rem < 10:
            break
        inits = generate_diverse_inits(n_len, rng, parent)
        for ii, init in enumerate(inits):
            if time.monotonic() > t_phase1 - 3:
                break
            try_seq(init, f"init_{n_len}_{ii}")

    # Quick optimization on best
    if best_seq is not None:
        rem = t_phase1 - time.monotonic()
        if rem > 5:
            p_opt, _ = multiplicative_weight_v2(best_seq, time_limit=min(rem, 20),
                                                 lr_init=0.05, beta_init=5.0, beta_growth=1.01)
            try_seq(p_opt, "phase1_quick")

    print(f"  Phase 1 done. Best: {best_score:.8f}")

    # === Phase 2: Log-space Adam (~400s) ===
    print("\n=== Phase 2: Log-space Adam ===")
    t_phase2 = time.monotonic() + min(400, (deadline - time.monotonic()) * 0.25)

    if best_seq is not None:
        adam_configs = [
            {'beta_init': 3.0, 'beta_final': 100.0, 'lr_init': 0.004, 'decay': 0.9996},
            {'beta_init': 10.0, 'beta_final': 300.0, 'lr_init': 0.006, 'decay': 0.9995},
            {'beta_init': 30.0, 'beta_final': 500.0, 'lr_init': 0.003, 'decay': 0.9997},
        ]

        rem = t_phase2 - time.monotonic()
        t_per = max(15, rem / len(adam_configs))

        for ci, cfg in enumerate(adam_configs):
            rem = t_phase2 - time.monotonic()
            if rem < 10:
                break
            t_alloc = min(t_per * 1.2, rem)
            print(f"  Adam cfg {ci} (beta {cfg['beta_init']}-{cfg['beta_final']}), t={t_alloc:.0f}s")
            p_opt, _ = adam_logspace_opt(best_seq, time_limit=t_alloc, **cfg)
            try_seq(p_opt, f"adam_{ci}")

            # Try from perturbed version
            rem = t_phase2 - time.monotonic()
            if rem > 20 and ci < len(adam_configs) - 1:
                noise = rng.standard_normal(len(best_seq)) * 0.008
                p_pert = project_simplex(best_seq + noise)
                p_opt2, _ = adam_logspace_opt(p_pert, time_limit=min(rem * 0.3, 40), **cfg)
                try_seq(p_opt2, f"adam_pert_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.8f}")

    # === Phase 3: MW + Palindromic (~400s) ===
    print("\n=== Phase 3: MW + Palindromic ===")
    t_phase3 = time.monotonic() + min(400, (deadline - time.monotonic()) * 0.25)

    if best_seq is not None:
        # MW with different settings
        mw_configs = [
            {'lr_init': 0.02, 'beta_init': 5.0, 'beta_growth': 1.003},
            {'lr_init': 0.04, 'beta_init': 15.0, 'beta_growth': 1.005},
            {'lr_init': 0.01, 'beta_init': 3.0, 'beta_growth': 1.002},
        ]

        rem = t_phase3 - time.monotonic()
        t_per_mw = max(10, rem * 0.6 / len(mw_configs))

        for ci, cfg in enumerate(mw_configs):
            rem = t_phase3 - time.monotonic()
            if rem < 10:
                break
            t_alloc = min(t_per_mw * 1.3, rem)
            print(f"  MW cfg {ci} (lr={cfg['lr_init']}), t={t_alloc:.0f}s")
            p_opt, _ = multiplicative_weight_v2(best_seq, time_limit=t_alloc, **cfg)
            try_seq(p_opt, f"mw_{ci}")

        # Palindromic optimization
        rem = t_phase3 - time.monotonic()
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Palindromic, t={t_alloc:.0f}s")
            p_opt, _ = palindromic_optimize(best_seq, time_limit=t_alloc, lr_init=0.015, beta_init=10.0)
            try_seq(p_opt, "palindromic")

            # Post-process: run MW from palindromic result without constraint
            rem = t_phase3 - time.monotonic()
            if rem > 15:
                p_opt2, _ = multiplicative_weight_v2(p_opt, time_limit=min(rem, 40),
                                                      lr_init=0.01, beta_init=20.0, beta_growth=1.003)
                try_seq(p_opt2, "palmw")

    print(f"  Phase 3 done. Best: {best_score:.8f}")

    # === Phase 4: Iterated Local Search (~500s) ===
    print("\n=== Phase 4: Iterated Local Search ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 50:
            t_alloc = min(rem, 500)
            print(f"  ILS, t={t_alloc:.0f}s")
            p_opt, _ = iterated_local_search(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(p_opt, "ils")

    print(f"  Phase 4 done. Best: {best_score:.8f}")

    # === Phase 5: Final Polish (~200s) ===
    print("\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        # Fine MW
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_alloc = min(rem, 70)
            print(f"  Fine MW, t={t_alloc:.0f}s")
            p_opt, _ = fine_polish(best_seq, time_limit=t_alloc)
            try_seq(p_opt, "fine_mw")

        # Fine Adam
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_alloc = min(rem, 50)
            print(f"  Fine Adam, t={t_alloc:.0f}s")
            p_opt, _ = adam_logspace_opt(best_seq, time_limit=t_alloc,
                                          beta_init=100.0, beta_final=500.0,
                                          lr_init=0.001, restart_every=50)
            try_seq(p_opt, "fine_adam")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            for pi in range(4):
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                noise = rng.standard_normal(len(best_seq)) * 0.001
                p_pert = project_simplex(best_seq + noise)
                t_alloc = min(rem, 12)
                p_opt, _ = fine_polish(p_pert, time_limit=t_alloc)
                try_seq(p_opt, f"perturb_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
