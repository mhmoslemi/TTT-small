def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:

    u = np.sort(v)[::-1]

    rho = np.nonzero(cond)[0][-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov accelerated gradient) on smoothed log-sum-exp max-convolution
2. Inverse temperature annealing for progressive sharpening
3. Periodic LP-based Frank-Wolfe direction-finding escapes

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_weights(b, tau):
    """Compute softmax weights for smoothed max: w_k = exp(tau*b_k)/sum_j exp(tau*b_j)."""
    b_max = b.max()
    scaled = tau * (b - b_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_s = np.exp(scaled)
    s = exp_s.sum()
    if s < 1e-30:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = exp_s / s
    return w


def compute_grad_from_weights(p, w):
    """Compute gradient of sum_k w_k * b_k where b = conv(p,p).
    
    grad[i] = 2 * sum_k w[k] * p[k-i] (valid range)
    = 2 * sum_j p[j] * w[i+j]  (with w indexed 0..2n-2)
    
    Using FFT: convolve p[::-1] with w, take slice [n-1 : 2n-1].
    """
    n = len(p)
    p_rev = p[::-1]
    # Full convolution length: n + (2n-1) - 1 = 3n - 2
    fft_len = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p_rev, fft_len)
    W = np.fft.rfft(w, fft_len)
    c = np.fft.irfft(P * W, fft_len)
    c = c[:3 * n - 2]
    grad = 2.0 * c[n - 1:2 * n - 1]
    return grad


def fista_optimize(p0, time_limit, tau_init=5.0, tau_end=300.0,
                   step_init=0.05, decay=0.9995, mom_anneal=True,
                   lp_every=400, verbose=False):

    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)

    b = fast_convolve(p)
    best_obj = b.max()
    best_p = p.copy()

    y = p.copy()
    t_mom = 1.0
    tau = tau_init
    step = step_init
    v_mom = 0.9

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        # Compute smoothed gradient at y
        b_y = fast_convolve(y)
        w = smooth_weights(b_y, tau)
        grad = compute_grad_from_weights(y, w)

        # FISTA momentum update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        p_new = project_simplex(y - step * grad)
        y_new = p_new + ((t_mom - 1.0) / t_next) * (p_new - p)
        p = p_new
        y = y_new
        t_mom = t_next

        # Evaluate at p
        b_p = fast_convolve(p)
        obj = b_p.max()

            best_obj = obj
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        tau = min(tau * 1.002, tau_end)
        step *= decay

        # LP escape

            g = _solve_lp_oracle(p, M_cur * 1.01, time_limit=15)

            t_mom = 1.0

def _solve_lp_oracle(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def subgrad_descent(p0, time_limit, lr=0.005, mom=0.93, top_k=64, temp=0.002,
                    lp_every=300, verbose=False):
    """Subgradient descent with momentum and softmax peak weighting."""
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)

    b = fast_convolve(p)
    best_obj = b.max()
    best_p = p.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        # Compute subgradient with softmax weighting
        if top_k >= len(b):
            peaks = np.arange(len(b))
        else:
            peaks = np.argpartition(b, -top_k)[-top_k:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w_vec = np.exp(scaled)
        wsum = w_vec.sum()
        if wsum < 1e-30:
            w_vec = np.zeros(len(peaks))
            w_vec[np.argmax(pv)] = 1.0
        else:
            w_vec /= wsum

        # Build subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w_vec[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w_vec[i] * p[k - idx]

        # Update with momentum
        v = mom * v - cur_lr * sg
        p = project_simplex(p + v)
        b = fast_convolve(p)
        obj = b.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp *= 1.003
        cur_lr *= 0.9998

        # LP escape
        if no_improve > lp_every and rem > 20:
            g = _solve_lp_oracle(p, best_obj * 1.02, time_limit=12)
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    for alpha in [0.3, 0.5, 0.7, 1.0]:
                        p_try = project_simplex((1 - alpha) * best_p + alpha * g)
                        b_try = fast_convolve(p_try)
                        if b_try.max() < best_obj:
                            best_obj = b_try.max()
                            best_p = p_try.copy()
                            p = p_try.copy()
                            v = np.zeros(n)
                            cur_lr = lr * 0.5
                            cur_temp = temp
                            break
            no_improve = 0

        # Restart
        if no_improve > 1500:
            rng = np.random.default_rng(it * 11 + 5)
            p = project_simplex(best_p + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            cur_lr = lr
            cur_temp = temp
            no_improve = 0

    return best_p, best_obj

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Pick the top peak
        k_peak = int(np.argmax(b))

        # Identify which p elements contribute to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contributors = np.arange(js, je + 1)
        other_idx = np.setdiff1d(np.arange(n), contributors)

        # Try to reduce the peak by redistributing mass from contributors to non-contributors
        # among the contributors: reduce the product by making them more equal
        # among non-contributors: increase them slightly

        # Strategy: reduce max contributor, increase a random non-contributor
        if len(contributors) > 1 and len(other_idx) > 0:
            # Find the contributor with max value
            contrib_vals = p[contributors]
            max_c = int(contributors[np.argmax(contrib_vals)])
            # Pick a random non-contributor to move mass to
            target = int(rng.choice(other_idx))

            for eps in [0.001, 0.002, 0.005, 0.01]:
                if p[max_c] < eps:
                    break
                p_try = p.copy()
                p_try[max_c] -= eps
                p_try[target] += eps
                b_try = fast_convolve(p_try)
                if b_try.max() < best_obj:
                    p = p_try.copy()
                    b = b_try
                    best_obj = b.max()
                    best_p = p.copy()
                    break

        # Also try: equalize contributors
        if len(contributors) > 2:
            avg_c = p[contributors].mean()
            p_try = p.copy()
            p_try[contributors] = avg_c
            b_try = fast_convolve(p_try)
            if b_try.max() < best_obj:
                p = p_try.copy()
                b = b_try
                best_obj = b.max()
                best_p = p.copy()

        # Periodic random perturbation
        if int((time.monotonic() - t0) * 2) % 3 == 0:
            p = project_simplex(best_p + rng.standard_normal(n) * 0.001)
            b = fast_convolve(p)
            if b.max() < best_obj:
                best_obj = b.max()
                best_p = p.copy()

    return best_p, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse structured initializations on the simplex."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = x - (n - 1) / 2.0  # centered

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window (raised cosine)
    c = np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2 + 0.01
    inits.append(c / c.sum())

    # Raised cosine (Hann)
    h = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1 + 1e-10))) + 0.01
    inits.append(h / h.sum())

    # Gaussian centered
    g = np.exp(-xm ** 2 / (2 * (n / 4.0) ** 2))
    inits.append(g / g.sum())

    # Bi-modal (two Gaussians)
    g1 = np.exp(-((xm - n * 0.25) ** 2) / (2 * (n / 10.0) ** 2))
    g2 = np.exp(-((xm + n * 0.25) ** 2) / (2 * (n / 10.0) ** 2))
    bi = g1 + g2 + 0.01
    inits.append(bi / bi.sum())

    # Triangular (tent)
    tri = 1.0 - np.abs(xm) / (n / 2.0)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Inverted triangular
    itr = np.abs(xm) / (n / 2.0) + 0.01
    inits.append(itr / itr.sum())

    # Quadratic U-shape
    q = xm ** 2 + 1.0
    inits.append(q / q.sum())

    # Sine
    sn = np.sin(np.pi * (x + 0.5) / n) + 0.05
    inits.append(sn / sn.sum())

    # Power law from center
    pw = np.abs(xm) ** 0.5 + 0.1
    inits.append(pw / pw.sum())

    # Random Dirichlet (concentrated)
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    # Random Dirichlet (spread)
    r2 = rng.dirichlet(np.ones(n) * 2.0)
    inits.append(r2)

    # Parent-based
    if parent is not None and len(parent) > 0:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend with cosine
            blended = 0.6 * (px / ps) + 0.4 * (c / c.sum())
            inits.append(blended / blended.sum())
            # Blend with uniform
            blended2 = 0.7 * (px / ps) + 0.3 * (np.ones(n) / n)
            inits.append(blended2 / blended2.sum())

    # Sinc-like (for frequency-domain flatness)
    sinc = np.sinc(xm / (n / 6.0))
    sinc = np.abs(sinc) + 0.01
    inits.append(sinc / sinc.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining FISTA on smoothed max-conv,
    subgradient descent, LP escapes, coordinate descent polishing,
    and multi-size exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.astype(a, np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p_arr = np.asarray(height_sequence_1, dtype=np.float64)
        p_arr = np.clip(p_arr, 0.0, 1004.0)
        if p_arr.ndim == 1 and p_arr.size > 0 and np.isfinite(p_arr).all() and p_arr.sum() > 1e-10:
            parent = p_arr / p_arr.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # === Phase 1: Quick FISTA on parent ===
    print("=== Phase 1: FISTA on Parent ===")
    if parent is not None:
        n_p = len(parent)
        t1 = min(120, max(30, (deadline - time.monotonic()) * 0.15))
        print(f"  FISTA on parent (n={n_p}), t={t1:.0f}s")
        a_opt, obj = fista_optimize(parent, time_limit=t1,
                                    tau_init=5.0, tau_end=200.0,
                                    step_init=0.05, lp_every=300)
        try_seq(a_opt, "parent_fista")

        # Subgrad follow-up
        rem = deadline - time.monotonic() - 200
        if rem > 30 and best_seq is not None:
            a_opt2, _ = subgrad_descent(best_seq, time_limit=min(rem, 80),
                                        lr=0.004, top_k=max(32, n_p // 8), temp=0.001)
            try_seq(a_opt2, "parent_sg")

    # === Phase 2: Exploration across sizes ===
    print("\n=== Phase 2: Multi-Size Exploration ===")

    sizes = [256, 384, 512, 768]
    # Remove parent size (already done)
    if parent is not None:
        parent_n = len(parent)
        sizes = [s for s in sizes if s != parent_n]
        sizes.insert(0, parent_n)

    time_per_size = max(60, (deadline - time.monotonic() - 200) / len(sizes))

    for n in sizes:
        if time.monotonic() > deadline - 150:
            break
        if parent is not None and n == len(parent) and best_seq is not None and len(best_seq) == n:
            # Already optimized parent size in phase 1, just do one more attempt
            pass

        print(f"  --- n={n}, time budget={time_per_size:.0f}s ---")
        inits = generate_inits(n, rng, parent)
        n_init = len(inits)
        t_per_init = max(15, time_per_size / max(1, n_init + 1))

        # Always try parent-interpolated init first if available
        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 80)
            if t_alloc < 10:
                break

            # Use FISTA for most, subgrad for some
            if i % 3 == 0:
                a_opt, obj = fista_optimize(f_init, time_limit=t_alloc,
                                            tau_init=5.0, tau_end=200.0,
                                            step_init=0.05, lp_every=max(200, n // 2))
            else:
                a_opt, obj = subgrad_descent(f_init, time_limit=t_alloc,
                                             lr=0.006, top_k=max(32, n // 10),
                                             temp=0.002, lp_every=max(200, n // 2))
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)
        best_norm = best_seq / best_seq.sum()

        # FISTA deep
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem // 3, 150)
            print(f"  FISTA deep (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = fista_optimize(best_norm, time_limit=t_alloc,
                                      tau_init=20.0, tau_end=500.0,
                                      step_init=0.02, decay=0.9997,
                                      lp_every=300)
            try_seq(a_opt, "deep_fista")

        # Subgradient deep with different params
        rem = deadline - time.monotonic() - 80
        if rem > 25:
            t_alloc = min(rem // 3, 120)
            print(f"  SG deep (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = subgrad_descent(best_seq, time_limit=t_alloc,
                                       lr=0.003, mom=0.95,
                                       top_k=max(64, n_best // 5), temp=0.0005,
                                       lp_every=250)
            try_seq(a_opt, "deep_sg")

        # Perturbed FISTA
        rem = deadline - time.monotonic() - 60
        if rem > 25:
            t_alloc = min(rem // 3, 100)
            f_pert = project_simplex(best_norm + rng.standard_normal(n_best) * 0.005)
            a_opt, _ = fista_optimize(f_pert, time_limit=t_alloc,
                                      tau_init=10.0, tau_end=400.0,
                                      step_init=0.03, lp_every=300)
            try_seq(a_opt, "perturbed_fista")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coord Descent Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 60)
            print(f"  Coord descent (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = coord_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

    # === Phase 5: Final micro-polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        best_norm = best_seq / best_seq.sum()

        # Final FISTA with very high tau
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_alloc = min(rem, 40)
            a_opt, _ = fista_optimize(best_norm, time_limit=t_alloc,
                                      tau_init=100.0, tau_end=1000.0,
                                      step_init=0.01, decay=0.9998,
                                      lp_every=500)
            try_seq(a_opt, "final_fista")

        # Tiny perturbations + quick optimize
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                f_pert = project_simplex(best_norm + rng.standard_normal(n_best) * 0.0005)
                a_opt, _ = subgrad_descent(f_pert, time_limit=min(rem, 10),
                                           lr=0.001, top_k=n_best // 3, temp=0.0001)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
