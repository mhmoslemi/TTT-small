    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_optimize(f0, time_limit, lp_timeout=10):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        A_cache = build_lp_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_timeout})
            if not res.success:
                no_improve += 1
                if no_improve > 5:
                    break
                continue
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g = g / g_sum
        except Exception:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 3:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_f, best_obj


def lse_adam_optimize(f0, time_limit, beta_start=10.0, beta_end=50000.0, top_k_frac=0.25):
    """LSE optimization with Adam optimizer and peak-targeted weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 3.0 / n
    b1 = 0.85
    b2 = 0.999
    eps = 1e-10
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    top_k = max(4, int(n * top_k_frac))
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        
        # Peak-targeted weights
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            scaled = beta * pv
            scaled -= scaled.max()
            w_peaks = np.exp(scaled)
            w_sum = w_peaks.sum()
            if w_sum < 1e-30:
                w_peaks = np.zeros(len(peaks))
                w_peaks[np.argmax(pv)] = 1.0
            else:
                w_peaks = w_peaks / w_sum
            w = np.zeros(len(conv))
            w[peaks] = w_peaks
        else:
            bc = beta * conv
            bc -= bc.max()
            w = np.exp(bc)
            w = w / w.sum()
        
        grad = compute_lse_grad(w, f)
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new  # Still move forward for exploration
            no_improve += 1
        
        # Adaptive lr
        if no_improve > 30:
            lr *= 0.93
        elif no_improve == 0:
            lr = min(lr * 1.005, 6.0 / n)
        
        # Beta schedule: ramp up gradually
        if beta < beta_end:
            beta *= 1.004
            if no_improve > 40:
                beta *= 1.005  # Accelerate when stuck
        
        # Restart
        if no_improve > 150:
            scale = 0.0005 + 0.002 * min(1.0, (no_improve - 150) / 100.0)
            rng = np.random.default_rng((t * 2654435761) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 3.0 / n
            no_improve = 0
            beta = max(beta_start, beta * 0.7)
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam it={t}, obj={best_obj:.10f}, beta={beta:.0f}, lr={lr:.6f}, no_imp={no_improve}, t={elapsed:.0f}s")
    
    return best_f, best_obj

        obj_new = conv_new.max()

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Gradient of LSE loss: 2 * cross_correlation(weights, f)."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix A where A[k,j] contributes to (f*f)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_optimize(f0, time_limit, lp_timeout=8):
    """Frank-Wolfe with LP oracle for global structure optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        A_cache = build_lp_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_timeout})
            if not res.success:
                no_improve += 1
                if no_improve > 6:
                    break
                continue
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-15:
                break
            g = g / gs
        except Exception:
            no_improve += 1
            if no_improve > 6:
                break
            continue

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 4:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def lse_adam_optimize(f0, time_limit, beta_start=10.0, beta_end=50000.0, top_k_frac=0.25):
    """LSE optimization with Adam optimizer and peak-targeted softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 4.0 / n
    b1 = 0.85
    b2 = 0.999
    eps = 1e-10
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    top_k = max(4, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Peak-targeted softmax weights
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            scaled = beta * pv
            scaled -= scaled.max()
            w_peaks = np.exp(scaled)
            w_sum = w_peaks.sum()
            if w_sum < 1e-30:
                w_peaks = np.zeros(len(peaks))
                w_peaks[np.argmax(pv)] = 1.0
            else:
                w_peaks = w_peaks / w_sum
            w = np.zeros(len(conv))
            w[peaks] = w_peaks
        else:
            bc = beta * conv
            bc -= bc.max()
            w = np.exp(bc)
            w = w / w.sum()

        grad = compute_lse_grad(w, f)

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 25:
            lr *= 0.94
        elif no_improve == 0:
            lr = min(lr * 1.003, 8.0 / n)

        # Beta schedule with acceleration when stuck
        if beta < beta_end:
            rate = 1.003
            if no_improve > 30:
                rate = 1.008
            beta *= rate

        # Restart from best when stuck
        if no_improve > 200:
            scale = 0.0005 + 0.003 * min(1.0, (no_improve - 200) / 150.0)
            rng = np.random.default_rng((t * 2654435761) % (2**31 - 1))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 4.0 / n
            no_improve = 0
            beta = max(beta_start, beta * 0.6)

        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam it={t}, obj={best_obj:.10f}, beta={beta:.0f}, lr={lr:.5f}, no_imp={no_improve}, t={elapsed:.0f}s")

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, beta=300.0, top_k_frac=0.15):
    """Mirror descent (exponential weights) on the simplex with LSE loss."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = 0.05
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(4, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Peak weights
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            scaled = beta * pv
            scaled -= scaled.max()
            w_peaks = np.exp(scaled)
            w_sum = w_peaks.sum()
            if w_sum < 1e-30:
                w_peaks = np.zeros(len(peaks))
                w_peaks[np.argmax(pv)] = 1.0
            else:
                w_peaks = w_peaks / w_sum
            w = np.zeros(len(conv))
            w[peaks] = w_peaks
        else:
            bc = beta * conv
            bc -= bc.max()
            w = np.exp(bc)
            w = w / w.sum()

        grad = compute_lse_grad(w, f)

        # Mirror descent: x_new[j] = x_old[j] * exp(-lr * grad[j]) / Z
        f_safe = np.maximum(f, 1e-30)
        log_update = -lr * grad
        log_update -= log_update.max()
        weights_vec = f_safe * np.exp(log_update)
        Z = weights_vec.sum()
        if Z < 1e-30:
            f_new = f.copy()
        else:
            f_new = weights_vec / Z

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            lr *= 0.92
        elif no_improve == 0:
            lr = min(lr * 1.01, 0.3)

        if no_improve > 120:
            rng = np.random.default_rng((it * 7919) % (2**31 - 1))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = 0.05
            no_improve = 0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror it={it}, obj={best_obj:.10f}, lr={lr:.5f}, no_imp={no_improve}, t={elapsed:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: FW-LP warm start (45s)
    Phase 2: Adam-LSE main optimization (850s)
    Phase 3: Mirror descent polish (200s)
    Phase 4: Final high-beta LSE (300s)
    Phase 5: Multi-length exploration (~250s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_seq = None
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"\nStarting: n={n_parent}, best={best_tracker[0]:.8f}, budget={budget_s}s")

    # === Phase 1: FW-LP warm start ===
    print("\n=== Phase 1: FW-LP Warm Start ===")
    t_alloc = min(45, deadline - time.monotonic() - 1500)
    if t_alloc > 10:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(parent, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: Adam-LSE main optimization ===
    print("\n=== Phase 2: Adam-LSE Main ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(850, deadline - time.monotonic() - 650)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lse_adam_optimize(current_best, time_limit=t_alloc,
                                        beta_start=6.0, beta_end=40000.0, top_k_frac=0.3)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: Mirror descent polish ===
    print("\n=== Phase 3: Mirror Descent ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 400)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = mirror_descent_optimize(current_best, time_limit=t_alloc,
                                              beta=200.0, top_k_frac=0.15)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: Final high-beta LSE ===
    print("\n=== Phase 4: Final High-Beta LSE ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(300, deadline - time.monotonic() - 180)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lse_adam_optimize(current_best, time_limit=t_alloc,
                                        beta_start=500.0, beta_end=100000.0, top_k_frac=0.08)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    # Try lengths near the current best
    candidates = []
    for n_alt in [320, 384, 416, 432, 448, 464, 480, 512, 576, 640]:
        if n_alt != n_best:
            candidates.append(n_alt)
    candidates.sort(key=lambda x: abs(x - n_best))
    candidates = candidates[:4]

    for n_alt in candidates:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 80:
            break
        t_alloc = min(120, t_remaining / max(1, (len(candidates) - 1)))
        if t_alloc < 40:
            break

        print(f"\n  Trying n={n_alt} (alloc={t_alloc:.0f}s)")

        # Interpolate from current best
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0.0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()
        else:
            f_interp = np.ones(n_alt) / n_alt

        # Two-stage optimization: coarse then fine
        t1 = t_alloc * 0.55
        t2 = t_alloc * 0.45
        a_opt, _ = lse_adam_optimize(f_interp, time_limit=t1,
                                      beta_start=8.0, beta_end=5000.0, top_k_frac=0.3)
        try_sequence(a_opt, best_tracker)

        cur_alt = best_tracker[1]
        if cur_alt is not None and len(cur_alt) == n_alt:
            a_opt, _ = lse_adam_optimize(cur_alt, time_limit=t2,
                                          beta_start=300.0, beta_end=30000.0, top_k_frac=0.1)
            try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
