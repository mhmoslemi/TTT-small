    """

    corr = fast_correlate_valid(w, a)
    # corr has length (2n-1) - n + 1 = n
    # The subgradient is 2 * corr

def compute_subgradient(a, conv, top_k=30, temp=0.01):
    """Compute smoothed subgradient using softmax weights on top-k peaks."""
    n = len(a)
    num_conv = len(conv)
    
    if top_k < num_conv:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(num_conv)
    
    vals = conv[peaks]
    scaled = vals / temp
    scaled -= scaled.max()
    w_peaks = np.exp(scaled)
    w_sum = w_peaks.sum()
    if w_sum < 1e-15:
        w_peaks = np.zeros(len(peaks))
        w_peaks[np.argmax(vals)] = 1.0
    else:
        w_peaks /= w_sum
    
    # Build full weight vector
    w = np.zeros(num_conv)
    w[peaks] = w_peaks
    
    grad = subgradient_via_fft(a, w)
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def build_toeplitz(a):
    """Build the Toeplitz matrix C where C[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, M, C=None):
    """
    Solve LP: max sum(b) s.t. C@b <= M, b >= 0.
    Returns (b, success).
    """
    n = len(a)
    if C is None:
        C = build_toeplitz(a)
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                                  'time_limit': 30})
        if result.success:
            b = result.x
            if b.sum() > 1e-10:
                return b, True
    except Exception:
        pass
    return None, False

                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    
                    if obj_try < best_obj - 1e-10:
                        a = a_try
                        conv = conv_try
                        M = conv.max()
                        best_obj = obj_try
                        best_a = a.copy()
                        C = build_toeplitz(a)
                        improved = True
                        no_improve = 0
                        break
                if not improved:
                    no_improve += 1
        
        iteration += 1
        
        # Perturbation on stagnation
        if no_improve >= 50:
            rng = np.random.default_rng(iteration * 31 + 7 + perturb_count)
            scale = 0.003 + 0.004 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            C = build_toeplitz(a)
            no_improve = 0
            perturb_count += 1
            if perturb_count > 50:
                break
    
    return best_a, best_obj


def gradient_descent_optimize(a0, time_limit, lr_init=0.02, top_k=40, temp_init=0.003):
    """Vanilla gradient descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    lr = lr_init
    temp = temp_init
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        grad = compute_subgradient(a, conv, top_k=top_k, temp=temp)
        
        # Normalized gradient step
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-12:
            break
        
        a_new = a - lr * grad / norm_grad * 0.5
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            a = a_new
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.5)
            temp = max(temp * 1.01, 0.0001)
        else:
            no_improve += 1
            lr *= 0.9
            temp *= 1.05
        
        if no_improve > 200:
            break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_trials=200000):
    """Pairwise coordinate descent: move mass between two positions."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    trial = 0
    
    # Precompute: for each pair (i,j) and delta, the change in conv
    # conv[k] changes by: delta*(a[k-i] + a[k-j]) for moving delta from j to i
    # (approximately, for small delta)
    
    while time.monotonic() - t0 < time_limit and trial < n_trials:
        batch = min(10000, n_trials - trial)
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            trial += 1
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try moving small amount from j to i
            for delta in [0.01, 0.005, -0.01, -0.005]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1
        
        if no_improve > 5000:
            break
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Cosine window
    cos_w = np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2 + 0.01
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Bimodal (edges)
    edge = max(2, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Anti-bimodal (center)
    center = np.zeros(n)
    center[n//4:3*n//4] = 1.0
    inits['center'] = center / center.sum()
    
    # Exponential decay from center
    center_idx = n / 2
    exp_d = np.exp(-0.5 * ((x - center_idx) / (n * 0.3)) ** 2)
    inits['gaussian'] = exp_d / exp_d.sum()
    
    # Alternating with damping
    alt = np.ones(n)
    alt[::2] *= 1.3
    alt[1::2] *= 0.7
    inits['alternating'] = alt / alt.sum()
    
    # Power law
    pw = (x + 1.0) ** 0.5
    inits['power'] = pw / pw.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits


def extend_sequence(a_short, n_target):
    """Extend sequence via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid FISTA + LP + coordinate descent search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    # =========================================================
    # Phase 1: FISTA scan across lengths and initializations
    # =========================================================
    print("=== Phase 1: FISTA Scan ===")
    t_start = time.monotonic()
    phase1_limit = 250.0
    
    lengths_to_try = [512, 768, 1024]
    if parent is not None:
        # Also try the parent's length
        if len(parent) not in lengths_to_try:
            lengths_to_try.insert(1, len(parent))
    
    scan_results = []
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_initializations(n, parent, rng)
        
        # Priority order
        priority = ['parent', 'parent_interp', 'uniform', 'cosine', 'bimodal', 'random', 'alternating', 'gaussian']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]
        
        t_per = max(8.0, (phase1_limit - 5) / (len(lengths_to_try) * 4))
        
        for name in init_names:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 200)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            
            # Try FISTA
            a_opt, obj = fista_optimize(a0, time_limit=t_alloc, top_k=30, temp_init=0.005)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  FISTA n={n}, {name}: {score:.6f} *BEST*")
                else:
                    print(f"  FISTA n={n}, {name}: {score:.6f}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 2: LP deep optimization on best candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()
    
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break
    
    if best_sequence is not None:
        if len(best_sequence) not in seen_lengths:
            top_configs.append((len(best_sequence), best_sequence.copy()))
            seen_lengths.add(len(best_sequence))
    
    remaining = deadline - time.monotonic() - 400.0
    n_candidates = max(1, len(top_configs))
    t_per_lp = max(30.0, remaining * 0.7 / n_candidates)
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 200)
        if t_alloc < 15:
            break
        
        print(f"  LP candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: FISTA polish with refined hyperparameters
    # =========================================================
    print("=== Phase 3: FISTA Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 250.0
        if remaining > 50:
            for pi in range(3):
                if time.monotonic() > deadline - 150:
                    break
                t_alloc = min(remaining / (3 - pi) * 0.8, deadline - time.monotonic() - 100)
                if t_alloc < 15:
                    break
                
                if pi == 0:
                    a0 = best_sequence.copy()
                    cfg = {'top_k': 50, 'temp_init': 0.003, 'lr_scale': 0.3}
                elif pi == 1:
                    noise = rng.standard_normal(len(best_sequence)) * 0.002
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    a0 = a0 / s if s > 1e-15 else best_sequence.copy()
                    cfg = {'top_k': 30, 'temp_init': 0.01, 'lr_scale': 0.2}
                else:
                    noise = rng.standard_normal(len(best_sequence)) * 0.005
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    a0 = a0 / s if s > 1e-15 else best_sequence.copy()
                    cfg = {'top_k': 100, 'temp_init': 0.001, 'lr_scale': 0.4}
                
                a_opt, obj = fista_optimize(a0, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  FISTA Polish {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  FISTA Polish {pi}: {score:.6f}")
    
    # Also try gradient descent
    if best_sequence is not None and time.monotonic() < deadline - 100:
        t_gd = min(remaining * 0.3, 60.0)
        if t_gd > 10:
            a_opt, obj = gradient_descent_optimize(best_sequence, time_limit=t_gd,
                                                     lr_init=0.03, top_k=50, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  GD: {score:.6f} *NEW BEST*")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 4: Length exploration - try extending/shrinking
    # =========================================================
    print("=== Phase 4: Length Exploration ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 200.0
        
        # Try a few nearby lengths
        candidate_lengths = [n_best // 2, n_best * 3 // 2, n_best * 2]
        candidate_lengths = [l for l in set(candidate_lengths) if 128 <= l <= 4096]
        
        t_per = max(10.0, remaining * 0.4 / max(1, len(candidate_lengths)))
        
        for n_try in candidate_lengths:
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break
            
            a0 = extend_sequence(best_sequence, n_try)
            a_opt, obj = fista_optimize(a0, time_limit=t_alloc * 0.6, top_k=40, temp_init=0.004)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                # Polish with LP if good
                if score < best_score + 0.01 and t_alloc > 20:
                    a_opt2, obj2 = lp_optimize(a_opt, time_limit=min(t_alloc * 0.4, 30))
                    score2 = float(evaluate_sequence(a_opt2.tolist()))
                    if np.isfinite(score2):
                        score = min(score, score2)
                        a_opt = a_opt2
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Length {n_try}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Length {n_try}: {score:.6f}")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 5: Final LP + coordinate descent polish
    # =========================================================
    print("=== Phase 5: Final Polish ===")
    t_start5 = time.monotonic()
    if best_sequence is not None and time.monotonic() < deadline - 20:
        # LP polish
        remaining = deadline - time.monotonic() - 20
        t_lp = min(remaining * 0.5, 60.0)
        if t_lp > 10:
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_lp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")
        
        # Coordinate descent
        remaining = deadline - time.monotonic() - 10
        t_cd = min(remaining * 0.5, 40.0)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd, n_trials=100000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:

                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")
    
    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
