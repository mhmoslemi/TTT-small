import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """C such that C @ g = conv(a, g)"""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def lp_find_direction(a, M):
    """Find g that minimizes max(conv(a, g)) subject to g>=0, sum(g)=1."""
    n = len(a)
    C = build_conv_matrix(a)
    # Minimize t s.t. C@g <= t, g >= 0, sum(g) = 1
    # Variables: g[0..n-1], t
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: C@g - t <= 0
    A_ub = np.hstack([C, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if result.success:
        g = result.x[:n]
        t_val = result.x[n]
        return g, t_val
    return None, None

def optimize_lp(a0, time_limit, verbose=False):
    """LP-based optimization."""
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() - t0 < time_limit:
        # Find direction
        g, t_val = lp_find_direction(a, M)
        if g is None:
            # Perturb
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            no_improve = 0
            continue
        
        # Line search
        d = g - a
        best_t = 0
        best_obj_try = best_obj
        
        for t_step in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            a_try = a + t_step * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:

            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t_step

            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:

            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= max_no_improve:
            # Perturb

            if s > 1e-15:
                a = a / s
            no_improve = 0
            max_no_improve = min(200, max_no_improve + 10)
            if no_improve > 50:
                break

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_active_gradient(a, conv, top_k=30, temp=0.001):
    """Compute weighted gradient using active peaks."""
    n = len(a)
    # Soft-max weighting over top peaks
    top_indices = np.argpartition(conv, -top_k)[-top_k:]
    top_vals = conv[top_indices]
    scaled = top_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(top_k)
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= w_sum
    
    d = np.zeros(n)
    for idx in range(top_k):
        if w[idx] < 1e-12:
            continue
        k = int(top_indices[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        d[idxs] += w[idx] * a[k - idxs]
    return 2.0 * d  # factor of 2 from the derivative

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_soft_gradient(a, conv, top_k=30, temp=0.001):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    top_indices = np.argpartition(conv, -top_k)[-top_k:]
    top_vals = conv[top_indices]
    scaled = top_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(top_indices))
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= w_sum

    d = np.zeros(n)
    for idx in range(len(top_indices)):
        if w[idx] < 1e-12:
            continue
        k = int(top_indices[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        d[idxs] += w[idx] * a[k - idxs]
    return 2.0 * d


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def lp_minimize_interaction(a, C=None):
    """
    Solve LP: minimize t s.t. conv(a,g)[k] <= t for all k, g >= 0, sum(g) = 1.
    Returns (g, t) or (None, None) on failure.
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t

    # Inequality: C@g - t <= 0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x[:n]
            t_val = result.x[n]
            return g, t_val
    except Exception:
        pass
    return None, None


def lp_step(a, conv, n, C=None):
    """One LP-based step with line search. Returns (a_new, improved)."""
    M = conv.max()
    if C is None:

    if g is None:
        return a.copy(), False

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False

    best_a = a.copy()
    best_obj = 2.0 * n * conv.max()

    # Line search with multiple candidates

            break

        return best_a, True
    return a.copy(), False

            i = rng.integers(0, n)

                    break
            else:
                stale += 1

        if stale > 300:

def make_geometric_init(n, r, rng):
    """Create a geometric sequence initialization."""
    if abs(r - 1.0) < 1e-10:
        a = np.ones(n)
    else:
        a = r ** np.arange(n, dtype=np.float64)
        # Also try reversed
        if rng.random() < 0.5:
            a = a[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_edge_init(n, edge_frac, rng):
    """Create an edge-heavy initialization."""
    a = np.ones(n)
    edge = max(1, int(n * edge_frac))
    # Boost edges
    a[:edge] *= 2.0
    a[-edge:] *= 1.5
    # Add some noise
    a += np.abs(rng.standard_normal(n)) * 0.1
    return a / a.sum()


def make_anti_palindromic_init(n, rng):
    """Create an anti-palindromic initialization: a[j] + a[n-1-j] ~ constant."""
    a = np.zeros(n)
    for j in range(n):
        a[j] = rng.random()
    # Make it anti-palindromic: a[j] = c - a[n-1-j]
    c = 1.0
    for j in range(n // 2):
        avg = (a[j] + a[n - 1 - j]) / 2
        a[j] = c - a[n - 1 - j]
        a[n - 1 - j] = c - a[j]
    a = np.maximum(a, 0.01)
    return a / a.sum()


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Geometric with various ratios
    for r in [1.05, 1.1, 1.2, 0.8, 0.9]:
        inits[f'geo_{r}'] = make_geometric_init(n, r, rng)

    # Edge-heavy
    inits['edge_0.2'] = make_edge_init(n, 0.2, rng)
    inits['edge_0.3'] = make_edge_init(n, 0.3, rng)

    # Anti-palindromic
    inits['anti'] = make_anti_palindromic_init(n, rng)

    # Sine-based
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.05)
    inits['sine1'] /= inits['sine1'].sum()

    # Power law
    pw = (x + 1.0) ** 0.5
    inits['power'] = pw / pw.sum()
    pw2 = (n - x) ** 0.5
    inits['power_r'] = pw2 / pw2.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def extend_to_length(a_short, n_target):
    """Extend a sequence to a longer length via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target),
                      np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext

                 'use_lp': True, 'lp_every': 20},
            ]

            n_runs = min(3, int(remaining / 30))
            t_per_run = remaining / max(1, n_runs) * 0.85

            for ci in range(n_runs):
                if time.monotonic() > deadline - 30:
                    break
                t_alloc = min(t_per_run, deadline - time.monotonic() - 20)
                if t_alloc < 10:
                    break

                cfg = configs[ci % len(configs)]
                a_opt, obj = active_gradient_optimize(
                    best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish-{ci}: {score:.6f} *** NEW BEST ***")
                else:
                    print(f"  Polish-{ci}: {score:.6f}")

    # =========================================================
    # Phase 4: Coordinate descent fine-tuning
    # =========================================================
    print("=== Phase 4: Coordinate Descent ===")
    t4 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 15:
        t_cd = min(30.0, deadline - time.monotonic() - 10)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *** NEW BEST ***")

    # =========================================================
    # Phase 5: Try extending to larger n
    # =========================================================
    print("=== Phase 5: Extension ===")
    if best_sequence is not None and time.monotonic() < deadline - 40:
        n_current = len(best_sequence)
        for n_try in [n_current * 2, n_current + 256]:
            if n_try > 2048 or time.monotonic() > deadline - 30:
                break
            a_ext = extend_to_length(best_sequence, n_try)
            # Add small structured perturbation
            rng2 = np.random.default_rng(n_try)
            anti = a_ext - a_ext[::-1]
            a_ext = project_simplex(a_ext + 0.01 * anti +
                                     rng2.standard_normal(n_try) * 0.005)

            t_alloc = min(20.0, deadline - time.monotonic() - 10)
            if t_alloc > 5:
                a_opt, obj = active_gradient_optimize(
                    a_ext, time_limit=t_alloc,
                    lr_init=0.006, top_k=40, temp_init=0.001)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Extend n={n_try}: {score:.6f} *** NEW BEST ***")
                else:
                    print(f"  Extend n={n_try}: {score:.6f}")

    # =========================================================
    # Final verification
    # =========================================================
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
