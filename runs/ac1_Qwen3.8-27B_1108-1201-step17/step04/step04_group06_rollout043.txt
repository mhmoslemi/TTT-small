import numpy as np

from scipy.optimize import minimize
from scipy.special import logsumexp


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_cross_convolve(p, a):
    """Linear convolution of p (length 2n-1) with a (length n) via FFT."""
    n = len(a)
    m = len(p)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(P * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def logsumexp_conv_objective(w, T):
    """Compute log(sum_k exp(conv(softmax(w), softmax(w))_k / T))."""
    a = softmax(w)
    c = fft_convolve(a)
    return logsumexp(c / T), a, c


def logsumexp_conv_gradient(w, T):
    """Compute gradient of log-sum-exp objective w.r.t. w."""

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on a log-sum-exp 
smoothed objective with temperature annealing. Combined with exponentiated 
gradient descent for exploration and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def fft_conv_auto(a):
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

def fft_conv(a, b):
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    B = np.fft.rfft(b, fs)
    return np.fft.irfft(A * B, fs)[:size]

def do_softmax(w):
    w = w - np.max(w)
    e = np.exp(w)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(w)) / len(w)
    return e / s

def lps_obj(w, T):
    a = do_softmax(w)
    c = fft_conv_auto(a)
    return logsumexp(c / T)

def lps_grad(w, T):
    a = do_softmax(w)
    c = fft_conv_auto(a)
    n = len(a)
    ct = c / T
    ct = ct - ct.max()
    p = np.exp(ct)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = p / ps
    c_mean = np.dot(p, c)
    q = p[::-1]
    conv_aq = fft_conv(a, q)
    g = conv_aq[n-1:2*n-1][::-1]
    grad = (2.0 / T) * a * (g - c_mean)
    return grad

def score_normalized(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    c = fft_conv_auto(a)
    return 2.0 * n * c.max()

def lps_optimize(a0, time_limit, T0_scale=0.1, T_min_scale=0.0005,
                 decay=0.7, batch=30, verbose=False):
    """L-BFGS optimization with temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    c = fft_conv_auto(a)
    max_c = c.max()
    T0 = T0_scale * max_c
    T_min = T_min_scale * max_c
    
    w = np.log(a + 1e-300)
    w = w - w.mean()
    
    best_a = a.copy()
    best_score = 2.0 * n * max_c
    t_start = time.monotonic()
    
    T = T0
    iteration = 0
    
    while time.monotonic() - t_start < time_limit:
        try:
            res = minimize(
                lps_obj, w, args=(T,),
                jac=lps_grad,
                method='L-BFGS-B',
                options={'maxiter': batch, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50}
            )
            w = res.x
            iteration += res.nit if hasattr(res, 'nit') else batch
        except Exception as e:
            if verbose:
                print(f"    minimize error: {e}")
            break
        
        a_new = do_softmax(w)
        c_new = fft_conv_auto(a_new)
        new_score = 2.0 * n * c_new.max()
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            if verbose:
                print(f"    LPS: T={T:.2e}, score={best_score:.8f}, iter={iteration}")
        
        if time.monotonic() - t_start > time_limit * 0.95:
            break
        
        T *= decay
        if T < T_min:
            T = T_min
            # Final iterations at min T
            try:
                res = minimize(
                    lps_obj, w, args=(T,),
                    jac=lps_grad,
                    method='L-BFGS-B',
                    options={'maxiter': batch * 2, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50}
                )
                w = res.x
            except Exception:
                pass
            a_new = do_softmax(w)
            c_new = fft_conv_auto(a_new)
            new_score = 2.0 * n * c_new.max()
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
            break
    
    return best_a, best_score

def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.005, temp_decay=0.9997,
                eta_decay=0.99995, top_k_frac=0.3, verbose=False):
    """Exponentiated gradient with softmax-weighted subgradients."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    
    c = fft_conv_auto(a)
    best_obj = c.max()
    best_a = a.copy()
    
    eta = eta0
    temp = temp0
    t_start = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t_start < time_limit:
        it += 1
        
        # Compute softmax-weighted subgradient
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
            peek = c[indices]
        else:
            indices = np.arange(len(c))
            peek = c.copy()
        
        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je+1] += 2.0 * w[idx] * a[k-je:k-js+1][::-1]
        
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = fft_conv_auto(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        eta *= eta_decay
        temp *= temp_decay
        
        if no_improve > 2000:
            rng_local = np.random.default_rng(hash((it, n)) % 2**32)
            a = best_a + rng_local.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fft_conv_auto(a)
            no_improve = 0
            eta = eta0 * 0.1
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization on a
log-sum-exp smoothed objective with temperature annealing, combined with
exponentiated gradient descent for exploration and multi-length search.
The key innovation is an analytically computed gradient of the log-sum-exp
of convolution entries via FFT cross-correlation, enabling efficient second-order
optimization. Multiple starting points and lengths are explored.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp


def fft_conv_auto(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def fft_conv(a, b):
    """Linear convolution of two arrays via FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    B = np.fft.rfft(b, fs)
    return np.fft.irfft(A * B, fs)[:size]


def do_softmax(w):
    """Numerically stable softmax."""
    w = w - np.max(w)
    e = np.exp(w)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(w)) / len(w)
    return e / s


def lps_obj(w, T):
    """Log-sum-exp of convolution entries divided by T."""
    a = do_softmax(w)
    c = fft_conv_auto(a)
    return logsumexp(c / T)


def lps_grad(w, T):
    """Analytic gradient of log-sum-exp objective w.r.t. log-parameter w."""
    a = do_softmax(w)
    c = fft_conv_auto(a)
    n = len(a)
    ct = c / T
    ct = ct - ct.max()
    p = np.exp(ct)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = p / ps
    c_mean = np.dot(p, c)
    q = p[::-1]
    conv_aq = fft_conv(a, q)
    g = conv_aq[n - 1:2 * n - 1][::-1]
    grad = (2.0 / T) * a * (g - c_mean)
    return grad


def eg_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(conv) using top-k peaks."""
    n = len(a)
    if top_k is not None and top_k < len(c) and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len(c))
        peek = c.copy()
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w = w / ws
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    return sg


def lps_optimize(a0, time_limit, T0_scale=0.05, T_min_scale=0.0003,
                 decay=0.7, batch=30, verbose=False):
    """L-BFGS optimization on log-sum-exp smoothed objective with T annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    c = fft_conv_auto(a)
    max_c = c.max()
    T0 = T0_scale * max_c
    T_min = T_min_scale * max_c

    w = np.log(a + 1e-300)
    w = w - w.mean()

    best_a = a.copy()
    best_score = 2.0 * n * max_c
    t_start = time.monotonic()
    iteration = 0

    T = T0
    while time.monotonic() - t_start < time_limit:
        try:
            res = minimize(
                lps_obj, w, args=(T,),
                jac=lps_grad,
                method='L-BFGS-B',
                options={'maxiter': batch, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50}
            )
            w = res.x
            iteration += res.nit
        except Exception:
            break

        a_new = do_softmax(w)
        c_new = fft_conv_auto(a_new)
        new_score = 2.0 * n * c_new.max()

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            if verbose:
                print(f"    LPS: T={T:.2e}, score={best_score:.10f}, iter={iteration}")

        if time.monotonic() - t_start > time_limit * 0.92:
            break

        T *= decay
        if T < T_min:
            T = T_min
            try:
                res = minimize(
                    lps_obj, w, args=(T,),
                    jac=lps_grad,
                    method='L-BFGS-B',
                    options={'maxiter': batch * 3, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50}
                )
                w = res.x
            except Exception:
                pass
            a_new = do_softmax(w)

                best_score = new_score
                best_a = a_new.copy()
                if verbose:
                    print(f"    LPS final: score={best_score:.10f}")
            break

    return best_a, best_score


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.004, temp_decay=0.9997,
                eta_decay=0.99995, top_k_frac=0.3, verbose=False):
    """Exponentiated gradient with softmax-weighted subgradients."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)

    n = len(a)

    best_obj = c.max()
    best_a = a.copy()

    eta = eta0
    temp = temp0
    t_start = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t_start < time_limit:
        it += 1

        sg = eg_subgradient(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fft_conv_auto(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new
        eta *= eta_decay
        temp *= temp_decay

        if no_improve > 2000:
            rng_local = np.random.default_rng((it * 7919 + n * 31) % (2**31))
            a = best_a + rng_local.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fft_conv_auto(a)
            no_improve = 0
            eta = eta0 * 0.1

    return best_a, best_obj


def projected_subgrad_polish(a0, time_limit, n_steps=500):
    """Polish using exact subgradient descent with diminishing step size on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fft_conv_auto(a)
    best_obj = c.max()
    best_a = a.copy()
    t_start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t_start > time_limit:
            break
        eta = 0.5 / np.sqrt(step + 1)

        # Use exact subgradient at argmax
        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        sg[js:je + 1] = 2.0 * a[k_star - je:k_star - js + 1][::-1]

        # Gradient step in log-space (exponentiated gradient step)
        log_update = -eta * sg
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new > 1e-300:
            a_new = a_new / s_new

        c_new = fft_conv_auto(a_new)
        obj_new = c_new.max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        c = c_new

    return best_a, best_obj


def generate_diverse_inits(a_best, rng, n, count=5):
    """Generate diverse initializations."""
    inits = [a_best.copy() if len(a_best) == n else np.ones(n) / n]
    x = np.arange(n, dtype=np.float64)

    # Edge-concentrated
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(rc / rc.sum())

    # Inverted Gaussian (peaked at edges)
    ig = np.exp(-2.0 * ((x - 0.5 * (n - 1)) / (0.4 * n)) ** 2)
    ig_inv = 1.0 / (ig + 0.5)
    inits.append(ig_inv / ig_inv.sum())

    # Random
    for _ in range(count):
        r = rng.dirichlet(np.ones(n) * 0.5)
        inits.append(r)

    # Perturbed best
    if len(a_best) == n:
        for scale in [0.01, 0.03]:
            p = a_best + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    return inits

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
