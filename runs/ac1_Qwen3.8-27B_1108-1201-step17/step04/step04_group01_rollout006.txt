"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses diverse initialization, accelerated LSE gradient descent, and coordinate descent polishing.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend((f[ks] for ks in (k - js)))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None):
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lse_fista(f0, time_limit, beta_start=20.0, beta_end=5000.0, restart_period=5000):
    """Log-sum-exp with FISTA acceleration."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    f_y = f.copy()  # FISTA iterate
    k_fista = 1.0
    beta = beta_start
    lr = 2.0 / (beta * n)  # initial learning rate
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient at f_y
        conv_y = fast_convolve(f_y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_lse_grad(w, f_y)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # FISTA update
        f_new = project_simplex(f_y - lr * grad)
        
        # Compute objective
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # FISTA momentum

        f_y = f_new + ((k_fista - 1.0) / k_new) * (f_new - f)
        f_y = np.maximum(f_y, 0.0)
        sy = f_y.sum()
        if sy > 1e-15:
            f_y = f_y / sy
        f = f_new
        k_fista = k_new

        # Adaptive parameters
        if no_improve >= 20:
            lr *= 0.9
            if no_improve > 100:
                # Restart
                f = best_f.copy()
                f_y = best_f.copy()
                k_fista = 1.0
                lr = 2.0 / (beta * n)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.002
            lr = 2.0 / (beta * n)

    return best_f, best_obj

        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        conv_look = fast_convolve(f_look)
        bc_look = beta * conv_look
        bc_look_max = bc_look.max()

        grad_look = compute_lse_grad(w_look, f_look)

        if gn > 1e-15:

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:

def coordinate_descent(f0, time_limit, max_sweeps=500):
    """Pairwise coordinate descent: try to improve by moving mass between pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve_sweeps = 0

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        # Try pairs of coordinates
        for i in range(n):
            if f[i] < 1e-12:
                continue
            # Try moving some mass from i to j
            # Optimal delta: minimize max_k (conv_k + delta*(a_{k-i} - a_{k-j}))
            # Approximate: pick j that minimizes the increase
            # Simple heuristic: try j = (i + n//2) % n (half-period shift)
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n
            
            # Try a few delta values
            fi = f[i]
            best_delta = 0
            best_val = conv.max()
            
            for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                delta = fi * frac
                if delta > f[j]:
                    delta = f[j] * 0.5
                    if delta > fi:
                        continue
                f[i] -= delta
                f[j] += delta
                c = fast_convolve(f)
                val = c.max()
                if val < best_val:
                    best_val = val
                    best_delta = delta
                # restore
                f[i] += delta
                f[j] -= delta
            
            if best_delta > 0:
                f[i] -= best_delta
                f[j] += best_delta
                improved = True
        
        if improved:
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve_sweeps = 0
            else:
                no_improve_sweeps += 1
        else:
            no_improve_sweeps += 1
        
        if no_improve_sweeps > 20:
            break

    return best_f, best_obj

                inits.append(c)
    
    # Raised cosine window
    for width in [0.3, 0.5, 0.7]:

        c = np.cos(width * (x - np.pi)) ** 2
        c = np.maximum(c, 0) + 1e-10
        c = c / c.sum()
        inits.append(c)
    
    # Random Dirichlet

        c = rng.dirichlet(np.full(n, alpha_val))
        inits.append(c)
    
    # Gaussian-like
    x = np.linspace(-3, 3, n)
    c = np.exp(-0.5 * x**2)
    c = c / c.sum()
    inits.append(c)
    
    return inits

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LSE gradient descent with Nesterov momentum, FW-LP structural
optimization, iterative perturbation-escaping, and multi-length exploration.
Uses FFT-based convolution and gradient computation throughout.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(weights, f):
    """Compute gradient of smoothed objective.
    weights: length 2n-1 (softmax of beta*conv)
    f: length n
    Returns: length n gradient vector
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse (2n-1, n) matrix where A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            fidx = k - js
            vals.extend(f[fidx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 25})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lse_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0,
                 lr_init=None, momentum_init=0.91):
    """LSE smoothed gradient descent with Nesterov momentum and adaptive params."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    if lr_init is None:
        lr = 3.0 / n
    else:
        lr = lr_init
    momentum = momentum_init
    beta = beta_start
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov look-ahead
        if momentum > 0:
            f_look = project_simplex(f - momentum * v)
        else:
            f_look = f

        conv_look = fast_convolve(f_look)
        bc = beta * conv_look
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_smooth_grad(w, f_look)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:

                lr *= 0.85

                lr *= 0.75

                v = np.zeros(n)

    print(f"\n--- Phase 3: High-beta LSE (beta 200->15000) ---", flush=True)
    t_alloc = min(300, deadline - time.monotonic() - 500)
    if t_alloc > 30:
        print(f"  Allocated: {t_alloc:.0f}s", flush=True)
        a_opt, obj = lse_optimize(cur, time_limit=t_alloc,
                                   beta_start=200.0, beta_end=15000.0,
                                   lr_init=2.0/n, momentum_init=0.90)
        try_sequence(a_opt, best_tracker)

    cur = best_tracker[1].copy() if best_tracker[1] is not None else parent

    # === Phase 4: Iterative perturbation + re-optimize ===
    print(f"\n--- Phase 4: Perturbation + Re-optimize ---", flush=True)
    n_perturb_rounds = 6
    for r in range(n_perturb_rounds):
        t_rem = deadline - time.monotonic() - 200
        if t_rem < 50:
            break
        t_each = t_rem / (n_perturb_rounds - r)
        t_lse = min(t_each * 0.55, 60)
        t_fw = min(t_each * 0.40, 45)

        # Structured perturbation
        if r % 3 == 0:
            # Cosine perturbation
            freq = rng.integers(1, 5)
            amp = 0.02 + 0.01 * r
            pert = amp * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2*np.pi))
            c = project_simplex(cur + pert * cur)
        elif r % 3 == 1:
            # Gaussian perturbation
            c = project_simplex(cur + rng.standard_normal(n) * (0.008 + 0.004 * r))
        else:
            # Block perturbation
            block_size = n // 4
            block = rng.integers(0, 4)
            pert = np.zeros(n)
            pert[block*block_size:(block+1)*block_size] += rng.uniform(0.02, 0.05)
            pert -= pert.mean()
            c = project_simplex(cur + pert)

        print(f"  Round {r+1}: LSE {t_lse:.0f}s + FW {t_fw:.0f}s", flush=True)

        a_opt, _ = lse_optimize(c, time_limit=t_lse,
                                  beta_start=50.0, beta_end=5000.0,
                                  lr_init=3.0/n, momentum_init=0.92)
        try_sequence(a_opt, best_tracker)

        cur = best_tracker[1].copy() if best_tracker[1] is not None else c
        t_rem = deadline - time.monotonic() - 200
        if t_rem > 40:
            a_opt, _ = fw_lp_optimize(cur, time_limit=t_fw)
            try_sequence(a_opt, best_tracker)
            cur = best_tracker[1].copy() if best_tracker[1] is not None else cur

    # === Phase 5: Alternative lengths ===
    print(f"\n--- Phase 5: Alternative Lengths ---", flush=True)
    cur = best_tracker[1].copy() if best_tracker[1] is not None else parent
    n_best = len(cur)

    alt_lengths = sorted(set([
        n_best - 8, n_best - 4, n_best - 2,
        n_best + 2, n_best + 4, n_best + 8,
        n_best + 16, n_best - 16
    ]), key=lambda x: abs(x - n_best))
    alt_lengths = [l for l in alt_lengths if 200 < l < 1000 and l != n_best]

    for i, n_alt in enumerate(alt_lengths[:4]):
        t_rem = deadline - time.monotonic() - 100
        if t_rem < 50:
            break
        t_alloc = min(80, t_rem - 30)
        if t_alloc < 20:
            break

        print(f"  n={n_alt}, {t_alloc:.0f}s", flush=True)

        # Interpolate
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, cur)
        f_interp = np.maximum(f_interp, 0)
        ss = f_interp.sum()
        if ss > 1e-10:
            f_interp = f_interp / ss

        a_opt, _ = lse_optimize(f_interp, time_limit=t_alloc,
                                  beta_start=15.0, beta_end=4000.0,
                                  lr_init=3.0/n_alt, momentum_init=0.92)
        try_sequence(a_opt, best_tracker)

    # === Phase 6: Final polish on absolute best ===
    print(f"\n--- Phase 6: Final Polish ---", flush=True)
    if best_tracker[1] is not None:
        t_rem = deadline - time.monotonic() - 10
        if t_rem > 30:
            n_fin = len(best_tracker[1])
            a_opt, _ = lse_optimize(best_tracker[1], time_limit=t_rem,
                                      beta_start=2000.0, beta_end=30000.0,
                                      lr_init=1.0/n_fin, momentum_init=0.88)
            try_sequence(a_opt, best_tracker)

    # Return best
    if best_tracker[1] is not None:
        final = best_tracker[1]
        final_score = float(evaluate_sequence(final.tolist()))
        print(f"\nFINAL: score={final_score:.10f}, n={len(final)}", flush=True)
        return [float(x) for x in final]
    else:
        fallback = np.ones(n) / n
        return [float(x) for x in fallback]
