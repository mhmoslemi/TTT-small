Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved exponentiated gradient
method with adaptive step sizes, momentum in log-space, support detection,
and multi-phase optimization including CVXPY polishing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c) using top-k peaks."""
    n = len(a)
    len_c = len(c)

    if top_k is not None and top_k < len_c and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        vals = c[indices]
    else:
        indices = np.arange(len_c)
        vals = c.copy()

    if temperature <= 0:
        # Hard: uniform weight on top-k
        w = np.ones(len(indices)) / len(indices)
    else:
        scaled = vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(vals)] = 1.0
        else:
            w /= ws

    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

    return sg


def compute_subgradient_fast(a, c, top_k, mode="hard"):
    """
    Faster subgradient computation. For 'hard' mode, uses uniform weights on top-k.
    For 'soft' mode, uses softmax weights.
    """
    n = len(a)
    len_c = len(c)

    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
    else:
        indices = np.arange(len_c)

    else:
        vals = c[indices]
        scaled = vals / 0.001  # small temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

    # Vectorized: sg[j] = 2 * sum_idx w[idx] * a[k_idx - j] for valid k_idx
    # This is: sg = 2 * sum_idx w[idx] * flip_shift(a, k_idx)
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

    return sg

    """
    Water-filling style update: push down peaks proportionally to how much
    they exceed the target level.
    """
    n = len(a)
    target = np.mean(c)  # or median
    excess = np.maximum(c - target, 0)
    
    if excess.max() < 1e-15:
        return a.copy()
    
    # For each lag k with excess, compute the "direction" to reduce it
    # Reducing (a*a)[k] by delta requires reducing a[j]*a[k-j] for various j
    # The steepest descent direction is -grad((a*a)[k]) = -2 * a_rev_shifted_k
    
    # Weighted sum of gradient directions, weighted by excess
    sg = np.zeros(n)
    active_indices = np.where(excess > excess.max() * 0.1)[0]  # top 10% excess
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je+1] += excess[k] * a[k-je:k-js+1][::-1]
    
    return sg

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient_fft(a, w):
    """
    Compute sg[j] = 2 * sum_k w[k] * a[k-j] using FFT cross-correlation.
    w: weight vector of length 2n-1 (zero for non-active lags)
    a: sequence of length n
    Returns: subgradient of length n
    """
    n = len(a)
    # Need N >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conjugate(A), N)[:n]
    return 2.0 * corr


def eg_optimize(a0, time_limit, eta0=1.0, top_k=128, mode="hard",
                momentum=0.9, support_thresh=1e-10, verbose=False):
    """
    Exponentiated gradient with FFT-accelerated subgradient, momentum, and adaptive step.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = min(top_k, 2 * n - 1)

    log_a = np.log(np.maximum(a, 1e-300))
    log_a_prev = log_a.copy()

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    eta = eta0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find top-k lags
        if top_k < 2 * n - 1:
            top_indices = np.argpartition(c, -top_k)[-top_k:]
        else:
            top_indices = np.arange(2 * n - 1)

        # Build weight vector
        if mode == "hard":
            w = np.zeros(2 * n - 1)
            w[top_indices] = 1.0 / top_k
        else:
            # Soft: use exponential weighting
            vals = c[top_indices]
            temp = (vals.max() - vals.min()) * 0.1 + 1e-10
            scaled = vals / temp
            scaled -= scaled.max()
            wv = np.exp(scaled)
            wv /= wv.sum()
            w = np.zeros(2 * n - 1)
            w[top_indices] = wv

        # Compute subgradient via FFT
        sg = compute_subgradient_fft(a, w)

        # Adaptive step: simple decay
        step = eta / np.sqrt(1.0 + 0.001 * it)

        # Log-space momentum update
        log_a_new = log_a + momentum * (log_a - log_a_prev) - step * sg
        # Normalize (subtract max for numerical stability)
        log_a_new -= log_a_new.max()
        log_a_prev = log_a.copy()
        log_a = log_a_new

        a_new = np.exp(log_a)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Support cleanup
        if support_thresh > 0:
            mask = a_new < support_thresh
            if mask.any():
                a_new[mask] = 0.0
                s_new2 = a_new.sum()
                if s_new2 > 1e-15:
                    a_new = a_new / s_new2

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"      EG iter {it}: obj={best_obj:.10f}, step={step:.8f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new
        log_a = np.log(np.maximum(a, 1e-300))

        # Decay
        eta *= 0.99995

        # Stagnation restart
        if no_improve > 3000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            log_a = np.log(np.maximum(a, 1e-300))
            log_a_prev = log_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.3

    return best_a.copy(), best_obj


def local_search(a0, time_limit, rng):
    """Local search with multiplicative perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        # Pick random coordinates
        n_perturb = max(5, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)

        improved = False
        for i in coords:
            old_val = a[i]
            if old_val < 1e-12:
                continue

            for factor in [0.5, 0.7, 0.85, 1/0.85, 1/0.7, 2.0]:
                a[i] = old_val * factor
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    break
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                a = best_a.copy()

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_a


def solve_qcqp(a0, time_limit=120):
    """Solve the QCQP using CVXPY."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    a_norm = a / s
    n = len(a)

    if n > 500:
        return None

    try:
        x = cp.Variable(n)
        t = cp.Variable()

        constraints = [
            cp.sum(x) == 1,
            x >= 0
        ]

        # Add convolution constraints
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # (a*a)[k] = sum_{j=js}^{je} a[j] * a[k-j]
            # Build as a quadratic form
            idx1 = []
            idx2 = []
            for j in range(js, je + 1):
                idx1.append(j)
                idx2.append(k - j)

            if len(idx1) == 1:
                # Linear term: 2*x[j]*x[k-j] if j != k-j, or x[j]^2 if j == k-j
                if idx1[0] == idx2[0]:
                    constraints.append(x[idx1[0]] ** 2 <= t)
                else:
                    constraints.append(2 * x[idx1[0]] * x[idx2[0]] <= t)
            else:
                # Quadratic form
                # Build sparse matrix
                from scipy.sparse import csr_matrix
                P = np.zeros((n, n))
                for i1, i2 in zip(idx1, idx2):
                    P[i1, i2] = 1.0
                # x^T P x = sum x[i1]*x[i2]
                constraints.append(cp.quad_form(x, P) <= t)

        prob = cp.Problem(cp.Minimize(t), constraints)

        t_start = time.monotonic()
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if time.monotonic() - t_start > time_limit:
                    break
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    result = np.maximum(x.value, 0)
                    rs = result.sum()
                    if rs > 1e-10:
                        return result / rs
                    break
            except Exception:
                continue

    except Exception:
        pass

    return None

        inits.append(a / a.sum())

    # Cosine
    for p in [0.5, 1.0, 2.0]:
        c = np.abs(np.cos(np.pi * t)) ** p + 0.001
        inits.append(c / c.sum())

    for p in [1.0, 2.0, 4.0]:
        s = np.sin(np.pi * t) ** p
        inits.append(s / s.sum())

    for sigma in [0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())
        g2 = np.exp(-0.5 * ((t - 0.3) / sigma) ** 2) + np.exp(-0.5 * ((t - 0.7) / sigma) ** 2)
        inits.append(g2 / g2.sum())

    # Linear ramp
    inits.append(t / t.sum())
    inits.append((1 - t) / (1 - t).sum())

    # V-shaped
    v = np.abs(2 * t - 1)
    inits.append(v / v.sum())
    inits.append((v + 0.1) ** 2 / ((v + 0.1) ** 2).sum())

    # Random Dirichlet
    for alpha in [0.1, 0.3, 0.5, 1.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    return inits
