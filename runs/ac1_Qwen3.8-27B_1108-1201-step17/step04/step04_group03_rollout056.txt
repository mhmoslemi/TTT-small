"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact 1D coordinate descent with incremental convolution updates (novel)
2. Multiplicative weight (mirror descent) polish
3. Subgradient polishing with momentum
4. Multiple restarts with diverse initializations
5. Adaptive length selection

Key innovation: Each coordinate update solves the exact 1D subproblem
(minimize max-conv/sum^2 over the single free coordinate), giving
much larger effective steps than gradient methods.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_desc_pass(a, conv_full, order, n_subdiv=45):
    """One pass of exact coordinate descent with incremental conv updates."""
    n = len(a)
    S = a.sum()

    for i in order:
        v = a[i]

        # Compute R = conv without a[i] contribution (from current conv_full)
        R = conv_full.copy()
        R[i:i + n] -= 2.0 * v * a
        R[2 * i] += v * v

        # C0: max of R where k is NOT in [i, i+n-1]
        C0 = 0.0
        if i > 0:
            C0 = max(C0, float(R[:i].max()))
        if i + n < 2 * n - 1:
            C0 = max(C0, float(R[i + n:].max()))

        # Pre-slice for fast evaluation
        R_slice = R[i:i + n]

        def obj_fn(vp):
            new_S = S + vp - v
            if new_S < 1e-15:
                return 1e10
            linear = R_slice + 2.0 * vp * a
            # Handle j=i: true value is R[2i]+vp^2, not R[2i]+2*vp*v
            old_li = linear[i]
            linear[i] = -1e30
            lin_max = float(linear.max())
            linear[i] = old_li
            quad = R[2 * i] + vp * vp
            total_max = max(C0, quad, lin_max)
            return total_max / (new_S * new_S)

        # Golden section search on [0, v_max]
        v_max = max(5.0 * v + 0.001, 0.05)
        # Check if expanding is needed
        obj_at_vmax = obj_fn(v_max)
        obj_at_0 = obj_fn(0.0)
        obj_at_v = obj_fn(v)

        if obj_at_vmax < obj_at_0 and obj_at_vmax < obj_at_v:
            # Optimum might be beyond v_max, expand
            v_max2 = v_max * 3
            obj_at_vmax2 = obj_fn(v_max2)
            if obj_at_vmax2 < obj_at_vmax:
                v_max = v_max2

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = v_max
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj_fn(c)
        fd = obj_fn(d)

        for _ in range(n_subdiv):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = obj_fn(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_fn(d)

        v_opt = (lo + hi) / 2.0

        # Ensure non-negative
        v_opt = max(v_opt, 0.0)

        if abs(delta) > 1e-15:
            conv_full[i:i + n] += 2.0 * delta * a
            conv_full[2 * i] += delta * delta
            a[i] = v_opt
            S += delta

    return a, conv_full


def coord_descent_optimize(a0, time_limit, verbose=False):
    """Run coordinate descent until time limit."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)

    conv_full = fast_convolve(a)
    current_obj = float(conv_full.max()) / (s * s)
    best_obj = current_obj
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Initial: sequential passes
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Choose order
        if passes < 5:
            order = list(range(n))
        else:
            order = list(range(n))
            rng.shuffle(order)

        a, conv_full = coord_desc_pass(a, conv_full, order)

        # Recompute conv_full (correct any floating point drift)
        if passes % 10 == 0:
            conv_full = fast_convolve(a)

        obj = float(conv_full.max()) / (a.sum() ** 2)
        passes += 1

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and passes % 100 == 0:
                print(f"    CD pass {passes}: obj={obj:.10f}")
        else:
            no_improve += 1

        # Perturbation to escape local minima
        if no_improve > 200:
            scale = 0.002 + 0.003 * min(1.0, no_improve / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv_full = fast_convolve(a)
            no_improve = 0

    return best_a, best_obj


def mult_weight_optimize(f0, time_limit, lr_init=0.04, top_k=12, temp_init=0.0004,
                         decay_lr=0.9995, decay_temp=1.0015):
    """Multiplicative weight (mirror descent) optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = float(conv.max())

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            ii = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[idx] * f[k - ii]
            log_update[ii] -= cur_lr * grad_contrib

        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = float(conv.max())

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= decay_lr
        cur_temp *= decay_temp

        if no_improve > 120:
            f = best_f * np.exp(rng.standard_normal(n) * 0.004)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.6, 0.01)
            cur_temp = temp_init * 1.5
            no_improve = 0

    return best_f, best_obj

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            ii = np.arange(js, je + 1)
            sg[ii] += 2.0 * w[idx] * f[k - ii]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = float(conv.max())

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.0008

        if no_improve > 300:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian in middle
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Two-bump (asymmetric)
    b1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.15 * n)) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2)
    inits.append((b1 + b2) / (b1 + b2).sum())

    # Linear ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())

    # Sine
    s1 = np.sin(np.pi * (x + 1) / (n + 1)) ** 2 + 0.05
    inits.append(s1 / s1.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining:
    1. Quick CD on multiple lengths to find best n
    2. Deep CD with restarts
    3. MW/SG polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"No parent: {e}")

    # === Phase 1: Quick length selection (~200s) ===
    print("\n=== Phase 1: Length Selection ===")
    candidate_ns = [200, 300, 384, 450, 500, 600]
    if parent is not None:
        candidate_ns.insert(0, len(parent))

    seen_ns = set()
    for n in candidate_ns:
        if n in seen_ns:
            continue
        seen_ns.add(n)
        rem = deadline - time.monotonic() - 1200
        if rem < 20:
            break
        t_alloc = min(rem, 25)
        print(f"  Quick CD n={n}, t={t_alloc:.0f}s")

        # Try a few inits
        inits = generate_inits(n, parent, rng)
        best_for_n = None
        best_obj_n = float('inf')

        for ii, init in enumerate(inits[:4]):
            rem = deadline - time.monotonic() - 1100
            if rem < 10:
                break
            t_sub = min(rem, 8)
            a_opt, obj = coord_descent_optimize(init, time_limit=t_sub)
            if obj < best_obj_n:
                best_obj_n = obj
                best_for_n = a_opt

        if best_for_n is not None:
            try_seq(best_for_n, f"quick_n{n}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep CD on best (~800s) ===
    print("\n=== Phase 2: Deep Coordinate Descent ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem, 600)
            print(f"  Deep CD n={n_best}, t={t_alloc:.0f}s")
            a_opt, obj = coord_descent_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_cd")

            # Try perturbed restarts
            rem = deadline - time.monotonic() - 400
            if rem > 30:
                for attempt in range(3):
                    rem = deadline - time.monotonic() - 300
                    if rem < 20:
                        break
                    noise = rng.standard_normal(n_best) * 0.003
                    f_pert = project_simplex(best_seq + noise)
                    t_sub = min(rem, 120)
                    a_opt2, _ = coord_descent_optimize(f_pert, time_limit=t_sub)
                    try_seq(a_opt2, f"cd_pert_{attempt}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: MW Polish (~300s) ===
    print("\n=== Phase 3: Multiplicative Weight Polish ===")

    if best_seq is not None:
        mw_configs = [
            {'lr_init': 0.03, 'top_k': 10, 'temp_init': 0.0003},
            {'lr_init': 0.05, 'top_k': 20, 'temp_init': 0.0005},
            {'lr_init': 0.02, 'top_k': 15, 'temp_init': 0.0002},
            {'lr_init': 0.04, 'top_k': 30, 'temp_init': 0.0008},
        ]

        rem = deadline - time.monotonic() - 250
        t_per = max(15, rem / len(mw_configs))

        for ci, cfg in enumerate(mw_configs):
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(t_per, rem)
            print(f"  MW config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = mult_weight_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=cfg['lr_init'], top_k=cfg['top_k'],
                temp_init=cfg['temp_init']
            )
            try_seq(a_opt, f"mw_{ci}")

            # CD polish after MW
            rem = deadline - time.monotonic() - 120
            if rem > 20:
                a_cd, _ = coord_descent_optimize(a_opt, time_limit=min(rem, 40))
                try_seq(a_cd, f"mw{ci}_cd")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: SG Polish + Final CD (~200s) ===
    print("\n=== Phase 4: Subgradient + Final Polish ===")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  SG polish, t={t_alloc:.0f}s")
            a_opt, _ = sg_polish(best_seq, time_limit=t_alloc,
                                 lr=0.0015, mom=0.94, top_k=30, temp=0.00003)
            try_seq(a_opt, "sg_polish")

        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, _ = coord_descent_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_cd")

        # Tiny perturbations + CD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            for _ in range(4):
                rem = deadline - time.monotonic() - 15
                if rem < 8:
                    break
                noise = rng.standard_normal(len(best_seq)) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = coord_descent_optimize(f_pert, time_limit=min(rem, 15))
                try_seq(a_opt, "micro_pert")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
