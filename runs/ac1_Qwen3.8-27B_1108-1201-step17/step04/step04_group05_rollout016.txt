def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_direct(a, w, peak_indices):
    """
    Compute gradient using only the specified peak indices.
    grad[m] = sum_{k in peaks} w[k] * d(conv[k])/d(a[m])
    d(conv[k])/d(a[m]) = 2*a[k-m] if 0<=k-m<n (interior), with edge corrections.

    grad = np.zeros(n)
    for idx in range(len(peak_indices)):

        wi = w[idx]
        if wi < 1e-15:
            continue
        # d(conv[k])/d(a[m]) for all m:
        # conv[k] = sum_{j} a[j]*a[k-j] where 0<=j<n, 0<=k-j<n
        # d/d(a[m]): a[k-m] (from first factor when j=m) + a[k-m] (from second when k-j=m, i.e., j=k-m)

        # For 0<=m<n and k-m >= n: no contribution
        # Wait, let me redo:
        # conv[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j]*a[k-j]
        # d/d(a[m]) = a[k-m] if m is in [max(0,k-n+1), min(k,n-1)] (from a[j] factor)
        #           + a[k-m] if (k-m) is in [max(0,k-n+1), min(k,n-1)] (from a[k-j] factor, j=k-m)
        # The second condition: 0 <= k-m <= n-1, i.e., k-n+1 <= m <= k
        # Combined with first: max(0,k-n+1) <= m <= min(k,n-1)
        
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # First factor contribution: a[k-m] for m in [js, je]
        # Second factor contribution: a[k-m] for m in [k-n+1, k] intersect [0, n-1]
        js2 = max(0, k - n + 1)
        je2 = min(n - 1, k)

        m_start = js
        m_end = je
        a_slice = a[k - m_end:k - m_start + 1][::-1]  # a[k-m] for m from m_start to m_end
        grad[m_start:m_end + 1] += 2.0 * wi * a_slice
    
    return grad


def softmax_gradient_descent(a0, time_limit, beta_start=50.0, beta_end=5000.0,
                             lr=0.01, momentum=0.9, verbose=False, seed=0):
    """
    Softmax gradient descent on the simplex for minimizing max(conv(a,a)).
    Uses analytic gradient computed from top peaks with softmax weights.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    # Adaptive parameters
    cur_lr = lr
    cur_mom = momentum
    n_top = max(10, n // 4)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        
        # Anneal beta
        cur_beta = beta_start * (beta_end / beta_start) ** progress
        # Decay learning rate
        cur_lr = lr * (1.0 - 0.5 * progress)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        # Select top peaks
        if n_top >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -n_top)[-n_top:]
        
        pv = conv[peaks]
        scaled = pv / (cur_beta * 1e-4)  # Scale for numerical stability
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        # Compute gradient
        grad = compute_gradient_direct(f, w, peaks)
        
        # Momentum update
        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new  # Still update to explore
        
        if verbose and it % 500 == 0:
            print(f"      SGD iter {it}: obj={best_obj:.10f}, beta={cur_beta:.0f}, lr={cur_lr:.5f}")
        
        # Restarts
        if no_improve > 200:
            scale = 0.005 + 0.003 * np.min(progress, 1.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            no_improve = 0
            # Reset lr on restart
            cur_lr = max(lr * 0.7, 0.001)
    
    return best_f, best_obj


def kkt_dual_primal(a0, time_limit, verbose=False, seed=0):
    """
    KKT-based dual-primal iteration:
    - Maintain dual weights w on peaks
    - Update primal via "water-filling" based on dual potential
    - Update dual by increasing weight on violated peaks
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    # Dual variables: weights on each convolution lag
    w = np.ones(conv_len) / conv_len
    eta = 0.05  # Step size for dual update
    alpha = 0.1  # Step size for primal update
    
    n_top = max(20, n // 3)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        # Focus on top peaks
        if n_top >= conv_len:
            active = np.arange(conv_len)
        else:
            active = np.argpartition(conv, -n_top)[-n_top:]
        
        # Update dual: increase weight on peaks that exceed current best
        peak_vals = conv[active]
        max_peak = peak_vals.max()
        
        # Softmax update on active peaks
        temp = 0.01 * max_peak + 1e-10
        w_active = np.exp(peak_vals / temp)
        w_active /= w_active.sum()
        
        # Update primal: potential V[m] = sum_k w[k] * 2*a[k-m]
        # On support: V should be constant (KKT condition)
        V = compute_gradient_direct(f, w_active, active)
        
        # Find lambda: the level at which we "clip"
        support = f > 1e-10
        if support.any():
            lambda_val = V[support].mean()
        else:
            lambda_val = V.mean()
        
        # Primal update: push down where V > lambda, push up where V < lambda
        diff = V - lambda_val
        f_new = f - alpha * diff
        
        # Project onto simplex
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:

            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:

            # Step back or stay
            f = f_new if obj_new < cur_obj + 1e-10 else best_f + rng.standard_normal(n) * 0.001
            f = project_simplex(np.maximum(f, 0))
        
        # Adapt step sizes
        alpha = max(0.01, alpha * (1.0 - 0.001 * progress))
        
        if verbose and it % 200 == 0:
            print(f"      KKT iter {it}: obj={best_obj:.10f}, alpha={alpha:.4f}")
        
        if no_improve > 150:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent_refine(a0, time_limit, verbose=False, seed=0):
    """
    Coordinate-wise refinement: optimize individual elements to reduce peak.
    For element m, changing a[m] affects conv[k] for k in [m, m+n-1].
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        
        # Random order of coordinates
        order = rng.permutation(n)
        
        for m in order:
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            # Current value
            am = f[m]
            if am < 1e-12:
                # Try setting it to a small positive value
                f[m] = best_f[m] * 0.5 + 1e-4
                s = f.sum()
                f /= s
                conv_new = fast_convolve(f)
                obj_new = conv_new.max()
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_f = f.copy()
                    improved = True
                else:
                    f[m] = best_f[m]
                    s = f.sum()
                    f /= s
                continue
            
            # Try modifying f[m]: the objective is 2*n*max(conv)/(sum)^2
            # After changing f[m] to f[m]+delta, the new sum is s+delta
            # The affected conv values are conv[m+j] for j=0,...,n-1 (and conv[k-m] for k...)
            # Specifically: conv[k] changes by 2*f[m]*a[k-m] + delta*2*a[k-m] + delta^2 (if k=2m)
            # For small delta: conv[k] new ≈ conv[k] + 2*delta*a[k-m]
            # We want to minimize max(conv) subject to sum = 1
            
            # Try a few delta values
            deltas = [am * (-0.3), am * (-0.5), am * (-0.7), am * 0.3, am * 0.5]
            best_delta = 0.0
            best_local_obj = conv.max()
            
            f_copy = f.copy()
            for d in deltas:
                new_val = am + d
                if new_val < 0:
                    continue
                f_copy[m] = new_val
                s_new = f_copy.sum()
                if s_new < 1e-10:
                    continue
                f_copy = f_copy / s_new
                conv_c = fast_convolve(f_copy)
                obj_c = conv_c.max()
                if obj_c < best_local_obj:
                    best_local_obj = obj_c
                    best_delta = d
            
            if best_delta != 0:

                f = np.maximum(f, 0)
                f /= f.sum()
                conv = fast_convolve(f)
                obj = conv.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and it > 5:
            break
    
    return best_f, best_obj


def generate_structured_inits(n, parent, rng):

        t = x / (n - 1) + 0.01
        b = t ** alpha_b * (1 - t + 0.01) ** beta_b
        inits.append(b / b.sum())
    
    # Random Dirichlet
    for conc in [0.3, 0.5, 1.0, 2.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * conc))
    
    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)

        ps = px.sum()
        if ps > 1e-10:

        pp = np.maximum(pp, 0)
        ps2 = pp.sum()
        if ps2 > 1e-10:
            inits.append(pp / ps2)
    
    return inits


def annealed_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Multi-stage annealed optimization:
    Stage 1: Low beta, large steps (exploration)
    Stage 2: Medium beta, medium steps (exploitation)
    Stage 3: High beta, small steps (refinement)
    """
    n = len(a0)
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Stage 1: Exploration (40% of time)
    t1 = time_limit * 0.35
    if t1 > 10:
        f, obj1 = softmax_gradient_descent(
            best_f, time_limit=t1, beta_start=20.0, beta_end=200.0,
            lr=0.02, momentum=0.92, verbose=verbose, seed=seed
        )
        if obj1 < best_obj:
            best_obj = obj1
            best_f = f.copy()
        if verbose:
            print(f"      Stage 1 (explore): obj={best_obj:.10f}")
    
    # Stage 2: Exploitation (40% of time)
    remaining = time_limit - (time.monotonic() - t0)
    t2 = remaining * 0.45
    if t2 > 10:
        f, obj2 = softmax_gradient_descent(
            best_f, time_limit=t2, beta_start=100.0, beta_end=2000.0,
            lr=0.005, momentum=0.95, verbose=verbose, seed=seed + 1
        )
        if obj2 < best_obj:
            best_obj = obj2
            best_f = f.copy()
        if verbose:
            print(f"      Stage 2 (exploit): obj={best_obj:.10f}")
    
    # Stage 3: Refinement (20% of time)
    remaining = time_limit - (time.monotonic() - t0)
    t3 = remaining * 0.5
    if t3 > 5:
        f, obj3 = softmax_gradient_descent(
            best_f, time_limit=t3, beta_start=2000.0, beta_end=10000.0,
            lr=0.001, momentum=0.97, verbose=verbose, seed=seed + 2
        )
        if obj3 < best_obj:
            best_obj = obj3
            best_f = f.copy()
        if verbose:
            print(f"      Stage 3 (refine): obj={best_obj:.10f}")
    
    # Final: coordinate descent
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 5:
        f, obj4 = coordinate_descent_refine(best_f, time_limit=min(remaining * 0.8, 60), seed=seed + 3)
        if obj4 < best_obj:
            best_obj = obj4
            best_f = f.copy()
        if verbose:
            print(f"      Stage 4 (coord): obj={best_obj:.10f}")
    
    return best_f, best_obj


def lp_polish(a0, time_limit, verbose=False):
    """
    LP-based polish: solve the linearized problem at the current point
    to get a good direction, then line search.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit - 5:
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        # Find top peaks
        k_top = max(10, n // 4)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        peaks = np.sort(peaks)
        
        # Build reduced LP: min t s.t. t >= 2*(B*g)[k] + conv[k] for k in peaks
        # g >= -f, sum(g) = 0
        # Variables: [t, g_0, ..., g_{n-1}]
        c = np.zeros(n + 1)
        c[0] = 1.0
        
        A_ub = np.zeros((k_top, n + 1))
        A_ub[:, 0] = -1.0
        b_ub = -conv[peaks]
        
        for ii, k in enumerate(peaks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for m in range(js, je + 1):
                A_ub[ii, m + 1] += 2.0 * f[k - m]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        
        if not res.success:
            break
        
        g = res.x[1:]
        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            break
        target = target / ts
        
        # Line search
        best_alpha = 0.0
        best_val = cur_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * target
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-8:
            break
        
        f = (1 - best_alpha) * f + best_alpha * target
        f = np.maximum(f, 0)
        f /= f.sum()
        
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            if verbose:
                print(f"      LP iter: obj={best_obj:.10f}, alpha={best_alpha:.3f}")
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point: multi-phase optimization combining FFT-based gradient descent,
    KKT dual-primal iteration, LP polish, and coordinate descent refinement.
    
    Strategy:
    1. Explore multiple lengths and initializations
    2. Annealed gradient descent (fast, many iterations)
    3. LP-based polish for final improvement
    4. Coordinate descent for final fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    n_default = 384
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            n_default = len(parent)
            print(f"Parent: n={n_default}, raw score={parent_score:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"No parent available: {e}")

    # === Phase 1: Quick screening of lengths and inits ===
    print("\n=== Phase 1: Screening ===")
    t_start = time.monotonic()
    
    lengths_to_try = sorted(set([n_default, max(32, n_default - 64), max(32, n_default + 64),
                                  max(32, n_default - 128), max(32, n_default + 128),
                                  256, 320, 448, 512, 640]))
    lengths_to_try = [l for l in lengths_to_try if 64 <= l <= 1024]
    # Remove duplicates while preserving order
    seen = set()
    unique_lengths = []
    for l in lengths_to_try:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)
    lengths_to_try = unique_lengths
    
    print(f"  Testing lengths: {lengths_to_try}")
    
    # Quick evaluation of best init for each length (just uniform + parent interp)
    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 600:
            break
        
        # Use parent interpolation as the best starting point
        if parent is not None:
            ini = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, len(parent)), parent)
            ini = np.maximum(ini, 0)
            ini = ini / ini.sum()
        else:
            ini = np.ones(n_try) / n_try
        
        # Quick anneal (short time budget)
        t_quick = 15  # seconds per length
        f_opt, obj = annealed_optimize(ini, time_limit=t_quick, seed=n_try)
        try_seq(f_opt, f"screen_n{n_try}")
    
    if best_seq is None:
        best_seq = np.ones(n_default) / n_default
        try_seq(best_seq, "fallback")
    
    print(f"  Phase 1 best: {best_score:.10f} (n={len(best_seq)})")

    best_n = len(best_seq)

    # === Phase 2: Deep optimization on top candidates ===
    print("\n=== Phase 2: Deep Optimization ===")
    
    # Try the best length with multiple initializations
    remaining = deadline - time.monotonic() - 100
    n_inits = 8
    t_per_init = max(5, remaining / (n_inits * 2))
    
    inits = generate_structured_inits(best_n, parent, rng)
    # Sort by quick evaluation
    init_scores = []
    for i, ini in enumerate(inits):
        conv_i = fast_convolve(ini)
        init_scores.append((conv_i.max(), i))
    init_scores.sort()
    
    # Take top initializations
    top_inits = [inits[i] for _, i in init_scores[:min(n_inits, len(inits))]]
    
    for i, ini in enumerate(top_inits):
        if time.monotonic() > deadline - 150:
            break
        
        print(f"  Optimization {i+1}/{len(top_inits)} (n={best_n}):")
        f_opt, obj = annealed_optimize(ini, time_limit=min(t_per_init, 120),
                                       verbose=False, seed=seed + i * 100)
        try_seq(f_opt, f"deep_{i}_n{best_n}")
    
    # Also try a few other lengths with the best structure
    for n_alt in [best_n + 32, best_n - 32, best_n + 64]:
        if n_alt < 64 or time.monotonic() > deadline - 120:
            continue
        px = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, best_n), best_seq)
        px = np.maximum(px, 0)
        px = px / px.sum()
        
        t_alt = min(60, deadline - time.monotonic() - 100)
        if t_alt > 10:
            f_opt, obj = annealed_optimize(px, time_limit=t_alt, seed=seed + 500 + n_alt)
            try_seq(f_opt, f"alt_n{n_alt}")
    
    print(f"  Phase 2 best: {best_score:.10f} (n={len(best_seq)})")

    # === Phase 3: LP Polish ===
    print("\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic() - 60
    t_lp = min(remaining * 0.5, 120)
    
    if t_lp > 15:
        print(f"  Running LP polish, budget={t_lp:.0f}s")
        f_opt, obj = lp_polish(best_seq, time_limit=t_lp, verbose=True)
        try_seq(f_opt, "lp_polish")
    
    # KKT iteration
    remaining = deadline - time.monotonic() - 40
    t_kkt = min(remaining * 0.4, 80)
    if t_kkt > 10:
        print(f"  Running KKT, budget={t_kkt:.0f}s")
        f_opt, obj = kkt_dual_primal(best_seq, time_limit=t_kkt, verbose=True, seed=seed)
        try_seq(f_opt, "kkt")
    
    print(f"  Phase 3 best: {best_score:.10f} (n={len(best_seq)})")

    # === Phase 4: Final refinement ===
    print("\n=== Phase 4: Final Refinement ===")
    remaining = deadline - time.monotonic() - 30
    
    # Multiple short refinement passes
    for attempt in range(5):
        if time.monotonic() > deadline - 15:
            break
        t_each = min(remaining / (5 - attempt), 40)
        if t_each < 5:
            break
        
        # Slight perturbation + short optimize
        noise_scale = 0.002 * (attempt + 1)
        f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * noise_scale)
        f_opt, obj = annealed_optimize(f_pert, time_limit=t_each, seed=seed + 1000 + attempt)
        try_seq(f_opt, f"final_{attempt}")
    
    # Coordinate descent on the absolute best
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        f_opt, obj = coordinate_descent_refine(best_seq, time_limit=min(remaining * 0.8, 30),
                                               seed=seed + 2000)
        try_seq(f_opt, "coord_final")
    
    # === Final verification ===
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
