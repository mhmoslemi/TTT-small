def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

    n = len(a)

    if s < 1e-15:

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g) (full linear)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def obj_normalized(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def solve_lp(n, M_target, C):
    """Solve: max sum(g) s.t. C@g <= M_target, g >= 0. Returns (g, success)."""
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-10,
                                                   'primal_feasibility_tolerance': 1e-10})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-12:
                return g, True
    except Exception:
        pass
    return None, False


def geometric_line_search(a, d, max_obj, n_steps=8, t_range=(0.01, 1.5)):
    """
    Geometric line search: try t values from large to small.
    Returns best a_new and whether improved.
    """
    best_a = a.copy()
    best_obj = max_obj
    t_lo, t_hi = t_range
    ratio = (t_lo / t_hi) ** (1.0 / n_steps)
    
    for i in range(n_steps):
        t = t_hi * (ratio ** i)
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * len(a_try) * conv_try.max()
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            break
    return best_a, best_obj


def soft_max_gradient(a, conv, top_k=30, temp=0.003):
    """Compute gradient of log-sum-exp approximation of max(conv)."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.91, top_k=40, temp_init=0.002):
    """Nesterov-accelerated gradient descent with log-sum-exp smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        conv_y = fast_convolve(y)
        grad = soft_max_gradient(y, conv_y, top_k=top_k, temp=temp)
        
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9997
        temp *= 1.004
        mom = min(0.97, mom + 0.0003)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def lp_optimize_forced(a0, time_limit, verbose=False):
    """
    LP-based optimization with forced peak reduction.
    Uses M*(1-epsilon) as the LP constraint to force the peak down.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    epsilon = 0.002
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0
    rng = np.random.default_rng(42 + n)
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        M_target = M * (1.0 - epsilon)
        C = build_conv_matrix(a)
        g, success = solve_lp(n, M_target, C)
        
        improved = False
        if success and g is not None:
            g_sum = g.sum()
            if g_sum > 1e-10:
                d = g - a
                norm_d = np.linalg.norm(d)
                if norm_d > 1e-12:
                    a_new, obj_new = geometric_line_search(a, d, best_obj)
                    if obj_new < best_obj - 1e-13:
                        a = a_new
                        conv = fast_convolve(a)
                        M = conv.max()
                        best_obj = obj_new
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                        epsilon = max(0.0002, epsilon * 0.97)
                    else:
                        no_improve += 1
                        epsilon = min(0.05, epsilon * 1.05)
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            epsilon = max(0.0002, epsilon * 0.8)
            no_improve += 1
        
        total_iters += 1
        
        if verbose and total_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {total_iters}, obj={best_obj:.8f}, eps={epsilon:.5f}, t={elapsed:.0f}s")
        
        if no_improve >= max_no_improve:
            # Peak-targeted perturbation
            top_peaks = np.argsort(conv)[-min(10, n // 4):]
            scale = 0.004 + 0.003 * min(perturb_count, 15)
            noise = rng.standard_normal(n) * scale
            # Reduce elements involved in top peaks
            for p in top_peaks:
                j_start = max(0, p - n + 1)
                j_end = min(n - 1, p)
                noise[j_start:j_end + 1] -= scale * 0.7
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(300, 60 + 25 * perturb_count)
            epsilon = 0.003
            
            if perturb_count > 60:
                # Try gradient escape
                a_g, obj_g = nesterov_optimize(best_a, time_limit=min(5.0, time_limit - (time.monotonic() - t0)),
                                                lr_init=0.008, top_k=50, temp_init=0.002)
                if obj_g < best_obj:
                    best_obj = obj_g
                    best_a = a_g.copy()
                    a = a_g.copy()
                    conv = fast_convolve(a)
                    M = conv.max()
                perturb_count = 0
                max_no_improve = 60
    
    return best_a, best_obj


def coordinate_refine(a0, time_limit):
    """
    Coordinate descent: try moving mass between pairs of positions.
    Uses convolution updates for efficiency.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    deltas = [0.002, 0.005, 0.01, 0.02, 0.05, -0.002, -0.005, -0.01, -0.02, -0.05]
    
    while time.monotonic() - t0 < time_limit:
        batch = max(1, int(200000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.3:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            found = False
            for delta in deltas:
                ai = a[i] + delta
                aj = a[j] - delta
                if ai < 0 or aj < 0:
                    continue
                a[i] = ai
                a[j] = aj
                conv_try = fast_convolve(a)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-11:
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    found = True
                    break
                else:
                    a[i] = ai - delta
                    a[j] = aj + delta
            if not found:
                no_improve += 1
        if no_improve > 5000:
            break
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine-based
    s1 = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] = s1 / s1.sum()
    
    s2 = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] = s2 / s2.sum()
    
    # Ramp (monotone - exploits non-palindromic structure)
    ramp = (x + 1.0)
    inits['ramp'] = ramp / ramp.sum()
    
    # Inverse ramp
    inits['inv_ramp'] = (n - x) / (n * (n + 1) / 2)
    
    # Bimodal
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Two-block with ratio
    c = 2.0
    two_block = np.ones(n)
    two_block[n // 2:] = c
    inits['two_block'] = two_block / two_block.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase LP + gradient + coordinate descent search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, score={evaluate_sequence(p.tolist()):.6f}")
    except Exception:
        pass
    
    # ============ Phase 1: Quick gradient scan across lengths ============
    print("=== Phase 1: Gradient Scan ===")
    t_start = time.monotonic()
    phase1_time = 150.0
    
    lengths = [512, 640, 768, 896, 1024]
    scan_results = []
    
    n_inits_per_len = 3
    t_per = max(5.0, (phase1_time - 5) / (len(lengths) * n_inits_per_len))
    
    for n in lengths:
        if time.monotonic() > deadline - 1200:
            break
        inits = generate_inits(n, parent, rng)
        priority = ['parent', 'parent_interp', 'ramp', 'inv_ramp', 'two_block',
                    'uniform', 'sine1', 'random', 'bimodal', 'gaussian', 'sine2']
        init_names = [nm for nm in priority if nm in inits]
        init_names = init_names[:n_inits_per_len]
        
        for name in init_names:
            if time.monotonic() > deadline - 1100:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 1000)
            if t_alloc < 3:
                break
            a0 = inits[name]
            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.012, momentum=0.92,
                                           top_k=40, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # ============ Phase 2: LP deep optimization ============
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()
    
    # Collect top candidates (unique by length, take best per length)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break
    
    # Ensure we include the best overall
    if best_sequence is not None:
        bl = len(best_sequence)
        if bl not in seen_lengths:
            top_configs.insert(0, (bl, best_sequence.copy()))
            seen_lengths.add(bl)
        else:
            # Replace with the best for that length
            for i, (ln, ao) in enumerate(top_configs):
                if ln == bl:
                    top_configs[i] = (bl, best_sequence.copy())
                    break
    
    remaining = deadline - time.monotonic() - 400.0
    n_cands = max(1, len(top_configs))
    t_per_lp = max(60.0, remaining * 0.55 / n_cands)
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 300)
        if t_alloc < 20:
            break
        print(f"  LP cand {i}: n={n}, alloc={t_alloc:.0f}s")
        a_opt, obj = lp_optimize_forced(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # ============ Phase 3: Extended LP on best ============
    print("=== Phase 3: Extended LP Polish ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 100:
            # Multiple LP runs with different perturbations
            n_runs = min(5, max(2, int(remaining / 100)))
            t_each = remaining / n_runs * 0.85
            
            for ri in range(n_runs):
                if time.monotonic() > deadline - 150:
                    break
                t_alloc = min(t_each, deadline - time.monotonic() - 120)
                if t_alloc < 20:
                    break
                
                if ri == 0:
                    a0 = best_sequence.copy()
                else:
                    scale = 0.002 * (ri + 1)
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()
                
                a_opt, obj = lp_optimize_forced(a0, time_limit=t_alloc, verbose=(ri == 0))
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {ri} (n={len(a_opt)}): {score:.6f} *NEW BEST*")
                else:
                    print(f"  Polish {ri}: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # ============ Phase 4: Coordinate descent + gradient fine-tuning ============
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        n_best = len(best_sequence)
        
        # Coordinate descent
        t_cd = min(remaining * 0.35, 80.0)
        if t_cd > 10:
            a_opt, obj = coordinate_refine(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")
            else:
                print(f"  CoordDesc: {score:.6f}")
        
        # Gradient fine-tuning with different configs
        remaining = deadline - time.monotonic() - 40.0
        if remaining > 20:
            configs = [
                {'lr_init': 0.004, 'momentum': 0.94, 'top_k': 60, 'temp_init': 0.001},
                {'lr_init': 0.002, 'momentum': 0.96, 'top_k': 100, 'temp_init': 0.0005},
                {'lr_init': 0.008, 'momentum': 0.90, 'top_k': 30, 'temp_init': 0.003},
                {'lr_init': 0.006, 'momentum': 0.93, 'top_k': 50, 'temp_init': 0.0015},
            ]
            n_grads = min(len(configs), max(1, int(remaining / 15)))
            for gi in range(n_grads):
                if time.monotonic() > deadline - 30:
                    break
                t_sg = min(remaining / (n_grads - gi) * 0.8, 30.0)
                if t_sg < 5:
                    break
                cfg = configs[gi % len(configs)]
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_sg, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad {gi}: {score:.6f} *NEW BEST*")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")
    
    # ============ Phase 5: Final LP micro-polish ============
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 15:
        t_final = min(40.0, deadline - time.monotonic() - 10)
        if t_final > 10:
            a_opt, obj = lp_optimize_forced(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
