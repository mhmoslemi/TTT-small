"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive LSE gradient descent
with periodic FW-LP restarts, multi-peak subgradient polishing, and

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, k):
    """Compute subgradient of c[k] = conv(a,a)[k] with respect to a."""
    n = len(a)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    sg[idx] = 2.0 * a[k - idx]
    return sg

def compute_multi_peak_subgrad(a, peaks, weights):
    """Weighted sum of subgradients at given peak positions."""
    n = len(a)
    sg = np.zeros(n)
    for k, w in zip(peaks, weights):
        if w < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * a[k - idx]
    return sg

def lse_gradient(a, beta):
    """Compute LSE-smoothed gradient using FFT."""
    n = len(a)
    conv = fast_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Gradient = sum_k w[k] * d(conv[k])/d(a) = sum_k w[k] * 2*a[reversed/shifted]
    # This is a cross-correlation of w with a (flipped)
    # grad[i] = 2 * sum_k w[k] * a[k-i] (when valid)
    # = 2 * (w flipped convolve a)[n-1 + i] ... let me think
    
    # Actually: grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
    # Let j = k-i: grad[i] = 2 * sum_j w[i+j] * a[j]
    # This is the cross-correlation of w and a, evaluated at lag i.
    
    # Using FFT: cross-correlation of w (length 2n-1) and a (length n)
    # We need: sum_j w[i+j] * a[j] for i = 0, ..., n-1
    # This is (w flipped) * a convolved, at specific positions
    
    # Let's use the direct approach with FFT
    w_rev = w[::-1]  # length 2n-1, reversed
    total_len = (2*n-1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_wa = np.fft.irfft(W * A, fft_size)
    # conv_wa[m] = sum_j w_rev[j] * a[m-j] = sum_j w[2n-2-j] * a[m-j]
    # We want sum_j w[i+j] * a[j] = sum_j w[(i+j)] * a[j]
    # Let's define: we want (w * a)[i] where the convolution is sum_j w[i+j]*a[j]
    # This is the cross-correlation <w, shift_i(a)> = sum_j w[j+i]*a[j]
    # = (w reversed shifted) convolve a
    # 
    # Actually, let me just use the formula:
    # sum_j w[i+j] * a[j] for j = 0..n-1, i+j ranges from i to i+n-1
    # This is the "sliding dot product" of w[i:i+n] with a
    # = cross-correlation of w and a at lag i (with w being the longer signal)
    #
    # Using FFT: define W_full as w zero-padded, A_full as a zero-padded
    # The cross-correlation R[i] = sum_j w[j+i]*a[j] 
    # = sum_j w_rev[2n-2-(j+i)] * a[j] ... this is getting confusing
    #
    # Simpler: R[i] = sum_{j=0}^{n-1} a[j] * w[j+i]
    # This is the convolution of a_rev with w, evaluated at position (n-1+i):
    # (a_rev * w)[n-1+i] = sum_j a_rev[j] * w[n-1+i-j] = sum_j a[n-1-j] * w[n-1+i-j]
    # Let k = n-1-j: = sum_k a[k] * w[i+k] = R[i] ✓
    
    a_rev = a[::-1]
    fft_size2 = 1 << ((n + (2*n-1) - 1) - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size2)
    Wf = np.fft.rfft(w, fft_size2)
    result = np.fft.irfft(Ar * Wf, fft_size2)
    return 2.0 * result[n-1:n-1+n]

def fw_lp_step(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv-matrix @ g <= M, g >= 0. Return g normalized."""
    n = len(a)
    # Build the constraint matrix A where A[k,j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend((a[k - idx]).tolist())
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def score_sequence(a):
    """Compute the objective value."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * conv.max()

def cosine_window(n):
    """Raised cosine window."""
    x = np.arange(n)
    return (1.0 + np.cos(2.0 * np.pi * (x + 0.5) / n)) / 2.0

def triangular_window(n):
    """Triangular (Bartlett) window."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    return 1.0 - np.abs(x - mid) / (mid + 0.5)

def exp_decay_window(n, asym=1.0):
    """Asymmetric exponential decay window."""
    x = np.arange(n)
    if asym == 1.0:
        return np.exp(-2.0 * x / n) + np.exp(-2.0 * (n-1-x) / n)
    else:
        return np.exp(-asym * x / n) + np.exp(-asym * (n-1-x) / n)

def lse_optimize(a0, time_limit, beta_start=5.0, beta_end=50000.0, lr_scale=1.0):
    """LSE-smoothed gradient descent with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    beta = beta_start
    lr = 3.0 / (n * lr_scale)
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_mom = momentum
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute LSE-softmax weights
        conv = fast_convolve(a)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Compute gradient via FFT
        grad = lse_gradient(a, beta)
        
        # Normalize
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm
        
        # Momentum update
        v = cur_mom * v - lr * grad
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.9
                cur_mom = max(0.85, cur_mom - 0.003)
            if no_improve > 100:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                cur_mom = 0.90
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
        
        # Ramp beta
        if beta < beta_end:
            beta *= 1.005
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj

def multi_peak_polish(a0, time_limit, n_peaks=5, lr_init=0.005, momentum=0.93):

    a = np.maximum(a0.astype(np.float64), 0.0)

    no_improve = 0
    lr = lr_init
    v = np.zeros(n)
    cur_mom = momentum
    temp = 0.0001
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        
        # Select top peaks with softmax weighting
        k = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Multi-peak subgradient
        sg = compute_multi_peak_subgrad(a, peaks, w)
        
        v = cur_mom * v - lr * sg
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        obj = conv.max()

            best_obj = obj
            best_a = a.copy()

                v = np.zeros(n)
                lr = lr_init * 0.5
                cur_mom = 0.91

def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent: for each peak, try to reduce it by adjusting individual elements."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Find top 3 peaks
        k = min(3, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        
        improved = False
        for k_peak in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            # Try to reduce conv[k_peak] by adjusting elements
            # The elements contributing to conv[k_peak] are a[i] * a[k_peak - i]
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            
            for _ in range(5):
                # Pick a random pair
                i = rng.integers(js, je + 1)
                j = k_peak - i
                if i == j:
                    continue
                if i < 0 or i >= n or j < 0 or j >= n:
                    continue
                
                # Try to reduce a[i]*a[j] by transferring mass
                # Option 1: decrease a[i], increase a[j] (or vice versa)
                # Try small transfers
                delta = rng.uniform(0.0005, 0.005) * min(a[i], a[j])
                if delta < 1e-10:
                    continue
                
                # Try: a[i] -= delta, a[j] += delta (keep sum)
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                obj_try = fast_convolve(a_try).max()
                if obj_try < conv.max() - 1e-12:
                    a = a_try
                    conv = fast_convolve(a)
                    improved = True
                    if conv.max() < best_obj - 1e-12:
                        best_obj = conv.max()
                        best_a = a.copy()
        
        if not improved:
            # Try random perturbations
            idx = rng.integers(0, n)
            delta = rng.uniform(-0.002, 0.002)
            a_try = a.copy()
            a_try[idx] += delta
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = fast_convolve(a_try).max()
            if obj_try < conv.max() - 1e-12:
                a = a_try
                conv = fast_convolve(a)
                if conv.max() < best_obj - 1e-12:
                    best_obj = conv.max()
                    best_a = a.copy()
            else:
                # No improvement, break after a while
                break
    
    return best_a, best_obj
