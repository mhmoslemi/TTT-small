"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast coordinate descent with incremental conv updates as the primary
method, combined with LP-based Frank-Wolfe for larger jumps, multi-length
exploration, and adaptive restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_descent(f0, time_limit, rng, verbose=False, restart_every=100, perturb_scale=0.008):
    """
    Fast coordinate descent with incremental convolution updates.
    Minimizes max(a*a) subject to sum(a)=1, a>=0.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            aj = f[j]
            if aj < 1e-10:
                continue

            # Precompute outside max (conv positions not affected by changing f[j])
            outside_max = 0.0
            if j > 0:
                outside_max = conv[:j].max()
            if j + n < conv_len:
                tmp = conv[j + n:conv_len].max()
                if tmp > outside_max:
                    outside_max = tmp

            # Grid search for optimal delta
            best_d = 0.0
            best_peak = conv.max()

            # Negative deltas (reduce f[j])
            for frac in (1.0, 0.6, 0.4, 0.2, 0.1):
                d = -aj * frac
                affected = conv[j:j + n] + 2.0 * d * f
                affected[j] += d * d
                pk = affected.max()
                if pk < best_peak or (pk > best_peak and pk > outside_max):
                    # Actually we want the minimum peak
                    pass
                peak = pk if pk > outside_max else outside_max
                if peak < best_peak:
                    best_peak = peak
                    best_d = d

            # Small positive deltas
            for frac in (0.05, 0.15):
                d = aj * frac
                affected = conv[j:j + n] + 2.0 * d * f
                affected[j] += d * d
                pk = affected.max()
                peak = pk if pk > outside_max else outside_max
                if peak < best_peak:
                    best_peak = peak
                    best_d = d

            if best_d != 0.0:
                conv[j:j + n] += 2.0 * best_d * f
                conv[2 * j] += best_d * best_d
                f[j] += best_d
                if f[j] < 0:
                    f[j] = 0.0

        # End of sweep: normalize
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
            conv = fast_convolve(f)
        else:
            f = f / s
            conv = fast_convolve(f)

        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

            conv = fast_convolve(f)
            no_improve = 0
            perturb_scale *= 0.8
            if perturb_scale < 0.001:
                perturb_scale = 0.001

    return best_f, best_obj


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, B=None, top_k=None):
    """Solve: min t s.t. t >= conv_f[k] + 2*(B*g)[k] for all k, g >= -f, sum(g)=0."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[0] = 1.0

    n_cons = 2 * n - 1
    if top_k is not None and top_k < n_cons:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, 0] = -1.0
        A_ub[:, 1:] = 2.0 * B[peaks, :]
        b_ub = -conv_f[peaks]
    else:
        A_ub = np.zeros((n_cons, n + 1))
        A_ub[:, 0] = -1.0
        A_ub[:, 1:] = 2.0 * B
        b_ub = -conv_f

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def line_search(f, target, current_obj, max_alpha=1.0):
    """Find best alpha for f_new = (1-alpha)*f + alpha*target."""
    best_alpha = 0.0
    best_obj = current_obj

    for alpha in np.linspace(0.05, max_alpha, 10):
        f_new = (1.0 - alpha) * f + alpha * target
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha

    if best_alpha > 0:
        lo = max(0.01, best_alpha * 0.5)
        hi = min(max_alpha, best_alpha * 1.5)
        for _ in range(6):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-15:
                break
            f_new = f_new / s
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid

    return best_alpha, best_obj

        else:
            g = solve_linearized_lp(f, conv)

                no_improve = 0

        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1

            no_improve += 1
            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
