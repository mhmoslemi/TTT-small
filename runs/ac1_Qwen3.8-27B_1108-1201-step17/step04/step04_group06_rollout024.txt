Key insights:

My approach combines:
- **Smoothed max objective** (log-sum-exp) for better gradient information

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed-max multiplicative weights
with adaptive temperature, diverse structured initializations, multi-length search,
coordinate descent polishing, and CVXPY QCQP for small instances.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smoothed_max_subgradient(a, c, beta, min_weight=1e-10):
    """
    Compute the weighted subgradient of the smoothed max of c.
    gradient = sum_k w_k * grad(b_k) where w_k = exp(beta*c_k)/sum(exp(beta*c_j))
    and grad(b_k)[i] = 2*a[k-i] if valid.
    """
    n = len(a)
    # Shift to avoid overflow
    shifted = beta * c - beta * c.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform over top values
        top_idx = np.argsort(c)[-max(1, n // 4):]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / len(top_idx)
        w_sum = 1.0
    else:
        w /= w_sum

    # Compute weighted gradient
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < min_weight:
            continue
        # b_k = sum_i a[i]*a[k-i], valid i: max(0,k-n+1) <= i <= min(n-1,k)
        imin = max(0, k - n + 1)
        imax = min(n - 1, k)
        if imin > imax:
            continue
        # grad[i] += 2 * a[k-i] for valid i
        # a[k-i] for i=imin..imax is a[k-imin], a[k-imin-1], ..., a[k-imax]
        # which is a[k-imax : k-imin+1] reversed
        vals = a[k - imax:k - imin + 1][::-1]
        grad[imin:imax + 1] += 2.0 * w[k] * vals

    return grad


def multiplicative_weights_smoothed(a0, time_limit, beta0=5.0, beta_end=500.0,
                                     eta0=0.8, eta_end=0.05, verbose=False):
    """
    Multiplicative weights on the simplex with smoothed max objective.
    a is normalized to sum=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    # Running average
    avg_a = a.copy()
    count = 0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        # Interpolate parameters
        beta = beta0 * (beta_end / beta0) ** frac
        eta = eta0 + (eta_end - eta0) * frac

        # Compute smoothed max subgradient
        grad = smoothed_max_subgradient(a, c, beta)
        grad_max = grad.max()
        if grad_max < 1e-15:
            break

        # Multiplicative update
        log_update = -eta * grad / grad_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    MW iter {it}: obj={best_obj:.10f}, beta={beta:.1f}, eta={eta:.4f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        # Check running average every 500 iters
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            obj_avg = c_avg.max()
            if obj_avg < best_obj - 1e-14:
                best_obj = obj_avg
                best_a = avg_a.copy()
                no_improve = 0

        # Stagnation: perturb and reset
        if no_improve > 5000:
            rng_local = np.random.default_rng(hash((it, n)) % (2**31))
            noise = rng_local.standard_normal(n) * 0.003
            a = np.maximum(best_a + noise, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            avg_a = a.copy()
            count = 0

    # Final check of average
    c_avg = fast_convolve(avg_a)
    if c_avg.max() < best_obj:
        best_obj = c_avg.max()
        best_a = avg_a.copy()

    return best_a.copy(), best_obj


def coordinate_descent_polish(a0, time_limit, n_trials_per_iter=50):
    """Coordinate descent: try scaling individual coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Pick random coordinates
        n_pert = max(1, min(n_trials_per_iter, n // 8))
        coords = rng.choice(n, size=n_pert, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting small positive value
                for scale in [0.01, 0.05, 0.1] / n:
                    a[i] = scale
                    s2 = a.sum()
                    a = a / s2
                    c2 = fast_convolve(a)
                    if c2.max() < best_obj - 1e-14:
                        best_obj = c2.max()
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a = best_a.copy()
                continue

            # Try multiplicative perturbations
            for factor in [0.5, 0.7, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 0.3]:
                a[i] = old_val * factor
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-14:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a.copy(), best_obj


def generate_structured_inits(n, rng, best_a=None):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Powers of a base (geometric)
    for base in [0.5, 0.7, 0.8, 0.9]:
        geo = np.power(base, x / n)
        inits.append(geo / geo.sum())
        geo_r = np.power(base, (n - 1 - x) / n)
        inits.append(geo_r / geo_r.sum())

    # Two bumps
    for ratio in [0.3, 0.5, 0.7]:
        center1 = n * ratio / 2
        center2 = n * (1 - ratio) / 2
        width = n * 0.1
        b1 = np.exp(-0.5 * ((x - center1) / width) ** 2)
        b2 = np.exp(-0.5 * ((x - center2) / width) ** 2)
        inits.append((b1 + b2) / (b1 + b2).sum())

    # Single bump (Gaussian)
    for width_frac in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n / 2) / (width_frac * n)) ** 2)
        inits.append(g / g.sum())

    # Sine-based
    for freq in [1, 2, 3]:
        sine = 0.5 + 0.5 * np.sin(np.pi * x / n * freq)
        inits.append(sine / sine.sum())

    # Cosine bump at edges
    edge = np.cos(np.pi * x / (2 * n)) ** 2
    inits.append(edge / edge.sum())

    # Random Dirichlet with various concentrations
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))

    # Nearly binary (random mask smoothed)
    mask = (rng.random(n) > 0.5).astype(float)
    # Smooth it
    kernel = np.ones(5) / 5
    smoothed = np.convolve(mask, kernel, mode='same')
    smoothed = np.maximum(smoothed, 0)
    if smoothed.sum() > 0:
        inits.append(smoothed / smoothed.sum())

    # Perturbed best
    if best_a is not None and len(best_a) == n:
        for scale in [0.01, 0.05, 0.1]:
            p = np.maximum(best_a + rng.standard_normal(n) * scale, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    return inits


def try_cvxpy_qcqp(a0, time_limit=120):
    """Solve QCQP with CVXPY for small n."""
    import cvxpy as cp

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    a = a / s
    n = len(a)

    if n > 200:
        return None

    try:
        x = cp.Variable(n)
        t = cp.Variable()

        constraints = [
            cp.sum(x) == 1,
            x >= 0
        ]

        # Add constraints for all lags
        for k in range(2 * n - 1):
            imin = max(0, k - n + 1)
            imax = min(n - 1, k)
            expr = cp.Constant(0)
            for i in range(imin, imax + 1):
                expr = expr + x[i] * x[k - i]
            constraints.append(expr <= t)

        prob = cp.Problem(cp.Minimize(t), constraints)

        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    break
            except Exception:
                continue

        if prob.status in ['optimal', 'optimal_inaccurate'] and x.value is not None:
            result = np.maximum(x.value, 0)
            rs = result.sum()
            if rs > 1e-10:
                return result / rs
    except Exception:
        pass

    return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-length, multi-init multiplicative weights with
    smoothed max, coordinate descent polishing, and CVXPY for small n.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        # Normalize for evaluation (score is scale-invariant, but ensure sum > 0.01)
        a_eval = a.copy()
        if a_eval.sum() < 0.01:
            a_eval = a_eval / a_eval.sum() * 1.0
        score = float(evaluate_sequence(a_eval.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy() / a.sum()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent if available
    parent = None
    parent_n = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_n = len(parent)
            parent_norm = parent / parent.sum()
            parent_score = evaluate_sequence(parent_norm.tolist())
            print(f"Parent: n={parent_n}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent_norm.copy()
    except Exception:
        pass

    if parent is None:
        parent_n = 256
        parent = np.ones(parent_n) / parent_n
        best_score = float('inf')
        best_seq = None
        print(f"No parent, starting with uniform n={parent_n}")

    print(f"Starting: n={parent_n}, best_score={best_score:.8f}")

    # Define candidate lengths to try
    candidate_lengths = []
    if parent_n is not None:
        candidate_lengths.append(parent_n)
    for n in [64, 96, 128, 160, 192, 256, 320, 384, 512, 768]:
        if n not in candidate_lengths and 32 <= n <= 1024:
            candidate_lengths.append(n)
    candidate_lengths = sorted(set(candidate_lengths))

    print(f"Testing lengths: {candidate_lengths}")

    # Phase 1: Quick scan over lengths with uniform init
    print("\n=== Phase 1: Length scan (uniform) ===")
    for n in candidate_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        a = np.ones(n) / n
        c = fast_convolve(a)
        obj = c.max()
        score = 2 * n * obj
        print(f"  n={n}: uniform score={score:.6f}")
        try_seq(a, f"uniform_n{n}")

    # Phase 2: MW optimization on each length with diverse inits
    print(f"\n=== Phase 2: MW Optimization ===")
    # Allocate time: more time for promising lengths
    lengths_to_optimize = []
    for n in candidate_lengths:
        uniform_obj = max(fast_convolve(np.ones(n) / n))
        score = 2 * n * uniform_obj
        lengths_to_optimize.append((score, n))
    lengths_to_optimize.sort()

    # Focus on top lengths (lowest uniform score = 2 for all, so use parent_n preference)
    # Actually all uniform scores are 2, so prioritize parent_n and nearby
    lengths_priority = sorted(candidate_lengths, key=lambda n: abs(n - (parent_n or 256)))

    total_remaining = deadline - time.monotonic() - 200
    n_lengths = min(len(lengths_priority), 6)  # Limit to 6 lengths
    time_per_length = total_remaining / max(1, n_lengths)

    for li, n in enumerate(lengths_priority[:n_lengths]):
        rem = deadline - time.monotonic() - 100
        if rem < 20:
            break
        t_alloc = min(time_per_length, max(30, rem / max(1, n_lengths - li)))

        print(f"\n  Optimizing n={n}, alloc={t_alloc:.0f}s")

        # Get init: use parent if available and right length
        if parent is not None and len(parent) == n:
            init = parent / parent.sum()
        else:
            init = np.ones(n) / n

        # Use best_seq if it has the right length
        if best_seq is not None and len(best_seq) == n:
            init = best_seq

        # Run MW
        a_opt, obj = multiplicative_weights_smoothed(
            init, time_limit=t_alloc * 0.6,
            beta0=5.0, beta_end=300.0, eta0=1.0, eta_end=0.05,
            verbose=(li == 0)
        )
        try_seq(a_opt, f"mw_n{n}")

        # Try a second config with different params
        if rem > t_alloc * 0.3:
            a_opt2, obj2 = multiplicative_weights_smoothed(
                a_opt, time_limit=min(t_alloc * 0.3, rem - t_alloc * 0.6),
                beta0=10.0, beta_end=500.0, eta0=0.5, eta_end=0.03
            )
            try_seq(a_opt2, f"mw2_n{n}")

        # Coordinate descent polish
        rem2 = deadline - time.monotonic() - 80
        if rem2 > 10:
            a_poly, _ = coordinate_descent_polish(a_opt, time_limit=min(rem2, 30))
            try_seq(a_poly, f"cd_n{n}")

    # Phase 3: Diverse inits on best length
    print(f"\n=== Phase 3: Diverse Inits ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        inits = generate_structured_inits(n_best, rng, best_a=best_seq)

        t0 = time.monotonic()
        time_budget_p3 = min(300, (deadline - time.monotonic() - 100) * 0.5)
        t_per_init = time_budget_p3 / max(1, len(inits))

        for ii, init in enumerate(inits):
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_alloc = min(max(10, t_per_init), rem / max(1, len(inits) - ii))
            if t_alloc < 5:
                break

            a_opt, _ = multiplicative_weights_smoothed(
                init, time_limit=t_alloc,
                beta0=3.0, beta_end=200.0, eta0=0.8, eta_end=0.05
            )
            try_seq(a_opt, f"init{ii}_n{n_best}")

    # Phase 4: CVXPY on small n
    print(f"\n=== Phase 4: CVXPY ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None and len(best_seq) <= 128:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            print(f"  Solving QCQP n={len(best_seq)}")
            result = try_cvxpy_qcqp(best_seq, time_limit=min(rem, 120))
            if result is not None:
                try_seq(result, "cvxpy")

    # Phase 5: Final polishing
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_final, _ = coordinate_descent_polish(

                    a_final, time_limit=min(rem2, 30),
                    beta0=100.0, beta_end=1000.0, eta0=0.3, eta_end=0.02
                )
                try_seq(a_final2, "final_mw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    a_final_check = best_seq / best_seq.sum()
    fs = float(evaluate_sequence(a_final_check.tolist()))
    if np.isfinite(fs):
        best_score = fs
        best_seq = a_final_check

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
