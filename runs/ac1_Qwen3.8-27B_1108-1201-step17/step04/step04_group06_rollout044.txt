"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using theoretical analysis and
exhaustive optimization.

Key insight: By Cauchy-Schwarz, sum(a_j^2) >= (sum(a_j))^2/n, and since

import numpy as np
import time
import scipy.optimize as so


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def score_from_normalized(a):
    """Compute score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def multiplicative_weights_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                                    temp_decay=0.9998, eta_decay=0.99995,
                                    top_k_frac=0.3):
    """Exponentiated gradient with softmax-weighted subgradients."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    eta = eta0
    temp = temp0
    no_improve = 0
    it = 0
    t0 = time.monotonic()

        it += 1

        # Compute softmax-weighted subgradient
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
            peek = c[indices]
        else:
            indices = np.arange(len(c))
            peek = c.copy()

        scaled -= scaled.max()

            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)

            best_a = a_new.copy()
            no_improve = 0
        else:

        c = c_new

def projected_gradient_optimize(a0, time_limit, lr0=0.01, decay=0.999):
    """Projected gradient descent on the simplex {a>=0, sum(a)=1}."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    # Find argmax of c
    best_obj = c.max()
    best_a = a.copy()

    lr = lr0
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Subgradient of max(c) at argmax
        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        sg[js:je + 1] = 2.0 * a[k_star - je:k_star - js + 1][::-1]

        # Gradient step
        a_new = a - lr * sg / sg.max()
        # Project onto simplex (a >= 0, sum = 1)
        a_new = project_simplex(a_new)

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95

        a = a_new
        c = c_new
        lr = max(lr0 * decay, 0.001)

        if no_improve > 3000:
            break

    return best_a.copy(), best_obj


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def scipy_optimize(a0, time_limit):
    """Use scipy's SLSQP to minimize the score."""
    a = np.maximum(a0.astype(np.float64), 0.0)

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = score_from_normalized(a)

    def obj(x):
        if time.monotonic() - t0 > time_limit:
            return float('inf')
        x = np.maximum(x, 0)
        xs = x.sum()
        if xs < 1e-10:
            return 1e6
        x = x / xs
        c = fast_convolve(x)
        return 2.0 * n * c.max()

    def obj_grad(x):
        x = np.maximum(x, 0)
        xs = x.sum()
        if xs < 1e-10:
            return np.zeros(len(x))
        x = x / xs
        c = fast_convolve(x)
        k = int(np.argmax(c))
        sg = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] = 2.0 * x[k - je:k - js + 1][::-1]
        return 2.0 * n * sg

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                    'jac': lambda x: np.ones(len(x))}]
    bounds = [(0, None)] * n

    try:
        result = so.minimize(obj, a, method='SLSQP', jac=obj_grad,
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 500, 'ftol': 1e-12})
        if result fun < best_obj:
            best_a = np.maximum(result.x, 0)
            bs = best_a.sum()
            if bs > 1e-10:
                best_a = best_a / bs
            best_obj = score_from_normalized(best_a)
    except Exception:
        pass

    return best_a.copy(), best_obj


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    inits.append(rc / rc.sum())

    # Won (sine squared)
    w = np.sin(np.pi * x / (n - 1)) ** 2
    inits.append(w / w.sum())

    # Hann window
    hn = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits.append(hn / hn.sum())

    # Triangular
    tri = 1.0 - np.abs(2.0 * x / (n - 1) - 1.0)
    inits.append(tri / tri.sum())

    # Edge-loaded
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        e = np.exp(-alpha * (x - (n-1)/2) ** 2 / n)
        inits.append(e / e.sum())

    # Dirichlet random
    for _ in range(5):
        r = rng.dirichlet(np.ones(n) * 0.5)
        inits.append(r)

    # Perturbed parent
    if parent is not None and len(parent) == n:
        for scale in [0.01, 0.05, 0.1]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    return inits
