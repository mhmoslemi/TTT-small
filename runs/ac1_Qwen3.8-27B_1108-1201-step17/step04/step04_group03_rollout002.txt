def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration with diverse initializations
2. Soft-peak reduction (p-norm based) for smooth optimization

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def soft_obj(a, p=20):
    """Soft objective using p-norm of conv."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * (np.sum(conv**p)) ** (1.0/p)

def soft_peak_reduction(a, time_limit, lr_init=0.02, p=20, p_decay=1.002):

    Peak reduction using p-norm objective.
    The p-norm is smooth and accounts for all peaks, not just the max.
    As p increases, it approaches the hard max.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_p = p
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        # Gradient of p-norm: d/d(a_i) (sum conv^p)^(1/p)
        # = (1/p) * (sum conv^p)^(1/p - 1) * p * sum_j conv_j^(p-1) * d(conv_j)/d(a_i)
        # = (sum conv^p)^(1/p - 1) * sum_j conv_j^(p-1) * d(conv_j)/d(a_i)
        
        # d(conv_k)/d(a_i) = 2*a_{k-i} (if valid) or a_i (if k-i=i)
        # More precisely: d(conv_k)/d(a_i) = a_{k-i} + a_{k-i} = 2*a_{k-i} (if k != 2i)
        # or a_i (if k = 2i)
        
        # Compute weighted gradient
        w = conv ** (cur_p - 1)  # weights for each conv value

        if w_sum < 1e-30:

def hard_peak_reduction(a, time_limit, lr_init=0.01, top_k=10, temp=0.001):

def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Build constraint matrix efficiently
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except:
            break
        
        if not res.success:
            break
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj / (2.0 * n) - 1e-14:
                best_obj = 2.0 * n * M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj

    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    
    # Bimodal

    g = np.exp(-0.5 * ((x - n*0.3) / sig) ** 2) + np.exp(-0.5 * ((x - n*0.7) / sig) ** 2)
    inits.append(g / g.sum())
    
    # Triangular
    tri = np.minimum(x + 1, n - x)

    inits.append(c / c.sum())
    
    # Edge-weighted
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    
    # Sine wave
    s1 = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x / n)
    inits.append(s1 / s1.sum())
    
    # Ragged (alternating high/low)
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(alt / alt.sum())
    
    return inits
