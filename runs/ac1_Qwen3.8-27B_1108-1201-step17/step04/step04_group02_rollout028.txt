import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_correlate(a, b):
    """Compute (a*b)[k] = sum_j a[j]*b[k-j] for k=0..len(a)+len(b)-2"""
    # This is the linear cross-correlation
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b[::-1], fft_size)
    return np.fft.irfft(A * B, fft_size)[:n]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def linear_conv(a, b):
    """Linear convolution of two sequences via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build the constraint matrix for the LP oracle.
    A[k, m] = f[k-m] if 0 <= k-m < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = f
    return A


def lp_oracle(f, M, A=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(f)
    if A is None:
        A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 45})
        if res.success and res.x is not None:
            s = res.x.sum()
            if s > 1e-15:
                return res.x / s
    except Exception:
        pass
    return None


def quadratic_line_search(cv_ff, cv_fg, cv_gg, M):
    """Given precomputed convolutions, find best alpha in [0,1] minimizing
    max_k [(1-a)^2 * ff[k] + a(1-a) * (fg[k]+fg[2n-2-k]) + a^2 * gg[k]]
    using golden-section search."""
    n2 = len(cv_ff) - 1  # 2n-2
    cv_fg_sym = cv_fg + cv_fg[::-1]  # fg[k] + fg[2n-2-k]

    def h(alpha):
        a2 = alpha * alpha
        a12 = alpha * (1.0 - alpha)
        val = (1.0 - alpha) ** 2 * cv_ff + a12 * cv_fg_sym + a2 * cv_gg
        return np.max(val)

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = h(c1)
    f2 = h(c2)

    for _ in range(60):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = h(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = h(c2)

    best_alpha = (lo + hi) / 2.0
    # Also check endpoints and a few extra points
    for a in [0.0, 1.0, best_alpha]:
        pass
    return best_alpha, h(best_alpha)


def fw_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle and quadratic-exact line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    cv_ff = fast_convolve(f)
    M = cv_ff.max()
    best_obj = M
    best_f = f.copy()
    A = build_lp_matrix(f)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = lp_oracle(f, M, A)
        if g is None:
            # Try with slightly larger M
            g = lp_oracle(f, M * 1.1, A)
            if g is None:
                no_improve += 1
                if no_improve > 30:
                    break
                continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Precompute convolutions for quadratic line search
        cv_fg = linear_conv(f, g)
        cv_gg = fast_convolve(g)

        best_alpha, best_val = quadratic_line_search(cv_ff, cv_fg, cv_gg, M)

        # Also try a few additional alphas for safety
        for a in [best_alpha * 0.5, best_alpha * 1.5, 0.3, 0.5, 0.7]:
            if a < 0 or a > 1:
                continue
            a2 = a * a
            a12 = a * (1.0 - a)
            val = np.max((1 - a) ** 2 * cv_ff + a12 * (cv_fg + cv_fg[::-1]) + a2 * cv_gg)
            if val < best_val:
                best_val = val
                best_alpha = a

        if best_alpha > 1e-10 and best_val < M - 1e-14:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            cv_ff = fast_convolve(f)
            M = cv_ff.max()
            A = build_lp_matrix(f)

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            cv_ff = fast_convolve(f)
            M = cv_ff.max()
            A = build_lp_matrix(f)
            no_improve = 0

    return best_f, best_obj


def smooth_momentum_optimizer(f0, time_limit, beta_init=100, beta_final=5000,
                              lr_init=0.08, mom_init=0.85):
    """
    Nesterov-style accelerated gradient with log-sum-exp smoothing.
    Minimizes log(sum exp(beta*(f*f)[k])) subject to f in simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    y = f.copy()
    t_k = 1.0
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 1:
        it += 1
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        beta = beta_init * (beta_final / beta_init) ** frac
        lr = lr_init / (1.0 + 0.02 * it)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        mu = (1.0 - t_k) / t_new
        y = f + mu * (y - f)
        y = project_simplex(y)

        # Gradient of log-sum-exp(beta * (f*f)) at y
        conv_y = fast_convolve(y)
        m = conv_y.max()
        shifted = beta * (conv_y - m)
        shifted -= shifted.max()
        p = np.exp(shifted)
        p /= p.sum()

        # grad_i = 2 * beta * sum_k p[k] * f[k-i] = 2*beta * (p * y_flipped correlation)
        # Using FFT: cross-correlation
        fft_size = 1 << (2 * n - 2).bit_length()
        P = np.fft.rfft(p, fft_size)
        Y = np.fft.rfft(y[:n], fft_size)
        cc = np.fft.irfft(P * np.conj(Y), fft_size)[:2 * n - 1]
        grad = 2.0 * beta * cc[:n]

        f_new = project_simplex(y - lr * grad)
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        t_k = t_new

        if no_improve > 300:
            rng = np.random.default_rng(it * 7 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            t_k = 1.0
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())

    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    q = (x - n / 2) ** 2 + 1.0
    inits.append(q / q.sum())

    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())

    e2 = np.exp(-x[::-1] / (n / 3.0))
    inits.append(e2 / e2.sum())

    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())

    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    sym = 1.0 / (1.0 + ((x - n / 2) / (n / 4)) ** 4)
    inits.append(sym / sym.sum())

    g = np.exp(-((x - n / 2) / (n / 3)) ** 2)
    inits.append(g / g.sum())

    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    bim = np.exp(-((x - n * 0.25) / (n / 8)) ** 2) + np.exp(-((x - n * 0.75) / (n / 8)) ** 2)
    inits.append(bim / bim.sum())

    pw = (x + 1.0) ** 0.3
    inits.append(pw / pw.sum())

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0.0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: hybrid LP-Frank-Wolfe + smoothed Nesterov + multi-size exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    parent_n = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_n = len(parent)
            print(f"Parent loaded: n={parent_n}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # === Phase 1: Parent optimization with FW-LP ===
    print("=== Phase 1: Parent FW-LP ===")
    if parent is not None:
        t1 = min(300, deadline - time.monotonic() - 600)
        if t1 > 30:
            print(f"  FW-LP on parent (n={parent_n}), t={t1:.0f}s")
            a_opt, obj = fw_optimize(parent, time_limit=t1)
            try_seq(a_opt, "parent_fw")

            rem = deadline - time.monotonic() - 500
            if rem > 20:
                print(f"  Nesterov follow-up, t={min(rem, 80):.0f}s")
                a_opt2, _ = smooth_momentum_optimizer(a_opt, time_limit=min(rem, 80),
                                                      beta_init=200, beta_final=5000)
                try_seq(a_opt2, "parent_nest")

    # === Phase 2: Multi-size exploration ===
    print("\n=== Phase 2: Multi-size Exploration ===")

    sizes = [256, 320, 384, 448, 512]
    if parent is not None:
        if parent_n not in sizes:
            sizes.append(parent_n)
        sizes = sorted(set(sizes))

    for n in sizes:
        if time.monotonic() > deadline - 250:
            break
        if parent is not None and n == parent_n and best_seq is not None and len(best_seq) == n:
            continue

        inits = generate_inits(n, parent, rng)
        rem = deadline - time.monotonic() - 200
        n_remaining = len(sizes) - sizes.index(n)
        t_per = max(15, rem / max(1, len(inits) * n_remaining))
        t_per = min(t_per, 45)

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break

            a_opt, _ = smooth_momentum_optimizer(f_init, time_limit=t_alloc,
                                                 beta_init=50, beta_final=3000,
                                                 lr_init=0.06)
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 2 best: {best_score:.8f} (n={len(best_seq) if best_seq is not None else '?'})")

    # === Phase 3: Deep FW-LP on best ===
    print(f"\n=== Phase 3: Deep FW-LP ===")

    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(6):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(rem // max(1, 6 - p), 120)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            elif p == 2:
                f_init = project_simplex(0.92 * best_seq + 0.08 * (np.ones(n_best) / n_best))
            elif p == 3:
                f_init = project_simplex(0.85 * best_seq + 0.15 * (best_seq[::-1]))
            elif p == 4:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)
            else:
                # Try a slightly different size by resampling
                n2 = n_best
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.015)

            print(f"  Deep FW-{p}, n={n_best}, t={t_alloc:.0f}s")
            a_opt, obj = fw_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_fw_{p}")

    # === Phase 4: Nesterov deep runs ===
    print(f"\n=== Phase 4: Nesterov Deep ===")

    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 40
            if rem < 15:
                break
            t_alloc = min(rem // max(1, 3 - p), 60)

            if p == 0:
                f_init = best_seq.copy()
                cfg = (500, 10000, 0.03, 0.9)
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                cfg = (2000, 20000, 0.01, 0.95)
            else:
                f_init = project_simplex(0.95 * best_seq + 0.05 * (np.ones(n_best) / n_best))
                cfg = (1000, 15000, 0.02, 0.92)

            print(f"  Nesterov deep-{p}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_momentum_optimizer(f_init, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_nest_{p}")

    # === Phase 5: Try slightly different sizes around best ===
    print(f"\n=== Phase 5: Size Search ===")

    if best_seq is not None:
        n_best = len(best_seq)
        nearby_sizes = [s for s in [n_best - 32, n_best - 16, n_best + 16, n_best + 32]
                       if 128 <= s <= 768]
        for n_alt in nearby_sizes:
            if n_alt == n_best:
                continue
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            f_init = np.interp(np.arange(n_alt), np.linspace(0, 1, n_best), best_seq)
            f_init = np.maximum(f_init, 0.0)
            f_init = f_init / f_init.sum()
            t_alloc = min(rem // 2, 30)
            a_opt, _ = smooth_momentum_optimizer(f_init, time_limit=t_alloc,
                                                 beta_init=200, beta_final=5000)
            try_seq(a_opt, f"size_{n_alt}")
            if best_seq is not None and len(best_seq) == n_alt:
                rem2 = deadline - time.monotonic() - 20
                if rem2 > 10:
                    a_opt2, _ = fw_optimize(best_seq, time_limit=min(rem2, 20))
                    try_seq(a_opt2, f"size_{n_alt}_fw")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = fw_optimize(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "final_fw")

        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = smooth_momentum_optimizer(best_seq, time_limit=min(rem, 25),
                                                 beta_init=5000, beta_final=20000,
                                                 lr_init=0.005, mom_init=0.95)
            try_seq(a_opt, "final_nest")

        # Micro perturbations
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0005)
                a_opt, _ = smooth_momentum_optimizer(f_pert, time_limit=min(rem, 8),
                                                     beta_init=10000, beta_final=30000,
                                                     lr_init=0.002, mom_init=0.97)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
