Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric parameterization (a[i] = a[n-1-i]) with smoothed gradient + L-BFGS-B

import numpy as np
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def sym_convolve(b):
    """Compute auto-convolution of symmetric sequence from first half b.
    b has length m = n//2, and a[i] = b[i] for i < m, a[i] = b[n-1-i] for i >= m."""
    n = 2 * len(b)
    a = np.concatenate([b, b[::-1]])
    return fast_convolve(a)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f):
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B

def solve_linearized_lp(f, conv_f, B=None):
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None

def solve_maxsum_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None

def line_search_for_direction(f, target, current_obj, max_alpha=1.0):
    best_alpha = 0.0
    best_obj = current_obj
    alphas = np.linspace(0.05, max_alpha, 12)
    for alpha in alphas:
        f_new = (1.0 - alpha) * f + alpha * target
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    if best_alpha > 0:
        lo = max(0.01, best_alpha * 0.6)
        hi = min(max_alpha, best_alpha * 1.4)
        for _ in range(8):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            s = f_new.sum()
            if s < 1e-15:
                break
            f_new = f_new / s
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    return best_alpha, best_obj

def smooth_subgrad_optimize(f0, time_limit, lr=0.003, mom=0.93, top_k=40, temp=0.0001):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            sg[idx] += 2.0 * w[i] * f[k - idx]

        cur_temp *= 1.004

    grad *= 2.0

    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(m)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        for j in range(m):
            idx1 = k - j
            idx2 = k - (n - 1 - j)
            contrib = 0.0
            if 0 <= idx1 <= n - 1:
                contrib += a_norm[idx1]
            if 0 <= idx2 <= n - 1:
                contrib += a_norm[idx2]
            grad[j] += w[i] * contrib

    grad *= 4.0 * n / s
    return grad


def sym_nesterov_descent(b0, time_limit, lr_init=0.1, temp_init=0.01,
                         top_k=30):
    b = np.maximum(np.asarray(b0, dtype=np.float64), 1e-8)
    m = len(b)
    b = b / b.sum() * 0.5
    n = 2 * m

    t0 = time.monotonic()
    cur_lr = lr_init
    cur_temp = temp_init
    y = b.copy()
    best_b = b.copy()
    best_obj = sym_objective_val(b)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_val = 1.0 + it
        beta = (1.0 - 1.0 / t_val) / (1.0 + 1.0 / (t_val + 1.0))
        y = b + beta * (b - y)
        y = np.maximum(y, 1e-10)
        y = y / y.sum() * 0.5

        grad = sym_gradient(y, temp=cur_temp, top_k=top_k)
        b_new = y - cur_lr * grad
        b_new = np.maximum(b_new, 1e-8)
        b_new = b_new / b_new.sum() * 0.5
        b = b_new

        obj = sym_objective_val(b)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_b = b.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.999
        cur_temp *= 1.005

        if no_improve > 200:
            scale = 0.002 + 0.002 * min(no_improve // 100, 5)
            b = best_b + rng.standard_normal(m) * scale
            b = np.maximum(b, 1e-8)
            b = b / b.sum() * 0.5
            y = b.copy()
            cur_lr = max(lr_init * 0.5, 0.01)
            no_improve = 0

    return best_b, best_obj


def coord_descent_sym(b0, time_limit):
    b = np.maximum(np.asarray(b0, dtype=np.float64), 1e-8)
    m = len(b)
    b = b / b.sum() * 0.5
    t0 = time.monotonic()
    best_obj = sym_objective_val(b)
    best_b = b.copy()
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(m):
            if time.monotonic() - t0 > time_limit:
                break
            base = b[j]
            best_delta = 0.0
            best_val = best_obj
            for frac in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3,
                         -0.1, 0.1, -0.02, 0.02, -0.08, 0.08]:
                delta = frac * base
                new_b = b.copy()
                new_b[j] = max(base + delta, 1e-8)
                new_b = new_b / new_b.sum() * 0.5
                obj = sym_objective_val(new_b)
                if obj < best_val - 1e-14:
                    best_val = obj
                    best_delta = delta
            if best_delta != 0.0:
                b[j] = max(b[j] + best_delta, 1e-8)
                b = b / b.sum() * 0.5
                if best_val < best_obj - 1e-14:
                    best_obj = best_val
                    best_b = b.copy()
                    improved = True
        if not improved:
            b = best_b + rng.standard_normal(m) * 0.001
            b = np.maximum(b, 1e-8)
            b = b / b.sum() * 0.5

    return best_b, best_obj


def fg_optimize(f0, time_limit, lr=0.003, mom=0.9, top_k=40, temp=0.0001):
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        cur_lr *= 0.9995
        cur_temp *= 1.004
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0005)
            no_improve = 0
    return best_f, best_obj


def gen_inits(n, rng):
    inits = []
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        inits.append(g / g.sum())
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / n)) + 0.01
        inits.append(c / c.sum())
    for alpha_b in [1.5, 2.0, 3.0, 4.0]:
        b = (x + 1) ** alpha_b * (max(n - x, 1)) ** alpha_b
        inits.append(b / b.sum())
    edge_w = 1.0 / (0.3 * np.abs(x - mid) / n + 0.05)
    inits.append(edge_w / edge_w.sum())
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = a.tolist()
        score = float(evaluate_sequence(a))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = np.array(a, dtype=np.float64)
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent, using uniform n=384")

    n_parent = len(parent)

    # Phase 1: Quick exploration of initializations
    print("\n=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    candidate_list = [("parent", parent)]
    for n_try in [n_parent - 16, n_parent, n_parent + 16, 320, 448]:
        if n_try < 64 or n_try > 1024:
            continue
        inits = gen_inits(n_try, rng)
        for i, ini in enumerate(inits[:4]):
            candidate_list.append((f"n{n_try}_i{i}", ini))

    for label, cand in candidate_list:
        if time.monotonic() > deadline - 400:
            break
        try_seq(cand, f"init_{label}")

    if best_seq is None:
        best_seq = parent.copy()
        try_seq(best_seq, "fallback")

    n_best = len(best_seq)
    print(f"  Phase 1 best: {best_score:.10f} (n={n_best})")

    # Phase 2: Symmetric Nesterov descent (new approach)
    print("\n=== Phase 2: Symmetric Nesterov Descent ===")
    remaining = deadline - time.monotonic() - 200
    t_sym = min(remaining * 0.3, 500)

    if t_sym > 30:
        # Convert best_seq to symmetric form
        m = n_best // 2
        if n_best % 2 == 0:
            b_init = best_seq[:m].copy()
            b_init = np.maximum(b_init, 1e-8)
            b_init = b_init / b_init.sum() * 0.5
        else:
            m = (n_best + 1) // 2
            b_init = best_seq[:m].copy()
            b_init = np.maximum(b_init, 1e-8)
            b_init = b_init / b_init.sum() * 0.5
            n_best += 1  # Make even

        print(f"  Sym Nesterov (n={n_best}, m={m}), budget={t_sym:.0f}s")
        b_opt, _ = sym_nesterov_descent(b_init, time_limit=t_sym * 0.5,
                                        lr_init=0.1, temp_init=0.01, top_k=30)
        a_sym = np.concatenate([b_opt, b_opt[::-1]])
        a_sym = a_sym / a_sym.sum()
        try_seq(a_sym, "sym_nesterov")

        # Coordinate descent on symmetric
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_cd = min(rem * 0.3, 100)
            b_opt, _ = coord_descent_sym(b_opt, time_limit=t_cd)
            a_sym = np.concatenate([b_opt, b_opt[::-1]])
            a_sym = a_sym / a_sym.sum()
            try_seq(a_sym, "sym_coord")

    print(f"  Phase 2 best: {best_score:.10f} (n={len(best_seq) if best_seq is not None else 0})")

    # Phase 3: LP-based Frank-Wolfe (improved)
    print("\n=== Phase 3: LP Frank-Wolfe ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 150
    t_fw = min(remaining * 0.4, 500)

    if t_fw > 30:
        print(f"  LinFW (n={n_best}), budget={t_fw:.0f}s")
        a_opt, _ = linearized_fw(best_seq, time_limit=t_fw * 0.6, reduced_k=20, verbose=True)
        try_seq(a_opt, "linfw")

        # Run again from result
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            a_opt2, _ = linearized_fw(a_opt, time_limit=min(rem * 0.5, 200), reduced_k=15)
            try_seq(a_opt2, "linfw2")

    print(f"  Phase 3 best: {best_score:.10f}")

    # Phase 4: Subgradient polishing
    print("\n=== Phase 4: Subgradient Polish ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 80
    t_sg = min(remaining * 0.35, 200)

    if t_sg > 20:
        print(f"  SG polish, budget={t_sg:.0f}s")
        a_opt, _ = fg_optimize(best_seq, time_limit=t_sg * 0.5,
                               lr=0.003, mom=0.92, top_k=max(20, n_best // 5), temp=0.00005)
        try_seq(a_opt, "sg1")

        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_opt, _ = fg_optimize(best_seq, time_limit=min(rem * 0.5, 80),
                                   lr=0.001, mom=0.95, top_k=max(30, n_best // 3), temp=0.00002)
            try_seq(a_opt, "sg2")

    print(f"  Phase 4 best: {best_score:.10f}")

    # Phase 5: Symmetric refinement from best
    print("\n=== Phase 5: Symmetric Refinement ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 50

    if n_best % 2 == 0:
        m = n_best // 2
        b_init = np.maximum(best_seq[:m], 1e-8)
        b_init = b_init / b_init.sum() * 0.5
        t_sym2 = min(remaining * 0.4, 150)
        if t_sym2 > 20:
            b_opt, _ = sym_nesterov_descent(b_init, time_limit=t_sym2 * 0.6,
                                            lr_init=0.05, temp_init=0.005, top_k=25)
            a_sym = np.concatenate([b_opt, b_opt[::-1]])
            a_sym = a_sym / a_sym.sum()
            try_seq(a_sym, "sym_refine")

            rem = deadline - time.monotonic() - 30
            if rem > 15:
                b_opt, _ = coord_descent_sym(b_opt, time_limit=min(rem * 0.5, 60))
                a_sym = np.concatenate([b_opt, b_opt[::-1]])
                a_sym = a_sym / a_sym.sum()
                try_seq(a_sym, "sym_coord_refine")

    # Phase 6: Final LP polish
    print("\n=== Phase 6: Final LP Polish ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 20
    if remaining > 20:
        t_final = min(remaining * 0.8, 100)
        a_opt, _ = linearized_fw(best_seq, time_limit=t_final, reduced_k=15)
        try_seq(a_opt, "final_lp")

    # Phase 7: Try nearby lengths
    print("\n=== Phase 7: Length Variants ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 10
    if remaining > 20:
        for n_try in [n_best - 8, n_best + 8, n_best + 16]:
            if n_try < 64 or remaining < 10:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                if remaining > 20:
                    a_opt, _ = linearized_fw(px, time_limit=min(remaining * 0.4, 30),
                                             reduced_k=15)
                    try_seq(a_opt, f"len{n_try}")
                    remaining = deadline - time.monotonic() - 10

    # Final
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.10f}")

    return [float(x) for x in best_seq]
