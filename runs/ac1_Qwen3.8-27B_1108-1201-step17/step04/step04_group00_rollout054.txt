Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using hybrid subgradient + LP-FW + 
coordinate descent with multi-length exploration and cutting-plane acceleration.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_top_indices(c, K):
    """Get indices of the K largest values in c."""
    K = min(K, len(c))
    return np.argpartition(c, -K)[-K:]

def compute_subgradient(a, active_idx):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c_len = 2 * n - 1
    w_vec = np.zeros(c_len)
    w_vec[active_idx] = 1.0 / len(active_idx)
    # sg[j] = 2 * sum_k w_vec[k] * a[k-j]
    N = 1 << (n + c_len - 1).bit_length()
    W = np.fft.rfft(w_vec, N)
    A = np.fft.rfft(a, N)
    sg = 2.0 * np.fft.irfft(np.conj(W) * A, N)[:n]
    return sg

def subgradient_nesterov(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient method with BB step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # Nesterov lookahead
    c_y = fast_convolve(y)
    M_y = c_y.max()
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    # Momentum
    v = np.zeros(n)
    moment = 0.9
    
    t0 = time.monotonic()
    no_improve = 0
    K = max(20, n // 4)  # number of active constraints
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at y
        active = get_top_indices(c_y, K)
        sg = compute_subgradient(y, active)
        
        # Projected step
        beta = 1.0 / ((time.monotonic() - t0) + 1.0)
        f_new = project_simplex(y - beta * sg)
        
        # Update momentum
        v = moment * v - beta * sg
        y_new = project_simplex(f_new + v)
        
        c_new = fast_convolve(y_new)
        M_new = c_new.max()
        
        if M_new < best_obj - 1e-12:
            best_obj = M_new
            best_f = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Perturb and restart
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            v = np.zeros(n)
            c_y = fast_convolve(y)
            M_y = c_y.max()
            no_improve = 0
            K = max(10, K // 2)
        elif no_improve == 50:
            K = max(10, K // 2)
        
        f = f_new
        y = y_new
        c_y = c_new
        M_y = M_new
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated LP-Frank-Wolfe with
cutting-plane constraint reduction, adaptive subgradient fallback, coordinate
descent polishing, and multi-length exploration (640-1024).

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_row(f, k, n):
    """Build row k of the (f*g) constraint matrix: row[j] = f[k-j] if valid."""
    row = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if je >= js:
        row[js:je + 1] = f[k - je:k - js + 1][::-1]
    return row
