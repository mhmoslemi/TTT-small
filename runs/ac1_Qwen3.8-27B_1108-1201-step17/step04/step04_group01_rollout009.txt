"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start with diverse initializations (uniform, U-shape, raised cosine, random)
2. Active-set gradient descent with exact line search
3. LP-based direction finding (Frank-Wolfe style)
4. Adaptive peak weighting (power-sum approximation to max)
5. Periodic random perturbation to escape local minima

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(f, conv, top_k=20, power=10):
    """Compute gradient of sum of top-k peaks of autocorrelation, weighted by power."""
    n = len(f)
    # Find top-k peaks
    if top_k >= len(conv):
        indices = np.arange(len(conv))
    else:
        indices = np.argpartition(conv, -top_k)[-top_k:]
    
    # Weights: power-sum
    vals = conv[indices]
    total = np.sum(vals ** power)
    if total < 1e-30:
        weights = np.ones(len(indices)) / len(indices)
    else:
        weights = (vals ** power) / total
    
    # Compute gradient: g[j] = 2 * sum_i w_i * f[indices[i] - j] (for valid indices)
    g = np.zeros(n)
    for i in range(len(indices)):
        k = indices[i]
        w = weights[i]
        if w < 1e-15:
            continue
        # g[j] += 2*w*f[k-j] for j where 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            g[js] += 2.0 * w * f[k - js]
    
    return g

def line_search_rigorous(f, g, max_iter=50):
    """Rigorous line search: find alpha in [0,1] minimizing max(conv((1-alpha)*f+alpha*g_dir))
    where g_dir = -gradient direction (projected)."""
    n = len(f)
    # g is already the gradient, we want to go in direction -g
    # But we need to project onto simplex at each step
    # Use backtracking + golden section
    
    direction = -g
    # Normalize direction
    dir_norm = np.linalg.norm(direction)
    if dir_norm < 1e-15:
        return 0.0, f.copy()
    direction = direction / dir_norm
    
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    
    # Try a grid first
    alphas = np.linspace(0, 1, 21)
    for alpha in alphas:
        f_new = project_simplex(f + alpha * direction)
        val = fast_convolve(f_new).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Refine with golden section
    golden_ratio = (np.sqrt(5) - 1) / 2
    lo = max(0, best_alpha - 0.1)
    hi = min(1, best_alpha + 0.1)
    
    for _ in range(20):
        c = hi - golden_ratio * (hi - lo)
        d = lo + golden_ratio * (hi - lo)
        f_c = project_simplex(f + c * direction)
        f_d = project_simplex(f + d * direction)
        val_c = fast_convolve(f_c).max()
        val_d = fast_convolve(f_d).max()
        if val_c < val_d:
            hi = d
        else:
            lo = c
        
        f_lo = project_simplex(f + lo * direction)
        f_hi = project_simplex(f + hi * direction)
        val_lo = fast_convolve(f_lo).max()
        val_hi = fast_convolve(f_hi).max()
        if val_lo < best_val:
            best_val = val_lo
            best_alpha = lo
        if val_hi < best_val:
            best_val = val_hi
            best_alpha = hi
    
    return best_alpha, best_val

def gradient_descent_step(f, lr, top_k=20, power=10):
    """One step of projected gradient descent with line search."""
    conv = fast_convolve(f)
    g = compute_gradient(f, conv, top_k=top_k, power=power)
    
    # Project gradient to tangent space of simplex
    g_mean = g.mean()
    g_tangent = g - g_mean
    
    # Direction: -gradient projected
    direction = -g_tangent
    
    # Line search
    # Try several step sizes
    best_alpha = 0.0
    best_val = conv.max()
    
    base_lr = lr
    for i in range(30):
        alpha = base_lr * (0.5 ** i)
        if alpha < 1e-8:
            break
        f_new = project_simplex(f + alpha * direction)
        val = fast_convolve(f_new).max()
        if val < best_val - 1e-14:
            best_val = val
            best_alpha = alpha
        else:
            break
    
    if best_alpha > 0:
        f_new = project_simplex(f + best_alpha * direction)
        return f_new, best_val, best_alpha
    return f, conv.max(), 0.0

def fw_lp_step(f, time_limit_per_lp=10):
    """One Frank-Wolfe step using LP oracle."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    
    # Build constraint matrix
    # A @ g <= M where A[k,j] = f[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit_per_lp})
        if res.success:
            g = np.maximum(res.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
                return g, M
    except Exception:
        pass
    return None, M

def build_initializations(n, rng):
    """Build diverse initial conditions."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine (peak in middle)
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())
    
    # Inverted raised cosine (U-shape, peak at edges)
    irc = 0.5 * (1 - np.cos(2 * np.pi * x))
    irc = np.maximum(irc, 0.01)
    inits.append(irc / irc.sum())
    
    # Triangular (peak in middle)
    tri = 1 - np.abs(2 * x - 1)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())
    
    # Inverted triangular (U-shape)
    itri = np.abs(2 * x - 1)
    itri = np.maximum(itri, 0.01)
    inits.append(itri / itri.sum())
    
    # Gaussian bump
    gauss = np.exp(-10 * (x - 0.5) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Random
    rand = rng.random(n)
    rand = np.maximum(rand, 0.001)
    inits.append(rand / rand.sum())
    
    # Power law (heavy tail)
    power = x ** 2
    power = np.maximum(power, 0.001)
    inits.append(power / power.sum())
    power2 = (1 - x) ** 2
    power2 = np.maximum(power2, 0.001)
    inits.append(power2 / power2.sum())
    
    return inits

def optimize_single(f0, time_limit, seed=42):
    """Optimize a single starting point using adaptive gradient + LP."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 2.0 / n
    no_improve = 0
    top_k = min(30, n // 5)
    power = 8
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Gradient step
        conv = fast_convolve(f)
        if conv.max() < best_obj:
            best_obj = conv.max()
            best_f = f.copy()
        
        g = compute_gradient(f, conv, top_k=top_k, power=power)
        
        # Project gradient to tangent
        g_tangent = g - g.mean()
        direction = -g_tangent
        
        # Adaptive step size
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            no_improve += 100
            break
        
        # Backtracking line search
        best_alpha = 0.0
        best_val = conv.max()
        alpha = lr
        for _ in range(25):
            f_new = project_simplex(f + alpha * direction)
            val = fast_convolve(f_new).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha
                alpha *= 1.5  # try bigger step
            else:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
        
        if best_alpha > 1e-10:
            f = project_simplex(f + best_alpha * direction)
            new_obj = fast_convolve(f).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.01, 4.0 / n)
            else:
                no_improve += 1
                lr *= 0.95
        else:
            no_improve += 1
            lr *= 0.9
        
        # Periodic LP step
        if no_improve > 50 and it % 10 == 0 and time.monotonic() - t0 < time_limit - 5:
            g_dir, M = fw_lp_step(f, time_limit_per_lp=5)
            if g_dir is not None and np.max(np.abs(g_dir - f)) > 1e-8:
                # Line search in LP direction
                best_a = 0.0
                best_v = M
                for a in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    f_t = project_simplex((1 - a) * f + a * g_dir)
                    v = fast_convolve(f_t).max()
                    if v < best_v:
                        best_v = v
                        best_a = a
                if best_a > 0:
                    f = project_simplex((1 - best_a) * f + best_a * g_dir)
                    conv = fast_convolve(f)
                    if conv.max() < best_obj:
                        best_obj = conv.max()
                        best_f = f.copy()
                        no_improve = 0
        
        # Random perturbation when stuck
        if no_improve > 100:
            scale = 0.001 + 0.002 * min(1.0, (no_improve - 100) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            best_obj = conv.max()
            no_improve = 0
            lr = max(2.0 / n, lr)
            # Increase power to focus more on peaks
            power = min(power * 1.3, 50)
        
        # Gradually increase power (focus more on the maximum)
        if it % 100 == 0:
            power = min(power * 1.1, 50)
            top_k = min(top_k + 1, 50)

        for j in range(n):
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            if k_start < k_end:
                g[j] = 2.0 * np.dot(w[k_start:k_end], f[:k_end - k_start])
        
        # This is O(n^2) = O(448^2) ≈ 200K ops, which is fine.
        
        # Actually, let me use a more efficient FFT-based approach:
        # g[j] = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]
        # Let's define the extension of w: W_full has length n + (2n-1) - 1 = 3n-2
        # No, simpler: the cross-correlation can be computed as:
        # corr = ifft(conj(fft(f, L)) * fft(w, L)) where L >= n + (2n-1) - 1
        # Then corr[j] = sum_k f[k] * w[k-j]... 
        # Ugh, the indexing is always confusing. Let me just use the direct method.
        
        # Normalize
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            no_improve += 100
            break
        g = g / g_norm
        
        # Momentum
        v = momentum * v - lr * g
        f_new = project_simplex(f + v)
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 3.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(1.0 / n, 0.001)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.005
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj
