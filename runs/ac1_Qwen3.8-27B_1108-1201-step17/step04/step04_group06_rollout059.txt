import numpy as np
import time
from scipy.fft import next_fast_len
from scipy.optimize import minimize as scipy_minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_correlate(p, a):
    """Compute cross-correlation: result[i] = sum_k p[k] * a[k-i]"""
    n = len(a)
    # result[i] = sum_k p[k] * a[k-i]
    # = (p * flip(a))[n-1-i]
    b = a[::-1]
    size = len(p) + n - 1
    fft_size = next_fast_len(size)
    P = np.fft.rfft(p, fft_size)
    B = np.fft.rfft(b, fft_size)
    conv = np.fft.irfft(P * B, fft_size)
    return conv[n-1::-1][:n]

def compute_grad_a(a, c, beta):
    """Compute gradient of logsumexp(beta*c)/beta w.r.t. a."""
    n = len(a)
    scaled = beta * c
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    grad_a = 2.0 * cross_correlate(p, a)
    return grad_a

def eg_with_linesearch(a0, time_limit, beta0=100, beta1=None, eta0=1.0, verbose=False):
    """EG with adaptive step size via line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    beta = beta0
    eta = eta0
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient
        grad_a = compute_grad_a(a, c, beta)
        grad_max = grad_a.max()
        if grad_max < 1e-15:
            break
        
        # Direction: multiplicative update
        # a_new = a * exp(-t * grad_a / grad_max)
        # Try different step sizes via line search
        best_t = eta
        best_new_obj = np.inf
        best_a_new = None

            if t <= 0:

            log_update = -t * grad_a / grad_max

            a_try = np.maximum(a_try, 1e-300)

            if obj_try < best_new_obj:

def fast_convolve(a):
    n = len(a)

    A = np.fft.rfft(a, fft_size)

def cross_correlate(p, a):
    n = len(a)
    b = a[::-1]
    size = len(p) + n - 1
    fft_size = next_fast_len(size)
    P = np.fft.rfft(p, fft_size)

def compute_sg(a, c, beta, top_k):
    n = len(a)
    len_c = len(c)
    if top_k is not None and top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()
    
    scaled = beta * peek
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)

        if w[idx] < 1e-12:
            continue
        k = int(indices[idx])

        je = min(n - 1, k)
        sg[js:je+1] += 2.0 * w[idx] * a[k-je:k-js+1][::-1]
    
    return sg

    return conv[n - 1::-1][:n]


def compute_sg(a, c, beta, top_k):
    """Compute softmax-weighted subgradient of max(c) at temperature 1/beta."""
    n = len(a)
    len_c = len(c)
    if top_k is not None and top_k < len_c and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()

    scaled = beta * peek
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-14:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

    return sg


def eg_optimize(a0, time_limit, eta0=1.0, beta0=50.0, beta_growth=1.0005,
                top_k_frac=0.3, use_linesearch=True, verbose=False):
    """EG/mirror descent with optional line search on step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(20, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.astype(np.float64).copy()
    count = 0
    eta = eta0
    beta = beta0
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_sg(a, c, beta, top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        if use_linesearch and it > 50:
            best_t = eta
            best_a_try = None
            best_obj_try = np.inf
            for t_mult in [1.0, 0.5, 0.25, 1.5]:
                t = eta * t_mult
                if t <= 0:
                    continue
                log_update = -t * sg / sg_max
                log_update -= log_update.max()
                a_try = a * np.exp(log_update)
                a_try = np.maximum(a_try, 1e-300)
                s_try = a_try.sum()
                if s_try < 1e-300:
                    continue
                a_try = a_try / s_try
                c_try = fast_convolve(a_try)
                obj_try = c_try.max()
                if obj_try < best_obj_try:
                    best_obj_try = obj_try
                    best_a_try = a_try
                    best_t = t
            if best_a_try is not None:
                a = best_a_try
                c = fast_convolve(a)
                obj = c.max()
            else:
                break
        else:
            log_update = -eta * sg / sg_max
            log_update -= log_update.max()
            a = a * np.exp(log_update)
            a = np.maximum(a, 1e-300)
            s = a.sum()
            a = a / s
            c = fast_convolve(a)
            obj = c.max()

        count += 1
        avg_a = (count * avg_a + a) / (count + 1)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-14:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
                no_improve = 0

        if no_improve > 3000:
            eta *= 0.7
            no_improve = 0
            if eta < eta0 * 0.005:
                rng_local = np.random.default_rng(it * 31 + 7)
                a = best_a + rng_local.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                ps = a.sum()
                if ps > 1e-10:
                    a = a / ps
                else:
                    a = best_a.copy()
                c = fast_convolve(a)
                eta = eta0 * 0.3
                avg_a = a.copy()
                count = 0
                no_improve = 0

        beta *= beta_growth
        if it % 500 == 0:
            eta *= 0.998

        if verbose and it % 20000 == 0:
            print(f"    EG iter {it}: obj={best_obj:.12f}, eta={eta:.6f}, beta={beta:.1f}")

    return best_a.copy(), best_obj


def lbfqs_polish(a0, time_limit, beta):
    """Polish using L-BFGS-B on log-softmax parameterization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), fast_convolve(a0).max()
    a = a / s
    n = len(a)
    x0 = np.log(np.maximum(a, 1e-25))

    t0 = time.monotonic()
    n_calls = [0]

    def obj_grad(x):
        n_calls[0] += 1
        if time.monotonic() - t0 > time_limit or n_calls[0] > 5000:
            return np.inf, np.zeros(len(x))
        x = x - x.max()
        a = np.exp(x)
        a = a / a.sum()
        c = fast_convolve(a)
        scaled = beta * c
        scaled -= scaled.max()
        logsumexp = scaled.max() + np.log(np.sum(np.exp(scaled)))
        obj = logsumexp / beta
        p = np.exp(scaled)
        p = p / p.sum()
        grad_a = 2.0 * cross_correlate(p, a)
        grad_x = a * grad_a - a * np.sum(a * grad_a)
        return obj, grad_grad_dummy(grad_x, a)

    try:
        result = sp_minimize(obj_grad, x0, method='L-BFGS-B',
                             jac=True, options={'maxiter': 300, 'maxfun': 3000,
                                                'ftol': 1e-14, 'gtol': 1e-9,
                                                'maxls': 20})
        if result.x is not None:
            x = result.x
            x = x - x.max()
            a_opt = np.exp(x)
            a_opt = a_opt / a_opt.sum()
            a_opt = np.maximum(a_opt, 0)
            return a_opt, fast_convolve(a_opt).max()
    except Exception as e:
        print(f"    L-BFGS failed: {e}")
    return a0.copy(), fast_convolve(a).max()
