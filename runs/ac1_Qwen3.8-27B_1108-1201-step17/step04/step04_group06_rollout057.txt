This is the "fourth moment" of the power spectrum. Minimizing this is equivalent to making the power spectrum as flat as possible.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def spectral_flatness_score(c):
    """How flat is the autocorrelation? Lower is better (more flat)."""
    c_mean = c.mean()
    if c_mean < 1e-30:
        return np.inf
    return c.max() / c_mean

def error_reduction_flat_spectrum(a, n_iters=100, damping=0.5):
    """
    Error reduction method to flatten the power spectrum.
    Works by alternating:
    - Time: clip to non-negative, normalize
    - Frequency: flatten magnitude
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(n_iters):
        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude
        target = mag.mean()
        
        # Damped update: blend current and target
        mag_new = (1 - damping) * mag + damping * target
        mag_new = np.maximum(mag_new, 1e-30)
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check improvement
        c = fast_convolve(a_new)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def spectral_initialization(n, method='random_phase', seed=42):
    """
    Generate initializations with approximately flat spectrum.
    """
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    if method == 'random_phase':
        # Random phase, flat magnitude
        mag = np.ones(fft_size // 2 + 1)
        phase = rng.standard_normal(fft_size // 2 + 1)
        # Enforce Hermitian symmetry for real signal
        # Actually for rfft, we just need the first half
        A = mag * np.exp(1j * phase)
        # Fix DC component
        A[0] = abs(A[0])
        a = np.fft.irfft(A, fft_size)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    
    elif method == 'gaussian':
        mag = np.abs(rng.standard_normal(fft_size // 2 + 1))
        mag[0] = 0  # zero DC for zero mean
        phase = rng.standard_normal(fft_size // 2 + 1)
        A = mag * np.exp(1j * phase)
        a = np.fft.irfft(A, fft_size)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    
    elif method == 'jack_knife':
        # Start with uniform and add structured perturbation
        x = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        a += 0.3 * np.sin(2 * np.pi * rng.integers(1, 5) * x / n)
        a += 0.3 * np.cos(2 * np.pi * rng.integers(1, 5) * x / n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        return a
    
    else:
        return np.ones(n) / n

def coordinate_descent_polish(a0, time_limit, max_rounds=500):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the max autocorrelation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        # Random order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            
            # Try different values
            candidates = [old_val * f for f in [0.5, 0.7, 0.85, 1.15, 1.3, 2.0, 0.0]]
            # Also try values based on neighbors
            if i > 0:
                candidates.append(a[i-1] * 0.9)
                candidates.append(a[i-1] * 1.1)
            if i < n-1:
                candidates.append(a[i+1] * 0.9)
                candidates.append(a[i+1] * 1.1)
            
            for val in candidates:
                a[i] = max(0, val)
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a_norm.copy()
                    improved = True
                    break
            a = best_a.copy()
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_a.copy(), best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Exponentiated gradient with momentum and adaptive temperature
2. Fourier-domain spectral flattening (error reduction / Gerchberg-Saxton)
3. Variance-based smooth surrogate for the max objective
4. Diverse spectral initializations (random phase, Gaussian magnitude)
5. Iterative balancing updates for autocorrelation equalization
6. Coordinate descent polishing
7. Multi-length and multi-initialization exploration
"""
import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute the max autocorrelation value (with sum normalized to 1)."""
    c = fast_convolve(a)
    return c.max()


def spectral_error_reduction(a, n_iters=200, damping=0.4, verbose=False):
    """
    Gerchberg-Saxton style error reduction to flatten the power spectrum.
    Alternates between:
    - Time domain: enforce non-negativity, normalize
    - Frequency domain: flatten magnitude spectrum
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = compute_objective(a)
    no_improve = 0

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)  # phase as unit complex

        # Target: geometric mean of magnitudes (preserves total power better)
        log_mag = np.log(np.maximum(mag, 1e-30))
        target = np.exp(np.mean(log_mag))

        # Damped magnitude update
        mag_new = (1.0 - damping) * mag + damping * target
        mag_new = np.maximum(mag_new, 1e-30)

        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj = compute_objective(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

        a = a_new

    return best_a


def balancing_update(a, n_iters=100, step_size=0.01, verbose=False):
    """
    Iterative balancing: push autocorrelation toward uniform.
    For lags where c_k > mean, reduce contributing a_i values.
    For lags where c_k < mean, increase contributing a_i values.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_objective(a)
    c = fast_convolve(a)
    target = 1.0 / (2 * n - 1)  # target for each c_k (since sum(c) = 1)

    for it in range(n_iters):
        c = fast_convolve(a)
        c_mean = c.mean()

        # Compute a "force" on each a_i based on autocorrelation imbalance
        force = np.zeros(n)
        # Only consider top excess lags for efficiency
        n_lags = min(len(c), 80)
        top_lags = np.argpartition(-c, -n_lags)[-n_lags:]

        for k in top_lags:
            excess = c[k] - c_mean
            if abs(excess) < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # Reduce a_i where c_k is too high, increase where too low
            # dc_k/da_i = 2 * a_{k-i} (for valid i)
            contrib = a[k - je:k - js + 1][::-1]
            force[js:je + 1] += -excess * contrib

        # Apply multiplicative update
        force_max = np.abs(force).max()
        if force_max < 1e-15:
            break
        a_new = a * np.exp(step_size * force / force_max)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj = compute_objective(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new
        # Decay step size
        step_size *= 0.999

    return best_a


def exp_gradient_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                          temp_decay=0.9997, eta_decay=0.99995,
                          top_k_frac=0.25, momentum=0.3,
                          variance_alpha=0.05, sr_interval=400,
                          verbose=False):
    """
    Exponentiated gradient with momentum, adaptive temperature,
    variance smoothing, and periodic spectral reconstruction.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    log_momentum = np.zeros(n)
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute softmax-weighted subgradient over top-k peaks
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
        else:
            indices = np.arange(len(c))

        peek = c[indices]
        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws

        # Subgradient of max(c)
        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-14:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

        # Variance smoothing: add gradient of sum(c_k^2) for top excess lags
        if variance_alpha > 0:
            c_mean = c.mean()
            n_excess = min(len(c), 40)
            top_excess = np.argpartition(c, -n_excess)[-n_excess:]
            for k in top_excess:
                excess = c[k] - c_mean
                if excess < 1e-10:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                sg[js:je + 1] += variance_alpha * 4.0 * c[k] * a[k - je:k - js + 1][::-1]

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        # Multiplicative update with momentum
        log_update = -eta * sg / sg_max
        log_momentum = momentum * log_momentum + (1.0 - momentum) * log_update
        # Numerical stability
        log_momentum -= log_momentum.max()
        exp_mom = np.exp(log_momentum)
        a_new = a * exp_mom
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        c = c_new

        # Periodic spectral error reduction
        if sr_interval > 0 and it % sr_interval == 0 and no_improve > 30:
            a_sr = spectral_error_reduction(a_new, n_iters=30, damping=0.3)
            c_sr = fast_convolve(a_sr)
            if c_sr.max() < best_obj - 1e-10:
                best_obj = c_sr.max()
                best_a = a_sr.copy()
                a = a_sr
                c = fast_convolve(a)
                no_improve = 0
                log_momentum = np.zeros(n)
                if verbose:
                    print(f"    SR improved at iter {it}: {best_obj:.10f}")

        # Stagnation restart
        if no_improve > 2500:
            rng_local = np.random.default_rng(it * 31 + 7)
            perturb_scale = 0.002 + 0.003 * (no_improve // 5000)
            a = best_a + rng_local.standard_normal(n) * perturb_scale
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            no_improve = 0
            eta = eta0 * 0.5
            temp = max(temp, temp0 * 0.05)
            log_momentum = np.zeros(n)

        if verbose and it % 2000 == 0:
            print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.4f}, temp={temp:.6f}, no_improve={no_improve}")

        eta *= eta_decay
        temp = max(temp * temp_decay, temp0 * 1e-4)

    return best_a.copy(), best_obj


def spectral_initialization(n, method, seed):
    """Generate initializations with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    if method == 'random_phase':
        mag = np.ones(fft_size // 2 + 1)
        # Slightly perturb magnitude for diversity
        mag *= (1.0 + 0.1 * rng.standard_normal(fft_size // 2 + 1))
        mag = np.maximum(mag, 0.1)
        phase = rng.standard_normal(fft_size // 2 + 1)
        A = mag * np.exp(1j * phase)
        a = np.fft.irfft(A, fft_size)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        return a

    elif method == 'gaussian_spectrum':
        mag = np.abs(rng.standard_normal(fft_size // 2 + 1))
        mag[0] = 0.0
        # Normalize
        mag /= mag.max() + 1e-10
        phase = rng.standard_normal(fft_size // 2 + 1)
        A = mag * np.exp(1j * phase)
        a = np.fft.irfft(A, fft_size)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        return a

    elif method == 'structured':
        # Sinusoidal perturbation of uniform
        x = np.arange(n, dtype=np.float64)
        freq1 = rng.integers(1, max(2, n // 10))
        freq2 = rng.integers(1, max(2, n // 10))
        a = np.ones(n)
        a += 0.5 * np.sin(2.0 * np.pi * freq1 * x / n)
        a += 0.3 * np.cos(2.0 * np.pi * freq2 * x / n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        return a

    elif method == 'boundary_boost':
        # Elevated boundaries based on theoretical analysis
        x = np.arange(n, dtype=np.float64)
        boundary_w = np.exp(-x / (n * 0.05)) + np.exp(-(n - 1 - x) / (n * 0.05))
        a = 0.5 + boundary_w
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        return a

    elif method == 'cosine_taper':
        x = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1 + 1e-10))
        a = np.maximum(a, 0.01)
        s = a.sum()
        a = a / s
        return a

    else:
        return np.ones(n) / n


def coordinate_descent_polish(a0, time_limit, max_rounds=300):
    """
    Coordinate descent: try multiplicative perturbations on random coordinates.
    More aggressive than the previous version with adaptive step sizes.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    # Factors to try, ordered by aggressiveness
    factors = [0.95, 1.05, 0.9, 1.1, 0.85, 1.18, 0.8, 1.25, 0.7, 1.4, 0.5, 2.0]

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        # Random permutation of coordinates
        n_perturb = max(1, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue

            for factor in factors:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a_norm.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a.copy(), best_obj


def try_and_update(best_score, best_seq, a, label):
    """Evaluate sequence and update best if improved. Returns (score, best_score, best_seq)."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return float('inf'), best_score, best_seq
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        best_score = score
        best_seq = a.copy()
    return score, best_score, best_seq

            if time.monotonic() > deadline - 60:
                break
            a_init = spectral_initialization(n, method=method, seed=sd)
            # Quick spectral reduction
            a_init = spectral_error_reduction(a_init, n_iters=30, damping=0.4)
            _, best_score, best_seq = try_and_update(best_score, best_seq, a_init, f"spec_{method}_{sd}")
            # EG from spectral init
            t_eg = min(25, (deadline - time.monotonic()) / 2)
            if t_eg > 8:
                a_opt, _ = exp_gradient_optimize(a_init, time_limit=t_eg,
                                                  eta0=1.0, temp0=0.01,
                                                  temp_decay=0.9998, eta_decay=0.9999,
                                                  top_k_frac=0.2, momentum=0.3,
                                                  variance_alpha=0.1, verbose=False)
                _, best_score, best_seq = try_and_update(best_score, best_seq, a_opt, f"eg_spec_{method}_{sd}")

    # === Phase 4: EG hyperparameter variations ===
    print(f"\n=== Phase 4: EG Hyperparameter Variations ===")
    print(f"  Best so far: {best_score:.8f}")

    configs = [
        {"eta0": 0.5, "temp0": 0.01, "temp_decay": 0.9998, "eta_decay": 0.9999, "momentum": 0.5, "variance_alpha": 0.0, "top_k_frac": 0.15},
        {"eta0": 2.0, "temp0": 0.002, "temp_decay": 0.9995, "eta_decay": 0.9999, "momentum": 0.1, "variance_alpha": 0.15, "top_k_frac": 0.35},
        {"eta0": 0.3, "temp0": 0.02, "temp_decay": 0.9999, "eta_decay": 0.99998, "momentum": 0.6, "variance_alpha": 0.1, "top_k_frac": 0.4},
        {"eta0": 1.5, "temp0": 0.003, "temp_decay": 0.9996, "eta_decay": 0.99993, "momentum": 0.2, "variance_alpha": 0.0, "top_k_frac": 0.1},
    ]
    for ci, cfg in enumerate(configs):
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t_alloc = min(70, rem / max(1, len(configs) - ci))
        if t_alloc < 10:
            break
        a_opt, _ = exp_gradient_optimize(best_seq, time_limit=t_alloc, verbose=False, **cfg)
        _, best_score, best_seq = try_and_update(best_score, best_seq, a_opt, f"eg_cfg{ci}")

    # === Phase 5: Balancing + EG hybrid ===
    print(f"\n=== Phase 5: Balance + EG Hybrid ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_bal = balancing_update(best_seq, n_iters=150, step_size=0.015)
        _, best_score, best_seq = try_and_update(best_score, best_seq, a_bal, "balance2")
        a_sr = spectral_error_reduction(best_seq, n_iters=50, damping=0.2)
        _, best_score, best_seq = try_and_update(best_score, best_seq, a_sr, "sr2")
        # EG from the best of these
        t_alloc = min(60, rem * 0.4)
        if t_alloc > 10:
            a_opt, _ = exp_gradient_optimize(best_seq, time_limit=t_alloc,
                                              eta0=0.5, temp0=0.005,
                                              temp_decay=0.9998, eta_decay=0.99995,
                                              momentum=0.4, variance_alpha=0.05, verbose=False)
            _, best_score, best_seq = try_and_update(best_score, best_seq, a_opt, "hybrid_eg")

    # === Phase 6: Coordinate descent polishing ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    if rem > 10:
        a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 100), max_rounds=200)
        _, best_score, best_seq = try_and_update(best_score, best_seq, a_opt, "coord_desc")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Different Lengths ===")
    print(f"  Best so far: {best_score:.8f}")

    alt_lengths = []
    for n_new in [int(n * 0.75), int(n * 1.25), int(n * 1.5), n // 2]:
        if n_new < 32 or n_new > 2000:
            continue
        if n_new == n:
            continue
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_new)
        a_new = np.interp(x_new, x_old, best_seq)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            alt_lengths.append(a_new)

    for li, a_alt in enumerate(alt_lengths):
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(35, rem / max(1, len(alt_lengths) - li))
        if t_alloc < 8:
            break
        a_opt, _ = exp_gradient_optimize(a_alt, time_limit=t_alloc,
                                          eta0=1.0, temp0=0.005,
                                          temp_decay=0.9997, eta_decay=0.99995,
                                          momentum=0.3, variance_alpha=0.05, verbose=False)
        _, best_score, best_seq = try_and_update(best_score, best_seq, a_opt, f"len_{len(a_alt)}")

    # === Phase 8: Final fine-tuning ===
    print(f"\n=== Phase 8: Final Fine-tuning ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    if rem > 5:
        a_opt, _ = exp_gradient_optimize(best_seq, time_limit=min(rem * 0.4, 50),
                                          eta0=0.1, temp0=0.001,
                                          temp_decay=0.99995, eta_decay=0.99999,
                                          momentum=0.5, variance_alpha=0.0,
                                          top_k_frac=0.5, sr_interval=0, verbose=False)
        _, best_score, best_seq = try_and_update(best_score, best_seq, a_opt, "eg_fine")

        rem2 = deadline - time.monotonic()
        if rem2 > 3:
            a_opt2, _ = coordinate_descent_polish(best_seq, time_limit=min(rem2 * 0.7, 25), max_rounds=50)
            _, best_score, best_seq = try_and_update(best_score, best_seq, a_opt2, "coord_final")

        # One more spectral pass
        rem3 = deadline - time.monotonic()
        if rem3 > 3:
            a_sr_final = spectral_error_reduction(best_seq, n_iters=50, damping=0.1)
            _, best_score, best_seq = try_and_update(best_score, best_seq, a_sr_final, "sr_final")
            a_bal_final = balancing_update(best_seq, n_iters=50, step_size=0.005)
            _, best_score, best_seq = try_and_update(best_score, best_seq, a_bal_final, "bal_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={elapsed:.1f}s)")
    return [float(x) for x in best_seq]
