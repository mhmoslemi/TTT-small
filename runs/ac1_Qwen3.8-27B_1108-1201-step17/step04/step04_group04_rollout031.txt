def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix_slice(a, k_list):
    """Build matrix C such that C @ g = conv(a, g) for lags in k_list only."""
    n = len(a)
    C = np.zeros((len(k_list), n))
    for row, k in enumerate(k_list):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            C[row, j] = a[k - j]
    return C


def active_set_lp(a, M, conv, active_frac=0.1, max_active=50):
    """
    Solve LP using only active (near-peak) constraints.
    max sum(g) s.t. conv(a, g)[k] <= M for active k, g >= 0
    """
    n = len(a)
    num_lags = 2 * n - 1
    
    # Find active set: lags where conv[k] is close to M
    threshold = M * (1 - active_frac)
    active_indices = np.where(conv >= threshold)[0]
    
    # Limit to top max_active
    if len(active_indices) > max_active:
        active_indices = np.argsort(conv[active_indices])[-max_active:]
    
    C = build_conv_matrix_slice(a, active_indices)
    c = -np.ones(n)
    b = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False

    fft_size = 1 << (size - 1).bit_length()

    """Vectorized soft-max weighted subgradient of peak autocorrelation."""
    n = len(a)
    flip_a = a[::-1]

    n = len(a)
    
    threshold = M * (1.0 - active_frac)
    active = np.where(conv >= threshold)[0]
    
    if len(active) > max_active:
        # Keep only the top max_active
        active = active[np.argsort(conv[active])[-max_active:]]
    
    if len(active) == 0:
        return None
    
    C = build_active_matrix(a, active)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            g = res.x
            g_sum = float(g.sum())
            if g_sum > 1e-10:
                return g
    except Exception:
        pass
    return None


def line_search(a, d, obj_current, n, max_iters=20):
    """Backtracking line search for step size in direction d."""
    # d should be a direction that (when added to a) reduces the objective
    # We want the largest t such that F(a + t*d) < F(a)
    # F(a) = 2*n*max(conv(a,a))  [with sum(a)=1]
    
    best_t = 0.0
    best_obj = obj_current
    
    # Try geometric sequence of step sizes
    t = 1.0
    for _ in range(max_iters):
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            t *= 0.5
            continue
        a_try = a_try / s
        conv = fast_convolve(a_try)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_t = t
            # Try to go bigger (but don't overshoot)
            t_try = t * 1.5
            if t_try > 2.0:
                break
            a_try2 = a + t_try * d
            a_try2 = np.maximum(a_try2, 0.0)
            s2 = a_try2.sum()
            if s2 > 1e-15:
                a_try2 = a_try2 / s2
                obj2 = 2.0 * n * fast_convolve(a_try2).max()
                if obj2 < best_obj - 1e-14:
                    continue  # Can go bigger, but let's be conservative
            break
        else:
            t *= 0.5
            if t < 1e-8:
                break
    
    if best_t > 0:
        a_new = a + best_t * d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        return a_new
    return a


def optimize_lp(a0, time_limit, verbose=False):
    """
    LP-based optimization with active-set constraints and backtracking line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    active_frac = 0.1
    max_active = 50
    
    while time.monotonic() - t0 < time_limit:
        # Get LP direction using active constraints
        g = active_lp_step(a, M, conv, active_frac=active_frac, max_active=max_active)
        
        if g is None:
            no_improve += 1
            # Try with more active constraints
            if no_improve % 5 == 0:
                active_frac = min(0.5, active_frac * 1.5)
                max_active = min(200, max_active + 20)
            if no_improve > 50:
                break
            continue
        
        d = g - a  # direction
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Normalize direction for line search
        d_norm = d / norm_d
        
        # Try step sizes
        improved = False
        for t_candidate in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_try = a + t_candidate * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            
            if obj_try < best_obj - 1e-12:
                a = a_try
                conv = conv_try
                M = conv.max()
                best_obj = obj_try
                best_a = a.copy()
                no_improve = 0
                # Reset active set params on improvement
                active_frac = max(0.02, active_frac * 0.8)
                max_active = max(30, int(max_active * 0.9))
                improved = True
                break
            elif obj_try < best_obj:
                # Slight improvement, not enough for threshold
                # Still take the step but don't update best
                a = a_try
                conv = conv_try
                M = conv.max()
                improved = True
                break
        
        if not improved:
            no_improve += 1
        
        iteration += 1
        if verbose and iteration % 50 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s, no_imp={no_improve}")
    
    return best_a, best_obj


def optimize_gradient(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=40, temp_init=0.002):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, top_k=top_k, temp=temp)
        
        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            break
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        lr *= 0.999
        temp *= 1.005
        mom = min(0.97, mom + 0.0001)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj
