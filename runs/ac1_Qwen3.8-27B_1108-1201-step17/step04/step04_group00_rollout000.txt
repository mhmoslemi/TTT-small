def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.005, decay=0.9995):
    """
    Mirror descent with negative entropy (multiplicative weight updates).
    Minimizes max_k (a*a)_k subject to a >= 0, sum(a) = 1.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.0001
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks for subgradient computation
        k_actual = min(20, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Softmax weights over peaks
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Multiplicative update: a_j *= exp(-lr * sg_j)
        # This is mirror descent with negative entropy
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= decay
        temperature *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: specifically targets the largest peaks
    by reducing contributions to them.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Find the top peaks
        top_k = min(10, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        # For each peak, find the pair (j, k-j) with maximum product
        # and try to reduce it
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if je < js:
                continue

            # Find j that maximizes f[j] * f[k-j] in the valid range
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            j_max = idx[np.argmax(products)]

            # Try reducing f[j_max] and redistributing to f[k-j_max]
            # or to the minimum element
            old_val = f[j_max]
            if old_val < 1e-12:
                continue

            best_trial = None
            best_trial_obj = conv.max()

            multipliers = [0.3, 0.5, 0.7, 0.85, 1.2, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                trial = f.copy()
                trial[j_max] = new_val
                # Redistribute the difference to the smallest elements
                diff = old_val - new_val
                if abs(diff) > 1e-15:
                    # Add proportional to current values (to keep smoothness)
                    weights = f.copy()
                    weights[j_max] = 0
                    wsum = weights.sum()
                    if wsum > 1e-15:
                        trial += diff * weights / wsum
                    else:
                        trial = f.copy()
                        trial[j_max] = new_val
                        trial /= trial.sum()
                        continue
                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-14:
                    best_trial_obj = obj
                    best_trial = trial

            if best_trial is not None:
                f = best_trial
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Try a different strategy: reduce all elements contributing to the top peak
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)

                # Scale down the contributing elements
                scale = 0.85
                trial = f.copy()
                trial[idx] *= scale
                # Redistribute to non-contributing elements
                diff = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[idx] = False
                if non_contrib.sum() > 0:
                    nc_weights = f[non_contrib]
                    nw = nc_weights.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += diff * nc_weights / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Create a bell-shaped initial sequence with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4  # Smooth bell
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

        print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  Peak shaving: t={t_alloc:.0f}s")
            a_opt, obj = peak_shaving(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "peak_shave")

    # === Phase 6: Coordinate descent on best ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_alloc = min(rem, 80)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 7: Deep LP-FW with perturbations ===
    print(f"\n=== Phase 7: Deep LP-FW with Perturbations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            t_alloc = min(rem / (5 - p), 80)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                # Reverse-symmetric blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif p == 3:
                # Blend with uniform
                f_init = 0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                # Scaled perturbation
                noise = rng.standard_normal(n_best) * 0.005
                f_init = np.maximum(best_seq + noise, 0)
                f_init = f_init / f_init.sum()

            print(f"  Deep LP-FW variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_seq(a_opt, f"deep_{p}")

    # === Phase 8: Final multiplicative descent polish ===
    print(f"\n=== Phase 8: Final MD Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  MD polish: t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc,
                                                verbose=True, lr_init=0.002, decay=0.9997)
            try_seq(a_opt, "md_polish")

    # === Phase 9: Ultra-fine coordinate descent ===
    print(f"\n=== Phase 9: Fine CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_alloc = min(rem, 40)
            print(f"  Fine CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fine_cd")

    # === Phase 10: Final LP-FW squeeze ===
    print(f"\n=== Phase 10: Final LP-FW Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
