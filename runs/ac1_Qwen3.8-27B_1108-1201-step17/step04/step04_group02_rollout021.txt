import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_project(f, sg, eta):
    """Mirror descent step with entropy proximal on the simplex."""
    log_f = np.log(f + 1e-300)
    log_new = log_f - eta * sg
    log_new -= log_new.max()  # numerical stability
    new = np.exp(log_new)
    s = new.sum()
    if s < 1e-30:
        return f.copy()
    return new / s

def solve_lp(f, M, time_limit=30):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    # where w_padded is w zero-padded appropriately
    # 
    # Simpler approach: g[i] = sum_j f[j]*w[i+j]
    # Let's define: h[m] = w[m] for m=0..2n-2, 0 otherwise
    # Then g[i] = sum_{j=0}^{n-1} f[j] * h[i+j]
    # This is the "cross-correlation" <f, shift_i(h)>
    # = (f * h)[n-1+i] in standard convolution where (f*h)[k] = sum_j f[j]*h[k-j]
    # (f*h)[n-1+i] = sum_j f[j]*h[n-1+i-j]
    # We want sum_j f[j]*h[i+j], so set n-1+i-j = i+j => j = (n-1)/2... no that's wrong
    #
    # Let me just do it the straightforward way with FFT:
    # g[i] = sum_{j=0}^{n-1} f[j] * w[i+j]
    # Define f_pad = [f, 0, ..., 0] of length n + 2n-2 - n + 1 = 2n-1
    # Then g is the "valid" part of the convolution of f with reverse(w)
    # 
    # OK I'll just use a simple formula: 
    # Create a matrix W where W[j, i] = w[i+j] if i+j < len(w), else 0
    # Then g = f @ W
    # But that's O(n^2). For n=384 that's fine (150K ops)
    # But let me use FFT for larger n
    #
    # Using FFT: g = ifft(fft(f, L) * fft(w_rev, L))[:n] where w_rev = w[::-1]
    # and L >= n + len(w) - 1
    w_rev = w[::-1]
    L = 1 << (n + len(w) - 1 - 1).bit_length()
    Ff = np.fft.rfft(f, L)
    Fw = np.fft.rfft(w_rev, L)
    g = np.fft.irfft(Ff * Fw, L)[:n]
    return (2.0 / T) * g


def compute_topk_gradient(f, conv, top_k, T):
    """Compute subgradient using top-k peaks with softmax weighting."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        pv = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
    scaled = T * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def mirror_descent_step(f, sg, eta):
    """Multiplicative weights update on the simplex."""
    log_f = np.log(f + 1e-300)
    log_new = log_f - eta * sg
    log_new -= log_new.max()
    new = np.exp(log_new)
    s = new.sum()
    if s < 1e-300:
        return f.copy()
    return new / s


def solve_lp(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def md_anneal(f0, time_limit, T_start=0.5, T_end=50.0, eta_start=0.05,
              eta_end=0.001, lp_every=200, top_k=None, verbose=False):
    """Mirror descent with smooth max and annealing temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    T = T_start
    eta = eta_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        # Anneal T and eta
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        # Compute gradient
        if top_k is not None and top_k < 2 * n - 1:
            sg = compute_topk_gradient(f, conv, top_k, T)
        else:
            sg = compute_smooth_gradient(f, conv, T)

        # Mirror descent step
        f_new = mirror_descent_step(f, sg, eta)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        conv = conv_new

        # LP escape
        if no_improve > lp_every and rem > 20:
            g = solve_lp(f, best_obj, time_limit=10)
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    # Try several blendings
                    for alpha in [0.3, 0.5, 0.7, 1.0]:
                        ft = project_simplex((1 - alpha) * f + alpha * g)
                        c_try = fast_convolve(ft).max()
                        if c_try < best_obj:
                            best_obj = c_try
                            best_f = ft.copy()
                            f = ft
                            conv = fast_convolve(f)
                            no_improve = 0
                            break
            no_improve = 0

        # Random restart from best
        if no_improve > lp_every * 3:
            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp(f, M * m_exp, time_limit=min(20, rem - 5))
        if g is None:
            m_exp *= 1.3
            if m_exp > 10.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        for _ in range(20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.7

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_sweeps_per_sec=50, verbose=False):
    """Coordinate descent: for each coordinate, try scaling it and renormalizing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    scales = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            old_val = f[i]
            if old_val < 1e-10:
                # Try setting it to a small positive value
                for sv in [0.001, 0.005, 0.01]:
                    f[i] = sv
                    rest = f.copy()
                    rest[i] = 0.0
                    rs = rest.sum()
                    if rs < 1e-15:
                        f[i] = old_val
                        continue
                    rest = rest / rs * (1.0 - sv)
                    f_try = rest.copy()
                    f_try[i] = sv
                    c_try = fast_convolve(f_try).max()
                    if c_try < best_obj - 1e-14:
                        best_obj = c_try
                        best_f = f_try.copy()
                        f = f_try.copy()
                        improved = True
                continue

            for sc in scales:
                new_val = old_val * sc
                if new_val >= 1.0:
                    continue
                f[i] = new_val
                rest = f.copy()
                rest[i] = 0.0
                rs = rest.sum()
                if rs < 1e-15:
                    f[i] = old_val
                    continue
                rest = rest / rs * (1.0 - new_val)
                f_try = rest.copy()
                f_try[i] = new_val
                c_try = fast_convolve(f_try).max()
                if c_try < best_obj - 1e-14:
                    best_obj = c_try
                    best_f = f_try.copy()
                    f = f_try.copy()
                    improved = True
                    break
            f[i] = 0.0
            rest = f.copy()
            rest[i] = 0.0
            rs = rest.sum()
            if rs > 1e-15:
                f = rest / rs

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                break

    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    # U-shaped (high at edges)
    u = 1.0 - np.cos(np.pi * x / (n - 1))
    u = u ** 2 + 0.01
    inits.append(u / u.sum())

    # Hann window (bell shape)
    h = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    h = h + 0.01
    inits.append(h / h.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc = rc + 0.01
    inits.append(rc / rc.sum())

    # Linear ramp
    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp / ramp.sum())

    # Exponential
    e = np.exp(-np.abs(x - n / 2) / (n / 4.0))
    inits.append(e / e.sum())

    # Double exponential (U-shape)
    de = np.exp(-np.abs(x - n / 2) / (n / 6.0)) + 0.3
    inits.append(de / de.sum())

    # Sine squared
    ss = np.sin(np.pi * x / (n - 1)) ** 2 + 0.01
    inits.append(ss / ss.sum())

    # Cosine window (narrower)
    cw = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 6 + 0.01
    inits.append(cw / cw.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend with various shapes
            for blend in [0.3, 0.5, 0.7]:
                b = blend * (px / ps) + (1 - blend) * (u / u.sum())
                inits.append(b / b.sum())

    return inits


def sg_descent(f0, time_limit, initial_lr=0.01, mom=0.9, top_k=64, temp=0.5,
               lp_escape=True, lp_every=300):
    """Subgradient descent with momentum and LP escape."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        sg = compute_topk_gradient(f, conv, top_k, cur_temp)

        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        conv = conv_new
        cur_temp *= 1.005
        cur_lr *= 0.9995

        if lp_escape and no_improve > lp_every and rem > 20:
            g = solve_lp(f, best_obj, time_limit=10)
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    for alpha in [0.3, 0.5, 0.7, 1.0]:
                        ft = project_simplex((1 - alpha) * f + alpha * g)
                        c_try = fast_convolve(ft).max()
                        if c_try < best_obj:
                            best_obj = c_try
                            best_f = ft.copy()
                            f = ft
                            conv = fast_convolve(f)
                            no_improve = 0
                            break
            no_improve = 0
            cur_lr = initial_lr * 0.5
            v = np.zeros(n)

        if no_improve > 1000:
            rng = np.random.default_rng(it * 13 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = initial_lr
            cur_temp = temp
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization using mirror descent, LP Frank-Wolfe,
    subgradient descent, and coordinate descent with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={float(evaluate_sequence(parent.tolist())):.8f}")
    except Exception:
        pass

    # === Phase 1: Optimize parent with multiple methods ===
    print("=== Phase 1: Parent Optimization ===")
    if parent is not None:
        n_p = len(parent)

        # Mirror descent with annealing
        t1 = min(200, deadline - time.monotonic() - 500)
        if t1 > 30:
            print(f"  MD-Anneal on parent (n={n_p}), t={t1:.0f}s")
            a_opt, obj = md_anneal(parent, time_limit=t1, T_start=0.3, T_end=30.0,
                                   eta_start=0.03, eta_end=0.0005, lp_every=150)
            try_seq(a_opt, "parent_md")

        # FW-LP
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            print(f"  FW-LP on parent (n={n_p}), t={rem:.0f}s")
            a_opt, obj = fw_lp_optimize(a_opt if best_seq is not None and len(a_opt) == n_p else parent,
                                        time_limit=min(rem, 150))

        # SG descent
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            f_start = best_seq if best_seq is not None and len(best_seq) == n_p else parent
            a_opt, obj = sg_descent(f_start, time_limit=min(rem, 100),
                                    initial_lr=0.005, top_k=max(32, n_p // 6), temp=0.3)

    print("\n=== Phase 2: Broad Exploration ===")

    lengths = [384]
    if parent is not None:
        if len(parent) != 384:

            break

        n_attempts = len(inits)

            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 80)

                                   T_start=0.5, T_end=20.0,

    print(f"\n=== Phase 3: Deep Optimization ===")

            try_seq(a_opt, "coord_descent")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine MD polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, obj = md_anneal(best_seq, time_limit=min(rem, 50),
                                   T_start=5.0, T_end=100.0,
                                   eta_start=0.005, eta_end=0.00005, lp_every=100)
            try_seq(a_opt, "polish_md")

        # FW-LP polish
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = fw_lp_optimize(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "polish_fw")

        # Micro perturbations + coordinate descent
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            for _ in range(3):
                rem = deadline - time.monotonic() - 8
                if rem < 5:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0005)
                a_opt, obj = coordinate_descent(f_pert, time_limit=min(rem, 15))
                try_seq(a_opt, "micro_cd")
                if a_opt is not None and best_seq is not None and len(a_opt) == len(best_seq):
                    f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
