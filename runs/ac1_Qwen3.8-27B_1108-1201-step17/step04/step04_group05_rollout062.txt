"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive log-sum-exp smoothed
gradient descent with Nesterov momentum, multi-peak targeting, random restarts,
and length variation. Built to refine the parent construction (n=384, obj≈1.506)
toward the target of 1.5027.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_max_and_weights(b, T):
    """Compute smoothed max and softmax weights."""
    shifted = T * b - T * b.max()
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    smoothed_val = b.max() + np.log(exp_shifted.sum()) / T
    return smoothed_val, weights

def compute_gradient(a, weights, peak_indices):
    """Compute gradient of weighted sum of b[peak_indices] w.r.t. a.
    grad[i] = 2 * sum_k w_k * a[peak_k - i] (if 0 <= peak_k - i < n)
    """
    n = len(a)
    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        w = weights[idx]
        if w < 1e-15:
            continue
        k = int(peak_indices[idx])
        # grad[i] += 2*w*a[k-i] for i where 0 <= k-i < n
        # i.e., i ranges from max(0, k-n+1) to min(k, n-1)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            # a[k-i] for i in [i_start, i_end]
            # k-i ranges from k-i_start to k-i_end
            idx_range = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end+1] += 2.0 * w * a[idx_range]
    return grad

def grad_descent_smoothed(a0, T, n_steps, lr, momentum, top_k, time_limit):
    """Projected gradient descent on the smoothed objective."""
    a = a0.copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    mu = momentum
    best_obj = 2 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    cur_lr = lr
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Get top-k peaks
        if top_k >= len(b):
            peaks = np.arange(len(b))
            w = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(b, -top_k)[-top_k:]
            peak_vals = b[peaks]
            # Softmax weights
            shifted = T * peak_vals - T * peak_vals.max()
            exp_vals = np.exp(shifted)
            w = exp_vals / exp_vals.sum()
        
        # Compute gradient
        grad = compute_gradient(a, w, peaks)
        
        # Nesterov momentum step
        v = mu * v - cur_lr * grad
        a_new = project_simplex(a + v)
        
        # Evaluate
        b_new = fast_convolve(a_new)
        obj_new = 2 * n * b_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        cur_lr *= 0.9999
        
        if step % 500 == 0:
            pass  # Could add logging
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive top-K power-smoothed
gradient descent with Nesterov momentum, basin hopping, LP-based large jumps,
and length exploration. Designed to refine the parent construction (n=384,
obj≈1.506) toward the target of 1.5027.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, weights, peak_indices):
    """Compute gradient of weighted sum of b[k] w.r.t. a.
    grad[i] = 2 * sum_k w_k * a[k-i] (if 0 <= k-i < n)"""
    n = len(a)
    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        w = weights[idx]
        if w < 1e-15:
            continue
        k = int(peak_indices[idx])
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            idx_range = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * w * a[idx_range]
    return grad


def objective_value(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_obj = objective_value(a)
    best_a = a.copy()
    best_K = K
    best_power = power

    cur_lr = lr
    step = 0
    no_improve = 0

    # Adaptive K and power
    cur_K = K if top_k_for_weights is None else top_k_for_weights
    cur_power = power

    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)

        # Get top-K peaks for gradient
        if cur_K >= len(b):
            peaks = np.arange(len(b))
        else:
            peaks = np.argpartition(b, -cur_K)[-cur_K:]

        peak_vals = b[peaks]

        # Weights proportional to b[k]^(power-1)
        if cur_power > 1.0:
            # Clip to avoid overflow
            pv = np.maximum(peak_vals, 1e-30)
            raw_w = pv ** (cur_power - 1.0)
        else:
            raw_w = np.ones(len(peaks))
        w_sum = raw_w.sum()
        if w_sum < 1e-30:
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            weights = raw_w / w_sum

        # Compute gradient
        grad = compute_gradient(a, weights, peaks)

        # Nesterov momentum step
        v = momentum * v - cur_lr * grad
        a_new = project_simplex(a + v)

        # Evaluate true objective
        b_new = fast_convolve(a_new)
        obj_new = 2.0 * n * b_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and step % 10000 == 0:
                print(f"    GD step {step}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        a = a_new
        cur_lr *= 0.99998

        # Adaptive: if stuck, try different K/power or perturb
        if no_improve > 2000:
            # Try a different K
            cur_K = rng.choice([3, 5, 8, 10, 15, 20])
            cur_power = rng.choice([2.0, 3.0, 4.0, 6.0, 8.0, 12.0])
            # Small perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.0005 * no_improve / 2000.0)
            v = np.zeros(n)
            cur_lr = 0.001 * (1.0 + no_improve / 5000.0)
            no_improve = 0

        if no_improve > 8000:
            # Larger restart
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = 0.002
            no_improve = 0

    return best_a, best_obj


def lp_large_jump(a, time_limit, max_iter=50):
    """
    Use linearized LP to make larger jumps. Solves:
    min t s.t. linearized b[k] <= t for top peaks, sum(a)=1, a>=0
    """
    n = len(a)
    rng = np.random.default_rng(99)
    best_obj = objective_value(a)
    best_a = a.copy()
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        K = min(10, len(b))
        peaks = np.argpartition(b, -K)[-K:]
        peaks_sorted = np.sort(peaks)

        # Build LP: min t s.t. b0[k] + 2*sum_j a0[k-j]*(a[j]-a0[j]) <= t
        # Variables: [t, a_0, ..., a_{n-1}]
        # Constraint for each peak k:
        #   -t + 2*sum_j a0[k-j]*a[j] <= b0[k] - 2*sum_j a0[k-j]*a0[j] + ... 
        # Actually: b0[k] + sum_j 2*a0[k-j]*(a[j]-a0[j]) <= t
        # => -t + 2*sum_j a0[k-j]*a[j] <= b0[k] - 2*b0[k] + ... 
        # Let me redo: b0[k] + 2*sum_j a0[k-j]*(a[j]-a0[j]) <= t
        # => -t + 2*sum_j a0[k-j]*a[j] <= -b0[k] + 2*sum_j a0[k-j]*a0[j]
        # Note: sum_j a0[k-j]*a0[j] is not b[k], it's a "cross term"
        # Actually b[k] = sum_j a[j]*a[k-j], and the linearization is:
        # b[k] ≈ b0[k] + sum_j [a0[k-j]*(a[j]-a0[j]) + a[j-(k-j)]... ]
        # The correct linearization: b[k] = sum_j a[j]*a[k-j]
        # ∂b[k]/∂a[i] = 2*a[k-i] (if valid)
        # So: b[k] ≈ b0[k] + sum_i 2*a0[k-i]*(a[i]-a0[i])
        # Constraint: b0[k] + sum_i 2*a0[k-i]*(a[i]-a0[i]) <= t
        # => -t + 2*sum_i a0[k-i]*a[i] <= -b0[k] + 2*sum_i a0[k-i]*a0[i]
        # Note: sum_i a0[k-i]*a0[i] = b0[k] (by definition!) when both indices are valid
        # Wait, b0[k] = sum_j a0[j]*a0[k-j]. And sum_i a0[k-i]*a0[i] is the same sum (just reindexed).
        # So: -t + 2*sum_i a0[k-i]*a[i] <= -b0[k] + 2*b0[k] = b0[k]

        # Variables: [t, a_0, ..., a_{n-1}], total n+1
        c = np.zeros(n + 1)
        c[0] = 1.0  # minimize t

        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)

        for ii, k in enumerate(peaks_sorted):
            A_ub[ii, 0] = -1.0
            # 2*sum_i a0[k-i]*a[i] for valid i
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                idx_range = k - np.arange(i_start, i_end + 1)
                A_ub[ii, i_start:i_end + 1] = 2.0 * a0[idx_range] if False else 2.0 * a[idx_range]
            b_ub[ii] = b[k]  # RHS

        # Equality: sum(a) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.array([1.0])

        bounds = [(None, None)] + [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if not res.success:
            # Perturb and try again
            a = project_simplex(a + rng.standard_normal(n) * 0.002)
            continue

        a_new = np.maximum(res.x[1:], 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a = project_simplex(a + rng.standard_normal(n) * 0.002)
            continue
        a_new = a_new / sn

        obj_new = objective_value(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Take a step toward the LP solution
            for alpha in [0.3, 0.5, 0.7]:
                a_try = project_simplex((1 - alpha) * a + alpha * a_new)
                obj_try = objective_value(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, perturb
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)

    return best_a, best_obj


def basin_hopping(a0, time_limit, rng, inner_time=10.0, n_cycles=50):
    """
    Basin hopping: optimize, perturb, optimize, ...
    Uses gradient descent as the inner optimizer.
    """
    n = len(a0)
    best_a = a0.copy()
    best_obj = objective_value(best_a)
    t0 = time.monotonic()
    cycle = 0

    while time.monotonic() - t0 < time_limit and cycle < n_cycles:
        cycle += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < inner_time * 2:
            break

        # Perturb
        scale = 0.002 + 0.003 * (cycle % 5)
        a_pert = project_simplex(best_a + rng.standard_normal(n) * scale)

        # Optimize
        t_inner = min(inner_time, remaining * 0.4)
        K_choices = [5, 8, 12, 15]
        p_choices = [3.0, 5.0, 8.0]

        best_cycle_obj = objective_value(a_pert)
        best_cycle_a = a_pert.copy()

        for K in K_choices:
            for p in p_choices[:2]:
                t_each = t_inner / (len(K_choices) * 2)
                a_opt, obj = grad_desc_topk(a_pert, time_limit=t_each, K=K,
                                             power=p, lr=0.0015, momentum=0.93,
                                             seed=int(time.monotonic() * 1000) % 100000)
                if obj < best_cycle_obj:
                    best_cycle_obj = obj
                    best_cycle_a = a_opt

        if best_cycle_obj < best_obj:
            best_obj = best_cycle_obj
            best_a = best_cycle_a.copy()
            print(f"  BH cycle {cycle}: new best obj={best_obj:.10f}")

    return best_a, best_obj


def try_lengths(a_best, n_best, time_limit, rng):
    """Try nearby lengths by interpolation."""
    best_obj = objective_value(a_best)
    best_a = a_best.copy()
    t0 = time.monotonic()

    for delta in [-4, -2, 2, 4, -8, 8, -16, 16]:
        n_try = n_best + delta
        if n_try < 64 or n_try > 2000:
            continue
        if time.monotonic() - t0 > time_limit - 10:
            break

        a_try = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_best), a_best)
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s

        t_alloc = min(20.0, (time_limit - (time.monotonic() - t0) - 5) / 2.0)
        a_opt, obj = grad_desc_topk(a_try, time_limit=t_alloc, K=10,
                                     power=5.0, lr=0.002, momentum=0.93,
                                     seed=42 + delta)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            print(f"  Length n={n_try}: new best obj={obj:.10f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    1. Load parent and evaluate
    2. Multi-configuration gradient descent (various K, power)
    3. LP-based large jumps
    4. Basin hopping (perturb + optimize cycles)
    5. Length exploration
    6. Final fine polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            a = p / p.sum()
            n_parent = len(a)
            parent_score = evaluate_sequence(a.tolist())
            print(f"Parent loaded: n={n_parent}, score={parent_score:.8f}")
    except Exception:
        pass

    if a is None:
        n_parent = 384
        a = np.ones(n_parent) / n_parent
        print("Using uniform initialization, n=384")

    n = n_parent
    try_seq(a, "parent")

    # === Phase 2: Multi-configuration gradient descent ===
    print("\n=== Phase 2: Multi-config GD ===")
    t_start = time.monotonic()
    t_alloc_p2 = min(deadline - t_start - 400, 500)

    configs = [
        (8, 4.0, 0.002, 0.92),
        (10, 5.0, 0.0015, 0.93),
        (12, 6.0, 0.001, 0.94),
        (5, 3.0, 0.003, 0.90),
        (15, 8.0, 0.001, 0.94),
    ]

    for ci, (K, p, lr, mom) in enumerate(configs):
        if time.monotonic() > deadline - 300:
            break
        t_each = t_alloc_p2 / len(configs)
        print(f"  Config {ci}: K={K}, power={p}, lr={lr}, mom={mom}, time={t_each:.0f}s")
        a_opt, obj = grad_desc_topk(a, time_limit=t_each, K=K, power=p,
                                     lr=lr, momentum=mom, seed=100 + ci)
        try_seq(a_opt, f"gd_K{K}_p{p}")

    # === Phase 3: LP large jumps ===
    print("\n=== Phase 3: LP Jumps ===")
    t_start = time.monotonic()
    t_alloc_p3 = min(deadline - t_start - 250, 150)

    if t_alloc_p3 > 10:
        a_opt, obj = lp_large_jump(best_seq if best_seq is not None else a,
                                    time_limit=t_alloc_p3, max_iter=30)
        try_seq(a_opt, "lp_jumps")

    # === Phase 4: Basin hopping ===
    print("\n=== Phase 4: Basin Hopping ===")
    t_start = time.monotonic()
    t_alloc_p4 = min(deadline - t_start - 150, 400)

    if t_alloc_p4 > 30:
        a_opt, obj = basin_hopping(best_seq if best_seq is not None else a,
                                    time_limit=t_alloc_p4, rng=rng,
                                    inner_time=8.0, n_cycles=80)
        try_seq(a_opt, "basin_hop")

    # === Phase 5: Length exploration ===
    print("\n=== Phase 5: Length Exploration ===")
    t_start = time.monotonic()
    t_alloc_p5 = min(deadline - t_start - 80, 120)

    if t_alloc_p5 > 15 and best_seq is not None:
        a_opt = try_lengths(best_seq, len(best_seq), time_limit=t_alloc_p5, rng=rng)
        try_seq(a_opt, "length_explore")

    # === Phase 6: Final fine polish ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()
    t_alloc_p6 = max(0, deadline - t_start - 10)

    if t_alloc_p6 > 5 and best_seq is not None:
        # Very fine gradient descent
        a_opt, obj = grad_desc_topk(best_seq, time_limit=t_alloc_p6 * 0.6,
                                     K=5, power=10.0, lr=0.0005, momentum=0.95,
                                     seed=777)
        try_seq(a_opt, "final_polish_1")

        # Another config
        rem = deadline - time.monotonic() - 8
        if rem > 5:
            a_opt, obj = grad_desc_topk(best_seq, time_limit=rem,
                                         K=8, power=6.0, lr=0.0003, momentum=0.96,
                                         seed=888)
            try_seq(a_opt, "final_polish_2")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
