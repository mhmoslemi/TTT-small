import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def subgradient_at_peak(a, k):
    """Subgradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    i = np.arange(n)
    g = np.zeros(n)
    j1 = k - i
    mask1 = (j1 >= 0) & (j1 < n)
    g[mask1] += a[j1[mask1]]
    j2 = i - k
    mask2 = (j2 >= 0) & (j2 < n)
    g[mask2] += a[j2[mask2]]
    return g

def soft_gradient(a, conv, top_k=50, temp=0.01):
    """Weighted subgradient using soft-max over top peaks."""
    n = len(a)
    # Get top-k peaks
    idx = np.argpartition(conv, -top_k)[-top_k:]
    vals = conv[idx]
    # Soft-max weights
    scaled = vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        if wi < 1e-12:
            continue
        g += wi * subgradient_at_peak(a, int(ki))
    return g

def lp_direction(a, M=None, C=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if M is None:
        conv = fast_convolve(a)
        M = conv.max()
    if C is None:
        C = build_conv_matrix(a)
    
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                     options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-10:
                return g
    except:
        pass
    return a.copy()

def mirror_descent_step(a, grad, lr):
    """Multiplicative weight update on the simplex."""
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(a_new, 1e-300)  # Avoid underflow
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def optimize_mirror(a0, time_limit, lr_init=0.1, temp_init=0.05, top_k=30, 
                    cool_rate=1.002, lr_decay=0.999):
    """Mirror descent with adaptive temperature and learning rate."""
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        grad = soft_gradient(a, conv, top_k=top_k, temp=temp)
        a = mirror_descent_step(a, grad, lr)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= lr_decay
        temp *= cool_rate

            break
    
    return best_a, best_obj

def generate_init(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Hann window (raised cosine)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits['hann'] = hann / hann.sum()
    
    # Gaussian (various widths)
    for sigma_frac in [0.2, 0.3, 0.4, 0.5]:
        name = f'gauss_{int(sigma_frac*100)}'
        g = np.exp(-0.5 * ((x - center) / (n * sigma_frac)) ** 2)
        inits[name] = g / g.sum()
    
    # Sine (single period)
    sine = (1 + np.sin(2 * np.pi * x / n)) / 2
    inits['sine'] = sine / sine.sum()
    
    # Bimodal
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps
    
    return inits

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Adaptive subgradient descent with Polyak-style step sizing
2. LP-based direction finding for aggressive flattening
3. Mirror descent (multiplicative weights) for exploration
4. Multi-length exploration with hierarchical initialization
5. Stagnation escape via structured perturbations

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def subgradient_single(a, k):
    """Subgradient of (a*a)[k] w.r.t. a at index k."""
    n = len(a)
    i = np.arange(n)
    g = np.zeros(n)
    j1 = k - i
    mask1 = (j1 >= 0) & (j1 < n)
    g[mask1] += a[j1[mask1]]
    j2 = i - k
    mask2 = (j2 >= 0) & (j2 < n)
    g[mask2] += a[j2[mask2]]
    return g


def weighted_subgradient(a, conv, top_k=50, temp=0.01):
    """Soft-max weighted subgradient over top peaks."""
    n = len(a)
    idx = np.argpartition(conv, -top_k)[-top_k:]
    vals = conv[idx]
    scaled = vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(idx))
        w[np.argmax(vals)] = 1.0
    else:
        w /= w_sum
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        if wi < 1e-12:
            continue
        g += wi * subgradient_single(a, int(ki))
    return g


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0)
            return g, g.sum()
    except Exception:
        pass
    return a.copy(), 1.0


def mirror_step(a, grad, lr):
    """Multiplicative weight update (mirror descent on simplex)."""
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(a_new, 1e-300)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def optimize_subgradient(a0, time_limit, lr_init=0.5, temp_init=0.02,
                         top_k=40, use_mirror=False, cool_rate=1.003,
                         lr_decay=0.9995):
    """
    Subgradient/mirror descent with adaptive step size.
    Uses Polyak-type step sizing when possible.
    """
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    running_avg = best_obj

    while time.monotonic() - t0 < time_limit:
        grad = weighted_subgradient(a, conv, top_k=top_k, temp=temp)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Polyak-style adaptive step
        current_obj = 2.0 * n * conv.max()
        gap = max(current_obj - best_obj, 1e-8)
        polyak_lr = gap / (grad_norm ** 2 + 1e-10)
        effective_lr = min(lr, polyak_lr * 0.5)
        effective_lr = max(effective_lr, 1e-6)

        if use_mirror:
            a = mirror_step(a, grad, effective_lr)
        else:
            a_new = a - effective_lr * grad / grad_norm
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()

        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        running_avg = 0.9 * running_avg + 0.1 * current_obj

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= lr_decay
        temp *= cool_rate

        if no_improve > 300:
            break

    return best_a, best_obj


def optimize_lp_iterative(a0, time_limit, aggressive=True):
    """
    LP-based iterative optimization with adaptive M and step sizing.
    Tries to take full steps to the LP solution when beneficial.
    """
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)

    no_improve = 0
    perturb_count = 0
    max_no_improve = 60

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        # Use a slightly relaxed constraint
        C = build_conv_matrix(a)
        g, g_sum = lp_solve(a, M * 1.00005, C)

        if g_sum <= 1.0 + 1e-7:
            no_improve += 1
        else:
            d = g / g_sum - a  # Direction to normalized LP solution
            improved = False

            if aggressive:
                # Try full step first (t=1), then smaller
                steps = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]
            else:
                steps = [0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]

            for t in steps:
                a_try = a + t * d
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-11:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                no_improve = 0
            else:
                no_improve += 1

            if perturb_count > 30:
                break

            else:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1

    return best_a, best_obj

def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Hann window
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    inits['hann'] = hann / hann.sum()

    # Gaussian with different widths
    for sf in [0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - center) / (n * sf)) ** 2)
        inits[f'gauss_{int(sf*100)}'] = g / g.sum()

    # Raised cosine (smooth taper)
    rc = np.sin(np.pi * x / (n - 1)) ** 2
    inits['sinc2'] = rc / rc.sum()

    # Bimodal
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    # Smooth edges
    for i in range(min(edge, 20)):
        bi[i] *= 0.5 * (1 + np.cos(np.pi * i / edge))
        bi[-1-i] *= 0.5 * (1 + np.cos(np.pi * i / edge))
    inits['bimodal'] = bi / bi.sum()

    # Power law
    pw = (x + 1.0) ** 0.5
    inits['power'] = pw / pw.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.4)

    # From parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps

    return inits
