"""

2. Exponentiated gradient with adaptive full log-sum-exp subgradients
3. L-BFGS polishing with analytic gradient in softmax parameterization

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_corr_valid(p, a, n):
    """Compute grad_j = sum_k p_k * a_{k-j} for j=0..n-1 (a out of range = 0)."""
    N = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(P * np.conj(A), N)
    return corr[:n]


def softmax_obj_grad(w, T):
    """
    Objective and gradient of smoothed score in log-space parameters w.
    a = softmax(w), score = 2*n*g_T(conv(a,a)) where g_T is log-sum-exp.
    Returns (value, grad_w).
    """
    w_shifted = w - w.max()
    exp_w = np.exp(w_shifted)
    a = exp_w / exp_w.sum()
    n = len(a)
    c = fast_convolve(a)
    Tc = T * c
    Tc = Tc - Tc.max()
    log_s = np.log(np.sum(np.exp(Tc)))
    value = 2.0 * n * (log_s + Tc.max()) / T
    # p_k = exp(T*c_k) / sum(exp(T*c_k))
    p = np.exp(Tc)
    p = p / p.sum()
    # gradient: 4*n/T * a_j * (conv_p[j] - E_p[c])
    conv_p = cross_corr_valid(p, a, n)
    Ep_c = np.dot(p, c)
    grad_w = (4.0 * n / T) * a * (conv_p - Ep_c)
    return value, grad_w


def softmax_obj_only(w, T):
    """Objective only (for numerical gradient if needed)."""
    val, _ = softmax_obj_grad(w, T)
    return val


def exp_gradient_optimize(a0, time_limit, eta0=1.5, temp0=0.003,
                          temp_decay=0.9996, eta_decay=0.99995,
                          top_k_frac=0.3, adaptive_temp=True,
                          verbose=False):
    """Improved EG with full log-sum-exp subgradient and adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    improvement_rate = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute subgradient using softmax over top-k peaks
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
            peek = c[indices]
        else:
            indices = np.arange(len(c))
            peek = c.copy()

        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w = w / ws

        # Build subgradient
        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        # Multiplicative update
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-12:
            improvement_rate = best_obj - obj_new
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}, improve={improvement_rate:.2e}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        # Adaptive temperature
        if adaptive_temp:
            if no_improve < 500:
                temp *= 0.9999  # keep exploring
            elif no_improve < 1500:
                temp *= 0.9995  # moderate decay
            else:
                temp *= 0.999  # aggressive decay to exploit
        else:
            temp *= temp_decay

        # Periodic check of running average
        if it % 2000 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay

        # Stagnation restart
        if no_improve > 5000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.1
            temp = max(temp, temp0 * 0.01)
            avg_a = a.copy()
            count = 0

    return best_a.copy(), best_obj


def lbfgs_polish(a0, time_limit, T=5.0):
    """L-BFGS polishing in softmax parameterization with analytic gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    # Convert to log-space (with symmetry breaking: w[0] = 0)
    a_clipped = np.maximum(a, 1e-30)
    w = np.log(a_clipped / a_clipped.sum())
    w = w - w[0]  # break symmetry

    c0 = fast_convolve(a)
    best_obj = c0.max()
    best_a = a.copy()

    t0 = time.monotonic()
    n_evals = 0

    def objective(w_x):
        nonlocal n_evals
        if time.monotonic() - t0 > time_limit:
            raise ValueError("Time limit exceeded")
        n_evals += 1
        val, _ = softmax_obj_grad(w_x, T)
        return val

    def gradient(w_x):
        nonlocal n_evals
        if time.monotonic() - t0 > time_limit:
            raise ValueError("Time limit exceeded")
        n_evals += 1
        _, g = softmax_obj_grad(w_x, T)
        return g

    try:
        result = minimize(objective, w, jac=gradient, method='L-BFGS-B',
                          options={'maxiter': 500, 'maxfun': 2000,
                                   'maxls': 50, 'ftol': 1e-14, 'gtol': 1e-8,
                                   'iprint': 0})
        if result.success or result.fun < objective(w):
            w_opt = result.x
            a_opt = np.exp(w_opt - w_opt.max())
            a_opt = a_opt / a_opt.sum()
            c_opt = fast_convolve(a_opt)
            if c_opt.max() < best_obj:
                best_obj = c_opt.max()
                best_a = a_opt.copy()
    except Exception as e:
        if time.monotonic() - t0 > time_limit:
            pass
        else:
            pass  # L-BFGS failed, return what we have

    # Try different temperatures
    for T2 in [3.0, 8.0, 15.0]:
        if time.monotonic() - t0 > time_limit - 5:
            break
        w2 = np.log(np.maximum(a, 1e-30))
        s2 = w2.sum()
        w2 = w2 - w2[0]

        try:
            def obj2(w_x):
                if time.monotonic() - t0 > time_limit:
                    raise ValueError("Time limit exceeded")
                val, _ = softmax_obj_grad(w_x, T2)
                return val

            def grad2(w_x):
                if time.monotonic() - t0 > time_limit:
                    raise ValueError("Time limit exceeded")
                _, g = softmax_obj_grad(w_x, T2)
                return g

            result2 = minimize(obj2, w2, jac=grad2, method='L-BFGS-B',
                               options={'maxiter': 300, 'maxfun': 1000,
                                        'ftol': 1e-14, 'gtol': 1e-8, 'iprint': 0})
            if result2.fun < obj2(w2):
                w_opt2 = result2.x
                a_opt2 = np.exp(w_opt2 - w_opt2.max())
                a_opt2 = a_opt2 / a_opt2.sum()
                c_opt2 = fast_convolve(a_opt2)
                if c_opt2.max() < best_obj:
                    best_obj = c_opt2.max()
                    best_a = a_opt2.copy()
        except Exception:
            pass

    return best_a.copy(), best_obj


def cosine_init(n, m, eps):
    """Cosine-modulated uniform distribution."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * m * x / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_cosine_init(n, n_modes=5, eps=0.3):
    """Sum of multiple cosines with zero mean."""
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        freq = n // (2 * m + 1)  # various frequencies
        if freq > 0:
            a += eps * np.cos(2.0 * np.pi * freq * x / n + 0.5 * m)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def generate_inits(parent, rng, n):
    """Generate diverse initializations."""
    inits = []

    # Cosine-modulated (key insight: m ~ n/4 reduces peak)
    for m_frac in [0.2, 0.25, 0.3, 0.15, 0.35]:
        m = max(1, int(n * m_frac))
        for eps in [0.2, 0.3, 0.4]:
            inits.append(cosine_init(n, m, eps))

    # Multi-cosine
    for n_modes in [3, 5]:
        for eps in [0.2, 0.3]:
            inits.append(multi_cosine_init(n, n_modes, eps))

    # Edge-concentrated
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Perturbed parent (if same length)
    if parent is not None and len(parent) == n:
        for scale in [0.001, 0.005, 0.02]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    # Random Dirichlet
    for alpha in [0.3, 1.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))

    return inits


def local_search(a0, time_limit, max_rounds=100):
    """Local search with multiplicative perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False
        n_perturb = max(1, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for factor in [0.7, 0.85, 1.0/0.85, 1.0/0.7,
                           0.8, 1.0/0.8, 0.9, 1.0/0.9]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-13:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_a.copy(), best_obj


def resample_sequence(a_old, n_new):
    """Resample sequence to new length using interpolation."""
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new

        print(f"  L-BFGS alloc={t_lbfgs:.0f}s")
        a_opt, _ = lbfgs_polish(best_seq, time_limit=t_lbfgs, T=5.0)
        try_seq(a_opt, "lbfgs_T5")

        rem = deadline - time.monotonic()
        if rem > 40:
            t_lbfgs2 = min(80, rem * 0.1)
            a_opt2, _ = lbfgs_polish(best_seq, time_limit=t_lbfgs2, T=10.0)
            try_seq(a_opt2, "lbfgs_T10")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()

    # Try a range of lengths
    n_candidates = [192, 256, 288, 320, 384, 448, 512, 576, 640]
    n_candidates = [x for x in n_candidates if x != n_parent]
    # Sort by proximity to best length (prefer nearby)
    n_candidates.sort(key=lambda x: abs(x - len(best_seq)))
    n_candidates = n_candidates[:6]  # top 6

    for n_new in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(120, rem / max(1, len(n_candidates)))
        if t_alloc < 15:
            break

        # Resample best to new length
        a_resampled = resample_sequence(best_seq, n_new)
        # Also try cosine init at this length
        m = max(1, int(n_new * 0.25))
        a_cos = cosine_init(n_new, m, 0.3)

        # Pick the better init
        c1 = fast_convolve(a_resampled).max()
        c2 = fast_convolve(a_cos).max()
        a_init = a_resampled if c1 < c2 else a_cos

        print(f"  n={n_new}: resample_peak={c1:.6f}, cos_peak={c2:.6f}")
        a_opt, _ = exp_gradient_optimize(a_init, time_limit=t_alloc,
                                         eta0=1.0, temp0=0.005,
                                         adaptive_temp=True,
                                         top_k_frac=0.3, verbose=False)
        try_seq(a_opt, f"len_n{n_new}")

        # Quick L-BFGS on this
        rem = deadline - time.monotonic()
        if rem > 30:
            a_opt2, _ = lbfgs_polish(a_opt, time_limit=min(30, rem * 0.3), T=5.0)
            try_seq(a_opt2, f"lbfgs_n{n_new}")

    # === Phase 5: Refine best with EG variations ===
    print(f"\n=== Phase 5: Final Refinement ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()

    if rem > 30 and best_seq is not None:
        # EG with very low temperature (exploitation)
        t1 = min(100, rem * 0.3)
        a_opt, _ = exp_gradient_optimize(best_seq, time_limit=t1,
                                         eta0=0.3, temp0=0.0005,
                                         adaptive_temp=False,
                                         temp_decay=0.9999,
                                         eta_decay=0.99999,
                                         top_k_frac=0.5, verbose=False)
        try_seq(a_opt, "eg_fine")

        # L-BFGS at low temperature
        rem = deadline - time.monotonic()
        if rem > 30:
            t2 = min(60, rem * 0.3)
            a_opt2, _ = lbfgs_polish(best_seq, time_limit=t2, T=8.0)
            try_seq(a_opt2, "lbfgs_final")

    # === Phase 6: Local search ===
    print(f"\n=== Phase 6: Local Search ===")
    rem = deadline - time.monotonic()
    if rem > 10 and best_seq is not None:
        t3 = min(60, rem * 0.5)
        a_opt, _ = local_search(best_seq, time_limit=t3, max_rounds=80)
        try_seq(a_opt, "local")

    # === Phase 7: One more EG + L-BFGS cycle ===
    rem = deadline - time.monotonic()
    if rem > 40 and best_seq is not None:
        t4 = min(80, rem * 0.5)
        a_opt, _ = exp_gradient_optimize(best_seq, time_limit=t4,
                                         eta0=0.5, temp0=0.001,
                                         adaptive_temp=True,
                                         top_k_frac=0.4, verbose=False)
        try_seq(a_opt, "eg_final2")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_opt2, _ = lbfgs_polish(a_opt, time_limit=min(30, rem * 0.6), T=5.0)
            try_seq(a_opt2, "lbfgs_final2")
            a_opt3, _ = local_search(a_opt2, time_limit=min(20, rem * 0.3), max_rounds=30)
            try_seq(a_opt3, "local_final2")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(256) / 256
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
