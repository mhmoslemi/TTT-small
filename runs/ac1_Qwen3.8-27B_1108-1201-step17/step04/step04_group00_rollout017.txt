"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-gradient mirror descent,
coordinate descent polishing, and multi-length exploration.
Key innovation: replaces expensive LP solves with efficient soft-max subgradients.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def soft_gradient(conv, a, tau, top_k=None):
    """Compute soft subgradient of max(conv) using soft-max weights."""
    n = len(a)
    if top_k is None:
        top_k = len(conv)
    top_k = min(top_k, len(conv))
    
    # Get top_k peaks
    if top_k < len(conv):
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
    else:
        top_indices = np.arange(len(conv))
    
    # Soft-max weights over top peaks
    pv = conv[top_indices]
    scaled = pv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(top_indices))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Compute gradient: sum_k w[k] * 2 * shifted_k(a)
    grad = np.zeros(n)
    for i in range(len(top_indices)):
        if w[i] < 1e-15:
            continue
        k = int(top_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # grad[m] += 2 * w[i] * a[k-m] for m in [js, je]
        m_range = np.arange(js, je + 1)
        grad[m_range] += 2.0 * w[i] * a[k - m_range]
    
    return grad

def mirror_descent_optimize(f0, time_limit, eta0=0.01, eta_decay=0.9995, tau0=0.001, tau_growth=1.002, top_k=100, verbose=False):
    """Mirror descent with soft-max gradient on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    eta = eta0
    tau = tau0
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad = soft_gradient(conv, f, tau, top_k)
        
        # Mirror descent update: f_new = f * exp(-eta * grad) / Z
        # To avoid underflow, use log-space
        log_update = -eta * grad
        log_update -= log_update.max()  # for numerical stability
        update = np.exp(log_update)
        f_new = f * update
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-300:
            break
        f_new = f_new / f_new_sum
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, tau={tau:.6f}")
        else:
            no_improve += 1
        
        f = f_new
        eta *= eta_decay
        tau *= tau_growth
        
        if no_improve > 500:
            # Restart with noise
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            obj = conv.max()
            eta = max(eta0 * 0.3, eta * 0.5)
            tau = max(tau0, tau * 0.5)
            no_improve = 0
    
    return best_f, best_obj

def gradient_descent_simplex(f0, time_limit, lr=0.001, mom=0.9, top_k=100, tau=0.001, verbose=False):
    """Gradient descent with momentum and simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_tau = tau
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad = soft_gradient(conv, f, cur_tau, top_k)
        
        v = mom * v - cur_lr * grad
        f_new = project_simplex(f - v)
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    GD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        f = f_new
        cur_lr *= 0.9998
        cur_tau *= 1.001
        
        if no_improve > 800:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0001)
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated soft-gradient
methods (mirror descent, Nesterov-accelerated GD), coordinate descent polishing,
and multi-length exploration. Replaces expensive LP solves with O(n log n) gradients.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_weights(conv, tau, top_k=None):
    """Compute soft-max weights over the peaks of conv."""
    size = len(conv)
    if top_k is not None and top_k < size:
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros(size)
        pv = conv[top_indices]
    else:
        top_indices = np.arange(size)
        pv = conv
        w = None
    
    scaled = pv / tau
    scaled -= scaled.max()
    weights = np.exp(scaled)
    ws = weights.sum()
    if ws < 1e-300:
        weights = np.zeros(len(pv))
        weights[np.argmax(pv)] = 1.0
    else:
        weights /= ws
    
    if w is not None:
        w[top_indices] = weights
        return w
    else:
        return weights


def soft_gradient_fft(conv, a, tau, top_k=None):
    """
    Compute soft subgradient of max(conv) using FFT-based cross-correlation.
    grad[m] = 2 * sum_k w[k] * a[k-m] = 2 * (w cross-correlate a)[m]
    """
    n = len(a)
    size = len(conv)  # 2n-1
    # Need fft_size >= size + n - 1 for linear cross-correlation
    need = size + n - 1
    fft_size = 1 << (need - 1).bit_length()
    
    w = compute_soft_weights(conv, tau, top_k)
    
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    grad = 2.0 * corr[:n]
    
    return grad


def mirror_descent(f0, time_limit, eta0=0.01, eta_decay=0.9995, tau0=0.001,
                   tau_growth=1.003, top_k=100, verbose=False):
    """Mirror descent with soft-max gradient on the probability simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    eta = eta0
    tau = tau0
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad = soft_gradient_fft(conv, f, tau, top_k)
        
        # Mirror descent: f_new = f * exp(-eta * grad) / Z
        log_update = -eta * grad
        max_log = log_update.max()
        if max_log > 50:
            log_update -= max_log
        update = np.exp(log_update)
        f_new = f * update
        f_sum = f_new.sum()
        if f_sum < 1e-300:
            break
        f_new = f_new / f_sum
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, tau={tau:.6f}")
        else:
            no_improve += 1
        
        f = f_new
        eta *= eta_decay
        tau *= tau_growth
        
        if no_improve > 1000:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            eta = max(eta0 * 0.3, eta * 0.5)
            tau = max(tau0 * 0.5, tau * 0.3)
            no_improve = 0
    
    return best_f, best_obj


def nesterov_gd(f0, time_limit, lr=0.001, top_k=100, tau0=0.001,
                tau_growth=1.002, verbose=False):
    """Nesterov-accelerated gradient descent with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()
    conv_y = fast_convolve(y)
    best_obj = conv_y.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    cur_lr = lr
    tau = tau0
    no_improve = 0
    it = 0
    t_prev = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        beta = (1.0 - t_prev) / t_curr
        
        # Nesterov momentum: y = f + beta * (f - f_prev)
        # Approximate: y = (1+beta)*f - beta*f_prev
        # Simplified: just use y = f + beta*(f - y_prev)
        y_new = f + beta * (f - y)
        y_new = project_simplex(y_new)
        y = y_new
        
        conv_y = fast_convolve(y)
        grad = soft_gradient_fft(conv_y, y, tau, top_k)
        
        # Gradient step
        f_new = project_simplex(y - cur_lr * grad)
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    NGD iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1
        
        f = f_new
        t_prev = t_curr
        cur_lr *= 0.9998
        tau *= tau_growth
        
        if no_improve > 800:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            cur_lr = max(lr * 0.5, 0.0001)
            tau = max(tau0, tau * 0.5)
            no_improve = 0
            t_prev = 1.0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.5, 0.7, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
