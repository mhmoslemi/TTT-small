        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        # Compute update with optional momentum
        log_update = -eta * sg / sg_max
        if momentum > 0:
            buf = momentum * buf + (1 - momentum) * log_update
            log_update = buf

        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fft_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"      EG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        # Periodic running average check
        if it % 1000 == 0:
            c_avg = fft_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-13:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay
        temp *= temp_decay
        if temp < 1e-5:
            temp = 1e-5

        # Stagnation restart
        if no_improve > 3000:
            rng_local = np.random.default_rng((it * 7919 + 13) % (2**31))
            scale = 0.001 + 0.002 * min(1.0, no_improve / 5000.0)
            a = best_a + rng_local.standard_normal(n) * scale
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fft_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.3
            temp = max(temp, temp0 * 0.1)
            buf = np.zeros(n)
            avg_a = a.copy()
            count = 0

    return best_a.copy(), best_obj


def scipy_smooth_polish(a0, time_limit):
    """Polish using L-BFGS with decreasing log-sum-exp temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    n = len(a)

    t0 = time.monotonic()
    epsilons = [0.05, 0.02, 0.008, 0.003, 0.001]
    result_a = a.copy()

    for eps in epsilons:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 3:
            break
        alloc = min((time_limit - elapsed) / len(epsilons), time_limit - elapsed)
        if alloc < 2:
            break

        # Initialize b from current a
        b = np.log(result_a + 1e-300)
        b = b - b.mean()

        def obj_func(bv, eps=eps):
            bs = bv - bv.max()
            eb = np.exp(bs)
            av = eb / eb.sum()
            cv = fft_convolve(av)
            cs = cv / eps
            cm = cs.max()
            return 2.0 * n * (cm + eps * np.log(np.sum(np.exp(cs - cm))))

        def grad_func(bv, eps=eps):
            bs = bv - bv.max()
            eb = np.exp(bs)
            av = eb / eb.sum()
            cv = fft_convolve(av)
            cs = cv / eps
            cm = cs.max()
            sigma = np.exp(cs - cm)
            sigma /= sigma.sum()
            ga = compute_smooth_grad_a(av, sigma, n)
            mg = np.dot(ga, av)
            return 2.0 * n * av * (ga - mg)

        try:
            res = minimize(
                obj_func,
                b,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': 200, 'maxfun': 2000, 'iprint': -1,
                         'gtol': 1e-8}
            )
            bopt = res.x
            bs = bopt - bopt.max()
            aopt = np.exp(bs)
            aopt = aopt / aopt.sum()

            c_opt = fft_convolve(aopt)
            c_cur = fft_convolve(result_a)
            if c_opt.max() < c_cur.max():
                result_a = aopt
                if verbose_flag:
                    print(f"      SciPy polish eps={eps}: {c_cur.max():.10f} -> {c_opt.max():.10f}")
        except Exception as e:
            pass

    return result_a


def local_search_polish(a0, time_limit):
    """Coordinate-wise local search with both multiplicative and additive perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    n = len(a)

    c = fft_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    max_no_improve = 50

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        n_perturb = max(4, n // 12)
        coords = rng.choice(n, size=n_perturb, replace=False)
        improved = False

        for i in coords:
            old_val = a[i]

            # Try multiplicative factors
            for factor in [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.25, 1.4]:
                a[i] = old_val * factor
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s2
                c2 = fft_convolve(a_norm)
                if c2.max() < best_obj - 1e-13:
                    best_obj = c2.max()
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
                else:
                    a[i] = old_val

            # Try small additive shifts
            if not improved:
                for delta in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                    new_val = old_val + delta
                    if new_val < 0:
                        continue
                    a[i] = new_val
                    s2 = a.sum()
                    if s2 < 1e-15:
                        a[i] = old_val
                        continue
                    a_norm = a / s2
                    c2 = fft_convolve(a_norm)
                    if c2.max() < best_obj - 1e-13:
                        best_obj = c2.max()
                        best_a = a_norm.copy()
                        a = a_norm.copy()
                        improved = True
                        break
                    else:
                        a[i] = old_val

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_a.copy()


def generate_diverse_inits(parent, n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Parent (resampled if needed)
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent) / (np.interp(x_new, x_old, parent).sum() + 1e-300))

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    h = np.sin(np.pi * x / (n - 1)) ** 2
    h = np.maximum(h, 1e-10)
    inits.append(h / h.sum())

    # Exponential decay (several rates)
    for beta in [0.5, 1.0, 2.0, 4.0]:
        e = np.exp(-beta * x / (n - 1))
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / (n - 1))
        inits.append(e2 / e2.sum())

    # U-shaped (edge concentrated)
    for alpha in [0.5, 1.0, 2.0]:
        u = (x / (n - 1)) ** alpha + ((n - 1 - x) / (n - 1)) ** alpha + 0.1
        inits.append(u / u.sum())

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            g = np.exp(-0.5 * ((x - center * (n - 1)) / (width * n)) ** 2)
            inits.append(g / g.sum())

    # Cosine-based
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / (n - 1))) + 0.05
        inits.append(c / c.sum())

    # Random smooth (sum of random Gaussians)
    for _ in range(5):
        n_bumps = rng.integers(2, 6)
        r = np.zeros(n)
        for _ in range(n_bumps):
            mu = rng.uniform(0, n - 1)
            sig = rng.uniform(0.05 * n, 0.3 * n)
            amp = rng.uniform(0.5, 2.0)
            r += amp * np.exp(-0.5 * ((x - mu) / sig) ** 2)
        r = np.maximum(r, 0.01)
        inits.append(r / r.sum())

    # Perturbed parent
    if parent is not None and len(parent) == n:
        for scale in [0.005, 0.02, 0.05]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    return inits


def try_symmetrize(a):
    """Try the symmetrized version and return the better one."""
    sym = (a + a[::-1]) / 2.0
    s = sym.sum()
    if s < 1e-15:
        return a
    sym = sym / s
    c1 = fft_convolve(a).max()
    c2 = fft_convolve(sym).max()
    if c2 < c1:
        return sym
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: exponentiated gradient + scipy polish + local search.
    """
    global verbose_flag
    verbose_flag = True

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            if np.isfinite(ps):
                best_score = ps
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(256) / 256
        print("No parent, using uniform n=256")
        best_score = evaluate_sequence(parent.tolist())
        best_seq = parent.copy()

    n = len(parent)
    print(f"Starting: n={n}, best={best_score:.10f}")

    # Phase 1: Main EG on parent
    print("\n=== Phase 1: EG on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(500, rem * 0.35)
    if t1 > 30:
        a_opt, obj = exp_gradient_optimize(parent, time_limit=t1,
                                           eta0=1.5, temp0=0.005,
                                           temp_decay=0.9996, eta_decay=0.99995,
                                           top_k_frac=0.3, verbose=True,
                                           momentum=0.1)
        try_seq(a_opt, "eg_main")
        # Also try symmetrized
        a_sym = try_symmetrize(a_opt)
        try_seq(a_sym, "eg_main_sym")

    # Phase 2: EG with different hyperparameters
    print(f"\n=== Phase 2: EG variations ===")
    print(f"  Best: {best_score:.10f}")
    configs = [
        {"eta0": 0.5, "temp0": 0.02, "temp_decay": 0.9998, "eta_decay": 0.9999, "top_k_frac": 0.15, "momentum": 0.3},
        {"eta0": 3.0, "temp0": 0.002, "temp_decay": 0.9994, "eta_decay": 0.9998, "top_k_frac": 0.4, "momentum": 0.0},
        {"eta0": 1.0, "temp0": 0.01, "temp_decay": 0.9999, "eta_decay": 0.99999, "top_k_frac": 0.2, "momentum": 0.2},
        {"eta0": 2.0, "temp0": 0.008, "temp_decay": 0.9997, "eta_decay": 0.9999, "top_k_frac": 0.35, "momentum": 0.15},
    ]
    for ci, cfg in enumerate(configs):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(150, rem / (len(configs) - ci + 1))
        if t_alloc < 20:
            break
        a_opt, _ = exp_gradient_optimize(best_seq, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"eg_cfg{ci}")

    # Phase 3: Scipy polish
    print(f"\n=== Phase 3: Scipy polish ===")
    print(f"  Best: {best_score:.10f}")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(200, rem * 0.25)
        a_opt = scipy_smooth_polish(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "scipy1")
        a_sym = try_symmetrize(a_opt)
        try_seq(a_sym, "scipy1_sym")

    # Phase 4: Diverse initializations
    print(f"\n=== Phase 4: Diverse inits ===")
    print(f"  Best: {best_score:.10f}")
    inits = generate_diverse_inits(best_seq, n, rng)
    for i, f_init in enumerate(inits[:15]):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(30, (rem - 20) / max(1, 15 - i))
        if t_alloc < 8:
            break
        a_opt, _ = exp_gradient_optimize(f_init, time_limit=t_alloc,
                                         eta0=1.0, temp0=0.01,
                                         temp_decay=0.9997, eta_decay=0.99995,
                                         top_k_frac=0.25, momentum=0.1)
        try_seq(a_opt, f"init{i}")

    # Phase 5: Local search
    print(f"\n=== Phase 5: Local search ===")
    print(f"  Best: {best_score:.10f}")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(120, rem * 0.2)
        a_opt = local_search_polish(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "local1")

    # Phase 6: More scipy polish (with lower eps)
    print(f"\n=== Phase 6: Fine scipy ===")
    print(f"  Best: {best_score:.10f}")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(150, rem * 0.3)
        a_opt = scipy_smooth_polish(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "scipy2")

    # Phase 7: Final EG + local
    print(f"\n=== Phase 7: Final refinement ===")
    print(f"  Best: {best_score:.10f}")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(100, rem * 0.4)
        a_opt, _ = exp_gradient_optimize(best_seq, time_limit=t_alloc,
                                         eta0=0.5, temp0=0.001,
                                         temp_decay=0.99995, eta_decay=0.99999,
                                         top_k_frac=0.5, momentum=0.2)
        try_seq(a_opt, "eg_final")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_opt = local_search_polish(best_seq, time_limit=min(60, rem - 5))
        try_seq(a_opt, "local_final")

    # Phase 8: One last scipy with very small eps
    rem = deadline - time.monotonic()
    if rem > 10:
        a_opt = scipy_smooth_polish(best_seq, time_limit=min(40, rem - 3))
        try_seq(a_opt, "scipy_final")

    # Final symmetrization attempt
    if best_seq is not None:
        a_sym = try_symmetrize(best_seq)
        try_seq(a_sym, "final_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
