def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_matrix_vec(a, g):
    """Compute a * g (linear convolution) efficiently using FFT."""
    return fast_convolve_cross(a, g)


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def objective_value(a):
    """Compute the objective value directly (sum normalized to 1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * n * b.max()


def compute_gradient_soft(a, conv, top_k=20, temp=0.005):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    if top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv))
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def lp_solve_multi(a, M_values, peak_indices, n):
    """
    Solve LP: max sum(g) s.t. (a*g)_{k} <= M_k for k in peak_indices, g >= 0.
    Returns (g, success).
    """
    c = -np.ones(n)
    # Build constraint matrix: for each peak index k, the row is (a * 1_{[0,n-1] shifted})
    A_ub_rows = []
    b_ub = []
    for k, M_k in zip(peak_indices, M_values):
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            row[j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
        A_ub_rows.append(row)
        b_ub.append(M_k)

    if not A_ub_rows:
        return a.copy(), True

    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5})
        if result.success:
            g = result.x
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass

    # Fallback: single peak constraint
    try:
        k = int(peak_indices[0])
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            row[j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
        result = linprog(c, A_ub=row.reshape(1, -1), b_ub=np.array([M_values[0]]),
                         bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 3})
        if result.success:
            g = result.x
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass

    return a.copy(), False


def bisection_line_search(a, d, best_obj, max_iter=30):
    """
    Find best step size t in [0, 1] such that obj(a + t*d) is minimized.
    Uses bisection on the derivative (approximated).
    """
    n = len(a)
    lo = 0.0
    hi = 1.0
    best_t = 0.0
    best_val = best_obj

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        a_try = a + mid * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            hi = mid
            continue
        a_try = a_try / s
        val = objective_value(a_try)

        if not np.isfinite(val):
            hi = mid
            continue

        if val < best_val:
            best_val = val
            best_t = mid

        # Check derivative sign via small perturbation
        if mid > 0.01:
            eps = 0.001
            a_plus = a + (mid + eps) * d
            a_plus = np.maximum(a_plus, 0.0)
            sp = a_plus.sum()
            if sp > 1e-15:
                a_plus = a_plus / sp
                val_plus = objective_value(a_plus)
            else:
                val_plus = best_val
            a_minus = a + (mid - eps) * d
            a_minus = np.maximum(a_minus, 0.0)
            sm = a_minus.sum()
            if sm > 1e-15:
                a_minus = a_minus / sm
                val_minus = objective_value(a_minus)
            else:
                val_minus = best_val

            if val_plus < val_minus:
                lo = mid  # derivative negative, go right
            else:
                hi = mid  # derivative positive, go left
        else:
            if val < best_obj:
                lo = mid
            else:
                hi = mid

    return a + best_t * d, best_val, best_t


def lp_optimize_adaptive(a0, time_limit, top_k_peaks=3, verbose=False):
    """
    LP-based optimization with adaptive peak tightening and bisection line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0
    rng = np.random.default_rng(42 + n)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2.0:
            break

        # Find top peaks
        k_peaks = min(top_k_peaks, len(conv))
        peak_indices = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]
        peak_vals = conv[peak_indices]

        # Adaptive tightening: start with small eps, increase over time
        base_eps = 0.001 + 0.008 * min(perturb_count, 10)
        M_values = peak_vals * (1.0 - base_eps)

        # Solve LP
        M_max = peak_vals[0] * (1.0 - base_eps * 0.5)
        g, success = lp_solve_multi(a, [M_max], [peak_indices[0]], n)

        if not success or g.sum() <= a.sum() * (1.0 + 1e-8):
            no_improve += 1
            if no_improve >= 50:
                # Perturb and reset
                scale = 0.002 + 0.004 * min(perturb_count, 5)
                noise = rng.standard_normal(n) * scale
                a = best_a + noise
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a = best_a.copy()
                else:
                    a = a / s
                conv = fast_convolve(a)
                no_improve = 0
                perturb_count += 1
                if perturb_count > 50:
                    break
            iteration += 1
            continue

        # Direction
        d = g - a
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-10:
            no_improve += 1
            iteration += 1
            continue

        # Bisection line search
        a_new, val_new, t_used = bisection_line_search(a, d, best_obj, max_iter=25)

        if val_new < best_obj - 1e-10:
            a = a_new
            conv = fast_convolve(a)
            best_obj = val_new
            best_a = a.copy()
            no_improve = 0
            if verbose and iteration % 50 == 0:
                print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")
        else:
            no_improve += 1
            # Try smaller step
            for t_try in [0.3, 0.1, 0.05, 0.02]:
                a_try = a + t_try * d
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try / st
                val_try = objective_value(a_try)
                if val_try < best_obj - 1e-10:
                    a = a_try
                    conv = fast_convolve(a)
                    best_obj = val_try
                    best_a = a.copy()
                    no_improve = 0
                    break

        iteration += 1

    return best_a, best_obj


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.003):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = compute_gradient_soft(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        temp *= 1.004
        mom = min(0.97, mom + 0.0003)

        if no_improve >= 300:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Random coordinate descent for fine-tuning."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, objective_value(a)
    a = a / s
    n = len(a)

    best_obj = objective_value(a)
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        batch = max(1, int(300000 / n))
        for _ in range(batch):
            if no_improve > 300:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.002, 0.005, 0.01, 0.001, -0.002, -0.005, -0.01, -0.001]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                val = objective_value(a_try)
                if val < best_obj - 1e-10:
                    a = a_try
                    best_obj = val
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations with emphasis on asymmetric shapes."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    xm = x.astype(np.float64)

    inits['uniform'] = np.ones(n) / n

    # Asymmetric linear ramp (tilted)
    ramp = np.linspace(0.3, 1.7, n)
    inits['tilted'] = ramp / ramp.sum()

    # Asymmetric quadratic
    quad = (x / n) ** 2 + 0.1
    inits['quad_asc'] = quad / quad.sum()
    quad2 = (1 - x / n) ** 2 + 0.1
    inits['quad_desc'] = quad2 / quad2.sum()

    # Sine-based
    for freq in [1, 2, 3]:
        s = (np.sin(np.pi * freq * x / n) ** 2 + 0.05)
        inits[f'sine{freq}'] = s / s.sum()

    # Asymmetric sine (phase shifted)
    s_asym = (np.sin(np.pi * (x / n + 0.3)) ** 2 + 0.05)
    inits['sine_asym'] = s_asym / s_asym.sum()

    # Bimodal (mass at edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.2  # Slightly asymmetric
    inits['bimodal'] = bi / bi.sum()

    # Gaussian centered off-center
    for center_frac in [0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - center_frac * n) / (0.2 * n)) ** 2)
        inits[f'gauss_{int(center_frac*10)}'] = g / g.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    # Exponential decay
    expd = np.exp(-x / (n * 0.5))
    inits['exp_decay'] = expd / expd.sum()
    expd2 = np.exp(-(n - x) / (n * 0.5))
    inits['exp_rise'] = expd2 / expd2.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
                # Tilted parent
                tilt = np.linspace(0.8, 1.2, n)
                pt = p / ps * tilt
                inits['parent_tilted'] = pt / pt.sum()
                # Phase-shifted parent
                shift = n // 8
                pshift = np.roll(p / ps, shift)
                pshift = np.maximum(pshift, 0)
                inits['parent_shift'] = pshift / pshift.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
                    tilt = np.linspace(0.85, 1.15, n)
                    pt = pp / pps * tilt
                    inits['parent_tilt'] = pt / pt.sum()

    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext

    # Select top candidates by (score, length diversity)
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = n
        if key not in seen:
            top_configs.append((n, a_opt, score))
            seen.add(key)
        if len(top_configs) >= 4:
            break

    if best_sequence is not None:
        blen = len(best_sequence)
        if blen not in seen:
            top_configs.append((blen, best_sequence.copy(), best_score))

    remaining2 = deadline - time.monotonic() - 400.0
    n_cand = max(1, len(top_configs))
    t_per_lp = max(60.0, remaining2 * 0.7 / n_cand)

    for i, (n, a0, sc) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 300)
        if t_alloc < 20:
            break

        print(f"  LP cand {i}: n={n}, init_score={sc:.6f}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize_adaptive(a0, time_limit=t_alloc, top_k_peaks=3, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
        elif np.isfinite(score):
            print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t2_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Nesterov + LP alternating on best (350s)
    # =========================================================
    print("=== Phase 3: Alternating Refinement ===")
    t3_start = time.monotonic()
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining3 = deadline - time.monotonic() - 150.0

        # Alternate between gradient and LP
        a_cur = best_sequence.copy()
        a_cur = a_cur / a_cur.sum()

        n_rounds = 4
        t_per_round = remaining3 / n_rounds

        for r in range(n_rounds):
            if time.monotonic() > deadline - 120:
                break
            t_ok = min(t_per_round, deadline - time.monotonic() - 100)
            if t_ok < 10:
                break

            # Gradient step
            a_g, obj_g = nesterov_optimize(a_cur, time_limit=t_ok * 0.5,
                                           lr_init=0.008, momentum=0.92,
                                           top_k=50, temp_init=0.0015)
            sc = float(evaluate_sequence(a_g.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = a_g.copy()
                a_cur = a_g.copy()
                a_cur = a_cur / a_cur.sum()
                print(f"  Round {r} grad: {sc:.6f} *NEW BEST*")

            # LP step
            a_l, obj_l = lp_optimize_adaptive(a_cur, time_limit=t_ok * 0.5,
                                              top_k_peaks=2)
            sc = float(evaluate_sequence(a_l.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = a_l.copy()
                a_cur = a_l.copy()
                a_cur = a_cur / a_cur.sum()
                print(f"  Round {r} LP: {sc:.6f} *NEW BEST*")

    print(f"Phase 3 done ({time.monotonic()-t3_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Try different lengths with best sequence as seed (150s)
    # =========================================================
    print("=== Phase 4: Length Exploration ===")
    t4_start = time.monotonic()
    if best_sequence is not None:
        n_cur = len(best_sequence)
        try_lengths = [n_cur - 128, n_cur + 128, n_cur + 256]
        try_lengths = [nl for nl in try_lengths if 128 <= nl <= 2048 and nl != n_cur]

        remaining4 = deadline - time.monotonic() - 50.0
        t_per_len = max(15.0, remaining4 / max(1, len(try_lengths)))

        for nl in try_lengths:
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(t_per_len, deadline - time.monotonic() - 30)
            if t_alloc < 10:
                break

            a_seed = extend_sequence(best_sequence, nl)
            a_opt, obj = lp_optimize_adaptive(a_seed, time_limit=t_alloc, top_k_peaks=3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Len {nl}: {score:.6f} *NEW BEST*")
            elif np.isfinite(score):
                print(f"  Len {nl}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t4_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final coordinate descent + micro LP polish (50s)
    # =========================================================
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining5 = min(50.0, deadline - time.monotonic() - 3)
        if remaining5 > 5:
            # Coordinate descent
            a_cd, obj_cd = coordinate_descent(best_sequence, time_limit=remaining5 * 0.6)
            score = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cd.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

            # Micro LP
            if time.monotonic() < deadline - 3:
                t_micro = min(15.0, deadline - time.monotonic() - 2)
                if t_micro > 3:
                    a_lp, obj_lp = lp_optimize_adaptive(best_sequence, time_limit=t_micro,
                                                        top_k_peaks=2)
                    score = float(evaluate_sequence(a_lp.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_lp.copy()
                        print(f"  MicroLP: {score:.6f} *NEW BEST*")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
