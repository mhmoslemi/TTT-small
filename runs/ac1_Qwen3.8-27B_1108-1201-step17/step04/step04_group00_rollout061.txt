def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe (main optimization)
2. Top-k subgradient projected gradient (fast polishing)
3. Coordinate descent (fine polishing)
4. Multi-length exploration with interpolation

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n):
    """Solve LP: max 1^T g s.t. (f*g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0
    m_relax = 1.0

        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        if g is None:
            m_relax = min(m_relax * 1.15, 5.0)
            if m_relax > 5.0:
                m_relax = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 60:

                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        f_init = make_initialization(n_parent, meth)
        # Blend with parent for better start
        f_init = 0.5 * f_init + 0.5 * (parent / parent.sum())
        f_init = project_simplex(f_init)
        sc_init = float(evaluate_sequence(f_init.tolist()))
        if np.isfinite(sc_init) and sc_init < best_score:
            best_score = sc_init
            best_seq = f_init.copy()
            print("  Better init [%s]: %.6f" % (meth, sc_init))

        t_alloc = min(rem * 0.5, 120.0)
        if t_alloc > 15:
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print("  NEW BEST [lpfw_%s]: %.6f" % (meth, sc))

    # === Phase 3: Multi-length exploration ===
    print("\n=== Phase 3: Multi-length exploration (best=%.6f) ===" % best_score)
    base = best_seq if best_seq is not None else parent

    target_lengths = [896, 640, 1024, 512]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300.0
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 180.0)
        print("  LP-FW n=%d, t=%.0fs" % (target_n, t_alloc))
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print("  NEW BEST [lpfw_n%d]: %.6f" % (target_n, sc))
        else:
            print("  n=%d: score=%.6f (no improvement)" % (target_n, sc))

    # === Phase 4: Top-k subgradient polishing ===
    print("\n=== Phase 4: Subgradient polishing (best=%.6f) ===" % best_score)
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200.0
        if rem > 20:
            t_sg = min(rem * 0.7, 200.0)
            print("  Top-k subgrad t=%.0fs" % t_sg)
            a_opt, _ = topk_subgrad_optimize(best_seq, time_limit=t_sg,
                                             top_k=max(10, n_best // 10),
                                             lr_init=0.3, lr_decay=0.9997)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print("  NEW BEST [subgrad]: %.6f" % sc)

            # Second round with different top_k
            rem = deadline - time.monotonic() - 150.0
            if rem > 15:
                t_sg2 = min(rem * 0.6, 100.0)
                a_opt2, _ = topk_subgrad_optimize(best_seq, time_limit=t_sg2,
                                                  top_k=max(5, n_best // 20),
                                                  lr_init=0.2, lr_decay=0.9998)
                sc2 = float(evaluate_sequence(a_opt2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt2.copy()
                    print("  NEW BEST [subgrad2]: %.6f" % sc2)

    # === Phase 5: Coordinate descent ===
    print("\n=== Phase 5: Coordinate descent (best=%.6f) ===" % best_score)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150.0
        if rem > 15:
            t_cd = min(rem * 0.7, 180.0)
            print("  CD t=%.0fs" % t_cd)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print("  NEW BEST [cd]: %.6f" % sc)

    # === Phase 6: Final LP-FW + subgradient ===
    print("\n=== Phase 6: Final refinement (best=%.6f) ===" % best_score)
    if best_seq is not None:
        n_best = len(best_seq)

        # LP-FW on current best
        rem = deadline - time.monotonic() - 80.0
        if rem > 15:
            t_final = min(rem * 0.5, 120.0)
            print("  Final LP-FW t=%.0fs" % t_final)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final, verbose=False)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print("  NEW BEST [final_lpfw]: %.6f" % sc)

        # Subgradient polish
        rem = deadline - time.monotonic() - 50.0
        if rem > 10:
            t_fin_sg = min(rem * 0.6, 60.0)
            a_opt, _ = topk_subgrad_optimize(best_seq, time_limit=t_fin_sg,
                                             top_k=max(5, n_best // 25),
                                             lr_init=0.15, lr_decay=0.9998)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print("  NEW BEST [final_sg]: %.6f" % sc)

        # Tiny perturbations + quick polish
        rem = deadline - time.monotonic() - 30.0
        for pert in range(3):
            rem = deadline - time.monotonic() - 20.0
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.4, 25.0)
            a_opt, _ = topk_subgrad_optimize(f_pert, time_limit=t_pert,
                                             top_k=max(5, n_best // 20),
                                             lr_init=0.1, lr_decay=0.9998)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print("  NEW BEST [perturb_%d]: %.6f" % (pert, sc))

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_seq)))
    return [float(x) for x in best_seq]
