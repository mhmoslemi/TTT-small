"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using sparse LP-based Frank-Wolfe
with multi-length exploration, diverse initializations, adaptive constraint
selection, and multi-phase polishing (coordinate descent + subgradient).

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_sparse_lp(f, M, n, top_k=None, warm_start=None):
    """Sparse LP: max sum(g) s.t. (f*g)[k] <= M for top_k most violated k, g >= 0.
    Uses a two-pass approach: first find violations, then solve with only those."""
    if top_k is None:
        top_k = min(2 * n - 1, max(64, n))
    
    # First pass: compute f*g for a unit vector to find constraint structure
    # Actually, we need to find which constraints are most likely to be active.
    # Strategy: solve with all constraints first (if n is small enough),
    # otherwise use a heuristic subset.
    
    if n <= 512:
        # Full LP for small n
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        return None
    else:
        # Sparse: use top_k constraints centered around the peak region
        # The most important constraints are near the middle (lag ~ n-1)
        center = n - 1
        half_k = top_k // 2
        k_lo = max(0, center - half_k)
        k_hi = min(2 * n - 2, center + half_k)
        
        # Add some constraints near the edges too
        edge_k = max(0, (top_k - (k_hi - k_lo + 1)) // 4)
        k_list = list(range(k_lo, k_hi + 1))
        if edge_k > 0:
            k_list.extend(range(0, min(edge_k, n)))
            k_list.extend(range(max(0, 2*n-2-edge_k), 2*n-1))
        
        k_list = sorted(set(k_list))
        num_constraints = len(k_list)
        
        c = -np.ones(n)
        A_ub = np.zeros((num_constraints, n))
        b_ub = np.full(num_constraints, M)
        
        for idx, k in enumerate(k_list):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[idx, j] = f[k - j]
        
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        return None


def solve_lp_full(f, M, n):
    """Full LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    top_k = min(2 * n, max(128, 4 * n // 3))

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

            g = solve_sparse_lp(f, M * m_exp, n, top_k=top_k)

            g = solve_lp_full(f, M * m_exp, n)

            if m_exp > 5.0:

            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        best_val = M

                    hi = mid

        if best_alpha > 1e-10:

            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        travel = passes % 2 == 0

        indices = range(n - 1, -1, -1) if travel else range(n)
        
        for j in indices:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()

            multipliers = [0.3, 0.5, 0.7, 1.3, 1.6, 2.0, 3.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 4:
            break
        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=None, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    if top_k is None:
        top_k = max(20, n // 6)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.002

        if no_improve > 400:
            rng = np.random.default_rng(hash(int(time.monotonic() * 10000)) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def random_initialization(n, rng, strategy=0):
    """Generate diverse random initializations."""
    if strategy == 0:
        # Uniform
        return np.ones(n) / n
    elif strategy == 1:
        # Gaussian centered
        x = np.linspace(0, 1, n)
        sigma = 0.2 + rng.random() * 0.15
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif strategy == 2:
        # Exponential decay
        tau = n * (0.2 + rng.random() * 0.3)
        a = np.exp(-np.arange(n) / tau)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif strategy == 3:
        # Random uniform
        a = rng.random(n)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif strategy == 4:
        # Cosine window
        a = np.cos(np.pi * np.arange(n) / n) ** 2
        a = np.maximum(a, 0)
        return a / a.sum()
    elif strategy == 5:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.3) / 0.12) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.12) ** 2)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif strategy == 6:
        # Ramp + noise
        a = np.linspace(0.5, 1.5, n) + rng.standard_normal(n) * 0.1
        a = np.maximum(a, 0)
        return a / a.sum()
    else:
        # Smooth random (low-pass filtered random)
        a = rng.random(n)
        kernel = np.exp(-0.5 * (np.arange(-5, 6) / 3.0) ** 2)
        from scipy.signal import convolve
        a = convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining sparse LP-FW, coordinate descent,
    subgradient polishing, and multi-length exploration with diverse
    initializations for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 0: Quick exploration of different lengths and initializations ===
    print("=== Phase 0: Quick Exploration ===")
    explorations = []

    # Try parent at its length
    if parent is not None:
        explorations.append(('parent', parent))

    # Try different lengths with diverse initializations
    for n in [512, 768, 1024]:
        for strat in [0, 1, 3]:
            a_init = random_initialization(n, rng, strategy=strat)
            explorations.append((f'n{n}_s{strat}', a_init))

    # Quick LP-FW on each (short time)
    for label, a_init in explorations:
        rem = deadline - time.monotonic() - 1200
        if rem < 20:
            break
        t_alloc = min(rem * 0.3, 60)
        if t_alloc < 10:
            t_alloc = 10
        a_opt, _ = lp_fw_optimize(a_init, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"explore_{label}")

    print(f"  After Phase 0: best={best_score:.8f}")

    # === Phase 1: Deep LP-FW on best candidate ===
    print("\n=== Phase 1: Deep LP-FW ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem, 400)
            print(f"  LP-FW on best (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw_1")

        # Try with perturbed initialization
        rem = deadline - time.monotonic() - 500
        if rem > 20:
            noise = rng.standard_normal(n_best) * 0.005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem, 200), verbose=True)
            try_seq(a_opt, "deep_lpfw_perturb")

        # Try mirrored version
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            f_mirror = 0.5 * best_seq + 0.5 * best_seq[::-1]
            f_mirror = project_simplex(f_mirror)
            a_opt, _ = lp_fw_optimize(f_mirror, time_limit=min(rem, 150), verbose=False)
            try_seq(a_opt, "deep_lpfw_mirror")

    # === Phase 2: Try other lengths ===
    print(f"\n=== Phase 2: Other Lengths (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        other_lengths = [l for l in [512, 768, 1024, 1280, 1536] if l != n_best]
        
        for target_n in other_lengths[:3]:
            rem = deadline - time.monotonic() - 300
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.5, 150)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"other_n{target_n}")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_cd = min(rem * 0.7, 200)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd")

            # SG after CD
            rem = deadline - time.monotonic() - 100
            if rem > 15:
                a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 80),
                                     lr=0.002, mom=0.94,
                                     top_k=max(20, len(best_seq) // 5),
                                     temp=0.00003)
                try_seq(a_opt, "sg_after_cd")

    # === Phase 4: Final Polish ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine SG
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 60),
                                 lr=0.0008, mom=0.95,
                                 top_k=max(30, n_best // 4),
                                 temp=0.00001)
            try_seq(a_opt, "sg_fine")

        # One more LP-FW
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 40), verbose=False)
            try_seq(a_opt, "final_lpfw")

        # Tiny perturbations + quick CD
        for pert in range(4):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = coordinate_descent(f_pert, time_limit=min(rem, 20))
            try_seq(a_opt, f"perturb_cd_{pert}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
