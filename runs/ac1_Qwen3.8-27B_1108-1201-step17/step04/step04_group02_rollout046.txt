    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve_sweeps = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        sweep_best = conv.max()
        
        # Randomize order
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 0.3:
                break
            
            x0 = f[j]
            x_new = cd_step(f, conv, j, n)
            
            if abs(x_new - x0) < 1e-14:
                continue
            
            # Apply update
            f[j] = x_new
            s = f.sum()
            if s < 1e-10:
                f[j] = x0
                continue
            f = f / s
            conv = fast_convolve(f)
            
            if conv.max() < sweep_best:
                sweep_best = conv.max()
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_f = f.copy()
        
        if sweep_best < best_obj - 1e-14:
            best_obj = sweep_best
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter {iteration}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")
        
        if no_improve_sweeps > 30:
            # Perturb and restart
            noise = rng.standard_normal(n) * 0.005
            f = np.maximum(best_f + noise, 0)
            s = f.sum()
            if s > 1e-10:
                f = f / s
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                no_improve_sweeps = 0
            else:
                f = best_f.copy()
                conv = fast_convolve(f)
                no_improve_sweeps = 0
    
    return best_f, best_obj


def generate_initializations(n, seed=42):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine (Hanning)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(rc / rc.sum())
    
    # Flat-top window (approximation)
    # 0.2156 + 0.4160*cos + 0.2780*cos^2 + 0.0835*cos^3 + 0.0069*cos^4
    t = 2 * np.pi * x / n - np.pi
    ft = (0.2156 - 0.4160*np.cos(t) + 0.2780*np.cos(2*t) 
          - 0.0835*np.cos(3*t) + 0.0069*np.cos(4*t))
    ft = np.maximum(ft, 0)
    if ft.sum() > 1e-10:
        inits.append(ft / ft.sum())
    
    # Gaussian
    g = np.exp(-((x - n/2)**2) / (2*(n/4)**2))
    inits.append(g / g.sum())
    
    # Exponential decay
    e1 = np.exp(-2.0 * x / n)
    inits.append(e1 / e1.sum())
    e2 = np.exp(-2.0 * (n-1-x) / n)
    inits.append(e2 / e2.sum())
    
    # Linear
    lin_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(lin_d / lin_d.sum())
    lin_i = np.arange(1, n+1, dtype=np.float64)
    inits.append(lin_i / lin_i.sum())
    
    # Sine
    sn = np.sin(np.pi * (x + 0.5) / n)
    sn = np.maximum(sn, 0)
    if sn.sum() > 1e-10:
        inits.append(sn / sn.sum())
    
    # Power law
    pw = (x + 1)**0.3
    inits.append(pw / pw.sum())
    ipw = 1.0 / (x + 1)**0.3
    inits.append(ipw / ipw.sum())
    
    # Dirichlet
    d = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(d)
    
    # Sinc-like (windowed)
    w = np.sinc((x - n/2) / (n/4)) * np.exp(-((x-n/2)**2)/(2*(n/3)**2))
    w = np.maximum(w, 0)
    if w.sum() > 1e-10:
        inits.append(w / w.sum())
    
    # Triangular
    tri = 1.0 - 2.0 * abs(x - n/2) / n
    tri = np.maximum(tri, 0)
    inits.append(tri / tri.sum())
    
    return inits


def solve_lp_escape(f, M, n, time_limit=30):
    """Try to find a direction that reduces the max convolution below M using LP."""
    from scipy.optimize import linprog
    
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None
