"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP direction finding, smooth log-sum-exp gradient polishing,
multi-length exploration, and adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that (C @ g)[k] = sum_j a[k-j]*g[j] (full convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lse_grad(a, conv, tau):
    """Compute gradient of log-sum-exp of auto-conv values w.r.t. a."""
    n = len(a)
    # Softmax weights on the conv values
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # grad_j = 2 * sum_k w[k] * a[k-j] (where a is indexed 0..n-1)
    # For each k, the contribution to grad_j is w[k] * 2 * a[k-j] for valid j
    # grad = 2 * (w circular-shifted and cross-correlated with a)
    
    # More efficiently: grad[j] = 2 * sum_k w[k] * a[k-j] for k-j in [0, n-1]
    # = 2 * sum_i w[i+j] * a[i] for i in [0, n-1] and i+j in [0, 2n-2]
    
    # Let's compute this as a convolution
    # grad[j] = 2 * sum_i a[i] * w[i+j]
    # This is like a cross-correlation of a with w
    
    # Pad w to handle boundaries
    # w has length 2n-1
    # For j=0: grad[0] = 2 * sum_i a[i] * w[i] = 2 * dot(a[:n], w[:n])
    # For j=n-1: grad[j] = 2 * sum_i a[i] * w[i+n-1] = 2 * dot(a[:n], w[n-1:2n-1])
    
    # So grad[j] = 2 * dot(a, w[j:j+n])
    # This is a sliding dot product, which can be computed as a convolution
    
    # Actually, let's use FFT for efficiency
    # grad[j] = 2 * sum_i a[i] * w[j+i]
    # This is the cross-correlation of a and w at lag j
    
    # Using FFT: cross-correlation = ifft(fft(a) * conj(fft(w)))
    # But w has length 2n-1 and a has length n, so we need to be careful
    
    # Simpler approach: just loop (n iterations, each O(n)) = O(n^2)
    # For n=1024, this is about 1M operations, which is fast enough
    
    grad = np.zeros(n)
    for j in range(n):
        # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * w[j+i]
        valid_end = min(n, len(w) - j)
        if valid_end > 0:
            grad[j] = 2.0 * np.dot(a[:valid_end], w[j:j + valid_end])
    
    return grad


def lse_grad_fft(a, conv, tau):
    """Compute gradient of LSE of auto-conv using FFT for efficiency."""
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len + n - 1 - 1).bit_length()
    
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * w[j+i]
    # This is the cross-correlation of w and a at lag j
    # = ifft(conj(fft(a, N)) * fft(w, N))[j] for j = 0..n-1
    # But we need to be careful with the definition
    
    # Actually: sum_i a[i] * w[j+i] for i=0..n-1
    # Let's define this as (a ⋆ w)[j] where ⋆ is cross-correlation
    # (a ⋆ w)[j] = sum_i a[i] * conj(w[i+j])
    # Since all real: = sum_i a[i] * w[i+j]
    
    # Using FFT: ifft(fft(a) * conj(fft(w)))
    N = fft_size
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    cross = np.fft.ifft(A * np.conj(W)).real
    grad = 2.0 * cross[:n]
    
    return grad


def lp_step(a, M, max_iter=1):
    """One LP step: find direction g that maximizes sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and result.x is not None:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def lse_polish(a0, time_limit, tau_init=0.001, lr_init=0.003):

    a = np.maximum(a0.astype(np.float64), 0.0)

    n = len(a)

    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    momentum_v = np.zeros(n)
    momentum = 0.9
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute LSE gradient
        grad = lse_grad_fft(a, conv, tau)
        
        # Momentum step
        momentum_v = momentum * momentum_v - lr * grad
        a_new = a + momentum_v
        a_new = np.maximum(a_new, 0.0)

        else:

        cur_obj = 2.0 * n * conv.max()
        step += 1

            no_improve = 0

def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.astype(np.float64).copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq

"""
Multi-phase optimization minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines LP direction finding, smooth log-sum-exp gradient polishing,
multi-length exploration (512-1536), and adaptive perturbation strategies.
Key improvements: fixed FFT sizing, U/V-shaped initializations, tau-scheduled
LSE polishing, and alternating LP/LSE refinement.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that (C @ g)[k] = full_conv(a, g)[k]."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lse_grad_fft(a, conv, tau):
    """Gradient of LSE(conv, tau) w.r.t. a, computed via FFT cross-correlation."""
    n = len(a)
    conv_len = 2 * n - 1
    # Need FFT size >= n + conv_len - 1 = 3n - 2
    fft_size = 1 << (3 * n - 3).bit_length()

    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # grad[j] = 2 * sum_m a[m] * w[j+m] = 2 * (a cross-corr w)[j]
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    cross = np.fft.ifft(A * np.conj(W)).real
    grad = 2.0 * cross[:n]
    return grad

    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def line_search(a, d, best_obj, n, max_candidates=12):
    """Coarse + fine line search along direction d."""
    best_t = 0.0
    best_o = best_obj

    # Coarse
    candidates = np.linspace(0.05, 1.0, max_candidates)
    for t in candidates:
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        o_try = 2.0 * n * conv_try.max()
        if o_try < best_o - 1e-14:
            best_o = o_try
            best_t = t

    # Fine (golden section)
    if 0.05 < best_t < 0.95:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0.01, best_t - 0.06)
        hi = min(0.99, best_t + 0.06)
        d1 = lo + (1.0 - gr) * (hi - lo)
        d2 = lo + gr * (hi - lo)
        for _ in range(5):
            a1 = np.maximum(a + d1 * d, 0.0)
            s1 = a1.sum()
            o1 = 2.0 * n * fast_convolve(a1 / s1).max() if s1 > 1e-15 else best_o
            a2 = np.maximum(a + d2 * d, 0.0)
            s2 = a2.sum()
            o2 = 2.0 * n * fast_convolve(a2 / s2).max() if s2 > 1e-15 else best_o
            if o1 < o2:
                hi, d2, o2 = d2, d1, o1
                d1 = lo + (1.0 - gr) * (hi - lo)
            else:
                lo, d1, o1 = d1, d2, o2
                d2 = lo + gr * (hi - lo)
            if hi - lo < 0.003:
                break
        mid = (lo + hi) / 2.0
        a_mid = np.maximum(a + mid * d, 0.0)
        s_mid = a_mid.sum()
        if s_mid > 1e-15:
            o_mid = 2.0 * n * fast_convolve(a_mid / s_mid).max()
            if o_mid < best_o:
                best_o = o_mid
                best_t = mid

    return best_t, best_o


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with adaptive tightness."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0
    tightness = 0.9995

    while time.monotonic() - t0 < time_limit:
        g, g_sum = lp_step(a, M * tightness)

        if g_sum > 1.0 + 1e-6:
            d = g - a
            if np.linalg.norm(d) > 1e-12:
                t_best, obj_best = line_search(a, d, best_obj, n)
                if t_best > 1e-4 and obj_best < best_obj - 1e-14:
                    a_new = np.maximum(a + t_best * d, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    conv = fast_convolve(a)
                    M = conv.max()
                    cur = 2.0 * n * M
                    if cur < best_obj:
                        best_obj = cur
                        best_a = a.copy()
                    no_improve = 0
                    tightness = max(0.9985, tightness - 0.00015)
                    if verbose and iteration % 30 == 0:
                        print(f"    LP it={iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.0f}s")
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            tightness = min(1.002, tightness * 1.004)
            no_improve += 1

        iteration += 1

        if no_improve >= 80:
            rng = np.random.default_rng(iteration * 13 + perturb_count * 7 + 1)
            scale = 0.002 * (1.0 + 0.3 * min(perturb_count, 6))
            # Mix random and structured perturbation
            if perturb_count % 3 == 0:
                noise = rng.standard_normal(n) * scale
            elif perturb_count % 3 == 1:
                freq = rng.integers(1, 6)
                noise = np.sin(2 * np.pi * freq * np.arange(n) / n) * scale + \
                        rng.standard_normal(n) * scale * 0.3
            else:
                edge = np.exp(-0.5 * (np.arange(n) / (0.25 * n)) ** 2) + \
                       np.exp(-0.5 * ((np.arange(n) - (n - 1)) / (0.25 * n)) ** 2)
                noise = edge * rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 1e-15 else best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            tightness = 0.9995
            if perturb_count > 25:
                break

    return best_a, best_obj


def lse_polish(a0, time_limit, tau_init=0.001, lr_init=0.003):
    """Smooth LSE gradient polishing with adaptive tau and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    v = np.zeros(n)
    momentum = 0.92
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        grad = lse_grad_fft(a, conv, tau)
        v = momentum * v - lr * grad
        a_new = np.maximum(a + v, 0.0)
        s_new = a_new.sum()
        a = a_new / s_new if s_new > 1e-15 else a
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        step += 1

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 80 == 0 and no_improve > 0:
            lr *= 0.996
            tau = max(0.00005, tau * 0.97)

        if no_improve >= 350:
            rng = np.random.default_rng(step * 3 + 11)
            noise = rng.standard_normal(n) * 0.0015
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 1e-15 else best_a.copy()
            v = np.zeros(n)
            conv = fast_convolve(a)
            no_improve = 0

    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return np.asarray(seq, dtype=np.float64).copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    return new_seq / s if s > 1e-15 else np.ones(new_n) / new_n


def get_inits(n, parent_seq, rng_local):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n

    if parent_seq is not None:
        inits['parent'] = resample_sequence(parent_seq, n)

    # U-shaped (high at edges)
    u = 0.4 + 0.6 * (np.cos(np.pi * x / max(n - 1, 1)) ** 2)
    inits['ushape'] = u / u.sum()

    # V-shaped (low in middle)
    v = 1.0 - 0.6 * np.sin(np.pi * x / max(n - 1, 1))
    inits['vshape'] = v / v.sum()

    # Two-lobe cosine
    lobe = 0.5 + 0.5 * np.cos(4.0 * np.pi * x / n)
    inits['tlobe'] = lobe / lobe.sum()

    # Four-lobe
    lobe4 = 0.6 + 0.4 * np.cos(8.0 * np.pi * x / n)
    inits['flobe'] = lobe4 / lobe4.sum()

    # Edge-peaked Gaussian
    edge = np.exp(-0.5 * (x / (0.15 * n)) ** 2) + np.exp(-0.5 * ((x - (n - 1)) / (0.15 * n)) ** 2)
    inits['edge'] = edge / edge.sum()

    # Center Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Random Dirichlet
    inits['random'] = rng_local.dirichlet(np.ones(n) * 0.3)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP scout -> Deep LP -> LSE polish -> Final refine.
    Explores lengths 512/768/1024/1536 with diverse inits and alternating methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(p)}, obj={2*len(p)*fast_convolve(parent).max():.6f}")
    except Exception:
        pass

    # Phase 1: Multi-length LP Scout
    print("=== Phase 1: Multi-length LP Scout ===")
    t_start = time.monotonic()
    p1_deadline = t_start + 280.0

    lengths = [512, 768, 1024, 1536]
    scan_results = []
    priority = ['parent', 'uniform', 'ushape', 'vshape', 'tlobe', 'edge', 'gauss', 'random']

    for n in lengths:
        if time.monotonic() > p1_deadline - 25:
            break
        inits = get_inits(n, parent, rng)
        init_names = [nm for nm in priority if nm in inits][:4]
        t_per = max(12.0, 240.0 / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > p1_deadline - 8:
                break
            t_alloc = min(t_per, p1_deadline - time.monotonic() - 4)
            if t_alloc < 6:
                break
            a0 = inits[name]
            a_opt, _ = lp_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: Deep LP
    print("=== Phase 2: Deep LP ===")
    t_start2 = time.monotonic()
    top_configs = []
    seen = set()
    for sc, n, nm, ao in scan_results:
        if n not in seen:
            top_configs.append((n, ao.copy()))
            seen.add(n)
        if len(top_configs) >= 3:
            break
    if best_sequence is not None and len(best_sequence) not in seen:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    rem_p2 = deadline - time.monotonic() - 420.0
    rem_p2 = max(60.0, rem_p2)
    t_per_deep = max(25.0, rem_p2 / max(1, len(top_configs)))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 420:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 390)
        if t_alloc < 10:
            break
        a_opt, _ = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: LSE + LP alternation
    print("=== Phase 3: LSE + LP Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        rem = deadline - time.monotonic() - 180.0
        if rem > 40:
            configs = [
                {'tau_init': 0.0008, 'lr_init': 0.0025},
                {'tau_init': 0.0015, 'lr_init': 0.004},
                {'tau_init': 0.0004, 'lr_init': 0.0015},
            ]
            t_per = rem / (len(configs) + 2.5)

            for ci, cfg in enumerate(configs):
                if time.monotonic() > deadline - 140:
                    break
                t_alloc = min(t_per, deadline - time.monotonic() - 120)
                if t_alloc < 8:
                    break
                a_opt, _ = lse_polish(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LSE {ci}: {score:.6f}")

            # LP refinement after LSE
            if time.monotonic() < deadline - 70:
                t_alloc = min(45.0, deadline - time.monotonic() - 50)
                if t_alloc > 12:
                    a_opt, _ = lp_optimize(best_sequence, time_limit=t_alloc)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  LP-after-LSE: {score:.6f}")

                    # Another LSE pass
                    if time.monotonic() < deadline - 30:
                        t_alloc2 = min(25.0, deadline - time.monotonic() - 15)
                        if t_alloc2 > 8:
                            a_opt, _ = lse_polish(best_sequence, time_limit=t_alloc2,
                                                   tau_init=0.0005, lr_init=0.001)
                            score = float(evaluate_sequence(a_opt.tolist()))
                            if np.isfinite(score) and score < best_score:
                                best_score = score
                                best_sequence = a_opt.copy()
                                print(f"  LSE-final: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Final perturb + refine
    print("=== Phase 4: Final Refinement ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        rem = deadline - time.monotonic() - 10.0
        if rem > 15:
            nb = len(best_sequence)
            n_pert = max(1, int(rem / 22))
            t_per = rem / n_pert

            for pi in range(n_pert):
                if time.monotonic() > deadline - 8:
                    break
                if t_per < 5:
                    break

                scale = 0.0012 * (1.0 + 0.25 * pi)
                mode = pi % 4
                if mode == 0:
                    noise = rng.standard_normal(nb) * scale
                elif mode == 1:
                    f = rng.integers(2, 7)
                    noise = np.sin(2 * np.pi * f * np.arange(nb) / nb + rng.uniform(0, 2 * np.pi)) * scale
                elif mode == 2:
                    edge = np.exp(-((np.arange(nb) - nb / 2) / (0.3 * nb)) ** 2)
                    noise = edge * rng.standard_normal(nb) * scale
                else:
                    noise = rng.standard_normal(nb) * scale * 0.5 + \
                            np.cos(4 * np.pi * np.arange(nb) / nb) * scale * 0.5

                a_pert = np.maximum(best_sequence + noise, 0.0)
                s = a_pert.sum()
                a_pert = a_pert / s if s > 1e-15 else best_sequence.copy()

                t_alloc = min(t_per, deadline - time.monotonic() - 4)
                if t_alloc < 4:
                    break

                if pi % 2 == 0:
                    a_opt, _ = lp_optimize(a_pert, time_limit=t_alloc)
                else:
                    a_opt, _ = lse_polish(a_pert, time_limit=t_alloc,
                                          tau_init=0.001, lr_init=0.002)

                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final {pi}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
