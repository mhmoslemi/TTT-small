Optimizer for minimizing 2*n*max(autoconv(a))/(sum(a))^2.
Uses smoothed subgradient descent, LP Frank-Wolfe, and coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_smooth(a, conv, T):
    """Compute gradient of smooth max of conv w.r.t. a."""
    n = len(a)
    shifted = conv - conv.max()
    exp_T = np.exp(T * shifted)
    w = exp_T / exp_T.sum()
    # grad[i] = 2 * T * sum_j w[i+j] * a[j] = 2*T * correlate(w, a, 'valid')[i]
    grad = 2.0 * T * np.correlate(w, a, mode='valid')
    return grad

def smooth_subgrad(a0, time_limit, T_init=100, T_growth=1.01,

        a = np.ones(n) / n
    else:
        a = a / s

def lp_frank_wolfe(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    a = a0.copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    # Precompute the convolution matrix B where (B@d)[k] = (a*d)[k]
    # B[k, j] = a[k-j] if 0 <= k-j < n, else 0
    B = np.zeros((2*n-1, n))
    for j in range(n):
        B[j:j+n, j] = a
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Update B if a changed significantly (actually B depends on a, so we need to rebuild each time)
        # For efficiency, rebuild B each iteration
        B = np.zeros((2*n-1, n))
        for j in range(n):
            B[j:j+n, j] = a
        
        # LP: min t s.t. t >= conv[k] + 2*(B@d)[k] for all k, d >= 0, sum(d) = 1
        # Variables: [t, d_0, ..., d_{n-1}]
        c = np.zeros(n + 1)
        c[0] = 1.0
        
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, 0] = -1.0
        A_ub[:, 1:] = 2.0 * B
        b_ub = -conv
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, 1:] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(None, None)] + [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if not res.success:
            no_improve += 5
            if no_improve > 50:
                break
            continue
        
        d = res.x[1:]
        
        # Check if d is close to a (fixed point)

            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search: find best alpha in (0, 1] for a_new = (1-alpha)*a + alpha*d
        best_alpha = 0
        best_val = M
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

            a_try = np.maximum(a_try, 0)

            if st < 1e-15:

            val = fast_convolve(a_try).max()
            if val < best_val:

                mid = (lo + hi) / 2
                a_try = (1 - mid) * a + mid * d
                a_try = np.maximum(a_try, 0)

                if val < best_val:

                else:
                    hi = mid

                no_improve = 0
            else:
                no_improve += 1

            no_improve += 1
