    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses top-k log-sum-exp with FFT-based gradients, adaptive parameters,
momentum with restarts, multi-start initialization, and length exploration.
Improved over prior SOTA by focusing gradient on critical entries and using
backtracking-style adaptive learning rates.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_topk_lse_grad(f, beta, top_k):
    """Compute gradient of top-k LSE of conv(f,f) using FFT.
    Returns (gradient, lse_value, conv).
    """
    n = len(f)
    conv = fast_convolve(f)
    conv_len = len(conv)

    if top_k >= conv_len:
        top_idx = np.arange(conv_len)
        top_vals = conv
    else:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]

    scaled = beta * top_vals
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w_top = exp_vals / exp_vals.sum()

    w_full = np.zeros(conv_len)
    w_full[top_idx] = w_top

    # FFT cross-correlation: grad = 2 * conv(w_full, f_flipped)[n-1 : 2n-1]
    f_flipped = f[::-1].copy()
    total_len = conv_len + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_wf = np.fft.irfft(W * F, fft_size)
    grad = 2.0 * conv_wf[n - 1:n - 1 + n]

    lse_val = top_vals.max() + np.log(exp_vals.sum()) / beta

    return grad, lse_val, conv

    s = f.sum()

        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    if lr_init is None:
        lr = 1.5 / n
    else:
        lr = lr_init
    momentum = 0.87
    v = np.zeros(n)
    beta = beta_start

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:

def build_lp_matrix(f):
    """Build sparse constraint matrix for FW-LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_step(f, M, A_cache=None):
    """Single FW-LP step: solve LP for direction, line search, return new f."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if not res.success:
            return None, None
        g = np.maximum(res.x, 0.0)
    except Exception:
        return None, None

    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, None
    g = g / g_sum

    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha < 1e-10:
        return None, None

    f_new = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
    f_new = f_new / f_new.sum()
    return f_new, best_val


def fw_lp_optimize(f0, time_limit, max_iters=200):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A_cache = build_lp_matrix(f)
        f_new, val = fw_lp_step(f, M * m_exp, A_cache)

        if f_new is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            continue

        M = fast_convolve(f_new).max()
        f = f_new
        m_exp = 1.0

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(len(f)) * 0.003)
            M = fast_convolve(f).max()
            no_improve = 0
            m_exp = 1.0

    return best_f, best_obj


def local_search_refine(f0, time_limit, n_trials=50):
    """Local search: try random coordinate perturbations and keep improvements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    scale = 0.01

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        # Perturb a random subset of coordinates
        n_perturb = max(1, n // 20)
        idx = rng.choice(n, size=n_perturb, replace=False)
        f_try = best_f.copy()
        f_try[idx] += rng.standard_normal(n_perturb) * scale
        f_try = np.maximum(f_try, 0.0)
        f_try = f_try / f_try.sum()

        obj = fast_convolve(f_try).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_try
            scale = min(scale * 1.05, 0.05)
        else:
            scale = max(scale * 0.95, 0.001)

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score
