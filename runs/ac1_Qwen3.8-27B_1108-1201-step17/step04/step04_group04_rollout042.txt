    for k in range(len(full_w)):
        if full_w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idxs = np.arange(js, je + 1)
        grad[js:je + 1] += 2.0 * full_w[k] * a[ki_minus_idx]
    return grad


def lp_solve_min_t(a, S, C=None, timeout=15):
    n = len(a)
    if C is None:
        C = np.zeros((2 * n - 1, n))
        for j in range(n):
            C[j:j + n, j] = a
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            t_val = res.x[n]
            return g, t_val, True
    except Exception:
        pass
    return None, None, False


def lp_solve_max_sum(a, M, C=None, timeout=15):
    n = len(a)
    if C is None:
        C = np.zeros((2 * n - 1, n))
        for j in range(n):
            C[j:j + n, j] = a
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            return g, g.sum(), True
    except Exception:
        pass
    return None, None, False


def lp_step(a, conv, n, C=None, use_min_t=True):
    M = conv.max()
    S = a.sum()
    if use_min_t:
        g, t_val, ok = lp_solve_min_t(a, S, C)
        if not ok or g is None:
            return None, False
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None, False
        g = g / g_sum * S
    else:
        g, g_sum, ok = lp_solve_max_sum(a, M * 0.999, C)
        if not ok or g is None:
            return None, False
        if g_sum < S * 0.5:
            return None, False
        g = g / g_sum * S
    d = g - a
    nd = np.linalg.norm(d)
    if nd < 1e-12:
        return None, False
    best_a = a.copy()
    best_obj = obj_value(a)
    for alpha in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
        a_try = a + alpha * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s * S
        o = obj_value(a_try)
        if o < best_obj - 1e-12:
            best_obj = o
            best_a = a_try.copy()
            break
    if best_obj < obj_value(a) - 1e-12:
        return best_a, True
    return None, False


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.92,
                      top_k=40, temp_init=0.003, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        conv_y = fast_convolve(y)
        full_w = softmax_weights(conv_y, top_k=top_k, temp=temp)
        grad = gradient_fft(y, full_w)
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-15:
            a = a_new / sn
        else:
            a = best_a.copy()
            v = np.zeros(n)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        lr *= 0.9997
        temp *= 1.004
        mom = min(0.97, mom + 0.0001)
        it += 1
        if verbose and it % 500 == 0:
            print(f"      nesterov it={it} obj={best_obj:.8f} t={time.monotonic()-t0:.1f}s")
        if no_improve >= 300:
            break
    return best_a, best_obj


def simulated_annealing(a0, time_limit, rng, T_init=0.002, T_min=1e-6,
                        cool_rate=0.9995, n_perturb=5, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, obj_value(a)
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit and T > T_min:
        npert = n_perturb
        a_try = a.copy()
        for _ in range(npert):
            i = rng.integers(0, n)
            delta = rng.standard_normal() * 0.01
            a_try[i] += delta
        a_try = np.maximum(a_try, 0.0)
        st = a_try.sum()
        if st < 1e-15:
            T *= cool_rate
            continue
        a_try = a_try / st
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        delta_obj = obj_try - best_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            a = a_try
            conv = conv_try
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        it += 1
        if verbose and it % 2000 == 0:
            print(f"      SA it={it} T={T:.6f} best={best_obj:.8f} t={time.monotonic()-t0:.1f}s")
        if no_improve > 5000:
            break
    return best_a, best_obj


def informed_sa(a0, time_limit, rng, T_init=0.001, T_min=1e-7,
                cool_rate=0.9998, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, obj_value(a)
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit and T > T_min:
        peak_idx = int(np.argmax(conv))
        a_try = a.copy()
        for _ in range(3):
            j = rng.integers(0, n)
            k_val = peak_idx - j
            if 0 <= k_val < n:
                partner = a_try[k_val]
            else:
                partner = 0.0
            if partner > 0.01 * a_try.max():
                a_try[j] -= rng.random() * 0.005
            else:
                a_try[j] += rng.random() * 0.005
        a_try = np.maximum(a_try, 0.0)
        st = a_try.sum()
        if st < 1e-15:
            T *= cool_rate
            continue
        a_try = a_try / st
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        delta_obj = obj_try - best_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            a = a_try
            conv = conv_try
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        it += 1
        if verbose and it % 2000 == 0:
            print(f"      iSA it={it} T={T:.6f} best={best_obj:.8f} t={time.monotonic()-t0:.1f}s")
        if no_improve > 8000:
            break
    return best_a, best_obj


def lp_optimize_loop(a0, time_limit, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    S = a.sum()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    use_min_t = True
    while time.monotonic() - t0 < time_limit:
        C = None
        a_new, improved = lp_step(a, conv, n, C, use_min_t=use_min_t)
        if improved and a_new is not None:
            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 30 + perturb_count * 5:
            rng = np.random.default_rng(hash((n, perturb_count)) % (2**31))
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            use_min_t = not use_min_t
            if perturb_count > 30:
                break
        if verbose and no_improve % 10 == 0:
            print(f"      LP n={n} best={best_obj:.8f} perturb={perturb_count} t={time.monotonic()-t0:.1f}s")
    return best_a, best_obj


def coordinate_descent(a0, time_limit, rng):
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, obj_value(a)
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        batch = max(1, int(200000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.3:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.002, 0.005, 0.01, -0.002, -0.005, -0.01]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                st = a_try.sum()
                if st < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
            else:
                no_improve += 1
        if no_improve > 500:
            break
    return best_a, best_obj


def extend_sequence(a_short, n_target):
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    x_old = np.linspace(0, 1, n_short)
    x_new = np.linspace(0, 1, n_target)
    a_ext = np.interp(x_new, x_old, a_norm)
    a_ext = np.maximum(a_ext, 0)
    se = a_ext.sum()
    if se > 1e-15:
        a_ext = a_ext / se
    return a_ext


def generate_inits(n, parent, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['sine'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine'] /= inits['sine'].sum()
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits['alt'] = alt / alt.sum()
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = extend_sequence(p, n)
                inits['parent'] = pp
    return inits

                                           top_k=100, temp_init=0.0005)
            try_update(a_opt, "FinalNest")
    print(f"Phase 6 done ({time.monotonic()-t_s6:.0f}s). Best: {best_score:.8f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
