"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent with
analytical 1D subproblem solutions, diversified multi-length search, and
gradient-based polishing. Key innovation: each coordinate update exploits the
structure that M(ai) = max(linear terms + one quadratic) to find the exact

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_from_conv(conv, n, total_sum):
    """Compute objective from convolution and sum."""
    if total_sum < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (total_sum * total_sum)

def coordinate_descent_pass(a, conv, n, order):

    total_sum = a.sum()
    improved = False
    for i in order:

        best_ai, _ = optimize_single_coord(a, i, conv, n, total_sum)
        if abs(best_ai - old_val) > 1e-10:
            a[i] = best_ai
            total_sum = total_sum - old_val + best_ai
            # Update convolution incrementally (approximate: recompute affected parts)
            # For efficiency, we'll recompute conv at the end of the pass
            improved = True
    return a, improved

    no_improve = 0
    pass_count = 0
    order = list(range(n))

    while time.monotonic() - t0 < time_limit and pass_count < max_passes:
        # Shuffle order every few passes for better mixing
        if pass_count % 5 == 0:
            rng.shuffle(order)

        old_sum = a.sum()
        a, improved = coordinate_descent_pass(a, conv, n, order)

        # Recompute convolution periodically (every pass for accuracy)
        if pass_count % 3 == 0 or improved:
            conv = fast_convolve(a)
            total = a.sum()
            if total > 1e-15:
                cur_obj = 2.0 * n * conv.max() / (total * total)
            else:
                cur_obj = float('inf')

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and pass_count % 50 == 0:
                    print(f"      CD pass {pass_count}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        pass_count += 1

        # Restart if stuck
        if no_improve > 50:
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(best_a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            no_improve = 0

        # Check time more frequently
        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_obj


def gradient_polish(a0, time_limit, lr_init=0.01, momentum=0.9, verbose=False):
    """Projected gradient descent with momentum on the normalized problem."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    a = a / a.sum()  # normalize to sum=1
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        k_star = np.argmax(conv)

        # Subgradient of max(conv) w.r.t. a
        # d/d a[i] of conv[k_star] = a[k_star - i] + a[i] (if indices valid)
        # More precisely: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i... )
        # Actually: d/d(a[i]) sum_j a[j]*a[k-j]
        # = a[k-i] (from j=i term: d(a[i]*a[k-i])/d(a[i]) = a[k-i])
        # + a[i] (from j=k-i term: d(a[k-i]*a[i])/d(a[i]) = a[k-i]... wait)
        # Let me be careful. conv[k] = sum_{j} a[j]*a[k-j]
        # d(conv[k])/d(a[i]) = sum_j [d(a[j])/d(a[i])*a[k-j] + a[j]*d(a[k-j])/d(a[i])]
        # = a[k-i] (when j=i, if 0<=k-i<n) + a[k-i] (when k-j=i, i.e., j=k-i, if 0<=k-i<n and k-i != i)
        # If k-i == i (k=2i): just a[i] (single term a[i]*a[i], derivative = 2*a[i])
        # If k-i != i: 2*a[k-i]

        sg = np.zeros(n)
        j = k_star - i if (k_star - i) < len(a) else -1  # dummy
        # For index i in a:
        # contribution to conv[k_star] from a[i]:
        # if 0 <= k_star - i < n: a[k_star - i]
        # if 0 <= i - k_star < n... no, we need k_star - j = i, so j = k_star - i
        # So the gradient w.r.t. a[i] is:
        # a[k_star - i] if 0 <= k_star - i < n (from j=i term)
        # + a[k_star - i] if 0 <= k_star - i < n and k_star - i != i (from j=k_star-i term)
        # = 2*a[k_star-i] if k_star-i != i and 0 <= k_star-i < n
        # = a[i] (i.e., 2*(a[i]/2)... no, = 2*a[i] if k_star = 2*i)

        # Simpler: gradient of conv[k] w.r.t. a[i] is:
        # a[k-i] if 0 <= k-i < n, plus a[i-k] if 0 <= i-k < n (which is a[k-i] reversed... no)
        # Actually conv[k] = sum_j a[j]*a[k-j]. The derivative w.r.t. a[i]:
        # Term where j=i: a[i]*a[k-i], derivative = a[k-i] (if 0<=k-i<n)
        # Term where k-j=i, i.e., j=k-i: a[k-i]*a[i], derivative = a[k-i] (if 0<=k-i<n and k-i!=i)
        # If k-i=i: only one term a[i]^2, derivative = 2*a[i]

        j_val = k_star - 0  # not useful, let me just do it directly
        # For each i, the gradient of conv[k_star] w.r.t. a[i]:
        # = a[k_star - i] if 0 <= k_star - i < n
        #   + a[k_star - i] if 0 <= k_star - i < n and k_star - i != i
        # Simplified: 2*a[k_star-i] if 0<=k_star-i<n and k_star-i != i
        #              2*a[i] if k_star == 2*i
        #              0 otherwise (but there's always at least one valid)

        # Vectorized:
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        sg[valid] = a[idx[valid]]
        # Add the second contribution (from j = k_star - i)
        idx2 = np.arange(n) - k_star  # this is i - k_star, we need k_star - i... 
        # Wait, the second term is: when j = k_star - i, the contribution is a[j]*a[i] = a[k_star-i]*a[i]
        # d/d(a[i]) of this = a[k_star-i]
        # This is the same as the first term! So total gradient = 2*a[k_star-i] (when k_star-i != i)
        # or 2*a[i] when k_star = 2*i.
        # Which is exactly 2*a[k_star-i] in all cases (since when k_star=2i, k_star-i=i).
        sg[valid] *= 2.0

        # But we need to be careful: the "max" in the objective means we should use
        # a subgradient. For the max of conv[k], any k that achieves the max gives a valid
        # subgradient. Using random selection among max-achieving k's helps escape cycles.

        # Objective on simplex: 2n * max(conv) (with sum(a)=1)
        # Gradient of 2n*max(conv) w.r.t. a is 2n * sg (where sg is gradient w.r.t. the active k)
        # But we're on the simplex, so we project.

        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new  # still move, but track best separately

        lr *= 0.9995

        if no_improve > 100:
            rng_local = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            a = project_simplex(best_a + rng_local.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    # Convert back: objective was 2n*max(conv) with sum=1
    # Original objective: 2n*max(conv)/(sum)^2 - same thing since we normalized
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c1 / c1.sum())

    # U-shape
    edge_w = 1.0 / (0.5 * np.abs(x - 0.5 * (n - 1)) / n + 0.05)
    inits.append(edge_w / edge_w.sum())

    # Bimodal
    xm = n // 2
    bim = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(bim / bim.sum())

    # Power law
    pl = (x + 1) ** 0.5 * (n - x + 1) ** 0.5
    inits.append(pl / pl.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase coordinate descent with diversification.
    Phase 1: Quick CD on parent + several initializations
    Phase 2: Extended CD on best candidates with restarts
    Phase 3: Gradient polishing
    Phase 4: Final fine CD
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-15:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
    except Exception:
        pass

    n_p = len(parent) if parent is not None else 384

    # === Phase 1: Quick exploration ===
    print("=== Phase 1: Quick Exploration ===")
    t_phase1 = time.monotonic()
    phase1_budget = 120

    # Try parent first with quick CD
    if parent is not None:
        a_opt, _ = full_coordinate_descent(parent, time_limit=30, rng=rng)
        try_seq(a_opt, "parent_cd")

    # Try several different initializations at parent length
    inits = generate_inits(n_p, parent, rng)
    for idx, ini in enumerate(inits[:6]):
        if time.monotonic() - t_phase1 > phase1_budget - 30:
            break
        a_opt, _ = full_coordinate_descent(ini, time_limit=10, rng=rng)
        try_seq(a_opt, f"init_{idx}")

    # Try a few different lengths
    if best_seq is None:
        best_seq = np.ones(n_p) / n_p
        try_seq(best_seq, "fallback")

    for n_try in [n_p - 32, n_p + 32, n_p + 64]:
        if n_try < 64 or time.monotonic() - t_phase1 > phase1_budget - 20:
            continue
        ini = generate_inits(n_try, parent, rng)[0]
        a_opt, _ = full_coordinate_descent(ini, time_limit=8, rng=rng)
        try_seq(a_opt, f"len{n_try}_quick")

    print(f"  Phase 1 best: {best_score:.10f} (n={len(best_seq)})")

    # === Phase 2: Extended CD on best + diversified attempts ===
    print("\n=== Phase 2: Extended Coordinate Descent ===")
    t_phase2 = time.monotonic()
    phase2_budget = 900

    n_best = len(best_seq)

    # Long CD on the current best
    remaining = deadline - time.monotonic() - 300
    t_cd_main = min(remaining * 0.5, phase2_budget * 0.4)
    print(f"  Main CD (n={n_best}), budget={t_cd_main:.0f}s")
    a_opt, _ = full_coordinate_descent(best_seq, time_limit=t_cd_main, rng=rng, verbose=True)
    try_seq(a_opt, "cd_main")

    # CD on perturbed versions
    remaining = deadline - time.monotonic() - 250
    num_perturbs = 4
    t_each = min(remaining / num_perturbs, 100)
    for p in range(num_perturbs):
        if time.monotonic() > deadline - 250:
            break
        noise_scale = 0.005 * (p + 1)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
        a_opt, _ = full_coordinate_descent(f_pert, time_limit=t_each * 0.8, rng=rng)
        try_seq(a_opt, f"cd_pert_{p}")

    # Try different lengths with longer CD
    remaining = deadline - time.monotonic() - 200
    for n_try in [n_best - 16, n_best + 16, n_best + 48, n_best - 48]:
        if n_try < 64 or remaining < 60:
            continue
        px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
        px = np.maximum(px, 0)
        px = px / px.sum()
        t_this = min(remaining * 0.25, 80)
        a_opt, _ = full_coordinate_descent(px, time_limit=t_this, rng=rng)
        try_seq(a_opt, f"cd_len{n_try}")
        remaining = deadline - time.monotonic() - 200

    print(f"  Phase 2 best: {best_score:.10f} (n={len(best_seq)})")

    # === Phase 3: Gradient polishing ===
    print("\n=== Phase 3: Gradient Polish ===")
    t_phase3 = time.monotonic()
    remaining = deadline - time.monotonic() - 100

    if remaining > 60:
        n_best = len(best_seq)
        t_grad = min(remaining * 0.4, 120)
        print(f"  Gradient polish (n={n_best}), budget={t_grad:.0f}s")
        a_opt, _ = gradient_polish(best_seq, time_limit=t_grad)
        try_seq(a_opt, "grad1")

        # Polish the result with CD
        remaining = deadline - time.monotonic() - 80
        if remaining > 30:
            a_opt2, _ = full_coordinate_descent(a_opt, time_limit=min(remaining * 0.5, 60), rng=rng)
            try_seq(a_opt2, "grad_cd")

    print(f"  Phase 3 best: {best_score:.10f} (n={len(best_seq)})")

    # === Phase 4: Final fine-tuning ===
    print("\n=== Phase 4: Final Fine-tuning ===")
    remaining = deadline - time.monotonic() - 30

    if remaining > 30 and best_seq is not None:
        n_best = len(best_seq)
        # Very fine CD with small restarts
        t_fine = min(remaining * 0.6, 120)
        a_opt, _ = full_coordinate_descent(best_seq, time_limit=t_fine, rng=rng)
        try_seq(a_opt, "final_cd")

        # A few more perturbation attempts
        remaining = deadline - time.monotonic() - 20
        for attempt in range(3):
            if remaining < 20:
                break
            noise = rng.standard_normal(n_best) * 0.001 * (attempt + 1)
            f_pert = project_simplex(best_seq + noise)
            t_this = min(remaining * 0.3, 30)
            a_opt, _ = full_coordinate_descent(f_pert, time_limit=t_this, rng=rng)
            try_seq(a_opt, f"final_pert_{attempt}")
            remaining = deadline - time.monotonic() - 15

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
