"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FW-LP for coarse structure optimization
2. LSE-Adam gradient descent with warm beta schedule for fine polishing
3. Very high beta LSE-Adam for final convergence

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient: 2 * cross_correlation(weights, f) of length n."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend((f[ks] for ks in (k - js)))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, lp_timeout=25):

    f = np.maximum(f0.astype(np.float64), 0.0)

            conv = fast_convolve(f)

def lse_adam_optimize(f0, time_limit, beta_start=50.0, beta_end=50000.0,
                      lr_init=None, beta1=0.9, beta2=0.999, eps=1e-8):
    """LSE smoothed optimization with Adam and backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 3.0 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Softmax weights (numerically stable)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_sum = exp_bc.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(bc)] = 1.0
        else:
            w = exp_bc / w_sum

        # Gradient via FFT
        grad = compute_lse_grad(w, f)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Backtracking line search
        alpha = lr
        f_curr = f
        obj_curr = best_obj
        found = False
        for _ in range(20):
            f_trial = project_simplex(f_curr + alpha * m_hat / (np.sqrt(v_hat) + eps))
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            if obj_trial < obj_curr - 1e-16:
                found = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break

        if found:
            f = f_trial
            no_improve = 0
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_f = f.copy()
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr = max(lr * 0.85, lr_init * 0.01)
            if no_improve > 500:
                # Restart from best with fresh momentum
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0

        # Ramp up beta
        if beta < beta_end:
            beta *= 1.004

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam t={t}, obj={best_obj:.12f}, beta={beta:.1f}, "
                  f"lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj

        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 1.0 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_sum = exp_bc.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(bc)] = 1.0
        else:
            w = exp_bc / w_sum

        grad = compute_lse_grad(w, f)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # More aggressive backtracking
        alpha = lr
        f_curr = f
        obj_curr = best_obj
        found = False
        for _ in range(30):
            f_trial = project_simplex(f_curr + alpha * m_hat / (np.sqrt(v_hat) + eps))
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            if obj_trial < obj_curr - 1e-16:
                found = True
                break
            alpha *= 0.4
            if alpha < 1e-14:
                break

        if found:
            f = f_trial
            no_improve = 0
            if obj_trial < best_obj - 1e-15:
                best_obj = obj_trial
                best_f = f.copy()
                lr = min(lr * 1.001, 3.0 / n)
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr = max(lr * 0.8, lr_init * 0.005)
            if no_improve > 1000:
                rng = np.random.default_rng(t * 13 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init * 0.3
                no_improve = 0

        if beta < beta_end:
            beta *= 1.002

        if t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Fine t={t}, obj={best_obj:.12f}, beta={beta:.1f}, "
                  f"lr={lr:.8f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def active_set_polish(f0, time_limit):
    """Polish by identifying active set and doing constrained optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Identify active set (peaks within 0.5% of max)
        threshold = M * 0.9995
        active = np.where(conv >= threshold)[0]

        if len(active) == 0:
            break

        # Compute subgradient: average gradient over active peaks
        sg = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * f[k - idx]
        sg /= len(active)

        # Project: move opposite to subgradient
        f_trial = project_simplex(f - 0.001 * sg)
        conv_trial = fast_convolve(f_trial)
        obj_trial = conv_trial.max()

        if obj_trial < best_obj - 1e-15:
            best_obj = obj_trial
            best_f = f_trial.copy()
            f = f_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 7 + 1)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                m = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: FW-LP for coarse optimization
    Phase 2: LSE-Adam with warm beta schedule
    Phase 3: Very high beta LSE-Adam for final polish
    Phase 4: Active set polish
    Phase 5: Brief exploration of nearby lengths
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_seq = None
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Starting optimization with n={n_parent}")

    # === Phase 1: FW-LP ===
    print("\n=== Phase 1: FW-LP ===")
    t1 = min(250, deadline - time.monotonic() - 800)
    if t1 > 30:
        print(f"  Allocating {t1:.0f}s")
        a_opt, obj = fw_lp_optimize(parent, time_limit=t1)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 2: LSE-Adam warm ===
    print(f"\n=== Phase 2: LSE-Adam (warm) ===")
    t2 = min(500, deadline - time.monotonic() - 500)
    if t2 > 30:
        print(f"  Allocating {t2:.0f}s")
        a_opt, obj = lse_adam_optimize(current_best, time_limit=t2,
                                       beta_start=30.0, beta_end=10000.0,
                                       lr_init=3.0 / n_parent)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 3: LSE-Adam fine (very high beta) ===
    print(f"\n=== Phase 3: LSE-Adam (fine, high beta) ===")
    t3 = min(400, deadline - time.monotonic() - 200)
    if t3 > 30:
        print(f"  Allocating {t3:.0f}s")
        a_opt, obj = lse_adam_fine(current_best, time_limit=t3,
                                   beta_start=2000.0, beta_end=100000.0,
                                   lr_init=0.5 / n_parent)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 4: Active set polish ===
    print(f"\n=== Phase 4: Active Set Polish ===")
    t4 = min(100, deadline - time.monotonic() - 80)
    if t4 > 20:
        print(f"  Allocating {t4:.0f}s")
        a_opt, obj = active_set_polish(current_best, time_limit=t4)
        try_sequence(a_opt, best_tracker)

    # One more LSE-Adam fine pass with even higher beta
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t5 = min(150, deadline - time.monotonic() - 50)
    if t5 > 20:
        print(f"\n=== Phase 4b: Final LSE-Adam ===")
        print(f"  Allocating {t5:.0f}s")
        a_opt, obj = lse_adam_fine(current_best, time_limit=t5,
                                   beta_start=5000.0, beta_end=300000.0,
                                   lr_init=0.3 / n_parent)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Brief exploration of nearby lengths ===
    print(f"\n=== Phase 5: Nearby Lengths ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    nearby = []
    for dn in [-2, -1, 1, 2, 4, 8]:
        nl = n_best + dn
        if nl > 10 and nl not in nearby:
            nearby.append(nl)

    for n_alt in nearby[:3]:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 40:
            break
        t_alloc = min(60, t_remaining / max(1, len(nearby) - nearby.index(n_alt)))
        if t_alloc < 20:
            break

        # Interpolate to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        ss = f_interp.sum()
        if ss > 1e-10:
            f_interp = f_interp / ss

        # Quick optimization
        a_opt, _ = lse_adam_optimize(f_interp, time_limit=min(t_alloc, 40),
                                     beta_start=50.0, beta_end=5000.0,
                                     lr_init=3.0 / n_alt)
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = parent
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
