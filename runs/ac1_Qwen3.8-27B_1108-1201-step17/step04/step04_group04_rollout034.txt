"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive softmax gradient descent with FFT-based computation
2. Constraint-generation LP (avoids building full convolution matrix)
3. Multi-scale multi-start strategy
4. Conjugate gradient for the smooth subproblem

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """FFT-based linear convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_smooth_gradient(a, conv, temp, top_k=50):
    """
    Compute gradient of t*log(sum exp(conv[k]/t)) w.r.t. a.
    Uses only top-k peaks for efficiency.
    Returns (gradient, objective_value).
    """
    n = len(a)
    # Softmax weights
    if top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(top_k)
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        # Objective (approximate)
        obj = temp * (np.log(w_sum) + scaled.max() * 0)  # t * log(sum exp)
        # Actually: t * log(sum_k exp(conv[k]/t)) ≈ t * (log(w_sum) + max_conv/temp)
        # = t * log(w_sum) + max_conv
        # But we only computed over top_k, so this is an approximation
    else:
        scaled = conv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w /= w_sum
        peaks = np.arange(len(conv))
        peak_vals = conv
        obj = temp * (np.log(w_sum) + (conv.max()) / temp)

    # Gradient: d/d a_j [sum_k w_k * conv[k]] = 2 * sum_k w_k * a_{k-j}
    # This is 2 * cross_correlation(w_full, a)
    # But we only have weights for top_k peaks
    # Build sparse w vector
    w_full = np.zeros(len(conv))
    w_full[peaks] = w
    
    # cross_correlation: sum_k w_full[k] * a[k-j] for valid k-j
    # = sum_m w_full[j+m] * a[m] (m = k-j)
    # This is the (j-th element of) the convolution of reverse(w_full) with a
    # Actually: let's compute it as fft_convolve(w_full, a_reversed) or similar
    
    # sum_k w_full[k] * a[k-j] where 0 <= k-j < n
    # = sum_m w_full[j+m] * a[m] where 0 <= j+m < len(conv) and 0 <= m < n
    # This is the cross-correlation of w_full with a
    
    # Using FFT: cross_corr(w_full, a)[j] = sum_m w_full[j+m] * a[m]
    # = (reverse(w_full) * a)[j + len(w_full) - 1] ... hmm this gets complicated
    
    # Simpler: just use the direct formula with the convolution matrix approach
    # grad[j] = 2 * sum_k w_full[k] * a[k-j] (if 0 <= k-j < n, else 0)
    
    # This is 2 * (w_full convolved with reverse(a))[n-1-j] ... let me think again
    
    # conv(w_full, reverse(a))[p] = sum_k w_full[k] * reverse(a)[p-k]
    # reverse(a)[p-k] = a[n-1-(p-k)] = a[n-1-p+k]
    # So conv(w_full, rev_a)[p] = sum_k w_full[k] * a[n-1-p+k]
    # Let j = n-1-p+k... no this is getting messy.
    
    # Let's just use: grad[j] = 2 * sum_k w_full[k] * a[k-j]
    # where a[m] = 0 if m < 0 or m >= n
    # = 2 * sum_k w_full[k] * a[(k-j) mod n] (circular) -- NO, it's linear
    
    # Actually the cleanest way: use the fact that this is a convolution
    # Let's define: for fixed j, grad[j]/2 = sum_{k=0}^{2n-2} w_full[k] * a[k-j] * [0 <= k-j < n]
    # Let m = k-j: grad[j]/2 = sum_{m=0}^{n-1} w_full[m+j] * a[m]
    # This is the j-th element of the "upper diagonal" cross-correlation
    
    # We can compute this via FFT:
    # Consider the matrix W where W[k,j] = w_full[k] if 0 <= k-j < n else 0
    # Then grad/2 = W @ a
    # W is a Toeplitz matrix with first column [w_full[0], w_full[1], ..., w_full[n-1], 0, ...]
    # and each subsequent column is shifted down by 1
    
    # For efficiency, use FFT-based approach:
    # Pad a to length 2n-1, and compute the appropriate cross-correlation
    
    # Simplest efficient approach: use the relationship
    # sum_m w_full[j+m] * a[m] for m=0..n-1
    # = element j of (w_full ⊛ a) [cross-correlation]
    # Cross-correlation can be computed as: ifft(fft(w_full_full) * conj(fft(a_full)))
    
    # Let's pad both to length 2n-1
    w_pad = np.zeros(2*n-1)
    w_pad[:len(w_full)] = w_full  # w_full already has length 2n-1
    a_pad = np.zeros(2*n-1)
    a_pad[:n] = a
    
    # Cross-correlation: sum_m w_pad[j+m] * a_pad[m] -- but this isn't standard cross-corr
    # Standard cross-corr: (w ⊛ a)[j] = sum_m conj(w[m]) * a[m+j] = sum_m w[m] * a[m+j] (for real)
    # That's sum_m w[m] * a[m+j], which is what we want with j shifted
    # grad[j]/2 = sum_m w_full[m+j] * a[m] = sum_m w_full[m'] * a[m'-j] (letting m'=m+j)
    # = (a ⊛ w_full)[j] = cross-correlation of a with w_full at lag j
    
    # numpy/scipy: np.correlate(a, w_full, mode='full')[n-1+j] = sum_m a[m] * w_full[m-j]
    # Hmm, let me just use scipy signal or compute directly
    
    # Actually, the simplest correct formula:
    # grad[j] = 2 * sum_k w_full[k] * a[k-j] where 0<=k-j<n
    # = 2 * sum_{k=j}^{j+n-1} w_full[k] * a[k-j] (clipping k to [0, 2n-2])
    # = 2 * sum_{m=0}^{n-1} w_full[j+m] * a[m] (where w_full[m]=0 if m out of range)
    
    # This is element j of the "notch" of w_full cross-correlated with a
    # We can compute all of them at once using FFT:
    # Let's pad w_full to length >= 2n-1 (it already is)
    # cross_corr[j] = sum_m w_full[j+m] * a[m] for m=0..n-1
    # = sum_m w_full_full[j+m] * a_padded[m]
    # where w_full_full has length 2n-1 and a_padded has length n padded to 2n-1
    
    # This is the (n-1-j)-th element of convolve(reverse(w_full_full), a_padded)?
    # Let's verify: conv(rw, ap)[p] = sum_k rw[k] * ap[p-k]
    # rw[k] = w_full[2n-2-k], ap[m] = a[m] for m<n, 0 otherwise
    # conv(rw, ap)[p] = sum_k w_full[2n-2-k] * a[p-k] (when 0<=p-k<n)
    # Let q = 2n-2-k, so k = 2n-2-q:
    # = sum_q w_full[q] * a[p-(2n-2-q)] = sum_q w_full[q] * a[p-2n+2+q]
    # We want this to equal cross_corr[j] = sum_m w_full[j+m] * a[m]
    # Setting p-2n+2+q = m and j+m = q: p-2n+2+j+m = m, so p = 2n-2-j
    # conv(rw, ap)[2n-2-j] = cross_corr[j] ✓
    
    # So: grad[j] = 2 * conv(reverse(w_full), a_padded)[2n-2-j]
    # Or equivalently: grad = 2 * reverse(conv(reverse(w_full), a_padded))
    
    rw = w_full[::-1]
    ap = np.zeros(2*n-1)
    ap[:n] = a
    conv_result = fft_convolve(rw, ap)  # length 2*(2n-1)-1 = 4n-3, but we only need 2n-1
    # Actually fft_convolve gives full linear convolution, so length = (2n-1) + (2n-1) - 1 = 4n-3
    # We need elements [2n-2-j] for j=0..n-1, i.e., elements from index n-1 to 2n-2
    
    # Hmm, this is getting complicated. Let me just use a simpler approach for the gradient.
    
    # SIMPLE APPROACH: Direct computation using the structure
    # grad[j] = 2 * sum_{m=0}^{n-1} w_full[j+m] * a[m]
    # We can compute this as: for each peak k with weight w_k, add 2*w_k * a[k-j] to grad[j]
    # for j = max(0, k-n+1) to min(n-1, k)
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += 2.0 * w[idx] * a[k - idxs]
    
    return grad, conv.max()  # Return gradient and the actual max (not smooth approx)


def smooth_grad_optimize(a0, time_limit, temp_init=0.001, temp_growth=1.01, 
                         lr_init=0.05, lr_decay=0.999, top_k=50, 
                         momentum=0.9, seed=0):
    """
    Gradient descent on the smooth approximation of the peak autocorrelation.
    Uses softmax temperature that adapts over time.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    temp = temp_init
    lr = lr_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smooth gradient
        grad, cur_max = compute_smooth_gradient(a, conv, temp, top_k=top_k)
        
        # Gradient step with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = best_a.copy()
            conv = fft_convolve(a)
        else:
            a = a_new / s_new
            conv = fft_convolve(a)
        
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        temp *= temp_growth
        lr *= lr_decay
        momentum = min(0.97, momentum + 0.0001)
        
        if no_improve >= 300:
            # Perturb
            rng = np.random.default_rng(iteration * 31 + 7)
            scale = 0.002 * (1 + no_improve / 100)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fft_convolve(a)
            no_improve = 0
            v = np.zeros(n)
            lr = lr_init * 0.5
            temp = temp_init * 0.5
        
        iteration += 1
    
    return best_a, best_obj


def constraint_gen_lp(a, M, active_set=None, max_constraints=500):
    """
    Solve LP with only the most active constraints (constraint generation).
    max sum(g) s.t. conv(a,g)[k] <= M for k in active_set, g >= 0.
    Returns (g, active_set, success).
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    if active_set is None:
        # Start with the peak constraint
        active_set = set([int(np.argmax(fft_convolve(a)))])
    
    # Build constraint matrix from active set only
    k_list = sorted(active_set)[:max_constraints]
    C = np.zeros((len(k_list), n))
    for row, k in enumerate(k_list):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        C[row, j_start:j_end+1] = a[j_start:k+1][::-1]  # a[k-j] for j in [j_start, j_end]
    
    c = -np.ones(n)
    b = np.full(len(k_list), M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if not result.success:
            return a.copy(), active_set, False
        g = np.maximum(result.x, 0)
    except Exception:
        return a.copy(), active_set, False
    
    return g, active_set, True


def cg_lp_optimize(a0, time_limit, seed=0):
    """
    LP-based optimization using constraint generation.
    Much faster than full LP for large n.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fft_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    active_set = set([int(np.argmax(conv))])
    no_improve = 0
    iteration = 0
    perturb_count = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP with current active set
        M_current = conv.max()
        g, active_set, success = constraint_gen_lp(a, M_current * 1.0001, active_set)
        
        if not success:
            no_improve += 10
        else:
            g_sum = g.sum()
            if g_sum < 1e-10:
                no_improve += 10
            else:
                d = g - a
                norm_d = np.linalg.norm(d)
                if norm_d < 1e-10:
                    no_improve += 1
                else:
                    # Line search
                    best_t = 0
                    best_trial_obj = best_obj
                    for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
                        a_try = a + t * d
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        conv_try = fft_convolve(a_try)
                        obj_try = 2.0 * n * conv_try.max()
                        if obj_try < best_trial_obj - 1e-12:
                            best_trial_obj = obj_try
                            best_t = t
                            best_a_try = a_try.copy()
                            best_conv_try = conv_try
                            break
                    
                    if best_t > 0:
                        a = best_a_try
                        conv = best_conv_try
                        M = conv.max()
                        if best_trial_obj < best_obj - 1e-12:
                            best_obj = best_trial_obj
                            best_a = a.copy()
                        no_improve = 0
                        
                        # Update active set: add most violated constraint
                        # Check all constraints
                        # conv(a_new, g_new) should be <= M_new * 1.0001
                        # But we want to find which constraints are tight in the new direction
                        # Simply: recompute which lags have high values
                        peaks_new = np.argpartition(conv, -min(20, n))[-min(20, n):]
                        active_set = set(int(k) for k in peaks_new)
                    else:
                        no_improve += 1
        
        iteration += 1
        
        # Handle stagnation
        if no_improve >= 50 + 10 * perturb_count:
            scale = 0.002 + 0.001 * min(perturb_count, 10)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fft_convolve(a)
            no_improve = 0
            perturb_count += 1
            active_set = set([int(np.argmax(conv))])
            
            if perturb_count > 30:
                break
        
        # Periodic gradient burst
        if no_improve >= 30 and no_improve % 30 == 0:
            # Quick gradient step
            grad, _ = compute_smooth_gradient(a, conv, 0.002, top_k=30)
            a = a - 0.01 * grad
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fft_convolve(a)
    
    return best_a, best_obj


def get_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Chirp-based (frequency-varying)
    for f in [1, 2, 3, 5]:
        phase = np.pi * f * (x / n) ** 2
        sig = np.abs(np.exp(1j * phase))
        # This is always 1, so add position-dependent weighting
        weight = np.exp(-0.5 * ((x - n/2) / (n/4)) ** 2)
        inits[f'chirp_{f}'] = weight / weight.sum()
    
    # Cosine windows
    inits['hann'] = (0.5 + 0.5 * np.cos(2 * np.pi * x / n))
    inits['hann'] += 0.01
    inits['hann'] /= inits['hann'].sum()
    
    inits['blackman'] = (0.42 + 0.5 * np.cos(2 * np.pi * x / n) + 0.08 * np.cos(4 * np.pi * x / n))
    inits['blackman'] += 0.01
    inits['blackman'] /= inits['blackman'].sum()
    
    # Sine patterns
    for freq in [1, 2, 3]:
        sig = (np.sin(np.pi * freq * x / n) ** 2 + 0.01)
        inits[f'sine_{freq}'] = sig / sig.sum()
    
    # Bimodal
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sparse random
    sparse = np.zeros(n)
    n_nonzero = max(2, n // 4)
    idx = rng.choice(n, size=n_nonzero, replace=False)
    sparse[idx] = 1.0
    inits['sparse'] = sparse / sparse.sum()
    
    # Alternating with decay
    alt = np.ones(n)
    alt[:n//2] = 1.5
    alt[n//2:] = 0.5
    inits['alternating'] = alt / alt.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits

def fft_convolve(a, b=None):
    """FFT-based linear convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_gradient(a, conv, temp, top_k=50):
    """
    Compute gradient of softmax-approximated peak autocorrelation.
    grad[j] = 2 * sum_k w_k * a[k-j] (with a[m]=0 for out-of-range m)
    """
    n = len(a)
    conv_len = len(conv)
    
    if top_k < conv_len:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(conv_len)
    
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += 2.0 * w[idx] * a[k - idxs]
    
    return grad

def cg_lp_step(a, conv, active_set, M):
    """
    One constraint-generation LP step.
    Solve: max sum(g) s.t. conv(a,g)[k] <= M for k in active_set, g >= 0.
    Returns (g, new_active_set, success).
    """
    n = len(a)
    
    # Limit active set size for efficiency
    if len(active_set) > 200:
        # Keep the most violated constraints
        violations = np.array([conv[k] for k in active_set])
        order = np.argsort(violations)[::-1]
        active_set = set(int(list(active_set)[i]) for i in order[:200])
    
    k_list = sorted(active_set)
    if len(k_list) == 0:
        k_list = [int(np.argmax(conv))]
    
    # Build constraint matrix
    C = np.zeros((len(k_list), n))
    for row, k in enumerate(k_list):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            C[row, j_start:j_end+1] = a[k - np.arange(j_start, j_end+1)]
    
    c_obj = -np.ones(n)
    b_vec = np.full(len(k_list), M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_vec, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-9,
                                                   'ipm_gap_tolerance': 1e-9})
        if not result.success:
            return a.copy(), active_set, False
        g = np.maximum(result.x, 0)
        if g.sum() < 1e-12:
            return a.copy(), active_set, False
    except Exception:
        return a.copy(), active_set, False
    
    # Update active set for next iteration: add top peaks
    n_peaks = min(30, n)
    peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
    new_active = set(int(k) for k in peaks)
    # Also keep currently active constraints that are tight
    conv_ag = fft_convolve(a, g)
    tight = set(int(k) for k in range(len(conv_ag)) if conv_ag[k] > 0.9 * M)
    new_active = new_active | tight
    if len(new_active) > 300:
        # Prune: keep only most violated
        vals = np.array([conv[k] if k < len(conv) else 0 for k in new_active])
        order = np.argsort(vals)[::-1]
        new_active = set(int(list(new_active)[i]) for i in order[:300])
    
    return g, new_active, True

def get_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Window functions
    hann = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits['hann'] = (hann + 0.01) / (hann + 0.01).sum()
    
    blackman = 0.42 + 0.5 * np.cos(2 * np.pi * x / n) + 0.08 * np.cos(4 * np.pi * x / n)
    inits['blackman'] = (blackman + 0.01) / (blackman + 0.01).sum()
    
    # Sine patterns
    for freq in [1, 2, 3, 5]:
        sig = np.sin(np.pi * freq * x / n) ** 2 + 0.01
        inits[f'sine_{freq}'] = sig / sig.sum()
    
    # Cosine ridge
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        sig = np.exp(-alpha * (x / n - 0.5) ** 2) + 0.001
        inits[f'ridge_{alpha}'] = sig / sig.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Trapezoid
    trap = np.zeros(n)
    q = n // 4
    trap[:q] = np.linspace(0.3, 1.0, q)
    trap[q:3*q] = 1.0
    trap[3*q:] = np.linspace(1.0, 0.3, n - 3*q)
    inits['trapezoid'] = trap / trap.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sparse
    sparse = np.zeros(n)
    n_nz = max(2, n // 3)
    idx = rng.choice(n, size=n_nz, replace=False)
    sparse[idx] = rng.uniform(0.5, 1.5, n_nz)
    inits['sparse'] = sparse / sparse.sum()
    
    # Alternating block
    blk = np.ones(n)
    blk[0::2] = 1.3
    blk[1::2] = 0.7
    inits['alt_block'] = blk / blk.sum()
    
    # Quadratic
    quad = (x - n/2)**2 / n + 0.1
    inits['quadratic'] = quad / quad.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.all(np.isfinite(p)):
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits


def population_combine(candidates, rng, n_target):
    """Combine top candidates by weighted averaging + mutation."""
    if len(candidates) == 0:
        return np.ones(n_target) / n_target
    if len(candidates) == 1:
        c = candidates[0]
        if len(c) == n_target:
            return c.copy()
        ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, len(c)), c / max(c.sum(), 1e-15))
        return np.maximum(ext, 0) / max(ext.sum(), 1e-15)
    
    # Weight by quality (inverse score)
    n_max = max(len(c) for c in candidates)
    # Interpolate all to same length
    interp_cands = []
    for c in candidates:
        if len(c) == n_max:
            cn = c / max(c.sum(), 1e-15)
        else:
            cn = np.interp(np.linspace(0, 1, n_max), np.linspace(0, 1, len(c)), c / max(c.sum(), 1e-15))
        interp_cands.append(np.maximum(cn, 0))
    
    # Random weights
    weights = rng.dirichlet(np.ones(len(candidates)) * 2)
    combined = sum(w * c for w, c in zip(weights, interp_cands))
    combined = np.maximum(combined, 0)
    cs = combined.sum()
    if cs < 1e-15:
        combined = np.ones(n_max) / n_max
    else:
        combined = combined / cs
    
    # Small mutation
    combined += rng.standard_normal(n_max) * 0.005
    combined = np.maximum(combined, 0)
    cs = combined.sum()
    if cs < 1e-15:
        combined = np.ones(n_max) / n_max
    else:
        combined = combined / cs
    
    # Resample to target length
    if n_max != n_target:
        combined = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_max), combined)
        combined = np.maximum(combined, 0)
        cs = combined.sum()
        if cs < 1e-15:
            combined = np.ones(n_target) / n_target
        else:
            combined = combined / cs
    
    return combined


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid smooth-gradient + constraint-generation LP search
    with population-based exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    parent_n = len(parent) if parent is not None else 512
    
    # =========================================================
    # Phase 1: Fast gradient exploration (diverse starts)
    # =========================================================
    print("=== Phase 1: Gradient Exploration ===")
    t_start = time.monotonic()
    
    # Choose lengths to explore
    lengths_to_try = []
    candidates_n = [parent_n // 2, parent_n, parent_n * 3 // 2, parent_n * 2]
    for cn in candidates_n:
        cn = max(128, min(2048, cn))
        if cn not in lengths_to_try:
            lengths_to_try.append(cn)
    lengths_to_try.sort()
    
    all_results = []  # (score, n, a, method)
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 700:
            break
        inits = get_initializations(n, parent, rng)
        
        # Priority: parent-based first, then diverse
        priority = ['parent', 'parent_interp', 'uniform', 'hann', 'sine_1',
                    'ridge_1.0', 'random', 'bimodal', 'trapezoid', 'ridge_2.0']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits.keys() if nm not in priority][:3]
        init_names = init_names[:5]
        
        t_per = max(10.0, 250.0 / (len(lengths_to_try) * 5))
        
        for name in init_names:
            if time.monotonic() > deadline - 600:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 200)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            
            # Use different hyperparameters for diversity
            temp_cfgs = [
                (0.001, 1.005, 0.05, 0.9995, 50, 0.90),
                (0.002, 1.01, 0.03, 0.999, 80, 0.92),
                (0.0005, 1.003, 0.08, 0.9997, 30, 0.88),
                (0.003, 1.015, 0.02, 0.998, 100, 0.93),
                (0.001, 1.008, 0.04, 0.9993, 60, 0.91),
            ]
            cfg = temp_cfgs[hash(name) % len(temp_cfgs)]
            ti, tg, li, ld, tk, mo = cfg
            
            a_opt, obj = smooth_grad_optimize(a0, time_limit=t_alloc,
                                               temp_init=ti, temp_growth=tg,
                                               lr_init=li, lr_decay=ld,
                                               top_k=tk, momentum=mo,
                                               seed=hash(name) % 10000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                all_results.append((score, n, a_opt.copy(), f'grad_{name}'))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f} *BEST*")
    
    all_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 2: Deep CG-LP optimization on top candidates
    # =========================================================
    print("=== Phase 2: CG-LP Deep Optimization ===")
    t_start2 = time.monotonic()
    
    # Select top candidates (diverse by length)
    top_configs = []
    seen_lens = set()
    for score, n, a, method in all_results:
        if n not in seen_lens:
            top_configs.append((n, a))
            seen_lens.add(n)
        if len(top_configs) >= 4:
            break
    
    # Always include overall best
    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_lens:
            top_configs.append((bn, best_sequence.copy()))
            seen_lens.add(bn)
    
    remaining = deadline - time.monotonic() - 300.0
    n_cand = max(1, len(top_configs))
    t_per_lp = max(40.0, remaining * 0.55 / n_cand)
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 150)
        if t_alloc < 15:
            break
        
        print(f"  CG-LP {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj = cg_lp_optimize(a0, time_limit=t_alloc, seed=i * 137 + seed)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CG-LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  CG-LP {i} (n={n}): {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: Population-based exploration + alternating polish
    # =========================================================
    print("=== Phase 3: Population + Polish ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 100.0
        
        if remaining > 50:
            # Build population from top results
            population = [best_sequence.copy()]
            for score, n, a, method in all_results[:5]:
                if a is not best_sequence and len(a) == n_best:
                    population.append(a.copy())
            
            # Alternating: gradient -> LP -> combine -> gradient -> ...
            n_rounds = min(5, int(remaining / 40))
            for pi in range(n_rounds):
                if time.monotonic() > deadline - 60:
                    break
                t_each = min(remaining / (n_rounds - pi) * 0.6, 35.0)
                if t_each < 8:
                    break
                
                # Gradient pass on best
                a_opt, _ = smooth_grad_optimize(best_sequence, time_limit=t_each * 0.4,
                                                 temp_init=0.0005, temp_growth=1.01,
                                                 lr_init=0.03, lr_decay=0.999,
                                                 top_k=70, momentum=0.92,
                                                 seed=pi * 50 + seed)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    population.insert(0, a_opt.copy())
                
                # LP pass
                rem2 = deadline - time.monotonic() - 60
                if rem2 > 8:
                    a_opt, _ = cg_lp_optimize(best_sequence, time_limit=rem2 * 0.4,
                                               seed=pi * 200 + seed + 7)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        population.insert(0, a_opt.copy())
                        print(f"  Round {pi}: {score:.6f} *NEW BEST*")
                
                # Population combine
                if len(population) >= 2 and time.monotonic() < deadline - 40:
                    combined = population_combine(population[:4], rng, n_best)
                    # Quick gradient on combined
                    t_comb = min(8.0, deadline - time.monotonic() - 30)
                    if t_comb > 3:
                        a_c, _ = smooth_grad_optimize(combined, time_limit=t_comb,
                                                       temp_init=0.001, temp_growth=1.01,
                                                       lr_init=0.03, lr_decay=0.999,
                                                       top_k=50, momentum=0.9,
                                                       seed=pi + 999)
                        score_c = float(evaluate_sequence(a_c.tolist()))
                        if np.isfinite(score_c) and score_c < best_score:
                            best_score = score_c
                            best_sequence = a_c.copy()
                            print(f"  Combine {pi}: {score_c:.6f} *NEW BEST*")
            
            # Trim population
            population = population[:6]
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 4: Final micro-polish
    # =========================================================
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 8:
        t_final = min(20.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            # Very fine gradient
            a_opt, _ = smooth_grad_optimize(best_sequence, time_limit=t_final * 0.5,
                                             temp_init=0.0001, temp_growth=1.005,
                                             lr_init=0.01, lr_decay=0.9997,
                                             top_k=120, momentum=0.95,
                                             seed=seed + 7777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  GradFinal: {score:.6f} *NEW BEST*")
            
            # Very fine LP
            rem = deadline - time.monotonic() - 5
            if rem > 5:
                a_opt, _ = cg_lp_optimize(best_sequence, time_limit=rem,
                                           seed=seed + 8888)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LPFinal: {score:.6f} *NEW BEST*")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
