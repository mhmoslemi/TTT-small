        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    # Build the full response vector
    w_full = np.zeros(len_c)
    w_full[indices] = w
    
    # Subgradient: sg[j] = 2 * sum_k w_full[k] * a[k-j] for valid j
    # This is cross-correlation of w_full with a
    # corr[j] = sum_k w_full[k] * a[k-j] (with a out-of-bounds = 0)
    # Using FFT: pad both to length >= len_c + n - 1
    total_size = len_c + n - 1  # = 3n - 2
    fft_size = 1 << (total_size - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Linear cross-correlation: ifft(conj(W) * A) gives sum_k w[k]*a[k-j+circular]
    # For linear (non-circular), we need the "full" correlation
    # Actually, the standard FFT-based correlation:
    # corr_full[lag] = ifft(conj(FFT(w)) * FFT(a))[lag] (circular)
    # For linear, we pad to size >= len(w) + len(a) - 1
    # The result at index j gives sum_k w[k]*a[k-j] for the linear case
    # BUT with the right offset!
    
    # Let me use a simpler approach: 
    # sg[j] = sum_k w_full[k] * a[k-j]
    # = sum_k w_full[k] * a[k-j] where a is zero-padded
    # This is the convolution of w_full with flip(a)
    # conv(w_full, flip(a))[j + n - 1] = sum_k w_full[k] * flip(a)[j + n - 1 - k]
    # = sum_k w_full[k] * a[n-1 - (j+n-1-k)] = sum_k w_full[k] * a[k-j]
    # So sg = 2 * conv(w_full, flip(a))[n-1 : n-1+n]
    
    fa = a[::-1]
    conv_size = len(w_full) + n - 1  # = 3n-2
    fft_size2 = 1 << (conv_size - 1).bit_length()
    Wf = np.fft.rfft(w_full, fft_size2)
    Ff = np.fft.rfft(fa, fft_size2)
    conv = np.fft.irfft(Wf * Ff, fft_size2)[:conv_size]
    sg = 2.0 * conv[n-1:n-1+n]
    
    return sg

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved exponentiated gradient

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient_fft(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient using FFT-based convolution."""
    n = len(a)
    len_c = len(c)

    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()

    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0

        w /= ws

    w_full[indices] = w

    # = 2 * conv(w_full, flip(a))[n-1 + j]

    Wf = np.fft.rfft(w_full, fft_size2)
    Ff = np.fft.rfft(fa, fft_size2)
    conv = np.fft.irfft(Wf * Ff, fft_size2)[:conv_size]

    it = 0
    eshock_counter = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        sg = compute_subgradient_fft(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        # Adaptive temperature
        threshold = 0.85 * c.max()
        peak_width = np.count_nonzero(c > threshold) / len(c)
        if peak_width < 0.03:
            temp = max(temp0 * 0.05, temp * 0.998)
        elif peak_width > 0.4:
            temp = min(temp0 * 10.0, temp * 1.002)
        else:
            temp *= temp_decay

        # Check running average periodically
        if it % 800 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay
        eshock_counter += 1

        # Temperature shock
        if eshock_counter >= 6000:
            temp = max(temp, temp0 * 0.5)
            eshock_counter = 0

        # Stagnation restart with structured perturbation
        if no_improve > 2500:
            rng_local = np.random.default_rng(it * 31 + 7 + int(temp * 1e6))
            x = np.arange(n, dtype=np.float64)
            strategy = it % 4
            if strategy == 0:
                freq = rng_local.uniform(1.0, 4.0) * n
                phi = rng_local.uniform(0, 2 * np.pi)
                pert = np.sin(2 * np.pi * freq * x / n + phi) * 0.008
            elif strategy == 1:
                center = rng_local.uniform(0, n)
                width = rng_local.uniform(0.05, 0.15) * n
                pert = np.exp(-0.5 * ((x - center) / width) ** 2) * 0.015
            elif strategy == 2:
                pert = np.zeros(n)
                idx = rng_local.choice(n, size=max(1, n // 8), replace=False)
                pert[idx] = rng_local.standard_normal(len(idx)) * 0.008
            else:
                freq = rng_local.uniform(0.3, 2.0) * n
                pert = np.cos(2 * np.pi * freq * x / n) * 0.006

            a = best_a + pert
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.5
            temp = max(temp, temp0 * 0.5)
            avg_a = a.copy()
            count = 1

    return best_a.copy(), best_obj


def peak_tracking_cd(a0, time_limit, max_iters=5000):
    """Coordinate descent targeting specific peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 >= time_limit:
            break

        # Track top 3 peaks
        if n >= 6:
            top_peaks = np.argpartition(c, -3)[-3:]
            top_peaks = top_peaks[np.argsort(c[top_peaks])[::-1]]
        else:
            top_peaks = np.argsort(c)[::-1][:3]

        improved = False
        for k in top_peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            contribs = a[js:je + 1] * a[k - je:k - js + 1][::-1]
            total = contribs.sum()
            if total < 1e-15:
                continue

            max_idx = np.argmax(contribs) + js
            old_val = best_a[max_idx]
            if old_val < 1e-10:
                continue

            # Try reducing the most contributing coordinate
            for factor in [0.85, 0.7, 0.5, 0.3, 0.92, 0.95, 1.05, 1.15, 1.3]:
                a_trial = best_a.copy()
                a_trial[max_idx] = old_val * factor
                if a_trial[max_idx] < 0:
                    continue
                s2 = a_trial.sum()
                if s2 < 1e-10:
                    continue
                a_trial = a_trial / s2
                c2 = fast_convolve(a_trial)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-13:
                    best_obj = obj2
                    best_a = a_trial.copy()
                    c = c2
                    improved = True
                    break

            # Also try the paired coordinate
            pair_idx = k - max_idx
            if 0 <= pair_idx < n and not improved:
                old_pv = best_a[pair_idx]
                if old_pv > 1e-10:
                    for factor in [0.85, 0.7, 0.5, 0.92, 1.05, 1.15]:
                        a_trial = best_a.copy()
                        a_trial[pair_idx] = old_pv * factor
                        if a_trial[pair_idx] < 0:
                            continue
                        s2 = a_trial.sum()
                        if s2 < 1e-10:
                            continue
                        a_trial = a_trial / s2
                        c2 = fast_convolve(a_trial)
                        obj2 = c2.max()
                        if obj2 < best_obj - 1e-13:
                            best_obj = obj2
                            best_a = a_trial.copy()
                            c = c2
                            improved = True
                            break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Random coordinate perturbation
            n_try = max(1, n // 16)
            for _ in range(n_try):
                i = rng.integers(0, n)
                for factor in [0.6, 0.8, 1.2, 1.5, 0.4, 2.0]:
                    a_trial = best_a.copy()
                    a_trial[i] = best_a[i] * factor
                    s2 = a_trial.sum()
                    if s2 < 1e-10:
                        continue
                    a_trial = a_trial / s2
                    c2 = fast_convolve(a_trial)
                    obj2 = c2.max()
                    if obj2 < best_obj - 1e-13:
                        best_obj = obj2
                        best_a = a_trial.copy()
                        c = c2
                        improved = True
                        break
                if improved:
                    break
            if not improved and no_improve > 80:
                break

    return best_a.copy(), best_obj


def evaluate_local(a):
    """Evaluate sequence using the external evaluate_sequence function."""
    return float(evaluate_sequence(a.tolist()))


def run_eg_and_evaluate(a_init, time_limit, eta0, temp0, temp_decay, eta_decay, top_k_frac):
    """Run EG optimization and return the best sequence found."""
    a_opt, obj = exp_gradient_core(a_init, time_limit, eta0, temp0, temp_decay, eta_decay, top_k_frac)
    return a_opt


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: improved EG with FFT subgradients, adaptive temperature,
    peak-tracking CD, structured restarts, and multi-scale refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent, using uniform n=384")
        best_score = evaluate_local(parent)
        best_seq = parent.copy()

    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.10f}")

    # Helper to try a sequence
    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = evaluate_local(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # === Phase 1: Primary EG with best hyperparameters ===
    print("\n=== Phase 1: Primary EG ===")
    t1 = min(550, max(60, (deadline - time.monotonic()) - 600))
    if t1 > 60:
        print(f"  EG (n={n}, alloc={t1:.0f}s, eta0=0.8, temp0=0.002)")
        a_opt = run_eg_and_evaluate(best_seq, t1, 0.8, 0.002, 0.9994, 0.99985, 0.3)
        _try(a_opt, "eg_p1")

    # === Phase 2: Peak-tracking CD ===
    print(f"\n=== Phase 2: Peak CD ===")
    print(f"  Best: {best_score:.10f}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t2 = min(rem, 120)
            print(f"  CD alloc={t2:.0f}s")
            a_opt, _ = peak_tracking_cd(best_seq, time_limit=t2, max_iters=4000)
            _try(a_opt, "cd_p2")

    # === Phase 3: EG with varied hyperparameters ===
    print(f"\n=== Phase 3: EG Variations ===")
    print(f"  Best: {best_score:.10f}")

    configs = [
        (0.4, 0.005, 0.9997, 0.9999, 0.5),
        (2.5, 0.001, 0.9992, 0.9997, 0.15),
        (1.5, 0.003, 0.9995, 0.9999, 0.2),
        (0.6, 0.01, 0.9999, 0.99995, 0.6),
    ]
    for ci, (e0, t0, td, ed, tk) in enumerate(configs):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(80, rem / max(1, len(configs) - ci))
        if t_alloc < 15:
            break
        print(f"  cfg{ci}: eta0={e0}, temp0={t0}, alloc={t_alloc:.0f}s")
        a_opt = run_eg_and_evaluate(best_seq, t_alloc, e0, t0, td, ed, tk)
        _try(a_opt, f"eg_c{ci}")

    # === Phase 4: Structured perturbation restarts ===
    print(f"\n=== Phase 4: Structured Restarts ===")
    print(f"  Best: {best_score:.10f}")

    if best_seq is not None:
        x = np.arange(n, dtype=np.float64)
        restart_specs = [
            ("sin1", np.sin(2 * np.pi * 2.5 * x / n) * 0.01),
            ("sin3", np.sin(2 * np.pi * 3.7 * x / n + 0.5) * 0.008),
            ("cos2", np.cos(2 * np.pi * 1.8 * x / n) * 0.007),
            ("gauss", np.exp(-0.5 * ((x - n * 0.3) / (0.1 * n)) ** 2) * 0.02),
            ("gauss2", np.exp(-0.5 * ((x - n * 0.7) / (0.08 * n)) ** 2) * 0.015),
        ]
        for ri, (rlabel, pert) in enumerate(restart_specs):
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alloc = min(50, rem / max(1, len(restart_specs) - ri))
            if t_alloc < 10:
                break
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps > 1e-10:
                a_pert = a_pert / ps
            a_opt = run_eg_and_evaluate(a_pert, t_alloc, 0.8, 0.003, 0.9996, 0.9999, 0.25)
            _try(a_opt, f"rst_{rlabel}")

    # === Phase 5: Random inits + quick EG ===
    print(f"\n=== Phase 5: Diverse Inits ===")
    print(f"  Best: {best_score:.10f}")

    if best_seq is not None:
        inits = []
        # Exponential decays
        for beta in [0.5, 1.5, 4.0]:
            e = np.exp(-beta * x / n)
            inits.append((f"exp{beta}", e / e.sum()))
            e2 = np.exp(-beta * (n - 1 - x) / n)
            inits.append((f"expR{beta}", e2 / e2.sum()))
        # Edge concentrated
        edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
        inits.append(("edge", edge_w / edge_w.sum()))
        # Perturbed best
        for scale in [0.003, 0.01, 0.03]:
            p = best_seq + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            if p.sum() > 1e-10:
                inits.append((f"pert{scale}", p / p.sum()))

        for ii, (ilabel, a_init) in enumerate(inits):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(35, rem / max(1, len(inits) - ii))
            if t_alloc < 8:
                break
            a_opt = run_eg_and_evaluate(a_init, t_alloc, 1.0, 0.005, 0.9997, 0.9999, 0.3)
            _try(a_opt, f"init_{ilabel}")

    # === Phase 6: Try slightly different lengths ===
    print(f"\n=== Phase 6: Length Variants ===")
    print(f"  Best: {best_score:.10f}")

    if best_seq is not None:
        alt_lengths = [n // 2, n * 3 // 2, int(n * 0.75), int(n * 1.25)]
        for an in alt_lengths:
            if an < 32 or an > 1024:
                continue
            rem = deadline - time.monotonic() - 40
            if rem < 20:
                break
            t_alloc = min(30, rem)
            # Interpolate
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, an)
            a_alt = np.interp(x_new, x_old, best_seq)
            a_alt = np.maximum(a_alt, 0)
            s_alt = a_alt.sum()
            if s_alt > 1e-10:
                a_alt = a_alt / s_alt
                a_opt = run_eg_and_evaluate(a_alt, t_alloc, 0.8, 0.003, 0.9996, 0.9999, 0.25)
                _try(a_opt, f"len{an}")

    # === Phase 7: Final fine EG + CD ===
    print(f"\n=== Phase 7: Final Polish ===")
    print(f"  Best: {best_score:.10f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t7 = min(rem, 50)
            print(f"  Fine EG alloc={t7:.0f}s")
            a_opt = run_eg_and_evaluate(best_seq, t7, 0.15, 0.0005, 0.99995, 0.99999, 0.5)
            _try(a_opt, "fine_eg")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = peak_tracking_cd(best_seq, time_limit=min(rem, 25), max_iters=3000)
            _try(a_opt, "fine_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = evaluate_local(best_seq)

    # Final verification
    fs = evaluate_local(best_seq)
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
