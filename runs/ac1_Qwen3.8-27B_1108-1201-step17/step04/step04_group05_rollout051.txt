import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, peaks, weights):
    n = len(a)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * weights[i] * a[k - idx]
    return sg

def nesterov_subgrad(a0, time_limit, K=20, lr0=0.005, mom0=0.9, temp0=1e-4, 
                      schedule='cosine', verbose=True):
    """Nesterov-accelerated subgradient descent with softmax multi-peak targeting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # velocity
    mom = mom0
    
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        dt = time.monotonic() - t0
        
        # Nesterov step
        y = a + mom * v
        
        # Compute convolution of y
        conv_y = fast_convolve(y)
        
        # Select top K peaks
        if K >= len(conv_y):
            peaks = np.arange(len(conv_y))
        else:
            peaks = np.argpartition(conv_y, -K)[-K:]
        
        pv = conv_y[peaks]
        scaled = pv / temp0
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = compute_subgradient(y, conv_y, peaks, w)
        
        # Update velocity and position
        v = mom * v - lr0 * sg
        a = project_simplex(a + v)
        
        # Evaluate
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if schedule == 'cosine':
            alpha = dt / time_limit
            lr_eff = lr0 * (0.5 * (1 + np.cos(np.pi * alpha)))
        else:
            lr_eff = lr0
