    size = 2 * n - 1

def lp_newton_step(a, conv, active_peaks):
    """
    Solve LP: minimize t
    subject to: conv[k] + 2 * sum_i a[k-i] * da[i] <= t  for k in active_peaks
                sum(da) = 0
                da[i] >= -a[i]  (i.e., a[i] + da[i] >= 0)
    
    Returns (da, t_opt) or (None, inf) if infeasible.
    """
    n = len(a)
    K = len(active_peaks)
    
    # Decision variables: [da_0, ..., da_{n-1}, t]
    # Objective: minimize t → c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active peak k: 2*sum_i a[k-i]*da[i] - t <= -conv[k]
    # Plus: -da[i] <= a[i] for each i
    
    n_ineq = K + n
    A_ub = np.zeros((n_ineq, n + 1))
    b_ub = np.zeros(n_ineq)
    
    for idx, k in enumerate(active_peaks):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                A_ub[idx, i] = 2.0 * a[j]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -conv[k]
    
    for i in range(n):
        A_ub[K + i, i] = -1.0
        b_ub[K + i] = a[i]
    
    # Equality constraint: sum(da) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if result.success:
            da = result.x[:n]
            t_opt = result.x[n]
            return da, t_opt
    except Exception:
        pass
    
    return None, np.inf

def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()

"""
Semi-smooth Newton optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP subproblems with full active set, line search, and multi-start.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lp_newton_direction(a, conv, active_indices):
    """
    Solve LP to find Newton direction for reducing max peak.
    
    Minimize t
    s.t. conv[k] + 2 * sum_i a[k-i] * da[i] <= t  for k in active_indices
         sum(da) = 0
         da[i] >= -a[i]
    
    Returns (da, t_opt) or (None, inf).
    """
    n = len(a)
    K = len(active_indices)
    
    # Variables: [da_0, ..., da_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints
    # Rows 0..K-1: convolution constraints
    # Rows K..K+n-1: non-negativity (da[i] >= -a[i] → -da[i] <= a[i])
    n_rows = K + n
    A_ub = np.zeros((n_rows, n + 1))
    b_ub = np.zeros(n_rows)
    
    for idx, k in enumerate(active_indices):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A_ub[idx, i] = 2.0 * a[k - i]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -conv[k]
    
    for i in range(n):
        A_ub[K + i, i] = -1.0
        b_ub[K + i] = a[i]
    
    # Equality: sum(da) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            da = np.maximum(res.x[:n], -a)  # Ensure feasibility
            t_opt = res.x[n]
            return da, t_opt
    except Exception:
        pass
    
    return None, np.inf

    perturb_count = 0
    max_stall = 60

        iters_left_time = time_limit - (time.monotonic() - t0)

            stall = 0

def subgradient_optimize(a0, time_limit):
    """
    Subgradient method with Polyak step size as fallback.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    f_star_est = best_obj / (2.0 * n)  # estimate of optimal max(conv)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        cur_obj = 2.0 * n * M
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            f_star_est = best_obj / (2.0 * n)
            stall = 0
        else:
            stall += 1
        
        # Find active peaks
        thresh = M * 0.995
        active = np.where(conv >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        K = len(active)
        
        # Compute subgradient: average of gradients of active peaks
        # grad_k[m] = 2 * a[k-m] (with zero extension)
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                grad[i] += 2.0 * a[k - i] / K
        
        # Polyak step size
        gap = M - f_star_est
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-30:
            break
        lr = gap / g_norm_sq
        lr = min(lr, 1.0)  # Cap step size
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new
        a = a_new
        
        if stall > 200:
            break
    
    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        p = np.clip(parent, 0, None)
        ps = p.sum()
        if ps > 1e-10:
            p_interp = np.interp(np.linspace(0, 1, n),
                                 np.linspace(0, 1, len(p)), p)
            p_interp = np.maximum(p_interp, 0)
            psi = p_interp.sum()
            if psi > 1e-10:
                inits.append(('parent', p_interp / psi))
    
    # Sine squared
    s = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits.append(('sine', s / s.sum()))
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(('gauss', g / g.sum()))
    
    # Bimodal (edges)
    w = max(1, n // 6)
    b = np.zeros(n)
    b[:w] = 1.0
    b[-w:] = 1.0
    inits.append(('bimodal', b / b.sum()))
    
    # Random Dirichlet
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(('random', r))
    
    return inits

"""
Semi-smooth Newton optimization with LP subproblems for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses full-convolution LP directions, Nesterov subgradient fallback,
and coordinate descent polishing with multi-start across lengths.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lp_newton_direction(a, conv, active_indices):
    """
    Solve LP: minimize t
    s.t. conv[k] + 2*sum_i a[k-i]*da[i] <= t  for k in active_indices
         sum(da) = 0
         da[i] >= -a[i]
    Returns (da, t_opt) or (None, inf).
    """
    n = len(a)
    K = len(active_indices)

    c = np.zeros(n + 1)
    c[n] = 1.0

    n_rows = K + n
    A_ub = np.zeros((n_rows, n + 1))
    b_ub = np.zeros(n_rows)

    for idx, k in enumerate(active_indices):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A_ub[idx, i] = 2.0 * a[k - i]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -conv[k]

    for i in range(n):
        A_ub[K + i, i] = -1.0
        b_ub[K + i] = a[i]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                      b_eq=np.zeros(1), method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            da = res.x[:n]
            t_opt = res.x[n]
            return da, t_opt
    except Exception:
        pass

    return None, np.inf

def nesterov_subgradient(a0, time_limit):
    """Nesterov-accelerated subgradient with adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    v = np.zeros(n)
    y = a.copy()
    t0 = time.monotonic()
    lr = 0.01
    mom = 0.9
    top_k = 30
    temp = 0.004
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        # Soft gradient over top peaks
        conv_y = fast_convolve(y)
        peak_idx = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peak_idx]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-15:
            w = w / w_sum
        else:
            w = np.zeros(top_k)
            w[np.argmax(peak_vals)] = 1.0

        grad = np.zeros(n)
        for idx in range(top_k):
            if w[idx] < 1e-12:
                continue
            k = int(peak_idx[idx])
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                grad[i] += w[idx] * 2.0 * y[k - i]

        # Update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        temp *= 1.005
        mom = min(0.97, mom + 0.0003)

        if no_improve >= 300:
            break

    return best_a, best_obj


def coordinate_polish(a0, time_limit):

    a = np.maximum(a0.astype(np.float64), 0.0).copy()

        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s

    while time.monotonic() - t0 < time_limit:
        batch = max(1, int(300000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.2:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.02, 0.08, 0.15, 0.3, -0.02, -0.08, -0.15]:
                delta = a[j] * frac
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()

                    a = a_try

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n) / n))

    if parent is not None and len(parent) > 0:
        p = np.clip(parent, 0, None)
        ps = p.sum()
        if ps > 1e-10:
            p_int = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            p_int = np.maximum(p_int, 0)
            ps2 = p_int.sum()
            if ps2 > 1e-10:
                inits.append(('parent', p_int / ps2))

    s1 = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits.append(('sine1', s1 / s1.sum()))

    s2 = np.sin(2 * np.pi * (x + 0.5) / n) ** 2 + 0.05

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(('gauss', g / g.sum()))

    w = max(1, n // 5)
    b = np.zeros(n)
    b[:w] = 1.0
    b[-w:] = 1.0
    inits.append(('bimodal', b / b.sum()))

    r = rng.dirichlet(np.ones(n) * 0.4)
    inits.append(('random', r))

    alt = np.ones(n)
    alt[::2] = 1.4
    alt[1::2] = 0.6
    inits.append(('alt', alt / alt.sum()))

    return inits

                    best_sequence = a_opt.copy()
                    print(f"  SG{cfg_i}: *NEW BEST* {score:.8f}")
                else:
                    print(f"  SG{cfg_i}: {score:.8f}")

    print(f"Phase 3 done ({time.monotonic()-t3:.0f}s), best={best_score:.8f}")

    # === Phase 4: Coordinate polish ===
    print("=== Phase 4: Coordinate Polish ===")
    t4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            t_cd = min(remaining * 0.5, 100.0)
            a_opt, obj = coordinate_polish(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence([float(x) for x in a_opt]))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: *NEW BEST* {score:.8f}")
            else:
                print(f"  CoordDesc: {score:.8f}")

    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s), best={best_score:.8f}")

    # === Phase 5: Final micro-polish ===
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t5 = time.monotonic()
        t_final = min(25.0, deadline - time.monotonic() - 3.0)
        if t_final > 5:
            a_opt, obj = newton_optimize(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence([float(x) for x in a_opt]))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final: *NEW BEST* {score:.8f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence([float(x) for x in best_sequence]))

    # Final validation
    final_list = [float(x) for x in best_sequence]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return final_list
