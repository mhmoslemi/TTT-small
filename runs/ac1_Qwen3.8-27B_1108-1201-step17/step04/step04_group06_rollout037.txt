"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization on a 
log-sum-exp smoothed objective with softmax parameterization, combined with
exponentiated gradient (multiplicative weights) method, and local search polishing.
Key innovation: second-order optimization (L-BFGS) on the smooth surrogate,
enabling much faster convergence than first-order EG methods.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad_smooth(z, beta, n):
    """
    Compute objective value and gradient for f_beta(z) = (1/beta)*log(sum_k exp(beta*c_k))
    where a = softmax(z) and c = conv(a,a).
    
    Returns (objective, gradient).
    """
    # Softmax
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Convolution
    c = fast_convolve(a)
    
    # Log-sum-exp: (1/beta) * log(sum_k exp(beta * c_k))
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    log_sum = bc_max + np.log(exp_bc.sum())
    obj = log_sum / beta
    
    # Weights
    w = exp_bc / exp_bc.sum()
    
    # Gradient: grad_z[i] = 2 * a[i] * (corr[i] - wavg_c)
    # where corr[i] = sum_j a[j] * w[i+j] and wavg_c = sum_k w[k] * c[k]
    
    # Compute corr using vectorized approach
    # w has length 2n-1, a has length n
    # corr[i] = dot(a, w[i:i+n]) for i = 0..n-1
    
    # Use as_strided for efficient Hankel matrix construction
    from numpy.lib.stride_tricks import as_strided
    w_view = as_strided(w, shape=(n, n), strides=(w.strides[0], w.strides[0]))
    corr = w_view.T @ a  # corr[i] = sum_j w[j+i] * a[j]
    
    wavg_c = np.dot(w, c)
    
    grad_z = 2.0 * a * (corr - wavg_c)
    
    return obj, grad_z

def lbfgs_optimize(a_init, beta_start=50.0, beta_end=2000.0, 
                   n_betas=10, maxiter_per_beta=500, time_limit=60.0):
    """
    Run L-BFGS with annealing temperature schedule.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initial z from a: z = log(a)
    z = np.log(a + 1e-300)
    
    t0 = time.monotonic()
    best_obj = np.inf
    best_z = z.copy()
    
    # Create beta schedule
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for b in betas:
        if time.monotonic() - t0 > time_limit:
            break
        
        # L-BFGS minimization at this beta
        result = scipy_minimize(
            compute_grad_smooth,
            z,
            args=(b, n),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter_per_beta, 'ftol': 1e-15, 'gtol': 1e-12,
                     'maxls': 50, 'iprint': -1}
        )
        
        z = result.x
        obj, _ = compute_grad_smooth(z, b, n)
        
        # Extract a
        z_shifted = z - z.max()
        a_new = np.exp(z_shifted)
        a_new = a_new / a_new.sum()
        
        # Track best (using actual max_conv as the true objective)
        c_new = fast_convolve(a_new)
        true_obj = c_new.max()
        
        if true_obj < best_obj:
            best_obj = true_obj
            best_z = z.copy()
    
    # Final extraction
    z_shifted = best_z - best_z.max()
    a_best = np.exp(z_shifted)
    a_best = a_best / a_best.sum()
    
    return a_best, best_obj

def exp_gradient_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                          temp_decay=0.9997, eta_decay=0.99995,
                          top_k_frac=0.25, verbose=False):
    """Exponentiated gradient / multiplicative weights on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)

    top_k = max(10, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax-weighted subgradient
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
            peek = c[indices]
        else:
            indices = np.arange(len(c))
            peek = c.copy()
        
        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
        
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        
        if it % 1000 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
        
        eta *= eta_decay
        temp *= temp_decay
        
        if no_improve > 3000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.1
            temp = max(temp, temp0 * 0.01)
            avg_a = a.copy()
            count = 0
    
    return best_a.copy(), best_obj

def local_search_polish(a0, time_limit=30.0, max_rounds=200):
    """Local search with multiplicative perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False
        
        n_perturb = max(1, n // 16)
        coords = rng.choice(n, size=n_perturb, replace=False)
        
        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            for factor in [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_a.copy(), best_obj

def generate_inits(rng, n, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
        for scale in [0.001, 0.005, 0.02, 0.05]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine/Hamming window
    hamming = np.hamming(n)
    inits.append(hamming / hamming.sum())
    
    # Hann window
    hann = np.hanning(n)
    inits.append(hann / hann.sum())
    
    # Edge-concentrated
    edge = 1.0 / (np.minimum(x, n-1-x) + 1.0)
    inits.append(edge / edge.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.2*n))**2)
    inits.append(g / g.sum())
    
    # Exponential decay
    for beta in [0.5, 1.0, 2.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n-1-x) / n)
        inits.append(e2 / e2.sum())
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Plateau with raised edges
    t = x / (n - 1)
    plat = np.where(t < 0.2, 1.0 + 0.5*(0.2-t)/0.2, 
                    np.where(t > 0.8, 1.0 + 0.5*(t-0.8)/0.2, 1.0))
    inits.append(plat / plat.sum())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS second-order optimization
on a log-sum-exp smoothed objective with softmax parameterization, combined with
exponentiated gradient (multiplicative weights), multi-initialization, and
multi-length exploration. L-BFGS converges much faster than first-order EG
methods near the optimum, enabling deeper optimization within the time budget.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize
from numpy.lib.stride_tricks import as_strided


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_obj_grad(z, beta, n):
    """
    Objective and gradient for f_beta(z) = (1/beta)*log(sum_k exp(beta*c_k))
    where a = softmax(z), c = conv(a,a).
    Returns (objective_value, gradient_vector).
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    c = fast_convolve(a)

    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    log_sum = bc_max + np.log(exp_bc.sum())
    obj = log_sum / beta

    w = exp_bc / exp_bc.sum()

    # corr[i] = sum_j a[j] * w[i+j] for i=0..n-1
    w_contig = np.ascontiguousarray(w)
    M = as_strided(w_contig, shape=(n, n), strides=(w_contig.strides[0], w_contig.strides[0]))
    corr = M.T @ a

    wavg_c = np.dot(w, c)

    grad_z = 2.0 * a * (corr - wavg_c)

    return obj, grad_z


def lbfgs_optimize(a_init, beta_start=30.0, beta_end=2000.0,
                   n_betas=10, maxiter_per_beta=300, time_limit=60.0):
    """L-BFGS with annealing temperature schedule on softmax parameterization."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(a + 1e-300)
    t0 = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()

    betas = np.geomspace(beta_start, beta_end, n_betas)

    for bi, b in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break

        try:
            result = scipy_minimize(
                compute_smooth_obj_grad,
                z,
                args=(b, n),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxls': 50, 'iprint': 0}
            )
            z = result.x
        except Exception:
            break

        # Re-center z to prevent drift
        z = z - z.mean()

        # Extract a and compute true objective
        z_shifted = z - z.max()
        a_new = np.exp(z_shifted)
        a_new = a_new / a_new.sum()
        c_new = fast_convolve(a_new)
        true_obj = c_new.max()

        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a_new.copy()

    return best_a, best_obj


def exp_gradient_optimize(a0, time_limit, eta0=1.0, temp0=0.005,

        avg_a = (count * avg_a + a_new) / (count + 1)

            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        c = c_new

        if it % 500 == 0:

        best_a = avg_a.copy()

def local_search_polish(a0, time_limit=30.0, max_rounds=300, seed=42):
    """Local search with multiplicative perturbations on random coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    factors = [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5,
               0.75, 1.0/0.75, 0.9, 1.0/0.9, 0.6, 1.0/0.6]

    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False

        n_perturb = max(2, n // 8)
        coords = rng.choice(n, size=min(n_perturb, n), replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                a[i] = 0.01 / n
                s2 = a.sum()
                a = a / s2
                continue

            for factor in factors:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] = old_val

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a.copy(), best_obj


def spectral_shaping_init(n, rng):
    """
    Generate initialization via spectral shaping:
    Choose a target power spectrum that minimizes peak of inverse FFT.
    """
    fft_size = 1 << (2 * n - 1).bit_length()

    # Strategy: create a spectrum with relatively flat magnitude
    # but with slight lift at high frequencies to reduce concentration
    freqs = np.fft.rfftfreq(fft_size)

    # Flat spectrum
    target_spec = np.ones(fft_size // 2 + 1)

    # Slightly boost high frequencies
    high_mask = freqs > 0.3
    target_spec[high_mask] = 1.2

    # Random phase
    phase = rng.standard_normal(fft_size // 2 + 1)

    # Create complex spectrum
    spectrum = target_spec * np.exp(1j * phase)
    spectrum[0] = target_spec[0]  # DC component real

    # Inverse FFT
    a_full = np.fft.irfft(spectrum * fft_size, fft_size)

    # Take first n elements and make positive
    a = np.maximum(a_full[:n], 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    return a


def generate_inits(rng, n, parent=None):
    """Generate diverse initializations for a given length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
        for scale in [0.001, 0.005, 0.02, 0.05, 0.1]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Hamming window

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS second-order optimization
on a log-sum-exp smoothed objective with softmax parameterization, combined with
exponentiated gradient (multiplicative weights), multi-initialization, and
multi-length exploration. L-BFGS converges much faster than first-order EG
methods near the optimum, enabling deeper optimization within the time budget.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize
from numpy.lib.stride_tricks import as_strided


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_obj_grad(z, beta, n):
    """
    Objective and gradient for f_beta(z) = (1/beta)*log(sum_k exp(beta*c_k))
    where a = softmax(z), c = conv(a,a).
    Returns (objective_value, gradient_vector).
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    c = fast_convolve(a)

    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    log_sum = bc_max + np.log(exp_bc.sum())
    obj = log_sum / beta

    w = exp_bc / exp_bc.sum()

    # corr[i] = sum_j a[j] * w[i+j] for i=0..n-1
    w_contig = np.ascontiguousarray(w)
    M = as_strided(w_contig, shape=(n, n), strides=(w_contig.strides[0], w_contig.strides[0]))
    corr = M.T @ a

    wavg_c = np.dot(w, c)

    grad_z = 2.0 * a * (corr - wavg_c)

    return obj, grad_z


def lbfgs_optimize(a_init, beta_start=30.0, beta_end=2000.0,
                   n_betas=10, maxiter_per_beta=300, time_limit=60.0):
    """L-BFGS with annealing temperature schedule on softmax parameterization."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(a + 1e-300)
    t0 = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()

    betas = np.geomspace(beta_start, beta_end, n_betas)

    for bi, b in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break

        try:
            result = scipy_minimize(
                compute_smooth_obj_grad,
                z,
                args=(b, n),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxls': 50, 'iprint': 0}
            )
            z = result.x
        except Exception:
            break

        # Re-center z to prevent drift
        z = z - z.mean()

        # Extract a and compute true objective
        z_shifted = z - z.max()
        a_new = np.exp(z_shifted)
        a_new = a_new / a_new.sum()
        c_new = fast_convolve(a_new)
        true_obj = c_new.max()

        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a_new.copy()

    return best_a, best_obj
