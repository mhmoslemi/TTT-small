def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute the objective value 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def coordinate_descent_polish(a0, time_limit, max_rounds=1000):
    """
    Coordinate descent polishing: try small multiplicative changes on individual
    or pairs of coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0
    no_improve = 0
    
    # Factors to try
    factors = [0.8, 0.9, 0.95, 1.05, 1.1, 1.25]
    
    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False
        
        # Try pairs of coordinates (i, n-1-i) to exploit anti-symmetry
        n_pairs = n // 2
        for _ in range(max(5, n_pairs // 10)):
            i = rng.integers(0, n_pairs)
            j = n - 1 - i
            
            old_a, old_b = a[i], a[j]
            if old_a < 1e-15 and old_b < 1e-15:
                continue
            
            # Try scaling one up and the other down (preserves sum approximately)
            for f in factors:
                # Move weight from j to i
                delta = old_b * (1 - 1.0/f) / f  # approximate
                a[i] = old_a + old_b * (f - 1.0) * 0.1
                a[j] = old_b - old_a * (f - 1.0) * 0.1
                if a[i] < 0 or a[j] < 0:
                    a[i], a[j] = old_a, old_b
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i], a[j] = old_a, old_b
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-14:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break
            
            if improved:
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_a.copy(), best_obj

def generate_diverse_inits(n, best_a, rng):
    """Generate diverse initializations exploiting the anti-symmetric structure."""
    inits = []

        mu = center + mu_offset * n

    for frac in [0.3, 0.4, 0.6, 0.7]:

        b = np.maximum(b, 0)

        if bs > 1e-10:

    if best_a is not None and len(best_a) == n:

            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:

        for scale in [0.005, 0.01, 0.02, 0.05]:

            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    for pos_frac in [0.3, 0.4, 0.6, 0.7]:
        pos = int(pos_frac * n)
        width = max(3, n // 20)

        h = np.maximum(h, 0)
        hs = h.sum()
        if hs > 1e-10:

    for alpha in [0.3, 0.5, 0.7]:
        r = 1.0 + alpha * (x / n - 0.5) * 2

    for _ in range(5):

        inits.append(r)
    
    # 8. Best itself
    if best_a is not None and len(best_a) == n:
        inits.insert(0, best_a.copy())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an adaptive exponentiated gradient
method with sharp single-peak subgradients, anti-symmetric structural exploitation,
backtracking line search, and multi-phase optimization. Key insight: optimal sequences
break left-right symmetry to reduce the middle convolution peak by concentrating mass
asymmetrically (a "skewed bump" shape).

import numpy as np
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient_single(a, c, k_peak):
    """Compute subgradient for a single peak lag."""
    n = len(a)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    sg = np.zeros(n)
    sg[js:je + 1] = 2.0 * a[k_peak - je:k_peak - js + 1][::-1]
    return sg


def compute_subgradient_weighted(a, c, num_peaks=3, sharpness=15.0):
    """Compute weighted subgradient over top peaks with sharp weighting."""
    n = len(a)
    len_c = len(c)
    
    if num_peaks < len_c:
        indices = np.argpartition(c, -num_peaks)[-num_peaks:]
    else:
        indices = np.arange(len_c)
    
    order = np.argsort(c[indices])[::-1]
    indices = indices[order]
    
    peak_vals = c[indices]
    max_val = peak_vals[0]
    w = (peak_vals / max_val) ** sharpness
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(indices))
        w[0] = 1.0
    else:
        w = w / w_sum
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    
    return sg, indices


def exp_gradient_core(a0, time_limit, eta0=2.0, num_peaks=1, sharpness=15.0,
                      verbose=False, seed=42, kick_interval=400, kick_scale=0.015):
    """
    Core exponentiated gradient with:
    - Sharp subgradient focusing on binding constraints
    - Adaptive step size (increase on progress, decrease on stall)
    - Anti-symmetric periodic kicks
    - Running average tracking
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    avg_a = a.copy().astype(np.float64)
    count = 1
    
    eta = eta0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    last_kick = 0
    prev_obj = best_obj
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Anti-symmetric kick to escape local minima
        if it - last_kick >= kick_interval:
            d = rng.standard_normal(n)
            d = d - d[::-1]
            d = d - d.mean()
            dn = np.sqrt((d ** 2).sum())
            if dn > 1e-10:
                a = a + kick_scale * d / (dn * np.sqrt(n))
                a = np.maximum(a, 1e-300)
                a = a / a.sum()
                c = fast_convolve(a)
            last_kick = it
            no_improve = 0
            avg_a = a.copy()
            count = 1
            eta = max(eta, eta0 * 0.3)
            continue
        
        # Compute subgradient
        if num_peaks == 1:
            k_peak = int(np.argmax(c))
            sg = compute_subgradient_single(a, c, k_peak)
        else:
            sg, _ = compute_subgradient_weighted(a, c, num_peaks=num_peaks, sharpness=sharpness)
        
        sg_norm = np.sqrt((sg ** 2).sum())
        if sg_norm < 1e-15:
            break
        
        # Backtracking line search
        alpha = eta
        accepted = False
        for _ in range(15):
            log_update = -alpha * sg / sg_norm
            log_update -= log_update.max()
            a_new = a * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-300)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
                s_new = 1.0
            a_new = a_new / s_new
            
            c_new = fast_convolve(a_new)
            obj_new = c_new.max()
            
            if obj_new <= prev_obj - 1e-15:
                accepted = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
        
        if not accepted or alpha < 1e-12:
            a_new = a.copy()
            c_new = c.copy()
            obj_new = prev_obj
        
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            # Increase step size on progress (adaptive)
            if alpha > eta * 0.3:
                eta = min(eta * 1.05, eta0 * 2.0)
            if verbose and it % 500 == 0:
                print(f"    iter {it}: obj={best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1
            # Decrease step size on stall
            eta = max(eta * 0.999, eta0 * 0.01)
        
        a = a_new
        c = c_new
        prev_obj = obj_new
        
        # Check running average
        if it % 300 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-14:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
        
        if no_improve > 3000:
            # Full restart with perturbation
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 1e-300)
            a = a / a.sum()
            c = fast_convolve(a)
            prev_obj = c.max()
            no_improve = 0
            eta = eta0
            avg_a = a.copy()
            count = 1
        
        # Gradual eta decay
        eta *= 0.99995
    
    return best_a.copy(), best_obj


def polish_coordinate(a0, time_limit, max_rounds=2000):
    """
    Coordinate descent: try multiplicative perturbations on anti-symmetric pairs
    (i, n-1-i) and individual coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    it = 0
    no_improve = 0
    
    factors = [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]
    
    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False
        
        # Try anti-symmetric pair updates
        n_attempts = max(10, n // 8)
        for _ in range(n_attempts):
            i = rng.integers(0, n)
            j = n - 1 - i
            if i == j:
                # Single coordinate update
                old = a[i]
                if old < 1e-15:
                    continue
                for f in factors:
                    a[i] = old * f
                    s2 = a.sum()
                    if s2 < 1e-15:
                        a[i] = old
                        continue
                    a = a / s2
                    c2 = fast_convolve(a)
                    obj2 = c2.max()
                    if obj2 < best_obj - 1e-14:
                        best_obj = obj2
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a = best_a.copy()
                        break
            else:
                old_i, old_j = a[i], a[j]
                if old_i < 1e-15 and old_j < 1e-15:
                    continue
                # Try shifting weight between the pair
                for f in [0.7, 0.85, 1.0/0.85, 1.0/0.7]:
                    # Move weight: a[i] *= f, a[j] adjusted to keep sum
                    new_i = old_i * f
                    new_j = old_j + (old_i - new_i)
                    if new_i < 0 or new_j < 0:
                        continue
                    a[i] = new_i
                    a[j] = new_j
                    s2 = a.sum()
                    if s2 < 1e-15:
                        a = best_a.copy()
                        continue
                    a = a / s2
                    c2 = fast_convolve(a)
                    obj2 = c2.max()
                    if obj2 < best_obj - 1e-14:
                        best_obj = obj2
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a = best_a.copy()
                        break
            
            if improved:
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break
    
    return best_a.copy(), best_obj


def generate_inits(n, best_a, rng):
    """Generate diverse initializations with anti-symmetric structure."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Skewed Gaussians
    for mu_frac in [0.35, 0.4, 0.45, 0.55, 0.6, 0.65]:
        for sigma_frac in [0.15, 0.2, 0.25, 0.3]:
            mu = mu_frac * n
            sigma = sigma_frac * n
            g = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            gs = g.sum()
            if gs > 1e-10:
                inits.append(g / gs)
    
    # "Hammer" - concentrated bumps
    for pos_frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width_frac in [0.05, 0.1, 0.15]:
            pos = pos_frac * n
            width = max(2.0, width_frac * n)
            h = np.exp(-0.5 * ((x - pos) / width) ** 2)
            hs = h.sum()
            if hs > 1e-10:
                inits.append(h / hs)
    
    # Linear ramps (strong anti-symmetry)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        r = 1.0 + alpha * (x / (n - 1) - 0.5) * 2
        r = np.maximum(r, 0.01)
        inits.append(r / r.sum())
    
    # Cosine modulated
    for freq in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.5]:
            for phase in [0.0, np.pi / 2, np.pi]:
                c1 = 1.0 + amp * np.cos(2 * np.pi * freq * x / n + phase)
                c1 = np.maximum(c1, 0.01)
                inits.append(c1 / c1.sum())
    
    # Perturbed best with anti-symmetric structure
    if best_a is not None and len(best_a) == n:
        inits.insert(0, best_a.copy())
        for scale in [0.005, 0.01, 0.02, 0.03, 0.05]:
            d = rng.standard_normal(n)
            d = d - d[::-1]
            d = d - d.mean()
            denom = np.sqrt((d ** 2).sum())
            if denom > 1e-10:
                p = best_a + scale * d / (denom * np.sqrt(n))
                p = np.maximum(p, 0)
                ps = p.sum()
                if ps > 1e-10:
                    inits.append(p / ps)
        
        # Random perturbation of best
        for scale in [0.002, 0.005, 0.01]:
            p = best_a + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)
    
    # Random Dirichlet
    for alpha_d in [0.1, 0.3, 1.0]:
        for _ in range(3):
            r = rng.dirichlet(np.ones(n) * alpha_d)
            inits.append(r)
    
    # "Edge bump" - mass concentrated near edges asymmetrically
    for edge_frac in [0.1, 0.2, 0.3]:
        for side in [0, 1]:
            if side == 0:
                e = np.exp(-x / (edge_frac * n))
            else:
                e = np.exp(-(n - x) / (edge_frac * n))
            e = np.maximum(e, 0.001)
            inits.append(e / e.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function: adaptive exponentiated gradient with sharp subgradients,
    anti-symmetric structural exploitation, diverse initializations, and multi-phase
    optimization with polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent
    n = 384
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            n = len(p)
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={n}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(n) / n
        print(f"No parent, using uniform n={n}")
        best_score = evaluate_sequence(parent.tolist())
        best_seq = parent.copy()
    
    print(f"Start: n={n}, score={best_score:.8f}")
    
    # === Phase 1: Long EG on parent ===
    print("\n=== Phase 1: Primary EG (parent) ===")
    rem = deadline - time.monotonic()
    t1 = min(500, rem * 0.35)
    if t1 > 60:
        print(f"  t={t1:.0f}s")
        a_opt, _ = exp_gradient_core(parent, time_limit=t1, eta0=2.5,
                                      num_peaks=1, sharpness=15.0,
                                      verbose=True, seed=1, kick_interval=300,
                                      kick_scale=0.02)
        try_seq(a_opt, "p1_main")
    
    # === Phase 2: EG variations from current best ===
    print(f"\n=== Phase 2: EG Variations (best={best_score:.8f}) ===")
    
    configs = [
        {"eta0": 1.0, "num_peaks": 3, "sharpness": 8.0, "kick_interval": 500, "kick_scale": 0.01, "seed": 10},
        {"eta0": 4.0, "num_peaks": 1, "sharpness": 15.0, "kick_interval": 200, "kick_scale": 0.03, "seed": 11},

    for ci, cfg in enumerate(configs):

        print(f"  cfg{ci}: t={t_alloc:.0f}s")

    print(f"\n=== Phase 3: Diverse Inits (best={best_score:.8f}) ===")

    # Score inits by their convolution max (lower is better)
    scored = []
    for i, f_init in enumerate(inits):
        c_i = fast_convolve(f_init)
        scored.append((c_i.max(), i, f_init))

    top_inits = scored[:15]

            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(x_new, x_old, best_seq)
            a_alt = np.maximum(a_alt, 0)
            a_alt = a_alt / a_alt.sum()
            print(f"  n_alt={n_alt}: t={t_alloc:.0f}s")

                                          num_peaks=1, sharpness=15.0,
                                          verbose=False, seed=300 + n_alt)

    # === Phase 7: Final fine-tune ===
    print(f"\n=== Phase 7: Final (best={best_score:.8f}) ===")
    
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:

        if rem > 5:
            a_opt, _ = polish_coordinate(best_seq, time_limit=min(rem * 0.8, 30), max_rounds=1500)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
