def fast_convolve(arr):
    """Computes the auto-convolution using FFT for speed."""
    n = len(arr)
    size = 2 * n - 1
    fft_size = 1 << (size-1).bit_length()
    X = np.fft.rfft(arr, fft_size)
    Y = np.fft.rfft(arr, fft_size)
    conv_result = np.fft.irfft(X * Y, fft_size)[:size]  # full convolution
    return conv_result
