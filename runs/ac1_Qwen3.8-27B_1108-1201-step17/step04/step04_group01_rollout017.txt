import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache):
    """Solve LP: max sum(g) s.t. f*g <= M (element-wise), g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def power_sum_gradient(f, p):
    """Compute gradient of (sum_k (conv(f,f)[k]^p))^(1/p) w.r.t. f.
    This equals 2*p^(1/p) * (sum_k (conv(f,f)[k]^p))^(1/p - 1) * conv(f,f)^(p-1) * cross_corr(...)
    Simplified: grad_i = 2 * sum_k w_k * f[k-i] where w_k = (conv(f,f)[k])^(p-1) / ||conv||_p^p
    """
    n = len(f)
    conv = fast_convolve(f)
    # Weight: w_k = conv[k]^(p-1) / sum(conv^p)
    if p > 1:
        conv_clipped = np.maximum(conv, 1e-30)
        wp1 = conv_clipped ** (p - 1)
        wp = conv_clipped ** p
        w = wp1 / wp.sum()
    else:
        w = np.ones(len(conv)) / len(conv)

    # grad[i] = 2 * sum_k w[k] * f[k-i] (where valid)
    # This is a cross-correlation of w and f
    f_flipped = f[::-1].copy()
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_result = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_result[n - 1:n - 1 + n]


def pnorm_gradient_descent(f0, time_limit, p_start=20, p_end=500, verbose=False):
    """Gradient descent with p-norm smoothing and Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    p = p_start
    lr = 3.0 / n
    momentum = 0.92
    y = f.copy()  # Nesterov momentum
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov lookahead
        y = f + momentum * v

        grad = power_sum_gradient(y, p)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                y = f.copy()
                lr = max(3.0 / n, 0.002)
                p = min(p * 1.3, p_end)
                momentum = 0.90
                no_improve = 0

        if p < p_end:
            p *= 1.004

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PNorm it={it}, obj={best_obj:.12f}, p={p:.0f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj

def lse_gradient_descent(f0, time_limit, beta_start=30, beta_end=10000, verbose=False):

    best_f = f.copy()

    lr = 2.5 / n
    momentum = 0.91
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        bc_max = bc.max()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. FW-LP with adaptive M-relaxation and accelerated line search
2. Power-sum (p-norm) gradient descent with Nesterov momentum (new smoothing)
3. LSE (log-sum-exp) gradient descent with adaptive beta
4. Targeted coordinate descent for peak reduction
5. Multi-restart with diverse initializations and length exploration
The objective is scale-invariant, so we work in the simplex (sum=1).

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix A where (A@g)[k] = (f*g)[k].
    Shape: (2n-1, n)
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 25,
                                                 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def compute_cross_corr_grad(w, f):
    """Compute gradient: 2 * sum_k w[k] * f[k-i] for each i.
    w has length 2n-1, f has length n.
    grad[i] = 2 * sum_{k} w[k] * f[k-i] (where 0<=k-i<n)
    This equals 2 * (w cross-correlated with f) at offset.
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def pnorm_gradient_descent(f0, time_limit, p_start=10, p_end=200, verbose=False):
    """Gradient descent using p-norm smoothing of the max.
    Objective approximation: (sum_k conv[k]^p)^(1/p) for large p approximates max.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    p = p_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    y = f.copy()  # Nesterov
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov lookahead
        y = f + momentum * v

        conv_y = fast_convolve(y)
        # p-norm weights
        conv_pos = np.maximum(conv_y, 1e-30)
        wp1 = conv_pos ** (p - 1)
        w_sum = wp1.sum()
        if w_sum < 1e-30:
            w = np.ones(len(conv_y)) / len(conv_y)
        else:
            w = wp1 / w_sum

        grad = compute_cross_corr_grad(w, y)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
                momentum = max(0.84, momentum - 0.004)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                y = f.copy()
                lr = max(3.0 / n, 0.003)
                p = min(p * 1.3, p_end)
                momentum = 0.90
                no_improve = 0

        if p < p_end:
            p *= 1.005

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PNorm it={it}, obj={best_obj:.12f}, p={p:.0f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lse_gradient_descent(f0, time_limit, beta_start=30, beta_end=10000, verbose=False):
    """Log-sum-exp gradient descent with adaptive beta and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.5 / n
    momentum = 0.91
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_sum = exp_bc.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_bc / w_sum

        grad = compute_cross_corr_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 12:
                lr *= 0.94
                momentum = max(0.84, momentum - 0.004)
            if no_improve > 90:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.5 / n, 0.002)
                beta = min(beta * 1.5, beta_end)
                momentum = 0.89
                no_improve = 0

        if beta < beta_end:
            beta *= 1.006

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def targeted_coord_descent(f0, time_limit, verbose=False):
    """Targeted coordinate descent focusing on reducing peak autocorrelation.
    For each peak position, identify contributing pairs and redistribute mass.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()
        improved_this_round = False

        # Find peak positions (top few)
        n_peaks = min(5, len(conv))
        peak_threshold = M * 0.90
        peak_positions = np.where(conv >= peak_threshold)[0]

        if len(peak_positions) == 0:
            break

        # For each peak, find the most contributing coordinate
        # and try to reduce it by transferring mass to a low-contributing coordinate
        for k in peak_positions:
            # Coordinates contributing to conv[k]:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            # Contribution of f[j] to conv[k] is f[j] * f[k-j]
            contributions = f[js] * f[k - js]
            total_contrib = contributions.sum()
            if total_contrib < 1e-15:
                continue

            # Find the coordinate with highest contribution
            max_idx_local = np.argmax(contribuations)
            i_hot = js[max_idx_local]
            partner = k - i_hot

            # Find a "cold" coordinate: one that contributes little to any peak
            # and is far from peak positions
            # Simplified: find coordinate with smallest total contribution to all peaks
            cold_contrib = np.zeros(n)
            for kp in peak_positions:
                js2 = np.arange(max(0, kp - n + 1), min(n - 1, kp) + 1)
                cold_contrib[js2] += f[kp - js2]
            cold_contrib[i_hot] = np.inf  # Exclude the hot coordinate
            # Also exclude the partner to avoid creating new peaks
            if 0 <= partner < n:
                cold_contrib[partner] = np.inf

            i_cold = np.argmin(cold_contrib)
            if cold_contrib[i_cold] > 1e-10:
                continue  # No good cold coordinate found

            # Transfer a small amount of mass from i_hot to i_cold
            # Try several transfer amounts
            best_delta = 0
            best_M = M
            for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
                delta = f[i_hot] * frac
                if delta < 1e-10:
                    continue
                f_try = f.copy()
                f_try[i_hot] -= delta
                f_try[i_cold] += delta
                f_try = np.maximum(f_try, 0)
                st = f_try.sum()
                if st < 1e-15:
                    continue
                f_try = f_try / st
                conv_try = fast_convolve(f_try)
                M_try = conv_try.max()
                if M_try < best_M - 1e-14:
                    best_M = M_try
                    best_delta = delta
                    best_f_try = f_try.copy()

            if best_delta > 0:
                f = best_f_try
                conv = fast_convolve(f)
                M = conv.max()
                improved_this_round = True
                if M < best_obj - 1e-14:
                    best_obj = M
                    best_f = f.copy()
                    no_improve = 0

        if not improved_this_round:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent it={it}, obj={best_obj:.12f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining diverse strategies:
    Phase 1: FW-LP for global structure (400s)
    Phase 2: P-norm GD with Nesterov momentum (350s) - different smoothing
    Phase 3: FW-LP refinement (250s)
    Phase 4: LSE polishing (200s)
    Phase 5: Targeted coordinate descent (100s)
    Phase 6: Alternative explorations (remaining time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 50.0)

    best_tracker = [float('inf'), None]

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent

    # === Phase 1: FW-LP on parent ===
    print("\n=== Phase 1: FW-LP on parent ===")
    rem = deadline - time.monotonic()
    t_alloc = min(400, rem * 0.24)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best

    # === Phase 2: P-norm gradient descent (different smoothing) ===
    print(f"\n=== Phase 2: P-norm Gradient Descent ===")
    rem = deadline - time.monotonic()
    t_alloc = min(350, rem * 0.22)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = pnorm_gradient_descent(current_best, time_limit=t_alloc,
                                            p_start=10, p_end=200)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best

    # === Phase 3: FW-LP refinement ===
    print(f"\n=== Phase 3: FW-LP Refinement ===")
    rem = deadline - time.monotonic()
    t_alloc = min(250, rem * 0.18)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best

    # === Phase 4: LSE polishing ===
    print(f"\n=== Phase 4: LSE Polishing ===")
    rem = deadline - time.monotonic()
    t_alloc = min(200, rem * 0.15)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lse_gradient_descent(current_best, time_limit=t_alloc,
                                          beta_start=40, beta_end=8000)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best

    # === Phase 5: Targeted coordinate descent ===
    print(f"\n=== Phase 5: Targeted Coordinate Descent ===")
    rem = deadline - time.monotonic()
    t_alloc = min(120, rem * 0.10)
    if t_alloc > 15:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = targeted_coord_descent(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best

    # === Phase 6: Alternative explorations ===
    print(f"\n=== Phase 6: Alternative Explorations ===")
    rem = deadline - time.monotonic()

    explorations = []

    # Different lengths (interpolate best)
    for n_alt in [384, 512, 320, 576]:
        if n_alt == n_parent:
            continue
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()
        explorations.append(f_interp)

    # Structured initializations
    n_struct = n_parent
    x = np.linspace(0, 1, n_struct)

    # Sine-based (smooth)
    sine_init = np.sin(np.pi * x) ** 1.5
    sine_init = np.maximum(sine_init, 0)
    if sine_init.sum() > 1e-10:
        sine_init = sine_init / sine_init.sum()
    explorations.append(sine_init)

    # Flatter top
    flat_top = np.where((x > 0.15) & (x < 0.85), 1.0, 0.4 * np.cos(np.pi * (x - 0.5)) ** 2)
    flat_top = np.maximum(flat_top, 0)
    if flat_top.sum() > 1e-10:
        flat_top = flat_top / flat_top.sum()
    explorations.append(flat_top)

    for exp in explorations:
        rem = deadline - time.monotonic() - 30
        if rem < 40:
            break
        # Quick FW-LP + LSE
        t1 = min(rem * 0.45, 70)
        t2 = min(rem * 0.35, 50)

        try_sequence(exp, best_tracker)
        a_opt, _ = fw_lp_optimize(exp, time_limit=t1)
        try_sequence(a_opt, best_tracker)

        cur = best_tracker[1].copy() if best_tracker[1] is not None else exp
        if len(cur) == len(exp):
            a_opt, _ = lse_gradient_descent(cur, time_limit=t2,
                                             beta_start=30, beta_end=5000)
            try_sequence(a_opt, best_tracker)

    # === Final polish: one more round on the best ===
    print(f"\n=== Final Polish ===")
    rem = deadline - time.monotonic() - 10
    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent
    if rem > 30:
        # Alternate between FW-LP and LSE
        t_fw = rem * 0.5
        t_lse = rem * 0.5
        a_opt, _ = fw_lp_optimize(current_best, time_limit=t_fw)
        try_sequence(a_opt, best_tracker)

        current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best
        a_opt, _ = lse_gradient_descent(current_best, time_limit=t_lse,
                                        beta_start=100, beta_end=12000)
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.8f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    final_seq = np.ones(448) / 448
    score = float(evaluate_sequence(final_seq.tolist()))
    print(f"\nFALLBACK: score={score:.8f}")
    return [float(x) for x in final_seq]
