"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp (LSE) smoothed objective with adaptive beta schedule
2. Momentum-based projected gradient with backtracking line search
3. Population-based iterated local search with crossover diversity
4. Pairwise coordinate descent for final polish
Key innovation: LSE smoothing provides dense gradient signal (all peaks weighted),
adaptive beta annealing transitions from global to local optimization,
and population crossover maintains diversity to escape local minima.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_value_and_grad(a, beta):
    """Compute LSE-smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    shifted = beta * (conv - m)
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w /= w.sum()
    val = m + (max_shifted + np.log(w.sum())) / beta

    # grad[j] = 2 * sum_m a[m] * w[m+j]
    # Use FFT: conv(rev_a, w)[n-1+j] = sum_m a[m] * w[m+j]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    rev_a = a[::-1]
    FA = np.fft.rfft(rev_a, fft_size)
    FW = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(FA * FW, fft_size)
    grad = 2.0 * conv_result[n - 1: n - 1 + n]
    return val, grad


def true_objective(a):
    """Compute the true (unsmoothed) objective: max of conv(a,a) for normalized a."""
    s = a.sum()
    if s < 1e-15:
        return 1e10
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return conv.max()


def momentum_lse_optimize(f0, time_limit, beta_start=3.0, beta_end=200.0,

    """

    f = np.maximum(f0.astype(np.float64), 0.0)

    n = len(f)

    best_obj = true_objective(f)

    cur_lr = lr_init
    no_improve = 0
    it = 0

        f = f_new

            no_improve = 0

            no_improve = 0

def subgradient_peak_optimize(f0, time_limit, lr_init=0.005, momentum=0.9,
                              top_k=20, seed=42):
    """Subgradient method focusing on top-k peaks with softmax weighting."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = true_objective(f)
    best_f = f.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    cur_lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        # Select top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Softmax weights (sharper for higher peaks)
        pv = conv[peaks]
        scale = max(1e-10, (M - pv.min()) * 5 + 1e-6)
        weights = np.exp((pv - M) / scale)
        weights /= weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * weights[i] * f[k - idx]

        # Momentum update
        v = momentum * v - cur_lr * grad
        f_new = project_simplex(f + v)

        obj = true_objective(f_new)
        f = f_new
        it += 1

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997

        if no_improve > 400:
            scale = 0.002 + 0.004 * min(1.0, no_improve / 800.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, seed=42, sweep_size=4):
    """
    Pairwise coordinate descent: for pairs (i,j), optimally redistribute mass.
    Uses ternary search on the 1D subproblem.
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        return f, 1e10
    f = f / s

    best_obj = true_objective(f)
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    def obj_with_move(f, i, j, t):
        """Objective when we add t to f[i] and subtract t from f[j]."""
        f2 = f.copy()
        f2[i] += t
        f2[j] -= t
        if f2[i] < 0 or f2[j] < 0:
            return 1e10
        return true_objective(f2)

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random pairs
        n_pairs = min(50, n * sweep_size)
        for _ in range(n_pairs):
            if time.monotonic() - t0 >= time_limit:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if f[i] < 1e-10 or f[j] < 1e-10:
                continue

            # Ternary search for optimal t in [-f[j]*0.9, f[i]*0.9]
            lo = -f[j] * 0.9
            hi = f[i] * 0.9
            if hi - lo < 1e-12:
                continue

            for _ in range(20):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                o1 = obj_with_move(f, i, j, m1)
                o2 = obj_with_move(f, i, j, m2)
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1

                if time.monotonic() - t0 >= time_limit:
                    break

            t_opt = (lo + hi) / 2.0
            f_new = f.copy()
            f_new[i] += t_opt
            f_new[j] -= t_opt
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()

            obj_new = true_objective(f_new)
            if obj_new < best_obj - 1e-11:
                f = f_new
                best_obj = obj_new
                best_f = f.copy()
                improved = True
            it += 1

        if not improved and it > 100:
            # Try larger perturbations
            scale = 0.01
            f_cand = project_simplex(best_f + rng.standard_normal(n) * scale)
            obj_cand = true_objective(f_cand)
            if obj_cand < best_obj:
                f = f_cand
                best_f = f.copy()
                best_obj = obj_cand

    return best_f, best_obj

def interpolate_sequence(src, n_target):
    """Interpolate a sequence to a different length."""
    n_src = len(src)
    if n_src == n_target:
        return src.astype(np.float64)
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    result = np.interp(x_tgt, x_src, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    else:
        result = np.ones(n_target) / n_target
    return result

                    lr_init=0.003, momentum=0.9, top_k=15, seed=seed + n_try + 1
                )
                try_seq(a_opt2, f"sg_len_{n_try}")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Coordinate descent polish (~250s) ===
    print("\n=== Phase 5: Coordinate descent ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 250
        t5 = max(20, min(rem, 250))
        print(f"  CD, t={t5:.0f}s")
        a_opt, _ = coordinate_descent_polish(
            best_seq, time_limit=t5, seed=seed + 100, sweep_size=6
        )
        try_seq(a_opt, "cd")

        # CD followed by short LSE
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            a_opt2, _ = momentum_lse_optimize(
                a_opt, time_limit=min(rem, 60),
                beta_start=50.0, beta_end=300.0,
                lr_init=0.002, momentum=0.9, seed=seed + 110
            )
            try_seq(a_opt2, "cd_lse")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Final polish (~200s remaining) ===
    print("\n=== Phase 6: Final polish ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple short LSE runs with different params
        final_configs = [
            {'beta_start': 10, 'beta_end': 200, 'lr': 0.003, 'mom': 0.88},
            {'beta_start': 30, 'beta_end': 400, 'lr': 0.0015, 'mom': 0.92},
            {'beta_start': 5, 'beta_end': 100, 'lr': 0.005, 'mom': 0.82},
        ]

        rem = deadline - time.monotonic() - 60
        t_per = max(10, rem / len(final_configs))

        for ci, cfg in enumerate(final_configs):
            rem = deadline - time.monotonic() - 40
            if rem < 15:
                break
            t_alloc = min(t_per, rem)
            a_opt, _ = momentum_lse_optimize(
                best_seq, time_limit=t_alloc,
                beta_start=cfg['beta_start'], beta_end=cfg['beta_end'],
                lr_init=cfg['lr'], momentum=cfg['mom'], seed=seed + 200 + ci
            )
            try_seq(a_opt, f"final_lse_{ci}")

        # SG with very small steps
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            a_opt, _ = subgradient_peak_optimize(
                best_seq, time_limit=min(rem, 50),
                lr_init=0.001, momentum=0.94, top_k=10, seed=seed + 300
            )
            try_seq(a_opt, "final_sg")

        # Tiny CD
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = coordinate_descent_polish(
                best_seq, time_limit=min(rem, 30), seed=seed + 400, sweep_size=3
            )
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
