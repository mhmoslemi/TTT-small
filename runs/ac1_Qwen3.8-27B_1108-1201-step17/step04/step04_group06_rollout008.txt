    # Exponential (both directions)
    for alpha in [1.0, 3.0]:
        e1 = np.exp(-alpha * x / n)
        inits.append(e1 / e1.sum())
        e2 = np.exp(-alpha * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())
    
    # Edge-concentrated
    edge = 1.0 / (np.abs(x - mid) + 1.0)
    inits.append(edge / edge.sum())
    
    # Random (Dirichlet)
    for conc in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * conc)
        inits.append(r)
    
    # Fourier-domain: IFFT of low-pass spectrum
    for lp_order in [2, 4, 8]:
        freqs = np.fft.rfftfreq(n)
        magnitude = 1.0 / (1.0 + (freqs * n * 0.5) ** (2 * lp_order))
        magnitude /= magnitude.sum()
        # Take IFFT of the magnitude (treating it as a real spectrum)
        spec = magnitude
        a_fourier = np.fft.irfft(spec, n * 2)[:n]
        a_fourier = np.maximum(a_fourier, 0)
        s_f = a_fourier.sum()
        if s_f > 1e-10:
            inits.append(a_fourier / s_f)
    
    return inits
