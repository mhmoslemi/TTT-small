def fast_convolve(a):
    """Linear convolution a * a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f):
    """Build the sparse matrix C such that conv(f, b) = C @ b.
    C[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = np.arange(max(0, k - n + 1), min(n - 1, k) + 1)
        if len(js) > 0:
            rows.extend(np.full(len(js), k).tolist())
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0):
    """Log-sum-exp smoothed gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 3.0 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 200 == 0:
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, t={time.monotonic()-t0:.0f}s")
        
        conv = fast_convolve(f)
        
        # Softmax weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Gradient
        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(3.0 / n, 0.001)
                beta = min(beta * 1.5, beta_end * 2)
                no_improve = 0
        
        if beta < beta_end * 2:
            beta *= 1.004
    
    return best_f, best_obj

def coord_descent_polish(f0, time_limit):
    """Coordinate descent: for each coordinate, optimize it given others fixed."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Save current value
            old_val = f[i]
            
            # Compute conv contributions involving a[i]
            # conv[k] = 2*f[i]*f[k-i] + c[k] (for k where both indices valid)
            # or conv[k] = f[i]^2 + c[k] (when k = 2i)
            
            # c[k] = conv[k] - (terms involving f[i])
            # Terms involving f[i]:
            #   j=i: f[i]*f[k-i] (if 0<=k-i<n)
            #   k-j=i => j=k-i: f[k-i]*f[i] (if 0<=k-i<n)
            # So if 0<=k-i<n: contribution is 2*f[i]*f[k-i] (or f[i]^2 if k=2i)
            
            contributions = np.zeros(2 * n - 1)
            for k in range(max(0, i), min(2 * n - 1, 2 * i + n - 1) + 1):
                j_other = k - i
                if 0 <= j_other < n:
                    if j_other == i:
                        contributions[k] = f[i] ** 2
                    else:
                        contributions[k] = 2.0 * f[i] * f[j_other]
            
            c = conv - contributions  # Part not involving f[i]
            
            # Now find f[i] >= 0 that minimizes max_k (contributions[k] with new f[i] + c[k])
            # This is: minimize max_k (2*f_new*f[k-i] + c[k]) subject to f_new >= 0
            # (and k ranges over 0..2n-2, but only k where 0<=k-i<n matter)
            
            # Get the affected indices
            affected_k = []
            coeff = []
            const = []
            for k in range(max(0, i), min(2 * n - 1, 2 * i + n) + 1):
                j_other = k - i
                if 0 <= j_other < n:
                    if j_other == i:
                        # f_new^2 + c[k]
                        affected_k.append(k)
                        coeff.append(1.0)  # will handle specially
                        const.append(c[k])
                    else:
                        affected_k.append(k)
                        coeff.append(2.0 * f[j_other])
                        const.append(c[k])
            
            if len(affected_k) == 0:
                continue
            
            # Bisection on the target level
            # We want the smallest M such that there exists x >= 0 with:
            #   x^2 + const[k] <= M (for self terms)
            #   2*f[j_other]*x + const[k] <= M (for cross terms)
            #   AND all non-affected conv[k] <= M (these are just c[k], should already be <= M from before)
            
            # Actually, let's just do a 1D golden section search on the objective:
            #   objective(x) = max_k (affine or quadratic in x)
            
            # For simplicity, use bisection + a few candidate points
            lo = 0.0
            hi = max(old_val * 3.0, 1.0 / n * 3.0)
            
            # Evaluate at a few points
            best_x = old_val
            best_m = np.inf
            candidates = [0.0, old_val * 0.5, old_val * 0.7, old_val, old_val * 1.3, old_val * 1.5, old_val * 2.0, hi / 2.0]
            for x in candidates:
                x = max(0.0, x)
                # Compute max conv with f[i] = x
                new_conv = c.copy()
                for idx, k in enumerate(affected_k):
                    if coeff[idx] == 1.0 and abs(coeff[idx] - 1.0) < 1e-15:
                        # self term: x^2 + const
                        new_conv[k] = x * x + const[idx]
                    else:
                        new_conv[k] = coeff[idx] * x + const[idx]
                m = new_conv.max()
                if m < best_m:
                    best_m = m
                    best_x = x
            
            # Refine with bisection on best region
            if best_x > 0:
                lo2 = max(0.0, best_x * 0.7)
                hi2 = best_x * 1.3
                for _ in range(10):
                    mid = (lo2 + hi2) / 2.0
                    new_conv = c.copy()
                    for idx, k in enumerate(affected_k):
                        if abs(coeff[idx] - 1.0) < 1e-15:
                            new_conv[k] = mid * mid + const[idx]
                        else:
                            new_conv[k] = coeff[idx] * mid + const[idx]
                    m = new_conv.max()
                    if m < best_m:
                        best_m = m
                        best_x = mid
                        lo2 = mid
                    else:
                        hi2 = mid
            
            # Update
            if best_x != old_val:
                # Adjust sum to maintain normalization
                f[i] = best_x
                f = np.maximum(f, 0.0)
                f = f / f.sum()  # Re-normalize
                conv = fast_convolve(f)
                obj = conv.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
            else:
                # Recompute conv since we didn't change f
                pass
    
    return best_f, best_obj

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Frank-Wolfe with LP oracle, (2) Accelerated LSE gradient descent,
(3) Mirror descent (multiplicative updates), (4) Coordinate descent polish,
(5) Randomized restarts with diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear convolution a * a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """
    Gradient of LSE(conv(f,f)) w.r.t. f.
    grad[i] = 2 * sum_{m=0}^{n-1} weights[i+m] * f[m]
    Computed via FFT cross-correlation.
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1: n - 1 + n]


def build_conv_matrix(f):
    """Build sparse banded matrix C where C @ b = conv(f, b)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = np.arange(max(0, k - n + 1), min(n - 1, k) + 1)
        if len(js) > 0:
            rows.extend(np.full(len(js), k).tolist())
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle for the convolution peak minimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f))
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 100 == 0:
            print(f"    FW it={it}, obj={best_obj:.12f}, m_exp={m_exp:.2f}, t={time.monotonic()-t0:.0f}s")

        C = build_conv_matrix(f)

        # Solve LP: max sum(b) s.t. C @ b <= M * m_exp, b >= 0
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_vec, A_ub=C, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 20})
            if res.success:
                g = np.maximum(res.x, 0.0)
                g_sum = g.sum()
                if g_sum < 1e-15:
                    no_improve += 1
                    m_exp = min(m_exp * 1.08, 3.0)
                    continue
                g = g / g_sum
            else:
                no_improve += 1
                m_exp = min(m_exp * 1.08, 3.0)
                continue
        except Exception:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 3.0)
            continue

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.03, 1.5)
            continue

        # Coarse line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.06)
            hi = min(1.0, best_alpha + 0.06)
            for _ in range(7):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 5.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 13 + 42)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 60) / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def lse_fista_optimize(f0, time_limit, beta_start=15.0, beta_end=3000.0):
    """Accelerated (FISTA) log-sum-exp gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f))
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 4.0 / n
    t_k = 1.0
    y = f.copy()
    v = np.zeros(n)
    momentum = 0.85
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 300 == 0:
            print(f"    LSE-FISTA it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, t={time.monotonic()-t0:.0f}s")

        # Compute smooth objective and gradient at y
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_lse_grad(w, y)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # FISTA step
        f_new = project_simplex(y - lr * grad)

        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            lr = min(lr * 1.002, 8.0 / n)
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.92
                momentum = max(0.80, momentum - 0.005)
            if no_improve > 120:
                # Restart
                f_new = best_f.copy()
                y = best_f.copy()
                v = np.zeros(n)
                lr = max(4.0 / n, 0.002)
                t_k = 1.0
                beta = min(beta * 1.5, beta_end * 3)
                no_improve = 0
                t_new = 1.0

        # FISTA update: y = f_new + ((t_k - 1)/t_new) * (f_new - f)
        y = f_new + ((t_k - 1.0) / t_new) * (f_new - f)
        f = f_new
        t_k = t_new

        # Also try momentum-based update as backup
        if it % 50 == 0:
            v = momentum * v - lr * grad
            f_mom = project_simplex(f + v)
            conv_mom = fast_convolve(f_mom)
            if conv_mom.max() < obj_new:
                f = f_mom
                y = f

        if beta < beta_end * 3:
            beta *= 1.005

    return best_f, best_obj


def mirror_descent(f0, time_limit, eta_start=0.5, beta_start=30.0):
    """Mirror descent (multiplicative updates) with LSE smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f))
    else:
        f = f / f
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    eta = eta_start
    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 200 == 0:
            print(f"    MD it={it}, obj={best_obj:.12f}, eta={eta:.4f}, beta={beta:.1f}, t={time.monotonic()-t0:.0f}s")

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_lse_grad(w, f)

        # Multiplicative update
        # f_new[i] proportional to f[i] * exp(-eta * grad[i])
        log_update = -eta * grad
        log_update -= log_update.max()  # numerical stability
        update = f * np.exp(log_update)
        update_sum = update.sum()
        if update_sum < 1e-15:
            break
        f_new = update / update_sum

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            eta = min(eta * 1.01, 5.0)
        else:
            no_improve += 1
            # Mix with previous
            f_new = 0.5 * f + 0.5 * f_new
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            if no_improve % 10 == 0:
                eta *= 0.95
            if no_improve > 100:
                f = best_f.copy()
                eta = max(eta * 0.5, 0.05)
                beta = min(beta * 1.5, 5000.0)
                no_improve = 0

        f = f_new
        if beta < 5000.0:
            beta *= 1.004

    return best_f, best_obj

    s = f.sum()

        f = np.ones(len(f))
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[i]

            # Compute the "baseline" convolution (without f[i]'s contribution)
            # conv[k] = sum_j f[j]*f[k-j]
            # Terms involving f[i]: j=i gives f[i]*f[k-i], and j=k-i gives f[k-i]*f[i]
            # (same term if k-i == i, i.e., k=2i)

            # c[k] = conv[k] - (contributions from f[i])
            c = conv.copy()
            affected = []
            for k in range(max(0, i), min(2 * n - 1, 2 * i) + 1):
                j_other = k - i
                if 0 <= j_other < n:
                    if j_other == i:
                        c[k] -= f[i] * f[i]
                        affected.append((k, 'self'))
                    else:
                        c[k] -= 2.0 * f[i] * f[j_other]
                        affected.append((k, 'cross', j_other))

            if len(affected) == 0:
                continue

            # 1D optimization: find x >= 0 minimizing max_k value
            # value[k] = x^2 + c[k] (self) or 2*f[j_other]*x + c[k] (cross)
            # Also need: c[k] <= M for k not in affected (already satisfied since they don't change)

            # Try candidate values
            best_x = old_val
            best_m = conv.max()

            # Build a function to evaluate
            def eval_1d(x):
                """Evaluate max conv when f[i] = x."""
                max_val = 0.0
                for item in affected:
                    if item[1] == 'self':
                        max_val = max(max_val, x * x + c[item[0]])
                    else:
                        max_val = max(max_val, 2.0 * f[item[2]] * x + c[item[0]])
                # Non-affected values stay the same
                max_val = max(max_val, best_m - 1e-10)  # at least current non-affected max
                # Actually we need the true max including non-affected
                # For efficiency, precompute non-affected max
                return max_val

            # Precompute non-affected max
            non_aff_ks = set(range(2 * n - 1)) - set(item[0] for item in affected)
            if non_aff_ks:
                non_aff_max = max(c[k] for k in non_aff_ks)
            else:
                non_aff_max = 0.0

            def eval_1d_fast(x):
                max_val = non_aff_max
                for item in affected:
                    if item[1] == 'self':
                        max_val = max(max_val, x * x + c[item[0]])
                    else:
                        max_val = max(max_val, 2.0 * f[item[2]] * x + c[item[0]])
                return max_val

            # Golden section search on [0, hi]
            hi = max(old_val * 2.5, 2.0 / n)
            lo = 0.0

            # First, evaluate at a few strategic points
            candidates = [0.0, old_val * 0.3, old_val * 0.6, old_val, old_val * 1.4, old_val * 2.0, hi * 0.5, hi]
            best_x = old_val
            best_m = eval_1d_fast(old_val)
            for xc in candidates:
                xc = max(0.0, min(xc, hi))
                m = eval_1d_fast(xc)
                if m < best_m - 1e-15:
                    best_m = m
                    best_x = xc

            # Refine with bisection
            if best_x > 0:
                lo2 = max(0.0, best_x - max(0.01, best_x * 0.3))
                hi2 = best_x + max(0.01, best_x * 0.3)
                for _ in range(12):
                    mid = (lo2 + hi2) / 2.0
                    m = eval_1d_fast(mid)
                    if m < best_m - 1e-15:
                        best_m = m
                        best_x = mid
                        hi2 = mid
                    else:
                        lo2 = mid

            # Apply update
            if abs(best_x - old_val) > 1e-10:
                f[i] = best_x
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s < 1e-15:
                    break
                f = f / s
                conv = fast_convolve(f)
                obj = conv.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                    sweep_improved = True

        if not sweep_improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 3:
                break
        else:
            no_improve_sweeps = 0

        if sweep % 10 == 0:
            print(f"    CD sweep={sweep}, obj={best_obj:.12f}, t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def normalize(f):
    """Normalize to sum=1, non-negative."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization with diverse strategies.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: FW-LP on parent ===
    print("\n=== Phase 1: FW-LP ===")
    t_alloc = min(350.0, deadline - time.monotonic() - 900.0)
    if t_alloc > 30:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(parent, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent
    n_best = len(current_best)

    # === Phase 2: LSE-FISTA ===
    print(f"\n=== Phase 2: LSE-FISTA ===")
    t_alloc = min(350.0, deadline - time.monotonic() - 500.0)
    if t_alloc > 30:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, obj = lse_fista_optimize(current_best, time_limit=t_alloc,
                                        beta_start=10.0, beta_end=2000.0)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    t_alloc = min(200.0, deadline - time.monotonic() - 300.0)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, obj = coord_descent_polish(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent

    # === Phase 4: Mirror Descent ===
    print(f"\n=== Phase 4: Mirror Descent ===")
    t_alloc = min(150.0, deadline - time.monotonic() - 200.0)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, obj = mirror_descent(current_best, time_limit=t_alloc,
                                    eta_start=0.3, beta_start=50.0)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent

    # === Phase 5: FW-LP refinement ===
    print(f"\n=== Phase 5: FW-LP Refinement ===")
    t_alloc = min(200.0, deadline - time.monotonic() - 150.0)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent

    # === Phase 6: Final LSE + CD ===
    print(f"\n=== Phase 6: Final Polish ===")
    t_alloc = min(120.0, deadline - time.monotonic() - 80.0)
    if t_alloc > 15:
        a_opt, _ = lse_fista_optimize(current_best, time_limit=t_alloc * 0.6,
                                      beta_start=100.0, beta_end=5000.0)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best

    t_alloc = min(80.0, deadline - time.monotonic() - 40.0)
    if t_alloc > 10:
        a_opt, _ = coord_descent_polish(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    # === Phase 7: Try alternative initializations ===
    print(f"\n=== Phase 7: Alternative Initializations ===")
    t_remaining = deadline - time.monotonic() - 30

    inits = []
    # Hann-window-like
    x = np.linspace(0, 1, n_best)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(normalize(hann))
    # Triangle
    tri = np.minimum(x, 1 - x) * 2
    inits.append(normalize(tri))
    # Skewed
    skew = np.power(x, 0.5)
    inits.append(normalize(skew))
    # Random
    inits.append(normalize(rng.dirichlet(np.ones(n_best))))

    for init in inits[:3]:
        t_per = min(60.0, t_remaining / max(1, 3))
        if t_per < 15:
            break
        t_remaining -= t_per

        # Quick FW then CD
        a_opt, _ = fw_lp_optimize(init, time_limit=t_per * 0.5)
        try_sequence(a_opt, best_tracker)
        if t_remaining > 10:
            a_opt, _ = coord_descent_polish(a_opt, time_limit=t_per * 0.3)
            try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
        else:
            final_seq = parent
            final_score = float(evaluate_sequence(final_seq.tolist()))
    else:
        final_seq = parent
        final_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nReturning sequence with score={final_score:.10f}")
    return [float(x) for x in final_seq]
