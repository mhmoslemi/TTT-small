def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:

    cond = u * np.arange(1, n + 1) > (cssv - 1)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 25})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Find active set: peaks within 2% of max
        threshold = M * 0.98
        active = np.where(conv >= threshold)[0]
        if len(active) > top_k:
            # Take top top_k
            order = np.argsort(conv[active])[::-1][:top_k]
            active = active[order]

        # Compute subgradient: average gradient over active peaks
        grad = np.zeros(n)
        for k in active:
            # d conv[k] / d a[i] = 2 * a[k-i] where 0<=k-i<n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]

        grad /= len(active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        # Diminishing step size
        lr = 2.0 / np.sqrt(it + 1.0) * n / (np.sqrt(n) + 1.0)
        lr = min(lr, 0.05)

        a_new = project_simplex(a - lr * grad)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        elif it % 500 == 0:
            a = best_a.copy()  # Revert if not improving

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SG it={it}, obj={best_obj:.12f}, n_active={len(active)}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def try_sequence(a, best_tracker):
    """Evaluate and update best tracker."""
    a = np.maximum(a, 0.0)
    if len(a) == 0:
        return float('inf')
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """Main entry point: multi-phase optimization for minimizing the objective."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_score = float('inf')
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Starting optimization, n_parent={n_parent}, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: FW-LP warm start ===
    t_remaining = deadline - time.monotonic()
    t_alloc = min(150, t_remaining * 0.08)
    if t_alloc > 20:
        print(f"\n=== Phase 1: FW-LP warm start ({t_alloc:.0f}s) ===")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 2: Nesterov LSE main optimization ===
    t_remaining = deadline - time.monotonic()
    t_alloc = min(1000, t_remaining * 0.55)
    if t_alloc > 50:
        print(f"\n=== Phase 2: Nesterov LSE ({t_alloc:.0f}s) ===")
        a_opt, _ = nesterov_lse(current_best, time_limit=t_alloc,
                                beta_start=8.0, beta_end=10000.0, beta_rate=1.004)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 3: Subgradient polish ===
    t_remaining = deadline - time.monotonic()
    t_alloc = min(150, t_remaining * 0.1)
    if t_alloc > 20:
        print(f"\n=== Phase 3: Subgradient polish ({t_alloc:.0f}s) ===")
        a_opt, _ = subgradient_polish(current_best, time_limit=t_alloc, top_k=15)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 4: Try alternative lengths ===
    t_remaining = deadline - time.monotonic()
    alt_lengths = [384, 512]
    if n_parent in alt_lengths:
        alt_lengths.remove(n_parent)
    # Add one more size
    alt_lengths.append(max(320, n_parent + 64))

    for n_alt in alt_lengths:
        t_remaining = deadline - time.monotonic()
        if t_remaining < 80:
            break
        t_alloc = min(120, t_remaining * 0.2)
        if t_alloc < 30:
            break

        print(f"\n=== Phase 4: Alt length n={n_alt} ({t_alloc:.0f}s) ===")
        # Interpolate
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0.0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()
        try_sequence(f_interp, best_tracker)

        # Quick Nesterov LSE
        t_sub = t_alloc * 0.7
        a_opt, _ = nesteror_lse_alt(f_interp, t_sub)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 5: Final high-beta LSE polish ===
    t_remaining = deadline - time.monotonic()
    t_alloc = min(200, t_remaining * 0.5)
    if t_alloc > 20:
        print(f"\n=== Phase 5: Final LSE polish ({t_alloc:.0f}s) ===")
        a_opt, _ = nesterov_lse(current_best, time_limit=t_alloc,
                                beta_start=50.0, beta_end=20000.0, beta_rate=1.006)
        try_sequence(a_opt, best_tracker)

    # === Phase 6: Final subgradient squeeze ===
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic()
    t_alloc = min(60, t_remaining * 0.3)
    if t_alloc > 10:
        print(f"\n=== Phase 6: Final SG squeeze ({t_alloc:.0f}s) ===")
        a_opt, _ = subgradient_polish(current_best, time_limit=t_alloc, top_k=10)
        try_sequence(a_opt, best_tracker)

    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.10f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
