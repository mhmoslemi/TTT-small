```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start L-BFGS with temperature-annealed smoothed max (log-sum-exp)
2. Subgradient descent with momentum on the true non-smooth objective
3. LP-based directional optimization with adaptive tightening
4. Final random perturbation + refinement cycles
Uses softmax parameterization for L-BFGS to handle non-negativity constraints.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.signal import correlate


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_obj_and_grad(x, T):
    """
    Compute smoothed max objective and gradient w.r.t. log-parameters x.
    a = softmax(x), obj = 2n * smooth_max(conv(a,a)) where smooth_max uses log-sum-exp with temp T.
    """
    n = len(x)
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = T * (conv - conv_max)
    shifted -= shifted.max()
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    smooth_max = conv_max + log_sum_exp / T
    obj = 2.0 * n * smooth_max
    
    # Softmax weights on conv
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # grad_a = 2 * correlate(w, a, mode='valid')
    # correlate(w, a, 'valid')[j] = sum_m w[j+m] * a[m]
    # Using FFT: conv(a, reverse(w))[n-1:2n-1][::-1]
    w_rev = w[::-1]
    conv_size = n + len(w) - 1  # n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (conv_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    Wr = np.fft.rfft(w_rev, fft_size)
    conv_aw = np.fft.irfft(A * Wr, fft_size)[:conv_size]
    grad_a = 2.0 * conv_aw[n-1:2*n-1][::-1]
    
    # Chain rule for softmax: grad_x[j] = a[j] * (grad_a[j] - dot(a, grad_a))
    dot_val = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot_val)
    
    return obj, grad_x


def lbfgs_optimize_smooth(x0, n, T_schedule, max_iter_per_T=2000):
    """
    Run L-BFGS with annealed temperature schedule.
    x0: initial log-parameters (length n)
    T_schedule: list of temperatures, from smallest (smoothest) to largest (sharpest)
    Returns: (best_x, best_obj)
    """
    x = x0.copy()
    best_x = x0.copy()
    best_obj = float('inf')
    
    for T in T_schedule:
        result = minimize(
            fun=lambda xx, TT=T: smooth_obj_and_grad(xx, TT)[0],
            x0=x,
            jac=lambda xx, TT=T: smooth_obj_and_grad(xx, TT)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter_per_T, 'maxfun': 10000, 'gtol': 1e-8}
        )
        x = result.x
        obj, _ = smooth_obj_and_grad(x, T)
        if obj < best_obj:
            best_obj = obj
            best_x = x.copy()
    
    return best_x, best_obj


def true_objective(a):
    """Compute the true objective 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def subgradient_polish(a0, time_limit, lr_init=0.005, momentum=0.94, top_k=50, temp=0.001):
    """Subgradient descent with momentum on the true objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient based on top-k peaks
        top_k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
        peak_vals = conv[peaks]
        
        # Softmax weighting
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        # grad_a[j] = 2 * sum_k w[k] * a[k-j] (with valid indices)
        # Build the weighted conv values
        w_full = np.zeros(2*n-1)
        w_full[peaks] = w
        
        # grad_a = 2 * correlate(w_full, a, 'valid')
        # Use FFT
        w_rev = w_full[::-1]
        conv_size = n + (2*n-1) - 1
        fft_size = 1 << (conv_size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        Wr = np.fft.rfft(w_rev, fft_size)
        conv_aw = np.fft.irfft(A * Wr, fft_size)[:conv_size]
        sg = 2.0 * conv_aw[n-1:2*n-1][::-1]
        
        # Momentum update
        v = momentum * v - lr * sg
        a_new = a + v

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (linear convolution, length 2n-1)."""
    n = len(a)
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # More efficiently: use the structure
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None


def lp_optimize(a0, time_limit, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    tightness = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        g = solve_lp(a, M * tightness)
        
        if g is None or np.allclose(g, a, atol=1e-10):
            no_improve += 1
        else:
            d = g - a
            # Line search
            best_t = 0.0
            best_obj_ls = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = np.maximum(a + t * d, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj_ls:
                    best_obj_ls = obj_try
                    best_t = t
            
            if best_t > 0 and best_obj_ls < best_obj - 1e-12:
                a_new = a + best_t * d
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
                best_obj = best_obj_ls
                best_a = a.copy()
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
                tightness = max(0.998, tightness - 0.0005)
            else:
                no_improve += 1
        
        iteration += 1
        
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 31 + 7)
            scale = 0.002 + 0.001 * min(no_improve // 50, 10)
            noise = rng.standard_normal(n) * scale

            a = np.maximum(a, 0.0)

            else:

            M = conv.max()
            no_improve = 0

def resample_sequence(seq, new_n):
    """Resample sequence to new length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq
```

So max(conv(a,a)) is the max of the inverse FFT of |FFT(a)|^2.

This gives us an alternative view: we want to choose a (non-negative, sum=1) such that the inverse FFT of |FFT(a)|^2 is as flat as possible.

In the frequency domain, |FFT(a)|^2 is the power spectral density. We want the time-domain signal (the auto-convolution) to be flat, which means the power spectral density should be "spread out" in a specific way.

But this doesn't immediately give us a simpler optimization. The non-negativity constraint on a is still the hard part.

**Idea: Iterative water-filling**

Another approach: think of the problem as distributing "water" (mass) over n positions to minimize the maximum of the resulting "wave" (auto-convolution). This is similar to the water-filling problem in information theory.

Algorithm:
1. Start with a flat distribution
2. Find the position k* where conv(a,a)(k*) is maximum
3. "Pour water out" from around position k*/2 (since conv(k*) = sum_j a_j * a_{k*-j}, the peak at k* is most influenced by a_{k*/2})
4. "Pour water in" at positions that contribute least to conv(k*)
5. Repeat

This is essentially what subgradient does, but the "water-filling" interpretation might suggest a more structured update.

**Idea: Coordinate descent on the active set**

The max of the auto-convolution is determined by a small "active set" of positions. If we fix the active set, we can optimize the non-active positions separately.

Algorithm:
1. Identify the active set: positions k where conv(a,a)(k) = max(conv(a,a)) (or close to it)
2. For the active set, we want to minimize conv(a,a)(k) for these k
3. This is a convex subproblem (minimizing a convex function over the active set)
4. Update the non-active positions to maintain sum(a) = 1 and a >= 0
5. Update the active set
6. Repeat

This is related to the LP approach but might be more efficient.
