"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE gradient descent with multi-stage beta schedule
2. Peak-targeted subgradient with coordinate refinement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient of LSE smooth max for auto-convolution."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - j] for j in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M):
    """Solve LP: max sum(g) s.t. A@g <= M*sum(f), g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lse_optimize_v2(f0, time_limit, beta_start=10.0, beta_end=10000.0, 
                    beta_ramp_power=0.7, lr_factor=2.5):
    """LSE gradient descent with power-law beta schedule and adaptive momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_factor / n
    momentum = 0.92
    v = np.zeros(n)
    
    # Power-law beta schedule: beta(t) = beta_start * (beta_end/beta_start)^(t^power)
    beta_ratio = beta_end / beta_start
    elapsed_frac = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        elapsed_frac = (time.monotonic() - t0) / time_limit
        beta = beta_start * (beta_ratio ** (elapsed_frac ** beta_ramp_power))
        
        conv = fast_convolve(f)
        
        # Softmax weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient
        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break
        grad = grad / grad_norm

        # Adaptive momentum
        if no_improve < 10:
            momentum = 0.93
        elif no_improve < 30:
            momentum = 0.90
        else:
            momentum = 0.85

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.005, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
            if no_improve > 100:
                # Smart restart: keep best, perturb
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_factor / n, 0.002)
                no_improve = 0
                # Add small random perturbation
                rng = np.random.default_rng(hash((it, n)) % 2**31)
                f = project_simplex(f + rng.standard_normal(n) * (0.005 + 0.01 * min(1, no_improve / 200)))

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-v2 it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, no_imp={no_improve}, t={elapsed:.0f}s")

    return best_f, best_obj


def peak_subgradient(f0, time_limit, n_peaks=5, lr_init=0.004):
    """Subgradient method targeting the top-k peaks of the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = 0.0001
    v = np.zeros(n)
    cur_mom = 0.95

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        
        # Weighted subgradient using top peaks
        k = min(n_peaks, len(conv))
        peaks = np.argpartition(-conv, k)[:k]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_peaks /= w_peaks.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k):
            if w_peaks[i] < 1e-14:
                continue
            k_idx = int(peaks[i])
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w_peaks[i] * f[k_idx - idx]

        v = cur_mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            cur_lr = min(cur_lr * 1.003, 0.01)
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                cur_lr *= 0.94
            if no_improve > 400:
                rng = np.random.default_rng(hash((no_improve, n)) % 2**31)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.001)
                no_improve = 0

        cur_temp *= 1.004

    return best_f, best_obj


def coordinate_refine(f0, time_limit, batch_size=50):
    """Coordinate-wise refinement: try moving mass from high-peak-contributing coords."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        
        # Find peak
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]
        
        # Find which coordinates contribute most to this peak
        # conv[peak_idx] = sum_i f[i] * f[peak_idx - i]
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = f[idx] * f[peak_idx - idx]
        
        # Try reducing the largest contributors
        top_contrib = np.argsort(-contrib)[:batch_size]
        
        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            if f[i] < 1e-10:
                continue
            # Try moving a small amount of mass from i to a random other coordinate
            j = rng.integers(0, n)
            if j == i:
                continue
            delta = min(f[i] * 0.1, 0.001)
            f_trial = f.copy()
            f_trial[i] -= delta
            f_trial[j] += delta
            if f_trial[i] < 0:
                continue
            f_trial = f_trial / f_trial.sum()
            conv_trial = fast_convolve(f_trial)
            if conv_trial.max() < best_obj - 1e-12:
                f = f_trial
                best_obj = conv_trial.max()
                best_f = f.copy()
                improved = True
                break
        
        if not improved:
            # Try a random perturbation
            perturb_idx = rng.integers(0, n)
            target_idx = rng.integers(0, n)
            if perturb_idx != target_idx and f[perturb_idx] > 1e-8:
                delta = f[perturb_idx] * 0.05
                f_trial = f.copy()
                f_trial[perturb_idx] -= delta
                f_trial[target_idx] += delta
                f_trial = f_trial / f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-12:
                    f = f_trial
                    best_obj = conv_trial.max()
                    best_f = f.copy()

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp_for_direction(f, M)
        if g is None:
            M *= 1.1
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = np.maximum(g, 0.0)
        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient: 2 * cross_correlation(weights, f) of length n.
    weights has length 2n-1, f has length n.
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    v = np.zeros(n)
    cur_mom = 0.95
    seed_counter = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        k = min(n_peaks, len(conv))
        peaks = np.argsort(-conv)[:k]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_peaks /= w_peaks.sum()

        sg = np.zeros(n)
        for i in range(k):
            if w_peaks[i] < 1e-14:
                continue
            k_idx = int(peaks[i])
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w_peaks[i] * f[k_idx - idx]

        v = cur_mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            cur_lr = min(cur_lr * 1.003, 0.012)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                cur_lr *= 0.94
            if no_improve > 300:
                seed_counter += 1
                rng = np.random.default_rng(seed_counter * 31627 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.001)
                no_improve = 0

        cur_temp *= 1.005

    return best_f, best_obj


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - j] for j in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_optimize(f0, time_limit, lp_timeout=30):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    seed_counter = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        A = build_lp_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            g = res.x if res.success else None
        except Exception:
            g = None

        if g is None:
            m_exp = min(m_exp * 1.15, 5.0)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = np.maximum(g, 0.0)
        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Refine with bisection
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 40:
            seed_counter += 1
            rng = np.random.default_rng(seed_counter * 9973 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_refine(f0, time_limit, batch_size=40):
    """Coordinate-wise refinement targeting peak reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        peak_idx = int(np.argmax(conv))

        # Find contributors to peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = f[idx] * f[peak_idx - idx]

        top_contrib = np.argsort(-contrib)[:batch_size]

        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            if f[i] < 1e-12:
                continue
            # Try moving mass to a few candidate targets
            for trial in range(3):
                j = int(rng.integers(0, n))
                if j == i:
                    continue
                delta = min(f[i] * (0.03 + 0.04 * trial), 0.002)
                if f[i] - delta < -1e-12:
                    continue
                f_trial = f.copy()
                f_trial[i] = max(0.0, f[i] - delta)
                f_trial[j] += delta
                f_trial = f_trial / f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-13:
                    f = f_trial
                    best_obj = conv_trial.max()
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Random pair move
            for _ in range(5):
                pert = int(rng.integers(0, n))
                target = int(rng.integers(0, n))
                if pert == target or f[pert] < 1e-10:
                    continue
                delta = f[pert] * 0.05
                f_trial = f.copy()
                f_trial[pert] -= delta
                f_trial[target] += delta
                f_trial = f_trial / f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-13:
                    f = f_trial
                    best_obj = conv_trial.max()
                    best_f = f.copy()
                    improved = True
                    break

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: p-norm optimization (coarse structure)
    Phase 2: LSE warm-start chain (progressive refinement)
    Phase 3: FW-LP (global structure)
    Phase 4: Peak subgradient (targeted polish)
    Phase 5: Multi-start diversity
    Phase 6: Coordinate refinement (final)
    """
    best_tracker = [float('inf'), None]
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_par = len(parent)

    # === Phase 1: p-norm optimization ===
    print("\n=== Phase 1: p-norm ===")
    t_alloc = min(250, deadline - time.monotonic() - 700)
    if t_alloc > 40:
        print(f"  Time: {t_alloc:.0f}s, n={n_par}")
        a_opt, _ = pnorm_optimize(parent, time_limit=t_alloc,
                                   p_start=4, p_end=48, p_ramp=0.45)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: LSE warm-start chain ===
    print("\n=== Phase 2: LSE warm-start chain ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(500, deadline - time.monotonic() - 500)
    if t_alloc > 60:
        print(f"  Time: {t_alloc:.0f}s, n={len(current)}")
        a_opt, _ = lse_warm_start_chain(current, time_limit=t_alloc, n_stages=3,
                                         beta_base_ratios=[(5, 500), (50, 5000), (500, 40000)])
        try_sequence(a_opt, best_tracker)

    # === Phase 3: FW-LP ===
    print("\n=== Phase 3: FW-LP ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 300)
    if t_alloc > 30:
        print(f"  Time: {t_alloc:.0f}s, n={len(current)}")
        a_opt, _ = fw_lp_optimize(current, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: Peak subgradient ===
    print("\n=== Phase 4: Peak SG ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 150)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = peak_subgradient(current, time_limit=t_alloc,
                                     n_peaks=12, lr_init=0.006, temp_init=0.00008)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Multi-start diversity ===
    print("\n=== Phase 5: Multi-start ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    t_remaining = deadline - time.monotonic() - 40
    n_starts = min(6, max(2, int(t_remaining / 50)))
    t_per = t_remaining / n_starts
    print(f"  {n_starts} restarts, ~{t_per:.0f}s each")

    for si in range(n_starts):
        t_rem = deadline - time.monotonic() - 20
        if t_rem < 25:
            break
        t_alloc = min(t_per, t_rem * 0.9)

        # Diverse perturbations
        rng = np.random.default_rng(seed + 200 + si * 137)
        if si % 3 == 0:
            # Small perturbation
            f_pert = project_simplex(current + rng.standard_normal(n_cur) * 0.008)
        elif si % 3 == 1:
            # Medium perturbation on random subset
            f_pert = current.copy()
            mask = rng.random(n_cur) < 0.3
            f_pert[mask] = f_pert[mask] * (1 + rng.standard_normal(mask.sum()) * 0.1)
            f_pert = np.maximum(f_pert, 0)
            if f_pert.sum() > 1e-10:
                f_pert = f_pert / f_pert.sum()
            else:
                f_pert = current.copy()
        else:
            # Structured perturbation: shift mass between regions
            f_pert = current.copy()
            n_half = n_cur // 2
            shift = 0.05 * rng.uniform(0.5, 1.5)
            f_pert[:n_half] *= (1 - shift)
            f_pert[n_half:] *= (1 + shift)
            f_pert = np.maximum(f_pert, 0)
            if f_pert.sum() > 1e-10:
                f_pert = f_pert / f_pert.sum()
            else:
                f_pert = current.copy()

        # Quick LSE polish
        t_lse = t_alloc * 0.75
        a_opt, _ = lse_optimize_v2(f_pert, time_limit=t_lse,
                                     beta_start=15.0, beta_end=10000.0,
                                     beta_ramp_power=0.4, lr_factor=3.0)
        try_sequence(a_opt, best_tracker)

        # Brief SG if time
        t_rem2 = deadline - time.monotonic() - 15
        if t_rem2 > 20 and len(a_opt) == n_cur:
            a_opt2, _ = peak_subgradient(a_opt, time_limit=min(30, t_rem2 * 0.4),
                                          n_peaks=8, lr_init=0.003)
            try_sequence(a_opt2, best_tracker)

    # === Phase 6: Coordinate refinement ===
    print("\n=== Phase 6: Coordinate refine ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(120, deadline - time.monotonic() - 15)
    if t_alloc > 15:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = coordinate_refine(current, time_limit=t_alloc, batch_size=50)
        try_sequence(a_opt, best_tracker)

    # === Phase 7: Final LSE micro-polish ===
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(80, deadline - time.monotonic() - 10)
    if t_alloc > 15:
        print(f"\n=== Phase 7: Final LSE polish ({t_alloc:.0f}s) ===")
        a_opt, _ = lse_optimize_v2(current, time_limit=t_alloc,
                                     beta_start=100.0, beta_end=100000.0,
                                     beta_ramp_power=0.3, lr_factor=1.5)
        try_sequence(a_opt, best_tracker)

    # Final
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
    else:
        final_seq = np.ones(448) / 448
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"\nFINAL (fallback): score={final_score:.10f}")

    return [float(x) for x in final_seq]
