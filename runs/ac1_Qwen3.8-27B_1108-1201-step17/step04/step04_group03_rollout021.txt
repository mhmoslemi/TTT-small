def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(conv, a):
    """Compute gradient of max(conv) wrt a (subgradient).
    Returns the subgradient at the peak position."""
    n = len(a)
    peak = np.argmax(conv)
    grad = np.zeros(n)
    k = int(peak)
    for j in range(n):
        if 0 <= k - j < n:
            grad[j] += 2.0 * a[k - j]
    return grad


def compute_smooth_gradient(conv, a, temperature):
    """Compute smooth gradient using softmax over all conv positions."""
    n = len(a)
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    w /= ws
    
    # Gradient = 2 * (w cross-correlated with a)
    # w is of length 2n-1, a is of length n
    # (w * a reversed) via cross-correlation
    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i

        if k_start <= k_end:
            grad[i] = 2.0 * np.dot(w[k_start:k_end+1], a[0:k_end-k_start+1])
    return grad

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines mirror descent (multiplicative weight) with all-peak softmax attention,
efficient coordinate descent with normalization constraint, and periodic FW-LP steps.
Key innovation: O(n) incremental convolution updates for coordinate descent,
enabling hundreds of sweeps within the time budget.
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_quick(f, M, n_constraints=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Optionally use a subset of constraints for speed."""
    n = len(f)
    conv_len = 2 * n - 1
    if n_constraints is not None and n_constraints < conv_len:
        # Use only the top constraints
        fg = fast_convolve(f)
        top_idx = np.argpartition(fg, -n_constraints)[-n_constraints:]
        c = -np.ones(n)
        A_ub = np.zeros((n_constraints, n))
        for row_idx, k in enumerate(top_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[row_idx, js:je + 1] = f[k - (js:je + 1)]
        b_ub = fg[top_idx]
    else:
        c = -np.ones(n)
        A_ub = np.zeros((conv_len, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(conv_len, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5})
    if res.success:
        return res.x
    return None


def coord_descent_sweep(a, conv, max_iter_per_coord=30):
    """One full sweep of coordinate descent. Returns updated a, conv."""
    n = len(a)
    for i in range(n):
        a_i_old = a[i]
        scale = 1.0 - a_i_old
        if scale < 1e-15:
            continue

        # Precompute contrib and linear coefficients
        contrib = np.zeros(2 * n - 1)
        lin_coef = np.zeros(2 * n - 1)
        k_i = i
        for k in range(max(0, k_i), min(2 * n - 1, k_i + n)):
            j = k - i
            if 0 <= j < n:
                if k == 2 * i:
                    contrib[k] = a_i_old * a_i_old
                    lin_coef[k] = 0.0
                else:
                    contrib[k] = 2.0 * a_i_old * a[j]
                    lin_coef[k] = 2.0 * a[j]

        c_rest = conv - contrib  # conv[k] - contrib[k]

        def eval_obj(x):
            if x < 0 or x > 1:
                return np.inf
            s = (1.0 - x) / scale
            # new_conv[k] = s^2 * c_rest[k] + s*x*lin_coef[k] + x^2*(k==2i)
            new_conv = s * s * c_rest + s * x * lin_coef
            new_conv[2 * i] += x * x
            return new_conv.max()

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = eval_obj(c_pt)
        fd = eval_obj(d_pt)

        for _ in range(max_iter_per_coord):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = eval_obj(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = eval_obj(d_pt)

        x_opt = (lo + hi) / 2.0
        if abs(x_opt - a_i_old) < 1e-14:
            continue

        # Update a and conv
        s = (1.0 - x_opt) / scale
        a_new = a * s
        a_new[i] = x_opt
        a = a_new
        conv = fast_convolve(a)

    return a, conv


def mirror_descent_step(a, conv, lr, temperature, n_steps=1):
    """Mirror descent (multiplicative weight) step with all-peak softmax."""
    n = len(a)
    for _ in range(n_steps):
        scaled = conv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
        # This is a cross-correlation
        grad = np.zeros(n)
        for i in range(n):
            k_start = i
            k_end = min(i + n, len(conv)) - 1
            if k_start <= k_end:
                grad[i] = 2.0 * np.dot(w[k_start:k_end + 1], a[:k_end - k_start + 1])

        # Multiplicative update
        log_update = -lr * grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        conv = fast_convolve(a)

    return a, conv


def fw_lp_step(a, max_time=10.0):
    """One FW-LP step: solve LP for direction, line search."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()

    g = solve_lp_quick(a, M, n_constraints=min(2 * n - 1, 200))
    if g is None:
        # Try with relaxed M
        g = solve_lp_quick(a, M * 1.1, n_constraints=min(2 * n - 1, 100))
    if g is None:
        return a

    g_sum = g.sum()
    if g_sum < 1e-10:
        return a
    g = g / g_sum

    # Line search
    best_obj = conv.max()
    best_alpha = 0.0
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        ft = (1.0 - alpha) * a + alpha * g
        ft = np.maximum(ft, 0)
        st = ft.sum()
        if st < 1e-10:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_obj:
            best_obj = val
            best_alpha = alpha

    if best_alpha > 0:
        a = (1.0 - best_alpha) * a + best_alpha * g
        a = np.maximum(a, 0)
        a = a / a.sum()

    return a


def run_mirror_descent(a, time_limit, lr_init=0.005, temp_init=0.0005,

            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
