```python
def cutting_plane_lp(a, M, top_k=50, max_iter=20):
    n = len(a)
    c = fast_convolve(a)  # current autoconv of a
    # Start with top-k peaks
    peaks = np.argpartition(c, -top_k)[-top_k:]
    
    g = a.copy()  # initial guess
    for iter in range(max_iter):
        # Build constraint matrix for active constraints
        A_ub = np.zeros((len(peaks), n))
        for idx, k in enumerate(peaks):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    A_ub[idx, i] = a[j]
        
        b_ub = np.full(len(peaks), M)
        result = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, 
                        bounds=[(0, None)]*n, method='highs')
        if not result.success:
            break
        g = result.x
        
        # Check for violated constraints
        conv_g = fast_convolve(g)
        violations = np.where(conv_g > M + 1e-10)[0]
        
        if len(violations) == 0:
            break
        
        # Add violated constraints (up to top_k total)
        new_peaks = np.union1d(peaks, violations)
        # Keep only top-k by constraint violation
        if len(new_peaks) > top_k * 2:
            new_peaks = np.argpartition(c, -top_k*2)[-top_k*2:]
            # Actually, we should prioritize the most violated
            viol_vals = conv_g[new_peaks]
            sorted_idx = np.argsort(-viol_vals)
            new_peaks = new_peaks[sorted_idx[:top_k*2]]
        peaks = new_peaks
    
    return g
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent with
FFT-based gradient computation, cutting-plane LP refinement, and adaptive
temperature scheduling. Key innovations: (1) O(n log n) gradient via FFT,
(2) cutting-plane LP with only top-K active constraints, (3) multi-scale
initialization from parent sequence.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_weights(c, T):
    """Compute softmax weights for autoconv values c with temperature T."""
    c_max = c.max()
    if T < 1e-12:
        # Hard max
        w = np.zeros_like(c)
        w[c == c_max] = 1.0 / (np.sum(c == c_max))
        return w
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    return w


def compute_grad(a, w, n):
    """Compute gradient of smoothed objective using FFT cross-correlation.
    grad[i] = 2n * sum_k w[k] * a[k-i] for valid k-i
    """
    # N must be >= len(w) + n - 1 = (2n-1) + n - 1 = 3n - 2
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * n * C[:n]


def obj_and_conv(a, n):
    """Compute objective value and autoconvolution for normalized a."""
    c = fast_convolve(a)
    return 2.0 * n * c.max(), c

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s

    T = T_init
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        w = smooth_weights(c, T)
        grad = compute_grad(a, w, n)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr_init * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project
        a = normalized(a)
        
        # Update autoconv
        _, c = obj_and_conv(a, n)
        cur_obj = 2.0 * n * c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Decrease temperature
        T = max(T_min, T * 0.995)
        
        # Periodic LP refinement
        if lp_interval > 0 and t % lp_interval == 0:
            g = cutting_plane_lp(a, c.max(), top_k=top_k_lp, max_iters=10)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                # Line search
                for alpha in [0.3, 0.5, 0.7, 1.0]:
                    a_try = normalized(a + alpha * (g - a))
                    _, c_try = obj_and_conv(a_try, n)
                    obj_try = 2.0 * n * c_try.max()
                    if obj_try < best_obj:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
                        _, c = obj_and_conv(a, n)
                        break
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter {t}, obj={best_obj:.8f}, T={T:.6f}, "
                  f"elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def cutting_plane_lp(a, M, top_k=30, max_iters=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for active constraints, g >= 0.
    Uses cutting-plane: start with top-k peaks, add violated constraints.
    """
    n = len(a)
    c = fast_convolve(a)
    peaks = np.argpartition(c, -top_k)[-top_k:]
    
    g = None
    for it in range(max_iters):
        # Build constraint matrix
        n_cons = len(peaks)
        A_ub = np.zeros((n_cons, n))
        for idx, k in enumerate(peaks):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for i in range(j_start, j_end + 1):
                A_ub[idx, i] = a[k - i]
        
        b_ub = np.full(n_cons, M)
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n
        
        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break
        
        if not result.success:
            break
        g = result.x
        
        # Check violations
        if g is not None and g.sum() > 0:
            conv_g = fast_convolve(g)
            violations = np.where(conv_g > M + 1e-10)[0]
            if len(violations) == 0:
                break
            # Add top violated constraints
            viol_vals = conv_g[violations]
            top_viol = violations[np.argsort(-viol_vals)[:top_k]]
            peaks = np.unique(np.concatenate([peaks, top_viol]))
            if len(peaks) > top_k * 3:
                # Keep only most important constraints
                peaks = peaks[np.argsort(-conv_g[peaks])[:top_k * 2]]
    
    return g


def coordinate_descent(a, time_limit, max_rounds=50):
    """Coordinate descent: for each coordinate, find optimal value given others fixed."""
    a = normalized(a.copy())
    n = len(a)
    best_obj, c = obj_and_conv(a, n)
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Current value
            ai = a[i]
            # If we change a[i] to ai + delta, how does c_k change?
            # c_k += delta * (a[k-i] + a[i-k]) for the two contributions
            # (when ai is at position i, it contributes to c_k as a[i]*a[k-i] and a[k-i]*a[i])
            # Actually c_k = sum_j a[j]*a[k-j], so the terms involving a[i] are:
            # j=i: a[i]*a[k-i] and j=k-i: a[k-i]*a[i] (same term if we count both)
            # Wait: c_k = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # The terms where j=i: a[i]*a[k-i] (if k-i is valid)
            # The terms where k-j=i, i.e., j=k-i: a[k-i]*a[i] (if k-i is valid)
            # These are the same term! So c_k contains a[i]*a[k-i] exactly once (when j=i)
            # and a[k-i]*a[i] exactly once (when j=k-i). These are different terms unless i = k-i.
            # So the total contribution of a[i] to c_k is: a[i]*(a[k-i] + a[i-k]) where valid
            # Wait no. c_k = sum_j a[j] * a[k-j]. 
            # Term j=i: a[i] * a[k-i]
            # Term j=k-i: a[k-i] * a[i]
            # These are two separate terms in the sum (unless i = k-i, i.e., k = 2i)
            # So changing a[i] to a[i]+delta changes c_k by:
            # delta * a[k-i] + a[k-i] * delta = 2*delta*a[k-i] (if k != 2i)
            # delta * a[i] + a[i] * delta = 2*delta*a[i] (if k = 2i)
            # Hmm wait, I need to be more careful.
            # c_k = sum_j a[j] * a[k-j]
            # ∂c_k/∂a[i] = a[k-i] (from term j=i) + a[k-i] (from term j=k-i where a[k-j] = a[i])
            # Wait no: term j=k-i gives a[k-i]*a[i], so ∂/∂a[i] of this is a[k-i]
            # And term j=i gives a[i]*a[k-i], so ∂/∂a[i] of this is a[k-i]
            # So ∂c_k/∂a[i] = a[k-i] + a[k-i] = 2*a[k-i] (when both j=i and j=k-i are valid and distinct)
            # When k = 2i: j=i is the only valid term (j=k-i=i is the same), so ∂c_k/∂a[i] = a[i]
            # When k != 2i: ∂c_k/∂a[i] = 2*a[k-i] (if k-i is in [0,n-1])
            
            # This is getting complex. Let me just use a simpler approach:
            # Try a few candidate values for a[i] and pick the best
            if ai < 1e-15:
                continue
            # Try reducing a[i]
            for frac in [0.5, 0.7, 0.9]:
                a_try = a.copy()
                a_try[i] = ai * frac
                # Renormalize (keep sum the same by distributing to others? No, just renormalize)
                a_try = normalized(a_try)
                _, c_try = obj_and_conv(a_try, n)
                obj_try = 2.0 * n * c_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try increasing some small coordinates
            for i in range(0, n, 3):
                if time.monotonic() - t0 > time_limit:
                    break
                if a[i] < 0.01 * a.mean():
                    a_try = a.copy()
                    a_try[i] += 0.01 * a.mean()
                    a_try = normalized(a_try)
                    _, c_try = obj_and_conv(a_try, n)
                    obj_try = 2.0 * n * c_try.max()
                    if obj_try < best_obj - 1e-12:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
    
    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return normalized(new_seq)


def get_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None:
        inits['parent'] = resample_sequence(parent, n)
    
    # Multi-lobe cosine
    for nlobe in [2, 3, 4, 5, 6, 8]:
        lobe = 0.5 + 0.5 * np.cos(2 * np.pi * nlobe * x / n)
        inits[f'cos{nlobe}'] = normalized(lobe)
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gauss'] = normalized(g)
    
    # B-spline-like (raised cosine)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['rcos'] = normalized(rc)
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    return inits
