import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def softmax_gradient(a, T, top_k=50):
    """
    Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a.
    Uses only top_k peaks for efficiency.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Get top-k peaks
    if top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv))
    
    vals = conv[peaks]
    shifted = vals / T
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(vals)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    grad = np.zeros(n)
    for idx in range(len(peaks)):

        k = peaks[idx]
        wk = 2.0 * w[idx]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idxs = np.arange(i_start, i_end + 1)
        grad[idxs] += wk * a[k - idxs]
    
    return grad

            v = np.zeros(n)

Let me also think about a completely different approach: **simulated annealing with targeted moves**.

At each step:
1. Find the peak lag k*
2. The peak is determined by the positions that contribute: i where a[i] and a[k*-i] are both large
3. To reduce the peak, we can:
   a. Reduce a[i] for positions i that contribute to the peak
   b. Move the mass to positions j that don't contribute to many peaks
4. The "cost" of moving mass from i to j is the increase in other lags

This is essentially what the LP approach does, but let me try a simpler heuristic version.

**Peak-shaving heuristic:**
```
For each iteration:
1. Find top-K peaks: k_1, k_2, ..., k_K
2. For each peak k_m, identify "heavy" contributors: positions i where a[i]*a[k_m - i] > threshold
3. For each heavy contributor i, compute the "pressure" p[i] = sum_m w_m * a[k_m - i]
   (how much position i contributes to the peaks)
4. Find positions with low pressure
5. Move mass from high-pressure to low-pressure positions
6. Use a small step size to avoid destabilizing
```

This is essentially a projected subgradient method where the "gradient" is the pressure vector.

Let me combine all these ideas into a final algorithm:

            w = np.exp((pvals - max_p) / (spread * 0.3 + 1e-15))
            w /= w.sum()

        # Compute pressure on each position
        pressure = np.zeros(n)
        for idx in range(top_k):
            if w[idx] < 1e-8:
                continue
            k = peaks[idx]
            i0 = max(0, k - n + 1)
            i1 = min(n - 1, k)
            if i0 > i1:
                continue
            idxs = np.arange(i0, i1 + 1)
            pressure[idxs] += w[idx] * a[k - idxs]

        # Identify high and low pressure regions
        p_thresh_high = np.percentile(pressure, 80)
        p_thresh_low = np.percentile(pressure, 20)

        high_mask = pressure > p_thresh_high
        low_mask = pressure < p_thresh_low

        if not np.any(high_mask) or not np.any(low_mask):
            break

        # Move mass from high to low pressure
        amount = a[high_mask].sum() * 0.05 * step_mult
        if amount < 1e-12:
            break

        move_from = a[high_mask] * (amount / max(a[high_mask].sum(), 1e-15))
        total_low = a[low_mask].sum()
        if total_low < 1e-15:
            # Distribute uniformly to low positions
            move_to = np.zeros(n)
            low_idx = np.where(low_mask)[0]
            move_to[low_idx] = amount / len(low_idx)
        else:
            move_to = np.zeros(n)
            low_idx = np.where(low_mask)[0]
            move_to[low_idx] = a[low_idx] / total_low * amount

        a_trial = a - move_from + move_to
        a_trial = normalize(a_trial)

        obj_trial = obj_val(a_trial)
        if obj_trial < best_obj - 1e-10:
            a = a_trial
            best_obj = obj_trial
            best_a = a.copy()
            no_improve = 0
            step_mult = min(1.0, step_mult * 1.05)
        else:
            no_improve += 1
            step_mult *= 0.95
            if step_mult < 0.01:
                step_mult = 1.0
                # Larger perturbation
                rng = np.random.default_rng(no_improve)
                a = best_a + rng.standard_normal(n) * 0.005
                a = normalize(a)
                if no_improve > 300:
                    break

    return best_a, best_obj


def coordinate_refine(a0, time_limit, n_pairs=100000):
    """
    Coordinate descent: try pairwise mass transfers to reduce peak.
    Uses incremental convolution updates for speed.
    """
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Sample pairs
        batch = min(n_pairs, max(1000, int(5000000 / max(n, 100))))
        i_batch = rng.integers(0, n, size=batch)
        j_batch = rng.integers(0, n, size=batch)

        for b in range(batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = i_batch[b]
            j = j_batch[b]
            if i == j:
                continue

            # Try moving delta from i to j
            for delta in [0.005, 0.002, 0.001, -0.002, -0.005]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                st = a_try.sum()
                if st < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

        if no_improve > 5000:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1))) + 0.01
    inits['hann'] = hann / hann.sum()

    # Raised cosine (wider)
    rc = 0.5 * (1 - np.cos(2 * np.pi * x / n)) + 0.01
    inits['raised_cos'] = rc / rc.sum()

    # Sine squared
    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine'] = s1 / s1.sum()

    # Bimodal (two Gaussians)
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    bi = g1 + g2 + 0.001
    inits['bimodal'] = bi / bi.sum()

    # Asymmetric (linear decay)
    lin = (n - x) / n + 0.01
    inits['asym_linear'] = lin / lin.sum()

    # Power law
    pw = (x + 1.0) ** 0.5 + 0.01
    inits['power'] = pw / pw.sum()

    # Subtractive random (random - mean, clipped)
    r = rng.standard_normal(n)
    r = r - r.min() + 0.01
    inits['rand_shift'] = r / r.sum()

    # From parent
    if parent is not None and np.isfinite(parent).all() and parent.sum() > 1e-10:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def extend_to_length(a, n_target):
    """Extend/interpolate sequence to target length."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a / s
    if n == n_target:
        return a_norm.copy()
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext /= ses
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid optimization:
    1. Quick gradient optimization from parent
    2. Multi-restart with diverse initializations
    3. Deep optimization of best candidates (gradient + peak-shave + spectral)
    4. Length exploration
    5. Final coordinate descent polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(p.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
            print(f"Parent loaded: n={len(p)}, score={parent_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_sequence is None:
        # Fallback initialization
        n_fallback = 768
        best_sequence = np.ones(n_fallback) / n_fallback
        best_score = obj_val(best_sequence)
        print(f"Fallback uniform n={n_fallback}, score={best_score:.6f}")

    # =========================================================
    # Phase 1: Quick gradient optimization from parent (250s)
    # =========================================================
    print("\n=== Phase 1: Quick Gradient from Parent ===")
    t_start = time.monotonic()
    n_parent = len(best_sequence)
    t_p1 = min(250.0, deadline - time.monotonic() - 800)

    if t_p1 > 30:
        a1, obj1 = optimize_nesterov(best_sequence, t_p1,
                                      T_init=0.004, lr_init=0.05,
                                      mom_init=0.85, top_k=40)
        score1 = float(evaluate_sequence(a1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_sequence = a1.copy()
            print(f"  Phase 1: {score1:.6f} *NEW BEST*")
        else:
            print(f"  Phase 1: {score1:.6f}")

    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # =========================================================
    # Phase 2: Multi-restart exploration (400s)
    # =========================================================
    print("\n=== Phase 2: Multi-Restart Exploration ===")
    t_start = time.monotonic()
    t_p2 = min(400.0, deadline - time.monotonic() - 600)

    if t_p2 > 30:
        n_explore = len(best_sequence)
        inits = generate_inits(n_explore, parent, rng)

        # Prioritize promising inits
        priority = ['parent', 'parent_interp', 'hann', 'sine', 'raised_cos',
                    'bimodal', 'asym_linear', 'random', 'uniform', 'power', 'rand_shift']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]

        n_inits = min(len(init_names), max(3, int(t_p2 / 60)))
        t_per_init = t_p2 / n_inits

        for i, name in enumerate(init_names[:n_inits]):
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per_init * (1 + 0.5 * (len(init_names[:n_inits]) - i) / n_inits),
                          deadline - time.monotonic() - 450)
            if t_alloc < 15:
                break

            a0 = inits[name]
            a_opt, obj_opt = optimize_nesterov(a0, t_alloc,
                                                T_init=0.005, lr_init=0.06,
                                                mom_init=0.85, top_k=50)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Init '{name}': {score:.6f} *NEW BEST*")
                else:
                    print(f"  Init '{name}': {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # =========================================================
    # Phase 3: Deep optimization of best candidate (500s)
    # =========================================================
    print("\n=== Phase 3: Deep Optimization ===")
    t_start = time.monotonic()
    t_p3 = min(500.0, deadline - time.monotonic() - 300)

    if t_p3 > 30 and best_sequence is not None:
        n_best = len(best_sequence)

        # 3a: Nesterov with spectral shaping
        t_3a = t_p3 * 0.4
        if t_3a > 20:
            a_opt, obj_opt = optimize_nesterov(best_sequence, t_3a,
                                                T_init=0.003, lr_init=0.04,
                                                mom_init=0.88, top_k=60,
                                                use_spectral=True, spectral_interval=80,
                                                spectral_alpha=0.04)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  3a Spectral: {score:.6f} *NEW BEST*")
            else:
                print(f"  3a Spectral: {score:.6f}")

        # 3b: Peak shaving
        t_3b = t_p3 * 0.3
        if t_3b > 15 and time.monotonic() < deadline - 250:
            a_opt, obj_opt = peak_shave_optimize(best_sequence, t_3b)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  3b PeakShave: {score:.6f} *NEW BEST*")
            else:
                print(f"  3b PeakShave: {score:.6f}")

        # 3c: Another Nesterov round (sharper T)
        t_3c = t_p3 * 0.3
        if t_3c > 15 and time.monotonic() < deadline - 200:
            a_opt, obj_opt = optimize_nesterov(best_sequence, t_3c,
                                                T_init=0.001, lr_init=0.02,
                                                mom_init=0.92, top_k=20)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  3c Sharp: {score:.6f} *NEW BEST*")
            else:
                print(f"  3c Sharp: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # =========================================================
    # Phase 4: Length exploration (250s)
    # =========================================================
    print("\n=== Phase 4: Length Exploration ===")
    t_start = time.monotonic()
    t_p4 = min(250.0, deadline - time.monotonic() - 150)

    if t_p4 > 30 and best_sequence is not None:
        n_best = len(best_sequence)
        # Try nearby lengths
        candidate_lengths = []
        for dn in [-128, -64, 64, 128, 192, 256]:
            nl = n_best + dn
            if 128 <= nl <= 2048 and nl != n_best:
                candidate_lengths.append(nl)
        # Also try some standard lengths
        for nl in [512, 768, 1024, 1280]:
            if nl not in candidate_lengths and nl != n_best:
                candidate_lengths.append(nl)

        candidate_lengths = candidate_lengths[:4]  # Limit to 4
        t_per_len = t_p4 / max(1, len(candidate_lengths))

        for nl in candidate_lengths:
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_per_len, deadline - time.monotonic() - 80)
            if t_alloc < 20:
                break

            a0 = extend_to_length(best_sequence, nl)
            a_opt, obj_opt = optimize_nesterov(a0, t_alloc,
                                                T_init=0.004, lr_init=0.05,
                                                mom_init=0.87, top_k=40)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Length {nl}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Length {nl}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # =========================================================
    # Phase 5: Final polishing (remaining time)
    # =========================================================
    print("\n=== Phase 5: Final Polishing ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 10.0

    if remaining > 10 and best_sequence is not None:
        # Short Nesterov with very sharp T
        t_5a = min(remaining * 0.5, 60.0)
        if t_5a > 5:
            a_opt, obj_opt = optimize_nesterov(best_sequence, t_5a,
                                                T_init=0.0005, lr_init=0.01,
                                                mom_init=0.94, top_k=10)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  5a Sharp Nesterov: {score:.6f} *NEW BEST*")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 5.0
        t_5b = min(remaining * 0.5, 40.0)
        if t_5b > 5:
            a_opt, obj_opt = coordinate_refine(best_sequence, t_5b)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  5b CoordDesc: {score:.6f} *NEW BEST*")

    print(f"Phase 5 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Fallback
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
