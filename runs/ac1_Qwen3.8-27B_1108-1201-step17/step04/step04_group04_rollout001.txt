import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_nesterov(a0, time_limit, lr_init=0.01, momentum=0.9):
    """Nesterov-accelerated subgradient descent for min max(a*a) s.t. sum(a)=1, a>=0."""
    a = project_simplex(a0.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    v = np.zeros(n)
    y = a.copy()
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + momentum * v
        y = project_simplex(y)
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M = conv_y.max()
        # Use top-k active constraints for smoother gradient
        top_k = min(10, n)
        thresholds = np.partition(conv_y, -top_k)[-top_k]
        active = np.where(conv_y >= thresholds[0] - 1e-6)[0]
        
        # Compute gradient as weighted average of active subgradients
        weights = conv_y[active] - thresholds[0] + 1e-6
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, active):
            k = int(k)
            # grad_i = 2 * y_{k-i} (with y_m = 0 outside [0,n-1])
            # Vectorized: for each i, grad[i] += w * 2 * y[k-i] if 0<=k-i<n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            indices = np.arange(lo, hi + 1)
            grad[indices] += w * y[k - indices]
        
        grad *= 2.0  # Factor of 2 from the quadratic
        
        # Update
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Evaluate
        conv_a = fast_convolve(a_new)
        obj = 2.0 * n * conv_a.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr *= 0.999
        momentum = min(0.98, momentum + 0.0001)
        
        if no_improve > 500:
            break

    """Solve LP to find a direction that reduces peak.
    max 1^T g - eps * ||g-a||_1
    s.t. (a*g)_k <= M, g >= 0

    c = np.concatenate([-np.ones(n), epsilon * np.ones(n)])

    # (a*g)_k <= M for k = 0..2n-2

    # (a*g)_k = sum_j a_j * g_{k-j} for valid j
    # For variable g_i: it appears in (a*g)_k for k = i, i+1, ..., i+n-1

    size = 2 * n - 1
    A_rows = []
    b_vals = []

    for k in range(size):

        row[n + i] = -1.0

        b_vals.append(a[i])

    for i in range(n):

    conv_A = build_conv_matrix_rows(a)
    
    if epsilon > 0:
        # Variables: [g (n), t (n)]
        c = np.concatenate([-np.ones(n), epsilon * np.ones(n)])
        size = 2 * n - 1
        
        # Convolution constraints: conv_A @ g <= M
        A1 = np.hstack([conv_A, np.zeros((size, n))])
        b1 = np.full(size, M)
        
        # g_i - t_i <= a_i
        A2 = np.zeros((n, 2*n))
        for i in range(n):
            A2[i, i] = 1.0
            A2[i, n+i] = -1.0
        b2 = a.copy()
        
        # -g_i - t_i <= -a_i
        A3 = np.zeros((n, 2*n))
        for i in range(n):
            A3[i, i] = -1.0
            A3[i, n+i] = -1.0
        b3 = -a.copy()
        
        A_ub = np.vstack([A1, A2, A3])
        b_ub = np.concatenate([b1, b2, b3])
        bounds = [(0, None)] * (2 * n)
    else:
        # Simpler: max 1^T g s.t. conv_A @ g <= M, g >= 0
        c = -np.ones(n)
        A_ub = conv_A
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            g = result.x[:n]
            g = np.maximum(g, 0)
            return g
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None
