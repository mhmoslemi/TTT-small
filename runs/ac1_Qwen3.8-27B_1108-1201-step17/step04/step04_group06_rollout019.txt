    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    A_gs = target_mag * np.exp(1j * np.angle(A))
    a_gs = np.fft.irfft(A_gs, fft_size)[:size]

    # Project to non-negative
    a_gs = np.maximum(a_gs, 0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2 for the given sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def freq_flatten(a0, time_limit, beta=0.3, n_iters=1000, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)

    no_improve = 0

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        angle = np.angle(A)

        mag_mean = mag.mean()
        mag_flat = (1.0 - beta) * mag + beta * mag_mean
        # Ensure non-negative
        mag_flat = np.maximum(mag_flat, 0)
        
        # Reconstruct
        A_new = mag_flat * np.exp(1j * angle)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()

        else:

def local_search_pairs(a0, time_limit, n_rounds=500):

    a = np.maximum(a0.astype(np.float64), 0.0)

                k_j = k_star - j
                if k_j < 0 or k_j >= n or j == k_j:
                    continue
                
                old_j = a[j]

                if old_j < 1e-10 or old_kj < 1e-10:

def generate_inits(rng, n, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian with different widths
    for sigma_frac in [0.15, 0.2, 0.25, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (sigma_frac * n)) ** 2)
        inits.append(g / g.sum())
    
    # Cosine
    for phase in [0, 0.5]:
        c = np.abs(np.cos(np.pi * (x + phase) / n)) + 0.01
        inits.append(c / c.sum())
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n-1-x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Exponential decay
    for beta in [0.5, 1.0, 2.0, 4.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n-1-x) / n)
        inits.append(e2 / e2.sum())
    
    # "W-shape" (two peaks)
    center = 0.5 * (n - 1)
    w = np.exp(-((x - n*0.25)**2) / (2*(n*0.1)**2)) + np.exp(-((x - n*0.75)**2) / (2*(n*0.1)**2))
    inits.append(w / w.sum())
    
    # "U-shape"
    u = ((x - center) / max(center, 1)) ** 2 + 0.1
    inits.append(u / u.sum())
    
    # Random smooth (Dirichlet with low concentration)
    for alpha in [0.1, 0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)
    
    # Parent if available
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
        for scale in [0.005, 0.01, 0.02, 0.05]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse initialization from multiple sequence families (Gaussian, cosine, 
   edge-concentrated, W-shape, Dirichlet)
2. Exponential gradient (multiplicative weights) with EMA momentum and 
   halving temperature schedule
3. Frequency-domain soft flattening (iterative magnitude compression)
4. Coordinate-pair local search targeting peak-contributing pairs
5. Multi-length exploration via interpolation
6. Fine-grained final polish
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective_normalized(a):
    """Compute max(conv) assuming sum(a)=1. Score = 2*n*max_conv."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def try_and_update(a, best_seq, best_score, label=""):
    """Evaluate sequence via evaluate_sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.01,
                temp_halve_every=400, top_k_frac=0.3,
                momentum=0.85, eta_decay=0.9999, verbose=False):
    """Exponential gradient with EMA momentum and halving temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(20, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    sg_ema = np.zeros(n)
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute softmax-weighted subgradient over top-k peaks
        if top_k < len(c) and top_k > 0:
            indices = np.argpartition(c, -top_k)[-top_k:]
            peek = c[indices]
        else:
            indices = np.arange(len(c))
            peek = c.copy()

        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-12:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

        # EMA momentum update
        sg_ema = momentum * sg_ema + (1.0 - momentum) * sg

        # Multiplicative (exponentiated gradient) update
        sg_max = sg_ema.max()
        if sg_max < 1e-15:
            break
        log_update = -eta * sg_ema / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"      EG iter {it}: obj={best_obj:.10f}, eta={eta:.4f}, temp={temp:.5f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new
        eta *= eta_decay

        # Halve temperature periodically for aggressive cooling
        if it % temp_halve_every == 0:
            temp = max(temp * 0.5, 0.0003)

        # Stagnation: restart with perturbation
        if no_improve > 1500:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            c = fast_convolve(a)
            no_improve = 0
            eta = max(eta0 * 0.3, 0.05)
            temp = max(temp, temp0 * 0.05)
            sg_ema = np.zeros(n)

    return best_a.copy(), best_obj


def freq_flatten(a0, time_limit, beta=0.3, n_iters=2000, verbose=False):
    """Frequency-domain soft flattening via iterative magnitude compression."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    n = len(a)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < n_iters:
        it += 1

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Soft flatten: compress dynamic range of magnitude
        mag_mean = mag.mean()
        mag_min = mag[mag > 0].min() if np.any(mag > 0) else 0
        # Power compression: mag_new = mag^p where p < 1 flattens
        p = 1.0 - beta
        mag_flat = mag ** p
        # Blend with uniform to avoid zeros
        mag_blend = (1.0 - beta * 0.5) * mag_flat + beta * 0.5 * mag_flat.mean()
        mag_blend = np.maximum(mag_blend, 1e-15)

        A_new = mag_blend * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

    return best_a


def local_search_pairs(a0, time_limit, n_rounds=1000):
    """Local search: redistribute mass between coordinate pairs at peak lag."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    no_improve = 0

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False

        # Find top 3 peak lags
        top_lags = np.argpartition(c, -3)[-3:]
        top_lags = top_lags[np.argsort(c[top_lags])[::-1]]

        for k_star in top_lags:
            if improved:
                break
            k_star = int(k_star)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            n_pairs = je - js + 1
            if n_pairs == 0:
                continue

            n_try = min(n_pairs, max(10, n // 8))
            if n_pairs > n_try:
                pair_indices = rng.choice(n_pairs, size=n_try, replace=False)
            else:
                pair_indices = np.arange(n_pairs)

            for pi in pair_indices:
                j = js + int(pair_indices[pi])
                k_j = k_star - j
                if k_j < 0 or k_j >= n or j == k_j:
                    continue

                old_j = a[j]
                old_kj = a[k_j]
                if old_j < 1e-12 or old_kj < 1e-12:
                    continue

                # Try shifting mass to reduce the product
                fractions = [0.15, 0.25, 0.35, 0.1, 0.4, 0.2]
                for frac in fractions:
                    if old_j > old_kj:
                        transfer = old_j * frac
                        new_j = old_j - transfer
                        new_kj = old_kj + transfer
                    else:
                        transfer = old_kj * frac
                        new_kj = old_kj - transfer
                        new_j = old_j + transfer

                    if new_j < 0 or new_kj < 0:
                        continue

                    a[j] = new_j
                    a[k_j] = new_kj
                    c2 = fast_convolve(a)
                    obj2 = c2.max()

                    if obj2 < best_obj - 1e-13:
                        best_obj = obj2
                        best_a = a.copy()
                        c = c2
                        improved = True
                        break
                    else:
                        a[j] = old_j
                        a[k_j] = old_kj
                        break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break

    return best_a

def generate_inits(rng, n, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian with various widths
    for sigma_frac in [0.12, 0.18, 0.22, 0.28, 0.35, 0.45]:
        g = np.exp(-0.5 * ((x - center) / (sigma_frac * n)) ** 2)
        inits.append(g / g.sum())

    # Cosine (various phases and orders)
    for order in [1, 2, 3]:
        c = np.abs(np.cos(order * np.pi * x / n)) + 0.02
        inits.append(c / c.sum())

    # Edge-concentrated (U-shape)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Quadratic U-shape
    u = ((x - center) / max(center, 1)) ** 2 + 0.05
    inits.append(u / u.sum())

    # Exponential decay (both directions)
    for beta in [0.5, 1.0, 2.0, 3.0, 5.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    # W-shape (two peaks)
    for pk_w in [0.08, 0.12, 0.15]:
        w = (np.exp(-((x - n * 0.25) ** 2) / (2 * (n * pk_w) ** 2)) +
             np.exp(-((x - n * 0.75) ** 2) / (2 * (n * pk_w) ** 2)))
        inits.append(w / w.sum())

    # M-shape (three peaks)
    m = (np.exp(-((x - n * 0.2) ** 2) / (2 * (n * 0.08) ** 2)) +
         np.exp(-((x - center) ** 2) / (2 * (n * 0.08) ** 2)) +
         np.exp(-((x - n * 0.8) ** 2) / (2 * (n * 0.08) ** 2)))
    inits.append(m / m.sum())

    # Random smooth (Dirichlet with various concentrations)
    for alpha in [0.1, 0.3, 0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Parent and perturbations
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
        for scale in [0.003, 0.008, 0.015, 0.03]:
            p = parent + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    return inits
