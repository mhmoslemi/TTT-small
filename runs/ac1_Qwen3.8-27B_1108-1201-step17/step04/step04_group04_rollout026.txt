"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative weights (mirror descent) with soft-peak weighting
2. LP-based direction finding with binary search step sizing
3. Multi-scale search and coordinate descent polishing

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, conv, top_k=10, temp=0.01):
    """
    Compute smoothed subgradient of max_k (a*a)[k].
    Returns gradient vector of length n.
    
    The gradient of (a*a)[k] w.r.t. a[i] is 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    # Find top peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    
    # Soft-max weights
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient: grad[i] = sum_k w_k * 2 * a[k-i] for valid k-i
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        # For this k, grad[i] += w[k] * 2 * a[k-i] for i in [max(0, k-n+1), min(n-1, k)]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            # a[k-i] for i in [i_start, i_end]
            # When i = i_start, k-i = k - i_start
            # When i = i_end, k-i = k - i_end
            indices = np.arange(i_start, i_end + 1)
            a_vals = a[k - indices]  # a[k-i_start], a[k-i_start-1], ..., a[k-i_end]
            grad[indices] += w[idx] * 2.0 * a_vals
    
    return grad


def mirror_descent_step(a, grad, eta, floor=1e-15):
    """Multiplicative weights update: a_new[i] ∝ a[i] * exp(-eta * grad[i]), then normalize."""
    n = len(a)
    # Clip gradient to avoid extreme exponentials
    max_grad = np.max(np.abs(grad))
    if max_grad > 1e-10:
        grad_clipped = np.clip(grad, -5.0 / eta, 5.0 / eta)
    else:
        grad_clipped = grad
    
    log_a = np.log(np.maximum(a, floor))
    log_a_new = log_a - eta * grad_clipped
    # Shift for numerical stability
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def lp_direction(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                                  'time_limit': 30})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_optimize_fast(a0, time_limit, max_iters=5000):
    """
    LP-based optimization with binary search step sizing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    C = None  # Lazy build
    
    while time.monotonic() - t0 < time_limit and iter_count < max_iters:
        a_new, improved = lp_step_binary(a, conv, n, C)
        
        if improved:
            a = a_new
            conv = fast_convolve(a)
            M = conv.max()
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iter_count += 1
        
        # Stagnation handling
        if no_improve >= 50:
            rng = np.random.default_rng(iter_count * 31 + 7)
            scale = 0.005 + 0.003 * min(no_improve // 50, 10)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            if no_improve > 300:
                break
    
    return best_a, best_obj


def lp_step_binary(a, conv, n, C=None):
    """One LP step with binary search for optimal step size."""
    M = conv.max()
    
    if C is None:
        C = build_conv_matrix(a)
    
    g, g_sum, success = lp_direction(a, M * 1.00001, C)
    
    if not success or g_sum <= 1.0 + 1e-6:
        return a.copy(), False
    
    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False
    
    # Binary search for best t in [0, 1]
    best_t = 0.0
    best_obj = 2.0 * n * M

def mirror_descent_optimize(a0, time_limit, eta_init=0.1, eta_decay=0.9999, top_k=20, temp=0.01):
    """Mirror descent (multiplicative weights) optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smoothed subgradient
        grad = compute_subgradient(a, conv, top_k=top_k, temp=temp)
        
        # Mirror descent update
        a_new = mirror_descent_step(a, grad, eta)
        
        # Evaluate
        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Reduce eta when stuck
            eta *= 0.95
            if no_improve > 1000:
                # Add small perturbation
                rng = np.random.default_rng(iter_count * 17)
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                conv = fast_convolve(a)
                no_improve = 0
                eta = eta_init * 0.5
        
        eta *= eta_decay
        temp *= 1.001  # Slowly anneal the soft-max
        iter_count += 1
    
    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit, batch_size=100):
    """Fast coordinate descent: adjust pairs of elements."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()

    best_obj = 2.0 * n * conv.max()

    no_improve = 0

    # For changing a[i] by +d and a[j] by -d:
    # conv[k] changes by: d*a[k-i] + d*a[k-j] - d*a[k-i] - d*a[k-j]... 
    # Actually it's more complex. Let's just recompute conv.
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            for delta in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue

                a[j] -= delta

                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
                else:
                    a[i] -= delta
                    a[j] += delta
            else:
                no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, best_obj


def generate_flat_spectrum_init(n, rng):
    """Generate a sequence with approximately flat power spectrum."""
    # Random phases with conjugate symmetry for real output
    N = 1 << (n - 1).bit_length()  # Next power of 2
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[:N//2+1] = np.exp(1j * phases[:N//2+1])
    X[N//2+1:] = np.conj(X[1:N//2][::-1])
    # Ensure DC component is positive
    X[0] = 1.0
    
    a = np.fft.irfft(X, N)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative weights (mirror descent) with soft-peak subgradients
2. LP-based direction finding with binary search step sizing
3. Multi-scale exploration and alternating MD/LP refinement
4. Coordinate descent polishing
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, conv, top_k=20, temp=0.01):
    """
    Compute smoothed subgradient of max_k (a*a)[k].
    ∂(a*a)[k]/∂a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    k_use = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_use)[-k_use:]
    peak_vals = conv[peaks]

    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(k_use)
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(k_use):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            indices = np.arange(i_start, i_end + 1)
            a_vals = a[k - indices]
            grad[indices] += w[idx] * 2.0 * a_vals

    return grad


def mirror_descent_step(a, grad, eta, floor=1e-18):
    """Multiplicative weights update with normalization to sum=1."""
    n = len(a)
    max_abs = np.max(np.abs(grad))
    if max_abs > 1e-14:
        grad_clipped = np.clip(grad, -4.0 / eta, 4.0 / eta)
    else:
        grad_clipped = grad

    log_a = np.log(np.maximum(a, floor))
    log_a_new = log_a - eta * grad_clipped
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                                  'time_limit': 20})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step_binary(a, conv, n, C=None):
    """One LP step with binary search for optimal step size."""
    M = conv.max()

    if C is None:
        C = build_conv_matrix(a)

    g, g_sum, success = lp_direction(a, M * 1.0001, C)

    if not success or g_sum <= 1.0 + 1e-6:
        return a.copy(), False

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False

    best_t = 0.0
    best_obj = 2.0 * n * M
    best_a_local = a.copy()

    # First try t=1
    a_1 = np.maximum(a + d, 0.0)
    s_1 = a_1.sum()
    if s_1 > 1e-15:
        a_1 = a_1 / s_1
        conv_1 = fast_convolve(a_1)
        obj_1 = 2.0 * n * conv_1.max()
        if obj_1 < best_obj - 1e-10:
            best_t = 1.0
            best_obj = obj_1
            best_a_local = a_1.copy()

    if best_t < 1.0:
        lo, hi = 0.0, 1.0
        for _ in range(14):
            mid = (lo + hi) / 2.0
            a_try = np.maximum(a + mid * d, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                hi = mid
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-10:
                best_t = mid
                best_obj = obj_try
                best_a_local = a_try.copy()
                lo = mid
            else:
                hi = mid

    if best_t > 0:
        return best_a_local, True
    return a.copy(), False


def mirror_descent_optimize(a0, time_limit, eta_init=0.05, eta_decay=0.99997,
                            top_k=30, temp=0.005, temp_growth=1.0005):
    """Mirror descent (multiplicative weights) with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    temp_cur = temp
    no_improve = 0
    iter_count = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - t0 < time_limit:
        grad = compute_subgradient(a, conv, top_k=top_k, temp=temp_cur)
        a_new = mirror_descent_step(a, grad, eta)

        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Perturb and restart
                scale = 0.002 * (1 + 0.5 * min(no_improve // 200, 5))
                pert = rng.standard_normal(n) * scale
                a = best_a + pert
                a = np.maximum(a, 0.0)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                else:
                    a = best_a.copy()
                conv = fast_convolve(a)
                no_improve = 0
                eta = eta_init * 0.7
                if no_improve > 2000:
                    break

        eta *= eta_decay
        temp_cur *= temp_growth
        iter_count += 1

    return best_a, best_obj


def lp_optimize(a0, time_limit, max_iters=50000):
    """LP-based optimization with binary search step sizing and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    C = build_conv_matrix(a)
    rng = np.random.default_rng(777 + n)

    while time.monotonic() - t0 < time_limit and iter_count < max_iters:
        a_new, improved = lp_step_binary(a, conv, n, C)

        if improved:
            a = a_new
            conv = fast_convolve(a)
            M = conv.max()
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        if no_improve >= 40:
            scale = 0.003 + 0.002 * min(no_improve // 40, 10)
            pert = rng.standard_normal(n) * scale
            a = best_a + pert
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            C = build_conv_matrix(a)
            no_improve = 0

    return best_a, best_obj


def gradient_optimize(a0, time_limit, lr_init=0.01, top_k=40, temp=0.003):
    """Gradient descent with Nesterov momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    y = a.copy()
    lr = lr_init
    mom = 0.9
    no_improve = 0
    iter_count = 0
    temp_cur = temp

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, top_k=top_k, temp=temp_cur)

        v = mom * v - lr * grad
        a_new = np.maximum(a + v, 0.0)
        sn = a_new.sum()
        if sn > 1e-15:
            a = a_new / sn
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        lr *= 0.9998
        mom = min(0.97, mom + 0.0001)
        temp_cur *= 1.002
        iter_count += 1

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent: adjust pairs of elements."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    no_improve = 0
    batch = max(10, int(200000 / n))

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            found = False
            for delta in [0.002, 0.008, 0.02, -0.002, -0.008, -0.02]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a[i] += delta
                a[j] -= delta
                conv_try = fast_convolve(a)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    found = True
                    break
                else:
                    a[i] -= delta
                    a[j] += delta
            if not found:
                no_improve += 1

        if no_improve > 300:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    s1 = np.sin(np.pi * x / n + 0.3)
    s1 = np.maximum(s1, 0.01)
    inits['sine'] = s1 / s1.sum()

    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.05
    inits['sine2'] = s2 / s2.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Flat spectrum approx
    N = 1 << (n - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[:N // 2 + 1] = np.exp(1j * phases[:N // 2 + 1])
    if N // 2 + 1 > 1:
        X[N // 2 + 1:] = np.conj(X[1:N // 2][::-1])
    X[0] = 1.0
    a_flat = np.fft.irfft(X, N)[:n]
    a_flat = np.maximum(a_flat, 0.0)
    sf = a_flat.sum()
    if sf > 1e-10:
        inits['flat_spec'] = a_flat / sf

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid mirror descent + LP + gradient + multi-scale search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Mirror descent on parent (n=768)
    # =========================================================
    print("=== Phase 1: Mirror Descent ===")
    t1 = time.monotonic()

    if parent is not None:
        t_alloc = min(150.0, deadline - time.monotonic() - 800.0)
        a_opt, obj = mirror_descent_optimize(parent, time_limit=t_alloc,
                                              eta_init=0.04, eta_decay=0.99998,
                                              top_k=30, temp=0.004, temp_growth=1.0008)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  MD parent: {score:.6f}")

    # Try flat spectrum init
    t_alloc = min(60.0, deadline - time.monotonic() - 800.0)
    if t_alloc > 15 and parent is not None:
        inits = generate_inits(len(parent), parent, rng)
        if 'flat_spec' in inits:
            a_opt, obj = mirror_descent_optimize(inits['flat_spec'], time_limit=t_alloc,
                                                  eta_init=0.05, top_k=25, temp=0.003)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  MD flat_spec: {score:.6f} *NEW BEST*")

    print(f"Phase 1 done ({time.monotonic()-t1:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Gradient descent refinement
    # =========================================================
    print("=== Phase 2: Gradient Descent ===")
    t2 = time.monotonic()

    if best_sequence is not None:
        t_alloc = min(120.0, deadline - time.monotonic() - 700.0)
        a_opt, obj = gradient_optimize(best_sequence, time_limit=t_alloc,
                                        lr_init=0.008, top_k=50, temp=0.002)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Grad: {score:.6f} *NEW BEST*")
        else:
            print(f"  Grad: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP optimization
    # =========================================================
    print("=== Phase 3: LP Optimization ===")
    t3 = time.monotonic()

    if best_sequence is not None:
        t_alloc = min(400.0, deadline - time.monotonic() - 300.0)
        a_opt, obj = lp_optimize(best_sequence, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP: {score:.6f} *NEW BEST*")
        else:
            print(f"  LP: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Multi-scale exploration
    # =========================================================
    print("=== Phase 4: Multi-scale ===")
    t4 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        for n_target in [512, 1024, 1536]:
            if time.monotonic() > deadline - 200:
                break
            if n_target == n_best:
                continue

            a_interp = np.interp(np.linspace(0, 1, n_target),
                                  np.linspace(0, 1, n_best), best_sequence)
            a_interp = np.maximum(a_interp, 0.0)
            s = a_interp.sum()
            if s > 1e-15:
                a_interp = a_interp / s
            else:
                a_interp = np.ones(n_target) / n_target

            t_alloc = min(100.0, deadline - time.monotonic() - 200.0)
            if t_alloc < 20:
                break

            # MD then LP
            a_opt, obj = mirror_descent_optimize(a_interp, time_limit=t_alloc * 0.4,
                                                  eta_init=0.04, top_k=30, temp=0.004)
            if time.monotonic() < deadline - 200:
                a_opt2, obj2 = lp_optimize(a_opt, time_limit=min(40.0, deadline - time.monotonic() - 200))
                if obj2 < obj:
                    a_opt = a_opt2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  n={n_target}: {score:.6f} *NEW BEST*")
            else:
                print(f"  n={n_target}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Alternating MD + LP refinement
    # =========================================================
    print("=== Phase 5: Alternating Refinement ===")
    t5 = time.monotonic()

    if best_sequence is not None:
        a = best_sequence.copy()
        n_cur = len(a)
        for rnd in range(5):
            if time.monotonic() > deadline - 60:
                break

            t_rem = deadline - time.monotonic() - 40.0
            if t_rem < 30:
                break

            t_md = min(t_rem * 0.3, 50.0)
            t_lp = min(t_rem * 0.5, 80.0)

            if t_md > 10:
                a_md, obj_md = mirror_descent_optimize(a, time_limit=t_md,
                                                        eta_init=0.02, top_k=40, temp=0.002)
                score = float(evaluate_sequence(a_md.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_md.copy()
                    a = a_md
                    print(f"  Alt MD {rnd}: {score:.6f} *NEW BEST*")
                elif np.isfinite(score):
                    print(f"  Alt MD {rnd}: {score:.6f}")

            if t_lp > 10 and time.monotonic() < deadline - 40:
                a_lp, obj_lp = lp_optimize(a, time_limit=t_lp)
                score = float(evaluate_sequence(a_lp.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_lp.copy()
                    a = a_lp
                    print(f"  Alt LP {rnd}: {score:.6f} *NEW BEST*")
                elif np.isfinite(score):
                    print(f"  Alt LP {rnd}: {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t5:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 6: Coordinate descent + final gradient
    # =========================================================
    print("=== Phase 6: Final Polish ===")
    t6 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_cd = min(40.0, deadline - time.monotonic() - 20.0)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")
            else:
                print(f"  CoordDesc: {score:.6f}")

        if time.monotonic() < deadline - 10:
            t_sg = min(20.0, deadline - time.monotonic() - 10.0)
            if t_sg > 5:
                a_opt, obj = gradient_optimize(best_sequence, time_limit=t_sg,
                                                lr_init=0.003, top_k=60, temp=0.001)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final Grad: {score:.6f} *NEW BEST*")

    print(f"Phase 6 done ({time.monotonic()-t6:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
