def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_and_conv(a):
    """Compute objective value and full convolution for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    return 2.0 * n * max_b, conv, max_b

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def spectral_shape(a, M_target, n_iter=50):
    """
    Iterative spectral shaping: try to make |FFT(a)|^2 more flat
    while keeping (a*a)[k] <= M_target.
    Uses iterative projection between time and frequency domain.
    """
    n = len(a)
    a_work = a.copy()
    s = a_work.sum()
    if s < 1e-15:
        return a_work
    a_work = a_work / s
    
    best_a = a_work.copy()
    best_obj, _, _ = obj_and_conv(a_work)
    
    for it in range(n_iter):
        # Compute FFT
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a_work, fft_size)
        power = np.abs(A) ** 2
        
        # We want power to be more uniform
        # Use a "soft" flattening: compress dynamic range
        p_min = power.min()
        p_max = power.max()
        if p_max > 1e-15:
            # Compress: new_power = p_min + (p_max - p_min) * (power - p_min) / (p_max - p_min) ** alpha
            # alpha > 1 compresses, alpha < 1 expands
            alpha = 0.8 + 0.4 * (it / n_iter)  # gradually flatten
            normalized = (power - p_min) / (p_max - p_min + 1e-30)
            compressed = normalized ** alpha
            new_power = p_min + (p_max - p_min) * compressed
        
        # Reconstruct: keep phase of A, use new magnitude
        new_mag = np.sqrt(new_power)
        new_A = new_mag * np.exp(1j * np.angle(A))
        
        # IFFT
        new_a = np.fft.irfft(new_A, fft_size)[:n]
        new_a = np.maximum(new_a, 0.0)
        s_new = new_a.sum()
        if s_new < 1e-15:
            break
        new_a = new_a / s_new
        
        obj_new, _, _ = obj_and_conv(new_a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = new_a.copy()
        
        a_work = new_a
    
    return best_a

def nesterov_sg(a, time_limit, lr_init=0.003, momentum=0.9, 
                 top_k=50, temp_scale=0.0001):
    """
    Nesterov-accelerated subgradient descent with adaptive features.
    Focuses on top-k autocorrelation peaks with softmax weighting.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)
    
    obj, conv, _ = obj_and_conv(a)
    best_obj = obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        # Subgradient computation
        top_k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -top_k_eff)[-top_k_eff:]
        peak_vals = conv[peaks]
        
        # Softmax weighting (sharper = focus more on worst peaks)
        scaled = peak_vals / temp_scale
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]
        
        # Nesterov update
        v = momentum * v - lr * sg
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a
        else:
            y = y / s_y
        
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a = a_new / s_new
        
        obj_new, conv, _ = obj_and_conv(a)
        step += 1
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve % 100 == 0 and no_improve > 0:
            lr *= 0.995
            temp_scale *= 1.005
        
        # Escape from local minima
        if no_improve >= 500:
            rng = np.random.default_rng(step * 7 + 3)
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            y = a.copy()
            _, conv, _ = obj_and_conv(a)
            no_improve = 0
    
    return best_a, best_obj

def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq

def generate_initializations(n, parent_seq, rng):

    inits = {}

    inits['gaussian'] = g / g.sum()
    
    # Exponential decay from center

    inits['exponential'] = e / e.sum()
    
    # Sine window (raise-cosine like)

    inits['raise_cosine'] = rc / rc.sum()
    
    # Two-peak
    two = np.exp(-0.5 * ((x - 0.3*n) / (0.15*n))**2) + np.exp(-0.5 * ((x - 0.7*n) / (0.15*n))**2)
    inits['two_peak'] = two / two.sum()
    
    # Four-peak
    peaks = [0.15, 0.35, 0.65, 0.85]
    four = sum(np.exp(-0.5 * ((x - p*n) / (0.1*n))**2) for p in peaks)
    inits['four_peak'] = four / four.sum()
    
    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangular'] = tri / tri.sum()
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Super-Gaussian (sharper)
    sg = np.exp(-0.5 * ((x - mid) / (0.15 * n)) ** 4)
    inits['super_gauss'] = sg / sg.sum()
    
    return inits

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-strategy optimization:

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_and_conv(a):
    """Compute objective value and full convolution for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    return 2.0 * n * max_b, conv, max_b


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                contrib = a[j_start:j_end + 1] * a[k - j_end:k - j_start + 1][::-1]
                # Reduce elements that contribute most to this peak
                reduction = contrib * 0.0005
                a[j_start:j_end + 1] -= reduction
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            y = a.copy()
            _, conv, _ = obj_and_conv(a)

        if no_improve >= 400:
            rng_step = np.random.default_rng(step * 13 + 7)
            noise = rng_step.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            y = a.copy()
            _, conv, _ = obj_and_conv(a)
            no_improve = 0
            lr = lr_init * 0.5

    return best_a, best_obj


def spectral_shape(a, n_iter=100, compress_alpha=0.7):
    """
    Iterative spectral shaping: compress dynamic range of |FFT(a)|^2
    to reduce peak autocorrelation (PAPR reduction inspired).
    """
    n = len(a)
    a_work = a.copy()
    s = a_work.sum()
    if s < 1e-15:
        return a_work
    a_work = a_work / s

    best_a = a_work.copy()
    best_obj, _, _ = obj_and_conv(a_work)

    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    for it in range(n_iter):
        A = np.fft.rfft(a_work, fft_size)
        power = np.abs(A) ** 2

        p_min = power.min()
        p_max = power.max()
        if p_max - p_min < 1e-15:
            break

        alpha = compress_alpha + (1.0 - compress_alpha) * (it / max(1, n_iter))
        normalized = (power - p_min) / (p_max - p_min + 1e-30)
        compressed = normalized ** alpha
        new_power = p_min + (p_max - p_min) * compressed

        new_mag = np.sqrt(new_power)
        phase = np.exp(1j * np.angle(A))
        new_A = new_mag * phase

        new_a = np.fft.irfft(new_A, fft_size)[:n]
        new_a = np.maximum(new_a, 0.0)
        s_new = new_a.sum()
        if s_new < 1e-15:
            break
        new_a = new_a / s_new

        obj_new, _, _ = obj_and_conv(new_a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = new_a.copy()

        a_work = new_a

    return best_a


def lp_refine(a, time_limit, verbose=False):
    """
    LP-based direction finding with adaptive constraint tightness.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_and_conv(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 50
    perturb_count = 0
    tightness = 0.9995

    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        M_lp = M * tightness

        c_coeff = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)
        bounds = [(0.0, None)] * n
        result = linprog(c_coeff, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        improved = False
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10 and g_sum > 1.0 + 1e-5:
                g_norm = g / g_sum
                d = g_norm - a
                norm_d = np.linalg.norm(d)
                if norm_d > 1e-12:
                    best_t = 0.0
                    best_obj_ls = best_obj
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                        a_try = np.maximum(a + t * d, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        obj_try, _, _ = obj_and_conv(a_try)
                        if obj_try < best_obj_ls - 1e-14:
                            best_obj_ls = obj_try
                            best_t = t

                    if best_t > 0:
                        a_new = a + best_t * d
                        a_new = np.maximum(a_new, 0.0)
                        s_new = a_new.sum()
                        if s_new > 1e-15:
                            a = a_new / s_new
                        _, conv = obj_and_conv(a)
                        M = conv.max()
                        best_obj = 2.0 * n * M
                        best_a = a.copy()
                        no_improve = 0
                        tightness = max(0.999, tightness - 0.0001)
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve % 10 == 0:
                tightness = min(1.001, tightness * 1.003)

        iteration += 1

        if no_improve >= max_no_improve:
            rng_p = np.random.default_rng(iteration * 17 + perturb_count * 3)
            scale = 0.001 + 0.0008 * min(perturb_count, 12)
            noise = rng_p.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            _, conv = obj_and_conv(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(150, 50 + 15 * perturb_count)
            tightness = 1.0005

            if perturb_count > 50:
                break

    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def generate_initializations(n, parent_seq, rng):
    """Generate diverse structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    inits['uniform'] = np.ones(n) / n

    if parent_seq is not None:
        inits['parent'] = resample_sequence(parent_seq, n)

    # Gaussian (moderate width)
    sigma = 0.22 * n
    g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
    inits['gaussian'] = g / g.sum()

    # Narrow Gaussian
    sigma_n = 0.12 * n
    gn = np.exp(-0.5 * ((x - mid) / sigma_n) ** 2)
    inits['narrow_gauss'] = gn / gn.sum()

    # Exponential decay from center
    e = np.exp(-np.abs(x - mid) / (0.25 * n))
    inits['exponential'] = e / e.sum()

    # Raise-cosine window
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['raise_cosine'] = rc / rc.sum()

    # Two-peak
    two = (np.exp(-0.5 * ((x - 0.3 * n) / (0.12 * n)) ** 2) +
           np.exp(-0.5 * ((x - 0.7 * n) / (0.12 * n)) ** 2))
    inits['two_peak'] = two / two.sum()

    # Four-peak
    peaks = [0.125, 0.375, 0.625, 0.875]
    four = sum(np.exp(-0.5 * ((x - p * n) / (0.08 * n)) ** 2) for p in peaks)
    inits['four_peak'] = four / four.sum()

    # Super-Gaussian (4th power)
    sg = np.exp(-0.5 * ((x - mid) / (0.18 * n)) ** 4)
    inits['super_gauss'] = sg / sg.sum()

    # Linear ramp
    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangular'] = tri / tri.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.25)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Combines Nesterov subgradient, spectral shaping, LP refinement,
    and multi-length exploration with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(p)}, score={p_score:.6f}")
    except Exception:
        pass

    # ===== Phase 1: Multi-length scout =====
    print("=== Phase 1: Multi-length Scout ===")
    t_start = time.monotonic()
    phase1_budget = 180.0
    phase1_deadline = t_start + phase1_budget

    lengths = [512, 768, 1024, 1536]
    scan_results = []

    priority = ['parent', 'gaussian', 'narrow_gauss', 'exponential',
                'raise_cosine', 'two_peak', 'super_gauss', 'ramp',
                'uniform', 'four_peak', 'random']

    for n in lengths:
        if time.monotonic() > phase1_deadline - 25:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = [nm for nm in priority if nm in inits][:4]

        t_per_init = max(12.0, (phase1_budget - 15.0) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 8:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 4)
            if t_alloc < 6:
                break

            a0 = inits[name]
            a_opt, obj_opt = nesterov_sg(a0, time_limit=t_alloc,
                                         lr_init=0.005, momentum=0.93,
                                         top_k=50, temp_scale=0.00012)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: Deep optimization =====
    print("=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()

    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt.copy()))
            seen_lengths.add(n)
        if len(top_configs) >= 4:
            break

    if best_sequence is not None:
        best_n = len(best_sequence)
        if best_n not in seen_lengths:
            top_configs.append((best_n, best_sequence.copy()))

    remaining_for_p2 = deadline - time.monotonic() - 380.0
    if remaining_for_p2 < 80:
        remaining_for_p2 = 80
    t_per_deep = max(25.0, remaining_for_p2 / max(1, len(top_configs)))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 380:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 330)
        if t_alloc < 12:
            break

        print(f"  Deep {i} (n={n}), t={t_alloc:.0f}s")

        # First: Nesterov SG
        t_sg = t_alloc * 0.5
        a_opt, obj_opt = nesterov_sg(a0, time_limit=t_sg,
                                     lr_init=0.003, momentum=0.94,
                                     top_k=80, temp_scale=0.00008)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"    SG: {score:.6f}")

        # Then: LP
        if time.monotonic() < deadline - 330:
            t_lp = min(t_alloc * 0.35, deadline - time.monotonic() - 310)
            if t_lp > 8:
                a_opt, obj_opt = lp_refine(a_opt, time_limit=t_lp)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"    LP: {score:.6f}")

        # Then: Spectral
        if time.monotonic() < deadline - 310:
            t_spec = min(t_alloc * 0.15, deadline - time.monotonic() - 300)
            if t_spec > 5:
                n_iter_spec = max(20, int(t_spec * 10))
                a_opt = spectral_shape(a_opt, n_iter=n_iter_spec)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"    Spectral: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Fine polishing =====
    print("=== Phase 3: Fine Polishing ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 130.0
        if remaining > 30:
            sg_configs = [
                {'lr_init': 0.002, 'momentum': 0.96, 'top_k': 120, 'temp_scale': 0.00004},
                {'lr_init': 0.004, 'momentum': 0.92, 'top_k': 60, 'temp_scale': 0.00015},
                {'lr_init': 0.001, 'momentum': 0.97, 'top_k': 200, 'temp_scale': 0.00002},
                {'lr_init': 0.006, 'momentum': 0.90, 'top_k': 40, 'temp_scale': 0.0003},
            ]

            t_per = remaining / (len(sg_configs) + 2)

            for ci, cfg in enumerate(sg_configs):
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(t_per, deadline - time.monotonic() - 60)
                if t_alloc < 8:
                    break

                a_opt, obj_opt = nesterov_sg(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG{ci}: {score:.6f}")

            # Spectral shaping
            if time.monotonic() < deadline - 50:
                t_spec = min(40.0, deadline - time.monotonic() - 40)
                if t_spec > 5:
                    a_opt = spectral_shape(best_sequence, n_iter=max(30, int(t_spec * 8)),
                                           compress_alpha=0.6)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Spectral: {score:.6f}")

            # Final LP
            if time.monotonic() < deadline - 30:
                t_alloc = min(30.0, deadline - time.monotonic() - 20)
                if t_alloc > 8:
                    a_opt, obj_opt = lp_refine(best_sequence, time_limit=t_alloc)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  LP: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Intensive final search =====
    print("=== Phase 4: Final Intensive Search ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 8.0
        if remaining > 15:
            n_best = len(best_sequence)
            n_trials = max(2, int(remaining / 10))
            t_per = remaining / n_trials

            for pi in range(n_trials):
                if time.monotonic() > deadline - 6:
                    break
                if t_per < 4:
                    break

                # Diverse perturbations
                scale = 0.0008 + 0.0004 * pi
                noise = rng.standard_normal(n_best) * scale
                freq = rng.uniform(1, 8)
                structured = np.sin(2 * np.pi * freq * np.arange(n_best) / n_best) * scale * 0.5
                a_pert = best_sequence + noise + structured
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                t_alloc = min(t_per, deadline - time.monotonic() - 3)
                if t_alloc < 3:
                    break

                if pi % 3 == 0:
                    a_opt, _ = nesterov_sg(a_pert, time_limit=t_alloc,
                                           lr_init=0.003, momentum=0.95,
                                           top_k=100, temp_scale=0.00006)
                elif pi % 3 == 1:
                    a_opt, _ = lp_refine(a_pert, time_limit=t_alloc)
                else:
                    a_opt = spectral_shape(a_pert, n_iter=max(10, int(t_alloc * 12)))

                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Trial {pi} (n={len(a_opt)}): {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
