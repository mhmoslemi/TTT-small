Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a hybrid approach combining:
1. Smoothed-max subgradient with mirror descent (multiplicative weights)
2. LP-based direction finding for periodic large steps
3. Peak balancing via targeted coordinate adjustments
4. Averaged iterates for convergence

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_gradient(a, conv, mu, top_k=200):
    """Compute gradient of smoothed max: mu * log(sum exp(conv_k / mu)).
    
    For numerical efficiency, only considers top_k peaks.
    Returns gradient vector of length n.
    """
    n = len(a)
    m = len(conv)
    
    if top_k >= m:
        peaks = np.arange(m)
        peak_vals = conv
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    
    # Softmax weights (numerically stable)
    scaled = peak_vals / mu
    scaled -= scaled.max()
    weights = np.exp(scaled)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(peaks))
        weights[np.argmax(peak_vals)] = 1.0
    else:
        weights /= w_sum
    
    # grad[j] = 2 * sum_k weights[idx] * a[k-j] for valid (k,j)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = int(peaks[idx])
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            grad[js] += 2.0 * w * a[k - js]
    
    return grad


def mirror_descent_step(a, grad, t):
    """Mirror descent step: a_new[j] proportional to a[j] * exp(-t * grad[j])."""
    # Clip grad to avoid overflow
    grad_clipped = np.clip(grad, -50.0, 50.0)
    a_new = a * np.exp(-t * grad_clipped)
    a_new = np.maximum(a_new, 1e-300)
    a_new /= a_new.sum()
    return a_new


def lp_direction(a, conv, M_scale=1.001):
    """LP-based direction: find g that maximizes sum(g) subject to conv(a,g) <= M.
    
    Returns the direction g (normalized) and whether it was successful.
    """
    n = len(a)
    M = conv.max() * M_scale
    
    # Build constraint matrix: C @ g = conv(a, g)
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[k-j] for k = j, j+1, ..., j+n-1
        C[j:j+n, j] = a
    
    # LP: max sum(g) s.t. C@g <= M, g >= 0
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b = np.full(2 * n - 1, M)
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
                return g, True
    except Exception:
        pass
    return a.copy(), False


def lp_step(a, conv, n):
    """One LP-based step with line search."""
    g, success = lp_direction(a, conv)
    if not success:
        return a.copy(), False
    
    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False
    
    # Line search
    current_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    best_obj = current_obj
    
    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:

        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            break
    
    return best_a, best_obj < current_obj - 1e-13

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_gradient(a, conv, mu, top_k=300):
    """Compute gradient of smoothed max: mu * log(sum exp(conv_k / mu))."""
    n = len(a)
    m = len(conv)

    if top_k >= m:
        peaks = np.arange(m)
        peak_vals = conv
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    scaled -= scaled.max()
    weights = np.exp(scaled)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(peaks))
        weights[np.argmax(peak_vals)] = 1.0
    else:
        weights /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-12:

        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            grad[js] += 2.0 * w * a[k - js]

    return grad

    if s > 1e-15:
        starts.append(('sharpened', sh / s))

    n_starts = len(starts)
    # Reserve time for phases 2-3
    reserved = 350.0
    available = min(deadline - time.monotonic() - reserved, budget_s - 60)
    t_per_start = max(30.0, available / n_starts)

    for i, (name, a0) in enumerate(starts):
        remaining = deadline - time.monotonic() - reserved
        if remaining < 40:
            break

        t_alloc = min(t_per_start, remaining)
        print(f"  Start '{name}' ({i+1}/{n_starts}): t={t_alloc:.0f}s")

        a_opt, obj = main_optimize(a0, time_limit=t_alloc, seed=seed + i * 100)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"    '{name}': {score:.6f} *** NEW BEST ***")
            else:
                print(f"    '{name}': {score:.6f}")

    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Peak-targeted coordinate descent
    # =========================================================
    print("=== Phase 2: Peak-Targeted CD ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 120
        if remaining > 30:
            t_cd = min(remaining * 0.7, 200.0)
            a_opt = peak_targeted_cd(best_sequence, time_limit=t_cd, n_peaks=6, rng=rng)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CD: {score:.6f} *** NEW BEST ***")
            else:
                print(f"  CD: {score:.6f}")

    # =========================================================
    # Phase 3: Final refinement (short main_optimize + LP)
    # =========================================================
    print("=== Phase 3: Final Refinement ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 20:
            # Short main optimize on best
            t_final1 = min(remaining * 0.5, 100.0)
            if t_final1 > 15:
                a_opt, obj = main_optimize(best_sequence, time_limit=t_final1, seed=seed + 999)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final opt: {score:.6f} *** NEW BEST ***")

            # LP polish
            remaining = deadline - time.monotonic() - 10
            if remaining > 5:
                a = np.maximum(best_sequence, 0).astype(np.float64)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                conv = fast_convolve(a)
                n_a = len(a)
                for _ in range(15):
                    if time.monotonic() > deadline - 5:
                        break
                    a_new, improved = lp_step_with_search(a, conv, n_a)
                    if improved:
                        a = a_new
                        conv = fast_convolve(a)
                    score = float(evaluate_sequence(a.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a.copy()
                        print(f"  LP polish: {score:.6f} *** NEW BEST ***")
                    else:
                        if not improved:
                            break

    # Fallback
    if best_sequence is None:
        best_sequence = parent
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Ensure validity
    best_sequence = np.maximum(best_sequence, 0).astype(np.float64)
    best_sequence = np.minimum(best_sequence, 1000.0)
    if best_sequence.sum() < 0.01:
        best_sequence = np.ones(n) / n

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
