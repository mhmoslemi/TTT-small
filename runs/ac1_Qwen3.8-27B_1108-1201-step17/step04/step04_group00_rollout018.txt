def fast_autoconv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad(a, p):
    """Compute gradient of log-sum-exp(beta * autoconv(a)) w.r.t. a.
    p is the softmax vector of length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    # (p * a_rev)[j + n - 1] = g[j]
    size = len(p) + n - 1  # = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(P * AR, fft_size)
    g = conv[n-1 : 2*n-1]  # g[j] = conv[j + n - 1] for j = 0,...,n-1
