        # Also m <= k-js = k - max(0, k-n+1).
        # If k >= n-1: k-js = k-(k-n+1) = n-1. So m <= n-1, always true.
        # If k < n-1: k-js = k-0 = k. So m <= k, which is the same as m in [0, k] = [js, je].
        
        # So for m in [js, je], both indicators are 1!
        # Therefore d(conv[k])/da[m] = 2*a[k-m] for m in [js, je].
        # Except when m = k-m (i.e., 2m = k), where we should only count once.
        # But actually, even when 2m = k, the formula gives 2*a[m], which is correct
        # (since d(a[m]*a[m])/da[m] = 2*a[m]).
        
        # So the gradient contribution from peak k to position m (for m in [js, je]) is:
        # 2 * w[i] * a[k-m]
        
        grad[idx] += 2.0 * w[i] * f[k - idx]
    
    return grad

# Hmm wait, I realize there might be an issue. Let me re-derive.
# conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
# Let S_k = {j : 0<=j<n, 0<=k-j<n} = [max(0,k-n+1), min(n-1,k)]
# 
# d(conv[k])/da[m] = sum_{j in S_k} [1(j=m) * a[k-j] + a[j] * 1(k-j=m)]
# = a[k-m] (if m in S_k) + a[k-m] (if k-m in S_k)
# = a[k-m] * (1(m in S_k) + 1(k-m in S_k))
#
# Now, m in S_k means max(0,k-n+1) <= m <= min(n-1,k).
# k-m in S_k means max(0,k-n+1) <= k-m <= min(n-1,k)
#   = (k - min(n-1,k)) <= m <= (k - max(0,k-n+1))
#   = max(0, k-n+1) <= m <= min(k, n-1)  [since k-min(n-1,k) = max(0, k-(n-1)) = max(0,k-n+1)]
#   Wait: k - min(n-1,k) = max(k-(n-1), 0) = max(k-n+1, 0)
#   And: k - max(0, k-n+1) = min(k, n-1)
# So k-m in S_k means max(0,k-n+1) <= m <= min(k, n-1).
# 
# But S_k = [max(0,k-n+1), min(n-1,k)] which is the same as [max(0,k-n+1), min(k,n-1)].
# So m in S_k and k-m in S_k are the same condition!
# Therefore d(conv[k])/da[m] = 2*a[k-m] for m in S_k, and 0 otherwise.
#
# This confirms: the gradient is 2*a[k-m] for m in S_k.

def compute_gradient_correct(f, conv, top_k, temperature):
    n = len(f)
    m = len(conv)
    
    if top_k >= m:
        peaks = np.arange(m)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * f[k - idx]
    
    return grad

def subgradient_optimize(f0, time_limit, temperature_init=None, temperature_min=1e-6,
                         lr_init=0.005, momentum=0.9, top_k_frac=0.2, 
                         restart_interval=200, restart_scale=0.005,
                         cool_rate=0.9997, verbose=False):
    """Adaptive subgradient descent with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if temperature_init is None:
        conv = fast_convolve(f)
        temperature_init = conv.max() * 0.1
    
    T = temperature_init
    lr = lr_init
    top_k = max(5, int(n * top_k_frac))
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        grad = compute_gradient_correct(f, conv, top_k, T)
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new  # still update f (for momentum)
        
        T = max(T * cool_rate, temperature_min)
        lr = max(lr * 0.9999, 1e-6)
        
        if no_improve >= restart_interval:
            # Restart with perturbation
            scale = restart_scale * (1 + no_improve / restart_interval * 0.5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            conv = fast_convolve(f)
            T = max(conv.max() * 0.05, temperature_min)
            lr = lr_init * 0.5
            no_improve = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}: obj={best_obj:.10f}, T={T:.2e}, lr={lr:.2e}, no_improve={no_improve}, elapsed={elapsed:.1f}s")
    
    return best_f, best_obj

def lp_peak_shave(f0, time_limit, top_k=10, max_iter=None, verbose=False):
    """LP-based peak shaving: minimize max of top_k peaks via LP linearization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)
    
    if max_iter is None:
        max_iter = 5000
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(999)
    it = 0
    
    # Pre-build the B matrix structure (sparse conceptually, but we'll use dense for n<=512)
    # B[k, j] = 1 if 0 <= k-j < n, i.e., j is in S_k
    # Actually, for the LP we need: (B @ g)[k] = sum_j B[k,j] * g[j]
    # where (B @ g)[k] = sum_{j in S_k} g[j] * f[k-j]... no wait.
    
    # The linearization of conv at point f in direction g is:
    # conv(f+g)[k] ≈ conv(f)[k] + 2 * sum_{j in S_k} f[k-j] * g[j]
    # So (B @ g)[k] = sum_{j in S_k} f[k-j] * g[j]
    
    # B[k, j] = f[k-j] if j in S_k, else 0
    
    if n <= 512:
        B = np.zeros((2*n-1, n))
        for j in range(n):
            js = max(0, j)
            # B[k, j] = f[k-j] for k in [j, j+n-1]
            B[j:j+n, j] = f
    else:
        B = None  # Too large for dense
    
    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        
        # Find top peaks
        if top_k >= 2*n-1:
            peaks = np.arange(2*n-1)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Solve LP: min t s.t. t >= conv[k] + 2*(B@g)[k] for k in peaks
        #             sum(g) = 0, g >= -f
        # Variables: [t, g_0, ..., g_{n-1}]
        c = np.zeros(n+1)
        c[0] = 1.0
        
        A_ub = np.zeros((len(peaks), n+1))
        A_ub[:, 0] = -1.0
        if B is not None:
            A_ub[:, 1:] = 2.0 * B[peaks, :]
        else:
            # Compute on the fly
            for pi, k in enumerate(peaks):
                js = max(0, k-n+1)
                je = min(n-1, k)
                A_ub[pi, js:je+1] = 2.0 * f[k - np.arange(js, je+1)]
        
        b_ub = -conv[peaks]
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, 1:] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            no_improve += 5
            if no_improve > 50:
                break
            continue
        
        g = res.x[1:]
        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        
        if np.max(np.abs(target - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = best_obj
        
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            f_new = (1 - alpha) * f + alpha * target
            f_new = np.maximum(f_new, 0)
            sf = f_new.sum()
            if sf < 1e-15:
                continue
            f_new = f_new / sf
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Fine search
            lo = max(0.01, best_alpha * 0.7)
            hi = min(1.0, best_alpha * 1.3)
            for _ in range(6):
                mid = (lo + hi) / 2
                f_new = (1 - mid) * f + mid * target
                f_new = np.maximum(f_new, 0)
                sf = f_new.sum()
                if sf < 1e-15:
                    break
                f_new = f_new / sf
                val = fast_convolve(f_new).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
            
            f = (1 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            # Perturb and restart
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            no_improve = 0
    
    return best_f, best_obj

def freq_domain_perturb(f, rng, num_tries=100, max_phase_shift=0.5):
    """Try frequency-domain phase perturbations to escape local minima."""
    n = len(f)
    fft_size = 1 << (2*n-2).bit_length()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    A = np.fft.fft(f, fft_size)
    # Enforce conjugate symmetry for real signal
    # A[0] is real, A[fft_size/2] is real (if fft_size is even)
    
    for _ in range(num_tries):
        # Random phase perturbation
        phase = np.angle(A)
        mag = np.abs(A)
        # Perturb phases (preserving conjugate symmetry)
        dphase = rng.standard_normal(fft_size) * max_phase_shift * 0.1
        # Enforce conjugate symmetry: dphase[fft_size-m] = -dphase[m]
        dphase_sym = np.zeros(fft_size)
        dphase_sym[0] = 0
        if fft_size % 2 == 0:
            dphase_sym[fft_size//2] = 0
        for m in range(1, fft_size//2):
            dphase_sym[m] = dphase[m]
            dphase_sym[fft_size-m] = -dphase[m]
        
        new_phase = phase + dphase_sym
        A_new = mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_f = a_new.copy()
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, rng, step_sizes=None):
    """Coordinate descent: try moving mass between pairs of positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.001, 0.0005]
    
    # Precompute which pairs to try
    # For efficiency, focus on pairs where one is a "peak contributor" and the other is not
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Find current peak
        k_star = np.argmax(conv)
        
        # Find top contributors to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        contributors = f[k_star - np.arange(js, je+1)]
        
        # Sort by contribution (highest first)
        order = np.argsort(contributors)[::-1]
        
        # Try moving mass from top contributors to non-contributors
        improved = False
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a random top contributor and a random non-contributor
            if len(order) > 0:
                i_rel = order[rng.integers(min(1, len(order)))]  # top contributor
                i = js + i_rel
            
            # Pick a random position to move mass to
            j = rng.integers(n)
            if j == i or f[j] >= f[i]:
                continue
            
            delta = min(f[i] * 0.5, step_sizes[rng.integers(len(step_sizes))])
            
            f_new = f.copy()
            f_new[i] -= delta
            f_new[j] += delta
            
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                conv = conv_new
                improved = True
        
        if not improved and it > 10:
            # Try random pair
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or f[i] < f[j]:
                continue
            delta = min(f[i] * 0.3, 0.005)
            f_new = f.copy()
            f_new[i] -= delta
            f_new[j] += delta
            conv_new = fast_convolve(f_new)
            if conv_new.max() < best_obj - 1e-15:
                best_obj = conv_new.max()
                best_f = f_new.copy()
                f = f_new
                conv = conv_new
    
    return best_f, best_obj

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score

        a = np.clip(a, 0, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:

            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
