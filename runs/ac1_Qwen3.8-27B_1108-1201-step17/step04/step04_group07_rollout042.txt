def multiplicative_update(a, beta, eta):
    n = len(a)
    conv = fast_convolve(a)
    # Compute softmax weights
    shifted = beta * conv
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    # Compute pressure (gradient)
    # p[i] = 2 * sum_m a[m] * w[i+m]
    # Use FFT
    L = 1 << (2*n-1).bit_length()
    a_rev = np.zeros(L)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(L)
    w_pad[:len(w)] = w
    cross = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), L)
    p = 2.0 * cross[n-1:n-1+n]
    # Multiplicative update
    a_new = a * np.exp(-eta * p / max(1.0, p.max()))
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

import numpy as np

from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a

def solve_lp_direction(a, M, A_ub):
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0

    Uses log-sum-exp smoothing of the max-convolution objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    _, conv = obj_value(a, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    beta = beta_start
    eta = eta_start
    step = 0
    no_improve = 0
    
    L = 1 << (2*n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        # Compute LSE weights
        shifted = beta * conv
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(w))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute pressure: p[i] = 2 * sum_m a[m] * w[i+m]
        a_rev = np.zeros(L)
        a_rev[:n] = a[::-1]
        w_pad = np.zeros(L)
        w_pad[:len(w)] = w
        cross = np.fft.irfft(np.fft.rfft(a_rev, L) * np.fft.rfft(w_pad, L), L)
        p = 2.0 * cross[n-1:n-1+n]
        
        # Ensure non-negative pressure (numerical safety)
        p = np.maximum(p, 0.0)
        
        # Multiplicative update
        p_max = p.max()
        if p_max > 1e-15:
            p = p / p_max
        a_new = a * np.exp(-eta * p)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        _, conv = obj_value(a, n)
        cur_obj = 2.0 * n * conv.max()
        step += 1
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal
        if step % 100 == 0:
            beta *= 1.05  # increase sharpness
            eta *= 0.995  # decrease step size
        
        # Restart if stuck
        if no_improve >= 5000:
            rng = np.random.default_rng(step)
            noise = rng.standard_normal(n) * 0.005
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            _, conv = obj_value(a, n)
            no_improve = 0
            eta = max(eta_start * 0.5, eta)
            beta = max(beta_start, beta * 0.5)
        
        if step > 100000:
            break
    
    return best_a, best_obj

"""
Hybrid optimizer combining multiplicative (exponentiated gradient) continuation
with LP-based polishing for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based fast convolution, log-sum-exp smoothing for smooth gradients,
chirp-based initializations exploiting flat-autocorrelation structure,
and multi-length exploration (512/768/1024/2048).
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0. Returns normalized g and raw sum."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    A_ub = build_conv_matrix(a)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def obj_value(a, n):
    """Compute 2n*max(conv(a,a)) and return (obj, conv)."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def resample_sequence(seq, new_n):

    Uses LSE smoothing of max-convolution for smooth gradients.
    Very fast: O(n log n) per step via FFT.

    no_improve = 0
    total_steps = 0

    while time.monotonic() - t0 < time_limit:

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s

            no_improve = 0
            # Reset annealing to be more exploratory
            beta = max(beta_start, beta * 0.3)
            eta = min(eta_start, eta * 2.0)

        if no_improve >= 50000:
            break

    return best_a, best_obj


def lg_polish(a0, time_limit):
    """LP-based iterative optimization with adaptive tightening."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_value(a, n)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iteration = 0
    tightness = 0.9995

    while time.monotonic() - t0 < time_limit:
        M_lp = M * tightness
        g, g_sum = solve_lp_direction(a, M_lp)

        if g_sum <= 1.0 + 1e-7:
            tightness = min(1.001, tightness * 1.003)
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-12:
                no_improve += 1
            else:
                # Coarse line search
                best_t = 0.0
                best_trial = best_obj
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                    a_try = np.maximum(a + t * d, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    obj_try, _ = obj_value(a_try, n)
                    if obj_try < best_trial:
                        best_trial = obj_try
                        best_t = t

                if best_t > 0 and best_trial < best_obj - 1e-13:
                    a = np.maximum(a + best_t * d, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                    _, conv = obj_value(a, n)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    tightness = max(0.999, tightness - 0.0001)
                else:
                    no_improve += 1

        iteration += 1

        if no_improve >= 80:
            rng_local = np.random.default_rng(iteration * 7 + 13)
            scale = 0.002 + 0.002 * min(no_improve // 80, 5)
            noise = rng_local.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            _, conv = obj_value(a, n)
            M = conv.max()
            no_improve = 0
            tightness = 1.001

        if no_improve >= 400:
            break

    return best_a, best_obj


def sg_refine(a0, time_limit, lr_init=0.005, momentum=0.94, top_k=80, temp=0.0001):
    """Momentum subgradient descent for fine-tuning peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    _, conv = obj_value(a, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient using top-k peaks with softmax weighting
        top_k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -top_k_eff)[-top_k_eff:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]

        # Momentum update
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)

        _, conv = obj_value(a, n)
        cur_obj = 2.0 * n * conv.max()
        step_count += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 100 == 0:
            lr *= 0.997
            temp *= 1.003

        if no_improve >= 3000:
            rng_local = np.random.default_rng(step_count * 3 + 7)
            noise = rng_local.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            _, conv = obj_value(a, n)
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid optimizer:
    Phase 1: Quick multiplicative descent scout across multiple lengths/inits (250s)
    Phase 2: Deep continuation MD from best candidates (550s)
    Phase 3: LP + SG polishing (500s)
    Phase 4: Final perturb cycles + alternate MD/LP (400s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}")
    except Exception:
        pass

    # ===== Phase 1: Quick MD Scout =====
    print("=== Phase 1: MD Scout ===")
    t_start = time.monotonic()
    phase1_deadline = t_start + 250.0

    lengths_to_try = [768, 1024, 1536, 512]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths_to_try:
            lengths_to_try.insert(0, pn)

    phase1_results = []
    init_priority = ['parent', 'chirp', 'chirp2', 'uniform', 'gauss', 'sine', 'random']

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline - 25:
            break

        inits = get_initializations(n, parent, rng)
        init_names = [nm for nm in init_priority if nm in inits][:3]

        t_per = max(12.0, (phase1_deadline - time.monotonic() - 15) / (len(lengths_to_try) * 3))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break
            t_alloc = min(t_per, phase1_deadline - time.monotonic() - 8)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj_opt = multiplicative_descent(
                a0, time_limit=t_alloc,
                beta_start=30.0, beta_end=10000.0,
                eta_start=1.0, eta_end=0.01)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                phase1_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P1: n={n}, {name}: {score:.6f}")

    phase1_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: Deep Continuation MD =====
    print("=== Phase 2: Deep MD Continuation ===")
    t_start2 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 900.0
        if remaining > 60:
            # Build list of starting points
            starts = [best_sequence.copy()]
            seen_n = {len(best_sequence)}
            for score, n, name, a_opt in phase1_results:
                if n not in seen_n:
                    starts.append(a_opt.copy())
                    seen_n.add(n)
                if len(starts) >= 3:
                    break

            md_configs = [
                {'beta_start': 20.0, 'beta_end': 80000.0, 'eta_start': 1.2, 'eta_end': 0.002},
                {'beta_start': 100.0, 'beta_end': 200000.0, 'eta_start': 0.6, 'eta_end': 0.001},
                {'beta_start': 5.0, 'beta_end': 30000.0, 'eta_start': 1.5, 'eta_end': 0.005},
            ]

            t_per_md = max(20.0, remaining / (len(starts) * len(md_configs)))

            for sp in starts:
                if time.monotonic() > deadline - 850:
                    break
                for ci, cfg in enumerate(md_configs):
                    if time.monotonic() > deadline - 800:
                        break
                    t_alloc = min(t_per_md, deadline - time.monotonic() - 750)
                    if t_alloc < 15:
                        break

                    a_opt, obj_opt = multiplicative_descent(sp, time_limit=t_alloc, **cfg)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  P2: n={len(a_opt)}, cfg{ci}: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: LP + SG Polishing =====
    print("=== Phase 3: LP + SG Polish ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 350.0
        if remaining > 30:
            # LP polish first
            t_lp = remaining * 0.5
            a_opt, obj_opt = lg_polish(best_sequence, time_limit=t_lp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3-LP: {score:.6f}")

            # SG refine
            t_sg = remaining * 0.3
            a_opt, obj_opt = sg_refine(best_sequence, time_limit=t_sg,
                                       lr_init=0.004, momentum=0.95, top_k=100, temp=0.00008)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3-SG: {score:.6f}")

            # Another LP round
            t_lp2 = min(remaining * 0.2, deadline - time.monotonic() - 300)
            if t_lp2 > 15:
                a_opt, obj_opt = lg_polish(best_sequence, time_limit=t_lp2)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P3-LP2: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Final Perturb + Refine =====
    print("=== Phase 4: Final Refinement ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 25:
            n_best = len(best_sequence)
            n_perturb = max(2, int(remaining / 28))
            t_per = remaining / n_perturb

            for pi in range(n_perturb):
                if time.monotonic() > deadline - 12:
                    break
                if t_per < 8:
                    break

                # Structured perturbation
                scale = 0.0008 + 0.0015 * (pi / max(1, n_perturb))
                noise = rng.standard_normal(n_best) * scale
                # Add smooth component
                smooth = np.sin(np.pi * np.arange(n_best) / n_best) * scale * 0.3
                a_pert = best_sequence + noise + smooth
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                t_alloc = min(t_per, deadline - time.monotonic() - 8)
                if t_alloc < 6:
                    break

                # Alternate methods
                if pi % 3 == 0:
                    a_opt, obj_opt = lg_polish(a_pert, time_limit=t_alloc)
                elif pi % 3 == 1:
                    a_opt, obj_opt = multiplicative_descent(
                        a_pert, time_limit=t_alloc,
                        beta_start=200.0, beta_end=100000.0,
                        eta_start=0.4, eta_end=0.002)
                else:
                    a_opt, obj_opt = sg_refine(
                        a_pert, time_limit=t_alloc,
                        lr_init=0.003, momentum=0.95, top_k=90, temp=0.0001)

                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P4-{pi} (n={len(a_opt)}): {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
