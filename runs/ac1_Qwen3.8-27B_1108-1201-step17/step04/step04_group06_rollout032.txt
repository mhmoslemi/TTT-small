"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1) Fienup-style iterative projection for flat-spectrum initialization
2) Exponentiated gradient with adaptive softmax-weighted subgradients
3) Multi-length exploration with diverse starts
4) Coordinate descent polishing
5) Aggressive perturbation-based restarts to escape local minima

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT (zero-padded)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def circ_convolve(a):
    """Circular auto-convolution via FFT."""
    n = len(a)
    fft_size = 1 << (n - 1).bit_length()
    if fft_size < n:
        fft_size = n
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:n]


def score_from_conv(c):
    """Compute normalized score from convolution (assumes sum(a)=1)."""
    n = (len(c) + 1) // 2
    return 2.0 * n * c.max()


def fienup_flat_spectral(a0, n_iters=200, use_circular=False):
    """Fienup/gerchberg-saxton style: project to flat DFT magnitude and non-negativity."""
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length() if not use_circular else (1 << (n - 1).bit_length())
    if use_circular and fft_size < n:
        fft_size = n

    for _ in range(n_iters):
        # FFT
        if not use_circular:
            a_pad = np.zeros(fft_size)
            a_pad[:n] = a
            A = np.fft.rfft(a_pad)
        else:
            A = np.fft.rfft(a, fft_size)

        # Flatten magnitude
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-30)
        phase = A / mag_safe
        target_mag = 1.0 / np.sqrt(fft_size)
        A_new = phase * target_mag

        # IFFT
        if not use_circular:
            a_full = np.fft.irfft(A_new, fft_size)[:n]
        else:
            a_full = np.fft.irfft(A_new, fft_size)[:n]

        # Project to non-negative
        a_full = np.maximum(a_full, 0.0)
        s2 = a_full.sum()
        if s2 < 1e-15:
            break
        a = a_full / s2

    return a


def compute_subgradient(a, c, top_k=None, temperature=None, power_beta=None):
    """Compute weighted subgradient of max(conv) using top-k or all lags."""
    n = len(a)
    len_c = len(c)

    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()

    if power_beta is not None:
        c_max = peek.max()
        if c_max < 1e-30:
            w = np.zeros(len(indices))
            w[0] = 1.0
        else:
            w = (peek / c_max) ** power_beta
            ws = w.sum()
            if ws > 0:
                w /= ws
            else:
                w = np.zeros(len(indices))
                w[np.argmax(peek)] = 1.0
    elif temperature is not None:
        scaled = peek / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws
    else:
        w = np.ones(len(indices)) / len(indices)

    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

    return sg


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.005, temp_decay=0.9997,
                eta_decay=0.99995, top_k_frac=0.3, power_beta=None,
                verbose=False, restart_patience=2000, restart_scale=0.005):
    """Exponentiated gradient with adaptive parameters and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac)) if top_k_frac else None

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient(a, c, top_k=top_k, temperature=temp if power_beta is None else None,
                                 power_beta=power_beta)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"      EG iter {it}: obj={best_obj:.10f} n={n}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        if it % 2000 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-13:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay
        if power_beta is None:
            temp = max(temp * temp_decay, 1e-5)

        if no_improve > restart_patience:
            rng_local = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
            a = best_a + rng_local.standard_normal(n) * restart_scale
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = max(eta0 * 0.5, 0.1)
            if power_beta is None:
                temp = max(temp, temp0 * 0.1)
            avg_a = a.copy()
            count = 0

    return best_a.copy(), best_obj


def coordinate_polish(a0, time_limit, n_trials_per_coord=5):
    """Coordinate descent with multiplicative perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    no_improve = 0
    factors = [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5,
               0.6, 1.0/0.6, 0.75, 1.0/0.75, 0.9, 1.0/0.9,
               0.65, 1.0/0.65, 0.8, 1.0/0.8, 0.95, 1.0/0.95]

    while time.monotonic() - t0 < time_limit:
        improved = False
        n_perturb = max(4, n // 8)
        coords = rng.choice(n, size=min(n_perturb, n), replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < a.max() * 0.001:
                continue
            for factor in factors:
                a[i] = old_val * factor
                s2 = a.sum()
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-13:
                    best_obj = obj2
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
                else:
                    a[i] = old_val

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_a.copy(), best_obj


def aggressive_restart(a_best, n, rng, strategy):
    """Generate diverse candidates from best solution using different strategies."""
    candidates = []
    a = a_best.copy() if len(a_best) == n else np.ones(n) / n

    if strategy == 0:
        # Gaussian perturbation
        c = a + rng.standard_normal(n) * 0.02
        c = np.maximum(c, 0)
        if c.sum() > 1e-10:
            candidates.append(c / c.sum())

    elif strategy == 1:
        # Zero out random fraction
        c = a.copy()
        mask = rng.random(n) < 0.3
        c[mask] = 0
        if c.sum() > 1e-10:
            candidates.append(c / c.sum())

    elif strategy == 2:
        # Block swap
        c = a.copy()
        blen = max(4, n // 8)
        for _ in range(3):
            i = rng.integers(0, n - blen)
            j = rng.integers(0, n - blen)
            c[i:i+blen], c[j:j+blen] = c[j:j+blen].copy(), c[i:i+blen].copy()
        if c.sum() > 1e-10:
            candidates.append(c / c.sum())

    elif strategy == 3:
        # Scale edges up, center down
        x = np.arange(n, dtype=np.float64)
        edge_w = 1.0 + 0.5 * (2.0 * x / (n - 1) - 1.0) ** 2
        c = a * edge_w
        c = np.maximum(c, 0)
        if c.sum() > 1e-10:
            candidates.append(c / c.sum())

    elif strategy == 4:
        # Random projection
        v = rng.standard_normal(n)
        proj = a + 0.1 * v * (a.max() / (np.abs(v).max() + 1e-10))
        c = np.maximum(proj, 0)
        if c.sum() > 1e-10:
            candidates.append(c / c.sum())

    return candidates


def try_length(best_a, n_target, time_alloc, seed=0):

    else:

def dual_reweight_eg(a0, time_limit, beta0=2.0, beta_end=20.0, n_restarts=3):

    s = a.sum()

    n = len(a)

    best_a = a.copy()

    t0 = time.monotonic()
    eta = 1.0
    it = 0
    no_improve = 0

    target_time = time_limit / n_restarts

    for restart in range(n_restarts):
        beta = beta0 + (beta_end - beta0) * (restart / max(1, n_restarts - 1))
        no_improve = 0
        restart_start = time.monotonic()

        while time.monotonic() - restart_start < target_time and no_improve < 5000:
            it += 1
            sg = compute_subgradient(a, c, power_beta=beta)
            sg_max = sg.max()
            if sg_max < 1e-15:
                break

            log_update = -eta * sg / sg_max
            log_update -= log_update.max()
            a_new = a * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-300)
            s_new = a_new.sum()
            if s_new < 1e-300:

                no_improve += 1

            a = a_new
            c = c_new

            eta = max(eta, 0.05)

    return best_a.copy(), best_obj

                try_seq(a_opt, f"restart_{ri}_{ci}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate Polish ===")
    print(f"  Best so far: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_opt, obj = coordinate_polish(best_seq, time_limit=min(rem * 0.3, 120))
        try_seq(a_opt, "coord_polish")

    # === Phase 7: Final fine-tuning EG ===
    print(f"\n=== Phase 7: Final EG ===")
    print(f"  Best so far: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if rem > 10:
        # Run several short EG passes with different params
        final_configs = [
            {"eta0": 0.3, "temp0": 0.001, "temp_decay": 0.9999, "eta_decay": 0.99999, "top_k_frac": 0.5},
            {"eta0": 0.5, "temp0": 0.002, "temp_decay": 0.9999, "eta_decay": 0.99998, "top_k_frac": 0.4},
            {"eta0": 0.2, "temp0": 0.0005, "temp_decay": 0.99995, "eta_decay": 0.99999, "top_k_frac": 0.6},
            {"eta0": 1.0, "temp0": 0.003, "temp_decay": 0.9998, "eta_decay": 0.99997, "top_k_frac": 0.3},
        ]
        n_final = len(final_configs)
        t_per_final = max(10, rem * 0.5 / n_final)

        for fi, cfg in enumerate(final_configs):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t_alloc = min(t_per_final, rem * 0.6)
            if t_alloc < 5:
                break
            a_opt, _ = eg_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"final_{fi}")

            # Quick coordinate polish after each
            rem = deadline - time.monotonic()
            if rem > 10:
                a_opt2, _ = coordinate_polish(best_seq, time_limit=min(rem * 0.15, 30))
                try_seq(a_opt2, f"final_{fi}_polish")

    # === Phase 8: One more Fienup + EG from best ===
    print(f"\n=== Phase 8: Fienup+EG from best ===")
    print(f"  Best so far: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if rem > 20:
        a_f = fienup_flat_spectral(best_seq, n_iters=50, use_circular=True)
        a_opt, _ = eg_optimize(a_f, time_limit=min(rem * 0.3, 60),
                               eta0=1.0, temp0=0.003, temp_decay=0.9996,
                               eta_decay=0.99995, top_k_frac=0.3)
        try_seq(a_opt, "fienup_best_eg")

        # Also try power-beta from this
        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt2, _ = dual_reweight_eg(a_f, time_limit=min(rem * 0.4, 45),
                                         beta0=2.0, beta_end=30.0, n_restarts=2)
            try_seq(a_opt2, "fienup_beta")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
