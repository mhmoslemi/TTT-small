"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed subgradient descent
with power-law ramp initializations and adaptive annealing.
Key innovation: log-sum-exp smoothing of the max function for faster, more stable
convergence, combined with momentum and multi-scale search.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_and_grad(p, tau):
    """Compute smoothed objective and gradient.
    F_tau(p) = (1/tau) * log(sum_k exp(tau * conv(p,p)[k]))
    grad = sum_k w_k * grad_conv_k where w_k = softmax(tau * conv)
    """
    n = len(p)
    conv = fast_convolve(p)
    
    # Softmax weights (with numerical stability)
    scaled = tau * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    
    # Smoothed objective
    F = (1.0 / tau) * (scaled.max() + np.log(exp_vals.sum()))  # log-sum-exp / tau
    
    # Gradient: sum_k w_k * grad_k
    # grad_k[i] = 2*p[k-i] (if 0<=k-i<n and 2i != k) or p[k-i] (if 2i=k)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        # p[i] appears in conv[k] as p[i]*p[k-i] and p[k-i]*p[i]
        # So d(conv[k])/dp[i] = p[k-i] + p[k-i] (if i != k-i) or p[k-i] (if i == k-i)
        # i.e., 2*p[k-i] if 2i != k, p[i] if 2i == k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            if 2 * i == k:
                grad[i] += w[k] * p[i]
            else:
                grad[i] += 2 * w[k] * p[k - i]
    
    return F, grad

def smooth_obj_and_grad_fast(p, tau, top_k=None):
    """Faster version using only top_k peaks."""
    n = len(p)
    conv = fast_convolve(p)
    
    if top_k is not None and top_k < 2 * n - 1:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(2 * n - 1)
    
    # Softmax weights over top peaks
    pv = conv[peaks]
    scaled = tau * pv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    
    # Smoothed objective (approximation - using only top peaks)
    F = (1.0 / tau) * (np.log(exp_vals.sum()) + pv.max())
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            if 2 * i == k:
                grad[i] += w[idx] * p[i]
            else:
                grad[i] += 2 * w[idx] * p[k - i]
    
    return F, grad

def make_ramp(n, beta, rng=None):
    """Generate a power-law ramp: p[i] ∝ (i+0.5)^beta."""
    x = np.arange(n, dtype=np.float64) + 0.5
    p = x ** beta
    p = np.maximum(p, 1e-15)
    s = p.sum()
    return p / s

def make_bimodal(n, ratio, rng=None):
    """Generate a bimodal distribution."""
    x = np.arange(n, dtype=np.float64)
    center = n * ratio
    width = n * 0.1
    p = np.exp(-0.5 * ((x - center) / width) ** 2)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

def make_cosine(n, freq=1, rng=None):
    """Generate a cosine-modulated distribution."""
    x = np.arange(n, dtype=np.float64)
    p = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

def optimize_smoothed(p0, time_limit, tau_start=10.0, tau_end=200.0,
                       lr=0.01, momentum=0.9, top_k=None,
                       verbose=False):
    """Smoothed subgradient descent with momentum and tau annealing."""
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)
    if top_k is None:
        top_k = min(n, 2 * n - 1)
    
    v = np.zeros(n)
    t0 = time.monotonic()
    
    # Compute total iterations estimate (adaptive)
    best_obj = fast_convolve(p).max()
    best_p = p.copy()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal tau
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Get gradient
        F, grad = smooth_obj_and_grad_fast(p, tau, top_k=top_k)
        
        # Momentum update
        v = momentum * v - lr * grad
        p_new = project_simplex(p + v)
        
        # Evaluate
        obj = fast_convolve(p_new).max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        p = p_new
        
        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.9
            if no_improve > 200:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                p = project_simplex(best_p + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                no_improve = 0
                lr = max(lr, 0.001)
        
        if verbose and it % 500 == 0:
            print(f"    iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.5f}")
    
    return best_p, best_obj

def optimize_peak_targeting(p0, time_limit, lr=0.005, momentum=0.93,
                            num_peaks=10, verbose=False):
    """Peak-targeting subgradient: focus gradient on top peaks only."""
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)
    
    v = np.zeros(n)
    t0 = time.monotonic()
    best_obj = fast_convolve(p).max()
    best_p = p.copy()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(p)
        
        # Get top peaks
        k = min(num_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -k)[-k:]
        
        # Equal-weight gradient on top peaks
        grad = np.zeros(n)
        for pk in peaks:
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            for i in range(js, je + 1):
                if 2 * i == pk:
                    grad[i] += p[i]
                else:
                    grad[i] += 2 * p[int(pk) - i]
        grad /= k
        
        # Update
        v = momentum * v - lr * grad
        p_new = project_simplex(p + v)
        
        obj = fast_convolve(p_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        p = p_new
        lr *= 0.9999
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 7 + 3)
            p = project_simplex(best_p + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            no_improve = 0
            lr = max(lr, 0.002)
    
    return best_p, best_obj
