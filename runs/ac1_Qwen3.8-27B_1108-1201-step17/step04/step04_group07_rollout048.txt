"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines active-set LP, log-sum-exp smoothing, structured perturbations,
and multi-length exploration to beat the 1.5027 target.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, cap=1.0):
    """Project vector onto {x: sum(x)=1, 0<=x<=cap}."""
    n = len(v)
    v = np.minimum(v, cap)
    lam = 0.0
    for _ in range(50):
        u = np.clip(v - lam, 0.0, None)
        s = u.sum()
        if abs(s - 1.0) < 1e-12:
            break
        lam += (s - 1.0) / n
    u = np.clip(v - lam, 0.0, None)
    s = u.sum()
    if s > 1e-15:
        u /= s
    else:
        u = np.ones(n) / n
    return u


def obj_normalized(a):
    """Compute 2n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max(), conv


def build_conv_row(k, a):
    """Build a single row of the convolution matrix for lag k."""
    n = len(a)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    row = np.zeros(n)
    if j_start <= j_end:
        row[j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
    return row


def lp_direction_active(a, M, active_lags, max_constraints=80):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for k in active_lags, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    if len(active_lags) == 0:
        return a.copy(), 1.0
    if len(active_lags) > max_constraints:
        # Take the most important ones (highest conv values)
        active_lags = sorted(active_lags, key=lambda k: a  # placeholder, caller sorts
                           )[:max_constraints]
    b_ub = np.full(len(active_lags), M)
    A_ub = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        A_ub[idx] = build_conv_row(k, a)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0


def lp_direction_full(a, M):
    """Solve LP with all constraints (slower but more accurate)."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        A_ub[k] = build_conv_row(k, a)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0


def line_search(a, d, current_best, max_steps=12):
    """Simple geometric line search."""
    best_t = 0.0
    best_obj = current_best
    t = 1.0
    for _ in range(max_steps):
        if t < 0.005:
            break
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-12:
            t *= 0.5
            continue
        a_try = a_try / s
        obj_try, _ = obj_normalized(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t
            t *= 0.5  # Try smaller step to see if we can do better
        else:
            t *= 0.5
    return best_t, best_obj


def logsumexp_grad(a, beta):
    """Compute gradient of 2n * logsumexp(beta * conv) / beta for normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n), conv
    shifted = beta * (conv - max_c)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    # grad[i] = 2 * sum_k w[k] * a[k-i]
    # Compute via dot products
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += wk * a[k - np.arange(j_start, j_end + 1)]
    grad *= 2.0
    return grad, conv


def logsumexp_optimize(a0, time_limit, beta_init=5.0, beta_max=500.0,
                       lr_init=0.01, n_steps_per_phase=500, n_phases=50):
    """Optimize using log-sum-exp approximation with annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj, _ = obj_normalized(a)

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    momentum = 0.9

    step = 0
    while time.monotonic() - t0 < time_limit:
        # Anneal beta and lr
        phase = step // n_steps_per_phase
        beta = min(beta_max, beta_init * (1.5 ** phase))
        lr = lr_init * (0.95 ** phase)

        for _ in range(n_steps_per_phase):
            if time.monotonic() - t0 > time_limit:
                break
            grad, conv = logsumexp_grad(a, beta)
            v = momentum * v - lr * grad
            a_new = a + v
            a_new = project_simplex(a_new, cap=1.0)
            a = a_new

        # Evaluate true objective
        cur_obj, _ = obj_normalized(a)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()

        step += n_steps_per_phase

    return best_a, best_obj


def lp_optimize_fast(a0, time_limit, use_active=True):
    """Fast LP-based optimization with active set management."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_normalized(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 60
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Identify active lags
        threshold = M * 0.95
        active_lags = list(np.where(conv >= threshold)[0])
        if len(active_lags) == 0:
            active_lags = [int(np.argmax(conv))]

        # Sort by conv value, keep top ones
        active_lags.sort(key=lambda k: -conv[k])
        active_lags = active_lags[:60]

        # Solve LP
        if use_active:
            g, g_sum = lp_direction_active(a, M, active_lags)
        else:
            g, g_sum = lp_direction_full(a, M)

        d = g - a
        norm_d = np.linalg.norm(d)

        if g_sum <= 1.0 + 1e-6 or norm_d < 1e-12:
            no_improve += 1
        else:

            no_improve = 0

def structured_perturb(a, rng, scale=0.002):
    """Apply structured perturbation: shift mass from peak-contributing positions."""
    n = len(a)
    conv = fast_convolve(a)
    peak_k = int(np.argmax(conv))
    # Identify which a[i] contribute most to the peak
    j_start = max(0, peak_k - n + 1)
    j_end = min(n - 1, peak_k)
    contrib = np.zeros(n)
    contrib[j_start:j_end + 1] = a[peak_k - np.arange(j_start, j_end + 1)] * a
    # Subtract from top contributors, add to low contributors
    total_c = contrib.sum()
    if total_c < 1e-15:
        return a
    a_new = a.copy()
    a_new -= scale * contrib / total_c
    a_new += scale / n  # Redistribute uniformly
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-12:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def resample_seq(seq, new_n):
    """Resample a sequence to a different length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-12:
        new_seq /= s
    return new_seq


def get_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        inits['parent'] = resample_seq(parent, n)

    # Sine modulated
    for freq in [1, 2, 3]:
        s = 0.7 + 0.3 * np.sin(2.0 * np.pi * freq * x / n)
        inits[f'sine{freq}'] = s / s.sum()

    # Cosine lobes
    for freq in [2, 4, 6]:
        c = 0.6 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        inits[f'cos{freq}'] = c / c.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Two Gaussian
    g2 = np.exp(-0.5 * ((x - 0.3 * n) / (0.15 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2)
    inits['twogauss'] = g2 / g2.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    return inits
