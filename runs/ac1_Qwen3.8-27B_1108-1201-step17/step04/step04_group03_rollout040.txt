Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth max (log-sum-exp) subgradient descent with beta annealing
2. Pair-swap local search exploiting the structure of auto-convolution gradients
3. Coordinate descent with exact 1D optimization
4. Multiple diverse initializations and restarts

import numpy as np
import time

def fast_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def auto_conv(a):
    return fast_convolve(a, a)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(weights, f):
    """Compute gradient of smooth max: grad[i] = 2 * sum_j f[j] * weights[j+i]"""
    n = len(f)
    # Using cross-correlation via np.convolve
    f_rev = f[::-1]
    conv_result = np.convolve(f_rev, weights, mode='full')
    cross = conv_result[n-1:n-1+n]
    return 2.0 * cross

def smooth_max_descent(f0, time_limit, beta_init=20.0, beta_max=200.0,
                       lr_init=0.005, min_lr=1e-6, verbose=False):

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = auto_conv(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = auto_conv(f)
        M = conv.max()
        
        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal beta
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_init * (beta_max / beta_init) ** (progress ** 0.7)
        
        # Softmax weights
        shifted = beta * (conv - conv.max())
        exp_vals = np.exp(shifted)
        weights = exp_vals / exp_vals.sum()
        
        # Gradient
        grad = compute_grad(weights, f)
        
        # Step

        M_new = conv_new.max()

            f = f_new

                lr = min(lr * 1.05, lr_init)

            lr *= 0.7
            if lr < min_lr:
                lr = min_lr

            f = project_simplex(best_f + noise)
            lr = max(lr_init * 0.3, min_lr * 10)
            no_improve = 0

            print(f"    SMD it={it}, M={M:.8f}, best={best_obj:.8f}, beta={beta:.1f}, lr={lr:.2e}")
    
    return best_f, best_obj

    Local search via pair swaps. For the peak lag k*, 
    transfer mass from position i (where a[k*-i] is large) to position j (where a[k*-j] is small).
    """

    s = f.sum()

            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = min(10, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_lags]
        
        # For each peak lag, find best swap
        best_delta = 0
        best_i = -1
        best_j = -1
        best_k = -1
        
        for k in peak_lags:
            k = int(k)
            # For each position i, the "cost" of having mass there is a[k-i]
            # We want to move mass from high-cost to low-cost positions
            costs = np.zeros(n)
            valid = np.zeros(n, dtype=bool)
            for i in range(n):
                ki = k - i
                if 0 <= ki < 2 * n - 1:

                    if 0 <= k - i <= n - 1:

                        valid[i] = True
                    else:
                        costs[i] = 0.0
                        valid[i] = True  # still valid to put mass here, just no contribution to this peak

            # ΔM ≈ -2*eps*a[k-i] + 2*eps*a[k-j] = 2*eps*(a[k-j] - a[k-i])
            # We want a[k-j] < a[k-i], so move from i to j
            # Best: maximize a[k-i] - a[k-j]
            
            # For each valid i, j: score = a[k-i] - a[k-j] (want to maximize)
            # Best i: argmax a[k-i] (over valid i where a[i] > 0)

            # Target: minimize a[k-j] (prefer positions that contribute little to this peak)
            target_score = np.where(mask, a_k_minus, np.inf)

            j_best = int(np.argmin(target_score))

                j_best = int(np.argmin(target_score))

            delta_move = (source_score[i_best] - target_score[j_best]) * w

                best_i = i_best
                best_j = j_best
                best_k = k

"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_auto_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(weights, f):
    """
    Compute gradient of smooth max: grad[i] = 2 * sum_j f[j] * weights[j+i]
    where weights has length 2n-1 and f has length n.
    """
    n = len(f)
    f_rev = f[::-1]
    conv_result = fast_conv(f_rev, weights)
    cross = conv_result[n - 1:n - 1 + n]
    return 2.0 * cross


def smooth_max_descent(f0, time_limit, beta_init=20.0, beta_max=300.0,
                       lr_init=0.004, min_lr=1e-7, verbose=False):
    """
    Minimize max(conv(a,a)) via log-sum-exp smoothed subgradient descent.
    Maintains sum(a)=1, a>=0 via simplex projection.
    """
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_auto_conv(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_auto_conv(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal beta with easing
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_init * (beta_max / beta_init) ** (progress ** 0.6)

        # Softmax weights over conv values
        shifted = beta * (conv - conv.max())
        exp_vals = np.exp(shifted)
        weights = exp_vals / exp_vals.sum()

        # Compute gradient
        grad = compute_smooth_grad(weights, f)

        # Gradient step with projection
        f_new = project_simplex(f - lr * grad)

        conv_new = fast_auto_conv(f_new)
        M_new = conv_new.max()

        if M_new < M * (1.0 - 1e-10):
            f = f_new
            if no_improve == 0 and lr < lr_init * 0.9:
                lr = min(lr * 1.08, lr_init)
        else:
            lr *= 0.65

        if lr < min_lr:
            lr = min_lr

        # Adaptive restart
        if no_improve > 150:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 400.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            lr = max(lr_init * 0.4, min_lr * 20)
            no_improve = 0

        if verbose and it % 80 == 0:
            print(f"    SMD[{n}] it={it}, M={M:.8f}, best={best_obj:.8f}, "
                  f"beta={beta:.1f}, lr={lr:.2e}, no_imp={no_improve}")

    return best_f, best_obj


def pair_swap_search(f0, time_limit, verbose=False):
    """
    Greedy local search: for the peak lag k*, transfer mass from position i
    (high contribution a[k*-i]) to position j (low contribution a[k*-j]).
    """
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_auto_conv(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_auto_conv(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Consider top peaks
        npeaks = min(8, len(conv))
        peak_idx = np.argpartition(conv, -npeaks)[-npeaks:]
        peak_vals = conv[peak_idx]

        best_gain = 0.0
        best_i = -1
        best_j = -1

        for pk in peak_idx:
            k = int(pk)
            w = peak_vals[np.argmax(peak_idx == pk)] / max(peak_vals.sum(), 1e-15)

            # a[k-i] for each position i
            akm = np.zeros(n)
            valid = np.ones(n, dtype=bool)
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    akm[i] = f[ki]
                else:
                    akm[i] = 0.0

            # Source: maximize akm[i] * f[i] (potential mass to move that hurts peak)
            src_score = akm * f
            # Target: minimize akm[j] (position that helps least for this peak)
            tgt_score = akm.copy()

            i_cand = int(np.argmax(src_score))
            if src_score[i_cand] < 1e-12:
                continue

            j_mask = tgt_score.copy()
            j_mask[i_cand] = np.inf
            j_cand = int(np.argmin(j_mask))

            gain = (akm[i_cand] - akm[j_cand]) * w
            if gain > best_gain:
                best_gain = gain
                best_i = i_cand
                best_j = j_cand

        if best_i < 0 or best_gain < 1e-14:
            if no_improve > 30:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0015)
                no_improve = 0
            continue

        # Transfer mass
        eps = min(f[best_i] * 0.15, 0.002)
        f[best_i] -= eps
        f[best_j] += eps
        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs > 1e-15:
            f = f / fs

        if verbose and it % 200 == 0:
            print(f"    PSL[{n}] it={it}, M={M:.8f}, best={best_obj:.8f}, no_imp={no_improve}")

        if no_improve > 300:
            rng = np.random.default_rng(it * 23 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def fw_lp_step(f, M, n_top=5):
    """
    Single Frank-Wolfe step: solve LP to find best direction,
    then do line search. Returns new f and new M.
    """
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M * 1.01)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f, M, 0.0

    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-15:
        return f, M, 0.0
    g = g / g_sum

    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_auto_conv(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha > 0:
        f_new = (1.0 - best_alpha) * f + best_alpha * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        return f_new, best_val, best_alpha
    return f, M, 0.0


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, adaptive M relaxation."""
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_auto_conv(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        f_new, M_new, alpha = fw_lp_step(f, M * m_exp)

        if alpha > 1e-10 and M_new < best_obj - 1e-14:
            f = f_new
            M = M_new
            best_obj = M_new
            best_f = f.copy()
            no_improve = 0
            m_exp = 1.0
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 3.0)

        if no_improve > 20:
            rng = np.random.default_rng(it * 41 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_auto_conv(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

        if verbose and it % 5 == 0:
            print(f"    FW[{n}] it={it}, M={M:.8f}, best={best_obj:.8f}")

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)

    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Cosine pattern
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Multi-frequency
    mf = np.abs(np.cos(2 * np.pi * 3 * x / n) + 0.5 * np.cos(2 * np.pi * 7 * x / n)) + 0.1
    inits.append(mf / mf.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits.append(g / g.sum())

    # Sparse-like (every other element)
    sp = np.ones(n)
    sp[::2] = 2.0
    inits.append(sp / sp.sum())

    # Fibonacci-like spacing
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    fib = np.abs(np.cos(2 * np.pi * x * phi / n)) ** 2 + 0.01
    inits.append(fib / fib.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0.0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def multi_peak_equalize(f0, time_limit, verbose=False):
    """
    Focus on equalizing the top peaks simultaneously.
    Uses a combined gradient from all top peaks with inverse-value weighting.
    """
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    n = len(f)

    conv = fast_auto_conv(f)
    best_obj = conv.max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.003
    npeaks = min(12, len(conv))

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_auto_conv(f)
        M = conv.max()

            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Select top peaks
        peak_idx = np.argpartition(conv, -npeaks)[-npeaks:]
        peak_vals = conv[peak_idx]
        # Weight by inverse value (lower peaks get more weight to be raised... 
        # actually we want to lower all of them, weight proportional to value)
        weights_peak = peak_vals ** 2 / max((peak_vals ** 2).sum(), 1e-15)

        # Combined gradient

        for pi in range(len(peak_idx)):

            w = weights_peak[pi]
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * w * f[ki]

        f_new = project_simplex(f - lr * grad)
        conv_new = fast_auto_conv(f_new)
        M_new = conv_new.max()

        if M_new < M * (1.0 - 1e-10):
            f = f_new
        else:
            lr *= 0.7

        if no_improve > 100:
            rng = np.random.default_rng(it * 53 + 19)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            lr = 0.003
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    MPE[{n}] it={it}, M={M:.8f}, best={best_obj:.8f}")

    return best_f, best_obj

            t_alloc = min(150.0, t_rem * 0.5)
            print(f"  Multi-peak equalize, t={t_alloc:.0f}s")
            a_opt, obj = multi_peak_equalize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "multi_peak")

        t_rem = t_phase3_end - time.monotonic()
        if t_rem > 20:
            t_alloc = min(t_rem, 100.0)
            print(f"  Pair-swap round 2, t={t_alloc:.0f}s")
            a_opt, obj = pair_swap_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pair_swap2")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: FW-LP refinement (~300s) ===
    print("\n=== Phase 4: FW-LP Refinement ===")
    t_phase4_end = time.monotonic() + 300.0

    if best_seq is not None:
        n_best = len(best_seq)
        t_rem = t_phase4_end - time.monotonic()
        if t_rem > 30:
            t_alloc = min(180.0, t_rem)
            print(f"  FW-LP on best (n={n_best}), t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_lp")

            t_rem = t_phase4_end - time.monotonic()
            if t_rem > 20:
                # Try on a perturbed version
                noise = rng.standard_normal(n_best) * 0.003
                f_pert = project_simplex(best_seq + noise)
                t_alloc2 = min(t_rem, 80.0)
                print(f"  FW-LP perturbed, t={t_alloc2:.0f}s")
                a_opt2, obj2 = fw_lp_optimize(f_pert, time_limit=t_alloc2)
                try_seq(a_opt2, "fw_lp_pert")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final combined polish (~remaining time) ===
    print("\n=== Phase 5: Final Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)
        t_rem = deadline - time.monotonic()

        # Alternating SMD + pair swap
        while t_rem > 30:
            t_alloc_smd = min(60.0, t_rem * 0.4)
            t_alloc_ps = min(40.0, t_rem * 0.3)
            if t_alloc_smd < 10 or t_alloc_ps < 10:
                break

            print(f"  Polish cycle: SMD t={t_alloc_smd:.0f}s, PS t={t_alloc_ps:.0f}s")
            a_opt, obj = smooth_max_descent(best_seq, time_limit=t_alloc_smd,
                                             beta_init=100.0, beta_max=500.0,
                                             lr_init=0.002)
            try_seq(a_opt, "polish_smd")

            t_rem = deadline - time.monotonic()
            if t_rem < 20:
                break

            a_opt2, obj2 = pair_swap_search(best_seq, time_limit=t_alloc_ps)
            try_seq(a_opt2, "polish_ps")

            t_rem = deadline - time.monotonic()

        # Tiny perturbations
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            for pi in range(3):
                t_rem = deadline - time.monotonic()
                if t_rem < 10:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(15.0, t_rem - 5)
                if t_alloc < 5:
                    break
                a_opt, obj = smooth_max_descent(f_pert, time_limit=t_alloc,
                                                 beta_init=200.0, beta_max=800.0,
                                                 lr_init=0.001)
                try_seq(a_opt, f"tiny_pert_{pi}")

    print(f"  Phase 5 best: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
