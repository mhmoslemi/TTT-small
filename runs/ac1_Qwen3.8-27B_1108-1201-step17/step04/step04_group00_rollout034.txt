Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive LP-FW with structured
initializations, peak-focused polishing, and multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    # Clip small negative values from numerical error
    conv[np.abs(conv) < 1e-15] = 0
    return conv


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    k = np.nonzero(cond)[0][-1]
    theta = (cssv[k] - 1.0) / (k + 1.0)
    return np.maximum(v - theta, 0.0)


def get_active_lags(conv, top_k=50):
    """Get the indices of the top-k peaks in the convolution."""
    if len(conv) <= top_k:
        return np.arange(len(conv))
    return np.argpartition(conv, -top_k)[-top_k:]


def solve_lp_direction(f, M, n, lags=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in lags, g >= 0."""
    if lags is None:
        lags = np.arange(2 * n - 1)
    
    n_lags = len(lags)
    c = -np.ones(n)
    
    # Build constraint matrix
    # A[i, j] = f[j] if lag[i] - j is in [0, n-1]
    A_ub = np.zeros((n_lags, n))
    for j in range(n):
        # For each variable g_j, it appears in constraints for lags k where 0 <= k-j <= n-1
        # i.e., j <= k <= j+n-1
        # In our lags array, find which entries are in [j, j+n-1]
        if len(lags) > 0:
            lo = np.searchsorted(lags, j, side='left')
            hi = np.searchsorted(lags, j + n, side='left')
            if lo < hi:
                A_ub[lo:hi, j] = f[j]
    
    b_ub = np.full(n_lags, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_descent(f0, time_limit):
    """Coordinate descent for polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            cur_obj = fast_convolve(f).max()
            best_j_obj = cur_obj
            best_j_val = old_val

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or 1.0 - old_val < 1e-10:
                    continue
                scale = (1.0 - new_val) / (1.0 - old_val)
                ft = f.copy()
                ft[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= scale
                ft = ft / ft.sum()
                val = fast_convolve(ft).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / (1.0 - old_val)
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def generate_structured_init(n, seed=42):
    """Generate a structured initial sequence."""
    rng = np.random.default_rng(seed)
    choices = []
    
    # Uniform
    choices.append(np.ones(n) / n)
    
    # Raised cosine
    x = np.arange(n)
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    rc = np.maximum(rc, 0.01)
    choices.append(rc / rc.sum())
    
    # Gaussian
    sigma = n / 4.0
    g = np.exp(-0.5 * ((x - n/2) / sigma)**2)
    choices.append(g / g.sum())
    
    # Hamming-like
    h = 0.54 + 0.46 * np.cos(2 * np.pi * x / (n - 1)) if n > 1 else np.ones(1)
    h = np.maximum(h, 0.01)
    choices.append(h / h.sum())
    
    # Random (Dirichlet)
    alpha = np.ones(n) * 0.5
    d = rng.dirichlet(alpha)
    choices.append(d)
    
    # Bimodal
    b = np.exp(-0.5 * ((x - n*0.25) / (n*0.15))**2) + np.exp(-0.5 * ((x - n*0.75) / (n*0.15))**2)
    choices.append(b / b.sum())
    
    return choices


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""

structured initializations, peak-focused subgradient polishing,
coordinate descent, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    conv[conv < 0] = 0.0
    return conv


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    k = np.nonzero(cond)[0][-1]
    theta = (cssv[k] - 1.0) / (k + 1.0)
    return np.maximum(v - theta, 0.0)


def get_top_lags(conv, top_k):
    """Get indices of top-k peaks."""
    m = len(conv)
    if m <= top_k:
        return np.arange(m)
    return np.argpartition(conv, -top_k)[-top_k:]


def build_constraint_matrix(n, lags):
    """Build the (|lags| x n) constraint matrix for LP.
    A[i, j] = 1 if lag[i] - j is in [0, n-1] (i.e., f[j] contributes to conv at lag lags[i]).
    The actual constraint is sum_j A[i,j] * f[j] * g[j] <= M, but since we fix f,
    it becomes sum_j A[i,j]*f[j] * g[j] <= M, so the matrix is A[i,j]*f[j].
    """
    n_lags = len(lags)
    A = np.zeros((n_lags, n))
    for j in range(n):
        # g[j] contributes to conv[k] for k in [j, j+n-1]
        # Find which lags are in [j, j+n-1]
        lo = np.searchsorted(lags, j, side='left')
        hi = np.searchsorted(lags, j + n, side='left')
        if lo < hi:
            A[lo:hi, j] = 1.0
    return A


def solve_lp_direction(f, M, n, lags):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in lags, g >= 0."""
    n_lags = len(lags)
    c = -np.ones(n)

    A_base = build_constraint_matrix(n, lags)
    A_ub = A_base * f[None, :]  # scale each column by f[j]

    b_ub = np.full(n_lags, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, top_k=100):
    """LP-based Frank-Wolfe with adaptive constraint set and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        lags = get_top_lags(conv, top_k)
        g = solve_lp_direction(f, M * m_exp, n, lags)
        if g is None:
            m_exp *= 1.12
            if m_exp > 4.0:
                lags_full = np.arange(2 * n - 1)
                g = solve_lp_direction(f, M * 5.0, n, lags_full)
                m_exp = 1.0
                if g is None:
                    continue
            else:
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 1000033 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            else:
                m_exp = min(m_exp * 1.03, 2.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.25, 0.5, 0.75, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: M={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 3.5)

        if no_improve > 70:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.004 * min(no_improve // 15, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_optimize(f0, time_limit, lr=0.004, momentum=0.92, top_k=30, temp_init=5e-6):
    """Subgradient optimization with softmax-weighted peaks and Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_temp = temp_init
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        k_act = min(top_k, len(conv))
        peaks = get_top_lags(conv, k_act)

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Subgradient: sg[j] = sum_k w_k * 2 * f[k-j]
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp *= 1.003
        if no_improve > 0 and no_improve % 150 == 0:
            cur_lr *= 0.92

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0002)
            cur_temp = temp_init
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            cur_obj = fast_convolve(f).max()
            best_j_obj = cur_obj
            best_j_val = old_val

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                ft = f.copy()
                ft[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= scale
                ft_sum = ft.sum()
                if ft_sum < 1e-15:
                    continue
                ft = ft / ft_sum
                val = fast_convolve(ft).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    fs = f.sum()
                    if fs > 1e-15:
                        f = f / fs
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def generate_structured_init(n, seed=42):
    """Generate diverse structured initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    choices = []

    # Uniform
    choices.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    rc = np.maximum(rc, 0.005)
    choices.append(rc / rc.sum())

    # Gaussian (wide)
    sigma = n / 3.0
    g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    choices.append(g / g.sum())

    # Hamming window
    if n > 1:
        h = 0.54 + 0.46 * np.cos(2 * np.pi * x / (n - 1))
    else:
        h = np.ones(1)
    h = np.maximum(h, 0.005)
    choices.append(h / h.sum())

    # Dirichlet (concentrated)
    alpha = np.ones(n) * 0.3
    d = rng.dirichlet(alpha)
    choices.append(d)

    # Bimodal
    b = (np.exp(-0.5 * ((x - n * 0.3) / (n * 0.18)) ** 2) +
         np.exp(-0.5 * ((x - n * 0.7) / (n * 0.18)) ** 2))
    b = np.maximum(b, 0.001)
    choices.append(b / b.sum())

    # "Flat with edges" - suppress endpoints
    fe = np.ones(n)
    fade = n // 8
    if fade > 0:
        fe[:fade] *= np.linspace(0.3, 1.0, fade)
        fe[-fade:] *= np.linspace(1.0, 0.3, fade)
    choices.append(fe / fe.sum())

    return choices


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def peak_shave(f0, time_limit, n_trials=200):
    """Surgical peak shaving: find pairs that contribute most to the peak and swap mass."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    for _ in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))
        cur_max = conv[peak_k]

        # Find the pairs (i, peak_k - i) that contribute most to conv[peak_k]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        contributions = f[idx] * f[peak_k - idx]

        # Get top contributing indices
        top_contrib = np.argsort(contributions)[::-1][:5]

        # Try moving mass from high-contributing elements to low ones
        high_i = idx[top_contrib[0]]
        # Pick a target to receive mass (one with low contribution to this peak)
        low_candidates = np.argsort(contributions)[:5]
        for tc in low_candidates:
            low_i = idx[tc]
            if low_i == high_i:
                continue
            # Try moving epsilon from high_i to low_i
            for eps_frac in [0.01, 0.03, 0.05]:
                eps = f[high_i] * eps_frac
                if eps < 1e-15:
                    continue
                ft = f.copy()
                ft[high_i] -= eps
                ft[low_i] += eps
                ft = np.maximum(ft, 0)
                ft = ft / ft.sum()
                new_conv = fast_convolve(ft)
                new_max = new_conv.max()
                if new_max < cur_max - 1e-15:
                    f = ft
                    if new_max < best_obj:
                        best_obj = new_max
                        best_f = f.copy()
                    break

    return best_f, best_obj

        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_sg2 = min(rem * 0.6, 40)
            print(f"  SG (fine), t={t_sg2:.0f}s")
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg2, lr=0.001,
                                   momentum=0.95, top_k=15, temp_init=1e-6)
            try_seq(a_opt, "sg_fine")

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 25
        if rem > 10:
            t_cd = min(rem * 0.7, 50)
            print(f"  CD, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick LP-FW
        rem = deadline - time.monotonic() - 12
        if rem > 8:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 30), top_k=50)
            try_seq(a_opt, "final_lpfw")

        # Tiny perturbations + quick SG
        for attempt in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 6:
                break
            noise = rng.standard_normal(n_best) * 0.0008
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = sg_optimize(f_pert, time_limit=min(rem, 12), lr=0.0008,
                                   momentum=0.95, top_k=10, temp_init=5e-7)
            try_seq(a_opt, f"final_pert_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
