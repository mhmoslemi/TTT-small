def project_simplex(v):
    n=v.size
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho_candidates=np.arange(1,n+1)
    cond = u*rho_candidates > (cssv - 1.0)
    if not cond.any(): return np.ones(n)/n
    rho=np.nonzero(cond)[0][-1]
    theta=(cssv[rho]-1.0)/(rho+1.0)
    return np.maximum(v-theta,0.0)
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a.

The implementation combines several complementary strategies:
  1. FFT-based exact gradients for smooth surrogates:
       f_beta(a) = (1/beta) * logsumexp(beta * (a*a))
     and L2 flatteners sum((a*a)^2).
  2. L-BFGS in a reduced softmax parameterization (x[0],...,x[n-2], 0).
  3. Active-set least-squares polishing of the highest peaks.
  4. Adam/multiplicative polishing with peak-focused and smooth weights.
  5. A short Frank-Wolfe / LP phase carried over from the previous approach.

All helpers are top-level; no nested closures or lambdas are used.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize, least_squares
from scipy.sparse import csr_matrix

# Globals used by the L-BFGS callback / objective bookkeeping.
_LBF_DEADLINE = 0.0
_OPT_BEST_SEQ = None
_OPT_BEST_SCORE = float("inf")


def _lbf_callback(k):
    return time.monotonic() < _LBF_DEADLINE


def _maybe_record_best(a, c):
    global _OPT_BEST_SEQ, _OPT_BEST_SCORE
    if a is None or c is None:
        return
    try:
        s = 2.0 * float(a.size) * float(np.max(c))
    except Exception:
        return
    if np.isfinite(s) and s < _OPT_BEST_SCORE:
        _OPT_BEST_SCORE = s
        _OPT_BEST_SEQ = a.copy()


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def convolve_fft(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    if fft_size is None:
        fft_size = next_pow2(L)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:L]
    c = np.real_if_close(c, tol=1000).astype(np.float64)
    c = np.maximum(c, 0.0)
    return c


def grad_from_weights(w, a, fft_size=None):
    """
    Gradient of sum_k w_k * (a*a)_k with respect to a.

    For each k, d(a*a)_k/da_i = 2*a_{k-i} when 0 <= k-i < n, else 0.
    This can be computed by one short convolution.
    """
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    if w.size != L:
        # Defensive: resample weights to the correct length.
        if w.size == 1:
            w = np.full(L, float(w[0]) / L)
        else:
            w = np.interp(
                np.linspace(0, 1, L),
                np.linspace(0, 1, w.size),
                w
            )
    if fft_size is None:
        fft_size = next_pow2(L)

    R = w[::-1]
    Rf = np.fft.rfft(R, fft_size)
    Af = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(Rf * Af, fft_size)[:L]
    conv = np.real_if_close(conv, tol=1000).astype(np.float64)

    idx = L - 1 - np.arange(n)
    g = 2.0 * conv[idx]
    return g


def softmax(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x.copy()
    x = x - np.max(x)
    e = np.exp(x)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full_like(x, 1.0 / x.size)
    return e / s


def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full(a.size, 1.0 / a.size)
    return a / s


def score_from_a(a):
    """Own fast objective; scale-invariant, assumes non-negative finite a."""
    a = normalize(a)
    if a.size == 0:
        return float("inf")
    c = convolve_fft(a)
    mc = float(np.max(c)) if c.size else float("inf")
    if not np.isfinite(mc):
        return float("inf")
    return 2.0 * float(a.size) * mc


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(0.0, float(v[0]))])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def best_from_candidate(old_seq, old_score, cand, label=""):
    try:
        a = normalize(cand)
    except Exception:
        return old_seq, old_score
    s_own = score_from_a(a)
    if not np.isfinite(s_own) or s_own >= old_score - 1e-12:
        return old_seq, old_score
    try:
        s_eval = float(evaluate_sequence(a.tolist()))
    except Exception:
        s_eval = s_own
    if np.isfinite(s_eval) and s_eval < old_score:
        # print(f"NEW BEST [{label}]: {s_eval:.10f} (n={a.size})")
        print(f"NEW BEST [{label}]: {s_eval:.10f} (n={a.size})")
        return a.copy(), s_eval
    return old_seq, old_score


def symmetrize(a):
    a = normalize(a)
    return normalize((a + a[::-1]) * 0.5)


def upscale_by_interp(a, m):
    a = normalize(a)
    m = int(max(1, m))
    n = a.size
    if m == n:
        return a.copy()
    if n == 1:
        return np.full(m, 1.0 / m)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    v = np.interp(x_new, x_old, a)
    v = np.maximum(v, 1e-15)
    return normalize(v)


def upscale_by_block(a, q):
    a = normalize(a)
    q = int(max(0, q))
    if q == 0:
        return a.copy()
    b = np.ones(q + 1) / float(q + 1)
    res = np.convolve(a, b)
    return normalize(res)


def solve_lp(f, M):
    """
    Solve LP:
      max sum(g)
      s.t. (f * g)_k <= M for k=0..2n-2, g >= 0.
    """
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    L = 2 * n - 1
    c = -np.ones(n)

    try:
        ks = np.add.outer(np.arange(n), np.arange(n))
        rs = ks.ravel()
        cs = np.repeat(np.arange(n), n)
        ds = np.tile(f, n)
        A_ub = csr_matrix((ds, (rs, cs)), shape=(L, n))
        b_ub = np.full(L, float(M))
        bounds = [(0.0, None) for _ in range(n)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    f = normalize(f0)
    n = f.size
    c = convolve_fft(f)
    M = float(np.max(c)) if c.size else 1.0
    best_obj = M
    best_f = f.copy()

    deadline = time.monotonic() + max(1.0, time_limit)
    no_improve = 0
    it = 0
    m_exp = 1.0

    alphas = [0.05, 0.10, 0.20, 0.30, 0.50, 0.70, 1.00]

    while time.monotonic() < deadline:
        it += 1
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        gs = g.sum()
        if gs <= 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 35:
                break
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = normalize(ft)
            val = float(np.max(convolve_fft(ft)))
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(10):
                mid = 0.5 * (lo + hi)
                ft = (1.0 - mid) * f + mid * g
                ft = normalize(ft)
                val = float(np.max(convolve_fft(ft)))
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = normalize(f)
            c = convolve_fft(f)
            M = float(np.max(c))
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 45:
            rng = np.random.default_rng(12345 + 17 * it)
            scale = 0.002 + 0.003 * min(5.0, no_improve / 45.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            c = convolve_fft(f)
            M = float(np.max(c))
            m_exp = 1.0
            no_improve = 0

    return best_f, (2.0 * n * best_obj if np.isfinite(best_obj) else float("inf"))


def lbgfs_obj(x, beta, mode, l2_weight):
    """
    L-BFGS objective/gradient in reduced softmax parameterization.
    x has length n-1; the last softmax logit is fixed to 0.
    """
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return (0.0, np.array([], dtype=np.float64))
    if not np.all(np.isfinite(x)):
        return (1e100, np.zeros_like(x))

    n = x.size + 1
    xf = np.empty(n, dtype=np.float64)
    xf[:-1] = x
    xf[-1] = 0.0

    a = softmax(xf)
    c = convolve_fft(a)
    if c.size == 0:
        return (1e100, np.zeros_like(x))
    _maybe_record_best(a, c)

    mode = int(mode)
    beta = float(beta)
    l2_weight = float(l2_weight)

    if mode == 1:
        val = l2_weight * float(np.sum(c * c))
        g = l2_weight * grad_from_weights(2.0 * c, a)
    elif mode == 2:
        if beta <= 1e-8:
            b = np.zeros_like(c)
        else:
            b = beta * c
        bmax = float(np.max(b))
        e = np.exp(b - bmax)
        Z = float(e.sum())
        if Z <= 0.0 or not np.isfinite(Z):
            w = np.full(c.size, 1.0 / c.size)
            f0 = 0.0
        else:
            w = e / Z
            f0 = (bmax + float(np.log(Z))) / max(beta, 1e-12)
        g = grad_from_weights(w, a)
        val = f0
        if l2_weight != 0.0:
            val += l2_weight * float(np.sum(c * c))
            g = g + l2_weight * grad_from_weights(2.0 * c, a)
    else:
        if beta <= 1e-8:
            # Very smooth fallback: minimize sum(c^2).
            val = float(np.sum(c * c))
            g = grad_from_weights(2.0 * c, a)
        else:
            b = beta * c
            bmax = float(np.max(b))
            e = np.exp(b - bmax)
            Z = float(e.sum())
            if Z <= 0.0 or not np.isfinite(Z):
                w = np.full(c.size, 1.0 / c.size)
                val = 0.0
            else:
                w = e / Z
                val = (bmax + float(np.log(Z))) / beta
            g = grad_from_weights(w, a)

    if not np.all(np.isfinite(g)):
        g = np.zeros_like(g)

    return val, g[:-1]


def run_lbgfs(a0, mode, beta, time_limit, l2_weight=1.0, label=""):
    global _LBF_DEADLINE, _OPT_BEST_SEQ, _OPT_BEST_SCORE

    if time_limit <= 0.0:
        a0 = normalize(a0)
        return a0, score_from_a(a0)

    a0 = normalize(a0)
    n = a0.size
    af = np.maximum(a0, 1e-12)
    af = normalize(af)

    s0 = score_from_a(af)
    _OPT_BEST_SCORE = s0
    _OPT_BEST_SEQ = af.copy()

    xf = np.log(af)
    if n > 1:
        y0 = xf[:-1] - xf[-1]
    else:
        y0 = np.array([], dtype=np.float64)
    if not np.all(np.isfinite(y0)):
        y0 = np.zeros(max(0, n - 1), dtype=np.float64)

    _LBF_DEADLINE = time.monotonic() + max(1.0, float(time_limit))

    try:
        res = minimize(
            lbgfs_obj,
            y0,
            args=(float(beta), int(mode), float(l2_weight)),
            method="L-BFGS-B",
            options={
                "maxiter": 5000,
                "maxfun": 20000,
                "ftol": 1e-12,
                "gtol": 1e-8,
                "disp": False,
                "callback": _lbf_callback,
            },
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            a_fin = softmax(np.concatenate((np.asarray(res.x, dtype=np.float64), [0.0])))
            a_fin = normalize(a_fin)
            s_fin = score_from_a(a_fin)
            if np.isfinite(s_fin) and s_fin < _OPT_BEST_SCORE:
                _OPT_BEST_SCORE = s_fin
                _OPT_BEST_SEQ = a_fin.copy()
    except Exception as e:
        print(f"  L-BFGS [{label}] failed: {e}")

    if _OPT_BEST_SEQ is None:
        return a0, s0
    return _OPT_BEST_SEQ.copy(), float(_OPT_BEST_SCORE)


def _ls_residual(z, active, n, c_scale, t_weight, reg, x0):
    size = int(active.size) + 1 + max(0, n - 1)
    if not np.all(np.isfinite(z)):
        return np.full(size, 1e100)

    y = z[:n - 1]
    T = float(z[n - 1])
    x = np.empty(n, dtype=np.float64)
    x[:n - 1] = y
    x[n - 1] = 0.0

    if not np.all(np.isfinite(x)) or not np.isfinite(T):
        return np.full(size, 1e100)

    a = softmax(x)
    c = convolve_fft(a)

    res = np.empty(size, dtype=np.float64)
    res[:active.size] = (c[active] - T) / c_scale
    res[active.size] = t_weight * (T / c_scale)
    if n > 1:
        res[active.size + 1:] = reg * (y - x0)
    return res


def _ls_jac(z, active, n, c_scale, t_weight, reg, x0):
    m = int(active.size)
    nvar = n
    size = m + 1 + max(0, n - 1)

    if not np.all(np.isfinite(z)):
        return np.zeros((size, nvar), dtype=np.float64)

    y = z[:n - 1]
    x = np.empty(n, dtype=np.float64)
    x[:n - 1] = y
    x[n - 1] = 0.0

    if not np.all(np.isfinite(x)):
        return np.zeros((size, nvar), dtype=np.float64)

    a = softmax(x)
    J = np.zeros((size, nvar), dtype=np.float64)

    for r, k in enumerate(active):
        k = int(k)
        d = np.zeros(n, dtype=np.float64)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            idx = np.arange(js, je + 1)
            d[idx] = 2.0 * a[k - idx]
        mean_d = float(np.dot(a, d))
        g = a * (d - mean_d)
        if n > 1:
            J[r, :n - 1] = g[:n - 1]
        J[r, n - 1] = -1.0 / c_scale

    J[m, n - 1] = t_weight / c_scale

    if n > 1 and reg != 0.0:
        idx = np.arange(n - 1)
        J[m + 1 + idx, idx] = reg

    return J


def active_ls_polish(a0, time_limit, m=100, t_weight=0.10, reg=0.12, max_runs=2):
    a = normalize(a0)
    n = a.size
    best_seq = a.copy()
    best_score = score_from_a(a)

    deadline = time.monotonic() + max(1.0, float(time_limit))
    L = 2 * n - 1
    m = int(max(1, min(m, L)))

    for run in range(int(max_runs)):
        if time.monotonic() >= deadline:
            break

        c = convolve_fft(a)
        if m >= L:
            active = np.arange(L)
        else:
            active = np.argpartition(c, -m)[:m]

        c_scale = float(np.max(c[active])) if active.size else 1.0
        if c_scale <= 0.0 or not np.isfinite(c_scale):
            c_scale = 1e-12

        af = np.maximum(a, 1e-15)
        af = normalize(af)
        xf = np.log(af)
        if n > 1:
            y0 = xf[:-1] - xf[-1]
        else:
            y0 = np.array([], dtype=np.float64)
        if not np.all(np.isfinite(y0)):
            y0 = np.zeros(max(0, n - 1), dtype=np.float64)

        z0 = np.concatenate((y0, [c_scale]))

        try:
            res = least_squares(
                _ls_residual,
                z0,
                args=(active, n, c_scale, float(t_weight), float(reg), y0),
                jac=_ls_jac,
                method="trf",
                max_nfev=80,
                xtol=1e-12,
                ftol=1e-12,
                gtol=1e-12,
            )
            xnew = np.empty(n, dtype=np.float64)
            xnew[:n - 1] = np.asarray(res.x[:n - 1], dtype=np.float64)
            xnew[n - 1] = 0.0
            a_new = normalize(softmax(xnew))
            s_new = score_from_a(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_seq = a_new.copy()
                a = a_new
            else:
                break
        except Exception as e:
            print(f"  Active-LS failed: {e}")
            break

    return best_seq, float(best_score)


def adam_optimize(a0, time_limit, mode=0, beta=100.0, top_k=25, temp=1e-5, lr=0.04):
    a0 = normalize(a0)
    n = a0.size
    L = 2 * n - 1
    fft_size = next_pow2(L)

    af = np.maximum(a0, 1e-300)
    af = normalize(af)
    x = np.log(af)
    x -= x.mean()

    best_seq = af.copy()
    best_score = score_from_a(af)
    best_x = x.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    no_improve = 0

    beta_cur = float(beta)
    temp_cur = float(temp)
    lr_cur = float(lr)

    deadline = time.monotonic() + max(1.0, float(time_limit))

    while time.monotonic() < deadline:
        t += 1
        a = softmax(x)
        c = convolve_fft(a, fft_size)
        if c.size == 0:
            break

        if mode == 2:
            w = 2.0 * c
        elif mode == 1:
            wk = int(max(1, min(L, top_k + t // 300)))
            active = np.argpartition(c, -wk)[:wk]
            pv = c[active]
            pmax = float(np.max(pv))
            if temp_cur > 1e-14:
                e = np.exp((pv - pmax) / temp_cur)
                s = e.sum()
                if s <= 0.0 or not np.isfinite(s):
                    e = np.zeros_like(pv)
                    e[int(np.argmax(pv))] = 1.0
                    s = 1.0
                e = e / s
            else:
                order = np.argsort(pv)[::-1]
                e = np.zeros(wk, dtype=np.float64)
                e[order] = 1.0 / (np.arange(wk, dtype=np.float64) + 1.0)
                s = e.sum()
                if s <= 0.0 or not np.isfinite(s):
                    e = np.zeros_like(pv)
                    e[int(np.argmax(pv))] = 1.0
                    s = 1.0
                e = e / s
            w = np.zeros(L, dtype=np.float64)
            w[active] = e
        else:
            if beta_cur <= 1e-8:
                w = np.full(L, 1.0 / L)
            else:
                b = beta_cur * c
                bmax = float(np.max(b))
                e = np.exp(b - bmax)
                s = float(e.sum())
                if s <= 0.0 or not np.isfinite(s):
                    w = np.full(L, 1.0 / L)
                else:
                    w = e / s

        g_a = grad_from_weights(w, a, fft_size)
        mean_term = float(np.dot(a, g_a))
        g_x = a * (g_a - mean_term)

        if not np.all(np.isfinite(g_x)):
            g_x = np.zeros(n, dtype=np.float64)

        m = 0.90 * m + 0.10 * g_x
        v = 0.999 * v + 0.001 * (g_x * g_x)

        bias1 = 1.0 - 0.90 ** t
        bias2 = 1.0 - 0.999 ** t
        step = lr_cur * (m / bias1) / (np.sqrt(v / bias2) + 1e-8)
        x_new = x - step

        if not np.all(np.isfinite(x_new)):
            x = best_x.copy()
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            continue

        x = x_new
        x -= x.mean()

        s = 2.0 * float(n) * float(np.max(c))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a.copy()
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if mode == 0:
            beta_cur = min(30000.0, beta_cur * 1.005)
            if no_improve > 80:
                beta_cur = min(30000.0, beta_cur * 1.25)
        elif mode == 1:
            if no_improve > 120:
                temp_cur = max(1e-7, temp_cur * 2.0)

        if no_improve > 220:
            rng = np.random.default_rng(987654 + t)
            noise = 0.01 + 0.002 * min(20.0, no_improve / 50.0)
            x = best_x + rng.standard_normal(n) * noise
            x -= x.mean()
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            lr_cur = max(0.008, lr_cur * 0.75)
            if mode == 0:
                beta_cur = max(50.0, beta_cur * 0.5)
            elif mode == 1:
                temp_cur = max(1e-7, temp_cur * 2.0)
            no_improve = 0

    return best_seq, float(best_score)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 60.0)
    rng = np.random.default_rng(int(seed))

    best_seq = None
    best_score = float("inf")

    # Load parent if present.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.sum() > 1e-12:
                parent = normalize(p)
                if parent.size > 1200:
                    parent = upscale_by_interp(parent, 1024)
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(384, dtype=np.float64) / 384.0

    # Initial candidates.
    for cand, lab in (
        (parent, "parent"),
        (parent[::-1], "parent_rev"),
        (symmetrize(parent), "parent_sym"),
    ):
        best_seq, best_score = best_from_candidate(best_seq, best_score, cand, lab)

    if best_seq is None:
        best_seq = parent.copy()
        best_score = score_from_a(best_seq)
        try:
            es = evaluate_sequence(best_seq.tolist())
            if np.isfinite(es):
                best_score = float(es)
        except Exception:
            pass

    print(f"Initial best: {best_score:.10f} (n={best_seq.size})")

    def remaining():
        return max(0.0, deadline - time.monotonic())

    # Very quick active least-squares polish.
    if remaining() > 80:
        alloc = min(35.0, remaining() - 20.0)
        if alloc > 5:
            cand, _ = active_ls_polish(best_seq, alloc, m=60, t_weight=0.05, reg=0.25, max_runs=2)
            best_seq, best_score = best_from_candidate(best_seq, best_score, cand, "quick_ls")

    # Main L-BFGS schedule.
    lbf_configs = [
        # (time, mode, beta, l2_weight, label)
        (120.0, 1, 0.0, 1.0, "l2"),
        (80.0, 0, 20.0, 0.0, "soft20"),
        (120.0, 2, 100.0, 0.2, "soft100_l2"),
        (150.0, 0, 500.0, 0.0, "soft500"),
        (220.0, 2, 2000.0, 0.1, "soft2000_l2"),
    ]

    for alloc, mode, beta, l2w, lab in lbf_configs:
        rem = remaining()
        if rem < 700:
            break
        t_alloc = min(alloc, rem - 650)
        if t_alloc < 25:
            break
        cand, _ = run_lbgfs(best_seq, mode, beta, t_alloc, l2w, lab)
        best_seq, best_score = best_from_candidate(best_seq, best_score, cand, f"lbf_{lab}")
        print(f"After LBF [{lab}]: {best_score:.10f} (n={best_seq.size})")

    # Try a short Adam L2 smoother.
    rem = remaining()
    if rem > 350:
        t_alloc = min(80.0, rem - 300)
        if t_alloc > 20:
            cand, _ = adam_optimize(best_seq, t_alloc, mode=2, beta=0.0, lr=0.05)
            best_seq, best_score = best_from_candidate(best_seq, best_score, cand, "adam_l2")

    # Explore nearby lengths by interpolation / block convolution.
    rem = remaining()
    if rem > 550:
        n0 = int(best_seq.size)
        targets = []
        if n0 < 768:
            targets.append(int(min(1024, max(256, n0 * 2))))
        if n0 > 256:
            targets.append(int(max(128, n0 // 2)))
        # Average target can be helpful.
        if n0 < 1024:
            targets.append(int(max(192, int(0.75 * n0))))

        seen = set()
        for target in targets:
            if target in seen or target <= 0:
                continue
            seen.add(target)
            rem = remaining()
            if rem < 250:
                break

            cand = upscale_by_interp(best_seq, target)
            cand, _score = best_from_candidate(best_seq, best_score, cand, f"upscale_{target}")

            t_alloc = min(70.0, remaining() - 220)
            if t_alloc > 20:
                cand2, _ = run_lbgfs(cand, 0, 200.0, t_alloc, 0.0, f"upscale_{target}_lbf")
                best_seq, best_score = best_from_candidate(best_seq, best_score, cand2, f"upscale_{target}_lbf")

            # Simple block extension, capped length.
            qb = min(128, max(16, target // 4))
            cand3 = upscale_by_block(best_seq, qb)
            if cand3.size > 1200:
                cand3 = upscale_by_interp(cand3, 1024)
            cand3, _score = best_from_candidate(best_seq, best_score, cand3, "block_ext")

            t_alloc = min(45.0, remaining() - 200)
            if t_alloc > 15:
                cand4, _ = run_lbgfs(cand3, 0, 300.0, t_alloc, 0.0, "block_ext_lbf")
                best_seq, best_score = best_from_candidate(best_seq, best_score, cand4, "block_ext_lbf")

    # Active LSB polish on the current best.
    ls_configs = [
        # (m, t_weight, reg, time, label)
        (60, 0.04, 0.30, 80.0, "ls60"),
        (120, 0.10, 0.18, 100.0, "ls120"),
        (200, 0.05, 0.12, 120.0, "ls200"),
    ]
    for m, tw, reg, alloc, lab in ls_configs:
        rem = remaining()
        if rem < 250:
            break
        t_alloc = min(alloc, rem - 220)
        if t_alloc < 20:
            break
        cand, _ = active_ls_polish(best_seq, t_alloc, m=m, t_weight=tw, reg=reg, max_runs=2)
        best_seq, best_score = best_from_candidate(best_seq, best_score, cand, lab)

    # Adam peak-focused polish.
    rem = remaining()
    if rem > 300:
        t_alloc = min(160.0, rem - 260)
        if t_alloc > 25:
            cand, _ = adam_optimize(best_seq, t_alloc, mode=1, beta=0.0, top_k=25, temp=1e-5, lr=0.05)
            best_seq, best_score = best_from_candidate(best_seq, best_score, cand, "adam_top")

    # Adam smooth high-beta polish.
    rem = remaining()
    if rem > 260:
        cmax = float(np.max(convolve_fft(best_seq))) if best_seq.size else 1e-8
        if cmax <= 0.0 or not np.isfinite(cmax):
            cmax = 1e-8
        beta_est = float(min(20000.0, max(100.0, 6.0 / cmax)))
        t_alloc = min(140.0, rem - 220)
        if t_alloc > 25:
            cand, _ = adam_optimize(best_seq, t_alloc, mode=0, beta=beta_est, lr=0.03)
            best_seq, best_score = best_from_candidate(best_seq, best_score, cand, "adam_soft")

    # Frank-Wolfe / LP phase, especially useful for final structural moves.
    rem = remaining()
    if rem > 320:
        t_alloc = min(260.0, rem - 270)
        if t_alloc > 30:
            cand, _ = fw_lp_optimize(best_seq, t_alloc)
            best_seq, best_score = best_from_candidate(best_seq, best_score, cand, "fwlp")

    # Final L-BFGS with high beta and a little L2 regularization.
    rem = remaining()
    if rem > 180:
        cmax = float(np.max(convolve_fft(best_seq))) if best_seq.size else 1e-8
        if cmax <= 0.0 or not np.isfinite(cmax):
            cmax = 1e-8
        beta_final = float(min(30000.0, max(500.0, 10.0 / cmax)))
        t_alloc = min(120.0, rem - 140)
        if t_alloc > 20:
            cand, _ = run_lbgfs(best_seq, 2, beta_final, t_alloc, 0.1, "final_lbf")
            best_seq, best_score = best_from_candidate(best_seq, best_score, cand, "final_lbf")

    # A tiny perturbation + LS can avoid exact local stagnation.
    rem = remaining()
    if rem > 120:
        pert = project_simplex(best_seq + rng.standard_normal(best_seq.size) * 0.002)
        pert, _ = best_from_candidate(best_seq, best_score, pert, "final_perturb")
        t_alloc = min(60.0, remaining() - 80)
        if t_alloc > 15:
            cand, _ = active_ls_polish(pert, t_alloc, m=100, t_weight=0.08, reg=0.20, max_runs=2)
            best_seq, best_score = best_from_candidate(best_seq, best_score, cand, "perturb_ls")

    # Final symmetrization check.
    best_seq, best_score = best_from_candidate(best_seq, best_score, symmetrize(best_seq), "final_sym")

    # Fallback.
    if best_seq is None:
        best_seq = parent
        best_score = score_from_a(best_seq)

    # Make sure final sequence is a clean python list of floats.
    final_arr = normalize(best_seq)
    try:
        final_eval = float(evaluate_sequence(final_arr.tolist()))
        if np.isfinite(final_eval):
            print(f"FINAL SCORE: {final_eval:.10f} (n={final_arr.size})")
        else:
            print(f"FINAL SCORE (own): {best_score:.10f} (n={final_arr.size}, eval non-finite)")
    except Exception as e:
        print(f"FINAL SCORE (own): {best_score:.10f} (n={final_arr.size}, eval error {e})")

    return [float(x) for x in final_arr]
