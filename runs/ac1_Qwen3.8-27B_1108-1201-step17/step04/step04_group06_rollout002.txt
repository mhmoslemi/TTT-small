The objective is: f(a) = 2n * max_k (a*a)[k] / (sum a)^2

With sum(a) = 1: f(a) = 2n * max_k (a*a)[k]

We want to minimize this. The constraint is a ≥ 0, sum(a) = 1.

At the optimum, by KKT conditions:
- There exists a set S of "active" lags where (a*a)[k] = M (the max)
- For coordinates where a[i] > 0: the derivative of M with respect to a[i] should be constant (otherwise we could improve)
- The derivative of M with respect to a[i] is 2 * sum_{k in S} w_k * a[k-i] (where w_k are the weights from the subgradient)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage optimization:
1. Exponentiated gradient (multiplicative weights) with adaptive momentum and temperature
2. Multi-length exploration (resampling to different n values)
3. Coordinate descent polishing
4. Simulated annealing style restarts
Key insight: use sharp subgradients (low temperature) for exploitation and
high temperature for exploration, with momentum to escape shallow local minima.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c) using top-k peaks."""
    n = len(a)
    len_c = len(c)

    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()

    scaled = peek / temperature
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    ws = exp_vals.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w = exp_vals / ws

    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

    return sg


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.002, temp_decay=0.9995,
                eta_decay=0.9999, top_k_frac=0.15, momentum=0.4,
                restart_every=4000, restart_noise=0.003, verbose=False):
    """Exponentiated gradient with momentum and adaptive restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    log_mom = np.zeros(n)
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    best_temp = temp0

    while time.monotonic() - t0 < time_limit:
        it += 1

        sg = compute_subgradient(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max < 1e-18:
            break

        log_update = -eta * sg / sg_max
        if momentum > 0:
            log_mom = momentum * log_mom + (1.0 - momentum) * log_update
            log_update = log_mom

        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"      EG iter {it}: obj={2*n*best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new
        eta *= eta_decay
        temp *= temp_decay

        if no_improve >= restart_every:
            rng_local = np.random.default_rng(it * 999983 + 13)
            noise_scale = restart_noise * (1.0 + 0.5 * (no_improve // restart_every))
            a = best_a + rng_local.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = max(eta0 * 0.4, 0.05)
            temp = best_temp * (0.5 + 0.5 * rng_local.random())

    return best_a.copy(), 2.0 * n * best_obj

    """Coordinate descent: try multiplicative perturbations on each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    a = a / s
    n = len(a)

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 >= time_limit:
                break
            old_val = a[i]
            if old_val < 1e-12:

                    a[i] = 1.0 / n * f
                    s2 = a.sum()
                    a = a / s2
                    c2 = fast_convolve(a)
                    obj2 = c2.max()
                    if obj2 < best_obj - 1e-14:
                        best_obj = obj2
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a = best_a.copy()
                continue

            for f in factors:
                a[i] = old_val * f
                s2 = a.sum()
                if s2 < 1e-15:
                    a = best_a.copy()
                    break
                a = a / s2

    return best_a.copy(), 2.0 * n * best_obj


def resample_sequence(a, target_n):
    """Resample sequence to target length using interpolation."""
    n_orig = len(a)
    if n_orig == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, n_orig)
    x_new = np.linspace(0, 1, target_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def annealing_search(a0, time_limit, verbose=False):
    """Simulated annealing style search with multiplicative perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 1e10
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    current_obj = best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(123456)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        temp = 0.5 * (1 - progress) ** 2 + 0.01

        # Perturb a random subset
        n_perturb = max(1, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)
        log_factors = rng.standard_normal(n_perturb) * temp

        a_trial = a.copy()
        for idx in range(n_perturb):
            a_trial[coords[idx]] *= np.exp(log_factors[idx])
            if a_trial[coords[idx]] < 0:
                a_trial[coords[idx]] = 0

        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s

        c_trial = fast_convolve(a_trial)
        obj_trial = c_trial.max()

        # Accept if better or with probability exp(-(obj_trial - current)/temp)
        if obj_trial < current_obj:
            a = a_trial
            current_obj = obj_trial
            c = c_trial
            if obj_trial < best_obj - 1e-13:
                best_obj = obj_trial
                best_a = a.copy()
                no_improve = 0
                if verbose:
                    print(f"      Anneal iter {it}: best={2*n*best_obj:.10f}")
        else:
            ratio = np.exp(-(obj_trial - current_obj) / max(temp, 1e-10))
            if rng.random() < ratio:
                a = a_trial
                current_obj = obj_trial
                c = c_trial

        no_improve += 1
        if no_improve > 5000:
            break

    return best_a.copy(), 2.0 * n * best_obj
