import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_objective(a, tau):
    """Return (value, gradient) of smooth max approximation."""
    c = fast_convolve(a)
    scaled = tau * c
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    value = (c.max() + np.log(exp_vals.sum()) / tau)  # Actually: (1/tau) * log(sum exp(tau*c))
    # value = (1/tau) * log(sum_k exp(tau * c_k))
    # = (1/tau) * (tau * c.max() + log(sum_k exp(tau * (c_k - c.max()))))
    # = c.max() + (1/tau) * log(sum_k exp(tau * (c_k - c.max())))
    
    # Gradient: (2/tau) * sum_j w_{i+j} * a_j

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Smoothed gradient descent with log-sum-exp (adaptive temperature)
2. LP-based Frank-Wolfe with reduced constraints
3. Multi-length exploration
4. Local search polishing
Key innovation: efficient exact gradient computation for the log-sum-exp
smooth approximation of the peak autocorrelation, enabling fast and
precise optimization in high dimensions.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, tau):
    """
    Compute the value and gradient of (1/tau)*log(sum_k exp(tau*c_k))
    where c = a * a (linear convolution).
    
    Returns (value, gradient, c)
    """
    n = len(a)
    c = fast_convolve(a)
    
    # Compute value and softmax weights
    c_shifted = c - c.max()
    scaled = tau * c_shifted
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    
    # Value: (1/tau) * log(sum_k exp(tau * c_k))
    # = c.max() + (1/tau) * log(sum_k exp(tau * (c_k - c.max())))
    value = c.max() + np.log(sum_exp) / tau
    
    # Gradient: g_i = 2 * sum_j w_{i+j} * a_j
    # Use direct computation (O(n^2))
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(a, w[i:i+n])
    
    return value, grad, c

        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        
        # Adaptive parameters
        tau = tau_start * (tau_end / tau_start) ** progress
        lr = lr_start * (1.0 - 0.7 * progress)
        mom = momentum + 0.08 * progress
        
        c = fast_convolve(a)
        cur_obj = c.max()

        c_shifted = c - c.max()
        scaled = tau * c_shifted
        exp_vals = np.exp(scaled)

        # Update with momentum
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Evaluate
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-13:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    LSE-Grad iter {it}: obj={best_obj:.10f}, tau={tau:.0f}, lr={lr:.4f}")
        elif obj_new < cur_obj - 1e-13:
            a = a_new
            no_improve += 1
        else:
            v *= 0.5
            no_improve += 1
        
        if no_improve > 150:
            # Restart with perturbation
            scale = 0.002 + 0.003 * (no_improve // 300)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            no_improve = 0
            tau = max(tau_start, tau * 0.5)  # Reset temperature
    
    return best_a, best_obj


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def reduced_linearized_lp(f, conv_f, top_k=20):
    """Solve reduced LP using only top_k lags."""
    n = len(f)
    if top_k >= len(conv_f):
        peaks = np.arange(len(conv_f))
    else:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    peaks = np.sort(peaks)
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    B_full = build_conv_matrix(f)
    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B_full[peaks, :]
    b_ub = -conv_f[peaks]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def full_linearized_lp(f, conv_f):
    """Solve full LP with all constraints."""
    n = len(f)
    c = np.zeros(n + 1)
    c[0] = 1.0

def fw_with_lp(f0, time_limit, use_full=False, top_k=25, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        if use_full:
            g = full_linearized_lp(f, conv)
        else:
            g = reduced_linearized_lp(f, conv, top_k=top_k)
        
        if g is None:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        
        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        
        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = current_obj
        alphas = np.linspace(0.05, 1.0, 15)
        for alpha in alphas:
            f_new = (1.0 - alpha) * f + alpha * target
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new = f_new / fs
            obj = fast_convolve(f_new).max()
            if obj < best_val:
                best_val = obj
                best_alpha = alpha
        
        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.07)
            hi = min(1.0, best_alpha + 0.07)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                f_new = (1.0 - mid) * f + mid * target
                f_new = np.maximum(f_new, 0.0)
                fs = f_new.sum()
                if fs < 1e-15:
                    break
                f_new = f_new / fs
                obj = fast_convolve(f_new).max()
                if obj < best_val:
                    best_val = obj
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            
            obj_val = fast_convolve(f).max()
            if obj_val < best_obj - 1e-13:
                best_obj = obj_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * (0.002 + 0.001 * (no_improve // 10)))
            no_improve = 0
    
    return best_f, best_obj


def local_search(f0, time_limit, rng, scale=0.003):
    """Random perturbation local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        noise = rng.standard_normal(n) * scale
        f_pert = project_simplex(best_f + noise)
        obj = fast_convolve(f_pert).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_pert.copy()
    
    return best_f, best_obj
