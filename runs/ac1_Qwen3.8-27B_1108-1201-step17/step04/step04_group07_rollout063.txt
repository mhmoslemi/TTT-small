DFT_{N}(a*a)_k = A_k * A_k = |A_k|^2 (for the zero-padded DFT)

So: (a*a)_m = (1/N) * sum_k |A_k|^2 * exp(2πikm/N)

We want max_m (a*a)_m to be small. The average of (a*a)_m over m is (1/N) * sum_m (a*a)_m = (1/N) * (sum a)^2 = 1/N (if sum a = 1).

So we want the convolution to be close to its mean 1/N everywhere (flat). This means we want the "AC component" of the convolution to be small, i.e., |A_k|^2 should be concentrated at k=0.

But |A_0|^2 = (sum a)^2 = 1, and the sum of |A_k|^2 = N * sum(a^2) (by Parseval). So the "energy" in the AC components is N*sum(a^2) - 1.

To minimize the peak of the convolution, we want to minimize the AC energy, which means minimizing sum(a^2). For a non-negative sequence with sum=1, the minimum of sum(a^2) is 1/n (achieved by uniform). So the uniform sequence is optimal in this sense!

But wait, the uniform sequence gives a triangular auto-convolution (not flat), with peak = n/n^2 = 1/n. The objective is 2n * 1/n = 2.

The issue is that minimizing the AC energy (sum(a^2)) doesn't directly minimize the PEAK of the convolution. The peak can be large even if the total AC energy is small, if the AC energy is concentrated at a few frequencies.

So the spectral approach should be: minimize the peak of IFFT(|A_k|^2), which is equivalent to minimizing the maximum of the inverse DFT of |A_k|^2.

This is the "pilot sequence design" problem in communications. One approach is to iteratively:
1. Start with some a
2. Compute A = FFT(a, 2n-1)
3. Modify |A|^2 to reduce the peak of IFFT(|A|^2)
4. Update a = real(IFFT(A)) (keeping the phase of A)
5. Project to non-negative (clip to 0)
6. Repeat

This is similar to the "successive over-relaxation" method used in radar sequence design.

Actually, a simpler approach: 
1. Start with random a >= 0, sum = 1
2. Compute conv = a * a
3. Find the peak position k*
4. The contribution to the peak from a_i is a_i * a_{k*-i}
5. Reduce a_i proportionally to its contribution, and redistribute the mass

This is essentially subgradient descent with a specific step size.

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: Exact 1D coordinate descent via bisection on the max-convolution value,
combined with spectral initialization, LP refinement, and adaptive subgradient polish.
"""
import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_and_conv(a, n):
    """Compute 2n*max(conv) and the convolution array."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def coord_update(a, conv, i, n, n_bisect=35):
    """Find optimal a[i] given all other coordinates fixed (minimax 1D subproblem)."""
    ai = a[i]
    if ai < 1e-16:
        return 0.0

    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    num_aff = k_end - k_start + 1

    # partners = a[0:num_aff] since k-i ranges from 0 to num_aff-1
    partners_sub = a[:num_aff]
    conv_range = conv[k_start:k_end + 1]

    # Determine which k equals 2i
    two_i = 2 * i
    is_2i = np.zeros(num_aff, dtype=bool)
    if k_start <= two_i <= k_end:
        is_2i[two_i - k_start] = True

    # Base convolution (with a[i] set to 0)
    base = conv_range - 2.0 * ai * partners_sub
    if is_2i.any():
        base[is_2i] = conv_range[is_2i] - ai * ai

    # Unaffected max
    U = 0.0
    if k_start > 0:
        U = max(U, conv[:k_start].max())
    if k_end < 2 * n - 2:
        U = max(U, conv[k_end + 1:].max())

    cur_max = conv.max()
    if U >= cur_max - 1e-15:
        return ai

    # Bisection setup
    t_lo = max(U, base.max())
    t_hi = cur_max
    if t_lo >= t_hi - 1e-15:
        return ai

    non2i_mask = ~is_2i
    pos_mask = non2i_mask & (partners_sub > 1e-16)
    has_non2i = pos_mask.any()
    has_2i = is_2i.any()

    base_2i = 0.0
    if has_2i:
        base_2i = base[is_2i][0]

    # Bisection
    for _ in range(n_bisect):
        t_mid = (t_lo + t_hi) * 0.5
        feasible = True
        x_max = 1e18

        if has_non2i:
            ub = (t_mid - base[pos_mask]) / (2.0 * partners_sub[pos_mask])
            if ub.min() < 0:
                feasible = False
            else:
                x_max = ub.min()

        if has_2i and feasible:
            if t_mid < base_2i:
                feasible = False
            else:
                x_2i = np.sqrt(t_mid - base_2i)
                if x_2i < x_max:
                    x_max = x_2i

        if feasible:
            t_hi = t_mid
        else:
            t_lo = t_mid

    # Compute optimal x at t_hi
    x_opt = 1e18
    if has_non2i:
        ub = (t_hi - base[pos_mask]) / (2.0 * partners_sub[pos_mask])
        x_opt = min(x_opt, ub.min())
    if has_2i:
        if t_hi > base_2i:
            x_opt = min(x_opt, np.sqrt(t_hi - base_2i))

    x_opt = max(0.0, x_opt)
    # Limit step size for stability
    x_opt = min(x_opt, ai * 3.0 + 1e-6)
    return x_opt


def coord_descent_optimize(a0, n, time_limit, verbose=False):
    """Run coordinate descent with restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        # One full sweep
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            x_opt = coord_update(a, conv, i, n)
            if x_opt != a[i]:
                a[i] = x_opt

        # Recompute convolution
        conv = fast_convolve(a)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            conv = fast_convolve(a)
        else:
            a = a / s
            conv = fast_convolve(a)

        cur_max = conv.max()
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}, max_conv={best_max:.10f}, "
                  f"obj={2*n*best_max:.8f}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= max_no_improve:
            rng = np.random.default_rng(sweep * 17 + 5)
            scale = 0.003 * (1.0 + 0.2 * min(no_improve, 20))
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            max_no_improve = min(15, max_no_improve + 1)

    return best_a, 2.0 * n * best_max


def build_conv_matrix(a):
    """Build matrix C such that C @ g = full linear convolution of a and g."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    tightness = 0.9995
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        A_ub = build_conv_matrix(a)
        M_lp = M * tightness

        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)
        bounds = [(0.0, None)] * n

        try:
            result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        if result.success:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
                d = g - a
                norm_d = np.linalg.norm(d)

                if norm_d > 1e-12 and g_sum > 1.0 + 1e-8:
                    # Line search
                    best_t = 0.0
                    best_obj_try = best_obj
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = np.maximum(a + t * d, 0.0)
                        st = a_try.sum()
                        if st < 1e-15:
                            continue
                        a_try = a_try / st
                        obj_try = 2.0 * n * fast_convolve(a_try).max()
                        if obj_try < best_obj_try - 1e-14:
                            best_obj_try = obj_try
                            best_t = t

                    if best_obj_try < best_obj - 1e-13:
                        a_new = np.maximum(a + best_t * d, 0.0)
                        s_new = a_new.sum()
                        if s_new > 1e-15:
                            a = a_new / s_new
                        conv = fast_convolve(a)
                        M = conv.max()
                        best_obj = 2.0 * n * M
                        best_a = a.copy()
                        no_improve = 0
                        tightness = max(0.998, tightness - 0.0003)
                    else:
                        no_improve += 1
                        tightness = min(1.001, tightness + 0.002)
                else:
                    no_improve += 1
                    tightness = min(1.001, tightness + 0.002)
            else:
                no_improve += 1
                tightness = min(1.001, tightness + 0.002)
        else:
            no_improve += 1
            tightness = min(1.001, tightness + 0.005)

        iteration += 1

        if no_improve >= 30:
            rng = np.random.default_rng(iteration * 7 + 11)
            scale = 0.003 * (1.0 + 0.1 * min(perturb_count, 10))
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            tightness = 1.0

    return best_a, best_obj


def sg_optimize(a0, n, time_limit, lr_init=0.004, momentum=0.94, top_k=80, temp=0.0002):
    """Subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient from top-k peaks
        top_k_eff = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k_eff)[-top_k_eff:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx_arr = np.arange(j_start, j_end + 1)
            sg[idx_arr] += w[idx] * 2.0 * a[k - idx_arr]

        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        step += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 30 == 0:
            lr *= 0.997
            temp *= 1.005

        if no_improve >= 300:
            rng = np.random.default_rng(step * 3 + 7)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            conv = fast_convolve(a)
            no_improve = 0

    return best_a, best_obj


def spectral_init(n, rng, variant=0):
    """Generate spectral/structured initialization."""
    x = np.arange(n, dtype=np.float64)

    if variant == 0:
        # Cosine raised to power
        a = (1.0 + np.cos(2.0 * np.pi * (x + 0.5) / n)) ** 0.5
    elif variant == 1:
        # Two-lobe
        a = 0.6 + 0.4 * np.cos(4.0 * np.pi * x / n)
    elif variant == 2:
        # Gaussian center
        a = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.18 * n)) ** 2)
    elif variant == 3:
        # Random phases in frequency domain
        fft_size = 1 << (2 * n - 1).bit_length()
        phases = rng.standard_normal(fft_size // 2 + 1)
        A = np.zeros(fft_size // 2 + 1, dtype=complex)
        mags = np.ones(len(A))
        A[:] = mags * np.exp(1j * phases)
        # Enforce conjugate symmetry
        full_A = np.zeros(fft_size, dtype=complex)
        full_A[:fft_size // 2 + 1] = A
        for k in range(1, fft_size // 2):
            full_A[fft_size - k] = np.conj(A[k])
        a = np.fft.irfft(full_A)[:n]
        a = np.abs(a)
    elif variant == 4:
        # Triangular with noise
        tri = np.minimum(x, n - 1 - x) + 1.0
        a = tri + rng.standard_normal(n) * 0.1
        a = np.abs(a)
    elif variant == 5:
        # Sinusoidal modulation
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * (x + 0.5) / n + rng.uniform(0, 2*np.pi))
    else:
        # Uniform with random perturbation
        a = np.ones(n) + rng.standard_normal(n) * 0.3
        a = np.abs(a)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq
