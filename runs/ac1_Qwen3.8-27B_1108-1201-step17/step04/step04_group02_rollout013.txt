import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_from_conv(conv, n, s):
    """Compute objective from precomputed convolution and sum."""
    return 2.0 * n * conv.max() / (s * s)


def smooth_obj_and_grad(a, tau):
    """Compute smoothed objective and gradient.
    Smooths max(conv) with log-sum-exp.
    Returns (objective, gradient) of 2*n*logsumexp(conv/tau)*tau / (sum a)^2
    """

    s = a.sum()
    conv = fast_convolve(a)
    
    # Soft-max approximation of max
    m = conv.max()
    exp_conv = np.exp((conv - m) / tau)
    sm = m + tau * np.log(exp_conv.sum())  # log-sum-exp * tau = soft max
    p = exp_conv / exp_conv.sum()  # softmax weights
    
    # Gradient of soft-max w.r.t. a
    # d(softmax)/da[j] = sum_k p[k] * d(conv[k])/da[j]
    # d(conv[k])/da[j] = 2*a[k-j] if 0<=k-j<n and k!=2j, or 2*a[j] if k=2j, else 0
    grad_sm = np.zeros(n)
    for j in range(n):

        ks = np.arange(k_min, k_max + 1)
        offsets = ks - j  # these are 0, 1, ..., n-1

        # But if k = 2j (i.e., offset = j), it's still 2*a[j] (same formula)
        # Actually: for offset o = k-j, if o < n: derivative = 2*a[o] (since a[k-j]=a[o])
        # Wait, I need to double check. conv[k] = sum_{j'} a[j']*a[k-j']
        # d(conv[k])/da[j]: j' can be j (giving a[k-j]) or j' can be k-j (giving a[j])
        # So d(conv[k])/da[j] = a[k-j] + a[j] if k!=2j, or 2*a[j] if k=2j
        # In both cases, if 0<=k-j<n: d = a[k-j] + (a[j] if k-j!=j else 0) + ... 
        # Hmm let me just do it directly:
        # For a specific j, the terms in conv[k] that involve a[j] are:
        #   - j'=j: a[j]*a[k-j], derivative = a[k-j]
        #   - j'=k-j: a[k-j]*a[j], derivative = a[j] (only if k-j != j and 0<=k-j<n)
        # So total: if 0<=k-j<n:
        #   if k-j == j: derivative = a[j] (from j'=j term, a[k-j]=a[j], and j'=k-j is same as j'=j)
        #     Wait no. If k-j=j, then j'=j gives a[j]*a[j], d/da[j] = 2*a[j].
        #     And j'=k-j=j is the same term, so we don't add again.
        #   if k-j != j: derivative = a[k-j] + a[j]
        # else: derivative = 0
        
        # Let me just compute it numerically for correctness:
        dk_da = np.zeros(n)  # d(conv[k])/da[j] for all k
        # k-j must be in [0, n-1]
        o = k_max - j  # max offset
        # offsets are 0..n-1 (capped at k_max-j)
        for o in range(min(n, k_max - k_min + 1)):
            k = k_min + o
            if k == 2 * j:
                dk_da[k] = 2.0 * a[j]
            else:
                dk_da[k] = a[o] + a[j]  # a[k-j] = a[o], plus a[j] from the other term
        # Wait, this is O(n) per j, making the full gradient O(n^2). That's fine for n=384.
        grad_sm[j] = np.dot(p, dk_da)
    
    # Full objective: F = 2*n*sm / s^2
    obj = 2.0 * n * sm / (s * s)
    
    # Gradient: dF/da[j] = 2*n * (grad_sm[j]/s^2 - 2*sm*a[j]/s^3)
    grad = 2.0 * n * (grad_sm / (s * s) - 2.0 * sm * a / (s * s * s))
    
    return obj, grad


def smooth_obj_and_grad_fast(a, tau):

def coordinate_step(a, j):
    """Perform one coordinate descent step for coordinate j.
    Returns the new value of a[j] that minimizes the objective.
    """
    n = len(a)
    s = a.sum()
    aj = a[j]
    
    if s < 1e-15:
        return aj
    
    conv = fast_convolve(a)
    
    # Precompute which conv values are affected
    # conv[k] changes if 0 <= k-j < n (i.e., k in [j, j+n-1]) and k < 2n-1
    k_min = j
    k_max = min(2 * n - 2, j + n - 1)
    
    # Store original conv values for affected k
    affected = np.arange(k_min, k_max + 1)
    orig_vals = conv[affected].copy()
    
    # For each affected k, compute the coefficient of (x - aj) in the linear part,
    # and whether it's quadratic (k == 2j)
    # new_conv[k] = orig_vals[i] + 2*a[k-j]*(x-aj) if k!=2j, or x^2 - aj^2 if k==2j
    linear_coeffs = np.zeros(len(affected))
    is_quad = np.zeros(len(affected), dtype=bool)
    for idx, k in enumerate(affected):
        o = k - j  # offset, 0 <= o < n
        if k == 2 * j:
            is_quad[idx] = True
        else:
            linear_coeffs[idx] = 2.0 * a[o]
    
    # Unaffected conv values
    unaffected_max = 0.0
    if k_min > 0 or k_max < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_min:k_max + 1] = False
        if mask.any():
            unaffected_max = conv[mask].max()
    
    # Now search for optimal x in [0, x_max]
    # The objective is: 2*n*max(new_conv) / (s - aj + x)^2
    
    def eval_x(x):
        new_affected = orig_vals + linear_coeffs * (x - aj)
        if np.any(is_quad):
            new_affected[is_quad] = orig_vals[is_quad] + (x * x - aj * aj)
        new_M = max(new_affected.max(), unaffected_max)
        new_s = s - aj + x
        if new_s < 1e-10:
            return np.inf
        return 2.0 * n * new_M / (new_s * new_s)
    
    # Evaluate at several points to bracket the minimum
    # Try x = 0, x = aj, and some scaled versions
    candidates = [0.0, aj]
    # Also try points where the affected max might cross unaffected_max
    if unaffected_max > 0:
        for idx in range(len(affected)):
            if linear_coeffs[idx] > 1e-15:
                # orig_vals[idx] + linear_coeffs[idx]*(x-aj) = unaffected_max
                x_cross = aj + (unaffected_max - orig_vals[idx]) / linear_coeffs[idx]
                if x_cross > 0:
                    candidates.append(x_cross)
                x_cross2 = aj + (unaffected_max - orig_vals[idx]) / linear_coeffs[idx]
                # Also where the quadratic crosses
            if is_quad[idx]:
                # x^2 - aj^2 + orig_vals[idx] = unaffected_max
                x2 = np.sqrt(max(0, unaffected_max - orig_vals[idx] + aj * aj))
                candidates.append(x2)
                candidates.append(0.0)
    
    # Add some heuristic points
    candidates.append(aj * 0.5)
    candidates.append(aj * 1.5)
    candidates.append(aj * 0.1)
    candidates.append(aj * 3.0)
    
    # Evaluate all candidates
    best_x = aj
    best_val = eval_x(aj)
    for x in candidates:
        if x < 0:
            continue
        v = eval_x(x)
        if v < best_val:
            best_val = v
            best_x = x
    
    # Refine with golden section search around best_x
    # Determine search range
    lo = max(0.0, best_x * 0.5 if best_x > 0 else 0.0)
    hi = best_x * 1.5 if best_x > 0 else max(aj * 2, 0.01)
    
    # Make sure lo < hi
    if hi <= lo:
        hi = lo + max(aj, 0.01)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(40):
        x1 = hi - gr * (hi - lo)
        x2 = lo + gr * (hi - lo)
        v1 = eval_x(x1)
        v2 = eval_x(x2)
        if v1 < v2:
            hi = x2
        else:
            lo = x1
        if hi - lo < 1e-14:
            break
    
    x_opt = (lo + hi) / 2.0
    if eval_x(x_opt) < best_val:
        best_x = x_opt
    
    return best_x


def coordinate_descent(a, time_limit, max_sweeps=100, verbose=False):
    """Full coordinate descent optimization."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s  # Normalize
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = obj_from_conv(fast_convolve(a), n, 1.0)
    
    conv = fast_convolve(a)
    s = 1.0
    current_obj = 2.0 * n * conv.max() / s
    
    rng = np.random.default_rng(12345)
    no_improve_sweeps = 0
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        obj_before = current_obj
        
        # Random order for this sweep
        order = rng.permutation(n)
        total_change = 0.0
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            x_new = coordinate_step(a, j)
            if x_new != a[j]:
                total_change += abs(x_new - a[j])
                a[j] = x_new
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        
        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: obj={current_obj:.8f}, best={best_obj:.8f}, change={total_change:.6f}")
        
        if no_improve_sweeps > 20:
            # Perturbation
            perturb_size = 0.01 / np.sqrt(sweep)
            a = np.maximum(best_a + rng.standard_normal(n) * perturb_size, 0.0)
            a = a / a.sum()
            no_improve_sweeps = 0
            conv = fast_convolve(a)
            current_obj = 2.0 * n * conv.max()
    
    return best_a, best_obj


def smooth_optimize(a0, tau_init=0.05, tau_final=0.005, time_limit=60, verbose=False):
    """Optimize using smoothed objective with L-BFGS."""
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = np.inf
    
    tau = tau_init
    n_tau_steps = 5
    
    for tau_step in range(n_tau_steps):
        tau = tau_init * (tau_final / tau_init) ** (tau_step / max(1, n_tau_steps - 1))
        step_time = (time_limit / n_tau_steps) - (time.monotonic() - t0)
        if step_time < 2:
            break
        
        cur_tau = tau
        
        def func_grad(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10, np.zeros_like(x)
            x = x / sx
            obj, grad = smooth_obj_and_grad_fast(x, cur_tau)
            return obj, grad
        
        res = minimize(func_grad, a, method='L-BFGS-B', 
                      bounds=[(0, None)] * n,
                      options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8,
                              'maxfun': 10000})
        
        a = np.maximum(res.x, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    # Final evaluation with exact objective
    conv = fast_convolve(best_a)
    best_obj = 2.0 * len(best_a) * conv.max()
    
    return best_a, best_obj


def solve_sparse_lp(f, M, top_k=20):
    """Solve LP with only top-k binding constraints for speed."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Get top-k lag indices
    if top_k >= len(conv):
        peak_indices = np.arange(len(conv))
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    
    c = -np.ones(n)
    A_ub_rows = []
    b_ub_vals = []
    
    for k in peak_indices:
        row = np.zeros(n)
        for j in range(n):
            o = k - j
            if 0 <= o < n:
                row[j] = f[o]
        A_ub_rows.append(row)
        b_ub_vals.append(M)
    
    if not A_ub_rows:
        return None
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    
    return None


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    # Linear ramps
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())
    
    # Cosine
    c = np.cos(np.pi * (x / max(1, n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())
    
    # Exponential
    e1 = np.exp(-2.0 * x / n)
    inits.append(e1 / e1.sum())
    e2 = np.exp(-2.0 * x[::-1] / n)
    inits.append(e2 / e2.sum())
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - n / 2) / (n / 6)) ** 2)
    inits.append(g / g.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

    best_seq = None

    def try_seq(a, label=""):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed L-BFGS (coarse phase with log-sum-exp softening of max)
2. Exact coordinate descent with incremental convolution updates (refinement)
3. Sparse LP-based direction finding (escapes)
4. Multi-start with diverse initializations and adaptive time allocation

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def smooth_obj_and_grad(a, tau):
    """Smoothed objective and gradient using log-sum-exp softening of max(conv)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    
    conv = fast_convolve(a)
    
    # Log-sum-exp softening of max
    m = conv.max()
    shifted = (conv - m) / tau
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    sm = m + tau * (max_shifted + np.log(exp_shifted.sum()))
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient of soft-max w.r.t. a[j]:
    # d(sm)/da[j] = 2 * sum_{o=0}^{n-1} p[j+o] * a[o]  (where j+o < 2n-1)
    # This is a cross-correlation computed via FFT
    a_flip = a[::-1]
    fft_len = 1 << ((2 * n - 1) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_len)
    Af = np.fft.rfft(a_flip, fft_len)
    xcorr_full = np.fft.irfft(P * Af, fft_len)
    # cross[j] = xcorr_full[j + n - 1] for j = 0, ..., n-1
    grad_sm = 2.0 * xcorr_full[n - 1:2 * n - 1]
    
    # Objective: F = 2*n*sm / s^2
    obj = 2.0 * n * sm / (s * s)
    
    # Gradient: dF/da[j] = 2*n * (grad_sm[j]/s^2 - 2*sm*a[j]/s^3)
    grad = 2.0 * n * (grad_sm / (s * s) - 2.0 * sm * a / (s * s * s))
    
    return obj, grad


def smooth_optimize(a0, tau_init=0.05, tau_final=0.003, time_limit=60, n_tau_stages=5):

    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()

        return a, np.inf
    a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    for stage in range(n_tau_stages):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        frac = stage / max(1, n_tau_stages - 1)
        tau = tau_init * (tau_final / tau_init) ** frac
        
        cur_tau = tau
        
        def func(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10
            xn = x / sx
            obj, _ = smooth_obj_and_grad(xn, cur_tau)
            return obj
        
        def func_grad(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10, np.zeros_like(x)
            xn = x / sx
            obj, grad = smooth_obj_and_grad(xn, cur_tau)
            return obj, grad
        
        step_time = (time_limit / n_tau_stages) - (time.monotonic() - t0)
        if step_time < 3:

def find_optimal_coord(a, j, conv, s):
    """Find optimal value for coordinate j given current state.
    Uses the exact (unsmoothed) objective."""
    n = len(a)
    aj = a[j]
    
    if s < 1e-15:
        return aj
    
    # Determine affected lags: k in [j, j+n-1] intersect [0, 2n-2]
    k_min = j
    k_max = min(2 * n - 2, j + n - 1)
    
    # Store affected conv values
    n_affected = k_max - k_min + 1
    aff_vals = conv[k_min:k_max + 1].copy()
    
    # Determine linearity: for each affected k, delta = 2*a[k-j]*(x-aj) if k!=2j, else x^2-aj^2
    linear_c = np.zeros(n_affected)
    is_quad = np.zeros(n_affected, dtype=bool)
    for idx in range(n_affected):
        k = k_min + idx
        if k == 2 * j:
            is_quad[idx] = True
        else:
            o = k - j
            linear_c[idx] = 2.0 * a[o]
    
    # Unaffected max
    if k_min > 0 or k_max < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_min:k_max + 1] = False
        if mask.any():
            unaff_max = conv[mask].max()
        else:
            unaff_max = 0.0
    else:
        unaff_max = 0.0
    
    s0 = s - aj  # new_s = s0 + x
    
    def eval_obj(x):
        new_aff = aff_vals + linear_c * (x - aj)
        if np.any(is_quad):
            new_aff[is_quad] = aff_vals[is_quad] + (x * x - aj * aj)
        new_M = max(new_aff.max(), unaff_max)
        new_s = s0 + x
        if new_s < 1e-10:
            return np.inf
        return 2.0 * n * new_M / (new_s * new_s)
    
    # Evaluate at candidate points
    candidates = [0.0, aj]
    if aj > 0:
        candidates.extend([aj * 0.1, aj * 0.3, aj * 0.5, aj * 0.7,

    best_x = aj
    best_v = eval_obj(aj)
    
    for x in candidates:
        if x < 0 or x > 10.0:
            continue
        v = eval_obj(x)
        if v < best_v:
            best_v = v
            best_x = x
    
    # Golden section refinement
    # Bracket: expand around best_x
    span = max(best_x, aj, 1e-4) * 0.5
    lo = max(0.0, best_x - span)
    hi = best_x + span
    if hi <= lo:
        hi = lo + max(aj, 1e-4)
    
    # Ensure the minimum is bracketed
    v_lo = eval_obj(lo)
    v_hi = eval_obj(hi)
    if v_lo < best_v:
        best_v = v_lo
        best_x = lo
    if v_hi < best_v:
        best_v = v_hi
        best_x = hi
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(50):
        if hi - lo < 1e-15:
            break
        x1 = hi - gr * (hi - lo)
        x2 = lo + gr * (hi - lo)
        v1 = eval_obj(x1)
        v2 = eval_obj(x2)
        if v1 < v2:
            hi = x2
        else:
            lo = x1
    
    x_opt = (lo + hi) / 2.0
    v_opt = eval_obj(x_opt)
    if v_opt < best_v:
        best_x = x_opt
    
    return best_x


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with incremental convolution updates."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    conv = fast_convolve(a)
    s = 1.0
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break
        
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            x_new = find_optimal_coord(a, j, conv, s)
            if abs(x_new - a[j]) > 1e-15:
                aj_old = a[j]
                a[j] = x_new
                # Incrementally update conv
                k_min = j
                k_max = min(2 * n - 2, j + n - 1)
                for k in range(k_min, k_max + 1):
                    o = k - j
                    if k == 2 * j:
                        conv[k] += x_new * x_new - aj_old * aj_old
                    else:
                        conv[k] += 2.0 * a[o] * (x_new - aj_old)
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        scale = 1.0 / s
        a = a * scale
        conv = conv * (scale * scale)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: obj={current_obj:.8f}, best={best_obj:.8f}")
        
        if no_improve > 10:
            # Perturbation escape
            pert = 0.02 / np.sqrt(no_improve / 10 + 1)
            a = np.maximum(best_a + rng.standard_normal(n) * pert, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                conv = fast_convolve(a)
                s = 1.0
            else:
                a = a / s
                conv = fast_convolve(a)
                s = 1.0
            current_obj = 2.0 * n * conv.max()
            no_improve = 0
    
    return best_a, best_obj


def solve_sparse_lp(f, M, top_k=25, time_limit=15):
    """LP: max sum(g) s.t. (f*g)[k] <= M for top-k peaks, g >= 0."""
    n = len(f)
    conv = fast_convolve(f)
    
    if top_k >= len(conv):
        peak_indices = np.arange(len(conv))
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    
    A_ub_rows = []
    b_ub_vals = []
    
    for k in peak_indices:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            row[js:je + 1] = f[k - js:k - je + 1][::-1]
        A_ub_rows.append(row)
        b_ub_vals.append(M)
    
    if not A_ub_rows:
        return None
    
    c = -np.ones(n)
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    
    return None
