import numpy as np

def solve_convex_optimization(n, solver=None):
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    
    constraints = [cp.sum(a) == 1]
    
    for k in range(n):
        # Build Q_k matrix (sparse)
        # Q_k[i,j] = 1 if |i-j| = k
        if k == 0:
            mask = np.eye(n, dtype=bool)
        else:
            mask = np.zeros((n, n), dtype=bool)
            idx = np.arange(n - k)
            mask[idx, idx + k] = True
            mask[idx + k, idx] = True
        
        Q_k = cp.psd_wrap(cp.sparse.csr_matrix(mask))
        constraints.append(cp.quad_form(a, Q_k) <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    if solver:
        prob.solve(solver=solver)
    else:
        prob.solve()
    
    return a.value, t.value

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Structured initialization (flat DFT magnitude approach)
2. Log-sum-exp smoothed gradient descent with adaptive beta and BB step size
3. Frank-Wolfe with LP oracle for global improvement
4. Multi-start with diverse initializations and lengths
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a.
    Returns (smoothed_obj, gradient) where gradient = 2 * h,
    h[j] = sum_i a[i] * w[j+i]
    """
    n = len(a)
    c = fast_convolve(a)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    obj = (bc_max + np.log(exp_bc.sum())) / beta
    
    # h[j] = sum_{i=0}^{n-1} a[i] * w[j+i]
    # = convolve(a[::-1], w)[n-1 : 2n-1]
    a_rev = a[::-1]
    total_len = n + (2*n - 1) - 1  # = 3n-2
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(A * W, fft_size)
    h = conv[n-1:2*n-1]
    
    grad = 2.0 * h
    return obj, grad

def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(f, M):
    n = len(f)
    A = build_lp_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def lse_optimize(f0, time_limit, beta_start=10.0, beta_end=10000.0, 
                 lr_init=None, momentum_init=0.9):
    """LSE smoothed gradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if lr_init is None:
        lr_init = 1.0 / np.sqrt(n)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    prev_grad = None
    prev_f = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        obj, grad = lse_grad(f, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            beta *= 2
            continue
        grad_unit = grad / grad_norm
        
        # BB step size
        if prev_grad is not None:
            s_vec = f - prev_f
            y_vec = grad - prev_grad
            sy = s_vec @ y_vec
            yy = y_vec @ y_vec
            if sy > 1e-12 and yy > 1e-12:
                lr_bb = (s_vec @ s_vec) / sy
                lr_bb = np.clip(lr_bb, lr * 0.3, lr * 3.0)
                lr = 0.5 * lr + 0.5 * lr_bb
        
        v = momentum * v - lr * grad_unit
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.9
            if no_improve > 50:
                f = best_f.copy()
                v = np.zeros(n)
                lr = lr_init / (1 + no_improve * 0.1)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
        
        # Increase beta
        if beta < beta_end:
            beta *= (1.0 + 0.005 * min(1.0, it / 100.0))
        
        prev_grad = grad.copy()
        prev_f = f.copy()
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, max_conv={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")
    
    return best_f, best_obj

def fw_lp_step(f0, time_limit, max_lp_calls=50):
    """FW-LP: Find descent direction using LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit and it < max_lp_calls:
        it += 1
        
        g = solve_lp(f, M * m_exp)
        if g is None or g.sum() < 1e-15:
            m_exp = min(m_exp * 1.15, 8.0)
            continue
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-12:
            m_exp = min(m_exp * 1.1, 10.0)
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Refine with bisection
            lo = max(0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2
                ft = np.maximum((1 - mid) * f
