def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_smoothed_optimize(a0, time_limit, tau_init=0.001, tau_min=1e-6, 
                          lr_init=0.01, lr_decay=0.999, mom_init=0.9, 
                          verbose=False):
    """
    Optimize using log-sum-exp smoothed gradients with adaptive temperature.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    tau = tau_init
    lr = lr_init
    mom = mom_init
    v = np.zeros(n)
    rng = np.random.default_rng(777)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        
        # Compute softmax weights for peaks
        w = lse_gradient(conv, tau)
        
        # Build list of (lag, weight) for significant peaks
        peaks_weights = [(k, w[k]) for k in range(len(conv)) if w[k] > 1e-10]
        
        if not peaks_weights:
            peaks_weights = [(int(np.argmax(conv)), 1.0)]
        
        # Compute gradient
        grad = compute_conv_gradient(a, peaks_weights)
        
        # Gradient step with momentum
        v = mom * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Keep old a but update velocity
            a = a_new  # still move, but track best separately
        
        # Anneal
        lr *= lr_decay
        tau = max(tau_min, tau_init * (0.98 ** it))
        
        # Adaptive momentum
        if no_improve > 100:
            mom = max(0.5, mom * 0.95)
            lr = max(0.001, lr_init * 0.1)
        elif no_improve > 200:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.3
            mom = mom_init
        
        if verbose and it % 500 == 0:
            print(f"    LSE iter {it}: obj={best_obj:.10f}, tau={tau:.8f}, lr={lr:.6f}")
    
    return best_a, best_obj

def spectral_shape_optimize(a0, time_limit, n_freq_targets=32, verbose=False):
    """
    Optimize by targeting a flat power spectrum in the frequency domain.
    A flat spectrum corresponds to a "good" autocorrelation (low sidelobes).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.005
    v = np.zeros(n)
    mom = 0.9
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    # FFT size for the sequence
    fft_size = 1 << (n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Target: flat spectrum (all equal)
        P_target = P.sum() / len(P)
        
        # Gradient of sum((P - P_target)^2) w.r.t. a
        # P[k] = |A[k]|^2 = A[k] * conj(A[k])
        # dP[k]/da = 2 * Re(conj(A[k]) * dA[k]/da)
        # dA[k]/da[j] = exp(-2*pi*i*k*j/fft_size)
        # So d(sum(P^2))/da[j] = sum_k 2*(P[k]-P_target) * 2 * Re(conj(A[k]) * exp(-2*pi*i*k*j/fft_size))
        # = 4 * Re(sum_k (P[k]-P_target) * conj(A[k]) * exp(-2*pi*i*k*j/fft_size))
        # = 4 * Re(IFFT_conj((P-P_target) * conj(A)))... 
        
        # Actually: sum_k (P[k]-P_target) * conj(A[k]) * exp(-2*pi*i*k*j/N)
        # Let Q[k] = (P[k] - P_target) * conj(A[k])
        # Then the sum is the IFFT of Q evaluated at j, with a negative sign in the exponent
        # IFFT(Q)[j] = (1/N) * sum_k Q[k] * exp(2*pi*i*k*j/N)
        # But I need sum_k Q[k] * exp(-2*pi*i*k*j/N) = conj(IFFT(conj(Q))[j]) * N... 
        
        # This is getting complicated. Let me just use the time-domain approach.
        # The spectral shaping gradient in the time domain is:
        # grad_time = IFFT((P - P_target) .* conj(A)) * 4 (up to a constant and scaling)
        
        Q = (P - P_target) * np.conj(A)
        grad_spec = 4.0 * np.fft.irfft(Q, fft_size)[:n]
        
        # Also add the direct peak gradient for the main peak
        conv = fast_convolve(a)
        peak_lag = int(np.argmax(conv))
        
        # Simple gradient for the peak
        grad_peak = np.zeros(n)
        for j in range(n):
            m = peak_lag - j
            if 0 <= m < n:
                grad_peak[j] = 2.0 * a[m]
        
        # Combined gradient
        grad = 0.5 * grad_spec + 0.5 * grad_peak
        
        # Step
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)
        
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr *= 0.9995
        
        if no_improve > 150:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.005
        
        if verbose and it % 200 == 0:
            print(f"    Spectral iter {it}: obj={best_obj:.10f}")
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: at each step, find the coordinate that gives the best
    improvement if adjusted, and adjust it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    rng = np.random.default_rng(99)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    step_sizes = [0.001, 0.005, 0.01, 0.02, 0.05]
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Pick a random coordinate (or a few)
        n_try = min(5, n)
        indices = rng.choice(n, size=n_try, replace=False)
        
        improved = False
        for j in indices:
            for step in step_sizes:
                for direction in [1, -1]:
                    a_trial = a.copy()
                    delta = direction * step
                    a_trial[j] = max(0.0, a[j] + delta)
                    # Renormalize
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    obj = fast_convolve(a_trial).max()
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Random jump
                a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
                no_improve = 0
                if fast_convolve(a).max() < best_obj:
                    best_obj = fast_convolve(a).max()
                    best_a = a.copy()
    
    return best_a, best_obj

def multi_resolution_optimize(a0, time_limit, n_coarse=32, verbose=False):
    """
    Optimize at multiple resolutions: downsample, optimize coarse, upscale, refine.
    """
    n = len(a0)
    t0 = time.monotonic()
    remaining = time_limit
    
    # Coarse optimization
    t_coarse = min(remaining * 0.2, 60)
    if t_coarse > 5:
        # Downsample by averaging blocks
        n_blocks = n // n_coarse
        if n_blocks >= 1:
            coarse = np.zeros(n_coarse)
            for i in range(n_coarse):
                start = i * n_blocks
                end = (i + 1) * n_blocks
                coarse[i] = a0[start:end].mean()
            coarse = coarse / coarse.sum()
            
            # Optimize coarse
            coarse_opt, _ = lse_smoothed_optimize(coarse, t_coarse * 0.8, 
                                                   tau_init=0.01, lr_init=0.05)
            
            # Upscale back to fine resolution
            fine = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_coarse), coarse_opt)
            fine = np.maximum(fine, 0)
            fine = fine / fine.sum()
            
            if fast_convolve(fine).max() < fast_convolve(a0).max():
                a0 = fine
    
    # Fine optimization
    remaining = time_limit - (time.monotonic() - t0)
    a_opt, _ = lse_smoothed_optimize(a0, remaining, tau_init=0.001, lr_init=0.01)
    
    return a_opt

def smart_perturbation(a, peaks, n_perturb=10, rng=None):
    """
    Generate smart perturbations that specifically target the peak lags.
    For a peak at lag k, the peak value is sum_i a[i]*a[k-i].
    A perturbation that reduces this: decrease a[i] where a[k-i] is large,
    and increase a[j] where a[k-j] is small.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    results = []
    
    for _ in range(n_perturb):
        a_pert = a.copy()
        # Pick a peak lag
        k = int(peaks[rng.integers(len(peaks))])
        # Find indices where a[i]*a[k-i] contributes most to the peak
        js = range(max(0, k - n + 1), min(n, k + 1))
        contributions = np.array([a[j] * a[k - j] for j in js if 0 <= k - j < n])
        js_valid = np.array([j for j in js if 0 <= k - j < n])
        
        if len(contributions) == 0:
            results.append(a_pert)
            continue
        
        # Decrease high contributors, increase low contributors
        order = np.argsort(contributions)
        n_decrease = max(1, len(order) // 4)
        n_increase = max(1, len(order) // 4)
        
        for idx in order[:n_decrease]:
            a_pert[js_valid[idx]] *= 0.95
        for idx in order[-n_increase:]:
            a_pert[js_valid[idx]] *= 1.05
        
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        results.append(a_pert)
    
    return results

def concominance_based_optimize(a0, time_limit, verbose=False):
    """
    Optimize using a concominance-based approach: iteratively identify
    the most "concentrated" part of the sequence and spread it out,
    while monitoring the effect on the convolution peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    rng = np.random.default_rng(555)
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        peak_lag = int(np.argmax(conv))
        peak_val = conv[peak_lag]
        
        # Find the "alignment vector" at the peak lag
        # The peak is sum of a[j]*a[peak_lag-j]
        align = np.zeros(n)
        for j in range(max(0, peak_lag - n + 1), min(n, peak_lag + 1)):
            if 0 <= peak_lag - j < n:
                align[j] = a[peak_lag - j]
        
        # The "energy" in the alignment is dot(a, align) = peak_val
        # To reduce it, we want to rotate a away from align
        # Project a onto the direction orthogonal to align (in the simplex)
        
        # Simple approach: move mass from high-align positions to low-align positions
        # Sort positions by alignment value
        order = np.argsort(align)
        
        # Transfer a small amount from high-align to low-align positions
        amount = peak_val * 0.01  # 1% of peak value
        n_transfer = max(2, n // 20)
        
        a_trial = a.copy()
        # Take from positions with highest alignment
        for i in range(n_transfer):
            src = order[-1 - i]
            taken = min(a_trial[src], amount / n_transfer)
            a_trial[src] -= taken
            # Give to positions with lowest alignment
            dst = order[i]
            a_trial[dst] += taken
        
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()
        
        obj_trial = fast_convolve(a_trial).max()
        if obj_trial < best_obj - 1e-14:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try the opposite: move from low to high (sometimes helps by changing peak lag)
            a_trial2 = a.copy()
            for i in range(n_transfer):
                src = order[i]
                taken = min(a_trial2[src], amount / n_transfer)
                a_trial2[src] -= taken
                dst = order[-1 - i]
                a_trial2[dst] += taken
            a_trial2 = np.maximum(a_trial2, 0)
            a_trial2 = a_trial2 / a_trial2.sum()
            obj2 = fast_convolve(a_trial2).max()
            if obj2 < best_obj - 1e-14:
                best_obj = obj2
                best_a = a_trial2.copy()
                a = a_trial2.copy()
                no_improve = 0
        
        if no_improve > 100:
            # Random restart
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = 0
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    Concominance iter {it}: obj={best_obj:.10f}")
    
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer combining:
    1. LSE smoothed gradient descent (primary)
    2. Spectral shaping (frequency domain)
    3. Concominance-based peak suppression
    4. Smart perturbations and coordinate descent (polishing)
    5. Multi-resolution refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_a = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
