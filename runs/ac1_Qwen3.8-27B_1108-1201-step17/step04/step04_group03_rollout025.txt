Actually, let me think about this more carefully. The LP oracle is the most powerful tool. Let me focus on making it work better.

The Frank-Wolfe algorithm for this problem:
1. Start with f (normalized, sum=1)
2. Compute M = max(conv(f,f))
3. Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
4. Get g, normalize to sum=1
5. Line search: find best α in [0,1] to minimize max(conv((1-α)f + αg, (1-α)f + αg))
6. Update f

The key question is: how to make this converge faster?

Ideas:
- **Pairwise FW**: Use a "free set" of coordinates that can be updated, and the rest are fixed. This can converge faster than standard FW.
- **Truncated FW**: Only update a subset of coordinates (those with the most improvement potential).
- **Accelerated FW**: Use Nesterov-style acceleration.

Let me try a different angle. What if I use the **conditional gradient** (which is FW) but with a **better line search** that uses quadratic interpolation?

The function φ(α) = max_k ((1-α)f + αg) * ((1-α)f + αg))[k] is the max of quadratic functions of α. Each individual function is quadratic:
h_k(α) = ((1-α)f + αg) * ((1-α)f + αg))[k]
= (1-α)^2 (f*f)[k] + 2α(1-α)(f*g)[k] + α^2 (g*g)[k]

Since we know (f*f)[k] <= M and (f*g)[k] <= M (by the LP constraint), the function h_k(α) is bounded. The max of these quadratics is a convex piecewise-quadratic function of α. We can find its minimum by:
1. Evaluating at α = 0, 0.5, 1
2. Identifying the active constraint at the minimum
3. Solving the single quadratic for the minimum

Actually, the minimum of a convex piecewise-quadratic function can be found by:
- Binary search on the derivative (subgradient)
- Or simply: the minimum of h_k(α) for a specific k is at α_k* = -(f*g)[k] - (f*f)[k] + (g*g)[k] + (f*f)[k]... 

Let me just do a proper 1D optimization. Since φ(α) is convex in α (max of convex functions), I can use ternary search or golden section search.

Even better: since each h_k is quadratic, I can find the overall minimum by:
1. For each k, find α_k* = argmin h_k(α) (clipped to [0,1])
2. The overall minimum is at the smallest α_k* where the "active" k changes

This is O(n^2) per line search, which is fine for n=384.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

        # = (1-a)^2 * ff[k] + 2a(1-a) * fg[k] + a^2 * gg[k]

                v1 = eval_alpha(c1)
            else:

def subgradient_optimize(f0, time_limit, lr_init=0.008, momentum=0.92, top_k_frac=0.08, temp_init=0.0008):
    """Subgradient with momentum, softmax peak weighting, and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    top_k = max(5, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)

        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1

        cur_lr *= 0.99985
        cur_temp *= 1.0008

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            # Structured Fourier perturbation
            scale = 0.002 + 0.004 * min(1.0, no_improve / 800.0)
            x = np.arange(n, dtype=np.float64)
            noise = np.zeros(n)
            for m in range(1, 5):
                amp = scale * rng.uniform(0.5, 1.0)
                phase = rng.uniform(0, 2 * np.pi)
                noise += amp * np.cos(2 * np.pi * m * x / n + phase)
            f = project_simplex(best_f + noise)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.6, 0.001)
            cur_temp = temp_init * 2.0
            no_improve = 0

    return best_f, best_obj

def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet (various concentrations)
    for conc in [0.3, 0.7, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * conc))

    # Edge-weighted
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        ew = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** alpha
        inits.append(ew / ew.sum())

    # Gaussian bump
    for w in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (w * n)) ** 2)
        inits.append(g / g.sum())

    # Cosine modes
    for m in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.6]:
            a = 1.0 + amp * np.cos(2 * np.pi * m * x / n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                inits.append(a / s)

    # Cauchy-like
    for c in [2.0, 5.0, 10.0, 20.0]:
        a = 1.0 / (x + c)
        inits.append(a / a.sum())

    # Two-bump
    for frac in [0.3, 0.4]:
        a = np.exp(-((x - frac * n) / (0.1 * n)) ** 2) + np.exp(-((x - (1-frac) * n) / (0.1 * n)) ** 2)
        inits.append(a / a.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Perturbed parent
            for sc in [0.005, 0.01, 0.02]:
                pert = project_simplex(px / ps + rng.standard_normal(n) * sc)
                inits.append(pert)

    return inits

            try_seq(a_opt, "final_fw")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            for _ in range(3):
                rem = deadline - time.monotonic() - 10
                if rem < 3:
                    break
                x = np.arange(n_b, dtype=np.float64)
                noise = 0.001 * np.cos(2 * np.pi * rng.integers(1, 5) * x / n_b + rng.uniform(0, 2*np.pi))
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = subgradient_optimize(f_pert, time_limit=min(5, rem),
                                                lr_init=0.001, momentum=0.95,
                                                top_k_frac=0.2, temp_init=0.0001)
                try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
