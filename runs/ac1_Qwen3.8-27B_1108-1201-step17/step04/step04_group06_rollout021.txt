"""

import numpy as np
import time
import scipy.optimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_at_lag(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a. O(n)."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    grad[js:je + 1] = 2.0 * a[k - je:k - js + 1][::-1]
    return grad


def score_from_conv(a, c):
    """Compute the objective from sequence and its convolution."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(a) * c.max() / (s * s)


def eg_momentum(a0, time_limit, eta0=1.0, momentum=0.85, top_k=3,
                cycle=True, verbose=False, seed=42):
    """
    Exponentiated gradient with momentum and peak cycling.
    At each iteration, cycles through the top-k peaks and uses a single
    peak's gradient (or a small weighted combination) for the update.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    grad_ema = np.zeros(n)
    eta = eta0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Find top-k peaks
        if top_k < len(c):
            top_idx = np.argpartition(c, -top_k)[-top_k:]

def eg_smoothed(a0, time_limit, eta0=1.0, temp=0.005, temp_decay=0.9997,
                top_k_frac=0.2, verbose=False, seed=42):
    """
    Exponentiated gradient with softmax-weighted subgradient over top peaks.
    Similar to the previous approach but with different defaults.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    eta = eta0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Softmax-weighted subgradient
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
        else:
            indices = np.arange(len(c))

        peek = c[indices]
        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx_i in range(len(indices)):
            if w[idx_i] < 1e-14:
                continue
            k = int(indices[idx_i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx_i] * a[k - je:k - js + 1][::-1]

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        a_new = a * np.exp(-eta * sg / sg_max)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0

                a_new = np.maximum(a_new, 0)
                sn = a_new.sum()
                if sn > 1e-10:
                    a_new = a_new / sn
                else:
                    a_new = best_a.copy()
                c_new = fast_convolve(a_new)

        c = c_new
        temp *= temp_decay

        if verbose and it % 2000 == 0:
            print(f"    EG-smooth iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 100:
        it += 1

        k_star = int(np.argmax(c))
        grad = compute_grad_at_lag(a, k_star)

        # Identify coordinates to decrease (high gradient) and increase (low gradient)
        # Top contributors to the peak
        top_contrib = np.argsort(grad)[::-1][:max(1, n // 8)]
        low_contrib = np.argsort(grad)[:max(1, n // 8)]

        improved = False

        # Try decreasing high contributors and increasing low contributors
        for factor in [0.7, 0.8, 0.9, 0.5]:
            a_try = a.copy()
            # Decrease high contributors
            for j in top_contrib:
                if a_try[j] > 1e-10:
                    a_try[j] *= factor
            # Increase low contributors (redistribute)
            deficit = 1.0 - a_try.sum()
            if deficit > 1e-15:
                for j in low_contrib:
                    a_try[j] += deficit / len(low_contrib)
                a_try = np.maximum(a_try, 0)
                s2 = a_try.sum()
                if s2 > 1e-10:
                    a_try = a_try / s2

            c_try = fast_convolve(a_try)
            obj_try = c_try.max()
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                c = c_try.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            # Try random multiplicative perturbation
            n_pert = max(1, n // 16)
            coords = rng.choice(n, size=n_pert, replace=False)
            for j in coords:
                for factor in [0.8, 1.2, 0.7, 1.4, 0.9, 1.1]:
                    a_try = a.copy()
                    a_try[j] *= factor
                    a_try = np.maximum(a_try, 0)
                    s2 = a_try.sum()
                    if s2 < 1e-10:
                        continue
                    a_try = a_try / s2
                    c_try = fast_convolve(a_try)
                    if c_try.max() < best_obj - 1e-13:
                        best_obj = c_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        c = c_try.copy()
                        improved = True
                        break
                if improved:
                    no_improve = 0
                    break

        if not improved:
            no_improve += 1

    return best_a.copy(), best_obj


def scipy_softmax_optimize(a0, time_limit, verbose=False):
    """
    Optimize using scipy L-BFGS-B in log-space.
    Parameterization: a = softmax(z), optimize over z.
    Objective: 2n * max(conv(a,a)) / (sum(a))^2 = 2n * max(conv(softmax(z), softmax(z)))
    (since sum(softmax(z)) = 1)
    """
    a = np.maximum(a0.astype(np.float64), 1e-300)
    n = len(a)
    a = a / a.sum()

    # Initial z in log-space
    z = np.log(a + 1e-300)

    def objective_and_grad(z):
        # softmax
        z_shifted = z - z.max()
        ez = np.exp(z_shifted)
        a_s = ez / ez.sum()

        c = fast_convolve(a_s)
        c_max = c.max()
        k_star = int(np.argmax(c))

        obj = 2.0 * n * c_max  # since sum(a) = 1

        # Gradient of obj w.r.t. z
        # d(conv(a,a)[k])/da_j = 2*a[k-j] (for valid j)
        # da_j/dz_i = a_j*(delta_ji - a_i)
        # So d(obj)/dz_i = 2n * (d conv[k*]/da) · (da/dz)_i
        # = 2n * sum_j (2*a[k*-j]) * a_j*(delta_ji - a_i)
        # = 2n * 2 * [a[i]*a[k*-i] - a[i]*sum_j a[j]*a[k*-j]]  (with bounds)
        # = 4n * a[i] * [a[k*-i] - conv[k*]]  (with valid index handling)

        # More carefully:
        # dC/dz_i = sum_j (dC/da_j)(da_j/dz_i)
        # dC/da_j = 2*a[k*-j] if 0<=k*-j<n else 0
        # da_j/dz_i = a_j*(delta_{ji} - a_i)
        # dC/dz_i = 2*a[k*-i]*a_i*(1) + sum_{j != i} 2*a[k*-j]*a_j*(-a_i)
        #         = 2*a_i*a[k*-i] - 2*a_i*sum_j a_j*a[k*-j]
        #         = 2*a_i*a[k*-i] - 2*a_i*C (where C = conv[k*])
        # with a[k*-i] = 0 if k*-i out of [0,n-1]

        ki = k_star - np.arange(n)
        valid = (ki >= 0) & (ki < n)
        a_ki = np.zeros(n)
        a_ki[valid] = a_s[ki[valid]]

        grad_z = 2.0 * n * 2.0 * a_s * (a_ki - c_max)

        return obj, grad_z

    result = scipy.optimize.minimize(
        lambda z: objective_and_grad(z)[0],
        z,
        jac=lambda z: objective_and_grad(z)[1],
        method='L-BFGS-B',
        options={'maxiter': 500, 'factr': 1e6}
    )

    if result.success or result.fun < 2.0 * n * fast_convolve(a).max():
        z_opt = result.x
        a_opt = np.exp(z_opt - z_opt.max())
        a_opt = a_opt / a_opt.sum()
        return a_opt
    return a


def generate_two_block_init(n, alpha, rng):
    """
    Generate a two-block initialization: high at both ends, low in middle.
    alpha controls the block width (fraction of n).
    """
    a = np.zeros(n)
    block_size = max(1, int(n * alpha))
    a[:block_size] = 1.0
    a[-block_size:] = 1.0
    # Add some smooth transition
    if block_size > 5:
        fade = np.linspace(1.0, 0.3, min(5, block_size))
        a[block_size - len(fade):block_size] *= fade
        a[:len(fade)][::-1] *= fade
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def generate_geometric_init(n, r, reverse=False):
    """Generate a geometric sequence initialization."""
    x = np.arange(n, dtype=np.float64)
    if reverse:
        x = n - 1 - x
    a = r ** x
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def generate_random_init(n, rng, concentration=0.5):
    """Generate a random Dirichlet initialization."""
    a = rng.dirichlet(np.ones(n) * concentration)
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function combining multiple optimization strategies:
    1. EG with momentum and peak cycling (primary)
    2. EG with smoothed subgradient (secondary)
    3. Targeted local search
    4. SciPy L-BFGS-B in log-space
    5. Multi-start with diverse initializations
    6. Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent found, using uniform n=384")
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()

    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.8f}")
    t_start = time.monotonic()

    # === Phase 1: EG with momentum + cycling on parent (primary) ===
    print(f"\n=== Phase 1: EG Momentum+Cycling (n={n}) ===")
    t1 = min(400, deadline - time.monotonic() - 600)
    if t1 > 30:
        print(f"  Alloc={t1:.0f}s")
        a_opt, obj = eg_momentum(parent, time_limit=t1, eta0=1.5,
                                 momentum=0.85, top_k=3, cycle=True,
                                 verbose=True, seed=1)
        try_seq(a_opt, "eg_mom_p1")

    # === Phase 2: EG with different momentum settings ===
    print(f"\n=== Phase 2: EG Momentum Variants ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        configs = [
            {"eta0": 0.8, "momentum": 0.92, "top_k": 2, "cycle": True},
            {"eta0": 2.0, "momentum": 0.80, "top_k": 5, "cycle": True},
            {"eta0": 1.0, "momentum": 0.95, "top_k": 3, "cycle": False},
        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 400
            if rem < 20:
                break
            t_alloc = min(120, rem / max(1, len(configs) - ci))
            if t_alloc < 15:
                break
            a_opt, obj = eg_momentum(best_seq, time_limit=t_alloc,
                                     seed=100 + ci, **cfg)
            try_seq(a_opt, f"eg_mom_v{ci}")

    # === Phase 3: EG smoothed (from previous approach, as baseline) ===
    print(f"\n=== Phase 3: EG Smoothed Baseline ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(150, rem)
            a_opt, obj = eg_smoothed(best_seq, time_limit=t_alloc,
                                     eta0=1.0, temp=0.003,
                                     temp_decay=0.9996, top_k_frac=0.25,
                                     verbose=False, seed=7)
            try_seq(a_opt, "eg_smooth")

    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse Inits ===")
    print(f"  Best so far: {best_score:.8f}")

    inits = []
    # Two-block inits
    for alpha in [0.1, 0.15, 0.2, 0.3, 0.4]:
        inits.append(generate_two_block_init(n, alpha, rng))
    # Geometric inits
    for r in [0.95, 0.98, 0.995]:
        inits.append(generate_geometric_init(n, r))
        inits.append(generate_geometric_init(n, r, reverse=True))
    # Random
    for conc in [0.1, 0.5, 2.0]:
        inits.append(generate_random_init(n, rng, conc))
    # Parent perturbed
    for scale in [0.005, 0.02, 0.05]:
        p = parent + rng.standard_normal(n) * scale
        p = np.maximum(p, 0)
        if p.sum() > 1e-10:
            inits.append(p / p.sum())

    for i, f_init in enumerate(inits):
        rem = deadline - time.monotonic() - 200
        if rem < 15:
            break
        t_alloc = min(40, rem / max(1, len(inits) - i))
        if t_alloc < 8:
            break
        a_opt, obj = eg_momentum(f_init, time_limit=t_alloc,
                                 eta0=1.0, momentum=0.85, top_k=3,
                                 cycle=True, verbose=False, seed=200 + i)
        try_seq(a_opt, f"init{i}")

    # === Phase 5: Targeted local search on best ===
    print(f"\n=== Phase 5: Targeted Local Search ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(120, rem)
            a_opt, obj = targeted_local_search(best_seq, time_limit=t_alloc,
                                               verbose=True, seed=99)
            try_seq(a_opt, "tls_1")

    # === Phase 6: SciPy L-BFGS-B in log-space ===
    print(f"\n=== Phase 6: SciPy Log-Space ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None and len(best_seq) <= 600:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(120, rem)
            a_opt = scipy_softmax_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "scipy_log")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        alt_lengths = []
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0]:
            n_new = int(n * mult)
            if 16 <= n_new <= 2000:
                alt_lengths.append(n_new)

        for n_new in alt_lengths:
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            # Interpolate parent to new length
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, n_new)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 0)
            if a_interp.sum() > 1e-10:
                a_interp = a_interp / a_interp.sum()
                t_alloc = min(60, rem)
                a_opt, obj = eg_momentum(a_interp, time_limit=t_alloc,
                                         eta0=1.0, momentum=0.85,
                                         top_k=3, cycle=True,
                                         verbose=False, seed=500)
                try_seq(a_opt, f"len_n{n_new}")

    # === Phase 8: Final fine-tuning EG ===
    print(f"\n=== Phase 8: Final Fine EG ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(120, rem)
            a_opt, obj = eg_momentum(best_seq, time_limit=t_alloc,
                                     eta0=0.1, momentum=0.95, top_k=5,
                                     cycle=True, verbose=True, seed=777)
            try_seq(a_opt, "eg_final")

        # One more targeted local search
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_opt, obj = targeted_local_search(best_seq, time_limit=min(rem, 60),
                                               seed=1234)
            try_seq(a_opt, "tls_final")

    # === Phase 9: Last resort - fine EG with very small eta ===
    print(f"\n=== Phase 9: Ultra-Fine ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_opt, obj = eg_smoothed(best_seq, time_limit=min(rem, 60),
                                     eta0=0.05, temp=0.0005,
                                     temp_decay=0.99995, top_k_frac=0.5,
                                     verbose=False, seed=4242)
            try_seq(a_opt, "ultra_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.1f}s)")
    return [float(x) for x in best_seq]
