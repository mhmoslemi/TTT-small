    """Compute J_softmax^T @ grad_a = a * (grad_a - dot(a, grad_a))"""

        a_result = a_result / a_result.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

Key innovation: uses L-BFGS-B (quasi-Newton) on the log-sum-exp smoothed objective

gradually sharpens the approximation from smooth to near-exact.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient 2 * cross_correlation(weights, f) of length n."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):

    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        A_cache = build_lp_matrix(f)

        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]

def softmax_fn(x):
    """Compute softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()


def lse_obj_grad_x(x, beta):
    """
    Compute LSE objective and gradient w.r.t. x (softmax parameterization).
    F(x) = 2*n * LSE_beta(softmax(x))
    where LSE_beta(a) = (1/beta)*log(sum_k exp(beta * conv_k(a)))
    """
    n = len(x)
    a = softmax_fn(x)
    conv = fast_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    # LSE value
    lse_val = bc_max / beta + np.log(exp_bc.sum()) / beta
    obj = 2.0 * n * lse_val

    # Gradient w.r.t. a
    grad_a = compute_lse_grad(w, a)

    # Transform to x-space: grad_x_i = a_i * (grad_a_i - dot(a, grad_a))
    dot_ag = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot_ag)

    return obj, grad_x


def lse_lbfgs_optimize(f0, time_limit, beta_start=10.0, beta_end=100000.0,

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    # Geometric beta schedule
    betas = np.geomspace(beta_start, beta_end, num_betas)

    for beta in betas:
        if time.monotonic() - t0 > time_limit - 10:
            break

        # Initialize x from current f
        f_clipped = np.maximum(f, 1e-30)
        x0 = np.log(f_clipped / f_clipped.sum())

        # Optimize at this beta
        local_best_obj = 2.0 * n * fast_convolve(f).max()
        local_best_x = x0.copy()

        def fun_and_grad(x):
            return lse_obj_grad_x(x, beta)

        time_per_beta = min(60.0, (time_limit - (time.monotonic() - t0)) / max(1, len(betas) - np.where(betas == beta)[0][0]))

        for restart in range(3):
            if time.monotonic() - t0 > time_limit - 10:
                break
            try:
                result = minimize(
                    fun_and_grad, local_best_x, method='L-BFGS-B', jac=True,
                    options={'maxiter': maxiter_per_beta, 'ftol': 1e-18,
                             'gtol': 1e-12, 'maxls': 30,
                             'iprint': -1}
                )
            except Exception:
                break

            if result.fun < local_best_obj * 1.0001:
                a_result = softmax_fn(result.x)
                a_result = a_result / a_result.sum()
                true_obj = 2.0 * n * fast_convolve(a_result).max()
                if true_obj < best_obj - 1e-14:
                    best_obj = true_obj
                    best_f = a_result.copy()
                    f = a_result.copy()
                    local_best_obj = true_obj
                    local_best_x = result.x.copy()
                    print(f"    L-BFGS beta={beta:.1f}: obj={true_obj:.10f}")
                else:
                    # Update local best even if not global
                    local_best_obj = result.fun
                    local_best_x = result.x.copy()
                    f = a_result

        # If no improvement at this beta, increase minimum of f to avoid flat spots
        if time.monotonic() - t0 > time_limit:
            break

    return best_f, best_obj / (2.0 * n)


def projected_lse_polish(f0, time_limit, beta_start=10000, beta_end=500000,
                         lr_init=None):
    """Projected gradient with momentum for final high-beta polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    if lr_init is None:
        lr_init = 1.0 / n

    t0 = time.monotonic()
    no_improve = 0
    beta = beta_start
    lr = lr_init
    momentum = 0.85
    v = np.zeros(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_lse_grad(w, f)

        # Scale gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break
        grad = grad / grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.90
                momentum = max(0.80, momentum - 0.01)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.001 / n)
                beta = min(beta * 1.3, beta_end)
                momentum = 0.85
                no_improve = 0

        if beta < beta_end:
            beta *= 1.01

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polish it={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.8f}, t={elapsed:.0f}s")

    return best_f, best_obj / (2.0 * n)


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def cosine_window(n):
    """Create a squared cosine window of length n."""
    x = np.arange(n)
    w = np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def raised_cosine_window(n):
    """Create a raised cosine (Hann) window."""
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1 + 1e-10)))
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def power_window(n, p):
    """Create a power-law window: w_i = (i+1)^p normalized."""
    x = np.arange(1, n + 1)
    w = x ** p
    return w / w.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization combining:
    - Multi-start with structured and random initializations
    - FW-LP for structural optimization
    - L-BFGS-B on log-sum-exp (softmax parameterization) for fast convergence
    - Projected gradient polishing at very high beta
    - Alternative length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_seq = None
    best_tracker = [float('inf'), None]

    # --- Get initial candidates ---
    candidates = []
    n_default = 448

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is not None:
        candidates.append(('parent', parent))
    else:
        parent = np.ones(n_default) / n_default
        candidates.append(('uniform', parent))

    # Structured candidates
    candidates.append(('cosine', cosine_window(n_default)))
    candidates.append(('hann', raised_cosine_window(n_default)))
    candidates.append(('power0.5', power_window(n_default, 0.5)))
    candidates.append(('power2', power_window(n_default, 2.0)))

    # Random candidates
    for ri in range(3):
        r = rng.dirichlet(np.ones(n_default) * 0.5)
        candidates.append((f'random{ri}', r))

    print(f"Total candidates: {len(candidates)}")

    # --- Phase 1: Quick FW-LP on each candidate ---
    print("\n=== Phase 1: Quick FW-LP on candidates ===")
    t1_end = time.monotonic() + min(300, budget_s * 0.15)
    for name, cand in candidates:
        if time.monotonic() > t1_end:
            break
        t_alloc = min(60, t1_end - time.monotonic())
        if t_alloc < 10:
            break
        a_opt, _ = fw_lp_optimize(cand, time_limit=t_alloc)
        score = try_sequence(a_opt, best_tracker)
        if score < best_tracker[0]:
            candidates = [(name, a_opt)] + [(n2, c2) for n2, c2 in candidates if n2 != name]

    current_best = best_tracker[1] if best_tracker[1] is not None else candidates[0][1]
    print(f"Phase 1 best: {best_tracker[0]:.10f}")

    # --- Phase 2: L-BFGS-B with increasing beta ---
    print(f"\n=== Phase 2: L-BFGS-B (beta 10 -> 100000) ===")
    t2_end = time.monotonic() + min(700, budget_s * 0.35)
    t_alloc = t2_end - time.monotonic()
    if t_alloc > 60:
        a_opt, _ = lse_lbfgs_optimize(
            current_best, time_limit=t_alloc,
            beta_start=10.0, beta_end=100000.0,
            num_betas=30, maxiter_per_beta=200
        )
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    print(f"Phase 2 best: {best_tracker[0]:.10f}")

    # --- Phase 3: High-beta L-BFGS polish ---
    print(f"\n=== Phase 3: High-beta L-BFGS polish ===")
    t3_end = time.monotonic() + min(400, budget_s * 0.22)
    t_alloc = t3_end - time.monotonic()
    if t_alloc > 30:
        a_opt, _ = lse_lbfgs_optimize(
            current_best, time_limit=t_alloc * 0.6,
            beta_start=5000.0, beta_end=500000.0,
            num_betas=15, maxiter_per_beta=300
        )
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best

    # --- Phase 4: Projected gradient final polish ---
    print(f"\n=== Phase 4: Projected gradient polish ===")
    t4_end = time.monotonic() + min(250, budget_s * 0.14)
    t_alloc = t4_end - time.monotonic()
    if t_alloc > 20:
        a_opt, _ = projected_lse_polish(
            current_best, time_limit=t_alloc,
            beta_start=20000, beta_end=1000000,
            lr_init=0.5 / len(current_best)
        )
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    print(f"Phase 4 best: {best_tracker[0]:.10f}")

    # --- Phase 5: Alternative lengths ---
    print(f"\n=== Phase 5: Alternative lengths ===")
    n_best = len(current_best)
    alt_lengths = [384, 400, 420, 480, 512, 576]
    alt_lengths = [n for n in alt_lengths if n != n_best]

    t5_remaining = deadline - time.monotonic()
    n_alts = min(4, len(alt_lengths))
    t_per_alt = max(30, t5_remaining / (n_alts + 1))

    for i, n_alt in enumerate(alt_lengths[:n_alts]):
        t_remaining = deadline - time.monotonic() - 20
        if t_remaining < 40:
            break

        # Interpolate to new length
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()
        else:
            f_interp = np.ones(n_alt) / n_alt

        try_sequence(f_interp, best_tracker)

        t_alloc_fw = min(t_remaining * 0.35, 80)
        t_alloc_lbfgs = min(t_remaining * 0.5, 120)

        a_opt, _ = fw_lp_optimize(f_interp, time_limit=t_alloc_fw)
        try_sequence(a_opt, best_tracker)

        cur = best_tracker[1] if (best_tracker[1] is not None and len(best_tracker[1]) == n_alt) else a_opt
        a_opt, _ = lse_lbfgs_optimize(
            cur, time_limit=t_alloc_lbfgs,
            beta_start=10.0, beta_end=50000.0,
            num_betas=20, maxiter_per_beta=150
        )
        try_sequence(a_opt, best_tracker)

    # --- Phase 6: Final micro-polish on best ---
    if best_tracker[1] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== Phase 6: Final micro-polish ({remaining:.0f}s) ===")
            a_opt, _ = projected_lse_polish(
                best_tracker[1], time_limit=min(remaining - 10, 100),
                beta_start=50000, beta_end=2000000,
                lr_init=0.1 / len(best_tracker[1])
            )
            try_sequence(a_opt, best_tracker)

    # --- Final answer ---
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
    else:
        final_seq = np.ones(n_default) / n_default

    final_score = float(evaluate_sequence(final_seq.tolist()))
    if not np.isfinite(final_score):
        final_seq = np.ones(n_default)
        final_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
