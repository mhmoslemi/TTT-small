def project_simplex(v):

    n = len(v)

    je = min(n - 1, k)

            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def mult_weight_optimize(f0, time_limit, lr_init=0.04, top_k=15, temp_init=0.0004):
    """Multiplicative weight (mirror descent) optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[i] * f[k - idx]
            log_update[idx] -= cur_lr * grad_contrib
        
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9996
        cur_temp *= 1.002
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            f = best_f * np.exp(rng.standard_normal(n) * 0.004)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.008)
            cur_temp = temp_init * 1.5
            no_improve = 0
    
    return best_f, best_obj


def peak_chase_sg(f0, time_limit, lr=0.003, mom=0.93):
    """Subgradient method chasing top peaks with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        top_k = max(5, n // 15)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.3
        tw = weights.sum()
        if tw > 0:
            weights /= tw
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9997
        
        if no_improve > 250:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0005)
            no_improve = 0
    
    return best_f, best_obj


def paired_coordinate_descent(a, time_limit, max_rounds=50):
    """
    Optimize pairs of coordinates (j, m) using bisection on the 1D problem.
    For pair (j,m) with a[j]+a[m]=c, find optimal split.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    rng = np.random.default_rng(123)
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        # Try random pairs
        for _ in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if j == m:
                continue
            
            conv = fast_convolve(a)
            c = a[j] + a[m]
            if c < 1e-15:
                continue
            
            # For split t (a[j]=t, a[m]=c-t):
            # conv_new[k] = conv[k] - a[j]*a[k-j] - a[m]*a[k-m] + t*new_a[k-j] + (c-t)*new_a[k-m]
            # where new_a is a with a[j]=t, a[m]=c-t (before renorm)
            # For first order (ignoring the effect of new_a on the cross terms):
            # conv_new[k] ≈ conv[k] + (t-a[j])*a[k-j] + ((c-t)-a[m])*a[k-m]
            # = conv[k] + (t-a[j])*(a[k-j] - a[k-m])
            
            # Coefficient vector for the 1D problem
            coeff = np.zeros(2 * n - 1)
            # a[k-j] for valid k: k in [j, j+n-1]
            coeff[j:j+n] += a
            # -a[k-m] for valid k: k in [m, m+n-1]
            coeff[m:m+n] -= a
            
            # conv_new[k] = conv[k] + (t - a[j]) * coeff[k]
            # sum_new = 1 (since we're doing a[j]+a[m]=c, keeping sum the same)
            # Actually sum stays the same since a[j]+a[m] is fixed!
            # So objective = 2n * max_k (conv[k] + (t-a[j])*coeff[k])
            
            # Minimize max_k (conv[k] + u*coeff[k]) where u = t - a[j], u in [-a[j], a[m]]
            lo_u = -a[j]
            hi_u = a[m]
            
            if hi_u - lo_u < 1e-15:
                continue
            
            # Bisection on the "slope" of the max
            # The derivative of max_k (conv[k] + u*coeff[k]) w.r.t. u
            # is the max of coeff[k] over active k
            # At optimum, 0 is in the subgradient, so some active k has coeff[k] <= 0
            # and some has coeff[k] >= 0
            
            # Use bisection: find u where the "balance" point is
            for _ in range(50):
                mid = (lo_u + hi_u) / 2.0
                vals = conv + mid * coeff
                active = np.where(vals == vals.max())[0]
                # If all active have coeff > 0, increase u (decrease the peak)
                # If all active have coeff < 0, decrease u
                avg_coeff = coeff[active].mean()
                if avg_coeff > 0:
                    lo_u = mid
                elif avg_coeff < 0:
                    hi_u = mid
                else:
                    break
            
            u_opt = (lo_u + hi_u) / 2.0
            u_opt = np.clip(u_opt, -a[j] + 1e-15, a[m] - 1e-15)
            
            # Apply
            new_a = a.copy()
            new_a[j] = a[j] + u_opt
            new_a[m] = a[m] - u_opt
            new_a = np.maximum(new_a, 0)
            ns = new_a.sum()
            if ns < 1e-15:
                continue
            new_a = new_a / ns
            new_obj = 2.0 * n * fast_convolve(new_a).max()
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_a = new_a.copy()
                a = new_a
                improved = True
        
        if not improved and rnd > 5:
            break
    
    return best_a


def simulated_annealing_structured(a0, time_limit, rng):
    """
    Simulated annealing with structured moves:
    - Move mass between two coordinates
    - Apply a small smooth perturbation
    - Block swap
    Accept with probability exp(-delta/T)
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = 0.001
    T_min = 1e-7
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        T = max(T * cooling, T_min)
        
        move_type = rng.integers(0, 3)
        new_a = a.copy()
        
        if move_type == 0:
            # Mass transfer between two coordinates
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if j == m:
                continue
            scale = T * 50 + 0.001
            delta = rng.normal(0, scale)
            delta = np.clip(delta, -a[j] + 1e-10, a[m] - 1e-10)
            new_a[j] += delta
            new_a[m] -= delta
        elif move_type == 1:
            # Smooth perturbation (window of coordinates)
            center = rng.integers(0, n)
            width = rng.integers(3, 15)
            mask = np.zeros(n)
            for i in range(max(0, center - width), min(n, center + width + 1)):
                mask[i] = 1.0 - abs(i - center) / (width + 1)
            noise = rng.standard_normal(n) * (T * 20 + 0.001) * mask
            new_a = a + noise
        else:
            # Block swap: swap values in two blocks
            bl = rng.integers(2, max(3, n // 20))
            j = rng.integers(0, n - bl)
            m = rng.integers(0, n - bl)
            if abs(j - m) < bl:
                continue
            new_a[j:j+bl], new_a[m:m+bl] = a[m:m+bl].copy(), a[j:j+bl].copy()
        
        new_a = np.maximum(new_a, 0)
        ns = new_a.sum()
        if ns < 1e-15:
            continue
        new_a = new_a / ns
        
        new_conv = fast_convolve(new_a)
        new_obj = 2.0 * n * new_conv.max()
        delta_obj = new_obj - current_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            a = new_a
            current_obj = new_obj
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
    
    return best_a


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian blob
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    
    # Bimodal (two bumps)
    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2) + np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / (n - 1)
    inits.append(tri / tri.sum())
    
    # Sinusoidal modulation
    sin_mod = 1.0 + 0.5 * np.sin(4 * np.pi * x / n)
    inits.append(sin_mod / sin_mod.sum())
    
    # Parent (if available)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Random smooth (interpolated random)
    n_ctrl = 8
    ctrl = rng.dirichlet(np.ones(n_ctrl) * 2)
    smooth = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_ctrl), ctrl)
    smooth = np.maximum(smooth, 1e-10)
    inits.append(smooth / smooth.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Diverse exploration with quick optimization
    Phase 2: Main optimization (alternating CD + FW + MW + SG)
    Phase 3: Paired coordinate descent
    Phase 4: Simulated annealing + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    n_target = 384
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_target = len(parent)
            print(f"Parent loaded: n={n_target}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(n_target) / n_target
    
    # === Phase 1: Quick Exploration (120s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase1_end = time.monotonic() + 120
    
    inits = generate_diverse_inits(n_target, parent, rng)
    
    # Quick optimize top initializations
    # Always start with parent
    t_alloc = min(50, t_phase1_end - time.monotonic() - 60)
    if t_alloc > 10:
        print(f"  FW-LP on parent, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
        try_seq(a_opt, "fw_parent")
    
    # Try a few alternatives quickly
    for i in [1, 4, 5]:  # Dirichlet, bimodal, cosine
        if time.monotonic() > t_phase1_end - 5:
            break
        t_alloc = min(20, t_phase1_end - time.monotonic() - 3)
        if t_alloc < 5:
            break
        print(f"  Quick FW on init {i}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(inits[i], time_limit=t_alloc)
        try_seq(a_opt, f"init_{i}")
    
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: Main Optimization (900s) ===
    print("\n=== Phase 2: Main Optimization ===")
    t_phase2_end = time.monotonic() + 900
    n_best = len(best_seq)
    
    # Hall of fame
    hall_of_fame = [(best_score.copy() if hasattr(best_score, 'copy') else best_score, best_seq.copy())]
    
    stall_count = 0
    iteration = 0
    
    while time.monotonic() < t_phase2_end:
        iteration += 1
        prev_best = best_score
        
        # Sub-phase: Coordinate descent (30s)
        t_cd = min(30, t_phase2_end - time.monotonic() - 100)
        if t_cd > 5:
            a_cd, _ = coordinate_descent_pass(best_seq, time_limit=t_cd)
            sc = try_seq(a_cd, f"cd_iter{iteration}")
            if sc < best_score:
                pass  # already updated
        
        # Sub-phase: FW-LP (30s)
        t_fw = min(30, t_phase2_end - time.monotonic() - 80)
        if t_fw > 5:
            a_fw, _ = fw_lp_optimize(best_seq, time_limit=t_fw)
            try_seq(a_fw, f"fw_iter{iteration}")
        
        # Sub-phase: MW (15s)
        t_mw = min(15, t_phase2_end - time.monotonic() - 60)
        if t_mw > 5:
            a_mw, _ = mult_weight_optimize(best_seq, time_limit=t_mw,
                                           lr_init=0.03 + 0.01 * (iteration % 3),
                                           top_k=10 + 5 * (iteration % 4))
            try_seq(a_mw, f"mw_iter{iteration}")
        
        # Sub-phase: SG (15s)
        t_sg = min(15, t_phase2_end - time.monotonic() - 40)
        if t_sg > 5:
            a_sg, _ = peak_chase_sg(best_seq, time_limit=t_sg,
                                     lr=0.002 + 0.001 * (iteration % 2),
                                     mom=0.92 + 0.01 * (iteration % 3))
            try_seq(a_sg, f"sg_iter{iteration}")
        
        # Check improvement
        if best_score < prev_best - 1e-10:
            stall_count = 0
        else:
            stall_count += 1
        
        # Update hall of fame
        hall_of_fame.append((best_score, best_seq.copy()))
        hall_of_fame.sort(key=lambda x: x[0])
        hall_of_fame = hall_of_fame[:5]
        
        # Random restart if stuck
        if stall_count % 4 == 0 and stall_count > 0:
            if time.monotonic() > t_phase2_end - 30:
                break
            # Pick from hall of fame and perturb
            hof_idx = min(stall_count // 4, len(hall_of_fame) - 1)
            base = hall_of_fame[hof_idx][1].copy()
            noise_scale = 0.002 + 0.003 * min(1.0, stall_count / 20.0)
            f_pert = project_simplex(base + rng.standard_normal(n_best) * noise_scale)
            
            # Quick optimize the perturbation
            t_restart = min(20, t_phase2_end - time.monotonic() - 20)
            if t_restart > 5:
                a_r, _ = fw_lp_optimize(f_pert, time_limit=t_restart)
                try_seq(a_r, f"restart_{iteration}")
                # Also try CD on it
                t_rcd = min(10, t_phase2_end - time.monotonic() - 10)
                if t_rcd > 3:
                    a_rcd, _ = coordinate_descent_pass(a_r, time_limit=t_rcd)
                    try_seq(a_rcd, f"restart_cd_{iteration}")
        
        if iteration % 5 == 0:
            print(f"  Iter {iteration}, best={best_score:.8f}, time_left={t_phase2_end-time.monotonic():.0f}s")
    
    print(f"  Phase 2 best: {best_score:.8f}")
    
    # === Phase 3: Paired Coordinate Descent (350s) ===
    print("\n=== Phase 3: Paired Coordinate Descent ===")
    t_phase3_end = time.monotonic() + 350
    
    # Multiple passes of paired CD
    for pcd_round in range(3):
        if time.monotonic() > t_phase3_end - 30:
            break
        t_pcd = min(100, t_phase3_end - time.monotonic() - 20)
        if t_pcd < 10:
            break
        print(f"  Paired CD round {pcd_round}, t={t_pcd:.0f}s")
        a_pcd = paired_coordinate_descent(best_seq, time_limit=t_pcd)
        try_seq(a_pcd, f"pcd_{pcd_round}")
    
    # Single coordinate CD with fine grid
    if time.monotonic() < t_phase3_end - 20:
        t_scd = min(80, t_phase3_end - time.monotonic() - 10)
        if t_scd > 10:
            print(f"  Fine CD, t={t_scd:.0f}s")
            a_scd, _ = coordinate_descent_pass(best_seq, time_limit=t_scd)
            try_seq(a_scd, "fine_cd")
    
    print(f"  Phase 3 best: {best_score:.8f}")
    
    # === Phase 4: Simulated Annealing + Final Polish (300s) ===
    print("\n=== Phase 4: SA + Final Polish ===")
    t_phase4_end = time.monotonic() + 300
    
    # Simulated annealing
    if time.monotonic() < t_phase4_end - 60:
        t_sa = min(100, t_phase4_end - time.monotonic() - 50)
        if t_sa > 20:
            print(f"  SA, t={t_sa:.0f}s")
            a_sa = simulated_annealing_structured(best_seq, time_limit=t_sa, rng=rng)
            try_seq(a_sa, "sa")
    
    # Final: alternate CD and small perturbations
    while time.monotonic() < t_phase4_end - 30:
        t_rem = t_phase4_end - time.monotonic() - 20
        if t_rem < 5:
            break
        
        # CD pass
        t_cd = min(20, t_rem)
        a_fcd, _ = coordinate_descent_pass(best_seq, time_limit=t_cd)
        try_seq(a_fcd, "final_cd")
        
        # Tiny perturbation + CD
        if time.monotonic() < t_phase4_end - 30:
            noise = rng.standard_normal(n_best) * 0.0005
            f_pert = project_simplex(best_seq + noise)
            t_pt = min(15, t_phase4_end - time.monotonic() - 15)
            if t_pt > 3:
                a_pt, _ = coordinate_descent_pass(f_pert, time_limit=t_pt)
                try_seq(a_pt, "perturb_cd")
        
        # MW fine
        if time.monotonic() < t_phase4_end - 25:
            t_mw2 = min(10, t_phase4_end - time.monotonic() - 10)
            if t_mw2 > 3:
                a_mwf, _ = mult_weight_optimize(best_seq, time_limit=t_mw2,
                                                lr_init=0.005, top_k=8, temp_init=0.0001)
                try_seq(a_mwf, "final_mw")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
