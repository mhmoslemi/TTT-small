    # Try to get dual values for active set tracking
    duals = None
    if active is None:
        try:
            duals = np.asarray(prob.constraints[0].dual.value, dtype=np.float64)
        except Exception:
            duals = None
    return val, duals


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_conv_self(a)
    return 2.0 * n * conv.max() / (s * s)


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()

    # Adaptive temperature: start high, decrease based on progress
    T_curr = 50.0
    T_min = 0.1
    T_max = 100.0
    no_improve_count = 0
    last_best = best_score

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Adaptive temperature: decrease if improving, increase if stuck
        if max_c < last_best - 1e-10:
            T_curr = max(T_min, T_curr * 0.95)
            no_improve_count = 0
            last_best = max_c
        else:
            no_improve_count += 1
            if no_improve_count % 50 == 0:
                T_curr = min(T_max, T_curr * 1.05)

        shifted = (conv - max_c) / T_curr
        shifted = np.maximum(shifted, -80.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        step_base = 1.0 / (T_curr * np.sqrt(n))
        step = step_base
        for ls in range(20):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < max_c or ls == 0:
                a = a_new
                break
            step *= 0.5

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it} T={T_curr:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP Nesterov with dual-guided active set."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    warm_b = None

    # Track active constraints from duals
    active_duals = np.zeros(size)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive active set: use both threshold and dual information
        threshold_high = M * 0.85
        threshold_low = M * 0.50
        active = np.where(conv_y >= threshold_high)[0]
        
        # If not enough active, expand
        if len(active) < 50:
            active = np.where(conv_y >= threshold_low)[0]
        
        # Also include lags with high dual values
        if t > 100:
            dual_active = np.where(active_duals > 0.01 * np.max(active_duals))[0]
            active = np.unique(np.concatenate([active, dual_active]))
            # Cap to avoid too many constraints
            if len(active) > 200:
                # Keep only the most important
                scores = np.zeros(size)
                scores[active] = conv_y[active]
                top_idx = np.argsort(scores[active])[-200:]
                active = active[top_idx]

        b, duals = lp_solve(y, M, solver, active if len(active) < size else None, warm_b)
        
        if duals is not None and len(duals) == size:
            active_duals = 0.9 * active_duals + 0.1 * np.maximum(duals, 0)
        elif duals is not None:
            # Only got duals for active constraints
            new_duals = np.zeros(size)
            new_duals[active] = np.maximum(duals, 0)
            active_duals = 0.9 * active_duals + 0.1 * new_duals

        if b is None:
            b, _ = lp_solve(y, M, solver, None, warm_b)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        warm_b = b

        # Verify dropped constraints
        if len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.005)[0]
            if len(violated) > 5:
                b_full, _ = lp_solve(y, M, solver, None, warm_b)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum
                    warm_b = b

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_targeted_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent targeting the tallest peaks specifically."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_objective(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find top peaks
        n_peaks = min(20, size)
        peak_idx = np.argsort(conv)[-n_peaks:]
        peak_idx = peak_idx[peak_idx > cur_max * 0.85]
        
        # For each peak, compute which coordinates contribute most
        # and try to reduce the peak by transferring mass
        perm = rng.permutation(n)
        pass_improved = False
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Find which peaks this coordinate contributes to
            # Peak at lag k gets contribution a[i] * a[k-i] (if valid)
            # We want to reduce the tallest peak
            # Transfer from i to a j that doesn't contribute to the same peaks
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            # Try several candidate j's
            for _ in range(15):
                j = int(rng.integers(n))
                if j == i:
                    continue
                aj = a[j]

                # Compute the change in convolution
                # conv_new[k] = conv[k] + a[i]*a[j]*(delta terms)
                # Actually: reducing a[i] by d and increasing a[j] by d:
                # delta_conv[k] = -d*(a[k-i] + a[i-k]) + d*(a[k-j] + a[j-k]) 
                #                + d^2 terms (small)
                # For linear approximation:
                # delta_conv[k] = d * (contrib_j[k] - contrib_i[k])
                # where contrib_i[k] = a[k-i] if 0<=k-i<n else (a[i-k] if ... )
                # Actually for full convolution: (a*b)[k] = sum_m a[m]*b[k-m]
                # If we change a[i] -> a[i]-d and a[j] -> a[j]+d:
                # delta(a*a)[k] = -d*a[k-i] - d*a[i-k] + d*a[k-j] + d*a[j-k]
                #              = d*(a[k-j] + a[j-k] - a[k-i] - a[i-k])
                # where a[m] = 0 if m < 0 or m >= n

                # Precompute contribution vectors
                def contrib(idx):
                    """Compute a[k-idx] + a[idx-k] for all k."""
                    c = np.zeros(size)
                    # a[k-idx] is valid when 0 <= k-idx < n, i.e., idx <= k < idx+n
                    lo1 = idx
                    hi1 = idx + n
                    lo1 = max(lo1, 0)
                    hi1 = min(hi1, size)
                    if hi1 > lo1:
                        c[lo1:hi1] += a[:hi1-lo1]
                    # a[idx-k] is valid when 0 <= idx-k < n, i.e., idx-n+1 <= k <= idx
                    lo2 = idx - n + 1
                    hi2 = idx + 1
                    lo2 = max(lo2, 0)
                    hi2 = min(hi2, size)
                    if hi2 > lo2:
                        c[lo2:hi2] += a[idx - np.arange(lo2, hi2)]
                    return c

                contrib_i = contrib(i)
                contrib_j = contrib(j)
                delta_vec = contrib_j - contrib_i

                # Try different deltas
                max_d = ai * 0.9
                n_samples = 15
                for d in np.linspace(0.1 * max_d, max_d, n_samples):
                    new_conv = conv + d * delta_vec
                    # Add second-order terms at diagonals
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross term at i+j
                    if i + j < size:
                        new_conv[i + j] += 2 * d * (aj - ai) + 2 * d * d  # approximate
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    conv = fast_conv_self(a)
                    cur_max = conv.max()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                # Perturb to escape
                n_pert = max(5, n // 50)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and no_improve % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD no_improve={no_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def freq_domain_init(n, rng):
    """Initialize using frequency-domain flatness target."""
    # Start with a random phase spectrum and try to make |A| flat
    # while maintaining a >= 0
    a = np.ones(n) / n
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    
    for it in range(500):
        # Compute current spectrum
        A = fft(a, 2 * n)
        P = np.abs(A) ** 2
        
        # Target: flat power spectrum
        target_P = np.ones_like(P) * P.mean()
        
        # Adjust phase to try to maintain a >= 0
        # Use the current phase but adjust magnitude
        target_A = np.sqrt(target_P) * np.exp(1j * np.angle(A))
        
        # Inverse transform
        a_new = np.real(ifft(target_A, 2 * n))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Blend with previous
        t = 0.3
        a = (1 - t) * a + t * a_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Adaptive-temperature LSE gradient descent with backtracking
3. LP Nesterov with dual-guided active set tracking

_solver = None


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None, warm_b=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if warm_b is not None:
        b_var.value = warm_b
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    if active is None:
        try:
            duals = np.asarray(prob.constraints[0].dual.value, dtype=np.float64)
        except Exception:
            duals = None
    return val, duals


def compute_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_conv_self(a)
    return 2.0 * n * conv.max() / (s * s)


def _contrib_vec(a, idx):
    """Compute the vector v where v[k] = a[k-idx] + a[idx-k] for valid indices."""
    n = len(a)
    size = 2 * n - 1
    c = np.zeros(size)
    lo1 = max(idx, 0)
    hi1 = min(idx + n, size)
    if hi1 > lo1:
        c[lo1:hi1] += a[:hi1 - lo1]
    lo2 = max(idx - n + 1, 0)
    hi2 = min(idx + 1, size)
    if hi2 > lo2:
        c[lo2:hi2] += a[idx - np.arange(lo2, hi2)]
    return c


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with adaptive temperature and backtracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()

    T_curr = 40.0
    T_min = 0.05
    T_max = 100.0
    no_improve_count = 0
    last_best = best_score

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        if max_c < last_best - 1e-10:
            T_curr = max(T_min, T_curr * 0.93)
            no_improve_count = 0
            last_best = max_c
        else:
            no_improve_count += 1
            if no_improve_count % 100 == 0:
                T_curr = min(T_max, T_curr * 1.08)

        shifted = (conv - max_c) / T_curr
        shifted = np.maximum(shifted, -80.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step_base = 1.0 / (T_curr * np.sqrt(n))
        step = step_base
        for ls in range(15):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc <= max_c * 1.001 or ls == 0:
                a = a_new
                break
            step *= 0.5

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it} T={T_curr:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP-based Nesterov acceleration with dual-guided active set."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    warm_b = None
    active_duals = np.zeros(size)

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        threshold_high = M * 0.82
        threshold_low = M * 0.45
        active = np.where(conv_y >= threshold_high)[0]
        if len(active) < 40:
            active = np.where(conv_y >= threshold_low)[0]

        if t > 200:
            max_dual = np.max(active_duals)
            if max_dual > 1e-10:
                dual_active = np.where(active_duals > 0.05 * max_dual)[0]
                combined = np.unique(np.concatenate([active, dual_active]))
                if len(combined) > 300:
                    scores = np.zeros(size)
                    scores[combined] = conv_y[combined]
                    top_idx = np.argsort(scores[combined])[-300:]
                    combined = combined[top_idx]
                active = combined

        b, duals = lp_solve(y, M, solver, active if len(active) < size else None, warm_b)

        if duals is not None:
            if len(duals) == size:
                active_duals = 0.85 * active_duals + 0.15 * np.maximum(duals, 0)
            else:
                new_duals = np.zeros(size)
                new_duals[active[:len(duals)]] = np.maximum(duals, 0)
                active_duals = 0.85 * active_duals + 0.15 * new_duals

        if b is None:
            b, _ = lp_solve(y, M, solver, None, warm_b)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        warm_b = b.copy()

        if len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.003)[0]
            if len(violated) > 5:
                b_full, _ = lp_solve(y, M, solver, None, warm_b)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum
                    warm_b = b.copy()

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_targeted_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent targeting the tallest peaks with structured candidates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_objective(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        peak_threshold = cur_max * 0.88
        peak_lags = np.where(conv >= peak_threshold)[0]

        perm = rng.permutation(n)
        pass_improved = False

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            j_candidates = set()
            for _ in range(12):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:6]:
                j_s1 = int((kl - i) % n)
                j_s2 = int((kl + i) % n)
                j_candidates.add(j_s1)
                j_candidates.add(j_s2)
            j_candidates.discard(i)

            for j in list(j_candidates)[:20]:
                if j == i:
                    continue
                aj = a[j]

                contrib_i = _contrib_vec(a, i)
                contrib_j = _contrib_vec(a, j)
                delta_vec = contrib_j - contrib_i

                max_d = ai * 0.95
                n_samples = 18
                for d in np.linspace(0.05 * max_d, max_d, n_samples):
                    new_conv = conv + d * delta_vec
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    cross_idx = i + j
                    if cross_idx < size:
                        new_conv[cross_idx] += 2 * d * (aj - ai)
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1
                    conv = fast_conv_self(a)
                    cur_max = conv.max()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 2:
                n_pert = max(5, n // 60)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements > 0 and total_improvements % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD impr={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def freq_domain_init(n, rng):
    """Initialize using iterative spectral flattening with non-negativity projection."""
    a = np.ones(n)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = fast_conv_self(a).max()

    for it in range(300):
        A = fft(a, 2 * n)
        P = np.abs(A) ** 2
        target_P = P.mean() * np.ones_like(P)
        target_A = np.sqrt(target_P + 1e-10) * np.exp(1j * np.angle(A + 1e-10))
        a_new = np.real(ifft(target_A, 2 * n))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        blend = 0.25
        a = (1 - blend) * a + blend * a_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _try_update(best_state, a, label, t_global):
    """Update best_state[0] (score) and best_state[1] (sequence) if improved."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = eval_fn(a)
    if np.isfinite(sc) and sc < best_state[0]:
        best_state[0] = sc
        best_state[1] = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining:
    1. Frequency-domain informed initialization
    2. Adaptive-temperature LSE gradient descent
    3. LP Nesterov with dual-guided active set
    4. Peak-targeted coordinate descent
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_state = [float('inf'), None]  # [score, sequence]

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_state[1] = p / p.sum()
            best_state[0] = eval_fn(best_state[1])
            print(f"Parent loaded: n={len(best_state[1])}, score={best_state[0]:.6f}")
    except Exception:
        pass

    # Initialize
    if best_state[1] is None:
        n_base = 1600
        a_freq, _ = freq_domain_init(n_base, rng)
        best_state[1] = a_freq
        best_state[0] = eval_fn(a_freq)
        print(f"Freq domain init: n={n_base}, score={best_state[0]:.6f}")
    else:
        n_base = len(best_state[1])

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 1: LSE GD warmup
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_gradient_descent(best_state[1], t1, verbose=True)
        _try_update(best_state, a_out, "lse1", t_global)

    # Phase 2: LP Nesterov (main)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_state[1])
    t2 = min(remaining * 0.25, 420.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_state[1], t2, solver, verbose=True)
        _try_update(best_state, a_out, "lp1", t_global)

    # Phase 3: Peak-targeted CD
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_state[1])
    t3 = min(remaining * 0.12, 200.0)
    print(f"\n=== Phase 3: Peak CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_targeted_cd(best_state[1], t3, rng, verbose=True)
        _try_update(best_state, a_out, "pcd1", t_global)

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_state[1])
    t4 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 4: LP Nesterov r2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_state[1], t4, solver, verbose=True)
        _try_update(best_state, a_out, "lp2", t_global)

    # Phase 5: Peak CD round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_state[1])
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Peak CD r2 n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = peak_targeted_cd(best_state[1], t5, rng, verbose=True)
        _try_update(best_state, a_out, "pcd2", t_global)

    # Phase 6: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t6 = min(remaining * 0.18, 180.0)
    print(f"\n=== Phase 6: Longer n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 60 and remaining > 150:
        init_long = interp_seq(best_state[1], n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t6 * 0.6, solver, verbose=False)
        _try_update(best_state, a_out, f"lp_long{n_long}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, _ = lse_gradient_descent(init_long, min(remaining * 0.3, 50.0), verbose=False)
            _try_update(best_state, a_out2, "lse_long", t_global)

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_state[1])
    print(f"\n=== Phase 7: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        t7a = min(remaining * 0.4, 80.0)
        a_out, _ = nesterov_lp_optimize(best_state[1], t7a, solver, verbose=False)
        _try_update(best_state, a_out, "lp_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = peak_targeted_cd(best_state[1], min(remaining * 0.5, 40.0), rng, verbose=False)
            _try_update(best_state, a_out, "pcd_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = lse_gradient_descent(best_state[1], min(remaining * 0.5, 30.0), verbose=False)
            _try_update(best_state, a_out, "lse_final", t_global)

    # Final
    if best_state[1] is not None:
        final_score = eval_fn(best_state[1])
        if np.isfinite(final_score):
            best_state[0] = final_score
        print(f"\nFINAL: score={best_state[0]:.6f}, n={len(best_state[1])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_state[1]]
