```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Projected gradient descent with adaptive step size (fast, many iterations)
2. LP best-response for final polishing
3. Multiple initializations including lopsided windows and spectral shaping
4. Length adaptation to find optimal sequence length

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient(a, k_star, n_peaks=1):
    """Compute subgradient of max_k b[k] with respect to a[j]."""
    n = len(a)
    conv = fast_autocorr(a)
    # Get the top n_peaks
    sorted_idx = np.argsort(conv)[::-1]
    top_k = sorted_idx[:n_peaks]
    # Weight: use softmax-like weighting
    vals = conv[top_k]
    weights = np.exp(-(vals.max() - vals) / max(vals.max() * 0.01, 1e-10))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_k):
        for j in range(n):
            v = 0.0
            i1 = k - j
            if 0 <= i1 < n:
                v += a[i1]
            i2 = j - k
            if 0 <= i2 < n:
                v += a[i2]
            grad[j] += w * v
    return grad, int(top_k[0]), conv
```

```python
def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def subgradient_at_k(a, k, n):
    """Compute subgradient of b[k] = sum_i a[i]*a[k-i] with respect to a."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        grad[lo:hi] = 2.0 * a[k - hi + 1:k - lo + 1][::-1]
    return grad


def combined_subgradient(a, conv, n, n_peaks=5, weight_power=20.0):
    """Compute weighted combined subgradient over top peaks."""
    M = conv.max()
    if M < 1e-15:
        return np.zeros(n), 0
    # Select top peaks with softmax weighting
    threshold = M * 0.9
    active = np.where(conv >= threshold)[0]
    if len(active) > n_peaks:
        # Keep only top n_peaks
        order = np.argsort(conv[active])[::-1][:n_peaks]
        active = active[order]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    # Weights based on how close to max
    vals = conv[active]
    weights = np.exp(-weight_power * (M - vals) / max(M, 1e-10))
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, active):
        grad += w * subgradient_at_k(a, int(k), n)
    return grad, len(active)


def mirror_descent_optimize(a0, time_limit, verbose=False, seed=0):
    """Mirror descent with multiplicative updates for non-negative optimization."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = 2.0 * n * conv.max()

    step = 0.8
    momentum = 0.7
    v = np.zeros(n)
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        M = conv.max()
        score_curr = 2.0 * n * M

        grad, n_active = combined_subgradient(a, conv, n, n_peaks=min(10, max(3, n // 200)))
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Mirror descent step: a_new = a * exp(-step * grad / M) / Z
        # Use log-space for stability
        log_a = np.log(a + 1e-300)
        update = -step * grad / max(M, 1e-10)
        # Momentum in log-space
        v = momentum * v + (1 - momentum) * update
        log_a_new = log_a + v
        # Subtract max for numerical stability
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        new_score = 2.0 * n * fast_autocorr(a_new).max()
        iter_count += 1

        if new_score < best_score - 1e-12:
            a = a_new
            best_score = new_score
            best_a = a.copy()
            stall = 0
            # Slightly increase step if making good progress
            if step < 2.0:
                step = min(step * 1.005, 2.0)
        elif new_score < score_curr:
            a = a_new
            stall = 0
        else:
            stall += 1
            step *= 0.95
            momentum = min(momentum + 0.01, 0.9)
            if step < 0.01:
                break

        if stall > max_stall:
            # Restart from best with perturbation
            a = best_a * (1.0 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            v = np.zeros(n)
            step = 0.5
            momentum = 0.7
            stall = 0

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={iter_count} ({elapsed:.0f}s): score={best_score:.8f}, step={step:.4f}")

    return best_a, best_score


def projected_gradient_optimize(a0, time_limit, verbose=False, seed=0):
    """Projected gradient descent with backtracking line search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 1.0 / n
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        M = conv.max()
        score_curr = 2.0 * n * M

        grad, _ = combined_subgradient(a, conv, n, n_peaks=min(8, max(3, n // 200)))
        grad /= max(M, 1e-10)  # Normalize

        # Backtracking line search
        best_trial = a.copy()
        best_trial_score = score_curr
        s_curr = a.sum()
        for ls in range(20):
            trial_step = step * (0.5 ** ls)
            a_trial = a - trial_step * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            trial_score = 2.0 * n * fast_autocorr(a_trial).max()
            if trial_score < score_curr - 1e-13 * score_curr:
                best_trial = a_trial
                best_trial_score = trial_score
                break
            else:
                best_trial = a_trial
                best_trial_score = trial_score

        a = best_trial
        iter_count += 1

        if best_trial_score < best_score - 1e-12:
            best_score = best_trial_score
            best_a = a.copy()
            stall = 0
            step = min(step * 1.05, 5.0 / n)
        else:
            stall += 1
            step *= 0.85

        if stall > 300:
            a = best_a * (1.0 + 0.005 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            step = 1.0 / n
            stall = 0

        if step < 1e-10:
            break

        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD iter={iter_count} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lp_polish(a0, time_limit, solver, verbose=False):
    """LP best-response polishing for final precision."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None
    stall = 0

    for t in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(a)
        M_curr = conv_y.max()

        # Adaptive active set
        if active_set is None or t % 20 == 0:
            tol = max(0.005, 0.03 * np.exp(-t / 100.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            top_k = np.argsort(conv_y)[-max(3, n // 100):]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > 2 * n:
                active_set = np.arange(2 * n - 1)

        # Build constraint matrix
        k_indices = np.asarray(active_set)
        size_mat = len(k_indices)
        T = np.zeros((size_mat, n))
        for idx, k in enumerate(k_indices):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            T[idx, lo:hi] = a[k - hi + 1:k - lo + 1][::-1]

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr, cp.sum(b_var) >= 1e-6]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue

        # Blend
        g = b_val / b_sum
        alpha = 0.3 / (1.0 + t * 0.01)
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        new_score = 2.0 * n * fast_autocorr(a_new).max()
        if new_score < best_score - 1e-13:
            a = a_new
            best_score = new_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

    conv = fast_autocorr(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    pass_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        conv = fast_autocorr(a)

        # Process coordinates in random order for diversity
        order = rng.permutation(n) if pass_count > 1 else np.arange(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = a.sum() - aj

            # Try multiple candidate values
            candidates = [0.0, aj * 0.5, aj * 0.75, aj * 1.0, aj * 1.25, aj * 1.5]
            # Add values based on local structure
            if aj > 1e-10:
                candidates.append(aj * (1 + 0.1 * rng.standard_normal()))
            candidates = [max(0.0, min(x, 1.0)) for x in candidates]

            best_x = aj
            best_obj = 2.0 * n * conv.max()

            # Compute background efficiently
            # bg[k] = conv[k] - contributions involving a[j]
            c = np.zeros(size)
            for k in range(size):
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    c[k] += a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    c[k] += a[i2]

            bg = conv - 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            for x in candidates:
                new_conv = bg + 2.0 * x * c
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                new_sum = s_others + x
                if new_sum < 1e-10:
                    continue
                obj = 2.0 * n * max(new_conv.max(), 0.0) / (new_sum * new_sum)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            if best_x != aj and abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-10:
                    a = a / s_new
                    new_score = 2.0 * n * fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_a, best_score


def evaluate_safe(a):
    """Safely evaluate the score."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def make_initialization(n, kind, rng):
    """Create various initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'ramp':
        x = np.linspace(0.1, 1.0, n)
        return x / x.sum()
    elif kind == 'exp_decay':
        alpha = rng.uniform(0.5, 3.0)
        x = np.exp(-alpha * np.arange(n) / n)
        return x / x.sum()
    elif kind == 'asym_cos':
        x = 0.5 + 0.4 * np.cos(np.pi * np.arange(n) / n) + 0.1 * np.sin(2 * np.pi * np.arange(n) / n)
        x = np.maximum(x, 0.0)
        ss = x.sum()
        if ss < 1e-10:
            return np.ones(n) / n
        return x / ss
    elif kind == 'triangular':
        mid = n // 2
        x = 1.0 - np.abs(np.arange(n) - mid) / (mid + 1)
        x = np.maximum(x, 0.0)
        return x / x.sum()
    elif kind == 'side_loaded':
        # Mass concentrated on one side
        x = np.ones(n)
        x[:n // 2] *= 0.5
        return x / x.sum()
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate():
    """Main search function."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent sequence
    n_base = 1600
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
            n_base = len(parent)
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_initialization(n_base, 'asym_cos', rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    print(f"Starting with n={n_base}, score={best_score:.8f}")

    # === Phase 1: Mirror descent on parent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        print(f"\n=== Phase 1: Mirror descent n={n_base}, budget={t1:.0f}s ===")
        a_out, _ = mirror_descent_optimize(best_seq, t1, verbose=True, seed=0)
        try_update(a_out, f"md1_n{n_base}")

    # === Phase 2: Projected gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Projected gradient n={n_cur}, budget={t2:.0f}s ===")
        a_out, _ = projected_gradient_optimize(best_seq, t2, verbose=True, seed=1)
        try_update(a_out, f"pgd_n{n_cur}")

    # === Phase 3: LP polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP polish n={n_cur}, budget={t3:.0f}s ===")
        a_out, _ = lp_polish(best_seq, t3, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 4: Mirror descent round 2 (from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 30:
        print(f"\n=== Phase 4: MD round 2 n={n_cur}, budget={t4:.0f}s ===")
        a_out, _ = mirror_descent_optimize(best_seq, t4, verbose=True, seed=100)
        try_update(a_out, f"md2_n{n_cur}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length adaptation, remaining={remaining:.0f}s ===")
    n_cur = len(best_seq)
    candidate_lengths = []
    for factor in [0.9, 1.1]:
        nl = int(n_cur * factor)
        if 500 <= nl <= 5000 and nl != n_cur:
            candidate_lengths.append(nl)
    # Add some absolute candidates
    for nl in [1200, 2000, 2500]:
        if nl not in candidate_lengths and 500 <= nl <= 5000:
            candidate_lengths.append(nl)

    t5_per = min(remaining * 0.25 / max(len(candidate_lengths), 1), 120.0)
    for nl in candidate_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t5 = min(t5_per, remaining - 30)
        if t5 < 20:
            break
        print(f"  Trying n={nl}, budget={t5:.0f}s")
        a_interp = interpolate_sequence(best_seq, nl)
        a_out, _ = mirror_descent_optimize(a_interp, t5, verbose=False, seed=nl)
        try_update(a_out, f"md_len{nl}")
        # Quick LP polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(60.0, remaining * 0.3)
            a_out2, _ = lp_polish(a_out, t5b, solver, verbose=False)
            try_update(a_out2, f"lp_len{nl}")

    # === Phase 6: Perturbation + mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Perturbation restarts, remaining={remaining:.0f}s ===")
    n_restarts = 3
    t6_each = min(remaining * 0.15 / n_restarts, 80.0)
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t6 = min(t6_each, remaining - 30)
        if t6 < 20:
            break
        scale = 0.01 + 0.005 * r
        pert = best_seq * (1.0 + scale * rng.standard_normal(n_cur))
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum()
        a_out, _ = mirror_descent_optimize(pert, t6, verbose=False, seed=500 + r)
        try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final CD n={n_cur}, remaining={remaining:.0f}s ===")
        t7 = min(remaining * 0.30, 150.0)
        a_out, _ = coordinate_descent_final(best_seq, t7, seed=777)
        try_update(a_out, f"cd_final_n{n_cur}")

    # === Phase 8: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final LP n={n_cur}, remaining={remaining:.0f}s ===")
        t8 = min(remaining * 0.4, 100.0)
        a_out, _ = lp_polish(best_seq, t8, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
