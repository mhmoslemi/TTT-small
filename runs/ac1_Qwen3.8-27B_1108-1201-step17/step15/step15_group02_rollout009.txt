def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lse_fista(a0, time_limit, symmetric=True):

            break

        L = 4.0 * n / T_val
        lr = 1.0 / L
        
        t_k = 1.0

                break

            shifted = np.maximum(shifted, -100.0)

            w = exp_c / exp_c.sum()
            
            # Gradient of LSE: sum_l w_l * grad_l
            # grad of (a*a)[l] w.r.t. a is: a_rev shifted

            Y = rfft(y, fft_size)

            grad_part1 = irfft(np.conj(Y) * W, fft_size)[:n]
            grad_part2 = irfft(Y * W, fft_size)[:n]
            grad = 2.0 * (grad_part1 + grad_part2)
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            alpha = (1.0 - t_k) / t_new

def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0, g symmetric."""
    n = len(a)
    size = 2 * n - 1
    
    # Use half-dimension for symmetric case
    n_half = (n + 1) // 2
    g_half = cp.Variable(n_half, nonneg=True)
    
    # Build g from g_half (symmetric)
    # g[i] = g_half[min(i, n-1-i)]
    
    # For the convolution constraint, we need to express conv(a, g)[k] in terms of g_half
    # conv(a,g)[k] = sum_i a[i] * g[k-i]
    # where g[j] = g_half[min(j, n-1-j)]
    
    # This is complex to express directly. Let's use the full formulation instead.
    
    g_full = cp.Variable(n, nonneg=True)
    
    if active_set is not None and 0 < len(active_set) < size:
        # Build constraint matrix for active set only
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            # conv(a,g)[k] = sum_i a[i] * g[k-i]
            # valid i: max(0, k-n+1) <= i <= min(k, n-1)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                T[idx, j] += a[i]
        constraints = [T @ g_full <= M]
    else:
        # Full constraint set
        T = np.zeros((size, n))
        for k in range(size):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                T[k, j] += a[i]
        constraints = [T @ g_full <= M]
    
    # Add symmetry constraint: g[i] = g[n-1-i]
    for i in range(n // 2):
        constraints.append(g_full[i] == g_full[n - 1 - i])
    
    prob = cp.Problem(cp.Maximize(cp.sum(g_full)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(g_full.value, dtype=np.float64)
    if val is None:
        return None
    val = np.maximum(val, 0.0)
    return symmetrize(val)

def nesterov_lp(a0, time_limit, solver, symmetric=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)

    t0 = time.monotonic()
    
    y = a.copy()
    size = 2 * n - 1

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = prev_conv.max()
        if M < 1e-12:
            break
        
        # Adaptive active set: use constraints where conv > M * threshold
        # Start with a relatively tight threshold, relax if LP is infeasible
        threshold = 0.8
        active_set = np.where(prev_conv > M * threshold)[0]
        
        if len(active_set) < 5:
            active_set = np.where(prev_conv > M * 0.5)[0]
        if len(active_set) > min(300, size // 3):
            # Too many constraints, use tighter threshold
            threshold = 0.9
            active_set = np.where(prev_conv > M * threshold)[0]
            if len(active_set) < 3:
                active_set = np.where(prev_conv > M * 0.7)[0]
        
        b = lp_best_response(y, M, solver, active_set)
        
        if b is None:
            # Try without active set restriction
            b = lp_best_response(y, M, solver, None)
        
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        if symmetric:
            y_new = symmetrize(y_new)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            if symmetric:
                a_new = symmetrize(a_new)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        prev_conv = fast_conv_self(y)
        sc_y = prev_conv.max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, |active|={len(active_set)}")
    
    return best_a, best_score

    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), norm_val, symmetric)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 30
    temps = np.geomspace(60.0, 0.01, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        n_steps = max(30, min(150, int(remaining * 10)))
        L = 4.0 * n / T_val
        lr = 1.0 / L
        
        t_k = 1.0
        y = a.copy()
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv_y = fast_conv_self(y)
            max_c = conv_y.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv_y - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            Y = rfft(y, fft_size)
            
            grad = 2.0 * (irfft(np.conj(Y) * W, fft_size)[:n] + irfft(Y * W, fft_size)[:n])
            
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            alpha = (1.0 - t_k) / t_new
            
            x_new = y - lr * grad
            x_new = np.maximum(x_new, 0.0)
            if symmetric:
                x_new = symmetrize(x_new)
            s_new = x_new.sum()
            if s_new < 1e-15:
                x_new = y.copy()
            else:
                x_new = x_new * norm_val / s_new
            
            y = x_new + alpha * (x_new - a)
            y = np.maximum(y, 0.0)
            if symmetric:
                y = symmetrize(y)
            s_y = y.sum()
            if s_y < 1e-15:
                y = x_new.copy()
            else:
                y = y * norm_val / s_y
            
            a = x_new
            t_k = t_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def build_conv_matrix_row(a, k, n):
    """Build the k-th row of the convolution matrix: row[j] = a[i] where k-i=j, i valid."""
    row = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(k, n - 1)
    for i in range(i_start, i_end + 1):
        j = k - i
        row[j] = a[i]
    return row

def lp_best_response(a, M, solver, active_set=None, symmetric=True):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0, (g symmetric)."""
    n = len(a)
    size = 2 * n - 1
    g_full = cp.Variable(n, nonneg=True)
    
    if active_set is not None and 0 < len(active_set) < size:
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                T[idx, j] += a[i]
        constraints = [T @ g_full <= M]
    else:
        T = np.zeros((size, n))
        for k in range(size):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                T[k, j] += a[i]
        constraints = [T @ g_full <= M]
    
    if symmetric:
        for i in range(n // 2):
            constraints.append(g_full[i] == g_full[n - 1 - i])
    
    prob = cp.Problem(cp.Maximize(cp.sum(g_full)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(g_full.value, dtype=np.float64)
    if val is None:
        return None
    val = np.maximum(val, 0.0)
    if symmetric:
        val = symmetrize(val)
    return val

def nesterov_lp(a0, time_limit, solver, symmetric=True):
    """LP best-response with Nesterov acceleration and adaptive active set."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), norm_val, symmetric)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    size = 2 * n - 1
    
    # Precompute the convolution matrix rows for efficiency
    # Only store rows we actually need (active set changes each iteration)
    
    for t in range(1, 30000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive active set
        frac = 0.85
        active_set = np.where(conv_y > M * frac)[0]
        
        if len(active_set) < 10:
            active_set = np.where(conv_y > M * 0.6)[0]
        if len(active_set) > 200:
            active_set = np.where(conv_y > M * 0.92)[0]
            if len(active_set) < 5:
                active_set = np.where(conv_y > M * 0.75)[0]
        
        b = lp_best_response(y, M, solver, active_set, symmetric)
        
        if b is None:
            b = lp_best_response(y, M, solver, None, symmetric)
        
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        if symmetric:
            y_new = symmetrize(y_new)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            if symmetric:
                a_new = symmetrize(a_new)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, |A|={len(active_set)}")
    
    return best_a, best_score

def precompute_shifts(a, n, size):
    """Precompute the cross-correlation shifts for fast CD."""
    # shift_i[k] = a[k-i] if 0 <= k-i < n, else 0
    # This is the contribution of e_i to conv(a, .)
    shifts = np.zeros((n, size))
    for i in range(n):
        # a convolved with e_i: shift a by i positions (with zero padding on the right)
        # conv(a, e_i)[k] = a[k-i] if 0 <= k-i < n
        start = i
        end = min(i + n, size)
        shifts[i, start:end] = a[:end - start]
    return shifts

def pair_cd_fast(a0, time_limit, rng, symmetric=True):
    """Fast pair coordinate descent using precomputed convolution shifts."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), norm_val, symmetric)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Precompute: for each position i, the "correlation vector" c_i[k] = a[k-i] (shifted)
        # When moving d from i to j: new_conv = conv - 2d*c_i + 2d*c_j + O(d^2)
        # where c_i[k] = a[k-i] if valid, 0 otherwise (for linear conv)
        # Actually for self-conv: (a*a)[k] = sum_i a[i]*a[k-i]
        # d/da[j] (a*a)[k] = a[k-j] + a[j+k-(n-1)*2]... 
        # Let me use: grad_j[k] = a[k-j] (if 0<=k-j<n) + a[j-(k)] (if 0<=j-k<n)
        # = c_j[k] + c_j_rev[k] where c_j is a shifted by j
        
        # Precompute all shift vectors
        c = np.zeros((n, size))
        for j in range(n):
            # c[j, k] = a[k-j] if 0<=k-j<n else 0
            start = j
            end = min(j + n, size)
            c[j, start:end] = a[:end - start]
            # c[j, k] += a[j-k] if 0<=j-k<n else 0 (the "reflected" part)
            # a[j-k] is valid when 0<=j-k<n, i.e., -n<k-j<0, i.e., -n<k-j
            # k = j - i for i in [0,n), so k in [j-n+1, j]
            start2 = max(0, j - n + 1)
            end2 = min(j + 1, size)
            for k in range(start2, end2):
                idx = j - k
                if 0 <= idx < n:
                    c[j, k] += a[idx]
        
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm_val * 1e-10:
                continue
            
            # Candidate j's: positions near the peaks
            j_candidates = []
            peak_idx = np.argsort(conv)[-12:]
            for kl in peak_idx:
                # j should be such that moving mass there reduces the peak
                j_candidates.append(int((kl - i) % n))
                j_candidates.append(int((kl + i) % n))
            for _ in range(5):
                j_candidates.append(int(rng.integers(n)))
            
            j_set = list(set(j_candidates))
            j_set = [j for j in j_set if j != i and a[j] < 100 * ai]
            
            best_delta = 0.0
            best_v = best_score
            best_j = -1
            
            ci = c[i]
            
            for j in j_set[:15]:
                cj = c[j]
                diff = cj - ci  # direction of change
                d_max = 0.7 * ai
                n_samples = 15
                deltas = np.linspace(0.005 * d_max, d_max, n_samples)
                
                for d in deltas:
                    new_conv = conv + 2.0 * d * diff
                    # Add d^2 corrections
                    # At position 2*i: (a[i]-d)^2 + 2*(a[i]-d)*0... 
                    # Actually: the d^2 term at lag 2*i-n+1 (or wherever e_i*e_i lands)
                    # e_i * e_i has a 1 at position 2*i... wait no.
                    # For linear conv: (e_i * e_i)[k] = 1 if k = 2*i, 0 otherwise
                    # But only if 2*i < size
                    if 2 * i < size:
                        new_conv[2 * i] += d * d - 2.0 * d * (a[i] - 0.0)
                        # Wait, this is getting complicated. The d^2 term:
                        # (a - d*e_i + d*e_j)*(a - d*e_i + d*e_j)
                        # = a*a - 2d*(a*e_i) + 2d*(a*e_j) + d^2*(e_i*e_i - 2*e_i*e_j + e_j*e_j)
                        # The -2d*(a*e_i) and +2d*(a*e_j) are the linear terms we already have
                        # The d^2 term: e_i*e_i has 1 at lag 0 (position i), 
                        # no wait: (e_i * e_i)[k] = 1 if k = i + i = 2i? No!
                        # Linear conv: (x*y)[k] = sum_m x[m]*y[k-m]
                        # (e_i * e_i)[k] = e_i[i]*e_i[k-i] = 1 if k-i = i, i.e., k = 2i
                        # But only if 0 <= k-i < n, i.e., 0 <= i < n (always true)
                        # So (e_i*e_i)[2i] = 1 (if 2i < size)
                        # Similarly (e_j*e_j)[2j] = 1
                        # (e_i*e_j)[k] = 1 if k = i+j (if valid)
                    # For now, ignore d^2 (it's small for small d)
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                if symmetric:
                    a = symmetrize(a)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_val / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = np.maximum(a, 0.0)
                        if symmetric:
                            a = symmetrize(a)
                        a = a * norm_val / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                # Perturbation
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                if symmetric:
                    a = symmetrize(a)
                a = normalize(a, norm_val, symmetric)
                if no_improve >= 15:
                    break
        else:
            no_improve = 0
        
        if (idx // n) % 50 == 0 and pass_improved:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng, symmetric=True):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.3, 0.5)
        c2 = 1.0 - c1
        a = np.exp(-50.0 * (x - c1) ** 2) + np.exp(-50.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        centers = np.sort(centers)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
            if symmetric:
                a += np.exp(-(x - (1 - c)) ** 2 / (2 * width ** 2))
    elif choice == 7:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 2 * x) + 0.2 * np.sin(2 * np.pi * 5 * x)
    elif choice == 8:
        a = np.sinc(3.0 * (x - 0.5)) ** 2 + 0.2
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
    
    a = np.maximum(a, 0.0)
    if symmetric:
        a = symmetrize(a)
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    if symmetric:
        a = symmetrize(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a / s
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function trying multiple strategies to minimize the score."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng, symmetric=True)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")
    
    # Symmetrize the starting point
    a_work = symmetrize(best_seq).astype(np.float64)
    a_work = a_work / a_work.sum()
    sc_sym = eval_fn(a_work)
    if sc_sym < best_score:
        best_score = sc_sym
        best_seq = a_work.copy()
        print(f"  Symmetrized improvement: {best_score:.6f}")
    
    # Phase 1: LSE-FISTA (fast initial progress)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_fista(a_work, t1, symmetric=True)
        check_and_update(a_out, "fista1")
    
    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP-Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp(best_seq, t2, solver, symmetric=True)
        check_and_update(a_out, "lp_nest1")
    
    # Phase 3: LSE-FISTA round 2 from LP result
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.1, 120.0)
    print(f"\n=== Phase 3: LSE-FISTA round 2 (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = lse_fista(best_seq, t3, symmetric=True)
        check_and_update(a_out, "fista2")
    
    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: LP-Nesterov round 2 (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp(best_seq, t4, solver, symmetric=True)
        check_and_update(a_out, "lp_nest2")
    
    # Phase 5: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 5: Pair CD polish (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_fast(best_seq, t5, rng, symmetric=True)
        check_and_update(a_out, "pcd1")
    
    # Phase 6: Try slightly different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 6: Multi-length exploration (remaining={remaining:.0f}s) ===")
        n_options = [n_base - 100, n_base + 100, n_base - 200, n_base + 200]
        n_options = [x for x in n_options if 500 <= x <= 3000]
        
        for n_try in n_options:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t_try = min(remaining * 0.4, 40.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            init_try = symmetrize(init_try)
            init_try = init_try / init_try.sum()
            
            a_out, _ = lse_fista(init_try, t_try, symmetric=True)
            check_and_update(a_out, f"fista_n{n_try}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 25 and t_try > 0:
                t_lp = min(remaining * 0.5, 30.0)
                if t_lp > 15:
                    a_out, _ = nesterov_lp(best_seq, t_lp, solver, symmetric=True)
                    check_and_update(a_out, f"lp_n{n_try}")
    
    # Phase 7: Final aggressive CD
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7 = min(remaining * 0.6, 120.0)
        print(f"\n=== Phase 7: Final CD (budget={t7:.0f}s) ===")
        if t7 > 15:
            a_out, _ = pair_cd_fast(best_seq, t7, rng, symmetric=True)
            check_and_update(a_out, "pcd_final")
    
    # Phase 8: Try fresh start with different structure
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 8: Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            n_fresh = n_base + int(rng.integers(-100, 200))
            n_fresh = max(800, min(2500, n_fresh))
            a_fresh = structured_init(n_fresh, rng, symmetric=True)
            t_fresh = min(remaining * 0.4, 20.0)
            a_out, _ = lse_fista(a_fresh, t_fresh, symmetric=True)
            check_and_update(a_out, f"fresh{attempt}_n{n_fresh}")
    
    # Final symmetrization check
    if best_seq is not None:
        a_sym_final = symmetrize(best_seq)
        a_sym_final = a_sym_final / a_sym_final.sum()
        check_and_update(a_sym_final, "final_sym")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0, 1.0]
    
    return [float(x) for x in best_seq]
