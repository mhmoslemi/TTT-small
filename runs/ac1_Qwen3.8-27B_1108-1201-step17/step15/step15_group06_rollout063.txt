    a = np.maximum(a, 0)

    eps = 1e-8

    best_score = evaluate_local(a)

        if time.monotonic() - t0 > time_budget:
            break
        
        # Current beta (anneal from beta_start to beta_end)
        progress = t / max(n_iter - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute objective and gradient
        c = fast_autocorr(a)
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        log_sum_exp = c_max + np.log(exp_bc.sum())
        w = exp_bc / exp_bc.sum()
        
        S = a.sum()
        if S < 1e-10:
            a = np.ones(n) / n
            S = 1.0
        
        # Smooth objective
        g_beta = (1.0/beta) * log_sum_exp
        f = 2.0 * n * g_beta / (S * S)

        corr = 2.0 * np.correlate(w, a, mode='valid')
        grad = 2.0 * n / (S * S) * corr - 4.0 * n * g_beta / (S**3)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        
        # Check improvement
        score = 2.0 * n * c.max() / (S * S)  # Use actual max, not smooth
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

```python
def simplex_project(x, S):
    """Project x onto {a >= 0, sum(a) = S}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0]
    if len(rho) == 0:
        return x - (x.sum() - S) / n  # all positive, just shift
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)
```

This is O(n log n) due to the sort. For n=1600, this is trivial.

def simplex_project(x, S):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.maximum(x - (x.sum() - S) / n, 0.0)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def power_sum_grad(c, a, p):
    n = len(a)
    c_safe = np.maximum(c, 1e-30)
    if p == 1:
        w = np.ones(len(c)) / len(c)
    else:
        c_pow = c_safe ** (p - 1)
        c_sum = c_pow.sum()
        if c_sum < 1e-30:
            w = np.ones(len(c)) / len(c)
        else:
            w = c_pow / c_sum
    corr = 2.0 * np.correlate(w, a, mode='valid')
    return corr


def fista_power_sum(a0, p, S, time_limit, verbose=False, seed=0, max_iter=50000):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    y = a.copy()
    best_a = a.copy()
    c = fast_autocorr(a)
    best_val = (c ** p).sum() ** (1.0 / p)
    best_score_max = 2.0 * n * c.max() / (S * S)

    t_k = 1.0
    gamma = 1.0 / (2.0 * p * max(best_val, 0.01))
    gamma = min(gamma, 1.0)
    t0 = time.monotonic()
    stall = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        c_y = fast_autocorr(y)
        c_y_safe = np.maximum(c_y, 1e-30)
        f_val = (c_y_safe ** p).sum() ** (1.0 / p)
        f_val = max(f_val, 1e-15)

        grad = power_sum_grad(c_y, y, p)
        scale = 2.0 * f_val ** (1.0 - p)

        t_k = t_k_new
        y = y_new

        c_a = fast_autocorr(a_new)
        f_a = (np.maximum(c_a, 1e-30) ** p).sum() ** (1.0 / p)
        if f_a < best_val * (1.0 - 1e-10):
            best_val = f_a
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        score_max = 2.0 * n * c_a.max() / (S * S)
        if score_max < best_score_max:
            best_score_max = score_max

        if stall > 200:
            y = best_a.copy()
            t_k = 1.0
            gamma = max(gamma * 0.5, 1e-6)
            stall = 0

    c_best = fast_autocorr(best_a)
    return best_a, 2.0 * n * c_best.max() / (S * S)


def fista_continuation(a0, time_limit, verbose=True, seed=0):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    p_schedule = [2, 4, 8, 16, 32, 64, 128, 256]
    t0 = time.monotonic()
    best_score = evaluate_local(a)
    best_a = a.copy()

    for p in p_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        alloc = remaining / (len(p_schedule) + 1)
        alloc = min(alloc, remaining * 0.3)
        if alloc < 2:
            break
        a_out, score = fista_power_sum(a, p, S, alloc, verbose=verbose, seed=seed + p)
        if score < best_score:
            best_score = score
            best_a = a_out
            if verbose:
                print(f"  Continuation p={p}: score={score:.10f}")
        a = a_out

    return best_a, best_score


def build_active_constraints(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_max_sum(a, M, solver, active_idx=None):

    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= 2 * n - 1:

        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)

    return val
