2. Multi-start with diverse non-symmetric initializations
3. Targeted peak deflation (direct convolution peak reduction)
4. Length exploration and interpolation
5. Fine-tuning with very low temperature softmax

def fast_conv_self(a):

    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

def gradient_lse(a, T):
    n = len(a)
    c = fast_conv_self(a)
    cm = c.max()
    logits = np.clip((c - cm) / T, -80.0, 0.0)
    w = np.exp(logits)

    if w_sum < 1e-30:

        w.argmax() if c.size > 0 else None

    g = np.correlate(w, a, mode='valid')

def lse_momentum_optimize(a0, time_budget, T_start=40.0, T_end=0.05, n_stages=25, base_lr=None, momentum=0.9):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * 0.1 + 0.1 * np.arange(n)
        a = np.maximum(a, 0.01)
    a = a / a.sum()

    best_a = a.copy()
    best_cmax = fast_conv_self(a).max()
    t0 = time.monotonic()

    if base_lr is None:
        base_lr = 0.3 / np.sqrt(n)

    temps = np.geomspace(T_start, T_end, n_stages)
    vel = np.zeros(n)

    for T_val in temps:
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(15, min(200, int(remaining * 15)))

            if time.monotonic() - t0 > time_budget:

def deflation_step(a, n_iterations=200, peak_threshold=0.95):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_cmax = fast_conv_self(a).max()
    t0 = time.monotonic()

    for it in range(n_iterations):
        if time.monotonic() - t0 > 30:
            break
        c = fast_conv_self(a)
        cmax = c.max()
        if cmax < 1e-15:
            break
        threshold = cmax * peak_threshold
        peak_lags = np.where(c >= threshold)[0]

        if len(peak_lags) == 0:
            break

        pressure = np.zeros(n)
        for k in peak_lags:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            indices = np.arange(lo, hi)
            contrib = a[k - indices]
            pressure[indices] += contrib * (c[k] / cmax)

        pmax = pressure.max()
        if pmax < 1e-15:
            break

        eta = 0.05 / pmax
        a_new = a - eta * pressure
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        c_new = fast_conv_self(a_new).max()
        if c_new < best_cmax:
            best_cmax = c_new
            best_a = a_new.copy()
            a = a_new

def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    a_norm = a / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(x_new, x_old, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

        best_cmax = c_out
        best_a = a_out.copy()

    # Phase 2: Deflation
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 10:
        a_out, c_out = deflation_step(best_a, n_iterations=min(200, int(remaining * 5)))
        if c_out < best_cmax:
            best_cmax = c_out
            best_a = a_out.copy()

    # Phase 3: Fine LSE
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 10:
        a_out, c_out = lse_momentum_optimize(best_a, remaining * 0.5, T_start=1.0, T_end=0.02, n_stages=15, momentum=0.9, base_lr=0.1 / np.sqrt(n))
        if c_out < best_cmax:
            best_cmax = c_out
            best_a = a_out.copy()

    # Phase 4: Local perturbation
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 5:
        a_out, c_out = local_perturbation_search(best_a, remaining, rng)
        if c_out < best_cmax:
            best_cmax = c_out
            best_a = a_out.copy()

    return best_a, best_cmax

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = non_sym_init(n_base, rng, style=0)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Optimize from parent (primary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: Two-phase from parent n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = two_phase_optimize(best_seq, t1, rng)
        update_best(a_out, "phase1")

    # === Phase 2: Multi-start exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_starts = 5
    t_per_start = min(remaining / (n_starts + 1), 180.0)
    print(f"\n=== Phase 2: Multi-start ({n_starts} starts, {t_per_start:.0f}s each) ===")

    styles = [0, 1, 2, 3, 5]
    for i in range(min(n_starts, len(styles))):
        if time.monotonic() - t_global > time_budget - 60:
            break
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        a_init = non_sym_init(n_base, rng, style=styles[i])
        t_s = min(t_per_start, remaining * 0.4)
        a_out, _ = two_phase_optimize(a_init, t_s, rng)
        update_best(a_out, f"start{i}_s{styles[i]}")

    # === Phase 3: Deep refinement of best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.35, 400.0)
    print(f"\n=== Phase 3: Deep refinement n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        # LSE with very fine temperature
        n = len(best_seq)
        a_out, _ = lse_momentum_optimize(best_seq, t3 * 0.5, T_start=2.0, T_end=0.01, n_stages=20, momentum=0.92, base_lr=0.15 / np.sqrt(n))
        update_best(a_out, "deep_lse")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = deflation_step(best_seq, n_iterations=min(300, int(remaining * 3)))
            update_best(a_out, "deep_defl")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = lse_momentum_optimize(best_seq, min(remaining * 0.5, 80), T_start=0.5, T_end=0.005, n_stages=15, momentum=0.95, base_lr=0.05 / np.sqrt(n))
            update_best(a_out, "deep_fine")

    # === Phase 4: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 4: Length exploration (remaining={remaining:.0f}s) ===")
        n_current = len(best_seq)
        lengths = [n_current - 200, n_current + 200, n_current + 400, n_current - 400]
        lengths = [l for l in lengths if 200 <= l <= 3000]
        for n_try in lengths:
            if time.monotonic() - t_global > time_budget - 40:
                break
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t_try = min(remaining * 0.3, 80.0)
            a_interp = interpolate_to_length(best_seq, n_try)
            a_out, _ = two_phase_optimize(a_interp, t_try, rng)
            update_best(a_out, f"len{n_try}")

    # === Phase 5: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 5: Final local search (remaining={remaining:.0f}s) ===")
        a_out, _ = local_perturbation_search(best_seq, min(remaining * 0.5, 60), rng, n_trials=100)
        update_best(a_out, "final_local")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            n = len(best_seq)
            a_out, _ = lse_momentum_optimize(best_seq, min(remaining * 0.7, 30), T_start=0.1, T_end=0.001, n_stages=10, momentum=0.97, base_lr=0.02 / np.sqrt(n))
            update_best(a_out, "final_ultrafine")

    # === Final evaluation ===
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        n_final = len(best_seq)
        print(f"\nFINAL: score={best_score:.6f}, n={n_final}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, x)) for x in best_seq]
    return result
