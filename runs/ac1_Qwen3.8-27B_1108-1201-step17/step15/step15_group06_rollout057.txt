def fast_autocorr(a):
    """Compute full convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def evaluate_safe(a):
    """Call the official evaluator."""
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lse_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv(a,a)[k]))
    Returns (smoothed_value, gradient_array).
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    conv = fast_autocorr(a)
    # Softmax weights
    shifted = tau * (conv - conv.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return conv.max(), np.zeros(n)
    w = w / w_sum  # normalized probability distribution over lags
    # Gradient: grad[i] = c_full[i] + c_full[2n-2-i] where c_full = conv(w, a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    c_full = irfft(W * A, fft_size)[:3 * n - 2]
    # grad[i] = c_full[i] + c_full[2n-2-i]
    grad = c_full[:n] + c_full[n - 1:2 * n - 1][::-1]
    lse_val = (conv.max() + np.log(w_sum)) / tau
    return lse_val, grad / tau  # divide by tau since we have (1/tau)*log(...)


def build_sparse_T(a, active_idx):
    """Build sparse constraint matrix for conv(b, a) at active positions."""
    from scipy.sparse import lil_matrix
    n = len(a)
    m = len(active_idx)
    T = lil_matrix((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[row, j] = a[k - j]
    return T.tocsr()


def lp_max_sum_sparse(a, M, solver, T):
    """Maximize sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b.value is None:
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_min_max_sparse(a, solver, T, normalize_sum):
    """Minimize t s.t. T@b <= t, sum(b) = normalize_sum, b >= 0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    cons = [T @ b <= t, cp.sum(b) == normalize_sum]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    if b.value is None:
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def spectral_init(n, rng, style=0):
    """Initialize using spectral design ideas."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        # Tapered with oscillation
        a = np.sin(np.pi * x) ** 1.5 * (1.0 + 0.3 * np.cos(6 * np.pi * x))
    elif style == 1:
        # Asymmetric taper
        a = np.sin(np.pi * (x + 0.1)) ** 1.2 * np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0.0)
    elif style == 2:
        # Double peak
        a = np.exp(-20 * (x - 0.25) ** 2) + 0.7 * np.exp(-20 * (x - 0.75) ** 2)
    elif style == 3:
        # Inverse cosine taper
        a = 0.5 + 0.5 * np.cos(np.pi * (2 * x - 1))
    elif style == 4:
        # Power law taper
        a = (1.0 - (2.0 * x - 1.0) ** 4)
    elif style == 5:
        # Spectral: design flat power spectrum
        # Use random phases with flat amplitude
        N = n
        freqs = np.arange(N)
        amp = np.ones(N)
        phase = rng.uniform(0, 2 * np.pi, N)
        spectrum = amp * np.exp(1j * phase)
        # Enforce Hermitian symmetry for real signal
        spectrum[N // 2] = np.abs(spectrum[N // 2]) if N % 2 == 0 else 0
        a = np.real(np.fft.ifft(spectrum))
        a = np.maximum(a, 0.0)
    elif style == 6:
        # Modulated taper
        a = np.sin(np.pi * x) ** 0.8 * (0.7 + 0.3 * np.sin(4 * np.pi * x + 0.5))
        a = np.maximum(a, 0.0)
    else:
        # Smooth with high-freq component
        a = np.sin(np.pi * x) ** 2 + 0.15 * np.sin(8 * np.pi * x) * np.sin(np.pi * x)
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_v2(a0, time_limit, solver, verbose=True, seed=0):
    """Improved Nesterov LP with top-K active set and adaptive strategy."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    T_cache = None
    active_set = None
    T_L = None

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set: top-K most significant + margin
        if active_set is None or t % 30 == 0:
            K = int(0.15 * size) + 20
            # Use top values with some margin
            sorted_vals = np.sort(conv_y)[::-1]
            threshold = sorted_vals[min(K, size - 1)] * 0.95
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 10:
                active_set = np.argsort(conv_y)[-K:]
            if len(active_set) > 0.8 * size:
                active_set = np.argsort(conv_y)[-int(0.4 * size):]

        # Build sparse T (cache if active set unchanged)
        if T_L is None or not np.array_equal(active_set, T_L):
            T_cache = build_sparse_T(y, active_set)
            T_L = active_set.copy()

        # Solve max-sum LP
        b_max = lp_max_sum_sparse(y, M_curr, solver, T_cache)

        # Periodically solve min-max LP for blending
        b_min = None
        if t % 5 == 0 and t > 50:
            b_min, _ = lp_min_max_sparse(y, solver, T_cache, norm)

        if b_max is None:
            # Retry with full constraints
            if t % 50 == 0:
                active_set = np.arange(size)
                T_cache = build_sparse_T(y, active_set)
                T_L = active_set.copy()
                b_max = lp_max_sum_sparse(y, M_curr, solver, T_cache)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        # Blend with min-max if available
        if b_min is not None and b_min.sum() > 1e-10:
            g_min = b_min * norm / b_min.sum()
            blend = 0.2 + 0.1 * np.exp(-t / 200.0)
            g = (1.0 - blend) * g + blend * g_min

        # Verify constraints on inactive set
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0 and len(inactive) < size:
                T_check = build_sparse_T(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * 1.001)[0]
                if len(viol_idx) > 0:
                    new_active = np.union1d(active_set, inactive[viol_idx[:50]])
                    active_set = new_active
                    T_cache = build_sparse_T(y, active_set)
                    T_L = active_set.copy()

        # Nesterov update with adaptive step
        alpha_t = max(2.0 / (t + 2.0), 0.05)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = min(t / (t + 3.0), 0.92)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        sc_a = fast_autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average every 50 steps
        if t % 50 == 0:
            avg_candidate = y * 0.5 + a * 0.5
            s_avg = avg_candidate.sum()
            if s_avg > 1e-15:
                avg_candidate = avg_candidate * norm / s_avg
                sc_avg = fast_autocorr(avg_candidate).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_candidate.copy()
                    stall_count = 0

        # Restart on stall
        if stall_count > 60:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            T_cache = None
            T_L = None
            stall_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP2 t={t} ({elapsed:.0f}s): best_M={best_score:.10f}")

    return best_a, best_score


def lse_optimize(a0, time_limit, verbose=True, seed=0):
    """
    Log-sum-exp smoothed optimization with Adam.
    Minimizes (1/tau)*log(sum_k exp(tau*conv(a,a)[k]))
    with adaptive tau (annealing).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.01
    lr_decay = 0.9995

    # Adaptive tau: start smooth, get sharper
    tau = 5.0
    tau_min = 50.0
    tau_anneal = 1.005

    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        lse_val, grad = lse_gradient(a, tau)

        # Gradient step with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        lr *= lr_decay

        # Check improvement
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal tau periodically
        if step % 100 == 0:
            tau = min(tau * tau_anneal, tau_min)
            # Adaptive LR based on improvement
            if no_improve > 50:
                lr = max(lr * 0.5, 1e-5)
                no_improve = 0
            elif no_improve > 20:
                lr = max(lr * 0.7, 1e-5)

        # Random perturbation to escape plateaus
        if no_improve > 200:
            noise = rng.standard_normal(n) * 0.01 * norm
            a = np.maximum(a + noise, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a
            no_improve = 0

        if no_improve > 1000:
            break

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={step} ({elapsed:.0f}s): best_M={best_score:.10f}, tau={tau:.1f}, lr={lr:.6f}")

    return best_a, best_score


def block_cd(a0, time_limit, rng, block_size=16, verbose=True):
    """Block coordinate descent with multiple perturbation trials."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    n_blocks = (n + block_size - 1) // block_size

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        for bi in range(n_blocks):
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            block = a[start:end].copy()
            block_best = block.copy()
            block_best_score = best_score

            # Try several perturbations of this block
            for trial in range(8):
                if trial == 0:
                    candidate = block.copy()
                else:
                    scale = 0.1 * (trial / 8.0)
                    candidate = block * (1.0 + scale * rng.standard_normal(end - start))
                    candidate = np.maximum(candidate, 0.0)

                a_test = a.copy()
                a_test[start:end] = candidate
                s_test = a_test.sum()
                if s_test < 1e-15:
                    continue
                a_test = a_test * norm / s_test
                sc = fast_autocorr(a_test).max()

                if sc < block_best_score - 1e-12:
                    block_best_score = sc
                    block_best = candidate.copy()

            if block_best_score < best_score - 1e-12:
                a[start:end] = block_best
                best_score = block_best_score
                best_a = a.copy()
                pass_improved = True
                no_improve = 0
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD pass={n_passes} ({elapsed:.0f}s): best_M={best_score:.10f}")

    return best_a, best_score


def coord_descent_v2(a0, time_limit, rng, verbose=False):
    """Coordinate descent with binary search for optimal value."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-14 * norm:
                continue

            # Try several multipliers for a[i]
            best_mult = 1.0
            for mult in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]:
                a_test = a.copy()
                a_test[i] = ai * mult
                s_test = a_test.sum()
                if s_test < 1e-15:
                    continue
                a_test = a_test * norm / s_test
                sc = fast_autocorr(a_test).max()
                if sc < best_score * mult - 1e-12:
                    pass  # placeholder
            # Actually do proper search
            current_score = best_score
            improved = False
            for mult in [0.5, 0.7, 1.5, 2.0, 0.3]:
                a_test = a.copy()
                a_test[i] = ai * mult
                s_test = a_test.sum()
                if s_test < 1e-15:
                    continue
                a_test = a_test * norm / s_test
                sc = fast_autocorr(a_test).max()
                if sc < best_score - 1e-12:
                    a = a_test
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    pass_improved = True
                    no_improve = 0
                    break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

        a_out, _ = nesterov_lp_v2(a_norm, t4, solver, verbose=False, seed=200 + r * 13)
        try_update(a_out, f"pert{r}_n{n_cur}")

        # Quick LSE polish after each restart
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t4b = min(40.0, remaining - 30)
            if t4b > 20:
                a_out2, _ = lse_optimize(a_out, t4b, verbose=False, seed=300 + r)
                try_update(a_out2, f"pert{r}_lse_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Try longer length (10% time) ===
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 5: LP Nesterov n={n_long} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 40:
        init_long = interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_v2(a_norm, t5, solver, verbose=True, seed=777)
        try_update(a_out, f"lp_long_n{n_long}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Final LSE polish on best ===
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LSE polish n={n_best} ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t6, verbose=True, seed=999)
        try_update(a_out, f"lse_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Final LP polish ===
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 30:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_v2(a_norm, t7, solver, verbose=True, seed=1234)
            try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final CD ===
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent_v2(a_norm, t8, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
