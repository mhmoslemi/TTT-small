If we have a dual feasible solution y >= 0 (satisfying T^T y >= 1), then for any primal feasible b:
sum(b) = 1^T b <= (T^T y)^T b = y^T (Tb) <= y^T (M*1) = M * sum(y)

So M * sum(y) is an upper bound on the optimal primal value. If we also have a primal feasible b, then sum(b) is a lower bound. The gap is M * sum(y) - sum(b).

For constraint screening: if a constraint k is not active (i.e., (Tb)[k] < M at the optimal b), then by complementary slackness, y[k] = 0 at the optimal dual. So we can identify "inactive" constraints by checking if y[k] = 0.

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    print("  Using solver (fallback): ECOS")

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_constraint_matrix(a, active=None, size=None):
    """Build the constraint matrix T[active, :] for the LP."""
    n = len(a)
    if size is None:
        size = 2 * n - 1
    rows = active if active is not None else range(size)
    T = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[idx, j] = a[k - j]
    return T


def lp_solve_screened(a, M, solver, active_set, warm_b=None):
    """Solve LP with constraint generation: start with active_set, add violated constraints."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    constraints = [build_constraint_matrix(a, active_set, size) @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b = np.asarray(b_var.value, dtype=np.float64)
    b = np.maximum(b, 0.0)

    # Constraint generation: check for violations
    for _ in range(5):
        if b is None or b.sum() < 1e-15:
            break
        conv_b = fast_conv_cross(a, b)
        violated = np.where(conv_b > M * 1.001)[0]
        new_violated = violated[~np.in1d(violated, active_set)]
        if len(new_violated) == 0:
            break
        active_set = np.concatenate([active_set, new_violated])
        active_set = np.unique(active_set)

        constraints = [build_constraint_matrix(a, active_set, size) @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None, None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
        b = np.asarray(b_var.value, dtype=np.float64)
        b = np.maximum(b, 0.0)

    return b, active_set


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with adaptive temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(20, min(400, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            # Adaptive step size
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si}/15 T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and dual-based constraint screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Initial active set
    conv_y = fast_conv_self(y)
    M_init = conv_y.max()
    active = np.where(conv_y >= M_init * 0.85)[0]
    if len(active) < 50:
        active = np.where(conv_y >= M_init * 0.60)[0]

    max_iters = 500000
    stagnant_count = 0
    best_stagnant_score = best_score

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Update active set based on current y
        # Use dual-guided: keep constraints that are near the maximum
        threshold = M * 0.88
        new_active = np.where(conv_y >= threshold)[0]
        if len(new_active) < 30:
            threshold = M * 0.65
            new_active = np.where(conv_y >= threshold)[0]
        # Merge with previous active set (keep some for stability)
        if t > 10:
            old_important = np.where(conv_y >= M * 0.95)[0]
            active = np.unique(np.concatenate([new_active, old_important]))
        else:
            active = new_active

        b, active = lp_solve_screened(y, M, solver, active, warm_b=None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stagnant_count = 0
            best_stagnant_score = best_score
        else:
            if best_score < best_stagnant_score * (1 - 1e-6):
                stagnant_count += 1
            else:
                stagnant_count += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, n_act={len(active)}")

        # Early stop if very stagnant for a long time
        if stagnant_count > 500 and t > 200:
            # Perturb to escape
            n_pert = max(20, n // 20)
            idx_p = np.random.default_rng(t).choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.1 * np.random.default_rng(t + 1).standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            y = y * norm / s_y
            stagnant_count = 0

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Aggressive 2D pair coordinate descent targeting top autocorrelation peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Focus on top peaks
        peak_threshold = cur_max * 0.90
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Candidate j's: structured + random
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                d = kl - i
                j1 = int(np.clip(d, 0, n - 1))
                j2 = int(np.clip(-d, 0, n - 1))
                j3 = int(np.clip(kl - i + n, 0, n - 1))
                j_candidates.add(j1)
                j_candidates.add(j2)
                j_candidates.add(j3)
                # Also try symmetric positions
                j4 = int((n - 1 - i + kl) % n)
                j_candidates.add(j4)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:15]:
                if j == i:
                    continue
                aj = a[j]

                # Compute the gain vector for transferring mass from i to j
                # conv_new[k] = conv_old[k] + delta * (contribution from j) - delta * (contribution from i)
                # Plus quadratic corrections at lags 2i, 2j, i+j
                g = np.zeros(size)
                # +a shifted to position j (forward)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                # +a reversed shifted to position j (backward)
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # -a shifted to position i (forward)
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                # -a reversed shifted to position i (backward)
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Sample deltas (transfer from i to j)
                n_samples = 30
                max_d = min(ai * 0.95, norm * 0.5)
                if max_d < 1e-12:
                    continue
                deltas = np.linspace(0.0, max_d, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g
                    # Quadratic correction at lag 2i (a[i]^2 term)
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    # Quadratic correction at lag 2j (a[j]^2 term)
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross term at lag i+j (a[i]*a[j] -> (ai-d)(aj+d))
                    if i + j < size:
                        new_conv[i + j] = conv[i + j] - ai * d + aj * d - d * d
                    # Cross term at lag |i-j|
                    li = abs(i - j)
                    if li < size:
                        new_conv[li] = conv[li] + ai * d - aj * d - d * d

                    v = new_conv.max()
                    if v < best_val - 1e-14:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                # Perturb to escape local minimum
                n_perturb = max(15, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                perturb_size = 0.08 + 0.04 * no_improve
                a[idx_perturb] *= (1.0 + perturb_size * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def compute_score(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    conv = fast_conv_self(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(3.0 * (x - 0.5)) ** 2 + 0.1
    elif choice == 9:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / (0.15 + 0.1 * rng.random()))
    elif choice == 10:
        # Power-law shape
        a = (x + 0.1) ** (-0.3) * ((1 - x) + 0.1) ** (-0.3)
    else:
        # Multi-peak
        a = sum(np.exp(-8.0 * (x - c) ** 2) for c in [0.2, 0.5, 0.8])
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def momentum_subgradient(a0, time_limit, verbose=True):
    """Momentum subgradient with adaptive temperature for the max-convolution objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()

    momentum = np.zeros(n)
    beta = 0.85
    lr_base = 1.0 / np.sqrt(n)

    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        temperature = max(0.05, max_c * 0.03 / (1.0 + 0.005 * it))
        shifted = (conv - max_c) / temperature
        shifted = np.maximum(shifted, -50.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        lr = lr_base / (1.0 + 0.0005 * it)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    global _lp_warm_start
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = eval_fn(p_norm)
            best_seq = p_norm
            best_score = sc
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 90.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse) - first round
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Pair CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, "pcd1")

    # Phase 4: LP Nesterov round 2 (refined)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 4: LP Nesterov R2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 5: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (budget={remaining:.0f}s) ===")
    if remaining > 120:
        alt_lengths = [n_base - 100, n_base + 100, n_base + 200, n_base - 200]
        for alt_n in alt_lengths:
            if alt_n < 200:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_alt = min(remaining * 0.3, 120.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            # Quick LSE + LP on alternative length
            t_lse = t_alt * 0.25
            t_lp = t_alt * 0.75
            a_out, _ = lse_gradient_descent(init_alt, t_lse, verbose=False)
            remaining_alt = time_budget - (time.monotonic() - t_global)
            if remaining_alt > 30:
                a_out, _ = lp_nesterov_optimize(a_out, min(remaining_alt * 0.5, t_lp), solver, verbose=False)
            try_update(a_out, f"alt_n{alt_n}")

    # Phase 6: Final aggressive coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final CD polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        a_out, _ = pair_cd_optimize(best_seq, remaining * 0.7, rng, verbose=True)
        try_update(a_out, "pcd_final")

    # Phase 7: Final LSE + LP micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final micro-polish (budget={remaining:.0f}s) ===")
        t7a = min(remaining * 0.4, 40.0)
        a_out, _ = lse_gradient_descent(best_seq, t7a, verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = lp_nesterov_optimize(best_seq, remaining * 0.6, solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
