    # Simplest approach: just compute it directly using the structure.
    # grad[i] = 2 * sum_{k=0}^{2n-2} weights[k] * a[k-i] (where 0<=k-i<n, i.e., i<=k<i+n)
    # = 2 * sum_{k=i}^{min(i+n-1, 2n-2)} weights[k] * a[k-i]
    
    # This is a sliding dot product. Can be computed with FFT.
    # Let me define: for each i, grad[i]/2 = dot(a[0:n], weights[i:i+n])
    # This is the cross-correlation of a and weights at lag i.
    
    # Using FFT: corr[i] = ifft(fft(a, N) * conj(fft(weights, N)))[i] where N >= (n-1)+(2n-2) = 3n-3
    
    # Hmm, that's a large FFT. Let me think of another way.
    
    # Alternative: use the fact that grad[i] = 2 * (a reversed) cross-correlated with weights...
    
    # Actually, the simplest efficient way:
    # Define b = a (length n), c = weights (length 2n-1)
    # I want: grad[i] = 2 * sum_{j=0}^{n-1} b[j] * c[j+i] for i = 0, ..., 2n-2-n+... 
    
    # This is the "unnormalized cross-correlation" evaluated at positive lags.
    # Using the formula: corr[i] = ifft(fft(b, N) * conj(fft(c, N)))[i]
    # where N >= max(n, len(c)) + ... 
    # Actually for cross-correlation of b (length n) and c (length m=2n-1):
    # We need N >= n + m - 1 = n + 2n - 1 - 1 = 3n - 2... no.
    # The cross-correlation has length n + m - 1.
    # To compute it via FFT, we need N >= n + m - 1.
    # Here m = 2n-1, so N >= n + 2n - 1 - 1 = 3n - 2. For n=1600, N >= 4798, so FFT size 8192.
    
    # That's fine. Let me implement it.
    
    N = 1 << (n + (2*n - 1) - 1 - 1).bit_length()  # should be >= 3n-2
    # Actually, let me just use a safe size
    N = 1 << (3 * n).bit_length()
    
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(weights, N)
    corr = np.fft.irfft(A * np.conj(W), N)[:n]
    
    grad = 2.0 * corr
    return grad, conv

def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # We want to find the closest point to v on the simplex {x >= 0, sum = S}
    # This is the "water-filling" algorithm
    if S <= 0:
        return np.zeros(n)
    
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0)

def mirror_descent_smoothed(a0, time_limit, tau_start, tau_end, tau_decay, verbose=True, seed=0):
    """Mirror descent with log-sum-exp smoothed objective."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    eta = 1.0 / (tau_start * norm * norm)  # initial step size
    stall = 0
    stall_limit = 100
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        tau = tau_start * (tau_end / tau_start) ** (t / max(1, int(1.0 / (1.0 - tau_end/tau_start + 1e-10))))
        # Simpler: exponential decay
        tau = tau_start * tau_decay ** t
        tau = max(tau, tau_end)
        
        # Compute smoothed gradient
        grad, conv = get_smoothed_grad(a, tau)
        
        # Mirror descent update (multiplicative):
        # a_new[i] = a[i] * exp(-eta * grad[i]) / sum_j a[j] * exp(-eta * grad[j])
        # For numerical stability, use the "shifted" version
        exponents = -eta * grad
        exponents -= exponents.max()  # shift for stability
        unnormalized = a * np.exp(exponents)
        s_new = unnormalized.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = unnormalized * norm / s_new
        
        # Evaluate
        score_new = fast_autocorr(a_new).max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # If stalled, try a larger step or restart
        if stall > stall_limit:
            eta *= 2.0
            if eta > 10.0:
                # Restart from best
                a = best_a.copy()
                eta = 1.0 / (tau * norm * norm)
                stall = 0
        
        a = a_new
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} tau={tau:.2f} eta={eta:.4f} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def subgradient_nesterov(a0, time_limit, verbose=True, seed=0):
    """Nesterov-accelerated subgradient method for max-convolution minimization."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * norm / s
    
    y = a.copy()  # Nesterov extrapolation point
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    stall = 0
    stall_limit = 150
    
    # Compute Lipschitz constant estimate
    # The gradient of conv[k] has norm 2*||a||_inf (roughly)
    # So L ≈ 2 * max(a) * ... let's just use adaptive step size
    
    eta = 0.1 / max(a.max(), 1.0)
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        conv_y = fast_autocorr(y)
        k_star = np.argmax(conv_y)
        
        # Subgradient: grad[i] = 2*a_y[k_star - i] if 0 <= k_star - i < n
        # But we're using y, so:
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        grad = np.zeros(n)
        indices = np.arange(lo, hi)
        grad[indices] = 2.0 * y[k_star - indices]
        # Fix the center if k_star is even
        if k_star % 2 == 0 and 0 <= k_star // 2 < n:
            grad[k_star // 2] = y[k_star // 2]  # not 2*y, just y
        
        # Actually wait, I showed above that it's always 2*a[k-i]. Let me recheck.
        # conv[k] = sum_j a[j]*a[k-j]
        # If j=i: term is a[i]*a[k-i], derivative w.r.t. a[i] is a[k-i]
        # If k-j=i (j=k-i): term is a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i]
        # Total: 2*a[k-i] if 0<=k-i<n
        # If k=2i: the term a[i]*a[i] has derivative 2*a[i], and both "j=i" and "k-j=i" give the same term
        #   but we should only count it once in the sum. So the derivative of a[i]^2 is 2*a[i].
        #   In our formula: "j=i" gives a[k-i]=a[i], "k-j=i" gives a[k-i]=a[i], total=2*a[i]. Correct!
        # So the formula grad[i] = 2*a[k-i] (if 0<=k-i<n) is always correct.
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        # Step: y_new = y - eta_t * grad
        # Then: a_new = y_new + (t-1)/(t+2) * (y_new - y)
        
        eta_t = eta / (1.0 + 0.01 * t)  # decay step size
        
        y_step = y - eta_t * grad
        y_step = np.maximum(y_step, 0.0)
        s_y = y_step.sum()
        if s_y < 1e-15:
            y_step = y.copy()
            s_y = y.sum()
        y_step = y_step * norm / s_y
        
        if t == 1:
            a_new = y_step
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_step + momentum * (y_step - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_step.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y
        y = y_step
        a = a_new
        
        # Track best
        sc_y = fast_autocorr(y).max()
        sc_a = fast_autocorr(a).max()
        if min(sc_y, sc_a) < best_score:
            best_score = min(sc_y, sc_a)
            if sc_y < sc_a:
                best_a = y.copy()
            else:
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_limit:
            y = best_a.copy()
            a = best_a.copy()
            eta = 0.1 / max(best_a.max(), 1.0)
            stall = 0
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG-Nest t={t} eta={eta_t:.6f} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def peak_shaving(a0, time_limit, top_k=5, verbose=True, seed=0):
    """Directly try to reduce the top peaks of the convolution."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_autocorr(a)
        # Find top k peaks
        top_indices = np.argsort(conv)[::-1][:top_k]
        
        # Average subgradient at top peaks
        grad = np.zeros(n)
        for k in top_indices:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            indices = np.arange(lo, hi)
            grad[indices] += 2.0 * a[k - indices]
        grad /= top_k
        
        # Take a small step in the negative gradient direction
        # with random perturbation to escape flat spots
        step_size = 0.05 * norm / max(grad.max(), 1e-10)
        a_new = a - step_size * grad
        a_new += 0.01 * rng.standard_normal(n) * norm / n
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new
        
        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = a - 0.01 * step_size * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try * norm / s_try
                sc_try = fast_autocorr(a_try).max()
                if sc_try < best_score - 1e-12:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    # No improvement, skip
                    pass
    
    return best_a, best_score

def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        a = np.ones(n) / n  # uniform
    elif style == 8:
        a = np.sin(np.pi * x) ** 0.5
    elif style == 9:
        a = 1.0 / (1.0 + 5.0 * (x - 0.5) ** 2)
    else:
        a = np.sin(np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
