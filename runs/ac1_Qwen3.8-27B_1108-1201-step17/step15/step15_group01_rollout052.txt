def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, w):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T


def lp_best_response(a, M, solver, active_only=None):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    if active_only is not None:
        T_use = T[active_only, :]
    else:
        T_use = T
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def smooth_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.array([30.0, 15.0, 8.0, 4.0, 2.0, 1.2, 0.7, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.025, 0.015, 0.01, 0.006, 0.004, 0.002, 0.001])

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(20, min(400, int(remaining * 4)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()
            grad = compute_smooth_grad(y, w)
            step = norm / (max(n * T, 1.0) * (1.0 + 0.02 * it))
            step = min(step, norm / (n * 0.3))
            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)
            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)
            y = y_new
            a = a_curr
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.3f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        conv_y = fast_autocorr(y)
        # Adaptive constraint generation: start with top 30%, expand if stuck
        frac = 0.3 + 0.1 * min(n_iters / 100.0, 1.0)
        threshold = (1.0 - frac) * M_curr
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 5:
            active = np.where(conv_y >= 0.5 * M_curr)[0]
        if len(active) < 3:
            active = np.arange(len(conv_y))

        b = lp_best_response(y, M_curr, solver, active_only=active)
        if b is None:
            b = lp_best_response(y, M_curr, solver, active_only=None)
            if b is None:
                stuck_count += 1
                if stuck_count > 20:
                    break
                continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov with tuned momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()

                a_new = a_new * norm / s_a

        a = a_new

        score_y = fast_autocorr(y).max()

            best_score = score_y
            best_a = y.copy()

def peak_cd(a0, time_limit, rng, n_target_lags=5, verbose=True):
    """Peak-targeted coordinate descent: specifically reduce the highest lags."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, fast_autocorr(a).max()
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Find the top lags to target
        top_lags = np.argsort(conv)[::-1][:n_target_lags]
        top_lag_set = set(top_lags.tolist())

        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # For each target lag, find which partner index contributes
            # conv[k] includes a[i]*a[k-i] (when valid)
            # So for lag k, partner is j = k - i (must be in [0, n-1])
            # Also conv[k] includes a[i]*a[k-i] where k-i can be negative -> use reversed
            # Actually for full conv: conv[k] = sum_i a[i]*a[k-i] for 0<=i<n, 0<=k-i<n
            # So for element i, it contributes to conv[i+j] with partner a[j] for j in [0,n-1]
            # and to conv[i-j+n-1]... no wait.
            # np.convolve(a,a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
            # Element a[i] contributes a[i]*a[k-i] to conv[k] where k-i in [0,n-1], i.e., k in [i, i+n-1]

            # Try moving mass from i to j (cyclic)
            for offset in range(1, min(n // 4, 60) + 1):
                j = (i + offset) % n
                if j == i:
                    continue
                # Test moving a fraction of a[i] to j
                for frac in [0.5, 0.3, 0.15, 0.05]:
                    d = a[i] * frac
                    if d < norm * 1e-10:
                        break
                    a_test = a.copy()
                    a_test[i] -= d
                    a_test[j] += d
                    if a_test[i] < -1e-15:
                        continue
                    a_test = np.maximum(a_test, 0.0)
                    s_test = a_test.sum()
                    if s_test < 1e-15:
                        continue
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    new_max = conv_test.max()
                    if new_max < best_score - 1e-12:
                        a = a_test
                        conv = conv_test
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                        break
                    elif new_max < conv.max() - 1e-12:
                        a = a_test
                        conv = conv_test
                        pass_improved = True
                        no_improve = 0
                        break

        if not pass_improve:
            no_improve += 1
            if no_improve >= 3:
                # Try larger jumps
                if n_passes < 50:
                    for _ in range(10):
                        i = rng.integers(0, n)
                        j = rng.integers(0, n)
                        if i == j or a[i] < norm * 1e-8:
                            continue
                        d = a[i] * 0.4
                        a_test = a.copy()
                        a_test[i] -= d
                        a_test[j] += d
                        a_test = np.maximum(a_test, 0.0)
                        a_test = a_test * norm / a_test.sum()
                        conv_test = fast_autocorr(a_test)
                        if conv_test.max() < best_score - 1e-12:
                            a = a_test
                            conv = conv_test
                            best_score = conv_test.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            break
                    if not pass_improved:
                        break
                else:
                    break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coarse_to_fine(a0, time_limit, solver, group_size=4, verbose=True):
    """Coarse-to-fine: optimize grouped version, then unfold and refine."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    n_groups = max(n // group_size, 10)
    actual_group = n // n_groups
    if actual_group < 2:
        return nesterov_lp(a0, time_limit, solver, verbose)

    # Create coarse representation
    coarse = np.zeros(n_groups)
    for g in range(n_groups):
        start = g * actual_group
        end = start + actual_group
        coarse[g] = a[start:end].mean()

    # Optimize coarse
    t_coarse = time_limit * 0.4
    coarse_opt, _ = nesterov_lp(coarse, t_coarse, solver, verbose=False)

    # Unfold
    fine = np.zeros(n)
    for g in range(n_groups):
        start = g * actual_group
        end = start + actual_group
        fine[start:end] = coarse_opt[g]

    # Refine fine
    remaining = time_limit - (time.monotonic() - (time.monotonic() - time_limit))
    t_fine = time_limit - t_coarse - 1
    t0 = time.monotonic()
    fine_opt, _ = nesterov_lp(fine, min(t_fine, time_limit - (time.monotonic() - t0)), solver, verbose=False)
    fine_opt = np.maximum(fine_opt, 0.0)
    fine_opt = fine_opt * norm / fine_opt.sum()

    if verbose:
        print(f"    Coarse-to-fine: coarse_best={fast_autocorr(coarse_opt).max():.6f}, fine_best={fast_autocorr(fine_opt).max():.6f}")

    return fine_opt, fast_autocorr(fine_opt).max()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Hann window
    i = np.arange(n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * (i + 0.5) / n))
    inits.append(hann / hann.sum())
    # Raised cosine (different phase)
    rc = 0.5 * (1 + np.cos(np.pi * i / n))
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())
    # Linear ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    # V-shape
    vshape = np.abs(np.linspace(-1, 1, n)) + 0.1
    inits.append(vshape / vshape.sum())
    # Two bumps
    bump1 = np.exp(-0.5 * ((i - n // 4) / (n / 8)) ** 2)
    bump2 = np.exp(-0.5 * ((i - 3 * n // 4) / (n / 8)) ** 2)
    two_bump = bump1 + bump2 + 0.01
    inits.append(two_bump / two_bump.sum())
    # Random smooth
    r = rng.random(n)
    # Smooth it
    kernel = np.ones(5) / 5.0
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0.01)
    inits.append(r_smooth / r_smooth.sum())
    return inits


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1620.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 0: Try diverse initializations quickly ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 0: Diverse initializations (budget={min(remaining*0.08, 90):.0f}s) ===")
        t_init = min(remaining * 0.08, 90.0)
        inits = generate_diverse_inits(n_base, rng)
        # Quick smooth optimization on each (short time)
        t_per_init = t_init / len(inits)
        for idx, init in enumerate(inits):
            if time.monotonic() - t_global > time_budget:
                break
            a_norm = init * np.sqrt(2.0 * n_base) / init.sum()
            a_out, _ = smooth_optimize(a_norm, t_per_init, verbose=False)
            do_try_update(a_out, f"init_{idx}")

    # === Phase 1: LP Nesterov on best so far (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LP Nesterov n={n_c} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_c}")

    # === Phase 2: Peak-targeted CD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = remaining * 0.15
    print(f"\n=== Phase 2: Peak CD n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = peak_cd(a_norm, t2, rng, n_target_lags=8, verbose=True)
        do_try_update(a_out, f"peakcd_n{n_c}")

    # === Phase 3: Smooth gradient on best (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Smooth grad n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t3, verbose=True)
        do_try_update(a_out, f"smooth_n{n_c}")

    # === Phase 4: LP Nesterov round 2 (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: LP Nesterov r2 n={n_c} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_c}")

    # === Phase 5: Peak CD round 2 (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Peak CD r2 n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = peak_cd(a_norm, t5, rng, n_target_lags=5, verbose=True)
        do_try_update(a_out, f"peakcd2_n{n_c}")

    # === Phase 6: Coarse-to-fine (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: Coarse-to-fine n={n_c} (budget={t6:.0f}s) ===")
    if t6 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = coarse_to_fine(a_norm, t6, solver, group_size=4, verbose=True)
        do_try_update(a_out, f"ctf_n{n_c}")

    # === Phase 7: Final polish with LP + PeakCD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_c = len(best_seq)
        print(f"\n=== Phase 7: Final polish n={n_c} (budget={remaining:.0f}s) ===")
        # Split between LP and peak CD
        t_lp = remaining * 0.5
        t_cd = remaining * 0.4
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        if t_lp > 15:
            a_out, _ = nesterov_lp(a_norm, t_lp, solver, verbose=False)
            do_try_update(a_out, "final_lp")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            n_c = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
            a_out, _ = peak_cd(a_norm, min(t_cd, remaining), rng, n_target_lags=10, verbose=False)
            do_try_update(a_out, "final_cd")

    # === Try alternate lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        n_base_final = len(best_seq)
        for n_alt in [n_base_final + 200, n_base_final - 200, n_base_final + 400]:
            if n_alt < 200 or n_alt > 5000:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 30:
                break
            t_alt = min(remaining_sub * 0.3, 120.0)
            print(f"\n=== Alt length n={n_alt} (budget={t_alt:.0f}s) ===")
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            a_out_alt, _ = nesterov_lp(a_norm_alt, t_alt, solver, verbose=False)
            do_try_update(a_out_alt, f"alt_n{n_alt}")
            # Quick peak CD on the result
            remaining_sub2 = time_budget - (time.monotonic() - t_global)
            if remaining_sub2 > 15 and best_seq is not None:
                n_r = len(best_seq)
                if n_r == n_alt:
                    a_norm_r = best_seq * np.sqrt(2.0 * n_r) / best_seq.sum()
                    a_out_r, _ = peak_cd(a_norm_r, min(remaining_sub2 * 0.3, 40.0), rng, verbose=False)
                    do_try_update(a_out_r, f"alt_cd_n{n_alt}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
