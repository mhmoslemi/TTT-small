    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

- For each of the top few autocorrelation peaks, identify the pair (i, j) that contributes most to that peak.
- Try moving mass from i to a position that reduces the peak.
- Use the exact derivative to determine the optimal amount to move.

With R=5-10 bumps, I have 3R = 15-30 parameters to optimize, which is much more tractable. I can use scipy.optimize or manual gradient descent.

One more thought: the previous program's AMD approach uses a "soft subgradient" that weights the top-k peaks by their value^beta. This is similar to my log-sum-exp approach but with a different weighting. The log-sum-exp is smoother and gives a true gradient, which should be better for Adam.

Let me also consider using the "entropic regularization" approach:

This encourages the autocorrelation to be spread out (high entropy), which should reduce the maximum. But it's a different objective and might not directly minimize the max.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam-based smooth gradient descent (log-sum-exp relaxation with temperature annealing)
2. LP-based Frank-Wolfe for polishing (when n is small enough)
3. Coordinate descent with exact derivative for final polish
4. Multi-length interpolation and multi-initialization
"""

import numpy as np
import time

# --- Core computation functions ---

def fast_autocorr(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient(ax, ac, T):
    """
    Compute gradient of (2n/T)*log(sum_k exp(T*b_k)) w.r.t. a.
    Returns the gradient vector.
    """
    n = len(ax)
    # Softmax weights over autocorrelation lags
    ac_shifted = ac - ac.max()
    exp_ac = np.exp(T * ac_shifted)
    p = exp_ac / exp_ac.sum()
    
    # Cross-correlation c[m] = sum_k p[k] * a[k-m]
    L = 1 << (3 * n).bit_length()
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(ax, L)
    c = np.fft.irfft(P * np.conj(A), L)[:n]
    
    # grad = (2n/T) * 2 * c = (4n/T) * c
    return (4.0 * n / T) * c

def project_simplex(v):
    """Euclidean projection onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
   .cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(.cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

# --- Adam smooth descent ---

def adam_smooth_descent(a0, time_limit, T_start=10.0, T_end=500.0, 
                        lr_init=0.005, lr_decay=0.9999, 
                        verbose=True, rng_seed=42):
    """
    Adam-based smooth gradient descent with temperature annealing.
    Minimizes (2n/T)*log(sum exp(T*b_k)) as T increases.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.85, 0.999
    eps = 1e-10
    step = 0
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        lr = lr_init * (lr_decay ** step)
        
        # Temperature schedule: exponential ramp
        frac = min(step / 50000.0, 1.0)
        T = T_start * (T_end / T_start) ** frac
        
        ac = fast_autocorr(a)
        grad = compute_gradient(a, ac, T)
        
        # GPU/memory check: clip gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e6:
            grad = grad / grad_norm * 1e6
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)
        
        # Check improvement every 20 steps
        if step % 20 == 0:
            M_new = fast_autocorr(a).max()
            if M_new < best_m - 1e-12:
                best_m = M_new
                best_a = a.copy()
                if verbose and step % 2000 == 0:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    Adam step={step} ({elapsed:.1f}s): J={J:.6f}, T={T:.1f}, lr={lr:.2e}")
        
        # Periodic perturbation to escape local minima
        if step % 10000 == 0:
            sigma = 0.02 * (1.0 / (1.0 + step / 50000.0))
            a_pert = best_a + sigma * rng.standard_normal(n)
            a_pert = project_simplex(a_pert)
            M_pert = fast_autocorr(a_pert).max()
            if M_pert < best_m:
                best_m = M_pert
                best_a = a_pert.copy()
            a = a_pert
            m = np.zeros(n)
            v = np.zeros(n)
    
    # Final check
    M_final = fast_autocorr(a).max()
    if M_final < best_m:
        best_m = M_final
        best_a = a.copy()
    
    return best_a, best_m

# --- LP Frank-Wolfe (for smaller n) ---

def lp_frank_wolfe(a0, time_limit, verbose=True, max_iters=200):
    """LP-based Frank-Wolfe for polishing."""
    try:
        import cvxpy as cp
    except ImportError:
        return a0, fast_autocorr(a0).max()
    
    n = len(a0)
    if n > 3000:
        return a0, fast_autocorr(a0).max()  # Too large for LP
    
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix A: A[k,j] = y[k-j] if 0<=k-j<n else 0
        # Constraint: A @ b <= M_curr for all k
        b_var = cp.Variable(n, nonneg=True)
        
        # Build A efficiently
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j+n, j] = y
        
        constraints = [A @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = project_simplex(y_new)
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = project_simplex(x_new)
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

# --- Coordinate descent ---

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Coordinate descent with exact derivative evaluation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Find top peaks
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            # Find the pair (i, j=k-i) with maximum contribution to b_k
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Try to move mass from src to a position that reduces the peak
            # Compute sensitivity: for each dst, how much would b_k change?
            # Moving delta from src to dst changes b_k by:
            # -delta * a[k-src] - a[src] * 0 + delta * a[k-dst] - 0 (approximately)
            # More precisely, the change in b_k from moving delta: 
            #   delta*(a[k-dst] - a[k-src]) for the k-th lag
            #   but also affects other lags
            
            # For simplicity, try a few random destinations
            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                
                # Try different fractions
                for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m
