def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def lse_smoothed_max_obj(z, T, n):
    """Compute LSE-smoothed max of conv(a,a) where a = z/sum(z)."""
    S = z.sum()
    if S < 1e-15:
        return 1e10
    a = z / S
    conv = fast_conv_self(a)
    max_c = conv.max()
    shifted = (conv - max_c) * T
    shifted = np.maximum(shifted, -100.0)
    log_sum = np.log(np.exp(shifted).sum()) if shifted.max() < 700 else shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())
    return (max_c + log_sum / T) / (T * 0 + 1.0)  # = max_c + log_sum_exp / T


def lse_smoothed_max_grad(z, T, n):
    """Compute gradient of LSE-smoothed max conv w.r.t. z where a = z/sum(z)."""
    S = z.sum()
    if S < 1e-15:
        return np.zeros(n)
    a = z / S
    conv = fast_conv_self(a)
    max_c = conv.max()
    shifted = (conv - max_c) * T
    shifted = np.maximum(shifted, -100.0)
    exp_c = np.exp(shifted - shifted.max())
    p = exp_c / exp_c.sum()
    
    # p has length 2n-1
    # Gradient w.r.t. a: grad_a[i] = 2 * sum_j a[j] * p[j+i]
    # This is a cross-correlation, computed via FFT
    size = 2 * n - 1
    # r[i] = sum_{j=0}^{n-1} a[j] * p[j+i]
    # Using reversal trick: r[i] = (a * rev(p))[2n-2-i]
    p_rev = p[::-1]
    conv_len = n + size - 1  # n + (2n-1) - 1 = 3n-2
    fft_size2 = 1 << (conv_len - 1).bit_length()
    A2 = rfft(a, fft_size2)
    PR = rfft(p_rev, fft_size2)
    cross_full = irfft(A2 * PR, fft_size2)[:conv_len]
    
    indices = 2 * n - 2 - np.arange(n)
    r = cross_full[indices]
    
    grad_a = 2.0 * r
    
    # Convert to gradient w.r.t. z: grad_z[i] = (S * grad_a[i] - z dot grad_a) / S^2
    z_dot_grad = np.dot(z, grad_a)
    grad_z = (S * grad_a - z * z_dot_grad / S * S) / (S * S)
    # Simplify: grad_z[i] = (grad_a[i] - a[i] * mean_weight) where...
    # Actually: grad_z = (S * grad_a - z * (z·grad_a/S)) / S^2 = (grad_a - a * (z·grad_a)/S) / S
    # = (grad_a - a * (a·grad_a)) / 1  (since z = a*S, so z·grad_a = S*(a·grad_a), and (z·grad_a)/S = a·grad_a)
    # Wait: z = a * S (element-wise), so z·grad_a = S * (a·grad_a)
    # grad_z[i] = (S * grad_a[i] - S * a[i] * (a·grad_a)) / S^2 = (grad_a[i] - a[i]*(a·grad_a)) / S
    
    a_dot_grad = np.dot(a, grad_a)
    grad_z = (grad_a - a * a_dot_grad) / S
    
    return grad_z

def build_T_full(a):
    """Build the full constraint matrix for the LP: T[i,j] such that (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    size = 2 * n - 1
    # conv(a,b)[k] = sum_j a[j] * b[k-j] (valid indices)
    # = sum_j b[j] * a[k-j] (valid indices)  [by reindexing]
    # So T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for j in range(n):
        # T[k, j] = a[k-j] for valid k-j in [0, n-1]
        # k-j in [0, n-1] means k in [j, j+n-1]
        T[j:j+n, j] = a
    return T


def build_T_active(a, active_lags):
    """Build constraint matrix for only active lags."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(active_lags), n))
    for ii, k in enumerate(active_lags):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                T[ii, j] = a[idx]
    return T


def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active is not None and 0 < len(active) < size:
        T = build_T_active(a, active)
        constraints = [T @ b_var <= M]
    else:
        T = build_T_full(a)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # === Phase 1: L-BFGS with annealed temperature ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 1: L-BFGS n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lbfgs_phase(best_seq, t1, verbose=True)
        try_update(a_out, "lbfgs1")
    
    # === Phase 2: LP best-response (adaptive step) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: LP adaptive n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_best_response(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp_adapt1")
    
    # === Phase 3: Peak coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.18, 300.0)
    print(f"\n=== Phase 3: Peak CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_cd_optimize(best_seq, t3, verbose=True)
        try_update(a_out, "peakcd1")
    
    # === Phase 4: Nesterov LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 4: Nesterov LP n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, "nesterov1")
    
    # === Phase 5: L-BFGS round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 5: L-BFGS round 2 (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = lbfgs_phase(best_seq, t5, verbose=False)
        try_update(a_out, "lbfgs2")
    
    # === Phase 6: Peak CD round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 6: Peak CD round 2 (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = peak_cd_optimize(best_seq, t6, verbose=False)
        try_update(a_out, "peakcd2")
    
    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 7: Multi-length (budget={remaining:.0f}s) ===")
        for n_try in [n_base + 200, n_base + 400, n_base - 200]:
            if n_try < 100 or time_budget - (time.monotonic() - t_global) < 40:
                break
            init_try = interpolate_seq(best_seq, n_try)
            t_sub = min(60.0, (time_budget - (time.monotonic() - t_global)) * 0.4)
            if t_sub < 20:
                break
            a_try, _ = nesterov_lp_optimize(init_try, t_sub, solver, verbose=False)
            try_update(a_try, f"nl_{n_try}")
    
    # === Final evaluation ===
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
