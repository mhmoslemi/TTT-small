def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        threshold_1 = M * 0.85
        active = np.where(conv_y >= threshold_1)[0]
        if len(active) < 20:
            threshold_2 = M * 0.55
            active = np.where(conv_y >= threshold_2)[0]

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum

        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.005)[0]
            if len(violated) > 5:
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    g = b / b.sum()

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / s_a

        y = y_new
        a = a_new

        sc_y = 2.0 * n * fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = 2.0 * n * fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with greedy transfer."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                for offset in [-1, 0, 1]:
                    j_struct = int((kl - i + offset) % n)
                    j_candidates.add(j_struct)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:15]:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 20
                deltas = np.linspace(0.0, ai * 0.95, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-14:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                new_max = fast_conv_self(a).max()
                if new_max * 2 * n < best_score - 1e-10:
                    best_score = new_max * 2 * n
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(10, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 500 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_initializations(n, rng, count=5):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    inits.append(np.ones(n) / n)

    a = np.cos(np.pi * x / 2.0) ** 2 + 0.01 * rng.random(n)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())

    a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2 + 0.01 * rng.random(n)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())

    a = np.exp(-8.0 * (x - 0.5) ** 2) + 0.3 + 0.05 * rng.random(n)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())

    a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    for _ in range(max(0, count - 5)):
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x + phase) + 0.05 * rng.random(n)
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            inits.append(a / a.sum())

    return inits


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_score(a):
    """Compute the evaluate_sequence score without calling the actual function."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0 or a.sum() < 0.01:
        return float('inf')
    b = fast_conv_self(a)
    max_b = b.max()
    sum_a = a.sum()
    return float(2 * n * max_b / (sum_a ** 2))


def momentum_smooth_descent(a0, time_limit, verbose=True):
    """Momentum-based smooth descent with adaptive LR."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    momentum = np.zeros(n)
    beta = 0.85
    lr = 5.0 / n

    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        temp = max(0.5 / n, max_c * 0.02 / (1.0 + 0.005 * it))
        shifted = (conv - max_c) / temp
        shifted = np.maximum(shifted, -50.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)
        g1 = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
        g2 = np.fft.irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        lr = 5.0 / (n * (1.0 + 0.002 * it))
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        sc = 2.0 * n * fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MomSm it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_score(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = eval_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Coord Descent n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = coordinate_descent(best_seq, t3, rng, verbose=True)
        try_update(a_out, "cd1")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov r2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 5: Momentum smooth descent
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 5: MomSmooth n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = momentum_smooth_descent(best_seq, t5, verbose=True)
        try_update(a_out, "mom")

    # Phase 6: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 600
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: Longer n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 80:
        init_long = interpolate_seq(best_seq, n_long)
        t6a = min(t6 * 0.6, 100.0)
        a_out, _ = nesterov_lp_optimize(init_long, t6a, solver, verbose=False)
        try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.4, 80.0)
            a_out2, _ = coordinate_descent(best_seq, t6b, rng, verbose=False)
            try_update(a_out2, "cd_long")

    # Phase 7: Diversified restart
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Diversified restarts (budget={remaining:.0f}s) ===")
    if remaining > 60:
        inits = diverse_initializations(n_base, rng, count=4)
        t_per = min(remaining / max(1, len(inits)), 120.0)
        for i, init_a in enumerate(inits):
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_rem = min(t_per, time_budget - (time.monotonic() - t_global))
            if t_rem < 20:
                break
            print(f"    Restart {i+1}/{len(inits)}: initial score={eval_score(init_a):.6f}")
            a_out, _ = lse_gradient_descent(init_a, t_rem * 0.5, verbose=False)
            try_update(a_out, f"restart_{i}_lse")

            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 30:
                a_out2, _ = nesterov_lp_optimize(best_seq, min(remaining2 * 0.3, 60.0), solver, verbose=False)
                try_update(a_out2, f"restart_{i}_lp")

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t_half = min(remaining * 0.5, 60.0)
        a_out, _ = coordinate_descent(best_seq, t_half, rng, verbose=False)
        try_update(a_out, "cd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = momentum_smooth_descent(best_seq, min(remaining * 0.6, 40.0), verbose=False)
            try_update(a_out, "mom_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_score(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
