Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP best-response with Nesterov
acceleration, direct peak tracking, and coordinate descent polishing.

import numpy as np

import time
from scipy.fft import rfft, irfft

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])

def build_constraint_matrix(a, active_lags):
    """Build the constraint matrix T such that T @ b <= M for the active lags."""
    n = len(a)
    m = len(active_lags)
    T = np.zeros((m, n))
    for idx, k in enumerate(active_lags):
        # conv(a,b)[k] = sum_i a[i] * b[k-i]
        # So the row for lag k has T[idx, j] = a[k-j] where 0 <= k-j < n
        # i.e., j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            T[idx, j] = a[k - j]
    return T

def solve_lp(a, M, active_lags, solver):
    """Solve the LP best response."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_constraint_matrix(a, active_lags)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

    t0 = time.monotonic()

    size = 2 * n - 1

        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)

        if M < 1e-12:

            best_a = y.copy()

        # Simpler approach: gradient of max w.r.t. a
        # d(max_b)/da[p] = sum over k where b[k] = M of (db[k]/da[p])
        # db[k]/da[p] = a[k-p] (if 0<=k-p<n) + a[p-k] (if 0<=p-k<n and p!=k-p)
        # Actually for conv(a,a)[k] = sum_i a[i]a[k-i]:
        # d/db[k]/da[p] = a[k-p] (if 0<=k-p<n) + a[p-k] (if 0<=p-k<n)
        # But if p = k-p (i.e., 2p=k), it's 2*a[p]... no wait.
        # conv(k) = sum_i a[i]*a[k-i]. d/dx_p = a[k-p] + a[p-k] (if both valid and k-p != p-k)
        # If k-p = p-k, i.e., k=2p, then d/dx_p = 2*a[p]
        
        # For the peak lag k_star, the gradient component at position p is:
        # grad_p = a[k_star - p] (if 0 <= k_star-p < n) 
        #         + a[p - k_star] (if 0 <= p-k_star < n)
        #         (with special handling if k_star-p == p-k_star, i.e., p = k_star/2)
        
        # But we also need to account for the constraint sum(a) = norm
        # So we subtract the mean gradient
        
        grad = np.zeros(n)
        for p in range(n):
            g = 0.0
            q1 = k_star - p
            if 0 <= q1 < n:
                g += a[q1]
            q2 = p - k_star
            if 0 <= q2 < n:
                g += a[q2]
            # Handle the case where q1 == q2 (only possible if k_star = 2*p)
            # In that case we've double-counted, but actually:
            # conv(k) = sum_i a[i]*a[k-i]
            # d/dx_p: terms where i=p give a[k-p], terms where k-i=p (i=k-p) give a[p]
            # If p = k-p (i.e., 2p=k), these are the same term, contributing a[p] once
            # But we wrote it as a[k-p] + a[p-k] = a[p] + a[-p]... 
            # Wait, if k_star = 2p, then q1 = 2p-p = p and q2 = p-2p = -p
            # q2 is negative so only q1 contributes: g = a[p]. Correct!
            # If k_star != 2p, then q1 and q2 are different (one positive, one negative or both valid)
            # Actually q1 = k_star - p and q2 = p - k_star = -q1
            # So either both are in [0,n) or neither (since one is the negative of the other)
            # Wait no: q1 + q2 = 0, so they can't both be >= 0 unless both are 0 (k_star=p)
            # If k_star = p: q1 = 0, q2 = 0, so g = a[0] + a[0] = 2*a[0]
            # But conv(p) has term a[p]*a[0] + a[0]*a[p] = 2*a[0]*a[p]... 
            # Wait no. conv(k) = sum_i a[i]*a[k-i]. For i=p: a[p]*a[k-p]. For i=k-p: a[k-p]*a[p].
            # These are the same term if p = k-p, i.e., 2p = k.
            # If 2p != k, they're two different terms in the sum, both equal to a[p]*a[k-p].
            # So d conv(k)/da[p] = a[k-p] (from term i=p) + a[k-p] (from term i=k-p, if k-p != p)
            # If k-p = p (i.e., 2p=k): d conv(k)/da[p] = a[p] (only one term)
            # If k-p != p and both p and k-p are in [0,n): d conv(k)/da[p] = 2*a[k-p]
            # If only p is in range (k-p not in [0,n)): d conv(k)/da[p] = a[k-p]... 
            # Wait, if k-p is not in [0,n), then the term i=k-p doesn't exist, so only the i=p term contributes.
            
            # Let me redo this more carefully:
            # conv(k) = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
            # d conv(k)/da[p]:
            #   = a[k-p] if p is in the valid range (i.e., max(0,k-n+1) <= p <= min(n-1,k))
            #     AND (k-p) != p (i.e., the term i=k-p is different from i=p)
            #   + a[p] if (k-p) is in the valid range AND (k-p) != p
            #   But wait, if k-p is in range and (k-p) != p, then when i = k-p, the term is a[k-p]*a[p],
            #     and d/dx_p of that is a[k-p]. So we add a[k-p] again.
            #   Total: 2*a[k-p] if p is valid and k-p is valid and k-p != p
            #          a[k-p] if p is valid and (k-p not valid or k-p == p)
            #          a[p] if k-p is valid and (p not valid or ...) -- this can't happen if p is valid
            #   Hmm, this is getting complicated. Let me just use the simple formula:
            #   d conv(k)/da[p] = a[k-p] * 1{0<=k-p<n} + a[p-k] * 1{0<=p-k<n}
            #   Wait no. Let me just do it directly.
            #   conv(k) = sum_i a[i] * a[k-i]
            #   d/dx_p [sum_i a[i]*a[k-i]] = sum_i [delta_{i,p}*a[k-i] + a[i]*delta_{k-i,p}]
            #                                = a[k-p] * 1{0<=k-p<n} + sum_i a[i]*delta_{k-i,p}
            #                                = a[k-p] * 1{0<=k-p<n} + a[k-p] * 1{0<=k-p<n}
            #                                = 2 * a[k-p] * 1{0<=k-p<n}
            #   Wait that's not right either. The second sum: delta_{k-i,p} means i = k-p,
            #   so it's a[k-p] (the value of a[i] when i=k-p), times 1{0 <= k-p < n}.
            #   But the first term is also a[k-p] * 1{0<=k-p<n}.
            #   So total = 2*a[k-p]*1{0<=k-p<n}... 
            #   But this double-counts when i=p and i=k-p are the same (i.e., 2p=k).
            #   In that case, the original sum has the term a[p]*a[k-p] = a[p]*a[p] = a[p]^2,
            #   and d/dx_p [a[p]^2] = 2*a[p].
            #   Using the formula: 2*a[k-p]*1{0<=k-p<n} = 2*a[p] (since k-p=p). Correct!
            #   And when 2p != k: we have two separate terms a[p]*a[k-p] and a[k-p]*a[p],
            #   each contributing a[k-p] to the derivative, total 2*a[k-p]. Correct!
            #   
            #   So the formula is simply: d conv(k)/da[p] = 2 * a[k-p] if 0 <= k-p < n, else 0.
            #   
            #   Wait, that's too simple. Let me verify with a small example.
            #   a = [x0, x1, x2], k = 1.
            #   conv(1) = a[0]*a[1] + a[1]*a[0] = 2*x0*x1
            #   d/dx0 = 2*x1. Formula: 2*a[1-0] = 2*a[1] = 2*x1. ✓
            #   d/dx1 = 2*x0. Formula: 2*a[1-1] = 2*a[0] = 2*x0. ✓
            #   d/dx2 = 0. Formula: 2*a[1-2] = 2*a[-1] = 0 (since -1 not in [0,3)). ✓
            #   
            #   k=2: conv(2) = a[0]*a[2] + a[1]*a[1] + a[2]*a[0] = 2*x0*x2 + x1^2
            #   d/dx0 = 2*x2. Formula: 2*a[2] = 2*x2. ✓
            #   d/dx1 = 2*x1. Formula: 2*a[1] = 2*x1. ✓
            #   d/dx2 = 2*x0. Formula: 2*a[0] = 2*x0. ✓
            #   
            #   So the formula is: d conv(k)/da[p] = 2*a[k-p] if 0 <= k-p < n, else 0.
            #   Interesting - it's always either 0 or 2*a[k-p].
            
            pass  # We'll compute gradient differently below
        
        # Use the simple formula: grad[p] = 2*a[k_star - p] if 0 <= k_star-p < n
        # But we want to reduce the peak, so we want to DECREASE a[p] where grad is high
        # and INCREASE a[p] where grad is low, maintaining sum(a) = norm.
        
        # Gradient of the peak (at k_star):
        grad = np.zeros(n)
        q = k_star - np.arange(n)  # k_star - p for p = 0, ..., n-1
        valid = (q >= 0) & (q < n)
        grad[valid] = 2.0 * a[q[valid]]
        
        # Project gradient: subtract mean to maintain sum constraint
        grad -= grad.mean()
        
        # Also consider other lags that are close to the peak (secondary peaks)
        # Weight by how close they are to the max
        second_lags = np.where(conv >= M * 0.9)[0]
        for k2 in second_lags:
            if k2 == k_star:
                continue
            q2 = k2 - np.arange(n)
            valid2 = (q2 >= 0) & (q2 < n)
            grad[valid2] += 2.0 * a[q2[valid2]] * 0.1  # small weight for secondary peaks
        
        # Update: a -= lr * grad, then project to non-negative and normalize
        lr = 0.01 / np.sqrt(n)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new = a - lr * 0.1 * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDescent it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent for final polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find j that best improves when we move mass from i to j
            peak_lags = np.where(conv >= cur_max * 0.9)[0]
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            # Try several candidate j values
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j_candidates.add(int((kl - i + n) % n))
            j_candidates.discard(i)
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute change in convolution when moving delta from i to j
                # New conv = old conv + delta*(e_j - e_i) convolved with a + delta*(e_j - e_i) convolved with (new a)
                # Approximation: new conv ≈ old conv + delta * g where g = conv(e_j - e_i, a) + conv(a, e_j - e_i)
                # But we need to be more careful at the diagonal terms
                
                g = np.zeros(size)
                # conv(a, e_j) has a[k-j] at position k (shifted a)
                # conv(e_j, a) has a[k-j] at position k (same thing for convolution)
                # Actually conv(a, e_j): result[k] = sum_i a[i]*e_j[k-i] = a[k-j] (if 0<=k-j<n)
                # conv(e_j, a): result[k] = sum_i e_j[i]*a[k-i] = a[k-j] (if 0<=k-j<n)
                # So conv(e_j, a) = conv(a, e_j) (convolution is commutative for real sequences)
                
                # Moving delta from i to j: new_a = a - delta*e_i + delta*e_j
                # new_conv = conv(a,a) + delta*conv(e_j-a_i_correction, a) + ...
                # More precisely:
                # new_conv = conv(a - d*e_i + d*e_j, a - d*e_i + d*e_j)
                # = conv(a,a) + d*conv(e_j - e_i, a) + d*conv(a, e_j - e_i) - d^2*conv(e_i - e_j, e_i - e_j)
                # = conv(a,a) + 2d*conv(e_j - e_i, a) - d^2*(2 - 2*conv(e_i,e_j))
                # For small d, the d^2 term is negligible
                
                # conv(e_j, a)[k] = a[k-j] if 0<=k-j<n else 0
                # conv(e_i, a)[k] = a[k-i] if 0<=k-i<n else 0
                
                lo_j = max(0, j)
                hi_j = min(size, j + n)
                if hi_j > lo_j:
                    g[lo_j:hi_j] += a[0:hi_j - lo_j]
                
                lo_i = max(0, i)
                hi_i = min(size, i + n)
                if hi_i > lo_i:
                    g[lo_i:hi_i] -= a[0:hi_i - lo_i]
                
                g *= 2.0  # Factor of 2 from both directions
                
                # Try several delta values
                d_max = ai * 0.5
                n_samples = 15
                deltas = np.linspace(0.01, d_max, n_samples)
                
                for d in deltas:
                    new_conv = conv + d * g
                    # Second order correction at positions 2i, 2j, i+j
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    if i + j < size and i != j:
                        new_conv[i + j] = conv[i + j] + 2 * d * (aj - ai) + 2 * d * d if i + j < size else 0
                    # Hmm, the cross terms are getting complicated. Let me just use the first order for now.
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                # Perturb
                n_perturb = max(3, n // 100)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 8:
                    break
        else:
            no_improve = 0
    
    return best_a, best_score

def asymmetric_init(n, rng):
    """Create an asymmetric initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 8)
    
    if choice == 0:
        # Skewed Gaussian
        c = rng.uniform(0.3, 0.7)
        w = rng.uniform(0.1, 0.3)
        a = np.exp(-(x - c) ** 2 / (2 * w ** 2))
    elif choice == 1:
        # Linear ramp with noise
        a = 0.3 + 0.7 * x
    elif choice == 2:
        # Two bumps, asymmetric
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.5, 0.8)
        w1 = rng.uniform(0.05, 0.15)
        w2 = rng.uniform(0.05, 0.15)
        h1 = rng.uniform(0.5, 1.0)
        h2 = rng.uniform(0.5, 1.0)
        a = h1 * np.exp(-(x - c1) ** 2 / (2 * w1 ** 2)) + h2 * np.exp(-(x - c2) ** 2 / (2 * w2 ** 2))
    elif choice == 3:
        # Power law
        alpha = rng.uniform(0.5, 2.0)
        a = x ** alpha
    elif choice == 4:
        # Sine-based asymmetric
        f = rng.uniform(1, 5)
        a = 0.5 + 0.3 * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
    elif choice == 5:
        # Exponential decay from center (asymmetric)
        c = rng.uniform(0.3, 0.7)
        a = np.exp(-abs(x - c) * rng.uniform(2, 8))
    elif choice == 6:
        # Cubic
        a = (1 - x) ** 2 * (0.5 + 0.5 * x)
    else:
        # Random smooth
        a = np.ones(n)
        freqs = rng.integers(1, 10, 5)
        phases = rng.uniform(0, 6.28, 5)
        ams = rng.uniform(0.1, 0.3, 5)
        for f, p, m in zip(freqs, phases, ams):
            a += m * np.sin(2 * np.pi * f * x + p)
    
    a = np.maximum(a, 0.01)
    a = a + 0.005 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Using solver (fallback): ECOS")
    return cp.ECOS


def build_T(a, lags):
    n = len(a)
    m = len(lags)
    T = np.zeros((m, n))
    for idx in range(m):
        k = lags[idx]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            T[idx, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return T


def solve_lp(a, M, lags, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T(a, lags)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lp_nesterov(a0, time_limit, solver, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Active set with two levels
        thresh_high = M * 0.97
        lags_high = np.where(conv_y >= thresh_high)[0]
        if len(lags_high) < 5:
            lags_high = np.where(conv_y >= M * 0.85)[0]
        if len(lags_high) > 800:
            lags_high = np.where(conv_y >= M * 0.92)[0]

        b = solve_lp(y, M, lags_high, solver)
        if b is None:
            b = solve_lp(y, M, np.arange(size), solver)
            if b is None:
                continue

        # Constraint verification and repair
        cross = fast_conv_cross(y, b)
        violated = np.where(cross > M * 1.002)[0]
        if len(violated) > 0:
            new_lags = np.union1d(lags_high, violated)
            if len(new_lags) < size:
                b2 = solve_lp(y, M, new_lags, solver)
                if b2 is not None:
                    b = b2
            else:
                b2 = solve_lp(y, M, np.arange(size), solver)
                if b2 is not None:
                    b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, |active|={len(lags_high)}")

    return best_a, best_score


def peak_gradient_descent(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Adaptive learning rate
    lr_base = 0.05 / np.sqrt(n)
    lr = lr_base

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break

        # Find all lags within 5% of peak
        peak_lags = np.where(conv >= M * 0.95)[0]
        n_peaks = len(peak_lags)

        # Compute gradient of the "smooth max" at these lags
        # d conv(k)/da[p] = 2*a[k-p] if 0 <= k-p < n
        grad = np.zeros(n)
        for k in peak_lags:
            q = k - np.arange(n)
            valid = (q >= 0) & (q < n)
            grad[valid] += 2.0 * a[q[valid]]

        # Weight peaks by their value (heavier weight for higher peaks)
        # Already implicitly done since we only include near-peak lags

        # Project gradient to maintain sum constraint
        grad -= grad.mean()

        # Also add a small regularization to prevent concentration
        # Penalize large values
        grad += 0.001 * a

        # Line search
        tried = 0
        while tried < 5:
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                lr = min(lr * 1.1, lr_base * 5)
                break
            else:
                lr *= 0.5
                tried += 1
        else:
            # No improvement, try a small random perturbation
            pert = 0.02 * np.abs(np.random.default_rng(it).standard_normal(n))
            a = a + pert
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PD it={it} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score


def coord_descent_polish(a0, time_limit, rng, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()

    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            peak_lags = np.where(conv >= cur_max * 0.88)[0]

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + 1 - i + n) % n))
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # First-order change in convolution
                g = np.zeros(size)
                lo_j = max(0, j)
                hi_j = min(size, j + n)
                if hi_j > lo_j:
                    g[lo_j:hi_j] += a[0:hi_j - lo_j]
                lo_i = max(0, i)
                hi_i = min(size, i + n)
                if hi_i > lo_i:
                    g[lo_i:hi_i] -= a[0:hi_i - lo_i]
                g *= 2.0

                d_max = ai * 0.6
                d_min = 1e-4
                if d_max < d_min:
                    continue
                n_samples = 20
                deltas = np.geomspace(d_min, d_max, n_samples)

                for d in deltas:
                    new_conv = conv + d * g
                    # Second-order corrections at diagonal positions
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    if i + j < size and i != j:
                        new_conv[i + j] = conv[i + j] + 2 * d * aj - 2 * d * ai + 2 * d * d
                    v = new_conv.max()
                    if v < best_v - 1e-12:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                n_perturb = max(5, n // 80)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 12:
                    break
        else:
            no_improve = 0

        if verbose and total_improvements % 1000 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                        cur_best = a_out.copy()
                    if sc < best_score:
                        best_score = sc
                        best_seq = a_out.copy()
                        print(f"    *** NEW GLOBAL BEST: {sc:.6f} (n={n_try})")

        # Phase 4: Coordinate descent polish
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 40:
            t_cd = min(t_alloc * 0.15, remaining * 0.3)
            if t_cd > 15:
                print(f"  [4] CD Polish ({t_cd:.0f}s):")
                a_out, _ = coord_descent_polish(cur_best, t_cd, rng, verbose=True)
                if a_out is not None:
                    sc = eval_fn(a_out)
                    if np.isfinite(sc) and sc < cur_score:
                        cur_score = sc
                        cur_best = a_out.copy()
                    if sc < best_score:
                        best_score = sc
                        best_seq = a_out.copy()
                        print(f"    *** NEW GLOBAL BEST: {sc:.6f} (n={n_try})")

        print(f"  Best after n={n_try}: {best_score:.6f}")

    # Final phase: intensive LP on the best solution found
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 80 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== FINAL INTENSIVE LP (n={n_best}, remaining={remaining:.0f}s) ===")
        t_final = remaining * 0.6
        a_out, _ = lp_nesterov(best_seq, t_final, solver, verbose=True)
        if a_out is not None:
            sc = eval_fn(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW GLOBAL BEST: {sc:.6f}")

        # Final CD
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 30:
            t_cd_final = remaining * 0.5
            print(f"  Final CD ({t_cd_final:.0f}s):")
            a_out, _ = coord_descent_polish(best_seq, t_cd_final, rng, verbose=True)
            if a_out is not None:
                sc = eval_fn(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW GLOBAL BEST: {sc:.6f}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\n{'='*60}")
        print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
