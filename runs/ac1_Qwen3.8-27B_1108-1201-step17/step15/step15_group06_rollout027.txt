Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-based Nesterov

"""

import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def _get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.SCS
    return _solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_obj(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * len(a) * fast_autocorr(a).max() / (s * s))


def eval_safe(a):
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_T(a, idx):
    """Build constraint matrix rows for given autocorrelation indices."""
    n = len(a)
    m = len(idx)
    T = np.zeros((m, n))
    for row, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[row, j] = a[k - j]
    return T


def solve_lp_maxsum(a, M, idx, solver):
    """Solve: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = build_T(a, idx)
    cons = [T @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def solve_lp_minmax(a, idx, solver, norm_sum):
    """Solve: min t s.t. T@b <= t, sum(b) = norm_sum, b >= 0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_T(a, idx)
    cons = [T @ b <= t, cp.sum(b) == norm_sum, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val

def structured_init_n(n, rng, style):

    if style == 0:

    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)

            a[i] = 0.5 * (1 + np.cos(np.pi * i / edge))

    elif style == 9:

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb(a, rng, scale):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.01)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Nesterov acceleration with backtracking line search
2. Adaptive active set with tight tolerance scheduling and dual tracking
3. Subgradient fallback for robustness
4. Set-aware coordinate descent focusing on peak-affecting coordinates
5. Multi-length exploration with interpolation and perturbation restarts
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None


def _get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.SCS
    print(f"  Fallback solver: SCS")
    return _solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_obj(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * len(a) * fast_autocorr(a).max() / (s * s))


def eval_safe(a):
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_T(a, idx):
    n = len(a)
    m = len(idx)
    T = np.zeros((m, n))
    for row in range(m):
        k = int(idx[row])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[row, j] = a[k - j]
    return T


def solve_lp_maxsum(a, M, idx, solver):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = build_T(a, idx)
    cons = [T @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def project_nonneg_sum(b, target_sum):
    n = len(b)
    if target_sum <= 0:
        return np.zeros(n)
    lo = b.min() - 1.0
    hi = b.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        val = np.sum(np.maximum(b - mid, 0.0))
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(b - tau, 0.0)


def subgradient_step(a, eta):
    n = len(a)
    norm_t = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    a_n = a * norm_t / s
    conv = fast_autocorr(a_n)
    max_val = conv.max()
    tol = 0.03
    active = np.where(conv >= max_val * (1.0 - tol))[0]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            g[j] += 2.0 * a_n[k - j]
    g /= max(1, len(active))
    b = a_n - eta * g
    return project_nonneg_sum(b, norm_t)


def nesterov_lp_v2(a0, time_limit, solver, seed=0, verbose=True):
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm_t = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_t / n
    else:
        a = a * norm_t / s
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    x = a.copy()
    t0 = time.monotonic()
    stall = 0
    active_idx = None
    eta_sg = norm_t / n * 0.1

    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        if active_idx is None or t % 25 == 0:
            tol = max(0.015, 0.06 * np.exp(-t / 150.0))
            thresh = M * (1.0 - tol)
            active_idx = np.where(conv_y >= thresh)[0]
            if len(active_idx) < 5:
                active_idx = np.where(conv_y >= M * 0.85)[0]
            if len(active_idx) > 0.7 * size:
                active_idx = np.where(conv_y >= M * 0.93)[0]
        b = solve_lp_maxsum(y, M, active_idx, solver)
        if b is None:
            if t % 30 == 0:
                active_idx = np.arange(size)
                b = solve_lp_maxsum(y, M, active_idx, solver)
            if b is None:
                x_new = subgradient_step(y, eta_sg)
                y = x_new
                x = x_new
                eta_sg *= 0.95
                continue
        b_sum = b.sum()
        g = b * norm_t / b_sum
        alpha = max(2.0 / (t + 2.0), 0.04)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_t / sy
        momentum = min((t - 1.0) / (t + 1.5), 0.92)
        x_new = y_new + momentum * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new * norm_t / sx
        sc_x = fast_autocorr(x_new).max()
        sc_y = fast_autocorr(y_new).max()
        best_x = x_new if sc_x < sc_y else y_new
        best_xs = min(sc_x, sc_y)
        for frac in [0.6, 1.4]:
            trial = (1.0 - frac * alpha) * y + frac * alpha * g
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st > 1e-15:
                trial = trial * norm_t / st
                sc_t = fast_autocorr(trial).max()
                if sc_t < best_xs:
                    best_xs = sc_t
                    best_x = trial
        y = y_new
        x = best_x
        if best_xs < best_score:
            best_score = best_xs
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        sc_y2 = fast_autocorr(y).max()
        if sc_y2 < best_score:
            best_score = sc_y2
            best_a = y.copy()
            stall = 0
        if stall > 120:
            y = best_a.copy()
            x = best_a.copy()
            active_idx = None
            stall = 0
            eta_sg = norm_t / n * 0.1
        if verbose and t % 80 == 0:
            print(f"    NLP2 t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f} M={M:.6f}")
    return best_a


def cd_v2(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm_t = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm_t / s
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        max_k = int(np.argmax(conv))
        lo_f = max(0, max_k - n + 1)
        hi_f = min(n, max_k + 1)
        focus = np.arange(lo_f, hi_f)
        others = np.setdiff1d(np.arange(n), focus)
        perm = np.concatenate([rng.permutation(focus), rng.permutation(others)])
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-14:
                continue
            best_val = best_score
            best_nv = ai
            for frac in [0.4, 0.6, 0.8, 1.0, 1.25, 1.6, 2.2]:
                nv = ai * frac
                if abs(nv - ai) < 1e-14:
                    continue
                a_try = a.copy()
                a_try[i] = nv
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try * norm_t / st
                c_try = fast_autocorr(a_try)
                sc = c_try.max()
                if sc < best_val - 1e-13:
                    best_val = sc
                    best_nv = nv
            if best_nv != ai:
                a[i] = best_nv
                st = a.sum()
                if st > 1e-15:
                    a = a * norm_t / st
                conv = fast_autocorr(a)
                if conv.max() < best_score:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
        if verbose and it % 10 == 0:
            print(f"    CD2 it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")
    return best_a, best_score


def structured_init_n(n, rng, style):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    elif style == 8:
        w = 0.15
        a = np.ones(n)
        edge = max(1, int(w * n))
        for i in range(edge):
            v = 0.5 * (1.0 + np.cos(np.pi * i / edge))
            a[i] = v
            a[n - 1 - i] = v
    elif style == 9:
        a = np.exp(-((x - 0.5) * 4.0) ** 4)
    else:
        a = 1.0 + 0.2 * np.sin(2 * np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    sn = a.sum()
    if sn < 1e-15:
        return np.ones(n) / n
    return a / sn


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb(a, rng, scale):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.02)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
