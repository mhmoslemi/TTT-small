        g = b / b_sum  # Normalize

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        S_y = y_new.sum()
        if S_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / S_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            S_a = a_new.sum()
            if S_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / S_a
        
        y = y_new
        a = a_new
        
        # Track best
        M_y = fast_conv_self(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
        
        M_a = fast_conv_self(a).max()
        if M_a < best_M:
            best_M = M_a
            best_a = a.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-Nest t={t} ({elapsed:.0f}s): best_M={best_M:.10f}, cur_M={M_y:.10f}")
    
    return best_a, best_M

def lse_gradient_step(a, T_val, lr, momentum=None, velocity=None):
    """One step of LSE-approximated gradient descent."""
    n = len(a)
    fs = fft_size_for(n)
    conv = fast_conv_self(a)
    max_c = conv.max()
    
    shifted = (conv - max_c) / T_val
    shifted = np.clip(shifted, -50, 0)
    w = np.exp(shifted)
    w = w / w.sum()
    
    # Gradient of sum(w[k] * conv(a,a)[k]) w.r.t. a
    W = rfft(w, fs)
    A = rfft(a, fs)
    
    # d/d a[i] of sum_k w[k] * sum_j a[j]*a[k-j]
    # = sum_k w[k] * (a[k-i] + a[i+k-n]...) 
    # Using FFT: grad = ifft(conj(A) * W) + ifft(A * W) (roughly)
    
    ca = irfft(np.conj(A) * W, fs)[:n]
    aw = irfft(A * W, fs)[:n]
    grad = 2.0 * (ca + aw)
    
    if momentum is None:
        m = grad.copy()
    else:
        m = momentum * 0.9 * m + (1 - 0.9) * grad
    
    a_new = a - lr * m
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new / S
    
    return a_new, m

def lse_anneal(a_init, time_limit, verbose=False):
    """LSE-based gradient descent with temperature annealing."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0) / a.sum()
    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    n_stages = 30
    temps = np.geomspace(50.0, 0.01, n_stages)
    lr = 0.3 / np.sqrt(n)
    m = np.zeros(n)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        n_steps = max(10, min(100, int(remaining * 8)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            a, m = lse_gradient_step(a, T_val, lr, m, None)
            M = fast_conv_self(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.3f}: best={best_M:.8f}")
    
    return best_a, best_M

def pair_cd_polish(a_init, time_limit, rng, verbose=False):
    """Pair coordinate descent polishing."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0) / a.sum()
    best_a = a.copy()
    best_M = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue
            
            # Find candidate j's
            j_candidates = list(rng.integers(n, size=5))
            peak_lags = np.where(conv >= cur_max * 0.9)[0]
            for kl in peak_lags[:5]:
                j_candidates.append(int((kl - i) % n))
                j_candidates.append(int((kl + i) % n))
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in set(j_candidates):
                if j == i or j >= n:
                    continue
                aj = a[j]
                
                # Compute delta-convolution
                g = np.zeros(size)
                # a[j] increases by d: adds d*a shifted by j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[:hi-lo]
                # a[i] decreases by d: subtracts d*a shifted by i
                lo2 = max(0, i)
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[:hi2-lo2]
                # Also the reflected parts
                lo3 = max(0, j - n + 1)
                hi3 = min(size, j + 1)
                if hi3 > lo3:
                    g[lo3:hi3] += a[j - np.arange(lo3, hi3)]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                d_max = ai * 0.9
                for d in np.linspace(0.01, d_max, 10):
                    new_conv = conv + d * g
                    # Correct the diagonal terms
                    if 2*i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*d + d*d
                    if 2*j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*d + d*d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 1e-6 and best_v < best_M - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a / a.sum()
                new_max = fast_conv_self(a).max()
                if new_max < best_M - 1e-10:
                    best_M = new_max
                    best_a = a.copy()
                    improved = True
                else:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = a / a.sum()
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                # Perturb
                n_pert = max(5, n // 50)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0) / a.sum()
                if no_improve >= 8:
                    break
        else:
            no_improve = 0
        
        if verbose and t0 and (time.monotonic() - t0) % 30 < 1:
            print(f"    PCD ({time.monotonic()-t0:.0f}s): best_M={best_M:.10f}")
    
    return best_a, best_M

def structured_init(n, rng):

    a += 0.01 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

    if _solver is not None:
        return _solver
    for s in [cp.MOSEK, cp.GUROBI, cp.XPRESS, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(50, nonneg=True)
            A = np.random.rand(80, 50)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [A @ x <= 1, cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Solver (fallback): ECOS")
    return _solver

def fft_size_for(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def fast_conv_self(a):
    n = len(a)
    fs = fft_size_for(n)
    A = rfft(a, fs)
    return irfft(A * A, fs)[:2 * n - 1]

def fast_conv_cross(a, b):
    n = len(a)
    fs = fft_size_for(n)
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A * B, fs)[:2 * n - 1]

def compute_max_conv(a):
    a = np.maximum(a, 0.0)
    return fast_conv_self(a).max()

def eval_fn(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_T_matrix(a, active_set):
    n = len(a)
    T = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_end > i_start:
            T[idx, i_start:i_end] = a[k - np.arange(i_start, i_end)]
    return T

def solve_lp(a, M, active_set, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_matrix(a, active_set)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def select_active_set(conv, M, max_constraints=2000):
    size = len(conv)
    if size <= max_constraints:
        return np.arange(size)
    # Select top constraints by value
    n_top = min(max_constraints, max(100, size // 2))
    # Include the global max
    max_idx = np.argmax(conv)
    # Use a mix: top values and near-threshold
    threshold = M * 0.6
    above = np.where(conv >= threshold)[0]
    if len(above) < n_top:
        # Fill with next highest
        remaining = np.argsort(conv)[-n_top:]
        active = np.union1d(above, remaining)
    else:
        active = above
    # Cap the size
    if len(active) > max_constraints:
        active = np.argsort(conv[active])[-max_constraints:]
        active = np.sort(active)
    # Ensure max is included
    if max_idx not in active:
        active = np.append(active, max_idx)
    return np.sort(active)

def lp_nesterov(a_init, time_limit, solver, verbose=True):
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_M = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    iter_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-15:
            break

        # Select active constraints
        active_set = select_active_set(conv_y, M, max_constraints=1500)

        # Solve LP
        b = solve_lp(y, M, active_set, solver)

        if b is None or b.sum() < 1e-15:
            # Try with more constraints
            active_set2 = select_active_set(conv_y, M, max_constraints=3000)
            b = solve_lp(y, M, active_set2, solver)

        if b is None or b.sum() < 1e-15:
            continue

        # Verify constraint satisfaction, add violated constraints if needed
        conv_yb = fast_conv_cross(y, b)
        violated = np.where(conv_yb > M * 1.005)[0]
        if len(violated) > 0 and len(active_set) + len(violated) <= 3500:
            new_active = np.union1d(active_set, violated)
            b2 = solve_lp(y, M, new_active, solver)
            if b2 is not None and b2.sum() > 1e-15:
                b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        S_y = y_new.sum()
        if S_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / S_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            S_a = a_new.sum()
            if S_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / S_a

        y = y_new
        a = a_new
        iter_count += 1

        M_y = compute_max_conv(y)
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()

        M_a = compute_max_conv(a)
        if M_a < best_M:
            best_M = M_a
            best_a = a.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_M
            print(f"    LP t={t} ({elapsed:.0f}s): best_score={score:.8f}, M={best_M:.10f}")

    return best_a, best_M

def lse_grad_descent(a_init, time_limit, verbose=False):
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_M = compute_max_conv(a)
    t0 = time.monotonic()
    fs = fft_size_for(n)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0

    n_stages = 25
    temps = np.geomspace(60.0, 0.02, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(15, min(150, int(remaining * 6)))
        lr = 0.4 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            step += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -60.0, 0.0)
            w = np.exp(shifted)
            w = w / w.sum()

            W = rfft(w, fs)
            A = rfft(a, fs)

            ca = irfft(np.conj(A) * W, fs)[:n]
            aw = irfft(A * W, fs)[:n]
            grad = 2.0 * (ca + aw)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / S_new

            a = a_new
            M = compute_max_conv(a)
            if M < best_M:
                best_M = M
                best_a = a.copy()

        if verbose and si % 5 == 0:
            print(f"    LSE stage {si} T={T_val:.4f} ({time.monotonic()-t0:.0f}s): best_M={best_M:.8f}")

    return best_a, best_M

def pair_cd(a_init, time_limit, rng, verbose=False):
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_M = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-13:
                continue

            # Candidate j's from peak lags and random
            j_set = list(rng.integers(n, size=4))
            peaks = np.where(conv >= cur_max * 0.88)[0]
            for kl in peaks[:8]:
                j_set.append(int((kl - i) % n))
                j_set.append(int((kl + i) % n))

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in set(j_set):
                if j == i or j >= n or j < 0:
                    continue
                aj = a[j]

                # Delta convolution for moving d from i to j
                g = np.zeros(size)
                lo = j
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[:hi - lo]
                lo2 = j - n + 1
                hi2 = min(size, j + 1)
                if lo2 < 0:
                    lo2 = 0
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = i
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[:hi3 - lo3]
                lo4 = i - n + 1
                hi4 = min(size, i + 1)
                if lo4 < 0:
                    lo4 = 0
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                d_max = ai * 0.85
                if d_max < 1e-12:
                    continue
                deltas = np.linspace(0.02, d_max, 12)
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 1e-7 and best_v < best_M - 1e-11:
                a[i] -= best_delta
                a[best_j] += best_delta
                S_new = a.sum()
                if S_new > 1e-15:
                    a = a / S_new
                    new_max = compute_max_conv(a)
                    if new_max < best_M - 1e-11:
                        best_M = new_max
                        best_a = a.copy()
                        improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a / a.sum()

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                n_pert = max(5, n // 40)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                S_p = a.sum()
                if S_p > 1e-15:
                    a = a / S_p
                if no_improve >= 10:
                    break
        else:
            no_improve = 0

        if verbose and (time.monotonic() - t0) % 30 < 1.0:
            print(f"    PCD ({time.monotonic()-t0:.0f}s): best_M={best_M:.10f}")

    return best_a, best_M

def structured_init(n, rng):
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 12)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sqrt(np.sin(np.pi * x))
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = rng.integers(3, 10)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)

    a = np.maximum(a, 0.0)
    a += 0.005 * np.abs(rng.standard_normal(n))
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    S = f.sum()
    if S < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / S
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1680.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def track(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        S = a.sum()
        if S < 0.01:
            return
        a = a / S
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old:.6f}, n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    # Phase 1: LSE gradient descent for quick initial improvement
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE n={n_base} budget={t1:.0f}s ===")
    if t1 > 20:
        a_out, _ = lse_grad_descent(best_seq, t1, verbose=False)
        track(a_out, f"lse_n{n_base}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 420.0)
    print(f"\n=== Phase 2: LP-Nesterov n={len(best_seq)} budget={t2:.0f}s ===")
    if t2 > 30:
        a_out, _ = lp_nesterov(best_seq, t2, solver, verbose=True)
        track(a_out, f"lp1_n{len(best_seq)}")

    # Phase 3: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: PCD n={len(best_seq)} budget={t3:.0f}s ===")
    if t3 > 20:
        a_out, _ = pair_cd(best_seq, t3, rng, verbose=False)
        track(a_out, f"pcd1_n{len(best_seq)}")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 360.0)
    print(f"\n=== Phase 4: LP-Nesterov2 n={len(best_seq)} budget={t4:.0f}s ===")
    if t4 > 30:
        a_out, _ = lp_nesterov(best_seq, t4, solver, verbose=True)
        track(a_out, f"lp2_n{len(best_seq)}")

    # Phase 5: LSE round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.08, 90.0)
    print(f"\n=== Phase 5: LSE2 n={len(best_seq)} budget={t5:.0f}s ===")
    if t5 > 15:
        a_out, _ = lse_grad_descent(best_seq, t5, verbose=False)
        track(a_out, f"lse2_n{len(best_seq)}")

    # Phase 6: Final PCD
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 6: PCD final budget={t6:.0f}s ===")
    if t6 > 15:
        a_out, _ = pair_cd(best_seq, t6, rng, verbose=False)
        track(a_out, f"pcd_final_n{len(best_seq)}")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 90:
        print(f"\n=== Phase 7: Multi-length exploration ===")
        for n_try in [n_base - 200, n_base + 200, n_base + 400, n_base + 600]:
            if remaining < 50:
                break
            if n_try < 200 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            t_try = min(remaining * 0.3, 45.0)
            print(f"  n={n_try} budget={t_try:.0f}s")
            a_interp = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_grad_descent(a_interp, min(t_try, 25.0), verbose=False)
            track(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t_lp = min(remaining * 0.4, 25.0)
                a_out, _ = lp_nesterov(a_out, t_lp, solver, verbose=False)
                track(a_out, f"ml_lp_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)

    # Phase 8: Fresh random starts
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 8: Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(3):
            if remaining < 25:
                break
            n_fresh = n_base + int(rng.integers(-300, 400))
            if n_fresh < 200:
                n_fresh = 800
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.3, 25.0)
            a_out, _ = lse_grad_descent(a_fresh, t_fresh, verbose=False)
            track(a_out, f"fresh{attempt}_n{n_fresh}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = pair_cd(a_out, min(remaining * 0.4, 15.0), rng, verbose=False)
                track(a_out, f"fresh{attempt}_pcd_n{n_fresh}")
                remaining = time_budget - (time.monotonic() - t_global)

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in best_seq]
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
