                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new

    return best_a, best_max


def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE-smoothed Adam optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    n_stages = 25
    temps = np.geomspace(100.0, 0.02, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        n_steps = max(50, min(300, int(remaining * 15)))
        lr = 0.8 / np.sqrt(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            
            grad = 2.0 * (ca + aw)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and (si % 5 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam stage {si} T={T_val:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        n_peaks = rng.integers(3, 10)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 7:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 8:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def swap_optimize(a, time_limit, rng, verbose=False):
    """Optimize using swap operations (exchange values between two positions)."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    no_improve = 0
    size = 2 * n - 1
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find peak lags
        peak_lags = np.where(conv >= cur_max * 0.9)[0]
        
        # For each peak lag, the "offending" pairs are (i, k-i) where a_i * a_{k-i} is large
        # Try swapping a_i with a_j where j is chosen to reduce the peak
        
        improved = False
        
        # Sample some random pairs to try
        n_tries = min(200, n // 5)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            ai, aj = a[i], a[j]
            if abs(ai - aj) < 1e-10:
                continue
            
            # Try the swap
            a[i], a[j] = aj, ai
            
            new_max = fast_conv_self(a).max()
            
            if new_max < best_max - 1e-10:
                best_max = new_max
                best_a = a.copy()
                improved = True
            else:
                # Revert
                a[i], a[j] = ai, aj
        
        if not improved:
            no_improve += 1
            if no_improve >= 15:
                break
        else:
            no_improve = 0
    
    return best_a, best_max

def fast_conv_self(a):

    size = 2 * n - 1

def pair_cd_fast(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    sweep = 0

    n_delta = 25
    n_cand = 10
    n_important = min(150, n)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()

        threshold = cur_max * 0.88
        peak_lags = np.where(conv >= threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.where(conv >= cur_max * 0.95)[0]

        # Compute pressure for candidate selection
        pressure = np.zeros(n)
        for k in peak_lags:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                j_start = k - i_min
                j_end = k - i_max
                pressure[i_min:i_max + 1] += a[j_end:j_start + 1][::-1]

        pressure_w = pressure * a

        if n_important < n:
            important_idx = np.argsort(pressure_w)[::-1][:n_important]
        else:
            important_idx = np.arange(n)

        pass_improved = False
        n_moves = 0

        # Shuffle important indices for diversity
        shuffled = rng.permutation(len(important_idx))
        important_idx = important_idx[shuffled]

        for ii in range(len(important_idx)):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(important_idx[ii])
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            low_idx = np.argsort(pressure_w)[:n_cand * 3]
            for j in low_idx:

            if not cand_list:
                continue

            d_i = np.zeros(size)
            d_i[i:i + n] = a

            best_delta = 0.0
            best_new_max = cur_max
            best_j = -1

                d_j[j:j + n] = a

                    np.linspace(max_d * 0.3, max_d, 10)

                best_local = np.argmin(new_maxes)

                if new_maxes[best_local] < best_new_max - 1e-12:

                    best_delta = deltas[best_local]
                    best_j = j

            if best_j >= 0 and best_delta > 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()

                new_max = fast_conv_self(a).max()
                if new_max < best_max - 1e-10:

def lse_adam_optimize(a0, time_limit, verbose=True, log_space=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 25
    temps = np.geomspace(80.0, 0.03, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        n_steps = max(50, min(400, int(remaining * 20)))
        lr = 1.0 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            if log_space:
                b = np.log(np.maximum(a, 1e-10))
                b_new = b - lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = np.exp(b_new)
            else:
                a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam stage {si} T={T_val:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def swap_optimize(a0, time_limit, rng, verbose=False):

            no_improve += 1
            if no_improve >= 20:
                break
        else:
            no_improve = 0

    return best_a, best_max


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
