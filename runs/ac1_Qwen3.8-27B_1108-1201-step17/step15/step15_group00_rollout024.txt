**Alternative idea: Use the structure of the problem**

The problem is equivalent to: find a probability distribution $a$ on $\{0, \ldots, n-1\}$ that minimizes $\max_k \sum_i a_i a_{k-i}$.

This is related to the "minimum overlap" problem. If we think of $a$ as a probability distribution, then $(a*a)_k$ is the probability that the sum of two independent draws from $a$ equals $k$. We want to make this sum distribution as uniform as possible.

This is exactly the problem of finding a probability distribution on $\{0, \ldots, n-1\}$ whose "sumset distribution" is as uniform as possible. The uniform distribution on $\{0, \ldots, n-1\}$ gives a triangular sumset distribution (not uniform). To make it more uniform, we need to "flatten" it.

This is related to the theory of **asymptotically good codes** or **balanced sequences**. In particular, it's related to the concept of **perfect sequences** in radar/sonar (where all autocorrelation peaks are equal), but we're looking at the sum (convolution) rather than the correlation.

Actually, for the sum distribution to be uniform on $\{0, \ldots, 2n-2\}$, we'd need $(a*a)_k = \frac{1}{2n-1}$ for all $k$. This would require $|A(\omega)|^2 = \frac{1}{2n-1}$ for all frequencies $\omega$, where $A(\omega) = \sum_i a_i e^{i\omega i}$. This means $|A(\omega)| = \frac{1}{\sqrt{2n-1}}$ for all $\omega$.

But $A(0) = \sum a_i = 1$, so we'd need $1 = \frac{1}{\sqrt{2n-1}}$, which is impossible for $n > 1$. So a perfectly uniform sum distribution is impossible.

The best we can do is to minimize the peak of the sum distribution. This is a well-studied problem in the context of **constellation design** and **peak-to-average power ratio (PAPR)** minimization.

Known results:
- For real-valued sequences, the minimum PAPR of the aperiodic autocorrelation is related to the **Golay complementary sequences** and **Barker sequences**.
- For complex sequences, there are known constructions with PAPR close to 1.
- For real non-negative sequences, the problem is more constrained.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B on L2 norm of autocorrelation (smooth preprocessing)
2. L-BFGS-B on log-sum-exp smoothed max with adaptive lambda
3. Nesterov-accelerated gradient with full subgradient
4. Stochastic perturbation (simulated annealing) for exploration
5. Final coordinate polish on true objective
Different from prior approach: uses second-order L-BFGS, full gradient (not top-k),
L2 smoothing as warm-start, and SA exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Compute full autocorrelation (convolution with self) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Stable softmax."""
    x_shift = x - x.max()
    e = np.exp(x_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def ac_l2_squared(a):
    """Sum of squares of autocorrelation values."""
    ac = fast_autocorr(a)
    return np.dot(ac, ac)


def ac_l2_grad_x(x):
    """Gradient of sum_k (a*a)_k^2 w.r.t. x (softmax params)."""
    a = softmax(x)
    n = len(a)
    ac = fast_autocorr(a)
    # d/d a_m sum_k (a*a)_k^2 = 4 * sum_k (a*a)_k * a_{k-m}
    # = 4 * (ac * reversed_a) cross-correlation
    fft_size = 1 << (2 * n - 2).bit_length()
    AC_fft = np.fft.rfft(ac, fft_size)
    A_fft = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(A_fft) * AC_fft, fft_size)
    grad_a = 4.0 * cross[:n]
    # Softmax Jacobian
    sum_ga = np.dot(grad_a, a)
    grad_x = a * (grad_a - sum_ga)
    return grad_x


def ac_l2_val_and_grad_x(x):
    """Value and gradient of L2^2 autocorrelation w.r.t. softmax params."""
    a = softmax(x)
    val = ac_l2_squared(a)
    grad = ac_l2_grad_x(x)
    return val, grad


def smoothed_max_val_and_grad_x(x, lam):
    """
    Value and gradient of (1/lam)*log(sum exp(lam*(a*a)_k)) w.r.t. softmax params.
    """
    a = softmax(x)
    n = len(a)
    ac = fast_autocorr(a)

    lam_ac = lam * ac
    max_val = lam_ac.max()
    e = np.exp(lam_ac - max_val)
    e_sum = e.sum()
    F = (max_val + np.log(e_sum)) / lam
    p = e / e_sum

    # grad_a[m] = 2 * sum_j p[j+m] * a[j]  (cross-correlation of a and p)
    fft_size = 1 << (2 * n - 2).bit_length()
    P_fft = np.fft.rfft(p, fft_size)
    A_fft = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(A_fft) * P_fft, fft_size)
    grad_a = 2.0 * cross[:n]

    # Softmax Jacobian
    sum_ga = np.dot(grad_a, a)
    grad_x = a * (grad_a - sum_ga)

    return F, grad_x


def true_score(a):
    """Compute 2*n*max(conv)/sum^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    M = ac.max()
    n = len(a)
    return float(2.0 * n * M / (S * S))


def lbfgs_optimize(x0, func, maxiter=500, time_limit=60.0, verbose=False, label=""):
    """Run L-BFGS-B with time limit."""
    t0 = time.monotonic()

    def objective(x):
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        return func(x)[0]

    def gradient(x):
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        return func(x)[1]

    try:
        res = minimize(objective, x0, jac=gradient, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': maxiter * 10,
                                'iprint': -1})
        return res.x, res.fun, True
    except (TimeoutError, Exception) as e:
        if verbose:
            print(f"      L-BFGS stopped: {str(e)[:50]}")
        # Return best found so far via our own tracking
        return x0, float('inf'), False


def lbfgs_smoothed_max(a_init, lam, maxiter=300, time_limit=60.0, verbose=False):
    """Optimize smoothed max using L-BFGS-B in softmax space."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    x0 = np.log(a)  # log of the sequence as initial softmax params

    def func(x):
        return smoothed_max_val_and_grad_x(x, lam)

    x_out, obj, success = lbfgs_optimize(x0, func, maxiter, time_limit, verbose)
    a_out = softmax(x_out)
    return a_out


def lbfgs_l2(a_init, maxiter=500, time_limit=60.0, verbose=False):
    """Optimize L2^2 of autocorrelation using L-BFGS-B."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    x0 = np.log(a)

    def func(x):
        return ac_l2_val_and_grad_x(x)

    x_out, obj, success = lbfgs_optimize(x0, func, maxiter, time_limit, verbose)
    a_out = softmax(x_out)
    return a_out


def nesterov_smoothed(a_init, time_limit, lam_start=100.0, lam_end=3000.0,
                       lr_start=0.01, verbose=False):
    """Nesterov accelerated gradient on smoothed max with increasing lambda."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = true_score(a)
    t0 = time.monotonic()

    x = np.log(a + 1e-15)
    x_prev = x.copy()
    counter = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        counter += 1

        # Adaptive lambda: increase over time
        frac = min(1.0, t / max(1, int(time_limit / 0.05)))
        lam = lam_start * (lam_end / lam_start) ** frac

        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_t = (beta - 1.0) / (beta + 1.0)
        y = x + beta_t * (x - x_prev)

        # Gradient at y
        F_y, grad_y = smoothed_max_val_and_grad_x(y, lam)

        # Adaptive learning rate
        lr = lr_start / (1.0 + 0.01 * t)
        x_new = y - lr * grad_y

        x_prev = x
        x = x_new

        # Every 10 steps, check true objective
        if t % 10 == 0:
            a_curr = softmax(x)
            score = true_score(a_curr)
            if score < best_score:
                best_score = score
                best_a = a_curr.copy()
                if verbose and t % 100 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      Nesterov t={t} lam={lam:.0f} ({elapsed:.0f}s): score={score:.6f}")

    # Final check
    a_final = softmax(x)
    score_final = true_score(a_final)
    if score_final < best_score:
        best_a = a_final.copy()
        best_score = score_final

    return best_a


def stochastic_search(a_init, time_limit, verbose=True, rng_seed=42):
    """Simulated annealing style stochastic local search."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = true_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    current_score = best_score
    T = 0.001  # initial temperature
    T_min = 1e-7
    cooling = 0.9995

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1

        # Random perturbation: move mass between two positions
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        # Amount to move (fraction of a[i])
        frac = rng.uniform(0.001, 0.05)
        amt = a[i] * frac
        if amt < 1e-15:
            continue

        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        trial_score = true_score(a_trial)
        delta = trial_score - current_score

        # Accept if better, or with probability exp(-delta/T)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            a = a_trial
            current_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_a = a.copy()
                if verbose and step % 5000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      SA step={step} ({elapsed:.0f}s): score={best_score:.6f}, T={T:.6f}")
        else:
            # Reject: try a smaller perturbation
            pass

        T = max(T_min, T * cooling)

        # Periodically re-optimize with L-BFGS from current state
        if step % 50000 == 0 and time.monotonic() - t0 < time_limit - 30:
            a_opt = lbfgs_smoothed_max(a, lam=1000, maxiter=50, time_limit=10.0)
            opt_score = true_score(a_opt)
            if opt_score < best_score:
                best_score = opt_score
                best_a = a_opt.copy()
            a = a_opt
            current_score = true_score(a)

    return best_a


def coordinate_polish(a_init, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent polish: try moving mass from high-peak contributors to low-peak areas."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = true_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()

        # Find the peak lags
        peak_lags = np.where(ac > 0.95 * M)[0]

        improved = False
        # For each peak lag, try to reduce it by moving mass
        for k in peak_lags[:5]:
            k_int = int(k)
            # Contributing pairs: a[i] * a[k-i] for valid i
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start >= i_end:
                # Single term: a[k_int]^2 (only if k_int < n)
                if 0 <= k_int < n and a[k_int] > 1e-10:
                    # Try moving mass from k_int to a random position
                    dst = int(rng.integers(n))
                    if dst != k_int:
                        for frac in [0.01, 0.03, 0.05, 0.08]:
                            amt = a[k_int] * frac
                            a_trial = a.copy()
                            a_trial[k_int] -= amt
                            a_trial[dst] += amt
                            a_trial = np.maximum(a_trial, 0.0)
                            a_trial = a_trial / a_trial.sum()
                            s_trial = true_score(a_trial)
                            if s_trial < best_score:
                                a = a_trial
                                best_score = s_trial
                                best_a = a.copy()
                                improved = True
                                break
                continue

            idx = np.arange(i_start, i_end + 1)
            partners = k_int - idx
            contrib = a[idx] * a[partners]

            # Find the pair with maximum contribution
            max_pg = np.argmax(contrib)
            src = int(idx[max_pg])

            # Find destination: positions that don't contribute to any peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in peak_lags:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Try a few random destinations
            n_try = min(20, len(candidates))
            chosen = candidates[rng.permutation(len(candidates))[:n_try]]

            for dst in chosen:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.04, 0.06]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    s_trial = true_score(a_trial)
                    if s_trial < best_score:
                        a = a_trial
                        best_score = s_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        if verbose and (t0 and (time.monotonic() - t0) % 30 < 0.1):
            elapsed = time.monotonic() - t0
            print(f"      Coord ({elapsed:.0f}s): score={best_score:.6f}, stall={stall}")

    return best_a


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = true_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = true_score(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = true_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    print(f"Time budget: {time_budget:.0f}s")

    # === Phase 1: L2 smoothing (warm start) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L2 warm-start n={n0} ===")
    t1 = min(remaining * 0.10, 120.0)
    if t1 > 20:
        a_out = lbfgs_l2(best_seq, maxiter=500, time_limit=t1, verbose=True)
        try_update(a_out, "l2_warm")

    # === Phase 2: L-BFGS on smoothed max with increasing lambda ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: L-BFGS smoothed max n={n0} ===")
    lambdas = [200, 500, 1000, 2000, 4000, 6000, 8000]
    for lam in lambdas:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t2 = min(remaining * 0.12, 100.0)
        if t2 < 15:
            break
        print(f"  lambda={lam}:")
        a_out = lbfgs_smoothed_max(best_seq, lam=lam, maxiter=200, time_limit=t2, verbose=True)
        try_update(a_out, f"lbfgs_lam{lam}")

    # === Phase 3: Nesterov with high lambda ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Nesterov n={n0} ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        a_out = nesterov_smoothed(best_seq, time_limit=t3, lam_start=500.0,
                                  lam_end=8000.0, lr_start=0.02, verbose=True)
        try_update(a_out, "nesterov")

    # === Phase 4: Another L-BFGS round from improved point ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: L-BFGS round 2 n={n0} ===")
    for lam in [3000, 5000, 8000, 12000]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        t4 = min(remaining * 0.10, 80.0)
        if t4 < 10:
            break
        print(f"  lambda={lam}:")
        a_out = lbfgs_smoothed_max(best_seq, lam=lam, maxiter=300, time_limit=t4)
        try_update(a_out, f"lbfgs2_lam{lam}")

    # === Phase 5: Stochastic search (exploration) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: SA exploration n={n0} ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 30:
        a_out = stochastic_search(best_seq, time_limit=t5, verbose=True, rng_seed=77)
        try_update(a_out, "sa")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coord polish n={n0} ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 20:
        a_out = coordinate_polish(best_seq, time_limit=t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Final L-BFGS + Nesterov from best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final optimization n={n0} ===")
    if remaining > 60:
        # L-BFGS at very high lambda
        t7 = min(remaining * 0.30, 150.0)
        a_out = lbfgs_smoothed_max(best_seq, lam=10000, maxiter=500, time_limit=t7, verbose=True)

        t8 = min(remaining * 0.40, 150.0)
        if t8 > 15:

            try_update(a_out, "coord2")

    # === Phase 9: Final L-BFGS squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 9: Final squeeze ===")
        t9 = min(remaining * 0.7, 100.0)
        for lam in [5000, 10000, 15000]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 10:
                break
            t9a = min(remaining * 0.4, 60.0)
            if t9a < 5:
                break
            a_out = lbfgs_smoothed_max(best_seq, lam=lam, maxiter=300, time_limit=t9a)
            try_update(a_out, f"squeeze_lam{lam}")

    # Final validation
    if best_seq is not None:
        fs = true_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        print(f"{'='*60}")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
