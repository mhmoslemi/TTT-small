    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

            for i in range(j_max + 1):
                # a[i] * a[k-i]: contributes a[k-i] to d/d(a[i])
                # a[k-i] * a[i]: contributes a[k-i] to d/d(a[k-i])... 
                # Actually conv[k] = sum_j a[j]*a[k-j] for valid j
                # d/d(a[i]) = a[k-i] + a[i-k] ... but i-k < 0 since k >= n > i
                # So d/d(a[i]) = a[k-i] (if 0 <= k-i < n)
                ki = k - i
                if 0 <= ki < n:
                    g[i] += a[ki]
                    # Also a[ki]*a[i-k]... no, k-i >= 0 always here
                    # And the symmetric term: when j = k-i, we get a[k-i]*a[i]
                    # d/d(a[i]) from a[i]*a[k-i] is a[k-i]
                    # d/d(a[ki]) from a[ki]*a[i] ... ki = k-i, so d/d(a[ki]) contributes a[i]
                    # But we're iterating over i, so for the term a[j]*a[k-j]:
                    # if j = i: contributes a[k-i] to g[i]
                    # if k-j = i i.e. j = k-i: contributes a[i] to g[k-i]
                    # So g[i] gets a[k-i] (from j=i term) and g[k-i] gets a[i] (from j=k-i term)
                    # But we only process each i once, so we need to add both
                    g[ki] += a[i]
    return g, max_val, conv


def lp_best_response_fast(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(size):
        if k < n:
            # conv[k] = sum_{j=0}^{n-1-k} a[j] * b[j+k]
            # = a[:n-k] @ b[k:]
            if k >= n:
                continue
            coef = a[:n-k]
            if coef.sum() < 1e-15:
                continue
            constraints.append(coef @ b_var[k:] <= M)
        else:
            # k >= n: conv[k] = sum_{j=k-n+1}^{n-1} a[n-1-(k-n)+j-n+1]... 
            # Let's use: conv[k] = sum_j a[j] * b[k-j] for j in [k-n+1, min(n-1,k)]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi < lo:
                continue
            # conv[k] = sum_{j=lo}^{hi} a[j] * b[k-j]
            # Let i = k-j, so j = k-i, i ranges from k-hi to k-lo
            # conv[k] = sum_{i=k-hi}^{k-lo} a[k-i] * b[i]
            # = sum_{i=0}^{n-1} a_reg[k-i] * b[i] where a_reg is padded
            # Simpler: just build the constraint directly
            terms = []
            for j in range(lo, hi + 1):
                i = k - j  # index into b
                if 0 <= i < n and a[j] > 0:
                    terms.append(a[j] * b_var[i])
            if terms:
                constraints.append(cp.sum(terms) <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    y = a.copy()
    momentum_t = 1.0
    patience = 0
    stuck = 0
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response_fast(y, M_curr, solver)
        if b is None:
            stuck += 1
            if stuck > 50:
                break
            continue

        stuck = 0
        b = b / b.sum()
        n_iters += 1

        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = simplex_project(y_new)

        if momentum_t < 2.0:
            a_new = b.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = simplex_project(a_new)

        y = y_new
        a = a_new
        momentum_t += 1.0

        sc_y = fast_autocorr(y).max()
        if sc_y < best_score - 1e-14:
            best_score = sc_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        sc_a = fast_autocorr(a).max()
        if sc_a < best_score - 1e-14:
            best_score = sc_a
            best_a = a.copy()
            patience = 0

        if patience > 60 and n_iters % 30 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            if verbose:
                print(f"    LP restart at iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

        if verbose and n_iters % 50 == 0:
            print(f"    LP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a


def exponentiated_gradient(a0, time_limit, top_k=10, verbose=True):
    """Exponentiated gradient (multiplicative weights) with subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(42)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_val, conv = compute_subgradient_fast(a, top_k)
        # Normalize gradient
        g_norm = g.max() - g.min()
        if g_norm < 1e-15:
            g_norm = 1e-15
        g = g / g_norm

        # Multiplicative update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * g
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = simplex_project(a_new)

        # Adaptive eta
        sc_new = fast_autocorr(a_new).max()
        if sc_new > best_score + 0.1 * max_val:
            eta *= 0.7
            log_a_new = log_a - eta * g
            a_new = np.exp(log_a_new - log_a_new.max())
            a_new = simplex_project(a_new)
            sc_new = fast_autocorr(a_new).max()

        a = a_new
        n_iters += 1

        if sc_new < best_score - 1e-13:
            best_score = sc_new
            best_a = a.copy()
            patience = 0
            eta = min(eta * 1.05, 2.0)
        else:
            patience += 1
            if patience % 200 == 0:
                eta *= 0.95

        if patience > 500:
            noise = rng.laplace(0, 1.0/n, n)
            a_pert = a + noise
            a = simplex_project(a_pert)
            patience = 0
            eta = 1.0 / np.sqrt(n)

        if verbose and n_iters % 200 == 0:
            print(f"    EG iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}, eta={eta:.4f}")

    return best_a


def subgradient_nesterov(a0, time_limit, top_k=10, verbose=True):
    """Subgradient descent with Nesterov acceleration and simplex projection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    t_mom = 1.0
    n_iters = 0
    patience = 0
    eta0 = 2.0 / np.sqrt(n)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_val, conv = compute_subgradient_fast(y, top_k)
        g = g - np.mean(g)  # Center the gradient (since we're on the simplex)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        eta_t = eta0 / np.sqrt(it + 1)
        y_new = y - eta_t * g
        y_new = simplex_project(y_new)

        alpha = 2.0 / (t_mom + 2.0)
        a_new = (1.0 - alpha) * a + alpha * y_new
        if t_mom >= 2.0:
            momentum = (t_mom - 1.0) / (t_mom + 2.0)
            a_new = a_new + momentum * (a_new - a)
            a_new = simplex_project(a_new)

        a = a_new
        y = y_new
        t_mom += 1.0
        n_iters += 1

        sc = fast_autocorr(a).max()
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > 100:
            y = best_a.copy()
            a = best_a.copy()
            t_mom = 1.0
            patience = 0

        if verbose and n_iters % 500 == 0:
            print(f"    SG-N iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: move mass between pairs to reduce max autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve > 100:
            break

        perm = rng.permutation(n)
        improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-12:
                continue

            # Try moving a small amount from i to several nearby positions
            for offset in [1, -1, 2, -2, 3, -3, 5, -5, 10, -10]:
                j = (i + offset) % n
                if j == i:
                    continue
                # Try different fractions
                for frac in [0.1, 0.3, 0.5]:
                    d = a[i] * frac
                    if d < 1e-15:
                        continue
                    a[i] -= d
                    a[j] += d
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-14:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        no_improve = 0
                    else:
                        a[i] += d
                        a[j] -= d

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and n_passes % 20 == 0:
            print(f"    CD pass={n_passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a


def spectral_equalization(a0, time_limit, verbose=True):
    """Try to flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0

    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute DFT
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: flat power spectrum (equal power at all frequencies)
        # But DC component is fixed by sum(a)
        target_power = np.sum(power[1:]) / (len(power) - 1) if len(power) > 1 else 0
        target = np.copy(power)
        target[1:] = target_power

        # Interpolate: new amplitude = current * (target/current)^alpha
        alpha = 0.1
        ratio = target / (power + 1e-30)
        new_amps = np.abs(A) * np.power(ratio, alpha)
        phases = np.angle(A)
        A_new = new_amps * np.exp(1j * phases)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = simplex_project(a_new)

        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            n_iters += 1
        else:
            alpha *= 0.9
            if alpha < 0.001:
                break

        if verbose and n_iters % 50 == 0:
            print(f"    SE iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64).copy(), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load previous best
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = p / p.sum()
            p_score = evaluate_safe(p)
            print(f"Parent: n={len(p)}, score={p_score:.6f}")
            best_seq = p.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default uniform init: n={n_base}, score={best_score:.6f}")

    n_best = len(best_seq)

    # Phase 1: Exponentiated Gradient (fast, no LP) - 25% time
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.25
    print(f"\n=== Phase 1: Exponentiated Gradient n={n_best} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out = exponentiated_gradient(best_seq, t1, top_k=8, verbose=True)
        do_try_update(a_out, "eg1")

    # Phase 2: LP Best-Response - 30% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.30
    print(f"\n=== Phase 2: LP Best-Response n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out = lp_optimize(best_seq, t2, solver, verbose=True)
        do_try_update(a_out, "lp1")

    # Phase 3: Subgradient Nesterov - 15% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Subgradient Nesterov n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out = subgradient_nesterov(best_seq, t3, top_k=8, verbose=True)
        do_try_update(a_out, "sgn1")

    # Phase 4: Coordinate Descent - 15% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Coordinate Descent n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_out = coordinate_descent(best_seq, t4, verbose=True)
        do_try_update(a_out, "cd1")

    # Phase 5: LP Best-Response round 2 - 10% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: LP R2 n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out = lp_optimize(best_seq, t5, solver, verbose=True)
        do_try_update(a_out, "lp2")

    # Phase 6: Random restarts + quick optimization - remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t6 = remaining * 0.6
        print(f"\n=== Phase 6: Random Restarts n={n_best} (budget={t6:.0f}s) ===")
        for r in range(3):
            if time.monotonic() - t_global > time_budget - 10:
                break
            noise_scale = n_best * 0.02 * (r + 1)
            noise = rng.laplace(0, noise_scale, n_best)
            a_pert = best_seq + noise
            a_pert = simplex_project(a_pert)
            # Quick EG optimization
            t_sub = min(t6 / 3, time_budget - (time.monotonic() - t_global) - 5)
            if t_sub > 5:
                a_out = exponentiated_gradient(a_pert, t_sub, top_k=5, verbose=False)
                do_try_update(a_out, f"rr{r}")

    # Final CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_best = len(best_seq)
        t7 = min(remaining * 0.8, 60.0)
        print(f"\n=== Phase 7: Final CD n={n_best} (budget={t7:.0f}s) ===")
        a_out = coordinate_descent(best_seq, t7, verbose=False)
        do_try_update(a_out, "cd_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
