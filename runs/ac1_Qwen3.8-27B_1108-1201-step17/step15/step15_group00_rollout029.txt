    # For each column j, the non-zero entries are at rows k where 0 <= k-j < n
    # i.e., j <= k <= j+n-1, clamped to [0, 2n-2]
    rows = []
    cols = []

        k_end = min(2 * n - 2, j + n - 1)  # k <= j+n-1 (since k-j < n)

        cols.extend([j] * len(ks))
        vals.extend((y[ks - j]).tolist())

    k_max = min(top_k, len(conv_vals))
    top_idx = np.argsort(conv_vals)[::-1][:k_max]
    top_vals = np.maximum(conv_vals[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * y[j_idx]
    return grad
