```python
def lp_solve_fast(a, M, active_indices, verbose=False):
    """Fast LP solve using scipy HiGHS.
    Returns (b, duals_full) or (None, None)."""
    n = len(a)
    K = len(active_indices)
    if K == 0:
        return None, None
    
    # Build constraint matrix
    T = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            T[idx, lo:hi] = a[k-hi+1:k-lo+1][::-1]
    
    # Solve LP: max sum(b) s.t. T@b <= M, b >= 0
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=T, b_ub=np.full(K, M), 
                      bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None, None
    
    if not res.success:
        return None, None
    
    b = np.maximum(res.x, 0.0)
    
    # Get dual variables
    duals_full = np.zeros(2 * n - 1)
    try:
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            marginals = res.ineqlin.marginals
            if marginals is not None:
                duals_local = np.maximum(-marginals, 0.0)  # y = -marginal for <= constraints
                duals_full[active_indices] = duals_local
    except Exception:
        pass
    
    return b, duals_full
```
