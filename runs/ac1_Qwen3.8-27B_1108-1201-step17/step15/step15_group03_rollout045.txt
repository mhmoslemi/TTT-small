def fast_conv_self(a):
    """Compute autocorrelation conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

def lp_solve_constrained(a, M, solver, active_idx=None):
    """
    Solve LP: max sum(b) s.t. T[active] @ b <= M, b >= 0.
    If active_idx is None, use all constraints.
    Returns b or None on failure.
    """
    n = len(a)
    size = 2 * n - 1
    
    if active_idx is None:
        active_idx = np.arange(size)
    
    # Build the constraint matrix rows for active constraints
    # T[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # For efficiency, we build it as a sparse-looking dense matrix
    rows = len(active_idx)
    T = np.zeros((rows, n))
    for r, k in enumerate(active_idx):
        # T[k, i] = a[k-i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            T[r, i] = a[k - i]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

            active = np.where(conv_a >= M * 0.5)[0]

        b = lp_solve_constrained(a, M, solver, active)
        if b is None:
            # Try full constraint set
            b = lp_solve_constrained(a, M, solver, None)
            if b is None:
                return None
            return b
        
        # Check for violated constraints
        conv_ab = fast_conv_cross(a, b)
        violated = np.where(conv_ab > M * 1.001)[0]
        
        if len(violated) == 0:
            return b
        
        # Add violated constraints (most violated first)
        violated_val = conv_ab[violated]
        order = np.argsort(-violated_val)
        new_active = np.union1d(active, violated[order[:min(50, len(violated))]])
        
        if len(new_active) == len(active):
            # No progress, use all constraints
            b = lp_solve_constrained(a, M, solver, None)
            return b
        
        if np.array_equal(new_active, np.arange(size)):
            b = lp_solve_constrained(a, M, solver, None)
            return b
        
        active = new_active
    
    # Fallback to full
    return lp_solve_constrained(a, M, solver, None)
