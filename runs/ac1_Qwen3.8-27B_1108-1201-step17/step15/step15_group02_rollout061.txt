"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization:
1. LSE (Log-Sum-Exp) proximal gradient with Adam and adaptive temperature
2. LP best-response with constraint generation and Nesterov acceleration  
3. Efficient pair coordinate descent with FFT-based gradient computation
4. Multigrid downsample-upsample for escaping local minima
5. Final polishing with symmetrization and sparsification
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

# Global state
_solver = None
_best_global = None
_best_score_global = float('inf')

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def normalize(a, target_sum=None):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
        s = len(a)
    if target_sum is None:
        return a / s
    return a * target_sum / s

def lse_adam_optimize(a0, time_limit, verbose=False, T_start=100.0, T_end=0.01, n_stages=None):
    """LSE (Log-Sum-Exp) smoothed max optimization with Adam."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    if n_stages is None:
        n_stages = 30
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(300, int(remaining * 8)))
        lr = 0.3 / np.sqrt(n)
        # Adaptive LR: decrease as we go
        lr *= (0.5 + 0.5 * (1.0 - si / max(1, n_stages - 1)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # Smoothed max via LSE
            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -100.0, 0.0)
            exp_c = np.exp(shifted)
            w = exp_c / (exp_c.sum() + 1e-15)
            
            # Gradient of LSE max w.r.t. a
            # d/d a_i [max_k conv_k] ≈ sum_k w_k * (a_{k-i} + a_{i-k+n-1-i}) ... 
            # More precisely: grad_i = sum_k w_k * (T_k @ a)_i
            # where (T_k @ a)_i = a_{k-i} + a_{k-i+n} (with appropriate bounds)
            # Using FFT: grad = cross_corr(w, a) + cross_corr(a, w)
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            # cross-correlation of w and a: gives (w*a)_i for each shift
            ca = irfft(np.conj(W) * A, fft_size)[:n]  # w * a (cross)
            aw = irfft(A * W, fft_size)[:n]  # a * w (cross)
            
            grad = 2.0 * (ca + aw)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and (si % 5 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam stage {si}/{n_stages} T={T_val:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def mirror_descent_optimize(a0, time_limit, verbose=False):
    """Multiplicative weights (mirror descent on simplex) with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Adaptive step size
    eta = 0.1
    patience = 0
    best_eta = eta
    
    n_stages = 40
    etas = np.geomspace(0.2, 0.005, n_stages)
    
    for si, eta_cur in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(10, min(100, int(remaining * 5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break
            
            # Find the "pressure" - which elements contribute to the peak
            # Use all convolution positions weighted by their excess over threshold
            threshold = M * 0.8
            w = np.maximum(conv - threshold, 0.0)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = conv / (conv.sum() + 1e-15)
            else:
                w = w / w_sum
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(W) * A, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            pressure = np.maximum(pressure, 0.0)
            
            # Multiplicative weight update
            a = a * np.exp(-eta_cur * pressure)
            a = np.maximum(a, 1e-15)
            a = a * norm / a.sum()
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
        
        if verbose and si % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD stage {si}/{n_stages} eta={eta_cur:.5f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def build_active_constraints(a, conv, k=200):
    """Build the convolution matrix for the top-k most active constraints."""
    n = len(a)
    size = 2 * n - 1
    # Get top-k positions
    if k >= size:
        indices = np.arange(size)
    else:
        top_idx = np.argsort(conv)[-k:][::-1]
        indices = top_idx
    
    T = np.zeros((len(indices), n))
    for idx, kk in enumerate(indices):
        # Row kk of the convolution matrix: (T @ x)[kk] = sum_i x_i * a_{kk-i}
        # Non-zero when max(0, kk-n+1) <= i <= min(n-1, kk)
        lo = max(0, kk - n + 1)
        hi = min(n, kk + 1)
        for i in range(lo, hi):
            T[idx, i] = a[kk - i]
    
    return T

def lp_best_response(a, M, T_subset=None, solver=None):
    """Solve the LP: max sum(b) s.t. T @ b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if T_subset is not None and T_subset.shape[0] < 2 * n - 1:
        constraints = [T_subset @ b_var <= M]
    else:
        # Full constraint set (expensive)
        size = 2 * n - 1
        T = np.zeros((size, n))
        for kk in range(min(n, size)):  # only first n constraints for speed
            lo = max(0, kk - n + 1)
            hi = min(n, kk + 1)
            for i in range(lo, hi):
                T[kk, i] = a[kk - i]
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver if solver else cp.ECOS, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def lp_nesterov_optimize(a0, time_limit, solver, verbose=False, max_constraints=300, n_outer=None):
    """LP best-response with Nesterov acceleration and constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    y = a.copy()  # Nesterov momentum point
    
    if n_outer is None:
        n_outer = 100000
    
    for t in range(1, n_outer + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        threshold = M * 0.7
        active_idx = np.where(conv_y > threshold)[0]
        if len(active_idx) > max_constraints:
            active_idx = np.argsort(conv_y)[-max_constraints:]
        if len(active_idx) < 20:
            active_idx = np.argsort(conv_y)[-min(max_constraints, size):]
        
        T_sub = build_active_constraints(y, conv_y, k=len(active_idx))
        
        # Solve LP
        b = lp_best_response(y, M, T_sub, solver)
        
        if b is None:
            # Fallback: use a simple heuristic
            b = y.copy()
            b = b / b.sum() * norm
            continue
        
        # Check for violated constraints and regenerate
        check = fast_conv_cross(y, b)
        violated = np.where(check > M * 1.001)[0]
        if len(violated) > 0:
            # Add violated constraints and re-solve
            combined = np.unique(np.concatenate([active_idx, violated]))
            if len(combined) > max_constraints * 2:
                combined = np.argsort(check)[-max_constraints * 2:]
            T_sub2 = build_active_constraints(y, check, k=len(combined))
            b2 = lp_best_response(y, M, T_sub2, solver)
            if b2 is not None:
                b = b2
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t <= 2:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=False, n_passes=None):
    """Pair coordinate descent: transfer mass between pairs of positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    no_improve = 0
    total_improvements = 0
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        pass_count += 1
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue
            
            # Find positions where convolution is high
            peak_threshold = cur_max * 0.8
            peak_lags = np.where(conv >= peak_threshold)[0]
            
            # Candidate j's: positions that would reduce the peak if we move mass to them
            j_candidates = set()
            # Add some random
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            # Add positions related to peaks
            for kl in peak_lags[:8]:
                # If we move mass from i to j, the change in conv at position k is:
                # delta_conv_k = d*(a[k-j] + a[j-k] - a[k-i] - a[i-k]) (with bounds)
                # We want to minimize the max, so move to j where a[k-j]+a[j-k] is small
                # for all peak positions k
                j_candidates.add(int((kl - i + n) % n))
            
            j_candidates.discard(i)
            
            if not j_candidates:
                continue
            
            best_delta = 0.0
            best_new_max = cur_max
            best_j = -1
            
            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute the change in convolution
                # new_conv = conv + d*(a shifted to j - a shifted from i)
                # conv(a + d*(e_j - e_i), a + d*(e_j - e_i))
                # = conv + 2d*(a shifted) + d^2*(delta terms)
                
                # First-order change: 2d * (cross_corr at j - cross_corr at i)
                # where cross_corr at position k for source at i: a[k-i]
                
                # Compute the "pressure" at each position for moving from i to j
                # delta_conv_k = 2d * (a_{k-j} - a_{k-i}) + d^2 * (terms)
                
                # For simplicity, use first-order: find d that minimizes max(conv + d * grad)
                # where grad_k = 2*(a_{k-j} + a_{j-k} - a_{k-i} - a_{i-k})
                
                g = np.zeros(size)
                
                # Contribution from +d at position j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi-lo]
                
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                
                # Contribution from -d at position i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3-lo3]
                
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Binary search for optimal d
                d_max = min(ai, cur_max / (2.0 * (np.abs(g).max() + 1e-15)))
                d_max = max(d_max, ai * 0.5)
                
                n_samples = 15
                deltas = np.linspace(0.02, min(d_max, ai * 0.9), n_samples)
                
                for d in deltas:
                    new_conv = conv + d * g
                    # Second-order correction at diagonal
                    if 2 * i < size:
                        new_conv[2 * i] -= 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] += 2 * aj * d + d * d
                    # Cross term
                    if (i + j) < size:
                        new_conv[i + j] += 2 * d * (a[j] - a[i]) if i != j else 0
                        # Actually the cross term is: d^2 at position i+j
                        # Let me just use the linear approximation + diagonal correction
                    
                    v = new_conv.max()
                    if v < best_new_max:
                        best_new_max = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_new_max < cur_max - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        # Revert
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                # Perturb to escape local minimum
                n_perturb = max(10, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                no_improve = 0
            if no_improve >= 8:
                break
        else:
            no_improve = 0
        
        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={pass_count} ({elapsed:.0f}s): best={best_score:.8f}, imps={total_improvements}")
    
    return best_a, best_score

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s

def downsample(a, factor=2):
    """Downsample by averaging consecutive pairs."""
    n = len(a)
    if n % factor != 0:
        a = a[:n - n % factor]
    return a.reshape(-1, factor).mean(axis=1)

def upsample(a, target_n):
    """Upsample via linear interpolation."""
    old_n = len(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    a_up = np.interp(new_x, old_x, a)
    return np.maximum(a_up, 0.0)

def multigrid_optimize(a0, time_limit, verbose=False):
    """Multigrid: downsample, optimize coarsely, upsample, optimize finely."""
    n = len(a0)
    best_a = np.asarray(a0, dtype=np.float64).copy()
    best_score = compute_score(best_a)
    t0 = time.monotonic()
    
    # Downsample to coarse grid
    coarse_n = max(64, n // 4)
    a_coarse = downsample(best_a, factor=max(1, n // coarse_n))
    
    if verbose:
        print(f"    Multigrid: n={n} -> coarse_n={len(a_coarse)}")
    
    # Optimize on coarse grid (quick LSE-Adam)
    t_coarse = min(time_limit * 0.3, 60.0)
    remaining = time_limit - (time.monotonic() - t0)
    a_coarse, _ = lse_adam_optimize(a_coarse, min(t_coarse, remaining), verbose=False, T_start=50.0, T_end=0.1, n_stages=10)
    
    # Upsample back
    a_up = upsample(a_coarse, n)
    a_up = normalize(a_up)
    
    # Blend with original
    a_blend = 0.5 * normalize(best_a) + 0.5 * a_up
    a_blend = normalize(a_blend)
    
    sc = compute_score(a_blend)
    if sc < best_score:
        best_score = sc
        best_a = a_blend.copy()
    
    # Optimize the blended version
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 10:
        a_out, _ = lse_adam_optimize(a_blend, remaining, verbose=False, T_start=20.0, T_end=0.05, n_stages=15)
        sc = compute_score(a_out)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
    
    return best_a, best_score

def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 14)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.1, 0.4)
        c2 = rng.uniform(0.6, 0.9)
        a = np.exp(-50.0 * (x - c1) ** 2) + np.exp(-50.0 * (x - c2) ** 2) + 0.05
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.3
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(6 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
        a += 0.05
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * 2 * x) + 0.2 * np.sin(2 * np.pi * 5 * x)
    elif choice == 11:
        # Two-plateau
        a = np.where(x < 0.3, 1.0, np.where(x > 0.7, 1.0, 0.2))
    elif choice == 12:
        # Logarithmic decay from center
        d = np.abs(x - 0.5)
        a = np.exp(-3.0 * d)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    
    a = np.maximum(a, 0.0)
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    return normalize(a)

Wait, I need to double-check a few things:

1. The `build_active_constraints` function: The convolution matrix T has rows indexed by the convolution position k, and columns by the variable index i. The entry T[k, i] = a[k-i] if 0 <= k-i < n, else 0. This is correct because (a * b)_k = sum_i a_i * b_{k-i} = sum_i b_i * a_{k-i}.

Wait, I need to be more careful. The self-convolution is (a * a)_k = sum_i a_i * a_{k-i} where the sum is over valid i (0 <= i < n and 0 <= k-i < n). 

For the cross-convolution (a * b)_k = sum_i a_i * b_{k-i}.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. LSE (Log-Sum-Exp) smoothed gradient with Adam and geometric temperature annealing
2. Mirror descent (multiplicative weights) on the probability simplex
3. LP best-response with constraint generation and Nesterov acceleration
4. Pair coordinate descent with exact quadratic model per pair
5. Multigrid downsample-upsample for escaping local minima
6. Symmetrization and multi-length exploration
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def normalize(a, target_sum=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
        s = float(len(a))
    if target_sum is None:

def lse_adam_optimize(a0, time_limit, verbose=False, T_start=100.0, T_end=0.01, n_stages=None):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    if n_stages is None:
        n_stages = 30
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(300, int(remaining * 8)))
        lr = 0.3 / np.sqrt(n)
        lr *= (0.5 + 0.5 * (1.0 - si / max(1, n_stages - 1)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -100.0, 0.0)
            exp_c = np.exp(shifted)
            w = exp_c / (exp_c.sum() + 1e-15)

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(W) * A, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam stage {si}/{n_stages} T={T_val:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

def mirror_descent_optimize(a0, time_limit, verbose=False):

    a = normalize(a0, norm)

    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

        remaining = time_limit - (time.monotonic() - t0)

            if time.monotonic() - t0 > time_limit:

            M = conv.max()
            if M < 1e-15:

            if w_sum < 1e-15:

            pressure = np.maximum(ca + aw, 0.0)

            a = a * np.exp(-eta_cur * pressure)
            a = np.maximum(a, 1e-15)
            a = a * norm / a.sum()

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 7 ==
