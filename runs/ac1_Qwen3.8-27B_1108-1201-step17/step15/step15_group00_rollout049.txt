  - If k = 2i: -2*delta*a[i] + delta^2 + (change from j)

  - Generally: -2*delta*a[k-i] + 2*delta*a[k-j] (with appropriate validity checks)

  delta_conv[k] = -2*delta*a[k-i]*(validity of k-i) + 2*delta*a[k-j]*(validity of k-j)
  Plus quadratic terms if k=2i or k=2j.

  delta_conv[k] ≈ 2*delta * (a[k-j] - a[k-i]) (where a[x] = 0 if x not in [0,n-1])

  max_k (conv[k] + 2*delta*(a[k-j] - a[k-i]))

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient_exact(a, ac, top_k=30, beta=8.0, exact_weight=0.5):
    """Blend exact subgradient at argmax with soft top-k subgradient."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)

    # Soft weights
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]

    # Exact subgradient at argmax
    k_exact = int(np.argmax(ac))
    i_start = max(0, k_exact - n + 1)
    i_end = min(n, k_exact + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k_exact - idx
    exact_grad = np.zeros(n)
    exact_grad[idx] = 2.0 * a[j_idx]

    grad = exact_weight * exact_grad + (1.0 - exact_weight) * grad
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=6.0, c0=3.0, exact_weight=0.5, verbose=True):
    """Nesterov-accelerated mirror descent with adaptive subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    x_prev = a.copy()

    max_stall = 3000

        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient_exact(y, ac, top_k=top_k, beta=beta, exact_weight=exact_weight)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.005, n) * best_a
            x_restarted = best_a + perturbation
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            if verbose:
                elapsed = time.monotonic() - t0

            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}")

def lp_direction(a, M, solver, active_frac=0.05):
    """Solve LP to find best direction: max sum(b) s.t. b>=0, (a*b)[k]<=M for active k."""
    import cvxpy as cp
    n = len(a)
    ac = fast_autocorr(a)
    threshold = M * (1.0 - active_frac)
    active_k = np.where(ac >= threshold)[0]

    if len(active_k) == 0:
        active_k = np.argsort(ac)[::-1][:50]

    b_var = cp.Variable(n, nonneg=True)

    # Only include active constraints
    T = np.zeros((len(active_k), n))
    for row, k in enumerate(active_k):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for col, i in enumerate(range(i_start, i_end)):
            T[row, i] = a[k_int - i]

    constraints = [T @ b_var <= M, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

    """LP-based Frank-Wolfe with Nesterov acceleration."""

def coord_delta_conv(a, i, j, delta, n):
    """Compute the change in autocorrelation when moving 'delta' from position i to position j.
    Returns the delta_conv array (approximate, linear in delta)."""
    dc = np.zeros(2 * n - 1)
    # Change from removing delta from a[i]:
    # dc[k] += -2 * delta * a[k-i] (if k-i valid and k != 2i)
    # dc[k] += -2*delta*a[i] + delta^2 (if k == 2i)
    for k in range(2 * n - 1):
        ki = k - i
        if 0 <= ki < n:
            if k == 2 * i:
                dc[k] += -2.0 * delta * a[i] + delta * delta
            else:
                dc[k] += -2.0 * delta * a[ki]
        kj = k - j
        if 0 <= kj < n:
            if k == 2 * j:
                dc[k] += 2.0 * delta * a[j] + delta * delta
            else:
                dc[k] += 2.0 * delta * a[kj]
    return dc


def fast_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Fast coordinate descent using delta_conv formula for O(n) evaluation per move."""
    n = len(a0)
    a = norm_sum1(a0)

    best_m = fast_autocorr(a).max()

        M_curr = ac.max()

        in_peak = np.zeros(n, dtype=bool)

            s = max(0, k_int - n + 1)
            e = min(n - 1, k_int)
            in_peak[s:e + 1] = True

        sources = np.where(in_peak)[0]
        destinations = np.where(~in_peak)[0]

        if len(sources) == 0 or len(destinations) == 0:
            stall += 1
            break

        # Try moves from each source to best destination
        n_try = min(len(sources), 50)
        src_perm = rng.permutation(len(sources))[:n_try]

        for si in src_perm:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(sources[si])
            if a[src] < 1e-12:
                continue

            # Find best destination
            best_dst = -1
            best_score = float('inf')
            n_dest = min(len(destinations), 200)
            dest_subset = destinations[rng.permutation(len(destinations))[:n_dest]]

            for dst in dest_subset:
                dst = int(dst)
                # Estimate the max after move using delta_conv (linear approx)
                # delta_conv[k] = 2*delta*(a[k-dst] - a[k-src])
                # New max ≈ max_k (ac[k] + 2*delta*(a[k-dst] - a[k-src]))
                # We want to minimize this max.
                # For the top peaks, the change is:
                #   2*delta*(a[peak-dst] - a[peak-src])
                # We want largest negative change at the max peak.

                # Score: how much does the peak decrease?
                score = 0.0
                for k in top_k[:5]:
                    k_int = int(k)
                    a_at_dst = a[k_int - dst] if 0 <= k_int - dst < n else 0.0
                    a_at_src = a[k_int - src] if 0 <= k_int - src < n else 0.0
                    score += ac[k] * 2.0 * (a_at_src - a_at_dst)

                    best_score = score
                    best_dst = dst

                continue

                a_trial[src] -= mid
                a_trial[dst] += mid
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_new_m:
                    best_new_m = M_trial
                    best_delta = mid
                    lo = mid
                else:
                    hi = mid

            if best_delta > 0 and best_new_m < best_m - 1e-15:
                a[src] -= best_delta
                a[dst] += best_delta
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                ac = fast_autocorr(a)

                M_new = fast_autocorr(a).max()

                    ac = fast_autocorr(a)

            J = 2.0 * n * best_m

def generate_initializations(n, rng):
    """Generate diverse initializations for a sequence of length n."""
    x = np.linspace(0, 1, n)
    inits = {}

    # Sine squared
    s = np.sin(np.pi * x) ** 2
    inits['sine2'] = s / s.sum()

    # Cosine fourth
    c = np.cos(np.pi * (x - 0.5)) ** 4
    inits['cos4'] = c / c.sum()

    # Gaussian
    g = np.exp(-20 * (x - 0.5) ** 2)
    inits['gauss'] = g / g.sum()

    # Bimodal
    b = np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2)
    inits['bimodal'] = b / b.sum()

    # U-shape (edges)
    u = x ** 4 + (1 - x) ** 4
    inits['u_shape'] = u / u.sum()

    # Half (mass in first half, tapering)
    h = np.zeros(n)
    h[:n // 2] = np.linspace(1, 0, n // 2)
    h = np.maximum(h, 0)
    if h.sum() > 0:
        inits['half'] = h / h.sum()

    # Random smooth
    r = rng.random(n)
    # Smooth it
    kernel = np.ones(20) / 20
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0)
    if r_smooth.sum() > 0:
        inits['random_smooth'] = r_smooth / r_smooth.sum()

    # Triangular
    tri = 1 - np.abs(2 * x - 1)
    tri = np.maximum(tri, 0)
    inits['tri'] = tri / tri.sum()

    # Sine^p with different p
    for p in [3, 5]:
        sp = np.sin(np.pi * x) ** p
        inits[f'sine{p}'] = sp / sp.sum()

    return inits

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        print("No parent available, using default init")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}, target=1.5027")

    # === Phase 1: AMD at parent length with multiple hyperparameter settings ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD n={n0} ({remaining*0.35:.0f}s budget) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        # Run AMD with different params
        t_per = t1 / 3.0
        configs = [
            {'top_k': 50, 'beta': 5.0, 'c0': 4.0, 'exact_weight': 0.3},
            {'top_k': 30, 'beta': 8.0, 'c0': 3.0, 'exact_weight': 0.6},
            {'top_k': 70, 'beta': 4.0, 'c0': 5.0, 'exact_weight': 0.5},
        ]
        for ci, cfg in enumerate(configs):
            t_remaining = t1 - (time.monotonic() - t_global + (time.monotonic() - t_global - t_global))  # just use t_per
            t_use = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_use < 10:
                break
            a_out, _ = amd_optimize(a_in, t_use, verbose=(ci == 0), **cfg)
            if try_update(a_out, f"amd_c{ci}_n{n0}"):
                a_in = norm_sum1(best_seq)

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP n={len(best_seq)} ({remaining*0.2:.0f}s budget) ===")
    n_best = len(best_seq)
    if n_best <= 2500:
        t2 = min(remaining * 0.2, 300.0)
        if t2 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, active_frac=0.03)
            try_update(a_out, f"lp_n{len(best_seq)}")

    # === Phase 3: Fast coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord n={len(best_seq)} ({remaining*0.25:.0f}s budget) ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fast_coordinate_descent(a_in, t3, verbose=True, rng_seed=42)
        if try_update(a_out, f"coord_n{len(best_seq)}"):
            a_in = norm_sum1(best_seq)
        # Run again with different seed
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.15, 200.0)
        if t3b > 20:
            a_out, _ = fast_coordinate_descent(a_in, t3b, verbose=True, rng_seed=999)
            try_update(a_out, f"coord2_n{len(best_seq)}")

    # === Phase 4: Group coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: GroupCoord n={len(best_seq)} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = group_coordinate_descent(a_in, t4, verbose=True, rng_seed=123)
        try_update(a_out, f"groupcoord_n{len(best_seq)}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length ===")
    n_best = len(best_seq)
    n_candidates = []
    for dn in [-400, -200, 200, 400, 800]:
        nc = n_best + dn
        if 800 <= nc <= 5000:
            n_candidates.append(nc)

    for nc in n_candidates[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={nc}")
        a_long = interpolate_sequence(best_seq, nc)
        # Quick AMD
        t5 = min(remaining * 0.15, 120.0)
        if t5 > 10:
            a_in = norm_sum1(a_long)
            a_out, _ = amd_optimize(a_in, t5, top_k=40, beta=5.0, c0=3.0, exact_weight=0.5, verbose=False)
            try_update(a_out, f"amd_n{nc}")

    # === Phase 6: Diverse initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Diverse inits ===")
    if remaining > 100:
        n_best = len(best_seq)
        inits = generate_initializations(n_best, rng)
        # Evaluate all initializations
        for name, init in inits.items():
            sc = evaluate_safe(init)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = init.copy()
                print(f"  Init {name}: {sc:.6f} (new best)")
            elif sc < best_score + 0.05:
                print(f"  Init {name}: {sc:.6f}")

        # Run AMD on the best 2 diverse inits
        init_scores = [(evaluate_safe(v), k) for k, v in inits.items()]
        init_scores.sort()
        for name, init in init_scores[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t6 = min(remaining * 0.3, 100.0)
            if t6 > 10:
                a_out, _ = amd_optimize(init, t6, top_k=50, beta=5.0, c0=4.0, exact_weight=0.4, verbose=False)
                if try_update(a_out, f"amd_{name}"):
                    pass

    # === Phase 7: Final AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final AMD polish n={len(best_seq)} ===")
    t7 = min(remaining * 0.3, 200.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t7, top_k=60, beta=4.0, c0=3.5, exact_weight=0.7, verbose=True)
        try_update(a_out, f"amd_final_n{len(best_seq)}")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final coord n={len(best_seq)} ===")
    t8 = min(remaining * 0.4, 150.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fast_coordinate_descent(a_in, t8, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_final_n{len(best_seq)}")

    # === Phase 9: Final LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2500 and remaining > 30:
        print(f"\n=== Phase 9: Final LP n={n_best} ===")
        t9 = min(remaining * 0.5, 120.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t9, solver, verbose=True, active_frac=0.02)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 10: Ultra-fine coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 10: Ultra-fine coord ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = fast_coordinate_descent(a_in, min(remaining * 0.7, 100.0), verbose=True, rng_seed=2024)
        try_update(a_out, f"ultra_coord")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
