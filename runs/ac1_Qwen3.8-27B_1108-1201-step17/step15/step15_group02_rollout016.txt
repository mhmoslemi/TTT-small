def fast_conv_self(a):
    """Compute full auto-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Compute full cross-convolution of a and b using FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_max_conv(a):
    """Compute max of auto-convolution."""
    return fast_conv_self(a).max()

def eval_fn(a):

    if S < 0.01:
        return float('inf')
    n = len(a)

def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE-Adam optimization with temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)

            if max_c < 1e-15:

def solve_lp_cutting_plane(a, M, max_inner=20, verbose=False):
    """
    Solve the LP:
    maximize sum(b)
    subject to (a*b)[k] <= M for all k
               b >= 0
    using cutting plane method.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Start with a small active set (most binding constraints)
    conv_a = fast_conv_self(a)
    # Use top 30 constraints
    active = np.argsort(conv_a)[-min(30, size):]
    
    b = None
    for inner_iter in range(max_inner):
        # Build constraint matrix for active constraints
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active), n))
        for idx, k in enumerate(active):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[idx, i] = a[k - i]
        
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                return None
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        if b_val is None:
            return None
        b = np.maximum(b_val, 0.0)
        
        # Check for violations
        conv_ab = fast_conv_cross(a, b)
        violated = np.where(conv_ab > M * 1.001)[0]
        
        if len(violated) == 0:
            return b
        
        # Add violated constraints (up to 50 new ones)
        # Prioritize most violated
        viol_amount = conv_ab[violated]
        order = np.argsort(viol_amount)[::-1][:50]
        new_constraints = violated[order]
        active = np.unique(np.concatenate([active, new_constraints]))
        
        # If active set gets too large, keep only most violated
        if len(active) > 200:
            conv_ab_active = conv_ab[active]
            keep_order = np.argsort(conv_ab_active)[::-1][:200]
            active = active[keep_order]
    
    return b

        conv_ab = fast_conv_cross(a, b)
        violations = conv_ab - M
        if violations.max() <= 1e-10:
            break  # Feasible

        k_max = violations.argmax()

        # Contribution of b[i] to (a*b)[k_max] is a[k_max - i]
        contrib = np.zeros(n)
        lo = max(0, k_max - n + 1)

import numpy as np

from scipy.fft import rfft, irfft


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_max_conv(a):
    return fast_conv_self(a).max()


def eval_fn(a):

        n = len(a)

        return float(2.0 * n * max_b / (S * S))

            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            lr = base_lr / (1.0 + 0.1 * it)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = compute_max_conv(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def solve_lp_single(a, M, active_set, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        row_start = max(0, k - n + 1)
        row_end = min(n, k + 1)
        for i in range(row_start, row_end):
            T[idx, i] = a[k - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def solve_lp_cutting_plane(a, M, solver, max_inner=8):
    n = len(a)
    size = 2 * n - 1
    conv_a = fast_conv_self(a)
    n_active = min(50, size)
    active = np.argsort(conv_a)[-n_active:]

    b = None
    for inner_iter in range(max_inner):
        b = solve_lp_single(a, M, active, solver)
        if b is None:
            return None

        conv_ab = fast_conv_cross(a, b)
        violated = np.where(conv_ab > M * 1.001)[0]

        if len(violated) == 0:
            return b

        viol_amount = conv_ab[violated]
        order = np.argsort(viol_amount)[::-1][:30]
        new_c = violated[order]
        active = np.unique(np.concatenate([active, new_c]))

        if len(active) > 300:
            cva = conv_ab[active]
            keep = np.argsort(cva)[::-1][:300]
            active = active[keep]

    return b


def lp_best_response_waterfill(a, M, n_iters=500):
    n = len(a)
    b = np.ones(n)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > 2.0:
            break
        conv_ab = fast_conv_cross(a, b)
        violations = conv_ab - M
        max_v = violations.max()
        if max_v <= 1e-10:
            break

        k_max = int(violations.argmax())
        contrib = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n, k_max + 1)
        for i in range(lo, hi):
            contrib[i] = a[k_max - i]

        if contrib.max() > 0:
            total_contrib = contrib @ b
            if total_contrib > 1e-15:
                scale = max_v / (total_contrib + 1e-15) * 0.7
                b[lo:hi] -= scale * contrib[lo:hi]

        b = np.maximum(b, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n)
        else:
            b = b / b_sum * n

    return b


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    y = a.copy()
    no_improve = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        b = solve_lp_cutting_plane(y, M, solver, max_inner=6)
        if b is None:
            b = lp_best_response_waterfill(y, M, n_iters=200)

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        beta = (t - 1.0) / (t + 2.0) if t >= 2 else 0.0
        a_new = y_new + beta * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = compute_max_conv(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        sc_a = compute_max_conv(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Ne t={t} ({elapsed:.0f}s): score={best_score:.8f}, M={M:.6f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    y = a.copy()
    v = np.zeros(n)
    t_adam = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-10

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()

        threshold = M * 0.995
        active = np.where(conv_y >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_y)])

        weights = conv_y[active] - threshold
        weights = weights / (weights.sum() + 1e-15)

        grad = np.zeros(n)
        for idx, k in enumerate(active):
            w = 2.0 * weights[idx]
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                grad[i] += w * y[k - i]

        t_adam += 1
        v = beta1 * v + (1.0 - beta1) * grad
        v_hat = v / (1.0 - beta1 ** t_adam)

        lr = 2.0 * norm / ((t + 2.0) * (np.linalg.norm(grad) + eps))
        y = y - lr * v_hat
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y * norm / s_y

        if t >= 2:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y + beta * (y - a)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new * norm / s_a
            a = a_new
        else:
            a = y.copy()

        sc = compute_max_conv(y)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def vegas_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 20
    etas = np.geomspace(0.08, 0.003, n_stages)

    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(15, min(100, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break

            w = np.maximum(conv - M * 0.85, 0.0)
            ww = w / (w.sum() + 1e-15)

            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw

            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-12)
            a = a * norm / a.sum()

            sc = compute_max_conv(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 5 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.5f}: best={best_score:.8f}")

    return best_a, best_score


def sparsify(a, threshold_frac=0.0005):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a_norm = a / s
    threshold = threshold_frac * np.max(a_norm)
    mask = a_norm > threshold
    if np.sum(mask) < n // 5:
        mask = a_norm > threshold * 0.5
    a_s = np.where(mask, a, 0.0)
    s_new = a_s.sum()
    if s_new < 1e-15:
        return a
    return a_s / s_new * (s / s_new)


def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym * (a.sum() / s)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.45)
        c2 = rng.uniform(0.55, 0.8)
        a = np.exp(-40.0 * (x - c1) ** 2) + np.exp(-40.0 * (x - c2) ** 2) + 0.1
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 4
    elif choice == 5:
        a = 0.2 + 0.8 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(6 * np.pi * x) * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 8:
        a = np.sinc(x - 0.5) ** 2 + 0.15
    else:
        a = 0.4 + 0.4 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}, solver={solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LSE-Adam for quick initial progress
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{n_base}")

    # Phase 2: Vegas multiplicative updates
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Vegas n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = vegas_optimize(best_seq, t2, verbose=True)
        check_and_update(a_out, f"vegas_n{len(best_seq)}")

    # Phase 3: LP cutting-plane with Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t3, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{len(best_seq)}")

    # Phase 4: Subgradient polishing
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Subgradient n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = subgradient_polish(best_seq, t4, verbose=True)
        check_and_update(a_out, f"sg1_n{len(best_seq)}")

    # Phase 5: LP round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: LP round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = lp_nesterov_optimize(best_seq, t5, solver, verbose=False)
        check_and_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 6: LSE-Adam round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 6: LSE-Adam round 2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = lse_adam_optimize(best_seq, t6, verbose=False)
        check_and_update(a_out, f"lse2_n{len(best_seq)}")

    # Phase 7: Sparsification and symmetrization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Sparsify/Symmetrize n={len(best_seq)} ===")
    a_spar = sparsify(best_seq, 0.0005)
    check_and_update(a_spar, f"sparse_n{len(best_seq)}")
    a_sym = symmetrize(best_seq)
    check_and_update(a_sym, f"sym_n{len(best_seq)}")
    a_spar2 = sparsify(best_seq, 0.001)
    check_and_update(a_spar2, f"sparse2_n{len(best_seq)}")

    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        print(f"\n=== Phase 8: Multi-length (remaining={remaining:.0f}s) ===")
        lengths_to_try = [n_base + 200, n_base - 200, n_base + 400, n_base + 600, n_base - 400]
        for n_try in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            if n_try < 200 or n_try > 4000:
                continue
            t_try = min(remaining * 0.30, 50.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try, 25.0), verbose=False)
            check_and_update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t_lp = min(remaining * 0.4, 25.0)
                if t_lp > 15:
                    a_out, _ = lp_nesterov_optimize(a_out, t_lp, solver, verbose=False)
                    check_and_update(a_out, f"ml_lp_n{n_try}")

    # Phase 9: Fresh starts
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 9: Fresh starts (remaining={remaining:.0f}s) ===")
        for attempt in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 25:
                break
            n_fresh = n_base + int(rng.integers(-300, 500))
            if n_fresh < 300:
                n_fresh = 800
            if n_fresh > 3500:
                n_fresh = 3000
            a_fresh = structured_init(n_fresh, rng)
            t_fresh = min(remaining * 0.25, 18.0)
            a_out, _ = lse_adam_optimize(a_fresh, t_fresh, verbose=False)
            check_and_update(a_out, f"fresh{attempt}_n{n_fresh}")

    # Final: subgradient polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final Polish (remaining={remaining:.0f}s) ===")
        t_final = min(remaining * 0.7, 60.0)
        a_out, _ = subgradient_polish(best_seq, t_final, verbose=False)
        check_and_update(a_out, "final_sg")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
