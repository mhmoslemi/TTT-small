"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based gradient descent
with adaptive top-K peak smoothing, momentum, coordinate descent polish,
multi-restart with perturbation, and multi-length exploration.
Key innovation: replaces slow LP solver with fast gradient-based method
allowing 100x more iterations in the same time budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_topk_gradient(a, conv, K, beta):
    """Compute smoothed gradient using top-K peaks of convolution."""
    n = len(a)
    size = 2 * n - 1
    M = conv.max()

    if K >= size:
        idx = np.arange(size)
        exp_vals = np.exp(beta * (conv - M))
    else:
        part = np.argpartition(conv, -K)[-K:]
        idx = part[np.argsort(conv[part])[::-1]]
        exp_vals = np.exp(beta * (conv[idx] - M))

    w_sum = exp_vals.sum()
    if w_sum < 1e-30:
        exp_vals = np.ones(len(idx)) / len(idx)
    else:
        exp_vals = exp_vals / w_sum

    g = np.zeros(n)
    for kk in range(len(idx)):
        k = int(idx[kk])
        w = float(exp_vals[kk])
        if w < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        j_lo = k - i_hi + 1
        j_hi = k - i_lo + 1
        g[i_lo:i_hi] += 2.0 * w * a[j_lo:j_hi][::-1]

    return g

    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))

            f2 = fast_autocorr(a_t2).max()

            for _ in range(25):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1.0 - phi) * (hi - lo)
                    a_t1 = a.copy()
                    a_t1[i] = c1
                    a_t1 = a_t1 * norm / a_t1.sum()
                    f1 = fast_autocorr(a_t1).max()
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a_t2 = a.copy()
                    a_t2[i] = c2
                    a_t2 = a_t2 * norm / a_t2.sum()
                    f2 = fast_autocorr(a_t2).max()

            if f1 < f2:
                a_try = a_t1
                sc = f1
            else:
                a_try = a_t2
                sc = f2

            if sc < best_M - 1e-12:
                a = a_try
                best_M = sc
                best_a = a.copy()
                pass_improved = True
                no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and no_improve == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD ({elapsed:.0f}s): M={best_M:.10f}")

    return best_a, best_M


def block_cd(a0, time_limit, seed=0, block_size=64, verbose=False):
    """Block coordinate descent: optimize groups of coordinates together."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    conv = fast_autocorr(a)
    best_M = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_blocks = (n + block_size - 1) // block_size

    for outer in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        order = rng.permutation(n_blocks)

        for bi in order:
            if time.monotonic() - t0 > time_limit:
                break

            i_start = bi * block_size
            i_end = min(n, (bi + 1) * block_size)
            block = a[i_start:i_end].copy()
            block_sum = block.sum()
            rest_sum = s - block_sum

            best_local = fast_autocorr(a).max()
            best_block = block.copy()

            for inner in range(30):
                conv_cur = fast_autocorr(a)
                K_local = min(50, 2 * n - 1)
                beta_local = 5.0
                grad_full = compute_topk_gradient(a, conv_cur, K_local, beta_local)
                grad_block = grad_full[i_start:i_end]

                lr_b = 0.1 / (1.0 + 0.1 * inner)
                new_block = block - lr_b * grad_block
                new_block = np.maximum(new_block, 0.0)

                a_try = a.copy()
                a_try[i_start:i_end] = new_block
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try * norm / s_try
                new_M = fast_autocorr(a_try).max()

                if new_M < best_local - 1e-14:
                    best_local = new_M
                    best_block = new_block.copy()
                    a = a_try
                    improved = True

            a[i_start:i_end] = best_block
            a = a * norm / a.sum()

        if not improved:
            break

        if best_local < best_M - 1e-12:
            best_M = best_local
            best_a = a.copy()

        if verbose and outer % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      BCD iter={outer} ({elapsed:.0f}s): M={best_M:.10f}")

    return best_a, best_M


def structured_init(n, style=0):
    """Generate structured initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def propose_candidate():
    """Main search function: multi-phase gradient-based optimization."""
    t_global = time.monotonic()
    time_budget = 1750.0

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_sequence([float(x) for x in a])
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, delta={old_score - sc:.6f})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, local_score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        for sty in range(8):
            cand = structured_init(n_init, sty)
            sc = evaluate_local(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Main gradient descent (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: Gradient Descent n={n_base}, time={t1:.0f}s ===")
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = grad_descent(a_norm, t1, verbose=True, seed=42)
    try_update(a_out, "gd1")

    # === Phase 2: Block CD polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: Block CD n={n_cur}, time={t2:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = block_cd(a_norm, t2, seed=1, verdict=False if False else False)
        try_update(a_out, "bcd1")

    # === Phase 3: Coordinate descent (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Coord Descent n={n_cur}, time={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent_full(a_norm, t3, seed=2, verbose=True)
        try_update(a_out, "cd1")

    # === Phase 4: Gradient descent round 2 with different params (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    if t4 > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: GD Round 2 n={n_cur}, time={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = grad_descent(a_norm, t4, verbose=True, seed=123,
                                K_start=200, K_end=800,
                                beta_start=5.0, beta_end=40.0,
                                lr0=0.3, momentum=0.9)
        try_update(a_out, "gd2")

    # === Phase 5: Multi-restart with perturbation (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5_total = remaining * 0.15
    n_restarts = 4
    t5_each = t5_total / n_restarts
    print(f"\n=== Phase 5: Multi-restart ({n_restarts} restarts) ===")
    rng = np.random.default_rng(999)
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        t5 = min(t5_each, remaining - 50)
        if t5 < 30:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.015 * r
        pert = best_seq * (1.0 + scale * rng.standard_normal(n_cur))
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum()
        a_norm = pert * np.sqrt(2.0 * n_cur)
        print(f"  Restart {r+1}: scale={scale:.3f}")
        a_out, _ = grad_descent(a_norm, t5, verbose=False, seed=500 + r * 17,
                                K_start=150, K_end=500,
                                beta_start=3.0, beta_end=20.0,
                                lr0=0.4, momentum=0.85)
        try_update(a_out, f"pert{r}")

    # === Phase 6: Try longer length (7% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.07, 120.0)
    n_long = n_base + 400
    if t6 > 40:
        print(f"\n=== Phase 6: Longer length n={n_long}, time={t6:.0f}s ===")
        init_long = interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long)
        a_out, _ = grad_descent(a_norm, t6, verbose=True, seed=777)
        try_update(a_out, f"long_n{n_long}")
        # CD polish on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_norm2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out2, _ = coord_descent_full(a_norm2, min(remaining * 0.3, 60), seed=888)
            try_update(a_out2, "cd_long")

    # === Phase 7: Try shorter length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(1000, n_base - 200)
    t7 = min(remaining * 0.05, 80.0)
    if n_short != n_base and t7 > 30:
        print(f"\n=== Phase 7: Shorter length n={n_short}, time={t7:.0f}s ===")
        init_short = interpolate_seq(best_seq, n_short)
        a_norm = init_short * np.sqrt(2.0 * n_short)
        a_out, _ = grad_descent(a_norm, t7, verbose=False, seed=333)
        try_update(a_out, f"short_n{n_short}")

    # === Phase 8: Final gradient descent on best (use remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.5, 200.0)
    if t8 > 30:
        print(f"\n=== Phase 8: Final GD n={n_best}, time={t8:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = grad_descent(a_norm, t8, verbose=True, seed=2024,
                                K_start=300, K_end=1000,
                                beta_start=8.0, beta_end=50.0,
                                lr0=0.2, momentum=0.92)
        try_update(a_out, "gd_final")

    # === Phase 9: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.5, 100.0)
    if t9 > 20:
        print(f"\n=== Phase 9: Final CD n={n_best}, time={t9:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent_full(a_norm, t9, seed=4242, verbose=True)
        try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
