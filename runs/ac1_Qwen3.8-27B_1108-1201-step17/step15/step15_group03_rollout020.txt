        if f1 < f2:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE gradient descent for warmup
2. LP best-response with Nesterov acceleration (main workhorse)
3. Grid-based analytical coordinate descent for polishing

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a, _cache={}):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b, _cache={}):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    """Build the convolution matrix T such that T @ b = a * b."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None):
    """Solve: max sum(b) s.t. T_a @ b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent for fast initial progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(60.0, 0.3, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(40, min(400, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, adapt_M=False):
    """LP best-response with Nesterov acceleration and adaptive constraint screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Adaptive M reduction factor
    M_reduction = 0.999 if adapt_M else 1.0
    current_M_target = best_score

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()

        if adapt_M:
            # Try to tighten M slightly
            current_M_target = min(current_M_target, M * M_reduction)
            M_eff = max(current_M_target, M * 0.999)
        else:
            M_eff = M

        if M_eff < 1e-12:
            break

        # Adaptive constraint screening - use wider active set
        threshold_1 = M_eff * 0.85
        active = np.where(conv_y >= threshold_1)[0]
        if len(active) < 50:
            threshold_2 = M_eff * 0.55
            active = np.where(conv_y >= threshold_2)[0]

        b = lp_solve(y, M_eff, solver, active)
        if b is None:
            b = lp_solve(y, M_eff, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M_eff * 1.002)[0]
            if len(violated) > 5:
                b_full = lp_solve(y, M_eff, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, rng, verbose=True, n_pairs_per_sweep=5):
    """Grid-based coordinate descent: move mass from i to j, optimize delta via grid."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    n_grid = 64  # grid points for delta search

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Find good candidate j's based on peak structure
            peak_threshold = cur_max * 0.90
            peak_lags = np.where(conv >= peak_threshold)[0]

            j_candidates = set()
            # Random candidates
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            # Structured candidates from peaks
            for kl in peak_lags[:6]:
                j_s = (kl - i + n) % n
                if 0 <= j_s < n and j_s != i:
                    j_candidates.add(j_s)
                j_s2 = (i - kl + n) % n
                if 0 <= j_s2 < n and j_s2 != i:
                    j_candidates.add(j_s2)
                j_s3 = (kl + i) % n
                if 0 <= j_s3 < n and j_s3 != i:
                    j_candidates.add(j_s3)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:n_pairs_per_sweep + 8]:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector: how b changes per unit delta moved from i to j
                # b[k] += 2d*(a[k-j] - a[k-i]) + quadratic corrections at 2i, i+j, 2j
                s_vec = np.zeros(size)
                # a[k-j] for k in [0, 2n-2]: valid when 0 <= k-j < n
                lo1 = max(0, j)
                hi1 = min(size, j + n)
                if hi1 > lo1:
                    s_vec[lo1:hi1] += a[0:hi1 - lo1]
                # a[k-i]
                lo2 = max(0, i)
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    s_vec[lo2:hi2] -= a[0:hi2 - lo2]
                s_vec *= 2.0

                # Quadratic corrections
                quad = np.zeros(size)
                if 2 * i < size:
                    quad[2 * i] = 1.0
                if i + j < size:
                    quad[i + j] = -2.0
                if 2 * j < size:
                    quad[2 * j] = 1.0

                # Grid search over delta
                max_delta = ai * 0.95
                if max_delta < 1e-12:
                    continue
                deltas = np.linspace(0.0, max_delta, n_grid)
                for d in deltas[1:]:
                    new_b = conv + d * s_vec + d * d * quad
                    v = new_b.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                # Perturb to escape local minimum
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 300 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_fine(a0, time_limit, rng, verbose=True):
    """Finer coordinate descent: try all pairs with ternary search on delta."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)

        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    def ternary_min(b, s_vec, quad, d_max):
        """Find d in [0, d_max] minimizing max(b + d*s_vec + d^2*quad) via ternary search."""
        if d_max < 1e-12:
            return 0.0
        lo, hi = 0.0, d_max
        # Check endpoints
        f_lo = b.max()
        f_hi = (b + hi * s_vec + hi * hi * quad).max()
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = (b + m1 * s_vec + m1 * m1 * quad).max()
            f2 = (b + m2 * s_vec + m2 * m2 * quad).max()
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        d_opt = (lo + hi) / 2.0
        f_opt = (b + d_opt * s_vec + d_opt * d_opt * quad).max()
        if f_opt >= f_lo:
            return 0.0
        return d_opt

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)

                break

            ai = a[i]

                continue

                lo1 = max(0, j)
                hi1 = min(size, j + n)
                if hi1 > lo1:
                    s_vec[lo1:hi1] += a[0:hi1 - lo1]
                lo2 = max(0, i)
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    s_vec[lo2:hi2] -= a[0:hi2 - lo2]
                s_vec *= 2.0

                quad = np.zeros(size)
                if 2 * i < size:
                    quad[2 * i] = 1.0
                if i + j < size:
                    quad[i + j] = -2.0
                if 2 * j < size:
                    quad[2 * j] = 1.0

                d_opt = ternary_min(conv, s_vec, quad, ai * 0.98)
                if d_opt > 0:
                    new_max = (conv + d_opt * s_vec + d_opt * d_opt * quad).max()
                    if new_max < best_val:
                        best_val = new_max
                        best_delta = d_opt
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()

                    pass_improved = True

                if s_new > 1e-15:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def momentum_subgradient(a0, time_limit, verbose=True):
    """Momentum subgradient with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    momentum = np.zeros(n)
    beta = 0.85
    lr0 = 0.5 / np.sqrt(n)

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Adaptive temperature: start high, decrease over time
        temperature = max(0.05, max_c * 0.08 / (1.0 + 0.005 * it))
        shifted = (conv - max_c) / temperature
        shifted = np.maximum(shifted, -50.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        lr = lr0 * 1.0 / (1.0 + 0.002 * it)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimizer: LSE warmup -> LP Nesterov -> Coordinate Descent -> 
    LP Nesterov round 2 -> Fine CD -> Multi-length exploration.
    Adapts time allocation based on improvement rate.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Keep a working copy normalized
    work = best_seq.copy()
    n = len(work)
    norm_target = np.sqrt(2.0 * n)
    work = work * norm_target / work.sum()

    # === Phase 1: LSE Gradient Descent (warmup) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE GD n={n} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(work.copy(), t1, verbose=True)
        try_update(a_out, "lse1")

    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    norm_target = np.sqrt(2.0 * n)
    work = best_seq * norm_target / best_seq.sum()
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov n={n} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(work, t2, solver, verbose=True, adapt_M=False)
        try_update(a_out, "lp1")

    # === Phase 3: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    norm_target = np.sqrt(2.0 * n)
    work = best_seq * norm_target / best_seq.sum()
    t3 = min(remaining * 0.18, 300.0)
    print(f"\n=== Phase 3: Coordinate Descent n={n} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = coordinate_descent(work, t3, rng, verbose=True)
        try_update(a_out, "cd1")

    # === Phase 4: LP Nesterov round 2 (with adaptive M) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    norm_target = np.sqrt(2.0 * n)
    work = best_seq * norm_target / best_seq.sum()
    t4 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 4: LP Nesterov r2 (adaptive M) n={n} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(work, t4, solver, verbose=True, adapt_M=True)
        try_update(a_out, "lp2")

    # === Phase 5: Fine Coordinate Descent (ternary search) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    norm_target = np.sqrt(2.0 * n)
    work = best_seq * norm_target / best_seq.sum()
    t5 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 5: Fine CD (ternary) n={n} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_fine(work, t5, rng, verbose=True)
        try_update(a_out, "finecd")

    # === Phase 6: Momentum Subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    norm_target = np.sqrt(2.0 * n)
    work = best_seq * norm_target / best_seq.sum()
    t6 = min(remaining * 0.08, 120.0)
    print(f"\n=== Phase 6: Subgradient n={n} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = momentum_subgradient(work, t6, verbose=True)
        try_update(a_out, "subgrad")

    # === Phase 7: Final LP Nesterov polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    norm_target = np.sqrt(2.0 * n)
    work = best_seq * norm_target / best_seq.sum()
    t7 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 7: Final LP polish n={n} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = nesterov_lp_optimize(work, t7, solver, verbose=False, adapt_M=True)
        try_update(a_out, "lp_final")

    # === Phase 8: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Longer length exploration (remaining={remaining:.0f}s) ===")
    if remaining > 100:
        n_long = len(best_seq) + 400
        init_long = interpolate_seq(best_seq, n_long)
        t8a = min(remaining * 0.45, 150.0)
        print(f"  Trying n={n_long}, LP budget={t8a:.0f}s")
        if t8a > 30:
            a_out, _ = nesterov_lp_optimize(init_long, t8a, solver, verbose=False, adapt_M=False)
            try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t8b = min(remaining * 0.4, 80.0)
            n = len(best_seq)
            norm_target = np.sqrt(2.0 * n)
            work = best_seq * norm_target / best_seq.sum()
            a_out, _ = coordinate_descent(work, t8b, rng, verbose=False)
            try_update(a_out, "cd_long")

    # === Final polish with remaining time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n = len(best_seq)
        norm_target = np.sqrt(2.0 * n)
        work = best_seq * norm_target / best_seq.sum()
        t_final = remaining * 0.9
        print(f"\n=== Final polish: n={n}, budget={t_final:.0f}s ===")
        # Quick CD pass
        a_out, _ = pair_cd_fine(work, min(t_final, 60.0), rng, verbose=False)
        try_update(a_out, "final_cd")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            n = len(best_seq)
            norm_target = np.sqrt(2.0 * n)
            work = best_seq * norm_target / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(work, min(remaining, 50.0), solver, verbose=False, adapt_M=True)
            try_update(a_out, "final_lp")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
