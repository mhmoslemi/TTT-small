def _lp_best_response(a_fixed, M, k_indices, solver, n):
    """
    Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in k_indices, b >= 0.
    """
    import cvxpy as cp
    
    size = len(k_indices)
    if size == 0:
        return np.ones(n) / n, None
    
    # Build constraint matrix
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a_fixed[k - j]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    
    return b_val, None

def _coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent with exact 1D optimization.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = _normalize(a0, norm)
    size = 2 * n - 1
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            aj = a[j]
            
            # Compute background and cross-term
            c = _fast_autocorr(a)  # current autocorrelation
            # c[k] = bg[k] + 2*aj*cross[k] + aj^2*delta_{k,2j}
            # cross[k] = sum of a[i]*a[k-i] where exactly one of i, k-i equals j
            # cross[k] = a[k-j] (if valid) + a[j-k] (if valid), excluding the j=j case
            
            cross = np.zeros(size)
            for k in range(size):
                val = 0.0
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    val += a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    val += a[i2]
                cross[k] = val
            
            # bg[k] = c[k] - 2*aj*cross[k] - aj^2*delta_{k,2j}
            bg = c - 2.0 * aj * cross
            if 2 * j < size:
                bg[2 * j] -= aj * aj
            
            # Minimize max_k (bg[k] + 2*x*cross[k] + x^2*delta_{k,2j}) over x in [0, norm]
            # This is a 1D convex minimization
            
            # Use bisection on the derivative
            def obj_at(x):
                vals = bg + 2.0 * x * cross
                if 2 * j < size:
                    vals[2 * j] += x * x
                return vals.max()
            
            # Binary search for the minimum
            lo = 0.0
            hi = norm
            # First, find the bracket
            mid = (lo + hi) / 2
            f_mid = obj_at(mid)
            f_lo = obj_at(lo)
            f_hi = obj_at(hi)
            
            # Golden section search (more reliable for this case)
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = obj_at(x1)
            f2 = obj_at(x2)
            
            for _ in range(50):  # 50 iterations gives precision ~ (0.618)^50 * range
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    f1 = obj_at(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    f2 = obj_at(x2)
            
            best_x = (lo + hi) / 2
            best_f = obj_at(best_x)
            
            # Compare with current
            curr_f = obj_at(aj)
            if best_f < curr_f - 1e-12:
                a[j] = best_x
                a = _normalize(a, norm)
                new_score = _fast_autocorr(a).max()
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
            # else: keep a[j] = aj (already is)
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def _perturb(a, rng, scale=0.02):
    """Multiplicative perturbation."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    a_p = a * noise
    return _normalize(a_p, np.sqrt(2.0 * len(a)))


def _interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss
