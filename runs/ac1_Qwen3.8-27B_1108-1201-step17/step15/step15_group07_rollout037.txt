def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    # T @ b = conv(a, b)
    # T[m, j] = a[m-j] if 0 <= m-j < n, else 0

def lp_best_response(a, M, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

        return None

def smooth_conv_value_grad(a, t):
    """Compute log-sum-exp smoothed max of conv(a,a) and gradient."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    
    conv_scaled = conv / t
    max_c = np.max(conv_scaled)
    log_sum_exp = max_c + np.log(np.sum(np.exp(conv_scaled - max_c)))
    value = t * log_sum_exp
    
    w = np.exp(conv_scaled - max_c)
    w /= np.sum(w)
    
    # grad[j] = 2*t * sum_m w[m] * a[m-j] (for 0 <= m-j < n)
    W = rfft(w, fft_size)
    grad = irfft(np.conj(A) * W, fft_size)[:n] * 2.0 * t
    
    return value, grad, conv

def smoothed_lbfgs(a0, time_limit, verbose=True):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    
    # Temperature schedule
    temperatures = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    time_per_t = time_limit / len(temperatures)
    
    for t in temperatures:
        if time.monotonic() - t0 > time_limit:
            break
        t_rem = time_limit - (time.monotonic() - t0)
        if t_rem < 5:
            break
        
        # Work in log-space to ensure positivity
        a_curr = a.copy()
        x0 = np.log(a_curr + 1e-30)
        
        def obj_grad(x):
            # Softmap parameterization
            x_shifted = x - x.max()
            a_try = np.exp(x_shifted)
            a_try /= a_try.sum()
            
            val, grad_a, _ = smooth_conv_value_grad(a_try, t)
            
            # Chain rule for softmax
            # da_try[i]/dx[j] = a_try[i] * (delta_ij - a_try[j])
            # dval/dx[j] = sum_i grad_a[i] * a_try[i] * (delta_ij - a_try[j])
            # = a_try[j] * (sum_i grad_a[i] - sum_i a_try[i]*grad_a[i])
            g = a_try * (np.sum(grad_a) - np.dot(a_try, grad_a))
            return val, g
        
        try:
            result = minimize(obj_grad, x0, method='L-BFGS-B',
                            options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-10})
            x_opt = result.x
            a_opt = np.exp(x_opt - x_opt.max())
            a_opt /= a_opt.sum()
        except Exception:
            a_opt = a_curr
        
        score = evaluate_safe(a_opt)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            if verbose:
                print(f"      L-BFGS t={t}: score={score:.6f} (t={time.monotonic()-t0:.0f}s)")
        
        a = a_opt
    
    return best_a, best_score

    no_improve = 0

        alpha = 2.0 / (momentum_t + 2.0)

        y_new = np.maximum(y_new, 0.0)

    return best_a, best_score

So let me make the L-BFGS phase shorter and use it more as an "exploration" tool rather than a "refinement" tool.

For the main optimization, I'll stick with the LP best-response approach but with some tweaks:
1. **Adaptive temperature in the LP**: Instead of using the exact `M = max(conv(y,y))`, use a slightly relaxed `M = (1+eps) * max(conv(y,y))` to allow the LP more freedom early on, then tighten as we converge.

2. **Multiple LP solves per iteration**: Solve the LP with slightly different `M` values to get a more robust best response.

3. **Sharp Nesterov switching**: Switch between "explore" mode (higher momentum, more aggressive steps) and "refine" mode (lower momentum, more precise steps) based on the rate of improvement.

Let me also add one more novel idea: ** "Gradient of the smoothing" approach.** Instead of the log-sum-exp, use the "entropic" smoothing:
`f_smooth(a) = t * log(sum_m exp(conv(a,a)[m] / t))`

And compute the gradient exactly (which I already derived). Then use a **subgradient descent with Polyak step size**:
`a_{k+1} = proj_simplex(a_k - eta_k * grad_k)`

where `eta_k = (f(a_k) - f*) / ||grad_k||^2` (Polyak step size, using a running estimate of `f*`).

This is a well-studied method for convex optimization and might converge faster than Nesterov in practice.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive Nesterov acceleration (primary driver)
2. Smoothed log-sum-exp gradient refinement via L-BFGS-B (exploration)
3. Multi-restart with structured perturbations (escape local minima)
4. Fine-grained targeted coordinate descent polish (final refinement)
Key: adaptive momentum with decay, structured restarts, smoothed gradient
phase for different landscape exploration, efficient targeted CD.

def smooth_conv_val_grad(a, t):
    """Log-sum-exp smoothed max of conv(a,a) and gradient via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]

    cs = conv / t
    mc = np.max(cs)
    val = t * (mc + np.log(np.sum(np.exp(cs - mc))))

    w = np.exp(cs - mc)
    w /= np.sum(w)

    # grad[j] = 2*t * sum_m w[m]*a[m-j] for 0<=m-j<n
    # = 2*t * (w cross-corr a)[j]
    W = rfft(w, fft_size)
    grad = irfft(np.conj(A) * W, fft_size)[:n] * 2.0 * t

    return val, grad


def smoothed_lbfgs_optimize(a0, time_limit, verbose=True):
    """L-BFGS-B with log-sum-exp smoothing and temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()

    temps = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003]
    t_per = time_limit / len(temps)

    for idx, t in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        t_rem = time_limit - (time.monotonic() - t0)
        if t_rem < 5:
            break

        x0 = np.log(a + 1e-30)
        max_iter = min(500, max(50, int(t_per * 150)))

        from scipy.optimize import minimize as _minimize

        result = _minimize(
            _lbfgs_obj_grad, x0, args=(n, t),
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-9}
        )

        x_opt = result.x
        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()

        sc = evaluate_safe(a_opt)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_opt.copy()
            if verbose:
                print(f"      L-BFGS t={t}: score={sc:.6f} (t={time.monotonic()-t0:.0f}s)")

        a = a_opt

    return best_a, best_score


def _lbfgs_obj_grad(x, n, t):
    """Objective and gradient for L-BFGS in log-space."""
    xs = x - x.max()
    a_try = np.exp(xs)
    a_try = a_try / a_try.sum()

    val, grad_a = smooth_conv_val_grad(a_try, t)

    # Softmax chain rule: dval/dx[j] = a[j]*(sum(grad) - a.dot(grad))
    g = a_try * (np.sum(grad_a) - np.dot(a_try, grad_a))
    return val, g


def targeted_cd_optimize(a0, time_limit, verbose=True):
    """Targeted coordinate descent on peak lags."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    size = 2 * n - 1
    no_imp = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-11:
                continue

            offsets = [1, -1, 2, -2, 3, -3, 4, -4, 6, -6, 9, -9, 13, -13]
            for off in offsets:
                j = (i + off) % n
                if j == i:
                    continue

                for frac in [0.05, 0.1, 0.2, 0.35, 0.5]:
                    d = a[i] * frac
                    if d < 1e-14 * norm:
                        continue
                    a[i] -= d
                    a[j] += d
                    nc = fast_autocorr(a).max()
                    if nc < best_score - 1e-13:
                        best_score = nc
                        best_a = a.copy()
                        improved = True
                        no_imp = 0
                        break
                    else:
                        a[i] += d
                        a[j] -= d

                if improved:
                    break

        if not improved:
            no_imp += 1
            if no_imp >= 15:
                noise = rng.normal(0, norm * 0.002, n)
                ap = np.maximum(a + noise, 0)
                sp = ap.sum()
                if sp > 1e-15:
                    a = ap * norm / sp
                no_imp = 0
            elif no_imp >= 80:
                break

        if verbose and no_imp % 30 == 0 and no_imp > 0:
            print(f"      TCD no_imp={no_imp} (t={time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mirror_descent_step(a0, time_limit, top_k=20, verbose=True):
    """Mirror descent with entropy-regularized subgradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    eta = 1.0 / np.sqrt(n)
    patience = 0
    n_iters = 0
    rng = np.random.default_rng(7777)

    for _ in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        mx = conv.max()

        k_act = min(top_k, size)
        act_idx = np.argsort(conv)[-k_act:]

        grad = np.zeros(n)
        for k in act_idx:
            ki = int(k)
            i_lo = max(0, ki - n + 1)
            i_hi = min(ki, n - 1)
            if i_hi >= i_lo:
                j_lo = ki - i_hi
                j_hi = ki - i_lo
                grad[i_lo:i_hi + 1] += a[j_lo:j_hi + 1][::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        la = np.log(a + 1e-30)
        best_try = a.copy()
        best_cm = mx

        for fac in [0.2, 0.4, 0.7, 1.0, 1.5, 2.0]:
            et = eta * fac
            at = np.exp(la - et * grad - (la - et * grad).max())
            at = at * norm / at.sum()
            cm = fast_autocorr(at).max()
            if cm < best_cm:
                best_cm = cm
                best_try = at
                eta = et

        a = best_try
        n_iters += 1

        if best_cm < best_score - 1e-12:
            best_score = best_cm
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 100 == 0:
                eta *= 0.9

        if patience > 200:
            noise = rng.laplace(0, norm * 0.01, n)
            ap = np.maximum(a + noise, 0)
            sp = ap.sum()
            if sp > 1e-15:
                a = ap * norm / sp
            patience = 0
            eta = 1.0 / np.sqrt(n)

        if verbose and n_iters % 200 == 0:
            print(f"      MD it={n_iters} (t={time.monotonic()-t0:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

    return best_a, best_score

                    print(f"  *** NEW BEST [lp2]: {sc:.6f}")

    # --- Phase 4: Mirror Descent (10%) ---
    rem = time_budget - (time.monotonic() - t_global)
    t4 = rem * 0.10
    n4 = len(best_seq)
    print(f"\n=== Phase 4: Mirror Descent n={n4} (budget={t4:.0f}s) ===")
    if t4 > 15:
        an = best_seq * np.sqrt(2.0 * n4) / best_seq.sum()
        ao, _ = mirror_descent_step(an, t4, top_k=25, verbose=True)
        if ao is not None:
            ao = np.maximum(ao, 0.0)
            if ao.sum() > 0.01:
                sc = evaluate_safe(ao)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_seq = ao.copy()
                    print(f"  *** NEW BEST [md]: {sc:.6f}")

    # --- Phase 5: Multi-restart (7%) ---
    rem = time_budget - (time.monotonic() - t_global)
    t5 = rem * 0.07
    n5 = len(best_seq)
    print(f"\n=== Phase 5: Multi-Restart n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        n_rst = 3
        t_pr = t5 / n_rst
        for r in range(n_rst):
            if time.monotonic() - t_global > time_budget:
                break
            if r == 0:
                noise = rng.laplace(0, best_seq.sum() * 0.03, n5)
            elif r == 1:
                sh = int(rng.integers(2, max(3, n5 // 30)))
                noise = (np.roll(best_seq, sh) - best_seq) * 0.3
            else:
                noise = rng.normal(0, best_seq.sum() * 0.015, n5)

            ap = np.maximum(best_seq + noise, 0)
            sp = ap.sum()
            if sp < 1e-10:
                continue
            ap = ap / sp
            t_sub = min(t_pr * 0.85, (time_budget - (time.monotonic() - t_global)) - 30)
            if t_sub < 10:
                break
            an = ap * np.sqrt(2.0 * n5) / ap.sum()
            ao, _ = nesterov_lp_optimize(an, t_sub, solver, verbose=False)
            if ao is not None:
                ao = np.maximum(ao, 0.0)
                if ao.sum() > 0.01:
                    sc = evaluate_safe(ao)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_seq = ao.copy()
                        print(f"  *** NEW BEST [rst{r}]: {sc:.6f}")

    # --- Phase 6: Targeted CD polish (5%) ---
    rem = time_budget - (time.monotonic() - t_global)
    t6 = max(rem * 0.05, 8.0)
    n6 = len(best_seq)
    print(f"\n=== Phase 6: TCD Polish n={n6} (budget={t6:.0f}s) ===")
    if t6 > 5:
        ao, _ = targeted_cd_optimize(best_seq, t6, verbose=False)
        if ao is not None:
            ao = np.maximum(ao, 0.0)
            if ao.sum() > 0.01:
                sc = evaluate_safe(ao)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_seq = ao.copy()
                    print(f"  *** NEW BEST [tcd]: {sc:.6f}")

    # --- Final LP with remaining time ---
    rem = time_budget - (time.monotonic() - t_global)
    if rem > 25:
        nf = len(best_seq)
        print(f"\n=== Final LP n={nf} (budget={rem:.0f}s) ===")
        an = best_seq * np.sqrt(2.0 * nf) / best_seq.sum()
        ao, _ = nesterov_lp_optimize(an, rem * 0.9, solver, verbose=False)
        if ao is not None:
            ao = np.maximum(ao, 0.0)
            if ao.sum() > 0.01:
                sc = evaluate_safe(ao)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_seq = ao.copy()
                    print(f"  *** NEW BEST [final]: {sc:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
