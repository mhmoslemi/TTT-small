            y_new = y_new / s_y
        
        # Nesterov momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        # Check improvement
        sc_y = compute_score(y)
        sc_a = compute_score(a)
        
        improved = False
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic perturbation
        if no_improve == 150:
            rng_p = np.random.default_rng(hash((t, n)) % (2**31))
            n_pert = max(5, n // 25)
            idx = rng_p.choice(n, size=n_pert, replace=False)
            y[idx] *= (1.0 + 0.15 * rng_p.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            warm_b = None
            no_improve = 0
        
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with exact 1D subproblem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        # Identify peak lags for structured candidate selection
        peak_threshold = cur_max * 0.95
        peak_lags = np.where(conv >= peak_threshold)[0]
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue
            total_iter += 1
            
            # Candidate j's: random + structured from peaks
            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                if kl < 2 * n - 1:
                    # Elements that contribute to peak at lag kl
                    # conv[kl] = sum_t a[t] * a[kl-t]
                    # Element i contributes a[i]*a[kl-i] to lag kl
                    # To reduce peak at kl, move mass from i to j where
                    # a[j]*a[kl-j] is small
                    j1 = int(kl - i)
                    j2 = int(kl // 2)
                    if 0 <= j1 < n:
                        j_candidates.add(j1)
                    if 0 <= j2 < n:
                        j_candidates.add(j2)
            j_candidates.discard(i)
            
            if not j_candidates:
                continue
            
            best_delta = 0.0
            best_new_max = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute g[k] and h[k] for the pair (i, j)
                # g[k] = -a[k-i] + a[k-j] + corrections
                # h[k] is non-zero only at 2i, i+j, 2j
                
                # Base linear term
                g_base = np.zeros(size)
                # -a[k-i] for valid k-i
                lo = max(0, i)
                hi = min(size, i + n)
                if hi > lo:
                    g_base[lo:hi] -= a[0:hi-lo]
                # +a[k-j] for valid k-j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g_base[lo:hi] += a[0:hi-lo]
                
                # Corrections
                g = g_base.copy()
                k2i = 2 * i
                kij = i + j
                k2j = 2 * j
                
                if k2i < size:
                    g[k2i] -= a[i]
                if kij < size:
                    g[kij] += a[i] - a[j]
                if k2j < size:
                    g[k2j] += a[j]
                
                # h is non-zero only at 3 positions
                h_positions = {}
                if k2i < size:
                    h_positions[k2i] = h_positions.get(k2i, 0) + 1
                if kij < size and i != j:
                    h_positions[kij] = h_positions.get(kij, 0) - 2
                if k2j < size:
                    h_positions[k2j] = h_positions.get(k2j, 0) + 1
                
                # Find optimal delta by scanning
                # Try a range of deltas
                n_samples = 15
                deltas = np.linspace(0.05, 0.95, n_samples) * ai
                
                best_d = 0.0
                best_v = cur_max
                
                for d in deltas:
                    # New conv = conv + d*g + d^2*h (at h positions)
                    new_conv = conv + d * g
                    for pos, hval in h_positions.items():
                        new_conv[pos] += d * d * hval
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d
                
                if best_d > 0 and best_v < best_new_max - 1e-14:
                    best_new_max = best_v
                    best_delta = best_d
                    best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a / a.sum()
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-14:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = a / a.sum()
        
        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                # Perturb
                n_pert = max(5, n // 40)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
        
        if verbose and total_iter % 5000 == 0 and total_iter > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={total_iter} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lse_descent(a0, time_limit, verbose=False):
    """Log-sum-exp gradient descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 8
    temps = np.geomspace(50.0, 0.5, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(200, int(remaining * 3)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            # Step size with backtracking
            step = 1.0 / (T_val * np.sqrt(n))
            for trial in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = a.copy()
            
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 10)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    else:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def dual_refine(a0, time_limit, solver, verbose=False):
    """Use LP dual variables for gradient refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(300):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break
        
        active = np.where(conv > M * 0.8)[0]
        b, duals = lp_best_response(a, M, solver, active)
        if duals is None or b is None:
            b2, duals2 = lp_best_response(a, M, solver, None)
            if duals2 is not None:
                duals = duals2
                b = b2
            else:
                duals = np.zeros(size)
                duals[np.argmax(conv)] = 1.0
        
        A = rfft(a, fft_size)
        D = rfft(duals, fft_size)
        g1 = irfft(np.conj(A) * D, fft_size)[:n]
        g2 = irfft(A * D, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        step = 0.05
        for trial in range(15):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new / s_new
            sc = fast_conv_self(a_new).max()
            if sc < M - 1e-12:
                a = a_new
                break
            step *= 0.5
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 50 == 0:
            print(f"    DualRefine it={it}: best={best_score:.8f}")
    
    return best_a, best_score
