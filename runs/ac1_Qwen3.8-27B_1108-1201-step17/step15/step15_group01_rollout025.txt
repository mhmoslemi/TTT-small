    """Gerchberg-Saxton style iteration for flat autocorrelation."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = project_nonneg_sum(a, S)
    
    if L is None:
        L = 1 << (2*n-1).bit_length()
    
    # Precompute target magnitude (Dirichlet kernel)
    k = np.arange(L // 2 + 1)
    # D_k = sin(pi*k*(2n-1)/L) / sin(pi*k/L)
    # Handle k=0: D_0 = 2n-1
    D = np.ones(L // 2 + 1)
    mask = k > 0
    D[mask] = np.sin(np.pi * k[mask] * (2*n-1) / L) / np.sin(np.pi * k[mask] / L)
    D = np.abs(D)
    # Add small epsilon to avoid zeros
    D = D + 1e-10
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    for it in range(n_gs):
        # Frequency domain step
        A = rfft(a, L)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        
        # Replace magnitude with target (Dirichlet kernel)
        # Scale to match current energy
        target_mag = C * D  # C chosen below
        C = mag.mean() / D.mean()
        target_mag = C * D

        A_new[0] = np.abs(A_new[0])  # DC is real

        a = a_new * S / s
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

        return np.ones(len(v)) * S / len(v)
    return v * S / s

def init_random(n, rng):
    a = rng.random(n)
    return clip_and_scale(a, np.sqrt(2.0 * n))

def init_sine(n, rng, n_harmonics=3):
    S = np.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf
    for h in range(1, n_harmonics + 1):
        for eps in [0.3, 0.5, 0.7, 0.9]:
            phase = rng.uniform(0, 2 * np.pi)
            a = 1.0 + eps * np.sin(2.0 * np.pi * h * np.arange(n) / n + phase)
            a = clip_and_scale(a, S)
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a

def init_fourier_target(n, rng):
    S = np.sqrt(2.0 * n)
    L = 1 << (2 * n - 1).bit_length()
    best_a = None
    best_score = np.inf
    for _ in range(10):
        k = np.arange(L // 2 + 1, dtype=np.float64)
        D = np.ones(L // 2 + 1)
        mask = k > 0
        denom = np.sin(np.pi * k[mask] / L)
        numer = np.sin(np.pi * k[mask] * (2 * n - 1) / L)
        D[mask] = np.abs(numer / (denom + 1e-30))
        D = D + 1e-8
        phases = np.exp(1j * rng.uniform(0, 2 * np.pi, L // 2 + 1))
        A = D * phases
        a = irfft(A, L)[:n].real
        a = clip_and_scale(a, S)
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a

def init_gaussian(n, rng):
    S = np.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf
    for center_frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.1, 0.2, 0.3, 0.5]:
            x = np.arange(n) / n
            a = np.exp(-((x - center_frac) / width) ** 2)
            a = a + 0.1 * rng.random(n)
            a = clip_and_scale(a, S)
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a

def gs_iterate(a, n_gs, verbose=False):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = clip_and_scale(a, S)
    L = 1 << (2 * n - 1).bit_length()
    k = np.arange(L // 2 + 1, dtype=np.float64)
    D = np.ones(L // 2 + 1)
    mask = k > 0
    D[mask] = np.abs(np.sin(np.pi * k[mask] * (2 * n - 1) / L) / (np.sin(np.pi * k[mask] / L) + 1e-30))
    D = D + 1e-8
    D_mean = D.mean()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    for it in range(n_gs):
        A = rfft(a, L)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        C = mag.mean() / D_mean
        target_mag = C * D
        A_new = target_mag * phase
        A_new[0] = np.abs(A_new[0])
        a_new = irfft(A_new, L)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * S / s
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if verbose and it % 20 == 0:
            print(f"    GS iter={it}: score={best_score:.8f}")
    return best_a, best_score

def compute_smooth_grad_fft(a, w):
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross

def smooth_optimize(a0, time_limit, verbose=True, use_annealing=True):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = clip_and_scale(np.asarray(a0, dtype=np.float64), S)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    rng_inner = np.random.default_rng(12345)
    for T_idx, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(20, min(300, int(remaining * 4)))
        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            exp_sum = exp_c.sum()
            if exp_sum < 1e-300:
                w = np.zeros_like(c)
                w[0] = 1.0
            else:
                w = exp_c / exp_sum
            w_n = w[:n]
            w_n_rev = w[n-1:2*n-1][::-1] if 2*n-1 <= len(w) else np.zeros(n)
            valid = np.arange(2*n-1)
            w_n_rev = np.zeros(n)
            for i in range(n):
                idx1 = i
                idx2 = 2*n-2-i
                if idx2 < len(w):
                    w_n_rev[i] = w[idx2]
                else:
                    w_n_rev[i] = 0.0
            grad1 = compute_smooth_grad_fft(y, w_n)
            grad2 = compute_smooth_grad_fft(y, w_n_rev)
            grad = grad1 + grad2
            step = S / (n * (T + 0.5))
            step = min(step, S / (n * 0.3))
            a_new = y - step * grad
            a_new = clip_and_scale(a_new, S)
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = clip_and_scale(y_new, S)
            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = clip_and_scale(a_curr, S)
            y = y_new
            a = a_curr
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if use_annealing and it > 10 and it % 20 == 0 and rng_inner.random() < 0.1:
                noise_scale = 0.05 * T / S
                y = clip_and_scale(y + rng_inner.normal(0, noise_scale, n), S)
                a = clip_and_scale(a + rng_inner.normal(0, noise_scale, n), S)
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Smooth T={T:.4f} done ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def peak_shaving_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = clip_and_scale(np.asarray(a0, dtype=np.float64), S)
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 5
    peak_threshold = 0.9
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        peak_val = conv.max()
        peak_pos = np.where(conv >= peak_threshold * peak_val)[0]
        if len(peak_pos) == 0:
            peak_pos = np.array([np.argmax(conv)])
        improved = False
        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < S * 1e-10:
                continue
            for pk in peak_pos[:5]:
                contrib = 0.0
                j1 = pk - i
                j2 = i - pk
                if 0 <= j1 < n:
                    contrib += a[j1]
                if 0 <= j2 < n and j2 != j1:
                    contrib += a[j2]
                if contrib < 1e-10:
                    continue
                for j in [(i + 1) % n, (i - 1) % n, (i + 3) % n, (i - 3) % n, (i + 7) % n]:
                    if j == i:
                        continue
                    for frac in [0.5, 0.3, 0.1]:
                        d = a[i] * frac
                        if d < 1e-15:
                            break
                        a_test = a.copy()
                        a_test[i] -= d
                        a_test[j] += d
                        a_test = clip_and_scale(a_test, S)
                        conv_test = fast_autocorr(a_test)
                        sc = conv_test.max()
                        if sc < best_score - 1e-12:
                            a = a_test
                            conv = conv_test
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                        elif sc < conv.max() - 1e-12:
                            a = a_test
                            conv = conv_test
                            if conv.max() < best_score:
                                best_score = conv.max()
                                best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        elif verbose:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD: best={best_score:.8f} ({elapsed:.0f}s)")
    return best_a, best_score

def full_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = clip_and_scale(np.asarray(a0, dtype=np.float64), S)
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < S * 1e-10:
                continue
            window = min(30, n // 4)
            for offset in range(1, window + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i or a[i] < S * 1e-10:
                        continue
                    lo, hi = 0.0, a[i] * 0.8
                    best_d = 0.0
                    best_val = conv.max()
                    for _ in range(6):
                        mid = (lo + hi) / 2
                        if mid <= 0:
                            break
                        a_t = a.copy()
                        a_t[i] -= mid
                        a_t[j] += mid
                        a_t = clip_and_scale(a_t, S)
                        conv_t = fast_autocorr(a_t)
                        if conv_t.max() < best_val - 1e-12:
                            best_val = conv_t.max()
                            best_d = mid
                            lo = mid
                        else:
                            hi = mid
                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        a = clip_and_scale(a, S)
                        conv = fast_autocorr(a)
                        if conv.max() < best_score - 1e-12:
                            best_score = conv.max()
                            best_a = a.copy()
                        improved = True
                        no_improve = 0
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        elif verbose:
            elapsed = time.monotonic() - t0
            print(f"    FullCD: best={best_score:.8f} ({elapsed:.0f}s)")
    return best_a, best_score

def optimize_single(a_init, n, time_budget, rng, verbose=True):
    S = np.sqrt(2.0 * n)
    a_init = clip_and_scale(np.asarray(a_init, dtype=np.float64), S)
    best_a = a_init.copy()
    best_score = fast_autocorr(a_init).max()
    t0 = time.monotonic()
    t_gs = min(30.0, time_budget * 0.15)
    if t_gs > 5:
        a_gs, _ = gs_iterate(a_init, 80, verbose=False)
        sc_gs = fast_autocorr(a_gs).max()
        if sc_gs < best_score:
            best_score = sc_gs
            best_a = a_gs.copy()
        if verbose:
            print(f"      After GS: {best_score:.8f}")
    remaining = time_budget - (time.monotonic() - t0)
    t_smooth = remaining * 0.5
    if t_smooth > 10:
        a_s, _ = smooth_optimize(best_a, t_smooth, verbose=verbose)
        sc_s = fast_autocorr(a_s).max()
        if sc_s < best_score:
            best_score = sc_s
            best_a = a_s.copy()
        if verbose:
            print(f"      After smooth: {best_score:.8f}")
    remaining = time_budget - (time.monotonic() - t0)
    t_cd = remaining * 0.5
    if t_cd > 5:
        a_c, _ = peak_shaving_cd(best_a, t_cd, rng, verbose=verbose)
        sc_c = fast_autocorr(a_c).max()
        if sc_c < best_score:
            best_score = sc_c
            best_a = a_c.copy()
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 10:
        a_f, _ = full_cd(best_a, remaining, rng, verbose=False)
        sc_f = fast_autocorr(a_f).max()
        if sc_f < best_score:
            best_score = sc_f
            best_a = a_f.copy()
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    best_seq = None
    best_score = float('inf')
    print(f"Total time budget: {time_budget:.0f}s")
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    if best_seq is None:
        n0 = 1600
        best_seq = np.ones(n0) / n0
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    # Define strategies: (n, init_func, label_prefix)
    n_parent = len(best_seq)
    lengths = sorted(set([n_parent, 1200, 1600, 2000, 2400, 3200]))
    lengths = [n for n in lengths if 500 <= n <= 5000]
    n_starts_per_length = 2
    n_total_starts = len(lengths) * n_starts_per_length
    time_per_start = time_budget / (n_total_starts + 2)
    print(f"Lengths to try: {lengths}")
    print(f"Time per start: {time_per_start:.0f}s")
    start_idx = 0
    for n in lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        for start in range(n_starts_per_length):
            start_idx += 1
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_start = min(time_per_start, remaining - 10)
            if t_start < 10:
                break
            if start == 0:
                if n == len(best_seq):
                    a_init = best_seq * np.sqrt(2.0 * n) / best_seq.sum()
                else:
                    a_init = init_sine(n, rng)
                label = f"sine_n{n}_s{start}"
            elif start == 1:
                r = rng.random()
                if r < 0.33:
                    a_init = init_fourier_target(n, rng)
                    label = f"fourier_n{n}_s{start}"
                elif r < 0.66:
                    a_init = init_gaussian(n, rng)
                    label = f"gauss_n{n}_s{start}"
                else:
                    a_init = init_random(n, rng)
                    label = f"rand_n{n}_s{start}"
            print(f"\n  Start {start_idx}/{n_total_starts}: {label}, t={t_start:.0f}s")
            a_out, _ = optimize_single(a_init, n, t_start, rng, verbose=False)
            try_update(a_out, label)
    # Final polish on the best solution found
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n  Final polish: n={n_best}, t={remaining:.0f}s")
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        # Smooth + CD
        t_s = remaining * 0.4
        a_s, _ = smooth_optimize(a_init, t_s, verbose=True)
        try_update(a_s, "final_smooth")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            a_c, _ = peak_shaving_cd(a_s if fast_autocorr(a_s).max() < fast_autocorr(a_init).max() else a_init, remaining2, rng, verbose=True)
            try_update(a_c, "final_peakcd")
    # Second final polish with different approach
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n  Final CD: n={n_best}, t={remaining:.0f}s")
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_c, _ = full_cd(a_init, remaining, rng, verbose=True)
        try_update(a_c, "final_fullcd")
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
