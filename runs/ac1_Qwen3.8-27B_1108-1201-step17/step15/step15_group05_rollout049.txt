def lp_best_response_full(a_fixed, M, n, solver, precomputed_indices=None):
    """
    Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for ALL k, b >= 0.

    """

        size = 2 * n - 1
        rows = []
        cols = []
        vals = []
        for k in range(size):

            hi = min(n, k + 1)
            for j in range(lo, hi):
                rows.append(k)
                cols.append(j)
                vals.append(a_fixed[k - j])

        T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    else:
        T = precomputed_indices

            col_list.append(j)
    row_arr = np.array(row_list)
    col_arr = np.array(col_list)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Build constraint matrix with current y values
        vals = np.zeros(len(row_arr))
        for idx in range(len(row_arr)):
            k = row_arr[idx]
            j = col_arr[idx]
            vals[idx] = y[k - j]
        T = csr_matrix((vals, (row_arr, col_arr)), shape=(size, n))

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):

        n_iters += 1

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)

        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.99)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

            a = best_a.copy()

            # More efficient:
        grad = np.zeros(n)

            lo = max(0, k - n + 1)
            hi = min(n, k + 1)

                grad[j] += 2.0 * w * a[k - j]

def coordinate_descent_v2(a0, time_limit, verbose=False):
    """
    Coordinate descent with exact 1D convex minimization via bisection.
    For coordinate j, minimize max_k (bg[k] + 2*x*c[k] + x^2*delta_{k,2j}) / (s+x-a_j)^2
    over x >= 0, using the fact that the objective is unimodal.
    """

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            conv = fast_autocorr(a)
            s_others = a.sum() - aj

            # Precompute c[k] = sum of a[i] for terms involving a[j] at lag k
            # (a*a)[k] = bg[k] + 2*x*c[k] + x^2*delta_{k,2j}
            c = np.zeros(size)
            for k in range(size):
                val = 0.0
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    val += a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    val += a[i2]
                c[k] = val

            bg = conv - 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            # Minimize f(x) = 2*n*max_k(bg[k]+2x*c[k]+x^2*delta)/(s_others+x)^2 over x>=0
            # Use ternary search (function is unimodal for x in [0, some_upper])

            # Find upper bound for x
            x_max = max(norm * 2, aj * 3, s_others)

            def obj_1d(x):
                if x < 0:
                    return float('inf')
                new_sum = s_others + x
                if new_sum < 1e-15:
                    return float('inf')
                new_conv = bg + 2.0 * x * c
                if 2 * j < size:
                    new_conv[2 * j] += x * x
                return 2.0 * n * new_conv.max() / (new_sum * new_sum)

            # Ternary search
            lo = 0.0
            hi = x_max
            # First check if current is already good
            obj_current = obj_1d(aj)

            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = obj_1d(m1)
                f2 = obj_1d(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            x_opt = (lo + hi) / 2.0
            obj_opt = obj_1d(x_opt)

            if obj_opt < obj_current - 1e-10:
                a[j] = x_opt
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm / a.sum()

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP with all constraints (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP all-constraints, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_all(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, "lp1_all")

    # === Phase 2: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: CD, remaining={remaining:.0f}s ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_v2(a_norm, t2, verbose=True)
        try_update(a_out, "cd1")

    # === Phase 3: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Subgrad, remaining={remaining:.0f}s ===")
    t3 = min(remaining * 0.12, 120.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish_v2(a_norm, t3, verbose=True)
        try_update(a_out, "subg1")

    # === Phase 4: LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP round 2, remaining={remaining:.0f}s ===")
    t4 = min(remaining * 0.25, 300.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_all(a_norm, t4, solver, verbose=True, seed=77)
        try_update(a_out, "lp2_all")

    # === Phase 5: Multi-start perturbations ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Multi-start, remaining={remaining:.0f}s ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 40:
        n_starts = 3
        t_each = t5 / n_starts
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = multi_start_lp(a_norm, n_starts, t_each, solver, verbose=True)
        try_update(a_out, "multistart")

    # === Phase 6: Final CD + subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final polish, remaining={remaining:.0f}s ===")
    t6a = min(remaining * 0.5, 100.0)
    t6b = min(remaining - t6a, 60.0)
    if t6a > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_v2(a_norm, t6a, verbose=False)
        try_update(a_out, "cd_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t6c = min(remaining * 0.6, 60.0)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish_v2(a_norm, t6c, verbose=False)
        try_update(a_out, "subg_final")

    # Final LP squeeze
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final LP squeeze, remaining={remaining:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
        a_out, _ = nesterov_lp_all(a_norm, remaining - 10, solver, verbose=False, seed=314)
        try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
