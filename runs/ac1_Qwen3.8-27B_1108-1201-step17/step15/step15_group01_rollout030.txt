def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix T where T[k,i] = a[k-i] if valid."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T


def lp_best_response(a, M, solver, active_indices=None):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns (b_val, dual_vars) or (None, None).
    """
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)

    if active_indices is not None and len(active_indices) < size:
        T_use = T[active_indices, :]
    else:
        T_use = T

    b_var = cp.Variable(n, nonneg=True)
    y_var = cp.Variable(len(T_use), nonneg=True)
    # Primal: max sum(b) s.t. T_use @ b <= M, b >= 0
    constraints = [T_use @ b_var <= M]
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    # Try to get dual variables
    dual = None
    try:
        if constraints[0].dual_value is not None:
            dual = np.asarray(constraints[0].dual_value, dtype=np.float64)
            dual = np.maximum(dual, 0.0)
    except Exception:
        dual = None
    return b_val, dual

            # Nesterov
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stagnant = 0

            # Early stop if very stagnant at low T
            if T < 0.1 and stagnant > 200:
                break

            if verbose and T <= 0.5 and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.4f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def full_w_to_conv_weights(w, n):
    """Convert convolution-domain weights w (length 2n-1) to the form needed
    for the Hankel matrix gradient computation.
    For grad_i = 2 * sum_j w_{i+j} * a[j], we need to compute 
    sum_j w_{i+j} * a[j] which is a cross-correlation-like operation.
    We reshape w into the appropriate form for FFT computation.
    """
    # We want: (H @ a)_i = sum_j w_{i+j} * a[j] for i = 0,...,n-1
    # where the sum is over valid j (0 <= j < n, 0 <= i+j < 2n-1)
    # This is: sum_j H[i,j] * a[j] where H[i,j] = w_{i+j}
    # H is a Hankel matrix. We can compute H @ a via FFT:
    # H[i,j] = w[i+j] = (w * flip(w))[i+j]... no that's not right.
    # Actually H[i,j] = w[i+j] is a Hankel matrix.
    # H @ a = sum_j w[i+j] * a[j]
    # Let m = i+j, then j = m-i: sum_m w[m] * a[m-i] (for valid m-i)
    # This is the cross-correlation (a * w)[i] with appropriate offset.
    # In numpy: np.correlate(a, w, mode='full')[n-1:] gives sum_j a[j] * w[j+i] for i=0,...,n-1
    # Wait: np.correlate(a, w)[k] = sum_j a[j+k] * w[j] for valid j
    # We want sum_j a[j] * w[i+j] = sum_j a[j] * w[j+i]
    # This is np.correlate(a, w, 'full')[offset] for the right offset.
    # Actually: correlation of a with w at lag i = sum_j conj(w[j]) * a[j+i]
    # We want sum_j a[j] * w[j+i]. Let k = j+i, j = k-i: sum_k a[k-i] * w[k]
    # = cross-correlation of a and w at lag -i... 
    # Let me just use the FFT approach directly.
    # sum_j w[i+j] * a[j]: define b[j] = w[i+j] for j=0,...,n-1 (with w padded to length 2n-1)
    # This is the inner product of a with b, where b_i is a shifted version of w.
    # For all i simultaneously: this is H @ a where H[i,j] = w[i+j]
    # H is a Hankel matrix. We can compute it via FFT:
    # H[i,j] = w[i+j]. Define c[m] = w[m] for m=0,...,2n-2 (padded to power of 2).
    # H[i,j] = c[i+j] = (c * reverse(c))[i+j]... no.
    # Actually, a Hankel matrix H[i,j] = v[i+j] can be factored as:
    # H = C @ P @ C^T or similar, but simplest is to just do it directly for small n.
    # For n=1600, H is 1600x1600, and H@a is 1600*1600 = 2.56M ops. That's fast.
    # But we can do better with FFT.
    # H[i,j] = w[i+j]. Let's define u[m] = w[m] for m = 0,...,2n-2.
    # Then (H@a)[i] = sum_j u[i+j] * a[j] for j=0,...,n-1 (with u[i+j]=0 if i+j >= 2n-1)
    # = sum_j u[i+j] * a[j]
    # This is the cross-correlation of u and a, evaluated at specific lags.
    # Specifically, if we define the linear cross-correlation (u ⊛ a)[d] = sum_j u[j+d] * a[j]
    # Then (H@a)[i] = (u ⊛ a)[i] for i = 0,...,n-1.
    # And (u ⊛ a) can be computed via FFT: irfft(fft(u) * conj(fft(a)))
    # But we need to be careful about the length. u has length 2n-1, a has length n.
    # The cross-correlation has length 3n-2 (or 2n-1+n-1 = 3n-2... no).
    # Actually for linear cross-correlation of signals of lengths L1 and L2:
    # result has length L1+L2-1.
    # Here L1 = 2n-1, L2 = n, so result has length 3n-2.
    # We want indices 0 to n-1 of this result.
    
    size = 2 * n - 1
    fft_size = 1 << (size + n - 1 - 1).bit_length()  # need at least size + n - 1
    U = rfft(w, fft_size)  # w has length 2n-1
    A = rfft(np.zeros(fft_size), fft_size)  # placeholder
    a_padded = np.zeros(fft_size)
    a_padded[:n] = "TEMP"  # will fix
    # Actually let me just compute it properly
    fft_size = 1 << (size + n - 1).bit_length()
    U = rfft(w, fft_size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    cross = irfft(U * np.conj(A), fft_size)[:n]
    return cross


def subgradient_method(a0, time_limit, verbose=True):
    """Subgradient method for max_k conv_k(a,a) with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    step_scale = norm / n  # base step scale

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        c_max = conv_y.max()

        # Find active set: top fraction of peaks
        # Use the top 5% or at least 3 peaks
        n_active = max(3, min(20, len(conv_y) // 20))
        top_idx = np.argsort(conv_y)[::-1][:n_active]
        # Only include those close to max
        threshold = c_max * 0.95
        active = top_idx[top_idx >= 0]  # top_idx is already sorted
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 3:
            active = np.arange(len(conv_y))

        # Compute subgradient as average over active set
        # grad_i = 2 * sum_{k in active} w_k * (Q_k a)_i
        # where (Q_k a)_i = a[k-i] (for valid k-i) + a[i] (if 2i=k)
        # Simplified: grad = 2/n_active * sum_{k in active} cross_corr(1_{k}, a)
        # But cross_corr(1_k, a) = a shifted by k... 
        # More precisely: (Q_k a)_i = sum_j Q_k[i,j] a[j]
        # Q_k[i,j] = 1/2 if i+j=k (i!=j), 1 if 2i=k
        # (Q_k a)_i = (1/2)*a[k-i] + (1/2)*a[k-i] (if i != k-i) = a[k-i]
        # or = a[i] (if i = k-i, i.e., 2i=k)
        # Actually (Q_k a)_i = a[k-i] for all valid i (where 0 <= k-i < n)
        # because when i = k-i, the 1/2 + 1/2 = 1, giving a[i] = a[k-i].
        # Wait no: Q_k[i,i] = 1 if 2i=k, and (Q_k a)_i gets contribution a[i]*1 = a[i].
        # Q_k[i,k-i] = 1/2 (if i != k-i), contribution (1/2)*a[k-i].
        # Q_k[k-i,i] = 1/2, contribution (1/2)*a[i].
        # Total for i != k-i: a[i]/2 + a[k-i]/2... no wait.
        # (Q_k a)_i = Q_k[i,i]*a[i] + Q_k[i,k-i]*a[k-i] (if k-i != i and valid)
        # = (1 if 2i=k else 0)*a[i] + (1/2 if i!=k-i else 0)*a[k-i]
        # Hmm, this is getting confusing. Let me just use the smooth gradient approach
        # with a very small temperature, which approximates the subgradient.
        
        # Use smooth gradient with very small T as subgradient approximation
        T_sub = max(c_max * 0.005, 0.01)
        scaled = (conv_y - c_max) / T_sub
        scaled = np.maximum(scaled, -50.0)
        w = np.exp(scaled)
        w /= w.sum()

        grad = compute_smooth_grad(w, y)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Step size: O(1/sqrt(t))
        step = step_scale * (norm / (grad_norm + 1e-10)) / np.sqrt(t)
        step = min(step, norm * 0.1)

        a_new = y - step * grad
        a_new = project_simplex(a_new, norm)

        # Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new
        y_new = project_simplex(y_new, norm)

        if t == 1:
            a_curr = a_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_curr = y_new + momentum * (y_new - y)
            a_curr = project_simplex(a_curr, norm)

        y = y_new
        a = a_curr

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Polyak average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        if n_avg >= 300:
            avg_a = a.copy()
            n_avg = 1

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_direct(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with direct evaluation and greedy selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    # Precompute: for each position i, which conv indices are affected
    # Moving mass from i to j affects conv[k] for k in [j, j+n-1] and [i, i+n-1]
    # Specifically: conv[k] changes by delta*(a[k-j] - a[k-i]) (approximately)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Find peak region
        peak_idx = conv.argmax()
        peak_val = conv.max()

        perm = rng.permutation(n)
        n_tried = 0

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            # Find best j to move to: try positions that would reduce the peak
            # The peak is at peak_idx. Moving mass from i to j changes conv[peak_idx] by:
            # delta * (a[peak_idx - j] - a[peak_idx - i]) (if valid)
            # We want this to be negative, so we want a[peak_idx - j] < a[peak_idx - i]
            
            best_j = -1
            best_improvement = 0
            best_delta = 0

            # Try a subset of candidate j's
            n_candidates = 50
            # Prioritize j's near the peak and at the edges
            candidates = set()
            # Near peak
            for off in range(-10, 11):
                j = (peak_idx + off) % n
                if 0 <= j < n and j != i:
                    candidates.add(j)
            # Random
            for _ in range(n_candidates // 2):
                candidates.add(rng.integers(0, n))
            # Edges
            candidates.add(0)
            candidates.add(n - 1)
            # Symmetric position
            candidates.add((n - 1 - i) % n)

            for j in candidates:
                if j == i or a[i] < norm * 1e-12:
                    continue
                # Quick check: does moving to j help?
                # Change in conv at peak: delta * (a[peak_idx - j] - a[peak_idx - i])
                # (simplified, ignoring boundary effects)
                a_pj = a[peak_idx - j] if 0 <= peak_idx - j < n else 0.0
                a_pi = a[peak_idx - i] if 0 <= peak_idx - i < n else 0.0
                if a_pj < a_pi - 1e-10:
                    # Potential improvement, try it
                    test_d = a[i] * 0.2
                    a_test = a.copy()
                    a_test[i] -= test_d
                    a_test[j] += test_d
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < peak_val - 1e-10:
                        # Try to find best delta via a few steps
                        best_delta = test_d
                        best_j = j
                        best_improvement = peak_val - conv_test.max()
                        # Try larger
                        for frac in [0.4, 0.6, 0.8]:
                            test_d2 = a[i] * frac
                            if test_d2 > a[i]:
                                break
                            a_test2 = a.copy()
                            a_test2[i] -= test_d2
                            a_test2[j] += test_d2
                            conv_test2 = fast_autocorr(a_test2)
                            if conv_test2.max() < peak_val - 1e-10:
                                best_delta = test_d2
                                best_j = j
                                best_improvement = peak_val - conv_test2.max()
                            else:
                                break

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a * norm / a.sum()
                conv = fast_autocorr(a)
                if conv.max() < best_score - 1e-12:
                    best_score = conv.max()
                    best_a = a.copy()
                pass_improved = True
                no_improve = 0
                n_tried += 1

            # Limit evaluations per pass
            if n_tried > 200:
                break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def generate_initial_structures(n, rng):
    """Generate diverse initial structures."""
    structures = []

    # 1. Linear ramp (decreasing)
    a = np.arange(n, 0, -1, dtype=np.float64)
    structures.append(a / a.sum())

    # 2. Linear ramp (increasing)  
    a = np.arange(1, n + 1, dtype=np.float64)
    structures.append(a / a.sum())

    # 3. Exponential decay
    alpha = 3.0
    a = np.exp(-alpha * np.arange(n) / n)
    structures.append(a / a.sum())

    # 4. Exponential growth
    a = np.exp(alpha * np.arange(n) / n)
    structures.append(a / a.sum())

    # 5. Asymmetric tent (fast rise, slow fall)
    mid = n // 3
    a = np.zeros(n)
    a[:mid] = np.linspace(1, 3, mid)
    a[mid:] = np.linspace(3, 0.1, n - mid)
    structures.append(a / a.sum())

    # 6. Slow rise, fast fall
    mid = 2 * n // 3
    a = np.zeros(n)
    a[:mid] = np.linspace(0.1, 3, mid)
    a[mid:] = np.linspace(3, 1, n - mid)
    structures.append(a / a.sum())

    # 7. Gaussian (off-center)
    center = n // 4
    sigma = n / 6.0
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    structures.append(a / a.sum())

    # 8. Another Gaussian
    center = 3 * n // 4
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    structures.append(a / a.sum())

    # 9. Sine (positive part)
    a = np.maximum(np.sin(np.pi * np.arange(n) / n), 0.01)
    structures.append(a / a.sum())

    # 10. Cosine ramp
    a = 0.5 * (1 + np.cos(np.pi * np.arange(n) / n))
    a = np.maximum(a, 0.01)
    structures.append(a / a.sum())

    # 11. Random smooth
    a = np.ones(n)
    for _ in range(5):
        center = rng.integers(0, n)
        width = rng.integers(n // 4, n // 2)
        amp = rng.uniform(0.5, 2.0)
        x = np.arange(n)
        a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0.01)
    structures.append(a / a.sum())

    # 12. Comb-like (alternating high/low with drift)
    a = np.ones(n) * 0.5
    a[::2] = 1.5
    a *= np.linspace(1.0, 0.3, n)
    structures.append(a / a.sum())

    return structures


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
            parent_loaded = True
    except Exception:
        pass

    n_target = 1600

    if not parent_loaded:
        # Generate diverse initial structures and pick the best
        print("Generating initial structures...")
        structures = generate_initial_structures(n_target, rng)
        best_init = None
        best_init_score = float('inf')
        for idx, s in enumerate(structures):
            s_norm = s * np.sqrt(2.0 * n_target) / s.sum()
            sc = fast_autocorr(s_norm).max()
            if sc < best_init_score:
                best_init_score = sc
                best_init = s.copy()
        best_seq = best_init
        best_score = fast_autocorr(best_seq * np.sqrt(2.0 * n_target) / best_seq.sum()).max()
        # Normalize to actual objective
        actual_score = evaluate_safe(best_seq)
        best_score = actual_score
        print(f"Best initial: score={best_score:.6f}")
    else:
        # Interpolate parent to target length if needed
        if len(best_seq) != n_target:
            best_seq = interpolate_sequence(best_seq, n_target)
            best_score = evaluate_safe(best_seq)
            print(f"Interpolated parent: n={n_target}, score={best_score:.6f}")
        n_target = len(best_seq)

    # === Phase 1: LP Nesterov (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LP Nesterov n={n_target} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_target) / best_seq.sum()
        a_out, _, dual_w = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_target}")
        # Store dual weights for next phase
        if dual_w is None:
            dual_w = None
    else:
        dual_w = None

    # === Phase 2: Smooth gradient with dual guidance (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.25
    n_c = len(best_seq)
    print(f"\n=== Phase 2: Smooth grad n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t2, dual_weights=dual_w, verbose=True)
        do_try_update(a_out, f"smooth_n{n_c}")

    # === Phase 3: Subgradient method (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    n_c = len(best_seq)
    print(f"\n=== Phase 3: Subgradient n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = subgradient_method(a_norm, t3, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_c}")

    # === Phase 4: LP Nesterov round 2 (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.10
    n_c = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov r2 n={n_c} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _, _ = nesterov_lp(a_norm, t4, solver, verbose=False)
        do_try_update(a_out, f"lp2_n{n_c}")

    # === Phase 5: Pair CD (7% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.07
    n_c = len(best_seq)
    print(f"\n=== Phase 5: Pair CD n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = pair_cd_direct(a_norm, t5, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_c}")

    # === Phase 6: Diversified restarts (3% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.03
    n_c = len(best_seq)
    print(f"\n=== Phase 6: Diversified restarts (budget={t6:.0f}s) ===")
    if t6 > 30:
        # Try a few perturbed versions
        for restart_idx in range(3):
            rem6 = time_budget - (time.monotonic() - t_global)
            if rem6 < 15:
                break
            t6_sub = min(rem6 / 3.0, t6 / 3.0)
            # Perturb
            a_pert = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
            # Mix with a different structure
            if restart_idx == 0:
                noise = rng.normal(0, 0.15, n_c)
                a_pert = a_pert + noise
            elif restart_idx == 1:
                # Mix with ramp
                ramp = np.arange(1, n_c + 1, dtype=np.float64)
                ramp = ramp / ramp.sum() * np.sqrt(2.0 * n_c)
                a_pert = 0.7 * a_pert + 0.3 * ramp
            else:
                # Mix with exponential
                exp_deg = np.exp(-2.0 * np.arange(n_c) / n_c)
                exp_deg = exp_deg / exp_deg.sum() * np.sqrt(2.0 * n_c)
                a_pert = 0.7 * a_pert + 0.3 * exp_deg

            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert * np.sqrt(2.0 * n_c) / a_pert.sum()

            # Quick smooth optimization
            a_out, _ = smooth_optimize(a_pert, t6_sub, verbose=False)
            do_try_update(a_out, f"restart{restart_idx}")

    # === Phase 7: Final smooth refinement (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        n_c = len(best_seq)
        print(f"\n=== Phase 7: Final refinement (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, remaining * 0.6, verbose=False)
        do_try_update(a_out, "final_smooth")
        
        rem7 = time_budget - (time.monotonic() - t_global)
        if rem7 > 15:
            a_norm2 = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
            a_out2, _ = pair_cd_direct(a_norm2, rem7 * 0.7, rng, verbose=False)
            do_try_update(a_out2, "final_pcd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        fallback = np.ones(100)
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
