This is equivalent to: for each active lag k, add `2*a[k-i]` to `grad[i]` for all valid i. In other words, for each active lag k, we add a shifted copy of `a` to the gradient.

def compute_subgradient_fast(a, active_lags):
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        # grad[i] += 2 * a[k-i] for 0 <= k-i < n, i.e., k-n+1 <= i <= k
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            # a[k-i] for i in [lo, hi] is a[k-hi : k-lo+1] reversed
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    return grad / len(active_lags)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with geometric (multiplicative) updates and adaptive annealing
2. Fast vectorized subgradient polishing on active lags
3. Multi-restart with structured perturbations
4. Final active-set LP polish
Key insight: d(conv[k])/da[i] = 2*a[k-i] (if valid), enabling O(n*|S|) subgradient computation.
Uses scipy HiGHS solver for speed, geometric updates for positivity preservation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def fast_autocorr(a):
    """Compute aperiodic autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_sparse_conv_matrix(a):
    """Build sparse matrix A where (A @ b)[k] = (a*b)[k] aperiodic."""
    n = len(a)
    size = 2 * n - 1
    n_entries = n * n
    rows = np.zeros(n_entries, dtype=np.int32)
    cols = np.zeros(n_entries, dtype=np.int32)
    data = np.zeros(n_entries)
    for j in range(n):
        idx = j * n
        rows[idx:idx + n] = np.arange(j, j + n)
        cols[idx:idx + n] = j
        data[idx:idx + n] = a
    return csr_matrix((data, (rows, cols)), shape=(size, n))

def lp_best_response(a, M):
    """Solve max 1^T b s.t. conv(a,b) <= M, b >= 0. Returns (b, dual) or (None, None)."""
    n = len(a)
    A = build_sparse_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None, None
    if res.status != 0:
        return None, None
    b_val = np.maximum(res.x, 0.0)
    dual = None
    try:
        dual = np.maximum(np.asarray(res.ineqlin.marginals, dtype=np.float64), 0.0)
    except Exception:
        pass
    return b_val, dual

def geometric_update(a, b, alpha):
    """Multiplicative update: a * (b_norm/a)^alpha, normalized to sum=1."""
    s_a = a.sum()
    s_b = b.sum()

        return a.copy()
    b_norm = b * (s_a / s_b)
    a_new = np.zeros_like(a)
    pos = a > 1e-20
    a_new[pos] = a[pos] * np.power(b_norm[pos] / a[pos], alpha)
    zero = ~pos
    if zero.any():
        a_new[zero] = b_norm[zero] * 1e-4
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def compute_subgrad_fast(a, active_lags):
    """Compute subgradient of max_k conv[k] using d(conv[k])/da[i] = 2*a[k-i]."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad / len(active_lags)
