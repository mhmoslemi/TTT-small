def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

def compute_sharp_subgradient(a, ac, top_k=3, beta=10.0):

    top_idx = np.argsort(ac)[::-1][:k_max]

    w_sum = w.sum()

            idx = np.arange(i_start, i_end)

def freq_init(n, rng):
    """Generate a sequence with approximately flat circular autocorrelation."""
    # Random phase, flat magnitude spectrum
    fft_size = 2 * n
    A_mag = np.ones(fft_size // 2 + 1)
    A_phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A = A_mag * np.exp(1j * A_phase)
    # Make it conjugate symmetric for real output
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:fft_size//2+1] = A
    for i in range(1, fft_size // 2 + 1):
        A_full[fft_size - i] = np.conj(A[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    # Make non-negative
    a = np.abs(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def coordinate_polish(a0, time_limit, verbose=False, rng_seed=42):

            best_dst = -1

                    best_dst = dst

                M_new = fast_autocorr(a).max()

                    stall = 0
                else:

                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with sharp peak-directed subgradient
2. Multiplicative (exponentiated gradient) updates
3. Frequency-domain and window-function initializations
4. Coarse-to-fine length scaling
5. Coordinate descent polishing
6. Multi-phase time allocation with adaptive restart

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_value(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient(a, ac, top_k=3, beta=10.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent(a0, time_limit, top_k=3, beta=12.0, c0=5.0,
                   restart_patience=2500, verbose=False, seed=42):

    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_patience:
            stall = 0
            noise_scale = 0.003 + 0.003 * (stall / restart_patience)
            noise = rng.normal(0, noise_scale, n)
            x = best_a + noise
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            cur_top_k = min(cur_top_k + 1, top_k + 3)
            cur_beta = max(cur_beta - 1.0, 4.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

def multiplicative_update(a0, time_limit, top_k=5, verbose=False, seed=7):

    a = normalize(a0)
    best_a = a.copy()

        top_idx = np.argsort(ac)[::-1][:top_k]

            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += a[j_idx]

        g_max = g.max()
        if g_max < 1e-30:
            break
        g_norm = g / g_max
        a = a * np.exp(-eta * g_norm)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            eta *= 0.97
            if eta < 0.05:
                noise = rng.normal(0, 0.01, n)
                a = best_a + noise
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                eta = 0.5
                stall = 0

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MU t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}")

    return best_a, best_m


def window_init(n, kind):
    x = np.linspace(0, 1, n)
    if kind == 'hann':
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif kind == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif kind == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif kind == 'blackman_harris':
        a = (0.35875 - 0.48829 * np.cos(2*np.pi*x) + 0.14128 * np.cos(4*np.pi*x)
             - 0.01168 * np.cos(6*np.pi*x))
    elif kind == 'gaussian':
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif kind == 'super_gaussian':
        a = np.cos(np.pi * (x - 0.5)) ** 6
    elif kind == 'parabola':
        a = 4 * x * (1 - x)
    elif kind == 'parabola_sq':
        a = (4 * x * (1 - x)) ** 2
    elif kind == 'quartic':
        a = (4 * x * (1 - x)) ** 4
    elif kind == 'sin4':
        a = np.sin(np.pi * x) ** 4
    elif kind == 'cos2':
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif kind == 'flat_cos':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def freq_init(n, rng):
    fft_size = 2 * n
    A_phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A_mag = np.ones(fft_size // 2 + 1)
    A = A_mag * np.exp(1j * A_phase)
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:fft_size // 2 + 1] = A
    for i in range(1, fft_size // 2 + 1):
        A_full[fft_size - i] = np.conj(A[i])
    a = np.fft.irfft(A_full, fft_size)[:n]
    a = np.abs(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def coordinate_polish(a0, time_limit, verbose=False, seed=42):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:min(8, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            best_dst = -1
            best_gain = float('inf')
            candidates = rng.choice(n, size=min(n, 300), replace=False)

            for dst in candidates:
                if dst == src:
                    continue
                add = 0.0
                rem = 0.0
                for k2 in top_k[:6]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    if 0 <= d1 < n:
                        add += a[d1]
                    d2 = dst - k2i
                    if 0 <= d2 < n:
                        add += a[d2]
                    d3 = k2i - src
                    if 0 <= d3 < n:
                        rem += a[d3]
                    d4 = src - k2i
                    if 0 <= d4 < n:
                        rem += a[d4]
                gain = add - rem
                if gain < best_gain:
                    best_gain = gain
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst
            found = False
            for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.12]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                pass

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    rng = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = 2.0 * len(a_norm) * fast_autocorr(a_norm).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    n_parent = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            p_score = 2.0 * n_parent * fast_autocorr(parent).max()
            best_seq = parent.copy()
            best_score = p_score
            print(f"Parent: n={n_parent}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_parent = 1600
        best_seq = window_init(n_parent, 'sin4')
        best_score = 2.0 * n_parent * fast_autocorr(best_seq).max()

    # === Phase 1: Sharp mirror descent on parent ===
    remaining = time_budget - (time.monotonic() - t_global)

                                  restart_patience=2000, verbose=True, seed=1)

    # === Phase 2: Multiplicative update ===

    if t2 > 20:
        a_out, _ = multiplicative_update(best_seq, t2, top_k=5, verbose=True, seed=3)
        try_update(a_out, f"mul_n{len(best_seq)}")

                                  restart_patience=2000, verbose=True, seed=5)
        try_update(a_out, f"md2_n{len(best_seq)}")

    # === Phase 4: Window initializations at parent length ===

            break

                                  restart_patience=1500, verbose=False, seed=10)
        try_update(a_out, f"win_{w}_n{n_parent}")
