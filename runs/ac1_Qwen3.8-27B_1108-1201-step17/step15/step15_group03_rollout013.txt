import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    # Try solvers in order of expected quality
    candidates = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for s in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.SCS
    print("  Using solver (fallback): SCS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    """Build the (2n-1) x n matrix T where T[k,i] = a[i+k] if 0<=i+k<n else 0
    Actually, T[k, i] represents the coefficient of b[i] in conv(a,b)[k].
    conv(a,b)[k] = sum_i a[i] * b[k-i] (for valid indices)
    So T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    i.e., T[k, j] = a[k-j] if max(0, k-n+1) <= j <= min(k, n-1)
    """
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            T[k, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
    return T


def lp_solve_full(a, M, solver):
    """Solve the LP: max sum(b) s.t. T@b <= M, b >= 0, where T is built from a."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(20, min(200, int(remaining * 4)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n * it + n))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_full(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration, using ALL constraints."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_prev = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        b = lp_solve_full(y, M, solver)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Check for restart: if function increased, reset momentum
        sc_y_new = fast_conv_self(y_new).max()
        momentum = (t - 1.0) / (t + 2.0)
        if sc_y_new > fast_conv_self(y).max() * 1.001:
            momentum = 0.0  # restart

        a_new = y_new + momentum * (y_new - a_prev)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        a_prev = y
        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_A_matrix(a):
    """Build the (2n-1) x n constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def build_A_sparse(a):
    """Build sparse constraint matrix."""
    n = len(a)
    size = 2 * n - 1
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.concatenate([np.full(n, j) for j in range(n)])
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def lp_solve_hg(a, M, use_sparse=True):
    """Solve max sum(b) s.t. A@b <= M, b >= 0 using HiGHS."""
    n = len(a)
    size = 2 * n - 1
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    b_ub = np.full(size, M)

    if use_sparse:
        A = build_A_sparse(a)
    else:
        A = build_A_matrix(a)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lse_gradient_descent(a0, time_limit, verbose=True):
    """Smooth gradient descent using log-sum-exp approximation of max."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 18
    temps = np.geomspace(150.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(30, min(400, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.02 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, verbose=True, restart_patience=50):
    """LP best-response with Nesterov acceleration using HiGHS solver."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_prev = a.copy()
    stuck_counter = 0
    best_this_phase = best_score
    last_improve_t = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        b = lp_solve_hg(y, M, use_sparse=True)
        if b is None:
            b = lp_solve_hg(y, M, use_sparse=False)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum

        # Nesterov update with adaptive restart
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        sc_y_new = fast_conv_self(y_new).max()
        sc_y_old = M

        # Restart logic: reset momentum if no progress
        if sc_y_new >= sc_y_old * 0.999:
            stuck_counter += 1
        else:
            stuck_counter = 0

        if stuck_counter > restart_patience:
            momentum = 0.0
            stuck_counter = 0
        else:
            momentum = (t - 1.0) / (t + 2.0)

        a_new = y_new + momentum * (y_new - a_prev)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()

        a_prev = y
        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y

            if best_score < best_this_phase - 1e-6:

            momentum = 0.0
            stuck_counter = 0

            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

def coordinate_descent_polish(a0, time_limit, rng, verbose=True, max_rounds=50):
    """Coordinate descent with golden-section 1D sub-problems for each coordinate."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_rounds = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        round_improved = False
        n_improved = 0

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai_old = a[i]

            # Compute effect of changing a[i] by d on conv
            # conv[k] changes by:
            #   For k in [i, i+n-1], k != 2*i: delta = 2*d*a[k-i]
            #   For k = 2*i (if valid): delta = 2*d*a[i] + d^2
            c1 = np.zeros(size)
            c2 = np.zeros(size)

            k_lo = i
            k_hi = min(i + n - 1, size - 1)

            # Linear coefficients
            k_range = np.arange(k_lo, k_hi + 1)
            offsets = k_range - i  # these are 0, 1, ..., n-1 (clipped)
            c1[k_range] = 2.0 * a[offsets]

            # Fix the diagonal term k=2*i
            if 2 * i <= size - 1:
                c1[2 * i] = 2.0 * a[i]
                c2[2 * i] = 1.0

            # Golden section search for d
            # Constraint: a[i] + d >= 0 => d >= -a[i]
            d_lo = -ai_old * 0.999
            d_hi = max(0.0, norm / n) * 1.0

            if d_hi <= d_lo + 1e-12:
                d_hi = d_lo + max(0.1, norm / n)

            # Evaluate at golden section points
            c1d_c1 = c1 * d_lo + c2 * d_lo * d_lo
            c1d_dd = c1 * d_hi + c2 * d_hi * d_hi

            c_val = d_hi - gr * (d_hi - d_lo)
            dd_val = d_lo + gr * (d_hi - d_lo)
            f_c = np.max(conv + c1 * c_val + c2 * c_val * c_val)
            f_dd = np.max(conv + c1 * dd_val + c2 * dd_val * dd_val)

            for _ in range(40):
                if f_c < f_dd:
                    d_hi = dd_val
                    dd_val = c_val
                    f_dd = f_c
                    c_val = d_hi - gr * (d_hi - d_lo)
                    f_c = np.max(conv + c1 * c_val + c2 * c_val * c_val)
                else:
                    d_lo = c_val
                    c_val = dd_val
                    f_c = f_dd
                    dd_val = d_lo + gr * (d_hi - d_lo)
                    f_dd = np.max(conv + c1 * dd_val + c2 * dd_val * dd_val)

            d_opt = (d_lo + d_hi) / 2.0
            new_max_est = np.max(conv + c1 * d_opt + c2 * d_opt * d_opt)

            if new_max_est < best_score - 1e-10:
                a[i] = max(0.0, a[i] + d_opt)
                s_new = a.sum()
                if s_new < 1e-15:
                    a = a * norm / max(s_new, 1e-15)
                else:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score:
                    best_score = new_max
                    best_a = a.copy()
                    round_improved = True
                    n_improved += 1
                # Update conv for next coordinates
                conv = fast_conv_self(a)

        if not round_improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 4:
                break
            # Perturb to escape local minimum
            n_perturb = max(10, n // 30)
            idx_perturb = rng.choice(n, size=n_perturb, replace=False)
            a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
        else:
            no_improve_rounds = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD round {rnd} ({elapsed:.0f}s): best={best_score:.8f}, improved={n_improved}")

    return best_a, best_score


def peak_shave_pass(a0, time_limit, rng, verbose=True):
    """Targeted peak shaving: identify top peaks and reduce contributions from high-value coordinates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    for rnd in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        cur_max = conv.max()
        # Find top peaks
        peak_thresh = cur_max * 0.95
        peak_lags = np.where(conv >= peak_thresh)[0]

        if len(peak_lags) == 0:
            break

        round_improved = False

        # For each peak, try to reduce it by moving mass
        for kl in peak_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break

            # Find which coordinates contribute most to this peak
            # conv[kl] = sum_j a[j]*a[kl-j] for valid j
            contribs = np.zeros(n)
            for j in range(n):
                k_j = kl - j
                if 0 <= k_j < n:
                    contribs[j] = a[j] * a[k_j]

            # Sort by contribution (descending)
            top_idx = np.argsort(-contribs)[:min(20, n)]

            for i in top_idx:
                ai = a[i]
                if ai < norm * 1e-8:
                    continue

                # Try reducing a[i] and adding to a random low-contribution coordinate
                low_idx = np.argsort(contribs)[:10]
                for j in low_idx:
                    if j == i or contribs[j] > contribs[i] * 0.5:
                        continue

                    # Try a range of transfer amounts
                    best_d = 0.0
                    best_v = cur_max
                    for d in np.linspace(0.01, min(ai * 0.5, norm / (2 * n)), 10):
                        new_a = a.copy()
                        new_a[i] -= d
                        new_a[j] += d
                        new_a = np.maximum(new_a, 0.0)
                        s_new = new_a.sum()
                        new_a = new_a * norm / s_new
                        v = fast_conv_self(new_a).max()
                        if v < best_v:
                            best_v = v
                            best_d = d

                    if best_d > 0 and best_v < best_score - 1e-10:
                        a[i] -= best_d
                        a[j] += best_d
                        a = np.maximum(a, 0.0)
                        s_new = a.sum()
                        if s_new > 1e-15:
                            a = a * norm / s_new
                        new_max = fast_conv_self(a).max()
                        if new_max < best_score:
                            best_score = new_max
                            best_a = a.copy()
                            round_improved = True
                            conv = fast_conv_self(a)

        if not round_improved:
            # Perturb
            n_perturb = max(5, n // 50)
            idx_p = rng.choice(n, size=n_perturb, replace=False)
            a[idx_p] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new

        if verbose and rnd % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave rnd={rnd} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.sinc(2.0 * (x - 0.5)) ** 2 + 0.05
    else:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.2)
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    """Evaluate a sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 80.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq.copy(), t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse) - all constraints via HiGHS
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov (HiGHS) n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t2, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: CD Polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = coordinate_descent_polish(best_seq.copy(), t3, rng, verbose=True)
        try_update(a_out, "cd1")

    # Phase 4: Second LP Nesterov round
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t4, verbose=True, restart_patience=30)
        try_update(a_out, "lp2")

    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: Peak Shaving n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = peak_shave_pass(best_seq.copy(), t5, rng, verbose=True)
        try_update(a_out, "peak1")

    # Phase 6: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t6 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 6: Longer length n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t6a = t6 * 0.6
        a_out, _ = nesterov_lp_optimize(init_long, t6a, verbose=False)
        try_update(a_out, f"lp_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.3, 40.0)
            a_out2, _ = coordinate_descent_polish(best_seq.copy(), t6b, rng, verbose=False)
            try_update(a_out2, "cd_long")

    # Phase 7: Try shorter length
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(800, n_base - 400)
    t7 = min(remaining * 0.10, 80.0)
    print(f"\n=== Phase 7: Shorter length n={n_short} (budget={t7:.0f}s) ===")
    if t7 > 20 and remaining > 60:
        init_short = interpolate_seq(best_seq, n_short)
        a_out, _ = nesterov_lp_optimize(init_short, t7 * 0.7, verbose=False)
        try_update(a_out, f"lp_n{n_short}")

    # Phase 8: Final CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    t8 = min(remaining, 120.0)
    print(f"\n=== Phase 8: Final CD polish n={n_final} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_out, _ = coordinate_descent_polish(best_seq.copy(), t8, rng, verbose=True, max_rounds=100)
        try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
