def fast_conv_self(a):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Brief LSE gradient descent warmup
2. LP best-response with Nesterov acceleration and adaptive screening
3. Directed peak-reduction coordinate descent with analytic 1D search
4. Fine single-coordinate polishing
5. Optional multi-length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fft_size_for(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def fast_conv_self(a):
    n = len(a)
    fs = fft_size_for(n)
    A = rfft(a, fs)
    return irfft(A * A, fs)[:2 * n - 1]


def fast_conv_cross(a, b):
    n = len(a)
    fs = fft_size_for(n)
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A * B, fs)[:2 * n - 1]


def obj_val(a):
    """Compute 2*n*max(conv)/(sum)^2 for numpy array a (non-negative, sum>0)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_conv_self(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if not a_list:
            return float('inf')
        for v in a_list:
            if v < 0 or not np.isfinite(v):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a, target_sum=1.0):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a) * target_sum
    return a * target_sum / s


def lse_gd(a0, time_limit):
    """LSE-smoothed gradient descent for brief warmup."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = obj_val(a)
    t0 = time.monotonic()
    fs = fft_size_for(n)
    size = 2 * n - 1
    
    # Adaptive temperature schedule
    n_iters = 2000
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_conv_self(a)
        mx = conv.max()
        if mx < 1e-15:
            break
        # Temperature: start high (explore), end low (exploit)
        frac = it / n_iters
        T = mx * (0.15 * (1 - frac) + 0.01)
        shifted = (conv - mx) / T
        shifted = np.clip(shifted, -100, 0)
        w = np.exp(shifted)
        w /= w.sum()
        
        # Gradient via FFT
        W = rfft(w, fs)
        A = rfft(a, fs)
        # grad of sum_k w[k] * (a*a)[k] w.r.t. a
        g1 = irfft(np.conj(A) * W, fs)[:n]
        g2 = irfft(A * W, fs)[:n]
        grad = 2.0 * (g1 + g2)
        
        # Step with adaptive size
        lr = 0.5 / (np.sqrt(n) * (1.0 + 0.01 * it))
        a = np.maximum(a - lr * grad, 0.0)
        a = normalize(a)
        
        sc = obj_val(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def lp_best_response(a, M, active_indices=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_indices is not None and len(active_indices) < size:
        # Build only active rows of the convolution matrix
        # (a*b)[k] = sum_i a[i] * b[k-i] for valid i
        # For a given k, the constraint is: sum_i a[i] * b[k-i] <= M
        # which is a linear constraint on b
        # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
        # So the constraint row k is: sum_j T[k,j] * b[j] <= M
        # T[k,j] = a[k-j] when 0 <= k-j < n, i.e., j in [k-n+1, k] (clipped to [0,n-1])
        
        # For efficiency, build the matrix only for active rows
        rows = np.zeros((len(active_indices), n))
        for ii, k in enumerate(active_indices):
            # T[k,j] = a[k-j] for j in max(0,k-n+1) .. min(n-1,k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                rows[ii, j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi + 1)]
        constraints = [rows @ b_var <= M]
    else:
        # Full matrix (slow for large n but needed sometimes)
        T = np.zeros((size, n))
        for k in range(size):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                T[k, j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi + 1)]
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=get_solver(), verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_nesterov(a0, time_limit):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = obj_val(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    
    no_improve = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-15:
            break
        
        # Adaptive constraint screening with hysteresis
        # Include lags where conv >= 0.85*M (generous threshold)
        threshold = M * 0.85
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 20:
            threshold = M * 0.60
            active = np.where(conv_y >= threshold)[0]
        if len(active) < 5:
            active = None  # Use full set
        
        # Solve LP
        b = lp_best_response(y, M, active)
        if b is None:
            b = lp_best_response(y, M, None)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum  # Normalize to probability
        
        # Verify: check if dropped constraints are violated
        if active is not None and len(active) < size:
            cross = fast_conv_cross(y, g)
            violated = np.where(cross > M * 1.005)[0]
            if len(violated) > 5:
                # Re-solve with full constraints
                b_full = lp_best_response(y, M, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    g = b_full / b_full.sum()
        
        # Nesterov update with adaptive parameters
        alpha = 2.0 / (t + 2.0)
        # Add slight damping for stability
        alpha = min(alpha, 0.3)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new)
        
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = obj_val(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        sc_a = obj_val(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0
        
        # Restart if stuck
        if no_improve > 500 and t % 500 == 0:
            y = best_a.copy()
            no_improve = 0
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

        g2 = irfft(A * W, fs)[:n]
        grad = 2.0 * (g1 + g2)
        lr = 0.4 / (np.sqrt(n) * (1.0 + 0.005 * it))
        a = np.maximum(a - lr * grad, 0.0)
        a = normalize(a)
        sc = obj_val(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def build_conv_matrix_rows(a, k_indices):
    """Build rows of the convolution matrix for specific lag indices."""
    n = len(a)
    rows = np.zeros((len(k_indices), n))
    for ii, k in enumerate(k_indices):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            rows[ii, j_lo:j_hi + 1] = a[k - np.arange(j_lo, j_hi + 1)]
    return rows


def lp_min_max_cross(a, k_indices=None):
    """
    Solve: min t s.t. (a*b)[k] <= t for k in k_indices (or all k), sum(b)=1, b>=0.
    Returns (b, t) or (None, None).
    """
    n = len(a)
    size = 2 * n - 1
    if k_indices is None:
        k_indices = np.arange(size)
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    T = build_conv_matrix_rows(a, k_indices)
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=get_solver(), verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    s = val.sum()
    if s < 1e-15:
        return None, None
    val = val / s
    return val, float(t_var.value)


def lp_original(a, M, k_indices=None):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for k in k_indices, b>=0."""
    n = len(a)
    size = 2 * n - 1
    if k_indices is None:
        k_indices = np.arange(size)
    
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix_rows(a, k_indices)
    constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=get_solver(), verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    s = val.sum()
    if s < 1e-15:
        return None
    return val / s


def lp_nesterov_v2(a0, time_limit):
    """LP best-response with Nesterov using min-max cross formulation."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = obj_val(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    no_improve = 0
    use_orig_lp = False
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-15:
            break
        
        # Adaptive constraint screening
        thresh = M * 0.82
        active = np.where(conv_y >= thresh)[0]
        if len(active) < 15:
            active = np.where(conv_y >= M * 0.55)[0]
        if len(active) < 5:
            active = None
        
        # Solve LP (use min-max cross formulation)
        if active is not None:
            b, t_val = lp_min_max_cross(y, active)
        else:
            b, t_val = lp_min_max_cross(y, None)
        
        if b is None:
            # Fallback to original formulation
            if active is not None:
                b = lp_original(y, M, active)
            else:
                b = lp_original(y, M, None)
            t_val = None
        
        if b is None:
            continue
        
        g = b  # Already normalized
        
        # Verify dropped constraints
        if active is not None and len(active) < size:
            cross = fast_conv_cross(y, g)
            violated = np.where(cross > M * 1.01)[0]
            if len(violated) > 8 and len(violated) < size:
                b2, _ = lp_min_max_cross(y, np.unique(np.concatenate([active, violated])))
                if b2 is not None:
                    g = b2
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        alpha = min(alpha, 0.25)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new)
        
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        y = y_new
        a = a_new
        
        sc_y = obj_val(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        sc_a = obj_val(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0
        
        if no_improve > 300 and t % 300 == 0:
            y = best_a.copy()
            a = best_a.copy()
            no_improve = 0
        
        if t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest v2 t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def directed_cd(a0, time_limit, rng):
    """Directed coordinate descent targeting peaks."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = obj_val(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_moves = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        peak_thresh = cur_max * 0.88
        peak_lags = np.where(conv >= peak_thresh)[0]
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-10:
                continue
            
            # Build the "remove i" vector
            v = np.zeros(size)
            k_start_i = i
            k_end_i = min(size, i + n)
            v[k_start_i:k_end_i] -= 2.0 * a[0:k_end_i - k_start_i]
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            # Generate candidates
            candidates = []
            for _ in range(6):
                candidates.append(int(rng.integers(n)))
            for kl in peak_lags[:6]:
                jj = kl - i
                if 0 <= jj < n and jj != i:
                    candidates.append(jj)
                jj2 = i - kl
                if 0 <= jj2 < n and jj2 != i:
                    candidates.append(jj2)
            # Add neighbors of peak positions
            for kl in peak_lags[:3]:
                for off in [-2, -1, 1, 2]:
                    jj = kl - i + off
                    if 0 <= jj < n and jj != i:
                        candidates.append(jj)
            
            # Deduplicate and limit
            seen = set()
            unique_cands = []
            for c in candidates:
                if c != i and c not in seen:
                    seen.add(c)
                    unique_cands.append(c)
            unique_cands = unique_cands[:16]
            
            for j in unique_cands:
                if j == i:
                    continue
                
                # Build "add j" and combine
                v_total = v.copy()
                k_start_j = j
                k_end_j = min(size, j + n)
                v_total[k_start_j:k_end_j] += 2.0 * a[0:k_end_j - k_start_j]
                
                # Find best delta via sampling
                max_d = ai * 0.95
                if max_d < 1e-12:
                    continue
                
                deltas = np.array([0.05, 0.15, 0.3, 0.5, 0.7, 0.85, 0.95]) * max_d
                for d in deltas:
                    nc = conv + d * v_total
                    # Quadratic corrections
                    if 2 * i < size:
                        nc[2 * i] += d * d
                    if 2 * j < size:
                        nc[2 * j] += d * d
                    # Cross term correction: if i+j is a valid lag
                    # The cross terms a[i]*a[j] appear at lag i+j
                    # When we change a[i]->a[i]-d, a[j]->a[j]+d:
                    # The product a[i]*a[j] changes to (a[i]-d)(a[j]+d) = a[i]*a[j] + a[i]*d - a[j]*d - d^2
                    # This appears at lag i+j and lag j+i (same thing)
                    # Actually at lag i+j: the terms are a[i]*a[j] (m=i, k-m=j) and a[j]*a[i] (m=j, k-m=i)
                    # Both are the same product, so total contribution is 2*a[i]*a[j] (or a[i]*a[j] if i==j)
                    # After change: 2*(a[i]-d)(a[j]+d) = 2*a[i]*a[j] + 2*a[i]*d - 2*a[j]*d - 2*d^2
                    # Linear: 2*d*(a[i]-a[j])
                    # This is already captured in v_total at position i+j!
                    # v_total[i+j] includes 2*a[k-j] from j-contribution (= 2*a[i]) and -2*a[k-i] from i-contribution (= -2*a[j])
                    # So v_total[i+j] = 2*a[i] - 2*a[j] = 2*(a[i]-a[j]) ✓
                    # The quadratic correction at i+j is: -2*d^2 (from the -d^2 term, times 2 for symmetry... 
                    # actually let's not worry about it, the linear approx is good enough for small d)
                    
                    vv = nc.max()
                    if vv < best_val - 1e-14:
                        best_val = vv
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize(a)
                new_conv = fast_conv_self(a)
                if new_conv.max() < conv.max() - 1e-12:
                    pass_improved = True
                    total_moves += 1
                sc = obj_val(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            # Perturb to escape local minima
            n_pert = max(5, n // 40)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a[idx_pert] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            a = normalize(a)
        else:
            no_improve = 0
        
        if total_moves % 300 == 0 and total_moves > 0:
            elapsed = time.monotonic() - t0
            print(f"    DirCD moves={total_moves} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def fine_polish(a0, time_limit, rng):
    """Fine single-coordinate adjustments with adaptive step sizes."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = obj_val(a)
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-8:
                continue
            
            saved = a.copy()
            # Try multiplicative adjustments
            for factor in [1.002, 0.998, 1.005, 0.995, 1.01, 0.99, 1.02, 0.98]:
                a[i] = ai * factor
                if a[i] < 0:
                    a[i] = 0.0
                a = normalize(a)
                sc = obj_val(a)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    step_count += 1
                    break
            a = saved  # Restore
            
            if improved:
                # After improvement, the next evaluation of conv is stale, break
                break
        
        if not improved:
            no_improve += 1
            if no_improve >= 10:
                # Try larger steps
                for i in perm[:n//4]:
                    if time.monotonic() - t0 > time_limit:
                        break
                    ai = a[i]
                    if ai < 1e-6:
                        continue
                    saved = a.copy()
                    for factor in [1.1, 0.9, 1.15, 0.85, 1.2, 0.8]:
                        a[i] = ai * factor
                        if a[i] < 0:
                            a[i] = 0.0
                        a = normalize(a)
                        sc = obj_val(a)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            step_count += 1
                            break
                    a = saved
                if not improved:
                    if no_improve >= 15:
                        break
                    # Perturb
                    n_pert = max(3, n // 60)
                    idx_pert = rng.choice(n, size=n_pert, replace=False)
                    a[idx_pert] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                    a = np.maximum(a, 0.0)
                    a = normalize(a)
            else:
                no_improve = 0
        else:
            no_improve = 0
            a = best_a.copy()
        
        if step_count % 100 == 0 and step_count > 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish steps={step_count} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 1.0 + 0.1 * np.cos(2 * np.pi * x)
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # Phase 1: Short LSE GD warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 90.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gd(best_seq, t1)
        try_update(a_out, "lse1")
    
    # Phase 2: LP-Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.32, 550.0)
    print(f"\n=== Phase 2: LP-Nesterov v2 n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_v2(best_seq, t2)
        try_update(a_out, "lp1")
    
    # Phase 3: Directed coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: Directed CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = directed_cd(best_seq, t3, rng)
        try_update(a_out, "dcd1")
    
    # Phase 4: LP-Nesterov round 2 (on best so far)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 4: LP-Nesterov round 2 (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lp_nesterov_v2(best_seq, t4)
        try_update(a_out, "lp2")
    
    # Phase 5: Directed CD round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Directed CD round 2 (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = directed_cd(best_seq, t5, rng)
        try_update(a_out, "dcd2")
    
    # Phase 6: Fine polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 6: Fine Polish (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = fine_polish(best_seq, t6, rng)
        try_update(a_out, "fine1")
    
    # Phase 7: Try slightly different length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        for dn in [200, -200]:
            if remaining < 60:
                break
            n_try = n_base + dn
            if n_try < 400:
                continue
            t7 = min(remaining * 0.4, 120.0)
            print(f"\n=== Phase 7: Try n={n_try} (budget={t7:.0f}s) ===")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lp_nesterov_v2(init_try, min(t7 * 0.6, 70.0))
            try_update(a_out, f"lp_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out2, _ = directed_cd(best_seq, min(remaining * 0.3, 40.0), rng)
                try_update(a_out2, f"dcd_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
    
    # Final: one more fine polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Final Polish (budget={remaining:.0f}s) ===")
        a_out, _ = fine_polish(best_seq, min(remaining * 0.8, 60.0), rng)
        try_update(a_out, "final")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
