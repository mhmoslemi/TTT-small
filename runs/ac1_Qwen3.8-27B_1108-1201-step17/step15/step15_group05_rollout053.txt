def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to(a, target_sum):
    """Normalize a to have the given sum."""
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s

def lp_best_response(a_fixed, M, active_set, solver):
    """
    Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0.
    Returns (b, duals) or (None, None).
    """
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    
    # Build constraint matrix efficiently
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            mat[idx, lo:hi] = a_fixed[k - (lo:hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0, max_active_frac=0.5):
    """Nesterov-accelerated LP best-response with dual-guided active set."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    n_iters = 0
    t = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set selection
        if active_set is None or t % 30 == 0:
            # Use dual values to prioritize constraints
            if active_set is not None and duals is not None:
                # Keep high-dual constraints, add new peaks
                max_dual = duals.max() if len(duals) > 0 else 1.0
                tol = 0.02 * max_dual
                keep_mask = duals > tol
                kept = active_set[keep_mask]
                # Add current peaks
                tol_peak = max(0.02, 0.08 * np.exp(-t / 200.0))
                peak_set = np.where(conv_y >= M_curr * (1.0 - tol_peak))[0]
                # Ensure top peaks are included
                top_k = np.argsort(conv_y)[-max(5, n // 100):]
                active_set = np.unique(np.concatenate([kept, peak_set, top_k]))
                # Limit size
                max_size = int(max_active_frac * (2 * n - 1))
                if len(active_set) > max_size:
                    # Keep based on conv_y values
                    active_set = active_set[np.argsort(conv_y[active_set])[-max_size:]]
            else:
                tol = max(0.02, 0.1 * np.exp(-t / 100.0))
                active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
                top_k = np.argsort(conv_y)[-max(5, n // 100):]
                active_set = np.unique(np.concatenate([active_set, top_k]))

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            # Try with all constraints
            if t % 5 == 0:
                active_set = np.arange(2 * n - 1)
                b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        # Check for violations and expand
        size = 2 * n - 1
        complement = np.setdiff1d(np.arange(size), active_set)
        if len(complement) > 0:
            # Check violations
            viol = np.zeros(len(complement))
            for idx, k in enumerate(complement):
                lo = max(0, k - n + 1)
                hi = min(n, k + 1)
                if lo < hi:
                    viol[idx] = np.dot(y[k - (lo:hi)], b[lo:hi])
            viol_idx = np.where(viol > M_curr * (1 + 1e-6))[0]
            if len(viol_idx) > 0:
                new_active = np.union1d(active_set, complement[viol_idx])
                if len(new_active) > 0.9 * size:
                    new_active = np.arange(size)
                b, duals = lp_best_response(y, M_curr, new_active, solver)
                if b is not None:
                    active_set = new_active

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
            stall_count = 0

        # Restart if stalled
        if stall_count > 50:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.10f}")

    return best_a, best_M
