import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    import cvxpy as cp
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")

        except Exception:

    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_clipped = np.minimum(a, 1000.0)
    n = len(a_clipped)
    conv = fast_autocorr(a_clipped)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s ** 2))


def build_conv_matrix(a, top_k_constraints=None):
    """Build sparse or dense convolution matrix. Returns (T, active_rows)."""
    n = len(a)
    size = 2 * n - 1
    if top_k_constraints is not None and top_k_constraints < size:
        # Use top-k largest potential constraints
        T = np.zeros((min(top_k_constraints, size), n))
        active_rows = np.arange(size)
    else:
        T = np.zeros((size, n))
        active_rows = np.arange(size)
    for j in range(n):
        lo = 0
        hi = min(size, j + n)
        T[lo:hi, j] = a[j]
    return T, active_rows


def lp_best_response(a, M, solver, subsample=None):
    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1

    # Build constraint matrix: conv(a,b)[k] = sum_j a[j]*b[k-j] (valid indices)
    # As linear in b: row k has a[j] at position j for valid j
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a

    # Subsample constraints if requested
    if subsample is not None and subsample < size:
        # Always include the middle (most constrained) and random others
        mid = n - 1
        other = np.random.choice(np.setdiff1d(np.arange(size), [mid]),
                                  size=min(subsample-1, size-1), replace=False)
        rows = np.concatenate([[mid], other])
        T_sub = T[rows]
    else:
        T_sub = T

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, subsample=None):
    """LP best-response with Nesterov acceleration and adaptive restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    patience = 0
    restart_count = 0
    momentum_t = 1.0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        b = lp_best_response(y, M_curr, solver, subsample=subsample)
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue

        stuck_count = 0
        g = b * norm
        n_iters += 1

        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * (g / g.sum())
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if momentum_t < 3.0:
            a_new = y_new.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        # Adaptive restart
        if patience > 60 and n_iters > 20:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, restarts={restart_count}")

    return best_a, best_score


def spectral_shaping(a0, time_limit, verbose=True, n_iter_max=5000):
    """FFT-based spectral shaping: alternate between flattening |DFT|^2 and projecting to simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # FFT size
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    alpha = 0.1  # Shrinkage parameter
    n_iters = 0
    patience = 0

    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        n_iters += 1

        # Compute FFT
        A = fft(a, L)
        mag2 = np.abs(A) ** 2

        # Target: make non-DC bins equal
        target = np.mean(mag2[1:])
        if target < 1e-30:
            break

        # Shrink mag2 towards target
        mag2_new = mag2.copy()
        mag2_new[1:] = np.sqrt((1 - alpha) * mag2[1:] + alpha * target)
        mag2_new[0] = mag2[0]  # Keep DC

        # New magnitude
        mag_new = np.sqrt(mag2_new)
        # Keep phase
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
        A_new = mag_new * phase

        # Inverse FFT
        a_new = np.real(ifft(A_new, L))[:n]

        # Project to simplex: a >= 0, sum(a) = 1
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Blending (to ensure stability)
        blend = 0.5
        a_try = (1 - blend) * a + blend * a_new
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            a_try = a.copy()
        else:
            a_try = a_try / s_try

        conv_try = fast_autocorr(a_try)
        score_try = conv_try.max()

        if score_try < best_score - 1e-12:
            best_score = score_try
            best_a = a_try.copy()
            a = a_try
            patience = 0
        else:
            patience += 1
            # Reduce alpha to be more conservative
            if patience % 50 == 0:
                alpha *= 0.8
                if alpha < 0.01:
                    alpha = 0.1
                    # Perturb to escape
                    noise = rng.normal(0, 0.001, n)
                    a_pert = a + noise
                    a_pert = np.maximum(a_pert, 0)
                    sp = a_pert.sum()
                    if sp > 1e-15:
                        a = a_pert / sp
                    patience = 0
                    if verbose:
                        print(f"    SS perturbed at iter={n_iters}, alpha reset")

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SS iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, alpha={alpha:.4f}")

    return best_a, best_score


def kkt_coordinate_descent(a0, time_limit, verbose=True):
    """KKT-driven coordinate descent: at optimum, for active lags, gradient should vanish
    on the tangent space of the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(31415)
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if n_passes % 200 == 0 and no_improve > 300:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()

        # Find active lags (top few)
        top_k = min(20, size)
        active_idx = np.argsort(conv)[-top_k:]
        active_threshold = max_val * 0.95
        active_idx = active_idx[conv[active_idx] >= active_threshold]

        # Compute gradient of max conv: weighted average of active gradient vectors
        # grad_k[j] = d conv[k] / d a[j]
        # For conv(k) = sum_i a[i]*a[k-i] (valid):
        # d conv(k)/d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, j!=k/2)
        #                    or 2*a[k-j] if j = k/2 (and k even)

        grad = np.zeros(n)
        for k in active_idx:
            k = int(k)
            # d conv(k)/d a[j] for each j
            # a[j] appears in conv(k) when: a[j]*a[k-j] (if k-j in [0,n-1])
            # and a[k-j]*a[j] = same term, so derivative is 2*a[k-j] if j = k-j,
            # otherwise a[k-j] + a[j-k]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                other = k - j
                if 0 <= other < n:
                    if j == other:
                        grad[j] += 2.0 * a[other]
                    else:
                        grad[j] += a[other]
                # Also a[j] appears as the "second" factor: a[other]*a[j] where other+j=k
                # This is the same term, already counted above.
            # Wait, I need to be more careful.
            # conv(k) = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
            # d/d a[j]: the term a[j]*a[k-j] (if k-j valid) + a[k-j]*a[j] (if k-j valid, and k-j != j)
            # Actually: if j is a valid first index (0<=j<n and 0<=k-j<n), contribution is a[k-j]
            # If j is a valid second index (0<=j<n and 0<=k-j<n, and k-j != j), contribution is a[k-j]
            # If j = k/2 (k even), contribution is 2*a[j]

        # Gradient step on simplex
        # Project: minimize (a + eta*grad)^T(a + eta*grad) s.t. a>=0, sum=1
        # Simple approach: a_new = softmax(-eta * grad) or multiplicative update
        eta = 0.01
        log_a = np.log(a + 1e-30)
        a_new = np.exp(log_a - eta * grad)
        a_new = a_new / a_new.sum()

        # Try multiple step sizes
        best_eta_a = a
        best_eta_score = fast_autocorr(a).max()
        for et in [0.001, 0.005, 0.01, 0.05, 0.1]:
            la = np.log(a + 1e-30)
            at = np.exp(la - et * grad)
            at = at / at.sum()
            ct = fast_autocorr(at).max()
            if ct < best_eta_score - 1e-12:
                best_eta_score = ct
                best_eta_a = at

        if best_eta_score < best_score - 1e-12:
            best_score = best_eta_score
            best_a = best_eta_a.copy()
            a = best_eta_a
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                # Perturb
                noise = rng.normal(0, 0.001, n)
                a_p = a + noise
                a_p = np.maximum(a_p, 0)
                sp = a_p.sum()
                if sp > 1e-15:
                    a = a_p / sp
                no_improve = 0

        if verbose and n_passes % 100 == 0:

            print(f"    KKT-CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

def pair_cd_polish(a0, time_limit, verbose=True):
    """Fast pair coordinate descent: try moving mass between pairs of coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(271)
    n_passes = 0
    no_improve = 0

    window = max(5, n // 100)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if n_passes % 50 == 0 and no_improve > 100:
            break

        pass_improved = False
        perm = rng.permutation(n)

        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-12:
                continue

            for offset in range(1, window + 1):
                for j_dir in [(i + offset) % n, (i - offset) % n]:
                    j = j_dir
                    if j == i:
                        continue
                    # Try moving small fraction of mass from i to j
                    for frac in [0.1, 0.3, 0.5]:
                        d = a[i] * frac
                        if d < 1e-14:
                            continue
                        a[i] -= d
                        a[j] += d
                        # Normalize
                        s_cur = a.sum()
                        a = a / s_cur
                        new_conv = fast_autocorr(a)
                        new_max = new_conv.max()
                        if new_max < best_score - 1e-12:
                            best_score = new_max
                            best_a = a.copy()
                            conv = new_conv
                            pass_improved = True
                            no_improve = 0
                            break
                        else:
                            # Revert
                            a[i] += d
                            a[j] -= d
                            s_cur = a.sum()
                            a = a / s_cur

                    if pass_improved:
                        break
                if pass_improved and offset > window // 2:
                    break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                # Perturb
                noise = rng.normal(0, 0.002, n)
                a_p = a + noise
                a_p = np.maximum(a_p, 0)
                sp = a_p.sum()
                if sp > 1e-15:
                    a = a_p / sp
                no_improve = 0

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    t4 = remaining * 0.15
    print(f"\n=== Phase 4: KKT-CD n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = kkt_coordinate_descent(best_seq, t4, verbose=True)
        do_try_update(a_out, f"kkt_n{n_best}")

    # Phase 5: Spectral + LP alternating (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5_spec = remaining * 0.05
    t5_lp = remaining * 0.05
    print(f"\n=== Phase 5: Spec+LP alt n={n_best} (budget={t5_spec+t5_lp:.0f}s) ===")
    if t5_spec > 10:
        a_out, _ = spectral_shaping(best_seq, t5_spec, verbose=False)
        do_try_update(a_out, f"spec2_n{n_best}")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_out, _ = nesterov_lp_optimize(best_seq, remaining * 0.6, solver, verbose=False, subsample=None)
        do_try_update(a_out, f"lp3_n{n_best}")

    # Phase 6: Pair CD polish (7% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: Pair CD n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = pair_cd_polish(best_seq, t6, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 7: Final LP + polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t7a = min(remaining * 0.5, 120.0)
        print(f"\n=== Phase 7a: Final LP n={n_best} (budget={t7a:.0f}s) ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t7a, solver, verbose=False, subsample=None)
        do_try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            n_best = len(best_seq)
            t7b = remaining * 0.7
            print(f"\n=== Phase 7b: Final CD n={n_best} (budget={t7b:.0f}s) ===")
            a_out, _ = pair_cd_polish(best_seq, t7b, verbose=False)
            do_try_update(a_out, "cd_final")

    # Phase 8: Try a different n if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_alt = n_base + 400  # Try n+400
        print(f"\n=== Phase 8: Alt n={n_alt} (budget={remaining:.0f}s) ===")
        # Interpolate parent to new n
        a_alt = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, n_base), best_seq)
        a_alt = np.maximum(a_alt, 0)
        a_alt = a_alt / a_alt.sum()

        t8a = remaining * 0.5
        a_out, _ = nesterov_lp_optimize(a_alt, t8a, solver, verbose=True, subsample=2000)
        do_try_update(a_out, f"lp_alt_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out, _ = spectral_shaping(best_seq if len(best_seq) == n_base else a_alt,
                                        remaining * 0.4, verbose=False)
            n_check = len(a_out)
            if n_check == n_base:
                do_try_update(a_out, f"spec_alt")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0] * 100

    return [float(x) for x in best_seq]
