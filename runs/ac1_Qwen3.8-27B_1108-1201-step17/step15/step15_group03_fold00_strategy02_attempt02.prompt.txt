def evaluate_sequence(sequence: list[float]) -> float:
    """

    Returns np.inf if the input is invalid.

        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    # Protect against negative numbers

        return np.inf

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def momentum_subgradient(a0, time_limit, verbose=True):
    """Subgradient method for the max function with momentum and non-negativity projection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()

    momentum = np.zeros(n)
    beta = 0.9
    lr0 = 1.0 / np.sqrt(n)

    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        # Smooth subgradient: weight each lag by soft assignment to max
        temperature = max(0.1, max_c * 0.05 / (1.0 + 0.01 * it))
        shifted = (conv - max_c) / temperature
        shifted = np.maximum(shifted, -50.0)
        w = np.exp(shifted)
        w = w / w.sum()

        # Gradient of sum w[k] * conv[k] with respect to a
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        # Polyak step size
        lr = lr0 * 1.0 / (1.0 + 0.001 * it)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
