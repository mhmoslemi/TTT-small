"""
Minimizes 2*n*max(np.convolve(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with LP oracle (full constraint set, 2n-1 constraints)
2. Log-sum-exp smoothed gradient descent for grinding
3. Adaptive subgradient with backtracking
4. Multi-start perturbation restarts
Key improvements: full-constraint LP for reliability, smoothed gradient
for faster local convergence, and better time allocation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    return _solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    conv = fast_autocorr(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)

def project_sum(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_oracle(a, M, solver):
    n = len(a)
    T = build_T(a)
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b.value, dtype=np.float64)
    return np.maximum(b_val, 0.0)

def fw_optimize(a0, time_limit, solver, verbose=False):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = project_sum(a0.copy(), S)
    best_a = a.copy()
    best_score = score_of(a)
    t_start = time.monotonic()
    stale = 0
    n_iter = 0

    for t in range(1, 10000000):
        if time.monotonic() - t_start > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        if M < 1e-14:
            break

        b = lp_oracle(a, M, solver)
        if b is None:
            stale += 1
            if stale > 30:
                break
            continue
        stale = 0

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b * (S / b_sum)

        gamma = 2.0 / (t + 2.0)
        a_new = (1.0 - gamma) * a + gamma * b
        a = project_sum(a_new, S)
        n_iter += 1

        sc = score_of(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        if verbose and n_iter % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    FW iter={n_iter} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def smoothed_gradient(a0, time_limit, verbose=False):

    S = np.sqrt(2.0 * n)
    a = project_sum(a0.copy(), S)

    t_start = time.monotonic()

    step = 0.5 / np.sqrt(n)
    patience = 0

        if time.monotonic() - t_start > time_limit:
            break
        if patience > 3000:
            break
        if step < 1e-10:
            break

        conv = fast_autocorr(a)
        conv_max = conv.max()
        if conv_max < 1e-14:
            break

        conv_shifted = conv - conv_max
        exp_vals = np.exp(beta * conv_shifted)
        wsum = exp_vals.sum()
        if wsum < 1e-15:
            break
        weights = exp_vals / wsum

        w_view = np.lib.stride_tricks.sliding_window_view(weights, n)
        grad = 2.0 * (w_view @ a)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        a_new = a - step * grad
        a_new = project_sum(a_new, S)

        sc = score_of(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
            step = min(step * 1.03, 1.0)
        else:
            patience += 1
            step *= 0.97

        if it % 1000 == 0 and beta < 100.0:
            beta *= 1.3

        if verbose and it % 2000 == 0:

            print(f"    SmoothGrad it={it} ({elapsed:.0f}s): best={best_score:.10f}, beta={beta:.1f}, step={step:.6f}")

    return best_a, best_score

    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = project_sum(a0.copy(), S)
    best_a = a.copy()
    best_score = score_of(a)
    t_start = time.monotonic()
    step = 2.0 / n
    patience = 0

    for it in range(1, 10000000):
        if time.monotonic() - t_start > time_limit:
            break
        if patience > 5000:
            break
        if step < 1e-10:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        # Use top peaks with softmax weighting
        k_top = max(3, n // 200)
        top_idx = np.argsort(conv)[-k_top:]
        top_vals = conv[top_idx]
        w = np.exp(10.0 * (top_vals - M_curr) / max(M_curr * 0.01, 1e-10))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            if wi < 1e-6:
                continue
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                grad[j] += wi * 2.0 * a[k - j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        a_new = a - step * grad
        a_new = project_sum(a_new, S)
        sc = score_of(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
            step = min(step * 1.02, 2.0)
        else:
            patience += 1
            step *= 0.96

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score

def propose_candidate():
    t_global = time.monotonic()
    time_budget = 1770.0

    solver = get_solver()
    print(f"Solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            n = len(p)
            S = np.sqrt(2.0 * n)
            best_seq = p * (S / p.sum())
            best_score = score_of(best_seq)
            print(f"Parent: n={n}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n = 1600
        x = np.linspace(0.0, 1.0, n)
        best_seq = 0.5 + 0.3 * np.cos(2.0 * np.pi * x) + 0.15 * np.cos(4.0 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        S = np.sqrt(2.0 * n)
        best_seq = best_seq * (S / best_seq.sum())
        best_score = score_of(best_seq)
        print(f"Default init: n={n}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = a * (S / s)
        sc = score_of(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={n}, t={elapsed:.0f}s)")

    # === Phase 1: FW main run ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 700.0)
    print(f"\n=== Phase 1: FW n={len(best_seq)}, budget={t1:.0f}s ===")
    a_out, _ = fw_optimize(best_seq, t1, solver, verbose=True)
    try_update(a_out, "fw1")

    # === Phase 2: Smoothed gradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: SmoothGrad budget={t2:.0f}s ===")
    if t2 > 30:
        a_out, _ = smoothed_gradient(best_seq, t2, verbose=True)
        try_update(a_out, "smooth1")

    # === Phase 3: FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: FW round 2, budget={t3:.0f}s ===")
    if t3 > 30:
        a_out, _ = fw_optimize(best_seq, t3, solver, verbose=True)
        try_update(a_out, "fw2")

    # === Phase 4: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: SubGrad budget={t4:.0f}s ===")
    if t4 > 20:
        a_out, _ = subgradient_polish(best_seq, t4, verbose=True)
        try_update(a_out, "subg1")

    # === Phase 5: Perturbation + FW restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Perturbations, remaining={remaining:.0f}s ===")
    rng = np.random.default_rng(12345)
    n_pert = 3
    t5_total = min(remaining * 0.40, 250.0)
    t5_each = t5_total / n_pert
    for r in range(n_pert):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t5 = min(t5_each, remaining - 30)
        if t5 < 20:
            break
        scale = 0.015 + 0.008 * r
        noise = 1.0 + scale * rng.standard_normal(len(best_seq))
        pert = best_seq * np.maximum(noise, 0.3)
        a_out, _ = fw_optimize(pert, t5, solver, verbose=False)
        try_update(a_out, f"pert{r}")

    # === Phase 6: Final smooth gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 6: Final polish, budget={remaining-10:.0f}s ===")
        a_out, _ = smoothed_gradient(best_seq, remaining - 15, verbose=False)
        try_update(a_out, "final_smooth")

    # === Phase 7: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_out, _ = subgradient_polish(best_seq, remaining - 10, verbose=False)
        try_update(a_out, "final_subg")

    if best_seq is not None:
        fs = score_of(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        best_seq = np.ones(1600) / 1600
        print("ERROR: returning fallback")

    return [float(x) for x in best_seq]
