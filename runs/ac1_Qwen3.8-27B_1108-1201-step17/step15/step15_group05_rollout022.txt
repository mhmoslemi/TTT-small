def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):

    s = a.sum()

    return 2.0 * n * conv.max() / (s * s)

def normalize_seq(a, n=None):
    """Normalize a to have non-negative entries and sum = sqrt(2n)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    if n is None:
        n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * np.sqrt(2.0 * n) / n
    else:
        a = a * np.sqrt(2.0 * n) / s
    return a


def build_A_matrix(a, k_indices, n):
    """
    Build constraint matrix A where A[row, j] = a[k - j] if 0 <= k-j < n.
    k_indices: array of k values (lag indices in [0, 2n-2]).
    Returns A of shape (len(k_indices), n).
    Uses FFT-based approach for speed when possible.
    """
    m = len(k_indices)
    A = np.zeros((m, n))
    for idx in range(m):
        k = k_indices[idx]
        # A[idx, j] = a[k-j] if 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[idx, lo:hi+1] = a[k - hi:k - lo + 1][::-1]
    return A


def solve_lp_best_response(a, M, active_set, solver, n):
    """
    Solve: max sum(b) s.t. A @ b <= M, b >= 0
    Uses the dual: min M * sum(y) s.t. A^T @ y >= 1, y >= 0
    Returns b (the primal solution).
    """
    import cvxpy as cp
    m = len(active_set)
    if m == 0:
        return None
    
    A = build_A_matrix(a, active_set, n)
    
    # Try dual (fewer variables if m < n)
    if m < n:
        y = cp.Variable(m, nonneg=True)
        # A^T @ y >= 1 means for each j: sum_k A[k,j]*y[k] >= 1
        # A^T is (n, m)
        constraints = [A.T @ y >= np.ones(n)]
        prob = cp.Problem(cp.Minimize(M * cp.sum(y)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try primal
            b_var = cp.Variable(n, nonneg=True)
            constraints_p = [A @ b_var <= M]
            prob_p = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints_p)
            try:
                prob_p.solve(solver=solver, verbose=False)
            except Exception:
                return None
            if prob_p.status not in ('optimal', 'optimal_inaccurate'):
                return None
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            if b_val.sum() < 1e-15:
                return None
            return b_val
    else:
        # Primal
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b_val.sum() < 1e-15:
            return None
        return b_val

        weights = weights / weights.sum()
        
        # Compute combined subgradient
        # d/d a[j] of conv(a,a)[k] = 2*a[k-j] if 0 <= k-j < n
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * w * a[k - j]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Projected gradient step
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 3.0)
        else:
            patience += 1
            step *= 0.995
            if step < 1e-10:
                break
        
        # Periodic random peak selection
        if t % 100 == 0:
            threshold = M_curr * 0.9
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 1:
                k_rand = candidates[np.random.randint(len(candidates))]
                grad_r = np.zeros(n)
                k_r = int(k_rand)
                lo = max(0, k_r - n + 1)
                hi = min(n - 1, k_r)
                for j in range(lo, hi + 1):
                    grad_r[j] = 2.0 * a[k_r - j]
                gr = np.linalg.norm(grad_r)
                if gr > 1e-15:
                    grad_r = grad_r / gr
                    a_t2 = a - 0.5 * step * grad_r
                    a_t2 = np.maximum(a_t2, 0.0)
                    s2 = a_t2.sum()
                    if s2 > 1e-15:
                        a_t2 = a_t2 * norm / s2
                        sc2 = fast_autocorr(a_t2).max()
                        if sc2 < best_score - 1e-13:
                            a = a_t2
                            best_score = sc2
                            best_a = a_t2.copy()
                            patience = 0
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")
    
    return best_a, best_score
