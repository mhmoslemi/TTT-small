    # np correlat ion: scipy.signal.correlate(weight, a, mode='full')[i + n - 1]
    # Or: sum_j a[j] * weight[j+i] = (a * weight_flipped)[i + n - 1] ... 
    
    # Let me just use the simpler approach: compute the gradient via FFT of the 
    # "outer product" interpretation.
    
    # Actually the cleanest way: 
    # grad_a = 2 * ifft(fft(weight_full) * conj(fft(a_full)))[:n]
    # where weight_full is padded to length 2n-1 and a_full is padded.
    
    # Wait, let me think again. 
    # d/d(a[i]) sum_k weight[k] * sum_j a[j]*a[k-j]
    # = sum_k weight[k] * sum_j (delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i})
    # = sum_k weight[k] * a[k-i] + sum_k weight[k] * a[k-i]  (second term: k-j=i, so j=k-i)
    # Wait, that's the same thing twice?
    # = 2 * sum_k weight[k] * a[k-i] (for valid indices)
    
    # So grad_a[i] = 2 * sum_k weight[k] * a[k-i] where 0 <= k-i < n
    # = 2 * (cross_correlation of weight and a at lag i)
    
    # Using FFT: if we compute c = irfft(fft(weight, N) * conj(fft(a, N)), N) where N >= 2n-1
    # then c[i] = sum_k weight[k] * a[k-i] (with appropriate wrapping)
    
    # Hmm, let me just verify with a small example and use a direct computation for small n,
    # and FFT for large n.
    
    # For the FFT approach:
    # Let W = rfft(weight_padded, fft_size) where weight_padded has length fft_size
    # Let A = rfft(a_padded, fft_size) where a_padded has length fft_size
    # Then the circular cross-correlation is:
    # xcorr[i] = sum_k weight_padded[k] * a_padded[k-i mod fft_size]
    # = irfft(W * conj(A), fft_size)
    
    # If fft_size >= 2n-1, then for i in [0, n-1], xcorr[i] = sum_k weight[k] * a[k-i]
    # (where a[k-i] = 0 if k-i < 0 or k-i >= n)
    
    # Actually I need to be more careful about the indexing. Let me just use the 
    # convolution theorem properly.
    
    # sum_k weight[k] * a[k-i] for k in [0, 2n-2] and k-i in [0, n-1]
    # = sum_j a[j] * weight[j+i] for j in [0, n-1] and j+i in [0, 2n-2]
    # This is the cross-correlation of a and weight at shift i.
    
    # If I define: h[m] = weight[m] for m in [0, 2n-2], 0 otherwise
    # Then sum_j a[j] * h[j+i] = (a * h_flipped)[n-1-i + ?] ... this is getting confusing.
    
    # Let me just use the direct FFT formula:
    # The cross-correlation c[i] = sum_j a[j] * weight[j+i]
    # can be computed as: c = ifft(conj(fft(a, N)) * fft(weight, N))[:n]
    # where N is large enough (>= 2n-1).
    
    # Let me verify: ifft(conj(A) * W)[i] = sum_k conj(A[k]) * W[k] * exp(2*pi*i*k/N)
    # This is the circular cross-correlation. For linear, we need N >= n + (2n-1) - 1 = 3n-2.
    # That's too large. Let me use N >= 2n-1 and just accept some wrapping for the last few elements.
    
    # Actually, for our purposes, let's just use N = 2n-1 (or the next power of 2) and 
    # accept that the last n-1 elements might have some error. Since we only need the first n
    # elements and the "wrapping" only affects the last n-1 elements of the full correlation,
    # we should be fine.
    
    # Hmm, actually this is getting too complicated. Let me use a simpler approach:
    # Since weight is symmetric (weight[k] = weight[2n-2-k] because conv is symmetric),
    # we can write:
    # grad_a[i] = 2 * sum_k weight[k] * a[k-i]
    # And sum_k weight[k] * a[k-i] is just the convolution of (reversed) weight with a, 
    # evaluated at the right position.
    
    # Let me just compute it as: 
    # grad = 2 * irfft(rfft(weight, fft_size) * rfft(a_extended, fft_size), fft_size)
    # where a_extended is a zero-padded version... 
    
    # You know what, let me just use the approach from the previous code:
    # grad_i = 2 * sum_k w_{i+k} * a_k (from the smooth gradient formula)
    
    # This is a cross-correlation: cross[i] = sum_k w[i+k] * a[k]
    # Using FFT: cross = irfft(conj(rfft(w, N)) * rfft(a, N), N) ... 
    
    # I'll just use the same formula as the previous code which worked:
    # cross = irfft(W * conj(A), fft_size)[:n]
    # where W = rfft(w, fft_size) and A = rfft(a, fft_size)
    # This gives cross[i] = sum_k w[k+i] * a[k] (circular, but correct for i < n if N >= 2n-1)
    
    cross = irfft(rfft(weight, fft_size) * np.conj(rfft(a, fft_size)), fft_size)[:n]
    return 2.0 * cross


def lbfgs_smooth_optimize(a0, time_limit, p=50, verbose=True):
    """Minimize p-norm of autocorrelation using L-BFGS-B with simplex constraint."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Use scipy L-BFGS-B on the p-norm objective, with x >= 0 and sum(x) = norm
    # We parameterize: x = simplex point, so we work in n-1 dimensions
    # Or just use the bound constraint x >= 0 and handle the sum constraint by 
    # normalizing at each evaluation
    
    def objective(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x_norm = x * norm / s
        return p_norm_objective(x_norm, p)
    
    def gradient(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return np.zeros(n)
        x_norm = x * norm / s
        g = p_norm_grad(x_norm, p)
        # Adjust for the normalization: d(x*norm/s)/dx_i = norm*(s - x_i)/s^2
        # So grad_obj wrt x_i = sum_j g_j * d(x_norm_j)/d(x_i)
        # This is complicated; let's just use the gradient of x_norm and scale
        # Actually, for L-BFGS, we can just use the gradient of the p-norm wrt x_norm
        # and the chain rule. But since we're on the simplex, let's project the gradient.
        return g
    
    # Actually, let's simplify: just optimize on the simplex directly using the 
    # retraction approach. At each step, compute the gradient of the p-norm, 
    # take a step, and project back to the simplex.
    
    # L-BFGS-B with bounds [0, inf] doesn't handle the equality constraint sum(x) = norm.
    # Alternative: use the "free" parameterization and project.
    
    # Simplest approach: use the previous smooth_optimize but with better step sizes
    # and L-BFGS-like memory.
    
    # Let me just use scipy's minimize with the constraint:
    from scipy.optimize import LinearConstraint
    
    # Actually, for simplicity and speed, let me use the projected subgradient 
    # method with adaptive step size and Nesterov momentum, similar to the previous
    # code but with the p-norm instead of log-sum-exp.
    
    y = a.copy()
    for iteration in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        objective_val = p_norm_objective(y, p)
        grad = p_norm_grad(y, p)
        
        # Adaptive step size (Barzilai-Borwein-like)
        step = 1.0 / (np.linalg.norm(grad, ord=np.inf) + 1e-10)
        step = min(step, 1.0)
        
        a_new = y - step * grad
        # Project to simplex {x >= 0, sum(x) = norm}
        a_new = project_simplex(a_new, norm)
        
        # Nesterov
        alpha = 2.0 / (iteration + 2.0)
        y = (1 - alpha) * y + alpha * a_new
        y = project_simplex(y, norm)
        
        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()
        
        score_a = fast_autocorr(a_new).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
        
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    p-norm iter={iteration} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    # Fix rounding
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        result[np.argmax(result)] += diff
    return result


def build_conv_matrix(a):
    """Build the convolution matrix T where (T @ b)[j] = (a*b)[j]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T


def lp_best_response(a, M, solver, active_only=None, tol=1e-8):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    
    if active_only is not None and len(active_only) > 0:
        T_use = T[active_only, :]
    else:
        T_use = T
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False, tol=tol)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, avg_window=100):
    """LP best-response with Nesterov acceleration and Polyak averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # Adaptive constraint generation
        conv_y = fast_autocorr(y)
        # Use more constraints as we progress (tighter threshold)
        progress = min(1.0, n_iters / 200.0)
        threshold = 0.7 + 0.2 * progress  # starts at 0.7, goes to 0.9
        active = np.where(conv_y >= threshold * M_curr)[0]
        if len(active) < 10:
            active = np.where(conv_y >= 0.5 * M_curr)[0]
        if len(active) < 5:
            active = np.arange(len(conv_y))
        
        b = lp_best_response(y, M_curr, solver, active_only=active)
        if b is None:
            b = lp_best_response(y, M_curr, solver, active_only=None)
            if b is None:
                stuck_count += 1
                if stuck_count > 50:
                    break
                continue
        
        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Polyak averaging
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        # Also check a sliding window average
        if n_avg >= avg_window and n_avg % 20 == 0:
            # Can't easily do sliding window without storing all iterates
            pass
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def focused_pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent focused on peak-contributing pairs."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        conv = fast_autocorr(a)
        pass_improved = False
        
        # Find which pairs contribute most to the peak
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]
        threshold = 0.9 * peak_val
        peak_positions = np.where(conv >= threshold)[0]
        
        # For each peak position k, the contribution is sum_{i+j=k} a[i]*a[j]
        # We want to reduce these by moving mass from high a[i] to low a[j]
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue
            
            # Find best j to move to
            best_j = -1
            best_improvement = 0
            
            # Try moving to positions that reduce peak contribution
            for offset in range(1, min(60, n)):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    # Quick check: does moving mass from i to j reduce the peak?
                    # The peak at position k involves a[i]*a[k-i] (if valid)
                    # Moving from i to j changes this to a[j]*a[k-j]
                    # If a[k-i] > a[k-j], then moving mass from i to j helps at peak k
                    
                    # Try a small move
                    d = a[i] * 0.1
                    a_test = a.copy()
                    a_test[i] -= d
                    a_test[j] += d
                    a_test = np.maximum(a_test, 0)
                    s_test = a_test.sum()
                    if s_test < 1e-15:
                        continue
                    a_test = a_test * norm / s_test
                    conv_test_max = fast_autocorr(a_test).max()
                    
                    if conv_test_max < conv.max() - best_improvement:
                        best_improvement = conv.max() - conv_test_max
                        best_j = j
                        best_d = d
            
            if best_j >= 0:
                a[i] -= best_d
                a[best_j] += best_d
                a = np.maximum(a, 0)
                a = a * norm / a.sum()
                conv = fast_autocorr(a)
                if conv.max() < best_score - 1e-12:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Focused PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, verbose=True):
    """Mirror descent (entropy) on the probability simplex with subgradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Mirror descent with entropy kernel: x_{t+1} = x_t * exp(-eta * grad) / Z
    # This is equivalent to multiplicative weights
    
    y = a.copy()
    step_scale = norm  # scale for step size
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        
        # Subgradient: for the active constraint (argmax), grad = 2 * cross_corr(e_k, a)
        # where e_k is the unit vector at the peak position
        # More precisely, if multiple positions are near-peak, average their gradients
        
        threshold = 0.95 * M
        active = np.where(conv_y >= threshold)[0]
        
        # Gradient at y for the max: average of gradients of active terms
        # d(conv(y,y))[k]/dy[i] = y[k-i] + y[i-k]
        # So grad[i] = (1/|active|) * sum_{k in active} (y[k-i] + y[i-k])
        
        # Compute gradient using FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        
        # weight[k] = 1/|active| if k in active, else 0
        weight = np.zeros(size)
        weight[active] = 1.0 / len(active)
        
        W = rfft(weight, fft_size)
        Y = rfft(y, fft_size)
        cross = irfft(W * np.conj(Y), fft_size)[:n]
        grad = 2.0 * cross
        
        # Mirror descent step: y_new = y * exp(-eta * grad)
        eta = step_scale / (np.linalg.norm(grad, ord=np.inf) * t)
        eta = min(eta, 1.0)
        
        y_new = y * np.exp(-eta * grad)
        y_new = np.maximum(y_new, 1e-30)
        y_new = y_new * norm / y_new.sum()
        
        score = fast_autocorr(y_new).max()
        if score < best_score:
            best_score = score
            best_a = y_new.copy()
        
        y = y_new
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror descent iter={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss
