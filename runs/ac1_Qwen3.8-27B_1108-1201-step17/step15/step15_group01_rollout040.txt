import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full linear autocorrelation (convolution of a with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_change_value(a, k, i, j, delta):
    """Compute exact change in conv[k] when moving delta from position i to position j."""
    n = len(a)
    aki = a[k - i] if 0 <= k - i < n else 0.0
    akj = a[k - j] if 0 <= k - j < n else 0.0
    first_order = 2.0 * delta * (akj - aki)
    second_order = 0.0
    if k == 2 * i:
        second_order += delta * delta
    if k == 2 * j:
        second_order += delta * delta
    if k == i + j:
        second_order -= 2.0 * delta * delta
    return first_order + second_order


def smooth_optimize(a0, time_limit, verbose=False):
    """Minimize smooth approximation of max(conv) using accelerated projected gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    temps = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1, 0.06, 0.03, 0.02, 0.01]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(50, min(300, int(remaining * 2)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            # Gradient: d/dy of T*log(sum exp(c/T)) where c = conv(y,y)
            # grad_i = 2 * sum_k w_k * y_{k-i} (cross-correlation of w and y)

            Y = rfft(y, fft_size)
            cross = irfft(W * np.conj(Y), fft_size)[:n]
            grad = 2.0 * cross

            step = norm / (np.linalg.norm(grad) + 1e-10) * 0.5
            step = min(step, norm / n)

            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()

                a_new = a_new * norm / s

            alpha = 2.0 / (it + 2.0)

            if s_y < 1e-15:

            a = a_curr

    a = a * norm / s0

    best_score = conv.max()

    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # Focus on positions that would reduce the peak
            # The peak is at argmax(conv)
            peak_k = np.argmax(conv)

            candidates = set()

            # Position opposite to peak contribution

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp annealed gradient (FFT-based) for global shaping
2. Fast peak-directed coordinate descent with exact O(1) convolution deltas
3. Multi-start with asymmetric initializations exploiting convolution asymmetry
4. Length exploration and final polishing
Key insight: conv(a,a) is NOT symmetric unless a is a palindrome, enabling
sub-Cauchy-Schwarz scores via asymmetric mass distribution.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def conv_delta(a, k, i, j, d):
    """Exact change in conv[k] when a[i]-=d, a[j]+=d. O(1)."""
    n = len(a)
    aki = a[k - i] if 0 <= k - i < n else 0.0
    akj = a[k - j] if 0 <= k - j < n else 0.0
    val = 2.0 * d * (akj - aki)
    if k == 2 * i:
        val += d * d
    if k == 2 * j:
        val += d * d
    if k == i + j:
        val -= 2.0 * d * d
    return val


def smooth_optimize(a0, time_limit, verbose=False):
    """Annealed smooth gradient (log-sum-exp) with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temps = [40.0, 20.0, 12.0, 7.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04, 0.025, 0.015, 0.01]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(40, min(200, int(remaining * 1.5)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > t0 + time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            if c_max < 1e-30:
                break
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            # Gradient of smooth max
            W = rfft(w, fft_size)
            Y = rfft(y, fft_size)
            cross = irfft(W * np.conj(Y), fft_size)[:n]
            grad = 2.0 * cross
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            # Adaptive step with damping
            step = 0.8 * norm / (gnorm * (1.0 + 0.1 * np.sqrt(it)))
            step = min(step, 2.0 * norm / n)

            a_new = y - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new * norm / s

            # Nesterov
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            if it == 0:
                a_curr = a_new.copy()
            else:
                mom = 1.0 / (1.0 + alpha)
                a_curr = y_new + mom * (y_new - y)
                a_curr = np.maximum(a_curr, 0.0)
                s_a = a_curr.sum()
                if s_a < 1e-15:
                    a_curr = y_new.copy()
                else:
                    a_curr = a_curr * norm / s_a

            y = y_new
            a = a_curr

            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

            if verbose and int(T * 100) % 5 == 0:
                el = time.monotonic() - t0
                print(f"    Smooth T={T:.3f} it={it} ({el:.0f}s): peak={best_score:.8f}")

    return best_a, best_score


def peak_directed_cd(a0, time_limit, rng, verbose=False):
    """Coordinate descent that targets the current peak of the convolution.
    Uses exact O(1) convolution change formula for fast evaluation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Get top positions of current conv
        top_count = min(80, len(conv))
        top_idx = np.argsort(conv)[-top_count:]
        current_max = conv.max()
        peak_k = int(np.argmax(conv))

        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(perm[idx])
            if a[i] < norm * 5e-11:
                continue

            # Candidate j positions: nearby + peak-contributing + random
            j_set = set()
            for off in range(1, 15):
                j_set.add((i + off) % n)
                j_set.add((i - off) % n)
            # Peak contributors
            lo_m = max(0, peak_k - n + 1)
            hi_m = min(n - 1, peak_k)
            if hi_m >= lo_m:
                for m in range(lo_m, hi_m + 1, max(1, (hi_m - lo_m) // 10 + 1)):
                    j_set.add(m)
                    j_set.add((peak_k - m) % n)
            for _ in range(3):
                j_set.add(int(rng.integers(0, n)))

            best_d = 0.0
            best_j = -1
            best_new_peak = current_max

            for j in j_set:
                j = int(j)
                if j == i:
                    continue
                # Quick check: would this reduce the peak at all?
                # At peak position: change = 2*d*(a[peak_k-j] - a[peak_k-i])
                # We want negative change (reduction)
                a_pj = a[peak_k - j] if 0 <= peak_k - j < n else 0.0
                a_pi = a[peak_k - i] if 0 <= peak_k - i < n else 0.0
                if a_pj >= a_pi and a[i] > 0:
                    # Moving from i to j would INCREASE the peak (or not help)
                    # unless secondary effects dominate; skip for speed
                    continue

                # Binary search for optimal delta
                lo, hi = 0.0, a[i] * 0.98
                for _ in range(5):
                    mid = (lo + hi) / 2.0
                    if mid < 1e-14:
                        lo = mid
                        continue
                    # Estimate new max using top positions
                    worst = 0.0
                    for k in top_idx:
                        new_val = conv[k] + conv_delta(a, int(k), i, j, mid)
                        if new_val > worst:
                            worst = new_val
                    if worst < best_new_peak - 1e-13:
                        best_new_peak = worst
                        best_d = mid
                        best_j = j
                        lo = mid
                    else:
                        hi = mid

            if best_d > 1e-14 and best_j >= 0:
                a[i] -= best_d
                a[best_j] += best_d
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                conv = fast_autocorr(a)
                new_max = conv.max()
                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            el = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({el:.0f}s): peak={best_score:.8f}")

    return best_a, best_score


def diffuse_smooth(a, sigma):
    """Apply mild Gaussian smoothing to reduce sharp features."""
    n = len(a)
    from scipy.ndimage import gaussian_filter1d
    a_s = gaussian_filter1d(a, sigma=sigma, mode='nearest')
    a_s = np.maximum(a_s, 0.0)
    s = a_s.sum()
    if s > 1e-15:
        a_s = a_s * a.sum() / s
    return a_s


def make_asymmetric_init(n, rng, style=0):
    """Create diverse asymmetric initializations."""
    if style == 0:
        # Power ramp
        x = np.arange(n, dtype=np.float64)
        a = (x / max(n - 1, 1)) ** 2 + 0.1
    elif style == 1:
        # Inverse ramp
        x = np.arange(n, dtype=np.float64)
        a = (1.0 - x / max(n - 1, 1)) ** 2 + 0.1
    elif style == 2:
        # Off-center bump
        center = 0.35 * n
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - center) ** 2) / (2 * (0.2 * n) ** 2)) + 0.05
    elif style == 3:
        # two bumps
        x = np.arange(n, dtype=np.float64)
        a = (np.exp(-((x - 0.3 * n) ** 2) / (2 * (0.15 * n) ** 2)) +
             0.5 * np.exp(-((x - 0.7 * n) ** 2) / (2 * (0.1 * n) ** 2))) + 0.05
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(5):
            center = rng.integers(0, n)
            width = rng.integers(n // 10, n // 3)
            x = np.arange(n)
            a += np.exp(-((x - center) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 0.0)
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum proportions."""
    old_n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
