def compute_subgradient(y, k):
    """Compute ∂(y*y)[k]/∂y[j] for all j."""
    n = len(y)
    g = np.zeros(n)
    # Term 1: y[k-j] for 0 <= k-j < n, i.e., j in [k-n+1, k] ∩ [0, n-1]

Now let me think about the Nesterov method more carefully.

The standard Nesterov accelerated gradient method for minimizing f(x) (convex, L-smooth):
- x_0, y_0 = x_0
- For t = 1, 2, ...:

But our problem is not smooth (the max is not differentiable), so we use the subgradient instead of the gradient. The Nesterov subgradient method has a convergence rate of O(1/sqrt(t)) for the subgradient method (without acceleration) or O(1/t) with acceleration (under additional conditions).

For our problem, the "L" (Lipschitz constant of the subgradient) is related to the norm of a, which is bounded.

Let me use the standard Nesterov subgradient method with Polyak step size:

where f* is a lower bound on the optimum.

def fast_autocorr(a):

    n = len(a)
    size = 2 * n - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def compute_subgradient(y, peak_indices):
    """

def compute_soft_subgradient(y, b, beta):
    """
    Compute the gradient of the soft-max objective:
    f_soft = (1/beta) * log(sum_k exp(beta * b[k]))

    max_b = b.max()
    exp_b = np.exp(beta * (b - max_b))

        if j_lo <= j_hi:

def build_constraint_matrix(a, k_indices, n):
    """Build constraint matrix for (a*b)[k] <= M rows."""
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):

        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            mat[idx, j] = a[k - j]
    return mat

    k_indices = np.asarray(active_set, dtype=int)
    T = build_constraint_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm_sq = 2.0 * n
    norm = np.sqrt(norm_sq)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    n_iters = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        
        # Adaptive active set
        if active_set is None or t % 30 == 0:
            tol = max(0.02, 0.08 * np.exp(-t / 200.0))
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            top_k = np.argsort(conv_y)[-max(3, n // 100):]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Limit active set size for speed
            if len(active_set) > 300:
                active_set = np.argsort(conv_y)[-300:]
        
        b, duals = lp_best_response(y, M_curr, active_set, solver)
        
        if b is None:
            active_set = np.arange(2 * n - 1)
            b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue
        
        # Verify and expand if needed
        size = 2 * n - 1
        complement = np.setdiff1d(np.arange(size), active_set)
        if len(complement) > 0:
            check_indices = complement[::max(1, len(complement) // 100)]
            T_check = build_constraint_matrix(y, check_indices, n)
            violations = T_check @ b
            viol_mask = violations > M_curr * (1 + 1e-5)
            if np.any(viol_mask):
                new_active = np.union1d(active_set, check_indices[viol_mask])
                b, _ = lp_best_response(y, M_curr, new_active, solver)
                if b is None:
                    continue
                active_set = new_active
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Restart if stalled
        if stall_count > 50:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.10f}")
    
    return best_a, best_M


def coordinate_descent_fast(a0, time_limit, verbose=False):
    """
    Coordinate descent with FFT-accelerated updates.
    For each coordinate j, try a few candidate values and keep the best.
    """
    n = len(a0)
    norm_sq = 2.0 * n
    norm = np.sqrt(norm_sq)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_passes = 0
    no_improve = 0
    max_no_improve = 4
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        current_conv = fast_autocorr(a)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            s_others = a.sum() - aj
            
            # Try a few candidate values
            candidates = [0.0, aj * 0.5, aj * 0.75, aj * 1.25, aj * 1.5,
                         s_others * 0.05, s_others * 0.25, s_others * 0.5]
            candidates = [max(0.0, min(c, norm)) for c in candidates]
            candidates.append(aj)
            
            best_x = aj
            best_score = 2.0 * n * current_conv.max()
            
            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                # Update autocorrelation: new_b = old_b + delta
                # delta[k] = 2*(x-aj)*a[k-j](valid) + (x^2-aj^2)*delta_{k,2j}
                delta = np.zeros(2 * n - 1)
                j_lo = max(0, 2 * j - n + 1)
                j_hi = min(2 * n - 2, 2 * j)
                # Actually: (a*a)[k] changes by 2*(x-aj)*c[k] + (x^2-aj^2)*[k==2j]
                # where c[k] = a[k-j] (if valid and k-j != j) + a[j-k] (if valid and j-k != j)
                # But simpler: just compute the change directly
                # (new_a * new_a)[k] - (a * a)[k] for the term involving coordinate j
                # = 2*(x-aj) * (a[k-j] if 0<=k-j<n) + (x^2 - 2*x*aj + aj^2) * [k == 2*j]... 
                # No, let me just recompute from scratch for correctness.
                new_a = a.copy()
                new_a[j] = x
                new_a = new_a * norm / new_a.sum()
                new_M = fast_autocorr(new_a).max()
                new_score = 2.0 * n * new_M
                if new_score < best_score - 1e-10:
                    best_score = new_score
                    best_x = x
            
            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                a = a * norm / s_new
                current_conv = fast_autocorr(a)
                M_new = current_conv.max()
                if M_new < best_M - 1e-12:
                    best_M = M_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = aj
                    a = a * norm / a.sum()
                    current_conv = fast_autocorr(a)
        
        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best_M={best_M:.10f}")
    
    return best_a, best_M


def perturb_sequence(a, rng, scale=0.02):
    """Add multiplicative noise and renormalize."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def evaluate_safe(a):
    """Safely evaluate the score."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def symmetrize(a):
    """Symmetrize: a = (a + reverse(a))/2."""
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def propose_candidate():
    """
    Main search function combining fast subgradient, LP refinement,
    and coordinate descent to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(777)
    t_global = time.monotonic()
    time_budget = 1750.0

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)
    print(f"Starting with n={n_base}, score={best_score:.8f}")

    # Try symmetrization
    a_sym = symmetrize(best_seq)
    sc_sym = evaluate_safe(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_seq = a_sym.copy()
        print(f"  Symmetrization improved: {best_score:.8f}")

    # === Phase 1: Fast Nesterov subgradient (hard max) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        print(f"\n=== Phase 1: Fast Nesterov subgrad, n={n_cur}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_subgrad(a_norm, t1, verbose=True, seed=0, use_soft=False)
        try_update(a_out, "ns1")

    # === Phase 2: Fast Nesterov with soft-max continuation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        print(f"\n=== Phase 2: Soft-max Nesterov, n={n_cur}, budget={t2:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_subgrad(a_norm, t2, verbose=True, seed=1, use_soft=True,
                                     beta_start=3.0, beta_end=80.0)
        try_update(a_out, "soft2")

    # === Phase 3: LP best-response ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 60:
        print(f"\n=== Phase 3: LP best-response, n={n_cur}, budget={t3:.0f}s ===")
        solver = get_solver()
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=0)
        try_update(a_out, "lp3")

    # === Phase 4: Fast Nesterov from improved point ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 40:
        print(f"\n=== Phase 4: Nesterov round 2, n={n_cur}, budget={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_subgrad(a_norm, t4, verbose=True, seed=100, use_soft=False)
        try_update(a_out, "ns4")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Coordinate descent, n={n_cur}, budget={t5:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t5, verbose=True)
        try_update(a_out, "cd5")

    # === Phase 6: Perturbation + Nesterov restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 3
    t6_total = min(remaining * 0.25, 300.0)
    t6_each = t6_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t6 = min(t6_each, remaining - 20)
        if t6 < 20:
            break
        print(f"\n=== Phase 6r{r}: Perturb+NS, budget={t6:.0f}s ===")
        pert = perturb_sequence(best_seq, rng, scale=0.015 + 0.01 * r)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_subgrad(a_norm, t6, verbose=False, seed=500 + r, use_soft=False)
        try_update(a_out, f"pert{r}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 7: Final LP, n={n_cur}, remaining={remaining:.0f}s ===")
        solver = get_solver()
        t7 = min(remaining * 0.4, 200.0)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=False, seed=999)
        try_update(a_out, "lp_final")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Final CD, remaining={remaining:.0f}s ===")
        t8 = min(remaining * 0.5, 100.0)
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t8, verbose=False)
        try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
