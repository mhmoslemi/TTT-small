            lo = mid
        else:
            hi = mid

    s = result.sum()

2. FISTA with FFT-based multi-subgradient for smoothing
3. Multiplicative weight LP iterations
4. Pair-coordinate local search for final polish
5. Iterite averaging for convergence acceleration
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def project_onto_simplex(v, target_sum):
    n = len(v)
    if n == 0:
        return v
    lo = v.min() - target_sum
    hi = v.max()
    for _ in range(64):
        mid = (lo + hi) / 2
        s = np.maximum(v - mid, 0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    result = np.maximum(v - tau, 0)
    s = result.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    return result * target_sum / s

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    return val

def lp_nesterov(a0, time_limit, solver, verbose=True, multiplicative=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    no_improve = 0
    avg_buffer = []
    AVG_WINDOW = 200

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening - progressively tighter
        if t < 50:
            thresh = 0.25
        elif t < 200:
            thresh = 0.5
        elif t < 1000:
            thresh = 0.7
        elif t < 5000:
            thresh = 0.85
        else:
            thresh = 0.92

        active = np.where(conv_y >= M * thresh)[0]
        if len(active) < 30:
            active = np.where(conv_y >= M * 0.3)[0]

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints every 50 iterations
        if active is not None and len(active) < size and t % 50 == 0:
            check = fast_conv_cross(y, g)
            if check.max() > M * (1 + 5e-3):
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    g = b_full * norm / b_full.sum()

        # Update with momentum
        alpha = 2.0 / (t + 2.0)
        if multiplicative and t > 10:
            ratio = g / np.maximum(y, 1e-30)
            log_r = np.log(np.clip(ratio, 1e-10, 1e10))
            y_new = y * np.exp(alpha * log_r)
        else:
            y_new = (1.0 - alpha) * y + alpha * g

        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        # Store in average buffer
        if t > 100:
            avg_buffer.append(a.copy())
            if len(avg_buffer) > AVG_WINDOW:
                avg_buffer.pop(0)

        # Check average periodically
        if len(avg_buffer) == AVG_WINDOW and t % 100 == 0:
            avg_a = np.mean(avg_buffer, axis=0)
            s_avg = avg_a.sum()
            if s_avg > 1e-15:
                avg_a = avg_a * norm / s_avg
                sc_avg = fast_conv_self(avg_a).max()
                if sc_avg < best_score - 1e-10:
                    best_score = sc_avg
                    best_a = avg_a.copy()
                    improved = True

        # Perturbation to escape local optima
        if no_improve == 300:
            n_p = max(15, n // 15)
            idx_p = np.random.default_rng(t).choice(n, size=n_p, replace=False)
            y[idx_p] *= (1.0 + 0.25 * np.random.default_rng(t + 1).standard_normal(n_p))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0
            avg_buffer = []

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Ne t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def fista_refine(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    y = a.copy()
    t_val = 1.0

    for it in range(80000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M_y = conv_y.max()
        if M_y < 1e-15:
            break

        # Multi-subgradient with adaptive temperature
        eta = 20.0 + 0.01 * it
        shifted = (conv_y - M_y) / eta
        shifted = np.maximum(shifted, -50.0)
        w = np.exp(shifted)
        w = w / w.sum()

        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        grad = 2.0 * irfft(A * W, fft_size)[:n]

        # FISTA step with backtracking
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        step = 1.0 / t_new

        a_trial = project_onto_simplex(a - step * grad, norm)
        M_trial = fast_conv_self(a_trial).max()

        n_bt = 0
        while M_trial > M_y + 1e-8 and n_bt < 10 and step > 1e-12:
            step *= 0.5
            a_trial = project_onto_simplex(a - step * grad, norm)
            M_trial = fast_conv_self(a_trial).max()
            n_bt += 1

        if M_trial <= M_y + 1e-8:
            a_next = a_trial
        else:
            a_next = a.copy()

        # FISTA extrapolation
        y = (1.0 - t_val / t_new) * a + (t_val / t_new) * a_next
        a = a_next
        t_val = t_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def local_perturb_search(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Pair move: transfer mass from i to j
        idx1 = rng.integers(n)
        idx2 = rng.integers(n)
        if idx1 == idx2:
            continue

        # Try several transfer amounts
        base_eps = a[idx1]
        if base_eps < 1e-12:
            continue

        found = False
        for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7]:
            eps = base_eps * frac
            a_try = a.copy()
            a_try[idx1] -= eps
            a_try[idx2] += eps
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            sc = fast_conv_self(a_try).max()
            if sc < best_score - 1e-13:
                a = a_try
                best_score = sc
                best_a = a.copy()
                found = True
                break

        # Also try simultaneous pair moves (4 indices)
        if not found and rng.random() < 0.3:
            idx3 = rng.integers(n)
            idx4 = rng.integers(n)
            if len(set([idx1, idx2, idx3, idx4])) == 4:
                eps1 = a[idx1] * 0.1
                eps2 = a[idx3] * 0.1
                a_try = a.copy()
                a_try[idx1] -= eps1
                a_try[idx2] += eps1
                a_try[idx3] -= eps2
                a_try[idx4] += eps2
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    sc = fast_conv_self(a_try).max()
                    if sc < best_score - 1e-13:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        found = True

        if found:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50000:
                # Larger perturbation
                n_p = max(20, n // 10)
                idx_p = rng.choice(n, size=n_p, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_p))
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * norm / s_a
                no_improve = 0

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Local it={it} ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):

        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate():
    t_global = time.monotonic()
    time_budget = 1730.0
    solver = get_solver()
    rng = np.random.default_rng(12345)
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting optimization with n={n_base}, current score={best_score:.8f}")

    # Phase 1: LP Nesterov additive (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.3, 450.0)
    print(f"\n=== Phase 1: LP Nesterov additive (budget={t1:.0f}s) ===")
    a_out, _ = lp_nesterov(best_seq, t1, solver, verbose=True)
    try_update(a_out, "lp1")

    # Phase 2: FISTA refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 2: FISTA refine (budget={t2:.0f}s) ===")
    a_out, _ = fista_refine(best_seq, t2, verbose=True)
    try_update(a_out, "fista")

    # Phase 3: LP Nesterov multiplicative
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.28, 400.0)
    print(f"\n=== Phase 3: LP Nesterov multiplicative (budget={t3:.0f}s) ===")
    a_out, _ = lp_nesterov(best_seq, t3, solver, verbose=True, multiplicative=True)
    try_update(a_out, "lp2")

    # Phase 4: Local pair-coordinate search
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.2, 280.0)
    print(f"\n=== Phase 4: Local pair search (budget={t4:.0f}s) ===")
    a_out, _ = local_perturb_search(best_seq, t4, rng, verbose=True)
    try_update(a_out, "local")

    # Phase 5: Final LP pass
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        t5 = min(remaining * 0.6, 250.0)
        print(f"\n=== Phase 5: Final LP pass (budget={t5:.0f}s) ===")
        a_out, _ = lp_nesterov(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp_final")

    # Phase 6: Quick local polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t6 = min(remaining * 0.7, 100.0)
        print(f"\n=== Phase 6: Quick local polish (budget={t6:.0f}s) ===")
        a_out, _ = local_perturb_search(best_seq, t6, rng, verbose=False)
        try_update(a_out, "polish")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
