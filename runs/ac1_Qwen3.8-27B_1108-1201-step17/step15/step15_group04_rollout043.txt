import numpy as np

_fft_cache = {}

def get_fft_size(n):
    size = 2*n - 1
    key = n
    if key not in _fft_cache:
        p = 1 << (size - 1).bit_length()
        _fft_cache[key] = p
    return p

def conv_self(a):
    n = len(a)
    fsize = get_fft_size(n)
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:2*n-1]

def conv_max(a):
    n = len(a)
    fsize = get_fft_size(n)
    A = np.fft.rfft(a, fsize)
    c = np.fft.irfft(A * A, fsize)
    return float(np.max(c[:2*n-1]))

def grad_from_weights(a, w):
    n=len(a); size=2*n-1; fsize=get_fft_size(n)
    A=np.fft.rfft(a,fsize)
    W=np.fft.rfft(w,fsize)
    g1=np.fft.irfft(np.conj(A)*W, fsize)[:n]
    g2=np.fft.irfft(A*W, fsize)[:n]
    return 2.0*(g1+g2)

def normalize_sum(a, S):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.array([1.0])
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-30:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a * (S/s)

    a=np.clip(a,0,None)
    s=float(np.sum(a))

    2*n*max(convolve(a,a))/(sum(a))^2
under nonnegativity.  The program starts from the supplied parent sequence
(when available), extends its length by interpolation, and then refines it
with smoothed-max projected gradient methods, active-lag gradient polishing,
and a local pair coordinate-descent/polishing stage.
"""

import numpy as np
import math
import time

try:
    from scipy.fft import rfft as _rfft, irfft as _irfft
except Exception:
    _rfft = np.fft.rfft
    _irfft = np.fft.irfft

_fft_cache = {}


def get_fft_size(n):
    val = _fft_cache.get(n)
    if val is None:
        size = 2 * int(n) - 1
        val = 1 << (size - 1).bit_length()
        _fft_cache[int(n)] = val
    return val


def conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0])
    fsize = get_fft_size(n)
    A = _rfft(a, fsize)
    return _irfft(A * A, fsize)[:2 * n - 1]


def conv_max(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0
    fsize = get_fft_size(n)
    A = _rfft(a, fsize)
    c = _irfft(A * A, fsize)
    return float(np.max(c[:2 * n - 1]))


def grad_from_weights(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    fsize = get_fft_size(n)
    A = _rfft(a, fsize)
    W = _rfft(w, fsize)
    g1 = _irfft(np.conj(A) * W, fsize)[:n]
    g2 = _irfft(A * W, fsize)[:n]
    return 2.0 * (g1 + g2)


def normalize_sum(a, S):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0])
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-30:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a * (S / s)


def score_obj(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    n = a.size
    maxb = conv_max(a)
    return float(2.0 * n * maxb / (s * s))


def eval_safe(a):
    try:
        vals = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        return float(evaluate_sequence(vals))
    except Exception:
        return float("inf")


def update_best(a, best_seq, best_score):
    if a is None:
        return best_seq, best_score
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return best_seq, best_score
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-12:
        return best_seq, best_score
    a = a / s
    sc = score_obj(a)
    if np.isfinite(sc) and sc < best_score:
        return a.copy(), float(sc)
    return best_seq, best_score


def interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(np.sum(a))
    if a.size == 0 or s < 1e-30:
        return np.ones(int(new_n), dtype=np.float64) / float(new_n)
    a = a / s
    old_n = a.size
    if old_n == int(new_n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, int(new_n))
    f = np.interp(x_new, x_old, a)
    f = np.maximum(f, 0.0)
    fs = float(np.sum(f))
    if fs < 1e-30:
        return np.ones(int(new_n), dtype=np.float64) / float(new_n)
    return f / fs


def make_shape(n, kind, rng):
    x = np.linspace(0.0, 1.0, int(n))
    if kind == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif kind == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif kind == 2:
        a = 0.5 + 0.35 * np.cos(2.0 * np.pi * 3.0 * x) + 0.18 * np.cos(2.0 * np.pi * 7.0 * x)
        a = np.maximum(a, 0.05)
    elif kind == 3:
        a = np.exp(-6.0 * (x - 0.22) ** 2) + 0.75 * np.exp(-6.0 * (x - 0.78) ** 2)
    elif kind == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif kind == 5:
        a = 0.35 + 0.65 * np.sin(np.pi * x) ** 0.8
    elif kind == 6:
        a = np.exp(-2.2 * x) + 0.55 * np.exp(-2.2 * (1.0 - x))
    elif kind == 7:
        a = 0.55 * np.exp(-4.5 * (x - 0.28) ** 2) + 0.45 * np.exp(-4.5 * (x - 0.72) ** 2)
        a = a * (1.0 + 0.25 * np.cos(18.0 * np.pi * x))
        a = np.maximum(a, 0.02)
    elif kind == 8:
        a = np.abs(np.sin(2.7 * np.pi * x)) ** 0.9
    elif kind == 9:
        a = 0.42 + 0.30 * np.cos(4.0 * np.pi * x) + 0.22 * np.cos(11.0 * np.pi * x)
        a = np.maximum(a, 0.04)
    elif kind == 10:
        a = np.abs(np.cos(1.7 * np.pi * x) + 0.35 * np.cos(4.3 * np.pi * x))
    else:
        a = 0.45 + 0.35 * np.cos(5.0 * np.pi * x) + 0.20 * np.sin(9.0 * np.pi * x)
        a = np.maximum(a, 0.03)

    noise = 1.0 + 0.04 * rng.standard_normal(int(n))
    a = a * noise
    a = np.clip(a, 0.0, 10.0)
    s = float(np.sum(a))
    if s < 1e-30:
        return np.ones(int(n), dtype=np.float64) / float(n)
    return a / s


def best_structured_init(n, seed, n_trials=24):
    rng = np.random.default_rng(seed)
    best = None
    best_sc = float("inf")
    for i in range(int(n_trials)):
        kind = int(rng.integers(0, 12))
        a = make_shape(int(n), kind, rng)
        sc = score_obj(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a
    if best is None:
        best = np.ones(int(n), dtype=np.float64) / float(n)
    return best, best_sc


def usable_step_base(a, T):
    anorm = math.sqrt(max(float(np.dot(np.asarray(a, dtype=np.float64), np.asarray(a, dtype=np.float64))), 1e-12))
    base = anorm * (0.10 + 0.50 * math.exp(-max(float(T), 0.0) / 2.0))
    return max(0.05, base)


def line_search_gradient(a, grad, m, T, sumS, decay=1.0):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(grad, dtype=np.float64)
    n = a.size
    if n == 0:
        return a, m
    g = g - float(np.mean(g))
    gn = float(np.linalg.norm(g))
    if gn < 1e-18:
        return a, m
    d = -g / gn
    step = usable_step_base(a, T) * max(0.05, float(decay))
    best_cand = a.copy()
    best_val = float(m)
    for _ in range(14):
        cand = a + step * d
        cand = np.clip(cand, 0.0, None)
        cs = float(np.sum(cand))
        if cs < 1e-30:
            step *= 0.5
            continue
        cand = cand * (float(sumS) / cs)
        cm = conv_max(cand)
        if cm < best_val - 1e-16:
            best_val = cm
            best_cand = cand
        tol = max(1e-10, 0.005 * max(float(T), 0.0))
        tol = min(tol, 0.05 * max(float(m), 1e-12))
        if cm <= float(m) + tol:
            best_cand = cand
            best_val = cm
            break
        step *= 0.65
    return best_cand, best_val


def anneal_accept(diff, m, rng):
    diff = float(diff)
    m = float(m)
    if diff < -1e-14:
        return True
    if diff <= 1e-10:
        return rng.random() < 0.45
    scale = 0.01 * max(m, 1e-6)
    if diff < 3.0 * scale:
        return rng.random() < math.exp(-diff / max(scale, 1e-15))
    return False


def lse_additive_optimize(a0, time_limit, seed):
    if time_limit <= 0.0:
        return normalize_sum(a0, 1.0)

    n = len(np.asarray(a0).ravel())
    if n <= 0:
        return np.array([1.0])
    sumS = math.sqrt(2.0 * float(n))
    a = normalize_sum(a0, sumS)
    best = a.copy()
    best_m = conv_max(a)

    t0 = time.monotonic()
    deadline = t0 + float(time_limit)
    temps = np.geomspace(30.0, 0.03, 22)
    rng = np.random.default_rng(seed)
    no_improve = 0

    for T in temps:
        if time.monotonic() >= deadline:
            break
        stage_deadline = min(deadline, time.monotonic() + max(1.0, float(time_limit) / len(temps)))
        it = 0
        while time.monotonic() < stage_deadline:
            c = conv_self(a)
            m = float(np.max(c))

            if m < best_m - 1e-14:
                best_m = m
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 220:
                npert = int(max(10, n // 22))
                npert = min(npert, n)
                idx = rng.choice(n, npert, replace=False)
                a[idx] *= np.exp(rng.standard_normal(npert) * 0.22)
                a = np.clip(a, 0.0, None)
                s = float(np.sum(a))
                if s < 1e-30:
                    a = normalize_sum(np.ones(n, dtype=np.float64), sumS)
                else:
                    a = a * (sumS / s)
                no_improve = 0

            shift = (c - m) / float(T)
            shift = np.clip(shift, -75.0, 0.0)
            w = np.exp(shift)
            w += 1e-12
            ws = float(np.sum(w))
            if ws <= 0.0:
                w = np.ones_like(c, dtype=np.float64) / float(c.size)
            else:
                w = w / ws

            grad = grad_from_weights(a, w)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-20:
                continue

            decay = 1.0 / (1.0 + 0.002 * it)
            cand, cm = line_search_gradient(a, grad, m, float(T), sumS, decay)
            if anneal_accept(cm - m, m, rng):
                a = cand
            it += 1

    return normalize_sum(best, 1.0)


def active_top_optimize(a0, time_limit, seed):
    if time_limit <= 0.0:
        return normalize_sum(a0, 1.0)

    n = len(np.asarray(a0).ravel())
    if n <= 0:
        return np.array([1.0])
    sumS = math.sqrt(2.0 * float(n))
    a = normalize_sum(a0, sumS)
    best = a.copy()
    best_m = conv_max(a)

    t0 = time.monotonic()
    deadline = t0 + float(time_limit)
    fracs = np.geomspace(0.50, 0.02, 18)
    rng = np.random.default_rng(seed)
    no_improve = 0

    for f in fracs:
        if time.monotonic() >= deadline:
            break
        stage_deadline = min(deadline, time.monotonic() + max(1.0, float(time_limit) / len(fracs)))
        it = 0
        while time.monotonic() < stage_deadline:
            c = conv_self(a)
            m = float(np.max(c))

            if m < best_m - 1e-14:
                best_m = m
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 260:
                npert = int(max(10, n // 25))
                npert = min(npert, n)
                idx = rng.choice(n, npert, replace=False)
                a[idx] *= np.exp(rng.standard_normal(npert) * 0.20)
                a = np.clip(a, 0.0, None)
                s = float(np.sum(a))
                if s < 1e-30:
                    a = normalize_sum(np.ones(n, dtype=np.float64), sumS)
                else:
                    a = a * (sumS / s)
                no_improve = 0

            size = c.size
            Q = int(max(8, min(size - 1, int(round(size * float(f))))))
            if Q >= size:
                w = np.ones(size, dtype=np.float64) / float(size)
            else:
                top = np.argpartition(c, size - Q)[-Q:]
                vals = c[top]
                vmax = float(np.max(vals))
                vmin = float(np.min(vals))
                span = max(vmax - vmin, 1e-12)
                w = np.zeros(size, dtype=np.float64)
                w[top] = np.exp((vals - vmax) / (0.15 * span))
                w += 1e-12
                ws = float(np.sum(w))
                if ws <= 0.0:
                    w = np.ones(size, dtype=np.float64) / float(size)
                else:
                    w = w / ws

            grad = grad_from_weights(a, w)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-20:
                continue

            T_eff = 0.04 + 0.12 * float(f)
            decay = 1.0 / (1.0 + 0.002 * it)
            cand, cm = line_search_gradient(a, grad, m, T_eff, sumS, decay)
            if anneal_accept(cm - m, m, rng):
                a = cand
            it += 1

    return normalize_sum(best, 1.0)


def pair_cd_optimize(a0, time_limit, seed, max_n=2200):
    n_total = len(np.asarray(a0).ravel())
    if time_limit <= 0.0 or n_total > int(max_n) or n_total < 8:
        return normalize_sum(a0, 1.0)

    n = n_total
    sumS = math.sqrt(2.0 * float(n))
    a = normalize_sum(a0, sumS)
    best = a.copy()
    best_m = conv_max(a)
    size = 2 * n - 1

    t0 = time.monotonic()
    deadline = t0 + float(time_limit)
    rng = np.random.default_rng(seed)

    coarse = n if n < 120 else min(900, max(120, n // 4))
    coarse = int(min(coarse, n))
    no_pass = 0
    dfracs = (0.10, 0.20, 0.35, 0.50, 0.65, 0.80, 0.90, 0.97)

    while time.monotonic() < deadline:
        c = conv_self(a)
        current_m = float(np.max(c))
        if current_m < best_m - 1e-14:
            best_m = current_m
            best = a.copy()

        Q = int(max(10, min(size // 8, 90)))
        top = np.argpartition(c, size - Q)[-Q:]
        top = top[np.argsort(c[top])[::-1]]
        peak = top[c[top] > current_m * 0.90]
        if peak.size == 0:
            peak = top[:min(20, top.size)]

        idxs = rng.choice(n, coarse, replace=False)
        improved = False
        moves_this_pass = 0

        for i in idxs:
            if time.monotonic() >= deadline or moves_this_pass >= 180:
                break
            i = int(i)
            ai = float(a[i])
            if ai <= 1e-14 * sumS:
                continue

            js_list = [int(x) for x in rng.integers(0, n, 4)]
            for k in peak[:10]:
                r = int(k) - i
                if 0 <= r < n and r != i:
                    js_list.append(r)
            if i > 0:
                js_list.append(i - 1)
            if i + 1 < n:
                js_list.append(i + 1)

            seen = set()
            best_d = 0.0
            best_j = -1
            best_v = current_m

            for j in js_list:
                if j < 0 or j >= n or j == i or j in seen:
                    continue
                seen.add(j)
                aj = float(a[j])

                g = np.zeros(size, dtype=np.float64)
                jlo = j
                jhi = min(size, j + n)
                if jhi > jlo:
                    g[jlo:jhi] += 2.0 * a[:jhi - jlo]
                ilo = i
                ihi = min(size, i + n)
                if ihi > ilo:
                    g[ilo:ihi] -= 2.0 * a[:ihi - ilo]

                for fr in dfracs:
                    d = float(fr) * ai
                    if d <= 0.0 or d >= ai:
                        continue
                    nc = c + d * g
                    if 2 * i < size:
                        nc[2 * i] += d * d
                    if 2 * j < size:
                        nc[2 * j] += d * d
                    kc = i + j
                    if kc < size and i != j:
                        nc[kc] -= 2.0 * d * d
                    v = float(np.max(nc))
                    if v < best_v - 1e-16:
                        best_v = v
                        best_d = d
                        best_j = j

            if best_d > 0.0 and best_j >= 0 and best_v < current_m - 1e-13:
                a[i] -= best_d
                a[best_j] += best_d
                if a[i] < 0.0:
                    a[i] = 0.0
                s = float(np.sum(a))
                if s > 1e-30:
                    a = a * (sumS / s)
                c = conv_self(a)
                current_m = float(np.max(c))
                if current_m < best_m - 1e-14:
                    best_m = current_m
                    best = a.copy()
                improved = True
                moves_this_pass += 1

        if improved:
            no_pass = 0
        else:
            no_pass += 1
            if no_pass >= 4:
                npert = int(max(10, n // 22))
                npert = min(npert, n)
                idx = rng.choice(n, npert, replace=False)
                a[idx] *= np.exp(rng.standard_normal(npert) * 0.16)
                a = np.clip(a, 0.0, None)
                s = float(np.sum(a))
                if s > 1e-30:
                    a = a * (sumS / s)
                no_pass = 0

    return normalize_sum(best, 1.0)


def remaining_time(deadline):
    r = float(deadline) - time.monotonic()
    return r if r > 0.0 else 0.0


def optimize_sequence(a0, time_limit, seed, use_pair=True):
    a = normalize_sum(a0, 1.0)
    best, best_sc = update_best(a, None, float("inf"))
    if best is None:
        return a

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_limit))

    rem = remaining_time(deadline)
    if rem > 40.0:
        t_lse = min(rem * 0.62, float(time_limit) * 0.62)
        if t_lse > 25.0:
            a1 = lse_additive_optimize(a, t_lse, int(seed))
            best, best_sc = update_best(a1, best, best_sc)

    rem = remaining_time(deadline)
    if rem > 30.0 and best is not None and best.size <= 3400:
        t_act = min(rem * 0.35, float(time_limit) * 0.20)
        if t_act > 20.0:
            a2 = active_top_optimize(best, t_act, int(seed) + 101)
            best, best_sc = update_best(a2, best, best_sc)

    rem = remaining_time(deadline)
    if rem > 25.0 and best is not None and use_pair and best.size <= 2200:
        t_pc = min(rem * 0.55, float(time_limit) * 0.22)
        if t_pc > 20.0:
            a3 = pair_cd_optimize(best, t_pc, int(seed) + 202, max_n=2200)
            best, best_sc = update_best(a3, best, best_sc)

    rem = remaining_time(deadline)
    if rem > 25.0 and best is not None:
        t_fin = min(rem * 0.85, float(time_limit) * 0.35)
        if t_fin > 15.0:
            a4 = lse_additive_optimize(best, t_fin, int(seed) + 303)
            best, best_sc = update_best(a4, best, best_sc)

    return best


def propose_candidate(seed=42, budget_s=1800.0):

    deadline = t0 + total_budget

    best_seq = None
    best_score = float("inf")

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and float(np.sum(p)) > 1e-10:
            best_seq, best_score = update_best(p, best_seq, best_score)
            print(f"Parent loaded: n={p.size}, score={best_score:.9f}")
    except Exception:
        best_seq = None
        best_score = float("inf")

    if best_seq is None:
        n0 = 2048
        a, sc = best_structured_init(n0, int(seed), 30)
        best_seq, best_score = update_best(a, None, float("inf"))
        print(f"Default construction: n={best_seq.size}, score={best_score:.9f}")

    # Quick local refinement of the current length.
    rem = remaining_time(deadline)
    if rem > 80.0 and best_seq is not None and best_seq.size <= 3200:
        t1 = min(150.0, 0.09 * total_budget, rem * 0.55)
        if t1 > 30.0:
            try:
                aopt = optimize_sequence(best_seq, t1, int(seed) + 1, use_pair=best_seq.size <= 2000)
                best_seq, best_score = update_best(aopt, best_seq, best_score)
                print(f"After quick polish: n={best_seq.size}, score={best_score:.9f}, t={time.monotonic()-t0:.1f}s")
            except Exception as e:
                print("quick polish error:", e)

    # Mid-length extension.
    rem = remaining_time(deadline)
    n_base = len(best_seq) if best_seq is not None else 0
    n_mid = int(max(n_base + 1, min(2800, n_base * 1.5)))
    n_mid = min(n_mid, 3400)
    if rem > 300.0 and n_mid > n_base:
        init_mid = interpolate_seq(best_seq, n_mid)
        best_seq, best_score = update_best(init_mid, best_seq, best_score)
        print(f"Mid init n={n_mid}, score={best_score:.9f}")
        t2 = min(620.0, 0.33 * total_budget, rem * 0.65)
        if t2 > 80.0:
            try:
                aopt = optimize_sequence(init_mid, t2, int(seed) + 1000, use_pair=n_mid <= 2000)
                best_seq, best_score = update_best(aopt, best_seq, best_score)
                print(f"After mid length: n={best_seq.size}, score={best_score:.9f}, t={time.monotonic()-t0:.1f}s")
            except Exception as e:
                print("mid length error:", e)

    # Long-length extension.
    rem = remaining_time(deadline)
    n_long = 4096
    if best_seq is not None and n_long > best_seq.size:
        init_long = interpolate_seq(best_seq, n_long)
        best_seq, best_score = update_best(init_long, best_seq, best_score)
        print(f"Long init n={n_long}, score={best_score:.9f}")
        t3 = min(950.0, 0.47 * total_budget, rem * 0.75)
        if t3 > 120.0:
            try:
                aopt = optimize_sequence(init_long, t3, int(seed) + 2000, use_pair=False)
                best_seq, best_score = update_best(aopt, best_seq, best_score)
                print(f"After long length: n={best_seq.size}, score={best_score:.9f}, t={time.monotonic()-t0:.1f}s")
            except Exception as e:
                print("long length error:", e)
    elif best_seq is not None and rem > 200.0:
        # If already long, just try interpolating to a slightly larger length.
        n_alt = int(best_seq.size) + 800
        if n_alt <= 6000:
            init_alt = interpolate_seq(best_seq, n_alt)
            best_seq, best_score = update_best(init_alt, best_seq, best_score)
            t3 = min(600.0, rem * 0.70)
            if t3 > 100.0:
                try:
                    aopt = optimize_sequence(init_alt, t3, int(seed) + 5000, use_pair=False)
                    best_seq, best_score = update_best(aopt, best_seq, best_score)
                    print(f"After alt extension: n={best_seq.size}, score={best_score:.9f}, t={time.monotonic()-t0:.1f}s")
                except Exception as e:
                    print("alt extension error:", e)

    # Final polish.
    rem = remaining_time(deadline)
    if rem > 70.0 and best_seq is not None:
        t4 = min(300.0, rem * 0.80)
        try:
            aopt = lse_additive_optimize(best_seq, t4, int(seed) + 3000)
            best_seq, best_score = update_best(aopt, best_seq, best_score)
        except Exception as e:
            print("final LSE error:", e)

        rem = remaining_time(deadline)
        if rem > 50.0 and best_seq is not None and best_seq.size <= 2200:
            t5 = min(160.0, rem * 0.85)
            try:
                aopt = pair_cd_optimize(best_seq, t5, int(seed) + 4000, max_n=2200)
                best_seq, best_score = update_best(aopt, best_seq, best_score)
            except Exception as e:
                print("final pair error:", e)

    if best_seq is None:
        best_seq = np.ones(1024, dtype=np.float64) / 1024.0

    # Final validation and safe normalization.
    final_sc = eval_safe(best_seq)
    if np.isfinite(final_sc) and final_sc < best_score:
        best_score = final_sc
    best_seq = normalize_sum(best_seq, 1.0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"FINAL: score={best_score:.9f}, n={best_seq.size}, time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in best_seq]
