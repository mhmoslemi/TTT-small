import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def objective(a):
    n = len(a)
    conv = fast_autocorr(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * conv.max() / (s * s))

def softmax_objective(a, T=0.01):
    """Smoothed objective using softmax approximation of max."""
    n = len(a)
    conv = fast_autocorr(a)
    s = a.sum()
    if s < 0.01:
        return float('inf'), np.zeros(n)
    # Softmax approximation of max
    conv_clipped = np.clip(conv, -100, 100)
    exp_conv = np.exp(conv_clipped / T - conv_clipped.max() / T)
    weights = exp_conv / exp_conv.sum()
    smooth_max = T * (np.log(exp_conv.sum()) + conv_clipped.max() / T)
    # Gradient w.r.t. a[i]
    # d smooth_max / d a[i] = sum_k weights[k] * d conv[k]/d a[i]
    # d conv[k]/d a[i] = a[k-i] if 0 <= k-i < n
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += weights[k] * a[idx]
    # Full objective gradient (accounting for (sum)^2)
    # f = 2n * smooth_max / s^2
    # df/da[i] = 2n * grad[i] / s^2 - 2 * 2n * smooth_max * a[i] / s^3
    score = 2.0 * n * smooth_max / (s * s)
    full_grad = 2.0 * n * grad / (s * s) - 4.0 * n * smooth_max * a / (s * s * s)
    return score, full_grad

    stall_count = 0
    stall_reset = 100

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set with softmax guidance
        if active_set is None or t % 30 == 0:
            T_active = max(0.01, 0.05 * np.exp(-t / 200.0))
            # Use softmax to identify important constraints
            exp_vals = np.exp((conv_y - M_curr) / T_active)
            weights = exp_vals / exp_vals.sum()
            cumulative = np.cumsum(weights)
            # Take constraints that account for 95% of the "mass"
            threshold_idx = np.searchsorted(cumulative, 0.95)
            active_set = np.where(weights >= weights[threshold_idx])[0]
            if len(active_set) < 10:
                active_set = np.where(conv_y >= M_curr * 0.8)[0]
            if len(active_set) > 0.8 * size:
                active_set = np.where(conv_y >= M_curr * 0.92)[0]

        # Max-sum LP
        b_max = lp_max_sum(y, M_curr, solver, active_set)

        # Min-max LP for blending
        b_min = None
        if t % 4 == 0:
            b_min, _ = lp_min_max(y, solver, active_set, normalize_sum=norm)

        if b_max is None:
            if t % 20 == 0:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        if b_min is not None and b_min.sum() > 1e-10:
            g_min_norm = b_min * norm / b_min.sum()
            g = (1.0 - beta_blend) * g + beta_blend * g_min_norm

        # Verify constraint satisfaction
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0 and len(inactive) < size // 2:
                T_check = build_active_constraints(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 1e-5))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        # Nesterov update
        alpha_t = max(2.0 / (t + 2.0), alpha_min)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.92)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_conv(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_conv(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_conv(avg_check).max()
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def coord_descent(a0, time_limit, rng, verbose=True):
    """Coordinate descent with exact 1D optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_conv(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            # Try several values of a[i] (keeping others fixed, then renormalizing)
            candidates = [0.0, ai * 0.5, ai * 0.7, ai * 1.3, ai * 1.5, ai * 2.0]
            best_val = best_score
            best_candidate = None

            for ci in candidates:
                if ci < 0:
                    continue
                a_try = a.copy()
                a_try[i] = ci
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                cv_try = fast_conv(a_try)
                val = cv_try.max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_candidate = a_try

            if best_candidate is not None:
                a = best_candidate
                conv = fast_conv(a)
                if conv.max() < best_score:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_conv(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            ai = a[i]
            aj = a[j]
            if ai < 1e-12 * norm and aj < 1e-12 * norm:
                continue

            best_val = fast_conv(a).max()
            best_delta = 0.0

            for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
                # Move from i to j
                d = ai * frac
                if d > 1e-15:
                    a_try = a.copy()
                    a_try[i] = ai - d
                    a_try[j] = aj + d
                    s_try = a_try.sum()
                    if s_try > 1e-15:
                        a_try = a_try * norm / s_try
                        val = fast_conv(a_try).max()
                        if val < best_val - 1e-13:
                            best_val = val
                            best_delta = d

                # Move from j to i
                d2 = aj * frac
                if d2 > 1e-15:
                    a_try = a.copy()
                    a_try[i] = ai + d2
                    a_try[j] = aj - d2
                    s_try = a_try.sum()
                    if s_try > 1e-15:
                        a_try = a_try * norm / s_try
                        val = fast_conv(a_try).max()
                        if val < best_val - 1e-13:
                            best_val = val
                            best_delta = -d2

            if abs(best_delta) > 1e-14:
                a_try = a.copy()
                if best_delta > 0:
                    a_try[i] -= best_delta
                    a_try[j] += best_delta
                else:
                    a_try[i] += abs(best_delta)
                    a_try[j] -= abs(best_delta)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    sc = fast_conv(a_try).max()
                    if sc < best_score - 1e-13:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        # New: raised cosine with phase
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x - np.pi) ** 1.5
        a = np.maximum(a, 0.0)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
