"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth optimization (log-sum-exp) with L-BFGS
2. Continuation method (decreasing smoothing parameter)
3. Multiple initializations and random restarts
4. Post-processing with direct subgradient steps
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_self_conv(a):
    """Compute self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the score."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_self_conv(a)
    return float(2 * len(a) * conv.max() / (s * s))

def smooth_obj_and_grad(theta, n, tau):
    """
    Compute smooth objective and gradient.
    a = softmax(theta)  (so sum(a) = 1, a >= 0)
    Objective: 2n * tau * log(sum_k exp(conv[k]/tau))
    where conv = self_convolution(a)
    """
    # Compute a from theta
    theta_max = theta.max()
    exp_theta = np.exp(theta - theta_max)
    a = exp_theta / exp_theta.sum()
    
    # Compute self-convolution
    conv = fast_self_conv(a)
    
    # Smooth max using log-sum-exp
    conv_max = conv.max()
    scaled = conv / tau
    scaled_max = scaled.max()
    exp_scaled = np.exp(scaled - scaled_max)
    log_sum_exp = scaled_max + np.log(exp_scaled.sum())
    
    obj = 2.0 * n * tau * log_sum_exp
    
    # Weights for the smooth max
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient of smooth max w.r.t. a:
    # grad_conv[j] = 2 * sum_{l=0}^{n-1} w[j+l] * a[l]
    # = 2 * correlate(w, a, 'valid')
    grad_conv = 2.0 * np.correlate(w, a, 'valid')
    
    # Gradient w.r.t. theta:
    # d obj / d theta[m] = 2n * tau * (d/d theta[m]) log(sum exp(conv/tau))
    # = 2n * tau * sum_k w[k] * (d conv[k] / d theta[m]) / tau
    # = 2n * sum_k w[k] * (d conv[k] / d theta[m])
    # d conv[k] / d theta[m] = sum_j (d conv[k] / d a[j]) * (d a[j] / d theta[m])
    # = sum_j 2*a[k-j]*I(0<=k-j<n) * a[j]*(delta[j,m] - a[m])
    # 
    # So: d obj / d theta[m] = 2n * sum_j grad_conv[j] * a[j] * (delta[j,m] - a[m])
    # = 2n * a[m] * (grad_conv[m] - sum_j grad_conv[j] * a[j])
    
    grad_conv_dot_a = np.dot(grad_conv, a)
    grad_theta = 2.0 * n * a * (grad_conv - grad_conv_dot_a)
    
    return obj, grad_theta

def optimize_smooth(a_init, n, time_limit, verbose=True, seed=0):
    """
    Optimize using smooth objective with continuation method.
    a_init: initial sequence (non-negative, will be normalized)
    """
    # Initialize theta from a_init
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-10)
    a = a / a.sum()
    theta = np.log(a)  # log of normalized a
    
    best_theta = theta.copy()
    best_obj = float('inf')
    t0 = time.monotonic()
    
    # Continuation schedule for tau
    # Start with large tau, decrease geometrically
    tau_start = 1.0  # in units of conv values (a is normalized, so conv values are ~ 1/n)
    tau_end = 0.001
    n_tau = 15
    
    # Scale tau: since a is normalized to sum=1, conv values are ~ O(1/n)
    # We want tau to start much larger than the range of conv values
    conv_init = fast_self_conv(a)
    conv_range = conv_init.max() - conv_init.min()
    if conv_range < 1e-10:
        conv_range = conv_init.max()
    tau_start = conv_range * 10.0
    tau_end = conv_range * 0.001
    n_tau = 20
    
    taus = np.geomspace(tau_start, tau_end, n_tau)
    
    for i, tau in enumerate(taus):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Optimize at this tau
        def obj_grad(theta):
            return smooth_obj_and_grad(theta, n, tau)
        
        try:
            result = minimize(
                obj_grad, theta, method='L-BFGS-B',
                jac=True, options={'maxiter': 200, 'gtol': 1e-8, 'ftol': 1e-10}
            )
            theta = result.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at tau={tau:.6f}: {e}")
            break
        
        # Evaluate
        a_curr = np.exp(theta - theta.max())
        a_curr = a_curr / a_curr.sum()
        score = compute_score(a_curr)
        
        if score < best_obj:
            best_obj = score
            best_theta = theta.copy()
        
        if verbose and (i % 5 == 0 or i == n_tau - 1):
            elapsed = time.monotonic() - t0
            print(f"    tau={tau:.6f} ({elapsed:.0f}s): score={score:.8f}, best={best_obj:.8f}")
    
    a_best = np.exp(best_theta - best_theta.max())
    a_best = a_best / a_best.sum()
    return a_best, best_obj

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

_global_best = None


def fast_self_conv(a):
    """Compute self-convolution (a * a) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the score: 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_self_conv(a)
    return float(2 * len(a) * conv.max() / (s * s))


def score_and_grad_smooth(theta, n, tau):
    """
    Compute smooth objective and gradient w.r.t. theta.
    a = softmax(theta), so sum(a)=1, a>=0.
    Objective: 2n * tau * log(sum_k exp(conv[k]/tau))
    Gradient: 4n * a[m] * (correlate(w,a,'valid')[m] - weighted_avg_conv)
    """
    # Compute a from theta (numerically stable softmax)
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    a = exp_theta / exp_theta.sum()

    # Compute self-convolution
    conv = fast_self_conv(a)

    # Log-sum-exp for smooth max
    conv_scaled = conv / tau
    cs_max = conv_scaled.max()
    exp_cs = np.exp(conv_scaled - cs_max)
    lse = cs_max + np.log(exp_cs.sum())

    obj = 2.0 * n * tau * lse

    # Weights w[k] = exp(conv[k]/tau) / sum(exp(conv/t))
    w = exp_cs / exp_cs.sum()

    # Weighted average of conv
    ew_conv = np.dot(w, conv)

    # correlate(w, a, 'valid'): result[m] = sum_{l=0}^{n-1} w[m+l]*a[l], m=0..n-1
    # w has length 2n-1, a has length n, result has length n
    corr = np.correlate(w, a, mode='valid')

    # Gradient
    grad = 4.0 * n * a * (corr - ew_conv)

    return obj, grad

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    theta = np.log(a)

    best_a = a.copy()
    best_score = compute_score(a)

    # Determine tau schedule
    conv_init = fast_self_conv(a)
    conv_range = conv_init.max() - conv_init.min()
    if conv_range < 1e-12:
        conv_range = conv_init.max() + 1e-6

    tau_start = conv_range * 20.0
    tau_end = conv_range * 0.005
    n_tau = 25
    taus = np.geomspace(tau_start, tau_end, n_tau)

    for i, tau in enumerate(taus):
        if time.monotonic() - t0 >= time_limit:
            break

        # Define objective and gradient
        def obj_grad(th):
            return score_and_grad_smooth(th, n, tau)

        # Run L-BFGS
        try:
            result = minimize(
                obj_grad, theta, method='L-BFGS-B',
                jac=True,
                options={'maxiter': 300, 'gtol': 1e-7, 'ftol': 1e-12,
                         'maxls': 50}
            )
            theta = result.x
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error (tau={tau:.2e}): {e}")
            break

        # Evaluate current solution
        a_curr = np.exp(theta - theta.max())
        a_curr = a_curr / a_curr.sum()
        score_curr = compute_score(a_curr)

        if score_curr < best_score:
            best_score = score_curr
            best_a = a_curr.copy()
            if _global_best is None or score_curr < _global_best[1]:
                _global_best = (a_curr.copy(), score_curr)

        if verbose and (i % 5 == 0 or i == n_tau - 1):
            elapsed = time.monotonic() - t0
            print(f"      tau={tau:.4e} ({elapsed:.0f}s): score={score_curr:.8f}, best={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a_init, time_limit, verbose=True):
    """
    Subgradient method with line search for polishing.
    Minimizes max_k conv[k] subject to sum(a)=1, a>=0.
    """
    global _global_best
    n = len(a_init)
    t0 = time.monotonic()

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    conv = fast_self_conv(a)
    best_conv_max = conv.max()

    step = 0.1 / n
    patience = 0
    max_patience = 500
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        if patience > max_patience:
            break

        conv = fast_self_conv(a)
        M = conv.max()

        # Find all active peaks (within 2% of max)
        active = np.where(conv >= M * 0.98)[0]
        k_star = int(active[np.argmax(conv[active])])

        # Subgradient at k_star: g[j] = 2*a[k_star - j] if 0 <= k_star-j < n
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        for j in range(lo, hi):
            g[j] = 2.0 * a[k_star - j]

        # Try a few step sizes (backtracking line search)
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            trial_step = step * frac
            a_trial = a - trial_step * g / g.max() if g.max() > 0 else a.copy()
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            conv_trial = fast_self_conv(a_trial)
            M_trial = conv_trial.max()

            if M_trial < M - 1e-14:
                a = a_trial
                score = compute_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    best_conv_max = M_trial
                    if _global_best is None or score < _global_best[1]:
                        _global_best = (a.copy(), score)
                step = min(step * 1.05, 1.0)
                patience = 0
                improved = True
                break
            else:
                # If no improvement, try next smaller step
                pass

        if not improved:
            patience += 1
            step *= 0.95
            if step < 1e-12:
                break

        # Every 100 iterations, try a random active peak for diversity
        n_iters += 1
        if n_iters % 100 == 0 and len(active) > 1:
            k_rand = active[np.random.randint(len(active))]
            g_r = np.zeros(n)
            lo_r = max(0, k_rand - n + 1)
            hi_r = min(n, k_rand + 1)
            for j in range(lo_r, hi_r):
                g_r[j] = 2.0 * a[k_rand - j]
            if g_r.max() > 0:
                a_trial = a - step * 0.5 * g_r / g_r.max()
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    conv_t = fast_self_conv(a_trial)
                    if conv_t.max() < M - 1e-14:
                        a = a_trial
                        score = compute_score(a)
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                            if _global_best is None or score < _global_best[1]:
                                _global_best = (a.copy(), score)
                        patience = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad iter={n_iters} ({elapsed:.0f}s): score={best_score:.8f}, step={step:.6f}")

    return best_a, best_score


def frank_wolfe_step(a, tau_smooth=None):
    """
    One Frank-Wolfe (conditional gradient) step.
    Linearizes the objective at the current point and finds the minimizer
    over the simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a)
    conv = fast_self_conv(a)
    M = conv.max()

    if tau_smooth is not None:
        # Smooth weights
        conv_scaled = conv / tau_smooth
        cs_max = conv_scaled.max()
        w = np.exp(conv_scaled - cs_max)
        w = w / w.sum()
        # Gradient of smooth objective w.r.t. a (before softmax):
        # grad_a[j] = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n)
        grad_a = 2.0 * np.correlate(w, a, mode='valid')
    else:
        # Hard: gradient at the peak
        k_star = int(np.argmax(conv))
        grad_a = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        for j in range(lo, hi):
            grad_a[j] = 2.0 * a[k_star - j]

    # FW step: minimize <grad_a, b> over {b >= 0, sum(b)=1}
    # Solution: b = e_{argmin(grad_a)}
    j_min = int(np.argmin(grad_a))
    b = np.zeros(n)
    b[j_min] = 1.0

    return b, grad_a


def nesterov_fw_optimize(a_init, time_limit, verbose=True, seed=0):
    """Nesterov-accelerated Frank-Wolfe optimization."""
    global _global_best
    n = len(a_init)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    y = a.copy()  # Nesterov momentum variable
    step_size = 1.0
    stall = 0

    # Soft threshold for FW (use smooth version initially)
    conv_init = fast_self_conv(a)
    tau_fw = conv_init.max() * 0.1
    min_tau = conv_init.max() * 0.001

    for t in range(1, 500000):
        if time.monotonic() - t0 >= time_limit:
            break

        # Nesterov: interpolate
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * best_a  # This isn't quite right for FW
        # Actually for FW: y = a, then compute step from y
        # Let me use the standard FW with Nesterov:
        # a_t = (1 - gamma_t) * a_{t-1} + gamma_t * s_t where s_t is the FW step from y_t
        # y_t = a_t + (t-1)/(t+2) * (a_t - a_{t-1})

        gamma = 2.0 / (t + 2.0)
        b, _ = frank_wolfe_step(y, tau_smooth=tau_fw)

        a_new = (1.0 - gamma) * a + gamma * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Nesterov update for y
        if t == 1:
            y_new = a_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = a_new + momentum * (a_new - a)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / s_y

        a = a_new
        y = y_new

        # Slowly decrease tau
        tau_fw = max(tau_fw * 0.999, min_tau)

        score = compute_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
            if _global_best is None or score < _global_best[1]:
                _global_best = (a.copy(), score)
        else:
            stall += 1

        if stall > 500:
            # Restart
            a = best_a.copy()
            y = best_a.copy()
            stall = 0
            tau_fw = min_tau * 10.0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FW iter={t} ({elapsed:.0f}s): score={best_score:.8f}, tau={tau_fw:.2e}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.03):
    """Add multiplicative noise and renormalize."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def coordinate_descent_fast(a_init, time_limit, rng, verbose=False):
    """
    Fast coordinate descent: for each coordinate, try a few candidate values
    and keep the best. Uses efficient update of the convolution.
    """
    global _global_best
    n = len(a_init)
    t0 = time.monotonic()

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    conv = fast_self_conv(a)
    best_conv_max = conv.max()

    no_improve = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        if no_improve >= max_no_improve:
            break
        improved = False
        conv = fast_self_conv(a)
        conv_max = conv.max()

        # Process coordinates in random order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 >= time_limit:
                break

            aj = a[j]
            if aj < 1e-15:
                # Try setting it to a small positive value
                candidates = [1e-4, 1e-3, aj * 10 if aj > 0 else 1e-3]
            else:
                # Try scaling aj by various factors
                candidates = [0, aj * 0.25, aj * 0.5, aj * 0.75,
                             aj * 1.25, aj * 1.5, aj * 2.0]
                # Also try values based on the "typical" scale
                typical = a.mean()
                candidates.extend([typical * 0.5, typical, typical * 2])

            best_x = aj
            best_local_conv = conv_max

            for x in candidates:
                x = max(0.0, min(x, 1.0))
                if abs(x - aj) < 1e-12:
                    continue
                a[j] = x
                s = a.sum()
                if s < 1e-15:
                    a[j] = aj
                    continue
                a_scaled = a / s
                # Quick check: compute new conv max only if potentially better
                # Use the fact that changing a[j] only affects certain lags
                # lags affected: k where 0 <= k-j < n, i.e., k in [j, j+n-1]
                # New conv at affected lags:
                # conv_new[k] = conv_old[k] + (x - aj) * a[k-j] + (x*aj - aj^2) * delta_{k, 2j}
                # Actually it's more complex due to normalization. Let's just compute.
                conv_new = fast_self_conv(a_scaled)
                new_max = conv_new.max()
                if new_max < best_local_conv - 1e-14:
                    best_local_conv = new_max
                    best_x = x

            if best_x != aj:
                a[j] = best_x
                a = a / a.sum()
                conv = fast_self_conv(a)
                conv_max = conv.max()
                score = compute_score(a)
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    if _global_best is None or score < _global_best[1]:
                        _global_best = (a.copy(), score)

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      CD pass done ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search function. Uses smooth L-BFGS optimization with continuation,
    subgradient polishing, Frank-Wolfe, and coordinate descent to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    global _global_best
    _global_best = None
    t_start = time.monotonic()
    time_budget = 1740.0  # 60s buffer

    rng = np.random.default_rng(42)

    def elapsed():
        return time.monotonic() - t_start

    def remaining():
        return time_budget - elapsed()

    # --- Load parent sequence ---
    best_a = None
    best_score = float('inf')

    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            s = parent.sum()
            if s > 0.01:
                best_a = parent / s
                best_score = compute_score(best_a)
                _global_best = (best_a.copy(), best_score)
                print(f"Parent loaded: n={len(best_a)}, score={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_a is None:
        n = 1600
        # Use a structured initialization
        x = np.linspace(0, 1, n)
        best_a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        best_a = np.maximum(best_a, 0.01)
        best_a = best_a / best_a.sum()
        best_score = compute_score(best_a)
        _global_best = (best_a.copy(), best_score)
        print(f"Default init: n={n}, score={best_score:.8f}")

    def try_update(a, label):
        """Try to update the global best."""
        global _global_best
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            if _global_best is None or sc < _global_best[1]:
                _global_best = (a.copy(), sc)
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}, t={elapsed():.0f}s)")

    n_orig = len(best_a)
    print(f"\nStarting: n={n_orig}, score={best_score:.8f}, budget={time_budget:.0f}s")

    # === Phase 1: Smooth L-BFGS from parent (main workhorse) ===
    print(f"\n=== Phase 1: Smooth L-BFGS n={n_orig} ===")
    t1 = min(remaining() * 0.30, 450.0)
    if t1 > 60:
        a_out, sc = optimize_smooth_lbfgs(best_a, t1, verbose=True, seed=0)
        try_update(a_out, f"lbfgs1_n{n_orig}")

    # === Phase 2: Subgradient polish ===
    print(f"\n=== Phase 2: Subgradient polish n={len(best_a)} ===")
    t2 = min(remaining() * 0.10, 120.0)
    if t2 > 20:
        a_out, sc = subgradient_polish(best_a, t2, verbose=True)
        try_update(a_out, f"subg1_n{len(best_a)}")

    # === Phase 3: Smooth L-BFGS with perturbation ===
    print(f"\n=== Phase 3: Perturbed L-BFGS ===")
    t3 = min(remaining() * 0.15, 200.0)
    if t3 > 40:
        pert = perturb_sequence(best_a, rng, scale=0.02)
        a_out, sc = optimize_smooth_lbfgs(pert, t3, verbose=True, seed=123)
        try_update(a_out, f"lbfgs_pert_n{len(best_a)}")

    # === Phase 4: Frank-Wolfe ===
    print(f"\n=== Phase 4: Frank-Wolfe n={len(best_a)} ===")
    t4 = min(remaining() * 0.12, 150.0)
    if t4 > 30:
        a_out, sc = nesterov_fw_optimize(best_a, t4, verbose=True, seed=7)
        try_update(a_out, f"fw_n{len(best_a)}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length ===")
    lengths_to_try = [2000, 2400]
    t5_total = min(remaining() * 0.20, 300.0)
    for new_n in lengths_to_try:
        if remaining() < 60:
            break
        t_each = t5_total / len(lengths_to_try)
        print(f"  Trying n={new_n} (budget={t_each:.0f}s)")
        a_interp = interpolate_sequence(best_a, new_n)
        # Quick smooth optimization
        a_out, sc = optimize_smooth_lbfgs(a_interp, min(t_each * 0.7, 150.0), verbose=False, seed=99)
        try_update(a_out, f"interp_n{new_n}")
        # Quick subgradient
        if remaining() > 30:
            a_out2, sc2 = subgradient_polish(a_out, min(t_each * 0.3, 60.0), verbose=False)
            try_update(a_out2, f"interp_subg_n{new_n}")

    # === Phase 6: Second L-BFGS refinement on best ===
    print(f"\n=== Phase 6: Final L-BFGS refinement ===")
    t6 = min(remaining() * 0.25, 300.0)
    if t6 > 30:
        a_out, sc = optimize_smooth_lbfgs(best_a, t6, verbose=True, seed=200)
        try_update(a_out, f"lbfgs_final_n{len(best_a)}")

    # === Phase 7: Subgradient polish on best ===
    if remaining() > 30:
        print(f"\n=== Phase 7: Final subgradient ===")
        t7 = min(remaining() * 0.30, 150.0)
        a_out, sc = subgradient_polish(best_a, t7, verbose=True)
        try_update(a_out, f"subg_final_n{len(best_a)}")

    # === Phase 8: Coordinate descent (if time remains) ===
    if remaining() > 30:
        print(f"\n=== Phase 8: Coordinate descent ===")
        t8 = min(remaining() * 0.40, 200.0)
        a_out, sc = coordinate_descent_fast(best_a, t8, rng, verbose=True)
        try_update(a_out, f"cd_final_n{len(best_a)}")

    # === Final: one more quick subgradient ===
    if remaining() > 15:
        a_out, sc = subgradient_polish(best_a, min(remaining() * 0.5, 30.0), verbose=False)
        try_update(a_out, "final_subg")

    # Final evaluation
    if best_a is not None:
        final_score = compute_score(best_a)
        best_score = min(best_score, final_score)
        print(f"\n{'='*60}")
        print(f"FINAL: score={best_score:.10f}, n={len(best_a)}, time={elapsed():.0f}s")
        print(f"{'='*60}")
        return [float(x) for x in best_a]
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]
