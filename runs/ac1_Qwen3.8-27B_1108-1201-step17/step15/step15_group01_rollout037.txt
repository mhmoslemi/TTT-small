`conv(a,a)[k] = sum_m a[m]*a[k-m]` is a quadratic form `a^T Q_k a` where `Q_k` is a symmetric matrix. Is it positive semidefinite?

`a^T Q_k a = sum_m a[m]*a[k-m]`. This is the inner product of `a` with its shift by `k` (with appropriate boundary handling). This is NOT necessarily positive for all `a`. For example, if `a = [1, -1]` and `k=1`: `a[0]*a[1] + a[1]*a[0] = -2 < 0`.

But we have the constraint `a >= 0`, so all terms are non-negative, and `conv(a,a)[k] >= 0` for all k. However, the quadratic form is not convex in general (it's not positive semidefinite without the non-negativity constraint).

So the problem is NOT a convex optimization problem in general. This means L-BFGS-B might get stuck in local minima, and we need multiple starts.

def lbfbs_smooth(a0, T_values, time_limit, maxiter=500):
    """Run L-BFGS-B with smooth objective at various temperatures."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for T in T_values:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        def obj_grad(x):
            x = np.maximum(x, 0)  # clip for safety
            obj, grad = smooth_objective_and_grad(x, T, None)
            return obj, grad
        
        res = minimize(
            obj_grad, a, jac=True, method='L-BFGS-B',
            options={'maxiter': maxiter, 'maxfun': 2000, 'iprint': 0}
        )
        
        a_new = np.maximum(res.x, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new
    
    return best_a, best_score

def pair_cd(a0, time_limit, rng, window=30, verbose=True):
    """Pair coordinate descent with binary search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-15:
                continue
            
            for offset in range(1, window + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i or a[i] < 1e-15:
                        continue
                    
                    # Try to move from i to j
                    lo, hi = 0.0, a[i] * 0.99
                    best_d = 0.0
                    best_val = conv.max()
                    
                    for _ in range(6):
                        mid = (lo + hi) / 2
                        if mid <= 1e-18:
                            lo = mid
                            continue
                        a_t = a.copy()
                        a_t[i] -= mid
                        a_t[j] += mid
                        conv_t = fast_autocorr(a_t)
                        if conv_t.max() < best_val - 1e-15:
                            best_val = conv_t.max()
                            best_d = mid
                            lo = mid
                        else:
                            hi = mid
                    
                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        conv = fast_autocorr(a)
                        if conv.max() < best_score - 1e-12:
                            best_score = conv.max()
                            best_a = a.copy()
                        improved = True
                        no_improve = 0
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
    
    return best_a, best_score

def fourier_init(n, rng, n_harmonics=5):
    """Create a structured initialization using Fourier components."""
    # Base level + cosine harmonics
    t = np.arange(n)
    a = np.ones(n)
    for k in range(1, n_harmonics + 1):
        freq = k + rng.integers(0, 3)  # random frequency near k
        if freq >= n // 2:
            freq = n // 2 - 1
        phase = rng.uniform(0, 2 * np.pi)
        amplitude = rng.uniform(0.1, 0.4) / k
        a += amplitude * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def linear_init(n, direction='up'):
    """Linear ramp initialization."""
    if direction == 'up':
        a = np.linspace(0.1, 1.0, n)
    elif direction == 'down':
        a = np.linspace(1.0, 0.1, n)
    else:  # 'v' shape
        a = np.abs(np.linspace(-1, 1, n)) + 0.1
    a = a / a.sum()
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""

1. L-BFGS-B with log-sum-exp smooth approximation of the objective ratio
2. Structured diverse initializations (Fourier, spectral, linear, random)
3. Adaptive temperature annealing
4. High-speed pair coordinate descent polishing
5. Multi-length exploration via interpolation
Key improvement over prior art: L-BFGS-B quasi-Newton optimizer directly on
the scale-invariant ratio objective, with faster convergence and better
landscape navigation than Nesterov-accelerated first-order methods.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _smooth_ratio_obj_grad(x, T, mu):
    """
    Top-level objective and gradient for L-BFGS-B.
    Minimizes 2n * smooth_max(conv(x,x)) / (sum(x))^2 + mu*(sum(x)-1)^2
    """
    n = len(x)
    x = np.maximum(x, 0.0)
    S = x.sum()
    if S < 1e-10:
        return 1e10, np.ones(n) * 1e6

    c = fast_autocorr(x)
    c_max = c.max()
    if c_max < 1e-300:
        return 0.0, np.zeros(n)

    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    smooth_max = c_max + T * np.log(exp_c.sum())

    # Cross-correlation for gradient: cross[i] = sum_j w[i+j] * a[j]
    size = 2 * n - 1
    # Need fft_size >= (2n-1) + n - 1 = 3n-2 for cross-correlation
    xcorr_size = 3 * n - 2
    fft_size_x = 1 << (xcorr_size - 1).bit_length()
    W = rfft(w, fft_size_x)
    A = rfft(x, fft_size_x)
    cross = irfft(np.conj(W) * A, fft_size_x)[:n]

    # Objective: 2n * smooth_max / S^2 + mu*(S-1)^2
    obj = 2.0 * n * smooth_max / (S * S) + mu * (S - 1.0) ** 2

    # Gradient: 2n * (S * 2*cross[i] - 2*smooth_max) / S^3 + 2*mu*(S-1)
    grad_conv = 2.0 * n * (2.0 * S * cross - 2.0 * smooth_max) / (S * S * S)
    grad_sum = 2.0 * mu * (S - 1.0)
    grad = grad_conv + grad_sum

    return obj, grad


def _lbfbs_run(a_init, T, mu, maxiter):
    """Run L-BFGS-B for one temperature setting."""
    def wrapper(x):
        return _smooth_ratio_obj_grad(x, T, mu)
    res = minimize(wrapper, a_init, jac=True, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'maxfun': maxiter * 3, 'iprint': 0})
    return res


def lbfbs_anneal(a0, temps, mu, time_limit, maxiter_per_T=300):
    """
    Run L-BFGS-B with decreasing temperatures.
    Returns best sequence found (normalized to sum=1).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for T in temps:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        alloc_iter = max(50, min(maxiter_per_T, int(remaining * 20)))
        try:
            res = _lbfbs_run(a, T, mu, alloc_iter)
        except Exception:
            continue

        a_new = np.maximum(res.x, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-10:
            continue
        a_new = a_new / S_new
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def pair_cd_fast(a0, time_limit, rng, window=30):

    Fast pair coordinate descent with limited binary search.

    """

    S = a.sum()
    if S < 1e-15:
        return a, float('inf')
    a = a / S

    best_score = conv.max()

    n_passes = 0

        n_passes += 1
        improved = False

            if time.monotonic() - t0 > time_limit:

                        continue

                    # Try moving fraction of a[i] to a[j]
                    # Use a few trial fractions rather than full binary search
                    for frac in (0.5, 0.25, 0.1, 0.75):
                        d = a[i] * frac
                        if d < 1e-15:
                            break
                        a[i] -= d
                        a[j] += d
                        conv_new = fast_autocorr(a)
                        if conv_new.max() < conv.max() - 1e-14:
                            conv = conv_new
                            if conv.max() < best_score:
                                best_score = conv.max()
                                best_a = a.copy()
                            improved = True
                            break
                        else:
                            # Revert
                            a[i] += d
                            a[j] -= d
                            break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                break

        if n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fourier_init(n, rng, n_harmonics=4, max_amp=0.4):
    """Structured initialization with cosine harmonics."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.8
    for k in range(1, n_harmonics + 1):
        freq = rng.integers(k, max(k + 3, n // 4))
        if freq >= n // 2:
            freq = n // 2 - 1
        phase = rng.uniform(0, 2 * np.pi)
        amplitude = max_amp / (k * 0.7) * rng.uniform(0.5, 1.0)
        a += amplitude * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(n) / n
    return a / S


def spectral_flat_init(n, rng, n_components=8):
    """
    Initialization designed to have flat power spectrum.
    Construct in frequency domain with random phases, then take real part.
    """
    fft_size = 1 << (2 * n - 1).bit_length()
    # Random spectrum with flat magnitude
    freqs = np.arange(fft_size // 2 + 1)
    # Zero out DC and Nyquist
    magnitude = np.ones(fft_size // 2 + 1) * 0.5
    magnitude[0] = np.sqrt(n) * 0.5  # DC component for non-negativity
    if fft_size % 2 == 0:
        magnitude[-1] = 0.0

    # Add some random variation to magnitude
    for _ in range(n_components):
        idx = rng.integers(1, fft_size // 2)
        magnitude[idx] *= rng.uniform(0.3, 2.0)
        if idx < fft_size // 2:
            mirror = fft_size - idx
            if mirror < len(magnitude):
                magnitude[mirror] = magnitude[idx]

    # Random phases
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    phases[0] = 0.0  # DC is real

    # Build spectrum
    spec = np.zeros(fft_size, dtype=np.complex128)
    spec[:fft_size // 2 + 1] = magnitude * np.exp(1j * phases)
    # Conjugate symmetry
    for k in range(1, fft_size // 2):
        spec[fft_size - k] = np.conj(spec[k])

    a = irfft(spec, fft_size)[:n]
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(n) / n
    return a / S


def linear_init(n, style):
    """Linear ramp initializations."""
    if style == 'up':
        a = np.linspace(0.05, 1.0, n)
    elif style == 'down':
        a = np.linspace(1.0, 0.05, n)
    elif style == 'v':
        a = np.abs(np.linspace(-1, 1, n)) + 0.05
    elif style == 'u':
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.05
    else:
        a = np.ones(n)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    S = f.sum()
    if S < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / S
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _make_starts(n, best_seq, rng):
    """Create diverse starting points."""
    starts = []
    # Parent
    if best_seq is not None and len(best_seq) == n:
        starts.append(('parent', best_seq.copy()))
    # Fourier inits
    for fi in range(4):
        starts.append((f'fourier_{fi}', fourier_init(n, rng, n_harmonics=fi + 2, max_amp=0.3 + 0.1 * fi)))
    # Spectral flat
    for si in range(3):
        starts.append((f'spectral_{si}', spectral_flat_init(n, rng, n_components=si * 4 + 4)))
    # Linear inits
    for style in ['up', 'down', 'v', 'u']:
        starts.append((f'linear_{style}', linear_init(n, style)))
    # Random
    a_rand = rng.random(n) + 0.01
    starts.append(('random', a_rand / a_rand.sum()))
    # Anti-symmetric (mass on one side)
    a_anti = np.exp(-np.arange(n) / (n * 0.3))
    starts.append(('exponential', a_anti / a_anti.sum()))
    a_anti2 = np.exp(-np.arange(n)[::-1] / (n * 0.3))
    starts.append(('exp_rev', a_anti2 / a_anti2.sum()))
    return starts


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Returns best sequence found within time budget.
    Uses L-BFGS-B with smooth approximation, diverse starts, and pair CD.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # Update helper (inline to avoid closure)
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        S = a.sum()
        if S < 0.01:
            return
        a = a / S
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    mu = 5.0  # Sum constraint penalty

    # === Phase 1: L-BFGS-B multi-start with coarse temps (45% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.45
    print(f"\n=== Phase 1: L-BFGS-B multi-start n={n_base} (budget={t1:.0f}s) ===")

    starts = _make_starts(n_base, best_seq, rng)
    temps_coarse = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3]

    time_per_start = t1 / len(starts)
    for label, a_start in starts:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 10:
            break
        alloc = min(time_per_start, remaining * 0.95)
        if alloc < 3:
            break
        print(f"  L-BFGS-B [{label}] (alloc={alloc:.0f}s)")
        t_start_phase = time.monotonic()
        a_out, _ = lbfbs_anneal(a_start, temps_coarse, mu, alloc, maxiter_per_T=200)
        elapsed_in_phase = time.monotonic() - t_start_phase
        try_update(a_out, f"lbfbs_{label}")

    # === Phase 2: L-BFGS-B refinement with fine temps on best (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: L-BFGS-B fine refinement n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 10:
        temps_fine = [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]
        a_out, _ = lbfbs_anneal(best_seq, temps_fine, mu, t2, maxiter_per_T=500)
        try_update(a_out, "lbfbs_fine1")

        # Second refinement from slightly perturbed
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            a_pert = best_seq + rng.normal(0, 0.02 * best_seq.std(), len(best_seq))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            temps_fine2 = [0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
            a_out2, _ = lbfbs_anneal(a_pert, temps_fine2, mu, min(remaining2 * 0.7, 300), maxiter_per_T=400)
            try_update(a_out2, "lbfbs_pert")

    # === Phase 3: Pair CD polishing (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: Pair CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_out, _ = pair_cd_fast(best_seq, t3, rng, window=35)
        try_update(a_out, "pair_cd_1")

        # L-BFGS-B after PCD
        remaining3 = time_budget - (time.monotonic() - t_global)
        if remaining3 > 30:
            temps_v = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_out2, _ = lbfbs_anneal(best_seq, temps_v, mu, min(remaining3 * 0.5, 200), maxiter_per_T=300)
            try_update(a_out2, "lbfbs_post_pcd")

    # === Phase 4: More pair CD + L-BFGS alternation (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.05
    print(f"\n=== Phase 4: Alternation (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_fast(best_seq, t4 * 0.5, rng, window=50)
        try_update(a_out, "pcd_final")
        remaining4 = time_budget - (time.monotonic() - t_global)
        if remaining4 > 20:
            a_out2, _ = lbfbs_anneal(best_seq, [0.03, 0.01, 0.005, 0.002, 0.001], mu,
                                      min(remaining4 * 0.7, 100), maxiter_per_T=500)
            try_update(a_out2, "lbfbs_final")

    # === Phase 5: Alternate lengths (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_base_final = len(best_seq)
        alt_lengths = [n_base_final + 200, n_base_final - 200, n_base_final + 500]
        for n_alt in alt_lengths:
            if n_alt < 200:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 30:
                break
            print(f"\n=== Alternate length n={n_alt} (remaining={remaining_sub:.0f}s) ===")
            t_alt = min(remaining_sub * 0.35, 150.0)
            init_alt = interpolate_sequence(best_seq, n_alt)
            temps_alt = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
            a_out_alt, _ = lbfbs_anneal(init_alt, temps_alt, mu, t_alt, maxiter_per_T=300)
            try_update(a_out_alt, f"alt_n{n_alt}")
            if try_update is not None and best_score < 1.503:
                print(f"  Breaking alt lengths (already at {best_score:.6f})")
                break

    # === Final: one last PCD pass with time remaining ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Final PCD polish (budget={remaining:.0f}s) ===")
        a_out, _ = pair_cd_fast(best_seq, min(remaining * 0.8, 300), rng, window=25)
        try_update(a_out, "final_pcd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
