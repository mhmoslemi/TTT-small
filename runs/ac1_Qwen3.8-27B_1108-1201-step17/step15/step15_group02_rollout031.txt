1. Fast LSE-Adam gradient descent for initial progress
2. Full-constraint LP best-response with Nesterov acceleration (main workhorse)
3. Pair coordinate descent for final polishing
4. Multi-n exploration for length tuning
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def _get_solver():
    global _cached_solver
    if '_cached_solver' in globals() and _cached_solver is not None:
        return _cached_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _cached_solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _cached_solver

_cached_solver = None

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def lse_adam(a0, time_limit, verbose=False):
    """LSE-Adam gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0
    lr = 0.5 / np.sqrt(n)
    
    n_stages = 15
    temps = np.geomspace(50.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(30, min(300, int(remaining * 15)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = np.maximum((conv - max_c) / T_val, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        if verbose:
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")
    return best_a, best_score

def build_full_T(a):
    """Build the full (2n-1) x n convolution matrix."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            T[k, j] = a[k - j]
    return T

def solve_lp_full(a, M, solver):
    """Solve LP with ALL constraints."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = build_full_T(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def solve_lp_active(a, M, active_set, solver):
    """Solve LP with only active constraints."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    m = len(active_set)
    T = np.zeros((m, n))
    for idx, k in enumerate(active_set):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            T[idx, j] = a[k - j]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and cutting-plane."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()

    best_a = a.copy()
    best_score = fast_conv_self(a).max()

    # Start with top-K active set
    K = min(200, size)
    y = a.copy()
    
    max_iters = 5000

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Determine active set
        if use_full:
            active_set = None  # Use full
        else:
            # Adaptive active set: top peaks
            threshold = M * 0.7
            active_set = np.where(conv_y > threshold)[0]
            if len(active_set) < 50:
                active_set = np.where(conv_y > M * 0.4)[0]
            if len(active_set) > 500:
                # Cap it
                top_k = np.argsort(conv_y)[-500:]
                active_set = top_k
        
        # Solve LP
        if active_set is None:
            b = solve_lp_full(y, M, solver)
        else:
            b = solve_lp_active(y, M, active_set, solver)
        
        if b is None:
            # Fallback: use full set
            b = solve_lp_full(y, M, solver)
        
        if b is None:
            continue
        
        # Check for violations if using active set
        if active_set is not None:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.0001)[0]
            if len(violated) > 0:
                new_active = np.union1d(active_set, violated)
                if len(new_active) <= size:
                    b2 = solve_lp_active(y, M, new_active, solver)
                    if b2 is not None:
                        b = b2
                    else:
                        b2 = solve_lp_full(y, M, solver)
                        if b2 is not None:
                            b = b2
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score

Let me reconsider the overall approach. The key insight is:

1. The LP best-response is the most powerful tool.
2. For n=1600, the full constraint LP (3199 × 1600) should be solvable in <1 second with CLARABEL.

def lse_adam_step(a, T_val, lr, m, v, t_adam):
    """One LSE-Adam step. Returns updated a, m, v, t_adam."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    beta1, beta2 = 0.85, 0.999
    eps = 1e-8
    
    conv = fast_conv_self(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return a, m, v, t_adam
    shifted = np.maximum((conv - max_c) / T_val, -80.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g1 = irfft(np.conj(A) * W, fft_size)[:n]
    g2 = irfft(A * W, fft_size)[:n]
    grad = 2.0 * (g1 + g2)
    
    t_adam += 1
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m / (1.0 - beta1 ** t_adam)
    v_hat = v / (1.0 - beta2 ** t_adam)
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new * norm / s_new
    return a_new, m, v, t_adam

def solve_lp(a, M, active_set, solver):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for k in active_set."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if active_set is None:
        size = 2 * n - 1
        active_set = np.arange(size)
    m = len(active_set)
    T = np.zeros((m, n))
    for idx, k in enumerate(active_set):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            T[idx, j] = a[k - j]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
