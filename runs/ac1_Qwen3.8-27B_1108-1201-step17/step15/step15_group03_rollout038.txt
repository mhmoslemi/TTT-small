import numpy as np
import cvxpy as cp

from scipy.fft import rfft, irfft

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a, n):
    """Build matrix T such that (T @ b)[k] = conv(a,b)[k] for all k."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

def solve_lp_response(a, M, solver, active_set=None, max_iter=5):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Uses iterative constraint addition."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a, n)
    
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and len(active_set) < size:
        constraints = [T[active_set, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Check for violated constraints and add them
    if active_set is not None and len(active_set) < size:
        for _ in range(max_iter):
            if b.sum() < 1e-15:
                break
            conv_ab = fast_conv_cross(a, b)
            violated = np.where(conv_ab > M * 1.001)[0]
            if len(violated) == 0:
                break
            active_set = np.unique(np.concatenate([active_set, violated]))
            
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T[active_set, :] @ b_var <= M]

def lse_smooth_descent(a0, time_limit, verbose=True):
    """Log-sum-exp smoothed gradient descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Geometric temperature schedule
    n_stages = 15
    temps = np.geomspace(50.0, 0.05, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        
        n_steps = max(20, min(200, int((time_limit / n_stages) * 4)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # LSE approximation of max
            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -100, 0)
            w = np.exp(shifted)
            w = w / w.sum()
            
            # Gradient of LSE: sum w[k] * d(conv[k])/d(a)
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            # Adaptive step size
            lr = 1.0 / (T_val * np.sqrt(n) * (1.0 + 0.1 * it))
            
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)

                a_new = a.copy()

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_m:

        if verbose:

def peak_shaving(a0, time_limit, rng, verbose=True):
    """Local search that identifies peak-contributing coordinates and redistributes mass."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Identify top peaks
        n_peaks = min(20, size)
        peak_lags = np.argsort(conv)[-n_peaks:]
        
        # For each peak, compute which coordinates contribute most
        # conv[k] = sum_i a[i] * a[k-i] (with valid indices)
        # Contribution of a[i] to conv[k] is a[k-i] (if valid) + a[i-k] (if valid, for i != k-i)
        
        # Strategy: for each peak, find the pair (i, k-i) that contributes most,
        # and reduce one of them while increasing a coordinate that doesn't contribute to any peak.
        
        improved = False
        
        for lag in peak_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find contributing pairs for this lag
            contributions = []
            i_lo = max(0, lag - n + 1)
            i_hi = min(n - 1, lag)
            for i in range(i_lo, i_hi + 1):
                j = lag - i
                if 0 <= j < n:
                    contributions.append((i, j, a[i] * a[j]))
            
            if len(contributions) == 0:
                continue
            
            # Sort by contribution
            contributions.sort(key=lambda x: -x[2])
            
            # Try to reduce the top contributor
            for (i, j, contrib_val) in contributions[:5]:
                if contrib_val < 1e-15:
                    continue
                
                # Try transferring from i to a random coordinate far from all peaks
                n_targets = 10
                best_gain = 0.0
                best_target = -1
                best_frac = 0.0
                
                for _ in range(n_targets):
                    target = int(rng.integers(n))
                    if target == i or target == j:
                        continue
                    
                    # Compute the effect of moving epsilon from i to target
                    # conv_new[k] = conv_old[k] - epsilon*(a[k-i] + a[i-k] - delta_{k,2i}) 
                    #                               + epsilon*(a[k-target] + a[target-k] - delta_{k,2*target})
                    # (approximately, for small epsilon)
                    
                    # Try several frac values
                    for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                        eps = a[i] * frac
                        if eps < 1e-15:
                            continue
                        
                        # Approximate new conv
                        new_conv = conv.copy()
                        # Remove contribution of a[i]
                        for k in range(max(0, i), min(size, i + n)):
                            j_idx = k - i
                            if 0 <= j_idx < n:
                                new_conv[k] -= a[i] * a[j_idx]
                        # Add contribution of a[target] += eps
                        for k in range(max(0, target), min(size, target + n)):
                            j_idx = k - target
                            if 0 <= j_idx < n:
                                new_conv[k] += eps * a[j_idx]
                        # Also the reverse (already handled by symmetry in conv)
                        # Actually conv is not symmetric in general for non-palindromic sequences
                        # conv[k] = sum_i a[i]*a[k-i], so we need both directions
                        
                        # Correction: the convolution formula means:
                        # conv new = conv old - contribution_from_i + contribution_from_target_increment
                        # But we also need to account for the fact that a[i] decreased and a[target] increased
                        
                        # Let's just compute directly for the affected indices
                        a_trial = a.copy()
                        a_trial[i] -= eps
                        a_trial[target] += eps
                        
                        # Only recompute affected conv values
                        # Actually for efficiency, let's just check the max
                        # The affected indices are those where either i or target appears
                        affected = set()
                        for k in range(max(0, i), min(size, i + n)):
                            affected.add(k)
                        for k in range(max(0, target), min(size, target + n)):
                            affected.add(k)
                        for k in range(max(0, i-n+1), min(size, i+1)):
                            affected.add(k)
                        for k in range(max(0, target-n+1), min(size, target+1)):
                            affected.add(k)
                        
                        new_max = cur_max
                        for k in sorted(affected):
                            # Recompute conv[k]
                            i_lo_k = max(0, k - n + 1)
                            i_hi_k = min(n - 1, k)
                            val = 0.0
                            for ii in range(i_lo_k, i_hi_k + 1):
                                jj = k - ii
                                if 0 <= jj < n:
                                    val += a_trial[ii] * a_trial[jj]
                            if val > new_max:
                                new_max = val
                        
                        gain = cur_max - new_max
                        if gain > best_gain:
                            best_gain = gain
                            best_target = target
                            best_frac = frac
                
                if best_target >= 0 and best_gain > 0:
                    eps = a[i] * best_frac
                    a[i] -= eps
                    a[best_target] += eps
                    s_new = a.sum()
                    a = a / s_new
                    new_m = fast_conv_self(a).max()
                    if new_m < best_m:
                        best_m = new_m
                        best_a = a.copy()
                    improved = True
                    conv = fast_conv_self(a)
                    cur_max = conv.max()
                    break  # Restart peak analysis
        
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
            # Perturb to escape local minimum
            n_pert = max(5, n // 50)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a[idx_pert] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
        else:
            no_improve = 0
    
    return best_a, 2 * n * best_m

def gradient_flow(a0, time_limit, verbose=True):
    """Direct gradient flow on the smoothed objective with adaptive parameters."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    momentum = np.zeros(n)
    beta = 0.95
    base_lr = 0.1
    
    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        max_c = conv.max()
        
        # Adaptive temperature
        T = max(0.01, max_c * 0.02 * (1.0 + 100.0 / (1.0 + it)))
        
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -80, 0)
        w = np.exp(shifted)
        w = w / w.sum()
        
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        
        # Project gradient onto tangent space (sum=0)
        grad_mean = grad.mean()
        grad_tan = grad - grad_mean
        
        lr = base_lr / (1.0 + 0.01 * it)
        momentum = beta * momentum - lr * grad_tan
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_m:
            best_m = sc
            best_a = a.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradFlow it={it} ({elapsed:.0f}s): M={best_m:.8f}, obj={2*n*best_m:.6f}")
    
    return best_a, 2 * n * best_m

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:

    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5

        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        # Approximate optimal: want flat convolution, try raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 7:
        # Multi-modal
        centers = [0.2, 0.5, 0.8]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-20 * (x - c) ** 2)
    elif choice == 8:
        # Power law
        a = x ** 0.5 * (1 - x) ** 0.5
    elif choice == 9:
        # Alternate high/low
        a = np.ones(n)
        a[::2] = 0.5
    elif choice == 10:
        # Sinc-like
        a = np.sinc(4 * (x - 0.5)) ** 2 + 0.05
    else:
        # Random smooth
        a = np.exp(-3.0 * (x - rng.uniform(0.3, 0.7)) ** 2 / 0.1)
    
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def solve_lp_with_constraints(a, M, solver, active_set, max_rounds=8):
    """Solve LP: max sum(b) s.t. (T@b)[active] <= M, b >= 0.
    Iteratively add violated constraints."""
    n = len(a)
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and 0 < len(active_set) < 2 * n - 1:
        constraints = [T[active_set, :] @ b_var <= M]
    else:
        idx = slice(None)
        constraints = [T[idx, :] @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b.sum() < 1e-15:
        return b

    if active_set is not None and len(active_set) < 2 * n - 1:
        for _ in range(max_rounds):
            conv_ab = fast_conv_cross(a, b)
            violated = np.where(conv_ab > M * 1.005 + 1e-10)[0]
            if len(violated) == 0:
                break
            active_set = np.unique(np.concatenate([active_set, violated]))
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T[active_set, :] @ b_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                return None
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                return None
            b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            if b_new.sum() < 1e-15:
                break
            b = b_new

    return b


def lp_best_response(a0, time_limit, solver, verbose=True):
    """LP best-response with adaptive constraints, Nesterov accel, and periodic full solves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        a = a / s

            break

        M = conv_y.max()
        if M < 1e-14:
            break

        # Adaptive constraint selection: use top constraints
        # Use a blend: always include top 5% + random 5% for diversity
        n_top = max(30, int(0.08 * size))
        top_idx = np.argsort(conv_y)[-n_top:]
        # Add some mid-range constraints for robustness
        n_mid = max(10, int(0.02 * size))
        mid_range = np.where((conv_y > M * 0.5) & (conv_y < M * 0.95))[0]
        if len(mid_range) > n_mid:
            mid_idx = rng_global.choice(mid_range, size=n_mid, replace=False)
            active = np.unique(np.concatenate([top_idx, mid_idx]))
        else:
            active = top_idx

        b = solve_lp_with_constraints(y, M, solver, active)
        if b is None:
            # Fallback to all constraints
            b = solve_lp_with_constraints(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum

        # Nesterov with adaptive step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / s_a

        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_m:
            best_m = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_m:
            best_m = sc_a
            best_a = a.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPBR t={t} ({elapsed:.0f}s): M={best_m:.10f}, obj={2*n*best_m:.8f}")

    return best_a, 2.0 * n * best_m


def lse_descent(a0, time_limit, verbose=True):
    """LSE smoothed gradient descent with adaptive temperature and step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Multi-stage temperature
    n_stages = 14
    temps = np.geomspace(60.0, 0.03, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        n_steps = max(20, int(remaining * 5 / (n_stages - si)))
        n_steps = min(n_steps, 400)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -120.0, 0.0)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
                break
            w = w / w_sum

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Project to tangent (sum=0)
            grad = grad - grad.mean()

            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            lr = min(0.5, 1.0 / (T_val * grad_norm * (1.0 + 0.05 * it)))

            a_new = a + lr * (-grad)  # negative gradient
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_m:
                best_m = sc
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si} T={T_val:.3f} ({elapsed:.0f}s): M={best_m:.10f}, obj={2*n*best_m:.8f}")

    return best_a, 2.0 * n * best_m


def momentum_gradient(a0, time_limit, verbose=True):
    """Heavy-ball momentum method on the smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    vel = np.zeros(n)
    beta = 0.93
    base_lr = 0.05

    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        # Adaptive temperature: start high, decay
        T = max(0.02, max_c * 0.015 * (1.0 + 200.0 / (1.0 + it)))

        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -80.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w = w / w_sum

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        grad = grad - grad.mean()  # tangent projection

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        lr = base_lr / (1.0 + 0.005 * it)
        vel = beta * vel - lr * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_m:
            best_m = sc
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MomGrad it={it} ({elapsed:.0f}s): M={best_m:.10f}, obj={2*n*best_m:.8f}")

    return best_a, 2.0 * n * best_m


def peak_shaving(a0, time_limit, rng, verbose=True):
    """Local search: identify peak-contributing coordinates and redistribute mass."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improves = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Find top peaks
        n_peaks = min(15, size)
        peak_lags = np.argsort(conv)[-n_peaks:]

        pass_improved = False

        for lag in peak_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break

            # Find contributing pairs for this lag
            i_lo = max(0, lag - n + 1)
            i_hi = min(n - 1, lag)
            pairs = []
            for i in range(i_lo, i_hi + 1):
                j = lag - i
                if 0 <= j < n:
                    pairs.append((i, j, a[i] * a[j]))

            if not pairs:
                continue

            pairs.sort(key=lambda x: -x[2])

            # Try to reduce top contributors
            for (i, j, cval) in pairs[:4]:
                if cval < 1e-18:
                    continue
                if i == j:
                    # Single coordinate contributes
                    # Try to shift mass from i to a random target
                    for attempt in range(8):
                        target = int(rng.integers(n))
                        if target == i:
                            continue
                        for frac in [0.2, 0.4, 0.6, 0.8]:
                            eps = a[i] * frac
                            if eps < 1e-15:
                                continue
                            # Quick check: compute new max approximately
                            a[i] -= eps
                            a[target] += eps
                            new_conv = fast_conv_self(a)
                            new_max = new_conv.max()
                            if new_max < cur_max - 1e-12:
                                # Accept
                                a = a / a.sum()
                                new_m = fast_conv_self(a).max()
                                if new_m < best_m:
                                    best_m = new_m
                                    best_a = a.copy()
                                conv = new_conv
                                cur_max = conv.max()
                                pass_improved = True
                                total_improves += 1
                                break
                            else:
                                # Revert
                                a[i] += eps
                                a[target] -= eps
                else:
                    # Two different coordinates
                    # Try shifting from i to a target, or from j to a target
                    for source in [i, j]:
                        for attempt in range(5):
                            target = int(rng.integers(n))
                            if target == source or target == (i if source == j else j):
                                continue
                            for frac in [0.2, 0.4, 0.6]:
                                eps = a[source] * frac
                                if eps < 1e-15:
                                    continue
                                a[source] -= eps
                                a[target] += eps
                                new_conv = fast_conv_self(a)
                                new_max = new_conv.max()
                                if new_max < cur_max - 1e-12:
                                    a = a / a.sum()
                                    new_m = fast_conv_self(a).max()
                                    if new_m < best_m:
                                        best_m = new_m
                                        best_a = a.copy()
                                    conv = new_conv
                                    cur_max = conv.max()
                                    pass_improved = True
                                    total_improves += 1
                                    break
                                else:
                                    a[source] += eps
                                    a[target] -= eps
                        if pass_improved:
                            break
                    if pass_improved:
                        break

            if pass_improved:
                break  # Restart with new conv

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 2:
                # Perturb
                n_pert = max(5, n // 40)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
        else:
            no_improve = 0

        if verbose and total_improves % 200 == 0 and total_improves > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improves={total_improves} ({elapsed:.0f}s): M={best_m:.10f}, obj={2*n*best_m:.8f}")

    return best_a, 2.0 * n * best_m


def multi_m_lp(a0, time_limit, solver, verbose=True):

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE warmup (fast) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE warmup n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # === Phase 2: LP best-response (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.28, 450.0)
    print(f"\n=== Phase 2: LP Best-Response n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_best_response(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: Peak Shaving n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = peak_shaving(best_seq, t3, rng, verbose=True)
        try_update(a_out, "peak1")

    # === Phase 4: Momentum gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 140.0)
    print(f"\n=== Phase 4: Momentum Grad n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = momentum_gradient(best_seq, t4, verbose=True)
        try_update(a_out, "momgrad")

    # === Phase 5: Multi-M LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.22, 350.0)
    print(f"\n=== Phase 5: Multi-M LP n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = multi_m_lp(best_seq, t5, solver, verbose=True)
        try_update(a_out, "multim")

    # === Phase 6: LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 6: LP Round 2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = lp_best_response(best_seq, t6, solver, verbose=True)
        try_update(a_out, "lp2")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        # Try longer
        n_long = len(best_seq) + 300
        t7 = min(remaining * 0.30, 120.0)
        print(f"\n=== Phase 7a: Longer n={n_long} (budget={t7:.0f}s) ===")
        if t7 > 20:
            init_long = interpolate_seq(best_seq, n_long)
            a_out, _ = lp_best_response(init_long, t7, solver, verbose=False)
            try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)

        # Try shorter
        if remaining > 50:
            n_short = max(500, len(best_seq) - 300)
            t7b = min(remaining * 0.35, 80.0)
            print(f"\n=== Phase 7b: Shorter n={n_short} (budget={t7b:.0f}s) ===")
            if t7b > 15:
                init_short = interpolate_seq(best_seq, n_short)
                a_out, _ = lse_descent(init_short, t7b, verbose=False)
                a_out2, _ = peak_shaving(a_out, t7b * 0.3, rng, verbose=False)
                try_update(a_out, f"lse_short{n_short}")
                try_update(a_out2, f"peak_short{n_short}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final Polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t8a = min(remaining * 0.4, 50.0)
        a_out, _ = lse_descent(best_seq, t8a, verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t8b = min(remaining * 0.5, 40.0)
            a_out, _ = peak_shaving(best_seq, t8b, rng, verbose=False)
            try_update(a_out, "peak_final")

    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
