    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    return val

def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP-based best response with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    no_improve = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        if t < 30:
            active = np.where(conv_y > M * 0.4)[0]
        elif t < 100:
            active = np.where(conv_y > M * 0.6)[0]
        elif t < 300:
            active = np.where(conv_y > M * 0.75)[0]
        else:
            active = np.where(conv_y > M * 0.85)[0]
        if len(active) < 20:
            active = None

        b = lp_solve_best_response(y, M, solver, active)
        if b is None:
            b = lp_solve_best_response(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        momentum = min(1.0, (t - 1.0) / (t + 2.0))
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / s_a

        y = y_new
        a = a_new

        # Check improvement
        sc = fast_conv_self(y).max()
        improved = False
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        # Perturbation
        if no_improve == 150:
            n_pert = max(5, n // 25)
            rng_p = np.random.default_rng(t)
            idx_p = rng_p.choice(n, size=n_p, replace=False)
            y[idx_p] *= (1.0 + 0.15 * rng_p.standard_normal(n_p))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            no_improve = 0

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            sc_avg = fast_conv_self(avg_check).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-10:
                continue

            # Find j candidates from peak lags
            peak_threshold = cur_max * 0.85
            peak_lags = np.where(conv >= peak_threshold)[0]
            j_candidates = set()
            for kl in peak_lags[:10]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                j_candidates.add(j1)
                j_candidates.add(j2)
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector for moving mass from i to j
                g = np.zeros(size)
                # Adding to j: conv += a shifted by j (both directions)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Removing from i: conv -= a shifted by i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta via sampling
                n_samples = 15
                deltas = np.linspace(0.05, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    # Correct diagonal terms
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v - 1e-14:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * 1.0 / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_pert = max(5, n // 30)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Asymmetric multi-peak
        a = 0.4 * np.exp(-8.0 * (x - 0.2) ** 2) + 0.3 * np.exp(-6.0 * (x - 0.5) ** 2) + 0.3 * np.exp(-8.0 * (x - 0.8) ** 2)
    elif choice == 7:
        # Oscillatory
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.cos(2 * np.pi * 13 * x)
        a = np.maximum(a, 0.05)
    elif choice == 8:
        # Serrated
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        # Power law
        a = np.power(1.0 - x, 0.5) + 0.3
    elif choice == 10:
        # Two-scale structure
        a = 0.6 + 0.4 * np.cos(2 * np.pi * 2 * x)
        a = a * (1.0 + 0.2 * np.cos(2 * np.pi * 17 * x))
        a = np.maximum(a, 0.05)
    else:
        # Random smooth
        freqs = rng.integers(1, 20, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        a = np.ones(n) * 0.5
        for f, amp, ph in zip(freqs, amps, phases):
            a += amp * np.cos(2 * np.pi * f * x + ph)
        a = np.maximum(a, 0.05)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines mirror descent, LP refinement, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Mirror Descent V2 (main fast optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 1: Mirror Descent V2 n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = mirror_descent_v2(best_seq, t1, verbose=True)
        try_update(a_out, f"mdv2_n{n1}")

    # === Phase 2: LP Best-Response Refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n2}")

    # === Phase 3: Mirror Descent round 2 (from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: MD V2 round 2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = mirror_descent_optimize(best_seq, t3, verbose=True)
        try_update(a_out, f"md_n{n3}")

    # === Phase 4: LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n4}")

    # === Phase 5: Pair CD Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Pair CD n={n5} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=False)
        try_update(a_out, f"pcd_n{n5}")

    # === Phase 6: Length extension ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(n_base + 400, 2200)
    print(f"\n=== Phase 6: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t_long_md = min(remaining * 0.4, 150.0)
        a_out, _ = mirror_descent_v2(init_long, t_long_md, verbose=False)
        try_update(a_out, f"md_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t_long_lp = min(remaining * 0.5, 120.0)
            a_out, _ = lp_nesterov_optimize(best_seq, t_long_lp, solver, verbose=False)
            try_update(a_out, f"lp_long_{n_long}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t_f1 = min(remaining * 0.4, 60.0)
        a_out, _ = pair_cd_optimize(best_seq, t_f1, rng, verbose=False)
        try_update(a_out, "pcd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_f2 = min(remaining * 0.6, 40.0)
            a_out, _ = mirror_descent_optimize(best_seq, t_f2, verbose=False)
            try_update(a_out, "md_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = min(best_score, final_score)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
