import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_conv_self(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Compute cross-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def get_solver():
    """Find a working CVXPY solver."""
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Using fallback solver: ECOS")
    return cp.ECOS

def lp_best_response(a, M, active_set, solver):
    """
    Solve LP: max sum(b) s.t. b >= 0, T_active @ b <= M
    where T_active is the constraint matrix for the active set.
    """
    n = len(a)
    size = 2 * n - 1
    
    if active_set is None or len(active_set) == 0:
        active_set = np.arange(size)
    
    active_set = active_set[np.sort(active_set)]
    n_active = len(active_set)
    
    # Build sparse constraint matrix
    # Row k of T: (T @ b)_k = sum_i a_{k-i} * b_i
    # For a given k, the non-zero entries are at positions i where 0 <= k-i < n
    # i.e., max(0, k-n+1) <= i <= min(n-1, k)
    
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_set):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])
    
    if len(rows) == 0:
        return np.ones(n) / n
    
    T_active = cp.SparseMatrix((vals, (rows, cols)), shape=(n_active, n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_active @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception as e:
        print(f"    LP solve error: {e}")
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def ipm_lp_best_response(a, M, solver, max_iter=5, verbose=False):
    """
    LP with iterative constraint generation (separation oracle).
    Start with a small active set, solve, check for violations, add violated constraints.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Initial active set: constraints where conv(a,a) is above threshold
    conv_aa = fast_conv_self(a)
    threshold = M * 0.5
    active_set = np.where(conv_aa > threshold)[0]
    
    if len(active_set) < 5:
        active_set = np.where(conv_aa > M * 0.3)[0]
    if len(active_set) < 5:
        active_set = np.arange(size)
    
    b = None
    for it in range(max_iter):
        b = lp_best_response(a, M, active_set, solver)
        if b is None:
            # Try full constraint set
            b = lp_best_response(a, M, np.arange(size), solver)
            if b is None:
                return None
            return b
        
        # Check for constraint violations
        cross_conv = fast_conv_cross(a, b)
        violated = np.where(cross_conv > M * 1.001)[0]
        
        if len(violated) == 0:
            if verbose:
                print(f"    IPM LP converged in {it+1} iterations, active_set size={len(active_set)}")
            return np.maximum(b, 0.0)
        
        # Add violated constraints to active set
        active_set = np.unique(np.concatenate([active_set, violated]))
        
        if verbose:
            print(f"    IPM LP iter {it+1}: added {len(violated)} violated constraints, total={len(active_set)}")
        
        # If active set is too large, reduce by keeping only the most violated
        if len(active_set) > 2 * size:
            # This shouldn't happen but just in case
            active_set = np.arange(size)
    
    return np.maximum(b, 0.0)

def subgradient_polish(a0, time_limit, verbose=True):
    """
    Polish using exact subgradient of max(conv(a,a)).
    The subgradient at the maximizing index k* is: grad_j = 2 * a_{k*-j}
    Use Polyak step size with moving average.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Track running best for Polyak step size
    f_star = best_score * 0.99  # optimistic lower bound estimate
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        k_star = np.argmax(conv)
        
        if M < 1e-12:
            break
        
        # Subgradient: grad_j = 2 * a_{k_star - j} (with boundary handling)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Polyak step size: eta = (f(a) - f*) / ||grad||^2
        # f(a) = M (since we normalized sum = sqrt(2n))
        # f* is a lower bound; use 0.99 * current best as proxy
        eta = max(1e-8, (M - f_star) / (grad_norm * grad_norm + 1e-15))
        eta = min(eta, 1.0)  # cap step size
        
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            f_star = best_score * 0.995  # update lower bound estimate
        
        a = a_new
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter {it} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score

def lse_adam_optimize(a0, time_limit, verbose=True):

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    n_stages = 15
    temps = np.geomspace(50.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(150, int(remaining * 5)))
        lr = 0.3 / np.sqrt(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            
            grad = 2.0 * (ca + aw)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and (si % 3 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")
    
    return best_a, best_score

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue
            
            # Try reducing a[i] and redistributing
            # The effect on convolution: removing a[i] subtracts a shifted copy of a
            # from the convolution
            g_remove = np.zeros(size)
            # (a*a) has contribution from a[i]: a[i] * a shifted
            # Specifically, (a*a)_k includes a[i]*a[k-i] and a[k-i]*a[i]
            for j in range(n):
                k1 = i + j  # a[i] * a[j] contributes to position i+j
                if k1 < size:
                    g_remove[k1] += a[i] * a[j]
                k2 = j + i  # same thing, already counted
            # Actually, the contribution of a[i] to the convolution is:
            # For each j where a[j] > 0, position i+j gets a[i]*a[j]
            # And position j+i gets a[j]*a[i] (same position)
            # So total contribution at position k is: a[i] * a[k-i] (if 0 <= k-i < n)
            # Plus a[k-i] * a[i] (same) -- wait, no.

            # Terms involving a[i]: j=i gives a[i]*a[k-i], and k-j=i gives j=k-i, term a[k-i]*a[i]
            # So contribution = a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i] (when k!=2i)
            # When k=2i: a[i]*a[i] = a[i]^2 (counted once in the sum, but derivative is 2*a[i])
            
            # For coordinate descent, if we change a[i] -> a[i] + delta:
            # New (a*a)_k = old (a*a)_k + delta * (a[k-i] + a[k-i]) + delta^2 * (1 if k=2i else 0)
            # = old (a*a)_k + 2*delta*a[k-i] + delta^2*(1 if k=2i else 0)
            
            # We want to minimize max_k of the new convolution
            # This is a 1D optimization problem
            
            # Use golden section search
            lo = 0.0
            hi = ai + 0.5 * ai  # try increasing by up to 50%
            # Actually, let's try a range: [0, ai*2]
            lo = 0.0
            hi = ai * 1.5 if ai > 0 else 0.1
            
            if hi <= lo:
                continue
            
            # Binary search for the optimal delta
            best_delta = 0.0
            best_obj = cur_max
            n_samples = 30
            candidates = np.linspace(lo, hi, n_samples)
            
            for new_val in candidates:
                if new_val < 0:
                    continue
                delta = new_val - ai
                if abs(delta) < 1e-15:
                    continue
                
                # Update only the affected positions
                # Positions affected: i + j for j in range(n), and j + i for j in range(n)
                # Simplified: position k is affected if 0 <= k-i < n (i.e., i <= k <= i+n-1)
                # or 0 <= k-i < n (same condition)
                # So affected positions are [i, i+n-1]
                
                # The change at position k is: 2*delta*a[k-i] + delta^2*(1 if k==2i else 0)
                k_start = i
                k_end = min(i + n, size)
                
                affected = conv[k_start:k_end].copy()
                for pos in range(k_start, k_end):
                    j = pos - i
                    affected[pos - k_start] += 2.0 * delta * a[j]
                if 2 * i < size:
                    affected[2 * i - k_start] += delta * delta
                
                obj = max(cur_max, affected.max())
                if obj < best_obj:
                    best_obj = obj
                    best_delta = new_val - ai
            
            if abs(best_delta) > 1e-12:
                a[i] = max(0, ai + best_delta)
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = ai
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[i] = ai
                        a = a * norm / a.sum()

            no_improve += 1

                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 8:
                    break
        else:
            no_improve = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD polish ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)

    elif choice == 3:

        c2 = rng.uniform(0.6, 0.8)
        a = np.exp(-40.0 * (x - c1) ** 2) + np.exp(-40.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-5.0 * (x - 0.5)) + 0.3
    elif choice == 7:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 8:
        a = np.sinc(x - 0.5) ** 2 + 0.05
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.01 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed Adam for initial optimization
2. LP best-response with iterative constraint generation (IPM) - key improvement

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_conv_self(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    """Compute cross-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def get_solver():
    """Find a working CVXPY solver."""
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Using fallback solver: ECOS")
    return cp.ECOS


def lp_best_response(a, M, active_set, solver):
    """
    Solve LP: max sum(b) s.t. b >= 0, T_active @ b <= M
    """
    n = len(a)
    size = 2 * n - 1

    if active_set is None or len(active_set) == 0:
        active_set = np.arange(size)

    active_set = active_set[np.sort(active_set)]
    n_active = len(active_set)

    if n_active == 0:
        return np.ones(n) / n

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []

    for idx, k in enumerate(active_set):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])

    if len(rows) == 0:
        return np.ones(n) / n

    T_active = cp.SparseMatrix((vals, (rows, cols)), shape=(n_active, n))
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_active @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def ipm_lp_best_response(a, M, solver, max_iter=5, verbose=False):
    """
    LP with iterative constraint generation.
    Start with a small active set, solve, check violations, add violated constraints.
    """
    n = len(a)
    size = 2 * n - 1

    conv_aa = fast_conv_self(a)
    threshold = M * 0.5
    active_set = np.where(conv_aa > threshold)[0]

    if len(active_set) < 5:
        active_set = np.where(conv_aa > M * 0.3)[0]
    if len(active_set) < 5:
        active_set = np.arange(size)

    b = None
    for it in range(max_iter):
        b = lp_best_response(a, M, active_set, solver)
        if b is None:
            b = lp_best_response(a, M, np.arange(size), solver)
            if b is None:
                return None
            return b

        # Check for constraint violations
        cross_conv = fast_conv_cross(a, b)
        violated = np.where(cross_conv > M * 1.001)[0]

        if len(violated) == 0:
            if verbose:
                print(f"    IPM LP converged in {it+1} iters, active size={len(active_set)}")
            return np.maximum(b, 0.0)

        active_set = np.unique(np.concatenate([active_set, violated]))

        if verbose:
            print(f"    IPM LP iter {it+1}: +{len(violated)} violated, total={len(active_set)}")

        if len(active_set) >= size:
            b = lp_best_response(a, M, np.arange(size), solver)
            if b is None:
                return None
            return b

    return np.maximum(b, 0.0)

        g = b * norm / b_sum

        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """
    Polish using exact subgradient of max(conv(a,a)).
    Subgradient at argmax k*: grad_j = 2 * a_{k*-j}
    Uses Polyak step size with adaptive lower bound.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    # Lower bound estimate (start optimistic, tighten over time)
    f_star = best_score * 0.99
    avg_a = a.copy()
    avg_count = 1

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()
        k_star = int(np.argmax(conv))

        if M < 1e-12:
            break

        # Subgradient: grad_j = 2 * a_{k_star - j}
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq < 1e-15:
            break

        # Polyak step size
        gap = M - f_star
        if gap <= 0:
            eta = 1e-6
        else:
            eta = gap / grad_norm_sq
        eta = np.clip(eta, 1e-8, 0.5)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        # Moving average (Rogozin's method)
        avg_a = avg_a + (a_new - avg_a) / avg_count
        avg_count += 1
        avg_a = avg_a * norm / avg_a.sum()

        sc = fast_conv_self(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            f_star = best_score * 0.997

        # Also check the averaged iterate
        sc_avg = fast_conv_self(avg_a).max()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
            f_star = best_score * 0.997

        a = a_new

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")

    return best_a, best_score


def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE-smoothed Adam optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 15
    temps = np.geomspace(50.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(150, int(remaining * 5)))
        lr = 0.3 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]

            grad = 2.0 * (ca + aw)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 3 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, rng, verbose=True):
    """
    Coordinate descent: for each coordinate, find best value via 1D search.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Sample candidate values for a[i]
            lo = 0.0
            hi = ai * 2.0 if ai > 0 else 0.1
            n_samples = 25
            candidates = np.linspace(lo, hi, n_samples)

            best_obj = cur_max
            best_new_val = ai

            for new_val in candidates:
                if new_val < 0:
                    continue
                delta = new_val - ai
                if abs(delta) < 1e-15:
                    continue

                # Only positions [i, i+n-1] are affected
                k_start = i
                k_end = min(i + n, size)

                affected = conv[k_start:k_end].copy()
                a_seg = a[0:min(n, k_end - k_start)]
                affected += 2.0 * delta * a_seg
                if 2 * i >= k_start and 2 * i < k_end:
                    affected[2 * i - k_start] += delta * delta

                obj = max(cur_max, float(affected.max()))
                if obj < best_obj:
                    best_obj = obj
                    best_new_val = new_val

            if best_new_val != ai:
                a[i] = max(0.0, best_new_val)
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = ai
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[i] = ai
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                n_perturb = max(3, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 8:
                    break
        else:
            no_improve = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD it={it} ({elapsed:.0f}s): best={best_score:.8f}, no_imp={no_improve}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        a = np.exp(-40.0 * (x - c1) ** 2) + np.exp(-40.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-5.0 * (x - 0.5)) + 0.3
    elif choice == 7:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.1, 0.9, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.03, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 8:
        a = np.sinc(x - 0.5) ** 2 + 0.05
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.01 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    """Evaluate using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def check_update(state, a, label):
    """Check if candidate improves and update state dict."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = eval_fn(a)
    if np.isfinite(sc) and sc < state['best_score']:
        old = state['best_score']
        state['best_score'] = sc
        state['best_seq'] = a.copy()
        elapsed = time.monotonic() - state['t_global']
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old:.6f}, n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    state = {'best_seq': None, 'best_score': float('inf'), 't_global': t_global}

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            state['best_seq'] = p / p.sum()
            state['best_score'] = eval_fn(state['best_seq'])
            print(f"Parent loaded: n={len(state['best_seq'])}, score={state['best_score']:.6f}")
    except Exception:
        pass

    if state['best_seq'] is None:
        n_base = 1600
        state['best_seq'] = structured_init(n_base, rng)
        state['best_score'] = eval_fn(state['best_seq'])
        print(f"Default init: n={n_base}, score={state['best_score']:.6f}")
    else:
        n_base = len(state['best_seq'])

    # Phase 1: Quick LSE-Adam warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(state['best_seq'], t1, verbose=True)
        check_update(state, a_out, f"lse1_n{n_base}")

    # Phase 2: LP Nesterov with IPM (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov (IPM) n={len(state['best_seq'])} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(state['best_seq'], t2, solver, verbose=True)
        check_update(state, a_out, f"lp1_n{len(state['best_seq'])}")

    # Phase 3: Subgradient polishing
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: Subgradient polish n={len(state['best_seq'])} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = subgradient_polish(state['best_seq'], t3, verbose=True)
        check_update(state, a_out, f"subgrad_n{len(state['best_seq'])}")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.22, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(state['best_seq'])} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(state['best_seq'], t4, solver, verbose=True)
        check_update(state, a_out, f"lp2_n{len(state['best_seq'])}")

    # Phase 5: Subgradient polish round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.1, 120.0)
    print(f"\n=== Phase 5: Subgradient round 2 n={len(state['best_seq'])} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = subgradient_polish(state['best_seq'], t5, verbose=True)
        check_update(state, a_out, f"subgrad2_n{len(state['best_seq'])}")

    # Phase 6: Coordinate descent final polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.18, 200.0)
    print(f"\n=== Phase 6: CD polish n={len(state['best_seq'])} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = coordinate_descent_polish(state['best_seq'], t6, rng, verbose=True)
        check_update(state, a_out, f"cd_n{len(state['best_seq'])}")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        lengths_to_try = [n_base + 200, n_base + 500, n_base - 300, n_base + 800]
        for n_try in lengths_to_try:
            if remaining < 40:
                break
            if n_try < 200 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            t_try = min(remaining * 0.3, 45.0)
            if t_try < 15:
                continue
            print(f"\n=== Multi-length: n={n_try} (budget={t_try:.0f}s) ===")
            init_try = interpolate_seq(state['best_seq'], n_try)
            a_out, _ = lse_adam_optimize(init_try, min(t_try, 20.0), verbose=False)
            check_update(state, a_out, f"lse_ml_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 25:
                a_out, _ = nesterov_lp_optimize(a_out, min(remaining * 0.4, 25.0), solver, verbose=False)
                check_update(state, a_out, f"lp_ml_n{n_try}")

    # Final evaluation
    if state['best_seq'] is not None:
        final_score = eval_fn(state['best_seq'])
        if np.isfinite(final_score) and final_score < state['best_score']:
            state['best_score'] = final_score
        print(f"\nFINAL: score={state['best_score']:.6f}, n={len(state['best_seq'])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in state['best_seq']]
