This is a linear program. But we can also do a simpler greedy approach:

- Take a gradient step in the negative direction (projected onto the non-negative simplex)

The subgradient of max_k conv(a,b)_k at the maximizing index k* is:
d/da conv(a,b)_{k*} = shift of b (or shift of a, depending on the definition)

Specifically, conv(a,b)[k] = sum_i a[i]*b[k-i], so:
d/da[j] conv(a,b)[k] = b[k-j]

So the subgradient at a with respect to the constraint conv(a,a)[k*] <= M is:
grad[j] = a[k*-j] (for valid j)

This is just a shifted version of a!

So the mirror descent update would be:
log(a_new) = log(a) - eta * grad
where grad[j] = a[k*-j] (the "reversed and shifted" version of a)

This is actually very efficient to compute!

def mirror_descent_step(a, eta):
    n = len(a)
    conv = fast_autocorr(a)
    k_star = np.argmax(conv)
    # Subgradient: grad[j] = a[k_star - j] for valid j
    # k_star - j ranges from 0 to n-1 when j ranges from k_star-(n-1) to k_star
    # So valid j: max(0, k_star-n+1) to min(n-1, k_star)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n-1, k_star)
    grad = np.zeros(n)
    # For j in [j_lo, j_hi]: grad[j] = a[k_star - j]
    # k_star - j ranges from k_star - j_hi to k_star - j_lo
    k_lo = k_star - j_hi  # = max(0, k_star-(n-1)) = 0 if k_star < n-1, else k_star-n+1
    k_hi = k_star - j_lo  # = min(n-1, k_star)
    grad[j_lo:j_hi+1] = a[k_lo:k_hi+1][::-1]
    # Mirror descent update
    log_a = np.log(a + 1e-30) - eta * grad
    a_new = np.exp(log_a - log_a.max())
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Principled initialization with ramp/sine profiles
2. Mirror descent (multiplicative weights) with adaptive step size
3. LP best-response with Nesterov acceleration (refinement)
4. Coordinate descent for fine-tuning

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_autocorr(a_norm)
    return 2.0 * n * conv.max()

def mirror_descent(a_init, time_limit, verbose=True, seed=42):
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.time()
    eta = 1.0 / np.sqrt(n)
    rng = np.random.default_rng(seed)
    n_iters = 0
    stagnation = 0
    
    while time.time() - t0 < time_limit:
        conv = fast_autocorr(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad[m] = a[k_star - m] if 0 <= k_star - m < n
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        idx = np.arange(j_lo, j_hi + 1)
        src = k_star - idx
        grad[idx] = a[src]
        
        # Try a few step sizes
        best_local_val = conv.max()
        best_local_a = a.copy()
        for eta_try in [eta * 0.5, eta, eta * 2.0]:
            log_a = np.log(a) - eta_try * grad
            a_try = np.exp(log_a - log_a.max())
            a_try = a_try / a_try.sum()
            val = fast_autocorr(a_try).max()
            if val < best_local_val:
                best_local_val = val
                best_local_a = a_try
                eta = eta_try  # Remember good step size
        
        if best_local_val < best_val - 1e-15:
            best_val = best_local_val
            best_a = best_local_a.copy()
            a = best_local_a
            stagnation = 0
        else:
            a = best_local_a  # Still move to best local
            stagnation += 1
            eta *= 0.95
            if stagnation > 200:
                # Add noise and restart
                noise = rng.normal(0, 0.01, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 1e-30)
                a = a_pert / a_pert.sum()
                eta = 1.0 / np.sqrt(n)
                stagnation = 0
        
        n_iters += 1
        if verbose and n_iters % 100 == 0:
            elapsed = time.time() - t0
            print(f"    MD iter={n_iters} ({elapsed:.1f}s): best={best_val:.8f}, eta={eta:.6f}")
    
    return best_a, best_val

def lp_best_response(a, M, solver):
    """Solve the LP best-response problem."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    
    # Build the constraint matrix T where (T @ b)[k] = conv(a, b)[k]
    # conv(a,b)[k] = sum_i a[i] * b[k-i]
    # So (T @ b)[k] = sum_j T[k,j] * b[j] where T[k,j] = a[k-j] if 0<=k-j<n
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraints: for each k, sum_j T[k,j] * b[j] <= M
    # T[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a banded matrix
    # For efficiency, use the fact that it's a convolution matrix
    
    # Actually, let's use scipy's convolve matrix or build it explicitly
    # For n up to a few thousand, building the full (2n-1) x n matrix is fine
    
    T = np.zeros((size, n))
    for j in range(n):
        # T[k,j] = a[k-j] for k = j, j+1, ..., j+n-1
        T[j:j+n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s  # Normalize to sum=1
