import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def make_conv_matrix(a, indices):
    """Build T matrix: T @ x = conv(a, x) evaluated at given lag indices."""
    n = len(a)
    T = np.zeros((len(indices), n))
    for row, k in enumerate(indices):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            idxs = k - np.arange(js, je + 1)
            T[row, js:je + 1] = a[idxs]
    return T


def lp_solve_active(a, M, solver, active_idx, warm_b=None):
    """Solve max sum(b) s.t. conv(a,b)[k] <= M for k in active_idx, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = make_conv_matrix(a, active_idx)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    if warm_b is not None:
        b_var.value = warm_b
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lse_gradient_descent(a0, time_limit, verbose=True):
    """Adaptive LSE (log-sum-exp) smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(100.0, 0.15, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(200, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def adaptive_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with adaptive constraint growth and Nesterov."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    warm_b = a.copy()

    # Start with top constraints
    conv_y = fast_conv_self(y)
    M = conv_y.max()
    n_active = min(40, size)
    active_idx = np.argsort(conv_y)[-n_active:]

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive active set: top constraints by value + some random
        frac = min(1.0, 30.0 / size + 0.02 * t / 200.0)
        n_act = max(20, int(frac * size))
        n_act = min(n_act, size)
        top_idx = np.argsort(conv_y)[-n_act:]

        # Add previously active to maintain continuity
        if len(active_idx) > 0:
            combined = np.unique(np.concatenate([top_idx, active_idx]))
            if len(combined) > n_act:
                # Keep top n_act by conv value
                vals = conv_y[combined]
                keep = np.argsort(vals)[-n_act:]
                combined = combined[keep]
            active_idx = combined
        else:
            active_idx = top_idx

        b = lp_solve_active(y, M, solver, active_idx, warm_b)
        if b is None:
            # Try with fewer constraints
            small_idx = np.argsort(conv_y)[-15:]
            b = lp_solve_active(y, M, solver, small_idx, warm_b)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify full constraint (sample-based for speed)
        check_conv = fast_conv_cross(y, g)
        violated = check_conv > M * 1.005
        if violated.sum() > 5:
            # Add violated to active set and re-solve
            add_idx = np.where(violated)[0][:20]
            active_idx = np.unique(np.concatenate([active_idx, add_idx]))
            active_idx = active_idx[np.argsort(conv_y[active_idx])][-60:]
            b2 = lp_solve_active(y, M, solver, active_idx, warm_b)
            if b2 is not None and b2.sum() > 1e-15:
                b = b2
                g = b * norm / b.sum()

        warm_b = b.copy()

        # Nesterov with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, rng, verbose=True):
    """Target top peaks by transferring mass from high-pressure to low-pressure coords."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        peak_threshold = cur_max * 0.88
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Generate candidate transfer targets
            candidates = set()
            for _ in range(8):
                candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                if kl >= i:
                    j1 = kl - i
                    if j1 != i:
                        candidates.add(j1)
                j2 = (i * 2 - kl + size) % n
                if j2 != i and j2 < n:
                    candidates.add(j2)
            candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]

                # Compute convolution change: moving delta from i to j
                # conv_new[k] = conv[k] + delta*(a[k-j]-a[k-i]) + delta^2*(delta terms at 2i, 2j, i+j)
                # Simplified: for small delta, conv change ~ delta * (row_j - row_i) of T
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 20
                max_d = min(ai * 0.9, norm * 0.1)
                if max_d < 1e-10:
                    continue
                deltas = np.linspace(0.0, max_d, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross term at i+j and |i-j|
                    ij = i + j
                    if ij < size:
                        new_conv[ij] = conv[ij] + 2 * d * (aj - ai) + 2 * d * d
                    diff = abs(i - j)
                    if diff < size and diff != 0:
                        # Already included in g
                        pass
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                    continue
                a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                conv = fast_conv_self(a)
                cur_max = conv.max()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(5, n // 50)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

    return best_a, best_score


def spectral_init(n, rng):
    """Initialize using spectral considerations - aim for flat power spectrum."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        # Raised cosine family
        beta = 0.5 + rng.random()
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (2 * beta)))
        a = np.where(np.abs(x - 0.5) < beta, a, 0.0)
    elif choice == 1:
        # Sinc-like
        a = np.sinc(2.0 * (x - 0.5) * rng.uniform(1.5, 3.0)) ** 2
    elif choice == 2:
        # Multi-lobe
        n_lobe = rng.integers(2, 5)
        centers = np.sort(rng.random(n_lobe))
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.05, 0.2)
            a += np.exp(-0.5 * ((x - c) / width) ** 2)
    elif choice == 3:
        # Power-law envelope
        p = rng.uniform(0.5, 2.0)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** p
    elif choice == 4:
        # Flat with taper
        taper = rng.uniform(0.1, 0.4)
        a = np.ones(n)
        for i in range(int(taper * n)):
            a[i] = 0.5 * (1 + np.cos(np.pi * i / (taper * n)))
            a[n - 1 - i] = a[i]
    elif choice == 5:
        # Beta distribution shape
        alpha_b = rng.uniform(1.5, 4.0)
        beta_b = alpha_b
        from scipy.stats import beta as beta_dist
        a = beta_dist.pdf(x, alpha_b, beta_b)
    elif choice == 6:
        # Polynomial
        coeffs = rng.random(rng.integers(3, 7))
        a = np.polyval(coeffs, 2 * x - 1)
    else:
        # Sum of cosines
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(1, 4)):
            f = rng.integers(1, 8)
            ph = rng.random() * np.pi
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * x + ph)

    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    # === PHASE 5: Try slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        for n_try in [n_base + 200, n_base - 200]:
            if n_try < 100:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            t5 = min(remaining * 0.3, 150.0)
            print(f"\n=== Phase 5: Length {n_try} (budget={t5:.0f}s) ===")
            init_long = interpolate_seq(best_seq, n_try)
            # Quick LSE then LP
            t5a = t5 * 0.3
            a_out, _ = lse_gradient_descent(init_long, t5a, verbose=False)
            try_update(a_out, f"lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t5b = min(remaining * 0.5, 100.0)
                a_out, _ = adaptive_lp_optimize(a_out if a_out is not None else init_long, t5b, solver, verbose=False)
                try_update(a_out, f"lp_n{n_try}")

    # === PHASE 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        # LSE polish
        t6a = min(remaining * 0.3, 60.0)
        a_out, _ = lse_gradient_descent(best_seq, t6a, verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.5, 50.0)
            a_out, _ = peak_coordinate_descent(best_seq, t6b, rng, verbose=False)
            try_update(a_out, "pcd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6c = min(remaining * 0.7, 80.0)
            a_out, _ = adaptive_lp_optimize(best_seq, t6c, solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
