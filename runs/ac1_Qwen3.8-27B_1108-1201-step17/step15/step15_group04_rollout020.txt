        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

            con = constraints[0]
            if con.dual_value is not None:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals

def fista_power_sum(a0, time_limit, p=8.0, verbose=False):
    """FISTA minimization of power-sum smoothed max of autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    x = a.copy()
    z = a.copy()
    t_k = 1.0
    step = 1.0 / (4.0 * norm)
    p_adapt = p

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_z = fast_conv_self(z)
        max_c = conv_z.max()
        if max_c < 1e-15:
            break

        w = np.power(conv_z, p_adapt - 1)
        w_sum = np.sum(np.power(conv_z, p_adapt))
        if w_sum < 1e-30:
            break

        scale = p_adapt * np.power(w_sum, 1.0 / p_adapt - 1.0)
        w_weighted = scale * w

        W = rfft(w_weighted, fft_size)
        A = rfft(z, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        # Backtracking line search
        for bs in range(20):
            x_new = z - step * grad
            x_new = np.maximum(x_new, 0.0)
            s_new = x_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            x_new = x_new * norm / s_new
            sc = fast_conv_self(x_new).max()
            f_z = np.power(np.sum(np.power(conv_z, p_adapt)), 1.0 / p_adapt)
            f_x = sc  # approximate
            if f_x <= f_z - 0.5 * step * step * np.dot(grad, x_new - z) + 1e-8:
                break
            step *= 0.5

        x_new = z - step * grad
        x_new = np.maximum(x_new, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new * norm / s_new

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        z = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        z = np.maximum(z, 0.0)
        s_z = z.sum()
        if s_z < 1e-15:
            z = x_new.copy()
        else:
            z = z * norm / s_z

        x = x_new
        t_k = t_new

        sc_x = fast_conv_self(x).max()
        if sc_x < best_score:
            best_score = sc_x
            best_a = x.copy()

        # Adaptive p: increase p as we converge (sharper approximation to max)
        if it % 100 == 0 and it > 0:
            p_adapt = min(p_adapt * 1.15, 50.0)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it} p={p_adapt:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lp_best_response(a0, time_limit, solver, verbose=True, use_mult=True):
    """LP best-response with adaptive step and multiplicative/additive weights."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    no_improve = 0
    prev_duals = None

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        if t < 20:
            thresh = 0.4
        elif t < 100:
            thresh = 0.6
        elif t < 500:
            thresh = 0.75
        else:
            thresh = 0.85
        active = np.where(conv_y > M * thresh)[0]
        if len(active) < 20:
            active = np.where(conv_y > M * 0.3)[0]

        b, duals = lp_solve(y, M, solver, active)
        if b is None or b.sum() < 1e-15:
            b, duals = lp_solve(y, M, solver, None)
        if b is None or b.sum() < 1e-15:
            continue

        b_sum = b.sum()
        g = b * norm / b_sum

        # Verify dropped constraints
        if active is not None and len(active) < size:
            check = fast_conv_cross(y, g)
            violated = np.where(check > M * (1 + 1e-3))[0]
            if len(violated) > 0:
                b_full, _ = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    g = b * norm / b.sum()

        # Adaptive step size based on LP gap
        gap = 1.0 - b_sum / norm
        gap = max(gap, 0.01)
        if use_mult:
            alpha = min(0.5, 2.0 / (t + 2.0)) * (1.0 + gap)
        else:
            alpha = min(0.5, 2.0 / (t + 2.0)) * (1.0 + gap)

        if use_mult:
            # Multiplicative (geometric mean) update
            safe_y = np.maximum(y, 1e-30)
            safe_g = np.maximum(g, 1e-30)
            y_new = np.power(safe_y, 1.0 - alpha) * np.power(safe_g, alpha)
        else:
            y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov extrapolation for the tracking variable
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0
            avg_a += a
            n_avg += 1

        # Periodic perturbation
        if no_improve == 80 and t > 100:
            rng = np.random.default_rng(t * 7919)
            n_perturb = max(20, n // 15)
            idx_perturb = rng.choice(n, size=n_perturb, replace=False)
            y[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0

        # Averages check
        if n_avg > 0 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
                avg_a = np.zeros(n)
                n_avg = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s) gap={gap:.4f}: best={best_score:.8f}")

    return best_a, best_score

def kkt_refine(a0, time_limit, solver, verbose=True):
    """Use LP dual variables for KKT-based gradient refinement."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()

        active = np.where(conv > M * 0.8)[0]
        b, duals = lp_solve(a, M, solver, active)

        if duals is None or b is None:
            b2, duals2 = lp_solve(a, M, solver, None)
            if duals2 is not None and b2 is not None:
                duals = duals2
            else:
                duals = np.zeros(size)
                duals[np.argmax(conv)] = 1.0

        # Gradient from duals
        A = rfft(a, fft_size)
        D = rfft(duals, fft_size)
        g1 = irfft(np.conj(A) * D, fft_size)[:n]
        g2 = irfft(A * D, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        step = 1.0
        found = False
        for trial in range(25):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < M - 1e-10:
                a = a_new
                found = True
                break
            step *= 0.5

        if not found:
            # Try smaller fixed step
            a_new = a - 0.01 * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc < M:
                    a = a_new
                    found = True

        if found:
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, seed, verbose=False):
    """Pair coordinate descent for final polish."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Find peak lags for structured candidates
        peak_thr = cur_max * 0.85
        peak_lags = np.where(conv >= peak_thr)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:6]:
                j1 = int(kl - i)
                j2 = int(kl + i)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector: change in conv when moving delta from i to j
                g = np.zeros(size)
                # Remove contribution of a[i]
                for m in range(n):
                    idx1 = m + i
                    if 0 <= idx1 < size:
                        g[idx1] -= a[m]
                    idx2 = m - i
                    if 0 <= idx2 < size:
                        g[idx2] -= a[m]
                # Add contribution of a[j]
                for m in range(n):
                    idx1 = m + j
                    if 0 <= idx1 < size:
                        g[idx1] += a[m]
                    idx2 = m - j
                    if 0 <= idx2 < size:
                        g[idx2] += a[m]

                # Binary search for optimal delta
                lo = 0.0
                hi = 0.9 * ai
                for _ in range(12):
                    mid = 0.5 * (lo + hi)
                    new_conv = conv + mid * g
                    # Correction for the quadratic term at 2i and 2j
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * mid + mid * mid
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * mid + mid * mid
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = mid
                        best_j = j
                        lo = mid
                    else:
                        hi = mid

            if best_j >= 0 and best_delta > 1e-10 and best_val < cur_max - 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(10, n // 25)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

    return best_a, best_score

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.2
    elif choice == 5:
        a = 0.3 + 0.5 * np.sin(np.pi * x) ** 0.5 + 0.2 * np.sin(2 * np.pi * x) ** 2
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.6 * np.exp(-3.0 * (1 - x)) + 0.3 * np.sin(np.pi * x)
    else:
        a = 0.6 * np.sin(np.pi * x) + 0.3 * np.sin(2 * np.pi * x) ** 0.5
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def optimize_length(n, start_seq, time_budget, solver, seed):
    """Run the full optimization pipeline for a given length."""
    t_start = time.monotonic()
    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"    *** [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if start_seq is not None and len(start_seq) == n:
        a = start_seq.copy()
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = structured_init(n, np.random.default_rng(seed))
    else:
        # Try multiple initializations
        rng = np.random.default_rng(seed)
        best_init = None
        best_init_sc = float('inf')
        for _ in range(4):
            cand = structured_init(n, rng)
            sc = eval_fn(cand)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = cand
        a = best_init

    norm = np.sqrt(2.0 * n)
    a = a * norm / a.sum()
    try_update(a, "init")

    # Phase 1: FISTA pre-optimization
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = min(remaining * 0.15, 150.0)
    if t1 > 20:
        a_out, _ = fista_power_sum(a, t1, p=6.0, verbose=False)
        try_update(a_out, "fista")

    # Phase 2: LP best-response (multiplicative)
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = min(remaining * 0.40, 500.0)
    if t2 > 30:
        a_out, _ = lp_best_response(a, t2, solver, verbose=True, use_mult=True)
        try_update(a_out, "lp_mult")

    # Phase 3: KKT refinement
    remaining = time_budget - (time.monotonic() - t_start)
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 20:
        a_out, _ = kkt_refine(a, t3, solver, verbose=True)
        try_update(a_out, "kkt")

    # Phase 4: LP again (additive) from KKT result
    remaining = time_budget - (time.monotonic() - t_start)
    t4 = min(remaining * 0.20, 200.0)
    if t4 > 20:
        a_out, _ = lp_best_response(a, t4, solver, verbose=True, use_mult=False)
        try_update(a_out, "lp_add")

    # Phase 5: CD polish
    remaining = time_budget - (time.monotonic() - t_start)
    t5 = min(remaining * 0.25, 200.0)
    if t5 > 15:
        a_out, _ = pair_cd_optimize(a, t5, seed + 123, verbose=False)
        try_update(a_out, "cd")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    """Main entry point: search for the best sequence of non-negative heights."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # Strategy: optimize at n=1600 and n=2000, pick the best
    # Give more time to n=1600 (starting from parent) and less to n=2000

    n_primary = 1600
    if best_seq is not None and len(best_seq) != n_primary:
        n_primary = len(best_seq)

    # Phase A: Optimize primary length (60% of budget)
    remaining = time_budget - (time.monotonic() - t_global)
    t_primary = remaining * 0.55
    print(f"\n=== Phase A: Primary n={n_primary} (budget={t_primary:.0f}s) ===")
    a_prim, sc_prim = optimize_length(n_primary, best_seq, t_primary, solver, seed)
    if a_prim is not None and sc_prim < best_score:
        best_score = sc_prim
        best_seq = a_prim

    # Phase B: Try longer length (30% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 200:
        n_long = n_primary + 400
        t_long = remaining * 0.35
        print(f"\n=== Phase B: Long n={n_long} (budget={t_long:.0f}s) ===")
        init_long = interpolate_seq(best_seq, n_long) if best_seq is not None else None
        a_long, sc_long = optimize_length(n_long, init_long, t_long, solver, seed + 1000)
        if a_long is not None and sc_long < best_score:
            best_score = sc_long
            best_seq = a_long
            print(f"  Long wins! score={best_score:.6f}")

    # Phase C: Final polish on the best (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_best = len(best_seq)
        t_final = remaining * 0.85
        print(f"\n=== Phase C: Final polish n={n_best} (budget={t_final:.0f}s) ===")

        # Quick FISTA
        t_f1 = min(t_final * 0.2, 60.0)
        if t_f1 > 15:
            a_out, _ = fista_power_sum(best_seq, t_f1, p=10.0, verbose=False)
            sc = eval_fn(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  FISTA polish: {best_score:.6f}")

        # LP refinement
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.4, 150.0)
        if t_f2 > 20:
            a_out, _ = lp_best_response(best_seq, t_f2, solver, verbose=True, use_mult=True)
            sc = eval_fn(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  LP polish: {best_score:.6f}")

        # CD final
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.5, 100.0)
        if t_f3 > 15:
            a_out, _ = pair_cd_optimize(best_seq, t_f3, seed + 9999, verbose=False)
            sc = eval_fn(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  CD final: {best_score:.6f}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
