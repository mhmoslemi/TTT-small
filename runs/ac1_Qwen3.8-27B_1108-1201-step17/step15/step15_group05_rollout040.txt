```python
def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

def build_lp_matrix(a, k_indices, n):
    """Build constraint matrix T where T[k_idx, j] = a[k-j] if valid."""
    rows = len(k_indices)
    T = np.zeros((rows, n))
    for i, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[i, j] = a[k - j]
    return T


def solve_lp(T, M, n):
    """Solve: max sum(b) s.t. T @ b <= M, b >= 0."""
    if T.shape[0] == 0:
        return None
    c = -np.ones(n)
    res = linprog(c, A_ub=T, b_ub=np.full(T.shape[0], M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        b = np.maximum(res.x, 0.0)
        if b.sum() > 1e-15:
            return b
    return None

        # Solve LP
        T = build_lp_matrix(y_new, active_set, n)
        b_resp = solve_lp(T, M_curr, n)

        if b_resp is None:
            stall += 1
            if stall > 50:
                y = best_b.copy()
                y_prev = best_b.copy()
                active_set = None
                stall = 0
            t += 1
            continue

        # Verify no violations on all lags
        b_resp_sum = b_resp.sum()
        if b_resp_sum < 1e-15:
            t += 1
            continue
        g = b_resp / b_resp_sum

        # Check for constraint violations outside active set
        ac_check = fast_autocorr(g)
        violations = ac_check > M_curr * (1 + 1e-5)
        if violations.any():
            viol_idx = np.where(violations)[0]
            active_set = np.unique(np.concatenate([active_set, viol_idx]))
            if len(active_set) > max_constraints * 2:
                active_set = np.arange(size)
            T = build_lp_matrix(y_new, active_set, n)
            b_resp = solve_lp(T, M_curr, n)
            if b_resp is None:
                t += 1
                continue
            b_resp_sum = b_resp.sum()
            if b_resp_sum < 1e-15:
                t += 1
                continue
            g = b_resp / b_resp_sum

        y_prev = y
        y = g
        stall = 0

        # Track best
        val = fast_autocorr(y).max()
        if val < best_val - 1e-14:
            best_val = val
            best_b = y.copy()

        t += 1

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_val:.10f}, |A|={len(active_set)}")

    return best_b, best_val


def frank_wolfe_nesterov(b0, time_limit, verbose=False, seed=0):
    """Frank-Wolfe (conditional gradient) with Nesterov acceleration."""
    rng = np.random.default_rng(seed)
    n = len(b0)
    b = simplex_project(np.asarray(b0, dtype=np.float64).copy(), 1.0)
    best_b = b.copy()
    best_val = fast_autocorr(b).max()

    y_prev = b.copy()
    t = 1
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if t > 1000000:
            break

        # Nesterov extrapolation
        momentum = min((t - 1.0) / (t + 2.0), 0.98)
        y = b + momentum * (b - y_prev)
        y = simplex_project(y, 1.0)

        # Compute subgradient (pick random active peak for diversity)
        ac = fast_autocorr(y)
        M_curr = ac.max()
        # Use top-3 peaks with softmax weighting
        top_n = min(3, len(ac))
        top_idx = np.argsort(ac)[-top_n:]
        weights = np.exp(10 * (ac[top_idx] - M_curr))
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            # d/db[j] (b*b)[k] = 2*b[k-j] if valid
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += w * 2.0 * y[idx]

        # FW step: pick j* = argmin grad[j], move towards e_{j*}
        j_star = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        b_new = (1.0 - gamma) * b + gamma * np.zeros(n)
        b_new[j_star] = gamma
        b_new = simplex_project(b_new, 1.0)

        y_prev = b
        b = b_new

        val = fast_autocorr(b).max()
        if val < best_val - 1e-14:
            best_val = val
            best_b = b.copy()
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                b = best_b.copy()
                y_prev = best_b.copy()
                stall = 0

        t += 1

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter={t} ({elapsed:.0f}s): best={best_val:.10f}")

    return best_b, best_val


def coordinate_descent(b0, time_limit, verbose=False):
    """Coordinate descent with exact 1D optimization per coordinate."""
    n = len(b0)
    b = simplex_project(np.asarray(b0, dtype=np.float64).copy(), 1.0)
    best_b = b.copy()
    best_val = fast_autocorr(b).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            bj = b[j]
            s_others = b.sum() - bj

            # Try several candidate values for b[j]
            candidates = [0.0, bj * 0.25, bj * 0.5, bj * 0.75, bj * 1.25,
                          bj * 1.5, s_others / (n - 1), s_others * 0.5]
            candidates = [max(0.0, min(c, 2.0)) for c in candidates]

            best_obj = 2.0 * n * fast_autocorr(b).max() / (b.sum() ** 2)
            best_x = bj

            for x in candidates:
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                # Compute new autocorrelation efficiently
                # (b*b)[k] changes only where b[j] appears
                # Use the formula: new_conv[k] = old_conv[k] + (x-bj)*(b[k-j] + b[j-k]) + (x^2-bj^2)*delta(k,2j)
                old_conv = fast_autocorr(b)
                delta = np.zeros(size)
                for k in range(size):
                    v = 0.0
                    i1 = k - j
                    if 0 <= i1 < n and i1 != j:
                        v += b[i1]
                    i2 = j - k
                    if 0 <= i2 < n and i2 != j:
                        v += b[i2]
                    delta[k] = v
                new_conv = old_conv + (x - bj) * delta
                if 2 * j < size:
                    new_conv[2 * j] += x * x - bj * bj
                obj = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_x = x

            if abs(best_x - bj) > 1e-12:
                b[j] = best_x
                b = simplex_project(b, 1.0)
                new_val = 2.0 * n * fast_autocorr(b).max() / (b.sum() ** 2)
                if new_val < best_val - 1e-14:
                    best_val = new_val
                    best_b = b.copy()
                    improved = True
                else:
                    b[j] = bj
                    b = simplex_project(b, 1.0)

        if not improved:
            no_improve += 1
            if no_improve >= 2:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_val:.10f}")

    return best_b, best_val


def symmetrize(a):
    """Symmetrize: a = (a + flip(a))/2, renormalize."""
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity and sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global

    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
            parent_loaded = True
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
