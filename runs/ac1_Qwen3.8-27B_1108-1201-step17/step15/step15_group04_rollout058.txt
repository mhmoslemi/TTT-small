            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient via FFT: grad_i = 2n * sum_k w_k * (a_{k-i} + a_{i-k})
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # conv(w, a) gives (w*a)_k = sum_i w_i * a_{k-i}
            g_cross = irfft(W * A, fft_size)[:n]
            # We need sum_k w_k * a_{k-i} for each i
            # This is the cross-correlation of w and a
            # conv(w, a)[i] = sum_k w_k * a[i-k] ... not quite
            # Let's use: (w * a)_i = sum_k w_k a_{i-k}
            # We want: sum_k w_k * a_{k-i} = (w_rev * a)_i where w_rev[j] = w_{size-1-j}
            # Actually: sum_k w_k * a_{k-i} for k-i in [0,n-1]
            # = sum_j w_{j+i} * a_j (substituting j=k-i)
            # This is cross-correlation of w and a at lag i
            # cross_corr[i] = sum_j w[j+i] * a[j]
            # = (w * flip(a))[i] where we pad appropriately
            # Simpler: just use the fact that sum_k w_k * a_{k-i} + a_{i-k}
            # = 2 * sum_k w_k * a_{k-i} if w is symmetric
            # In general: = (w * a)_reversed + (a * w)
            # Let's just compute both convolutions
            g1 = irfft(np.conj(A) * W, fft_size)[:n]  # a * w (cross-correlation)
            g2 = irfft(A * W, fft_size)[:n]  # a conv w

            grad = 2.0 * (g1 + g2)

            # Renormalize: we want to keep sum(a) = norm
            # Gradient of the normalized objective: subtract mean to maintain normalization
            grad -= grad.sum() / n

            adam_step += 1
            m_adam = beta1 * m_adam + (1 - beta1) * grad
            v_adam = beta2 * v_adam + (1 - beta2) * grad ** 2
            m_hat = m_adam / (1 - beta1 ** adam_step)
            v_hat = v_adam / (1 - beta2 ** adam_step)

            lr = 0.1 / (1.0 + 0.01 * si)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            sc = fast_conv_self(a_new).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Newton stage {si} T={T:.2f} ({elapsed:.0f}s): best_max={best_score:.8f}")

    # Final: normalize best_a and return
    s_best = best_a.sum()
    if s_best > 1e-15:
        best_a = best_a / s_best
    return best_a, best_score

def fast_lp_best_response(a, M, n_top=100, solver_name='CLARABEL'):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0.
    Uses only top n_top constraints for speed."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a

    conv_a = fast_conv_self(a)
    # Select top constraints
    if n_top < size:
        top_idx = np.argsort(conv_a)[-n_top:]
        T_sub = T[top_idx]
    else:
        T_sub = T
        top_idx = np.arange(size)

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver_name, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)

    # Get dual variables
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if len(top_idx) < size:
                full_duals = np.zeros(size)
                full_duals[top_idx] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass

    return val, duals

def lp_nesterov_fast(a0, time_limit, n_top=150, verbose=False):
    """LP best-response with Nesterov acceleration, using reduced constraints."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    no_improve = 0
    solver = None

    # Find a working solver
    import cvxpy as cp
    for s_name in ['CLARABEL', 'PDLP', 'ECOS', 'SCS']:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s_name)
            if p.status in ('optimal', 'optimal_inaccurate'):
                solver = s_name
                break
        except Exception:
            continue
    if solver is None:
        solver = 'ECOS'

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        elapsed = time.monotonic() - t0

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-14:
            break

        b, duals = fast_lp_best_response(y, M, n_top=n_top, solver_name=solver)
        if b is None or b.sum() < 1e-15:
            # Fallback: use gradient direction
            grad = -conv_y  # subgradient of max
            b = np.ones(n) * 0.1 + np.maximum(-grad[:n], 0) * 0.01
            # Verify
            check = fast_conv_cross(y, b)
            if check.max() > M * 1.5:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t - 1.0) / (t + 2.0))
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        # Periodic perturbation
        if no_improve == 150:
            rng_p = np.random.default_rng(t)
            n_pert = max(5, n // 25)
            idx_p = rng_p.choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.3 * rng_p.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0

        if verbose and t % 100 == 0:
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    s_best = best_a.sum()
    if s_best > 1e-15:
        best_a = best_a / s_best
    return best_a, best_score

def peak_cd(a0, time_limit, rng, verbose=False):
    """Coordinate descent focused on reducing peak convolution lags."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Identify peak lags (where conv is close to max)
        threshold = cur_max * 0.95
        peak_lags = np.where(conv >= threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(conv)])

        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Find candidate j's that would reduce peak lags
            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            # Candidates from peak lags
            j_candidates = set()
            for kl in peak_lags[:15]:
                j1 = kl - i
                j2 = i - kl
                if 0 <= j1 < n and j1 != i:
                    j_candidates.add(j1)
                if 0 <= j2 < n and j2 != i:
                    j_candidates.add(j2)
                # Also try j such that moving mass from i to j reduces conv at peak
                # conv[kl] changes by: -a_i*a_{kl-i} + a_{kl-j}*a_{kl-i} ... complex
                # Just try nearby indices
                j_candidates.add(min(n-1, max(0, j1)))
                j_candidates.add(min(n-1, max(0, j2)))

            # Add random candidates
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))

            j_candidates.discard(i)

            for j in list(j_candidates)[:10]:
                if j < 0 or j >= n or j == i:
                    continue
                aj = a[j]

                # Compute change in convolution when moving delta from i to j
                # conv_new = conv + delta * (contribution from j - contribution from i)
                # For the peak lags, we want this to be negative

                # Quick check: compute the gain vector
                g = np.zeros(size)
                # Contribution of a_j: for each lag k, a_j contributes a_j * a_{k-j} and a_{k-j} * a_j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[:hi-lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Subtract contribution of a_i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[:hi3-lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Diagonal corrections
                # conv[2*i] loses a_i^2, gains 0 (mass moved away)
                # conv[2*j] gains: was a_j^2, becomes (a_j+delta)^2
                # Net at 2*i: -2*a_i*delta + delta^2
                # Net at 2*j: +2*a_j*delta + delta^2

                # Try several delta values
                n_samples = 15
                deltas = np.linspace(0.02, 0.9, n_samples) * ai
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*d + d*d
                    if 2 * j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*d + d*d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve >= 5:
                n_pert = max(10, n // 20)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and no_improve % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD ({elapsed:.0f}s): best={best_score:.8f}")

    s_best = best_a.sum()
    if s_best > 1e-15:
        best_a = best_a / s_best
    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        # Smooth bump with oscillation
        a = np.exp(-3.0 * (x - 0.5)**2) * (1.0 + 0.2 * np.cos(30 * np.pi * x))
        a = np.maximum(a, 0.01)
    else:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)  # Hann window
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization combining LSE-Adam, LP Nesterov, and peak CD."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick LSE-Adam warm start
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 1: LSE-Adam init (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_grad_newton(best_seq, t1, T_init=50.0, T_min=2.0, verbose=True)
        try_update(a_out, "lse_adam_init")

    # Phase 2: LP Nesterov (main workhorse) with reduced constraints
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 2: LP Nesterov fast (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_fast(best_seq, t2, n_top=120, verbose=True)
        try_update(a_out, "lp_nest_1")

    # Phase 3: Peak-focused CD
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 3: Peak CD (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_cd(best_seq, t3, rng, verbose=True)
        try_update(a_out, "peak_cd_1")

    # Phase 4: LP Nesterov round 2 (from improved point, more constraints)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov_fast(best_seq, t4, n_top=200, verbose=True)
        try_update(a_out, "lp_nest_2")

    # Phase 5: LSE-Adam polish with lower temperature
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: LSE-Adam polish (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = lse_grad_newton(best_seq, t5, T_init=10.0, T_min=0.5, verbose=True)
        try_update(a_out, "lse_polish")

    # Phase 6: Multi-restart - try perturbed versions
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-restart (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        n_restarts = min(3, max(1, int(remaining // 80)))
        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            perturbed = best_seq.copy()
            n_pert = max(20, len(best_seq) // 10)
            idx_p = rng.choice(len(perturbed), size=n_pert, replace=False)
            perturbed[idx_p] *= (1.0 + 0.5 * rng.standard_normal(n_pert))
            perturbed = np.maximum(perturbed, 0.0)
            s_p = perturbed.sum()
            if s_p > 1e-15:
                perturbed = perturbed / s_p

            t_r = min(remaining * 0.4, 80.0)
            a_out, _ = lse_grad_newton(perturbed, t_r, T_init=30.0, T_min=2.0, verbose=False)
            try_update(a_out, f"restart_{r}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t_r2 = min(remaining * 0.4, 60.0)
                a_out, _ = lp_nesterov_fast(best_seq, t_r2, n_top=100, verbose=False)
                try_update(a_out, f"lp_restart_{r}")

    # Phase 7: Final peak CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final peak CD (remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t7 = min(remaining * 0.7, 100.0)
        a_out, _ = peak_cd(best_seq, t7, rng, verbose=True)
        try_update(a_out, "peak_cd_final")

    # Phase 8: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.8, 60.0)
        a_out, _ = lse_grad_newton(best_seq, t8, T_init=5.0, T_min=0.3, verbose=False)
        try_update(a_out, "lse_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
