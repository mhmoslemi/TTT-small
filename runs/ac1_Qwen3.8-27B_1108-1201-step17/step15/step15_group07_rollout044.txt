Given a (length n), conv(a,a) (length 2n-1), and active set S (subset of {0, ..., 2n-2}):

This can be computed as:

= (2/|S|) * sum_{i in S-j} a[i] * 1[0 <= i < n]

where S-j = {k-j : k in S, 0 <= k-j < n}.

If S is a contiguous range [k_min, k_max], then S-j = [k_min-j, k_max-j], and we can compute the sum using a prefix sum of a:
sum_{i=max(0,k_min-j)}^{min(n-1,k_max-j)} a[i] = prefix[k_max-j+1] - prefix[k_min-j] (with appropriate clamping).

This gives O(n) computation for the subgradient when S is contiguous!

In practice, the active set might not be perfectly contiguous, but it's often "nearly" contiguous (the lags near the peak that are within epsilon of the max). We can approximate it as contiguous for efficiency.

def fast_autocorr(a):

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def subgradient_contiguous(a, k_min, k_max):
    """
    Compute subgradient using contiguous active set [k_min, k_max].
    O(n) using prefix sums.
    """
    n = len(a)
    size = 2 * n - 1
    k_min = max(0, k_min)
    k_max = min(size - 1, k_max)
    
    # prefix sum of a
    prefix = np.zeros(n + 1)
    prefix[1:] = np.cumsum(a)
    
    subgrad = np.zeros(n)
    for j in range(n):
        # sum of a[k-j] for k in [k_min, k_max] where 0 <= k-j < n
        i_lo = max(0, k_min - j)
        i_hi = min(n - 1, k_max - j)
        if i_hi >= i_lo:
            subgrad[j] = prefix[i_hi + 1] - prefix[i_lo]
    
    return 2.0 * subgrad


def subgradient_active(a, active_idx):
    """Compute subgradient for a given active set."""
    n = len(a)
    size = 2 * n - 1
    subgrad = np.zeros(n)
    for k in active_idx:
        k = int(k)
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            subgrad[j] += 2.0 * a[k - j]
    return subgrad / max(1, len(active_idx))


def lp_best_response_cuting_plane(a, M, solver, max_constraints=200, verbose=False):

    """

    size = 2 * n - 1

        T = np.zeros((len(active_lags), n))
        for row, k in enumerate(active_lags):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for j in range(lo, hi + 1):
                T[row, j] = a[k - j]
        return T

    initial_active = sorted(initial_active.tolist())

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_cutting_plane=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    patience = 0
    restart_count = 0
    momentum_t = 1.0
    adaptive_eps = 0.01  # relative tolerance for active set

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if use_cutting_plane:
            b = lp_best_response_cuting_plane(y, M_curr, solver)
        else:
            b = lp_best_response_full(y, M_curr, solver)
        
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        if patience > 100:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score
