def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_exact_subgradient(a, ac, top_k=1):
    """
    Compute subgradient of max(ac) w.r.t. a.
    If top_k peaks are close, use weighted average of their gradients.
    """
    n = len(a)
    # Find the top peaks
    sorted_idx = np.argsort(ac)[::-1]
    
    # Determine how many peaks to include based on proximity
    max_val = ac[sorted_idx[0]]
    if max_val < 1e-30:
        return np.zeros(n)
    
    # Include peaks that are within 1% of the maximum
    threshold = max_val * 0.99
    active_peaks = []
    for idx in sorted_idx:
        if ac[idx] >= threshold:
            active_peaks.append(idx)
        else:
            break
    
    # Use softmax weighting over the active peaks
    peak_vals = np.array([ac[k] for k in active_peaks])
    # Temperature-based weighting
    temp = max_val * 0.01  # small temperature = focus on the max
    w = np.exp((peak_vals - max_val) / temp)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active_peaks)) / len(active_peaks)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, active_peaks):
        k_int = int(k)
        # db_k/da_j = 2 * a_{k-j} if 0 <= k-j < n
        # So grad_j = sum over active k: w_k * 2 * a_{k-j} (if valid)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def amd_exact(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated mirror descent with exact subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 3.0  # step size parameter
    stall = 0
    max_stall = 3000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        ac_y = fast_autocorr(y)
        g = compute_exact_subgradient(y, ac_y)
        
        # Mirror descent update (exponential family = multiplicative weights)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Evaluate
        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            # Perturb the best solution
            rng = np.random.default_rng(t)
            perturb = rng.normal(0, 0.005, n) * best_a
            x_restarted = best_a + perturb
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart at t={t} ({elapsed:.1f}s): J={J:.6f}")

def coord_descent_fast(a0, time_limit, verbose=False, rng_seed=42):
    """
    Fast coordinate descent using analytic sensitivity.
    Moving δ from i to j changes b_k by 2δ*(a_{k-j} - a_{k-i}).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Find the peak
        peak_k = int(np.argmax(ac))
        
        # Find the position that most contributes to the peak
        # Contribution of a_i to b_{peak_k} is proportional to a_{peak_k - i}
        # (since db_k/da_i = 2*a_{k-i})
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        contrib = a[i_start:i_end] * a[peak_k - np.arange(i_start, i_end)]
        
        # Find the source (highest contribution)
        src_offset = int(np.argmax(contrib))
        src = i_start + src_offset
        
        if a[src] < 1e-12:
            stall += 1
            continue
        
        # Find the best destination: minimize a_{peak_k - j}
        # We want a destination j where a_{peak_k - j} is small
        # (this reduces b_{peak_k})
        
        # Compute the "sensitivity" for each position j
        # Moving to j increases b_{peak_k} by 2*delta*a_{peak_k - j}
        # So we want j where a_{peak_k - j} is minimized
        
        sens = np.zeros(n)
        for j in range(n):
            val = 0.0
            # a_{peak_k - j} if valid
            if 0 <= peak_k - j < n:
                val += a[peak_k - j]
            # Also consider other peaks (top 5)
            top_indices = np.argsort(ac)[::-1][:5]
            for k2 in top_indices[1:]:
                k2i = int(k2)
                if 0 <= k2i - j < n:
                    val += 0.5 * a[k2i - j]
            sens[j] = val
        
        # Exclude positions near the peak (they're already important)
        peak_region = range(max(0, peak_k - 2), min(n, peak_k + 3))
        for j in peak_region:
            sens[j] = np.inf
        
        dst = int(np.argmin(sens))
        
        if sens[dst] == np.inf or dst == src:
            stall += 1
            # Try random destination
            dst = int(rng.integers(n))
            if dst == src:
                dst = (dst + 1) % n
        
        # Try different step sizes
        found = False
        for frac in [0.01, 0.005, 0.002, 0.01, 0.02, 0.05, 0.1, 0.2]:
            delta = a[src] * frac
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= delta
            a_trial[dst] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            
            ac_trial = fast_autocorr(a_trial)
            m_trial = ac_trial.max()
            
            if m_trial < best_m - 1e-15:
                a = a_trial
                ac = ac_trial
                best_m = m_trial
                best_a = a.copy()
                found = True
                stall = 0
                break
        
        if not found:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    ac_new = fast_autocorr(a)
                    m_new = ac_new.max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        ac = ac_new
                        stall = 0
                    else:
                        # Revert
                        a[i] += delta
                        a[j] -= delta
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient(a, ac, mode="focused"):
    """
    Compute subgradient of max(ac) w.r.t. a.
    mode='focused': only top 1-3 peaks (sharp gradient)
    mode='adaptive': adapt based on peak separation
    mode='broad': top 20 peaks with soft weighting
    """
    n = len(a)
    sorted_idx = np.argsort(ac)[::-1]
    max_val = ac[sorted_idx[0]]
    if max_val < 1e-30:
        return np.zeros(n)
    
    if mode == "focused":
        # Use only the top 1-3 peaks that are within 2% of max
        active = []
        for idx in sorted_idx[:5]:
            if ac[idx] >= max_val * 0.98:
                active.append(idx)
            else:
                break
        if not active:
            active = [sorted_idx[0]]
        peak_vals = np.array([ac[k] for k in active])
        w = peak_vals / peak_vals.sum()
    elif mode == "broad":
        active = list(sorted_idx[:20])
        peak_vals = np.maximum(np.array([ac[k] for k in active]), 1e-30)
        w = peak_vals ** 4
        w = w / w.sum()
    else:  # adaptive
        top2_ratio = ac[sorted_idx[1]] / max_val if len(sorted_idx) > 1 else 0
        if top2_ratio > 0.95:

            peak_vals = np.maximum(np.array([ac[k] for k in active]), 1e-30)
            w = peak_vals ** 3
            w = w / w.sum()
        elif top2_ratio > 0.8:
            active = list(sorted_idx[:4])

            w = peak_vals / peak_vals.sum()
        else:

            active = [sorted_idx[0]]

    for wi, k in zip(w, active):

        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad
