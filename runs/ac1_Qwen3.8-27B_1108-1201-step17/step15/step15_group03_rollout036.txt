"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE gradient descent for fast warmup
2. Full-constraint LP best response with Nesterov acceleration
3. Adaptive peak-shaving local search
4. Multi-length exploration with interpolation

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute the objective value (before the 2n/sum^2 normalization factor)."""
    n = len(a)
    conv = fast_conv_self(a)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def normalize_to_sqrt_2n(a):
    n = len(a)
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def lse_smooth_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent on the max-conv objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Multi-stage temperature annealing
    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(400, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            # Adaptive step size
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def build_convolution_matrix(a, active=None):
    """Build the (2n-1 x n) matrix T where T[k,i] = a[k-i] if valid."""
    n = len(a)
    size = 2 * n - 1
    if active is not None and 0 < len(active) < size:
        k_indices = active
    else:
        k_indices = np.arange(size)
    
    T = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        # T[k, i] = a[k-i] if 0 <= k-i < n
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:
            T[idx, i_min:i_max+1] = a[i_max:i_min-1:-1] if i_min > 0 else a[0:i_max+1][::-1]
    
    return T


def lp_best_response(a, M, solver, use_full=True):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    if use_full:
        active = None
    else:
        conv_check = fast_conv_self(a)
        threshold = M * 0.85
        active = np.where(conv_check >= threshold)[0]
        if len(active) < 20:
            threshold = M * 0.5
            active = np.where(conv_check >= threshold)[0]
    
    T = build_convolution_matrix(a, active)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def nesterov_lp(a0, time_limit, solver, verbose=True, use_full_lp=True):
    """Nesterov-accelerated LP best response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Track convergence
    prev_best = best_score + 1e-6
    no_improve_count = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Solve LP
        if use_full_lp:
            b = lp_best_response(y, M, solver, use_full=True)
        else:
            b = lp_best_response(y, M, solver, use_full=False)
            if b is None:
                b = lp_best_response(y, M, solver, use_full=True)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Track convergence
        if best_score < prev_best - 1e-10:
            no_improve_count = 0
            prev_best = best_score
        else:
            no_improve_count += 1

        # Restart mechanism for Nesterov
        if no_improve_count > 200:
            y = a.copy()
            no_improve_count = 0
            prev_best = best_score + 1e-6

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def momentum_subgradient(a0, time_limit, verbose=True):
    """Momentum-based subgradient descent with smooth max approximation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    v = np.zeros(n)  # momentum
    beta = 0.85
    lr0 = 2.0 / np.sqrt(n)

    for it in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()

        # Smooth max with decreasing temperature
        temp = max(0.05, max_c * 0.1 / (1.0 + 0.005 * it))
        shifted = (conv - max_c) / temp
        shifted = np.maximum(shifted, -60.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        lr = lr0 / (1.0 + 0.002 * it)
        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a different length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    for idx, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:
            T[idx, i_min:i_max + 1] = a[i_max:i_min - 1:-1]
    return T


def lp_best_response_full(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_convolution_matrix(a, None)
    b_var = cp.Variable(n, nonneg=True)
    d_var = cp.Variable(size, nonneg=True)
    constraints = [T @ b_var <= M, T.T @ d_var >= 1.0, d_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    d_val = np.asarray(d_var.value, dtype=np.float64) if d_var.value is not None else None
    return b_val, d_val


def lp_best_response_screened(a, M, solver, threshold_frac=0.85):
    n = len(a)
    size = 2 * n - 1
    conv_check = fast_conv_self(a)
    active = np.where(conv_check >= M * threshold_frac)[0]
    if len(active) < 15:
        active = np.where(conv_check >= M * 0.5)[0]
    T = build_convolution_matrix(a, active)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)


def lse_smooth_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(120.0, 0.08, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(500, int(remaining * 10)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.02 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_full=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    no_improve = 0
    prev_best = best_score + 1e-6

    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        if use_full:
            b, dual = lp_best_response_full(y, M, solver)
            if b is None:
                b = lp_best_response_screened(y, M, solver, 0.9)
        else:
            b = lp_best_response_screened(y, M, solver, 0.85)
            if b is None:
                b, dual = lp_best_response_full(y, M, solver)
                if b is None:
                    continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov with adaptive restart
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if best_score < prev_best - 1e-10:
            no_improve = 0
            prev_best = best_score
        else:
            no_improve += 1

        # Periodic restart
        if no_improve > 150:
            y = a.copy()
            no_improve = 0
            prev_best = best_score + 1e-6

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def momentum_subgradient(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    v = np.zeros(n)
    beta = 0.85
    lr0 = 3.0 / np.sqrt(n)

    for it in range(1, 80000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        temp = max(0.05, max_c * 0.08 / (1.0 + 0.003 * it))
        shifted = (conv - max_c) / temp
        shifted = np.maximum(shifted, -60.0)
        w = np.exp(shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        lr = lr0 / (1.0 + 0.001 * it)
        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        peak_lags = np.where(conv >= cur_max * 0.88)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            total_iters += 1
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            j_candidates = set()
            for kl in peak_lags[:min(10, len(peak_lags))]:
                j_candidates.add((kl - i + n) % n)
                j_candidates.add((kl + i) % n)
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]

                first_order = np.zeros(size)
                for k in range(max(0, j - n + 1), min(size, j + 1)):
                    if 0 <= k - j < n:
                        first_order[k] += 2.0 * a[k - j]
                for k in range(max(0, i - n + 1), min(size, i + 1)):
                    if 0 <= k - i < n:
                        first_order[k] -= 2.0 * a[k - i]

                n_samples = 20
                for d in np.linspace(0.02 * ai, 0.92 * ai, n_samples):
                    new_conv = conv + d * first_order
                    if 2 * i < size:
                        new_conv[2 * i] += (-2.0 * ai * d + d * d)
                    if 2 * j < size:
                        new_conv[2 * j] += (2.0 * aj * d + d * d)
                    v = new_conv.max()
                    if v < best_val - 1e-13:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                else:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                n_perturb = max(8, n // 35)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.18 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                no_improve = 0
        else:
            no_improve = 0

        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave iters={total_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(3.0 * (x - 0.5)) ** 2 + 0.1
    elif choice == 9:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / (0.15 + 0.1 * rng.random()))
    elif choice == 10:
        a = 0.4 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x + 0.5)
    else:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * rng.uniform(2, 6) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    a = a + 0.02 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_smooth_descent(best_seq.copy(), t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (full)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.28, 450.0)
    print(f"\n=== Phase 2: LP Nesterov full n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True, use_full=True)
        try_update(a_out, "lp_full1")

    # Phase 3: Subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.07, 90.0)
    print(f"\n=== Phase 3: SubGrad n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_out, _ = momentum_subgradient(best_seq, t3, verbose=True)
        try_update(a_out, "subgrad1")

    # Phase 4: LP Nesterov (screened, faster iterations)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 4: LP Nesterov screened (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True, use_full=False)
        try_update(a_out, "lp_screened")

    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Peak Shaving (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = peak_shaving(best_seq, t5, rng, verbose=True)
        try_update(a_out, "peak_shave1")

    # Phase 6: Try different length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        n_alt = n_base + 200
        t6 = min(remaining * 0.22, 200.0)
        print(f"\n=== Phase 6: Length {n_alt} (budget={t6:.0f}s) ===")
        if t6 > 40:
            init_alt = interpolate_seq(best_seq, n_alt)
            t6a = min(t6 * 0.3, 50.0)
            a_out, _ = lse_smooth_descent(init_alt, t6a, verbose=False)
            try_update(a_out, f"lse_n{n_alt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t6b = min(remaining * 0.5, 80.0)
                a_out2, _ = nesterov_lp_optimize(init_alt, t6b, solver, verbose=False, use_full=False)
                try_update(a_out2, f"lp_n{n_alt}")

    # Phase 7: Final polish cycle
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        t7a = min(remaining * 0.25, 40.0)
        a_out, _ = lse_smooth_descent(best_seq, t7a, verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.4, 80.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t7b, solver, verbose=True, use_full=True)
            try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t7c = min(remaining * 0.7, 30.0)
            a_out, _ = peak_shaving(best_seq, t7c, rng, verbose=False)
            try_update(a_out, "shave_final")

    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
