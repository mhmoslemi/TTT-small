def nesterov_subgradient(a0, time_limit, solver, verbose=True, seed=0):

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_conv(a, a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    stall_count = 0
    stall_patience = 150

    tol_schedule_start = 0.15
    tol_schedule_end = 0.01

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive tolerance: start loose, get tighter
        frac = min(t / 5000.0, 1.0)
        tol_frac = tol_schedule_start * (1.0 - frac) + tol_schedule_end * frac

        # Compute convolution at Nesterov point
        conv_y = fast_conv(y, y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Subgradient at y
        g, _ = compute_subgradient_direct(conv_y, y, tol_frac)
        g_max = g.max()
        if g_max < 1e-10:
            break

        # Polyak step size
        conv_a = fast_conv(a, a)
        f_curr = conv_a.max()
        lr = max(0.01, (f_curr - best_score) / (g_max * g_max + 1e-10))
        lr = min(lr, 1.0 / g_max)

        # Gradient step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        # Nesterov extrapolation
        beta = (t - 1) / (t + 1.5 + 0.1 * t)
        beta = min(beta, 0.95)
        y_new = a_new + beta * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-10:
            y_new = a_new.copy()
        else:
            y_new = y_new * norm / s_y

        prev_a = a
        a = a_new
        y = y_new

        # Ergodic average
        avg_a += a
        n_avg += 1

        # Track best
        conv_check = fast_conv(a, a)
        score_check = conv_check.max()
        if score_check < best_score:
            best_score = score_check
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Check ergodic average periodically
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            avg_check = project_onto_simplex(avg_check, norm)
            sc_avg = fast_conv(avg_check, avg_check).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()
                stall_count = 0

        # Restart if stalled
        if stall_count > stall_patience:
            a = best_a.copy()
            y = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall_count = 0
            # Small perturbation
            if t > 100:

                a = np.maximum(a + pert * norm / n, 0.0)
                a = a * norm / a.sum()
                y = a.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG t={t} ({elapsed:.0f}s): best={best_score:.10f}, tol={tol_frac:.3f}")

    # Final: also check the ergodic average
    if n_avg > 0:
        avg_final = avg_a / n_avg
        avg_final = project_onto_simplex(avg_final, norm)
        sc = fast_conv(avg_final, avg_final).max()
        if sc < best_score:
            best_score = sc
            best_a = avg_final.copy()

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_conv(a, a)
    best_score = best_conv.max()

    y = a.copy()
    t0 = time.monotonic()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv(y, y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Active set: top constraints
        tol = 0.05
        threshold = M_curr * (1.0 - tol)
        active_set = np.where(conv_y >= threshold)[0]
        if len(active_set) < 10:
            active_set = np.where(conv_y >= M_curr * 0.7)[0]
        if len(active_set) > 0.95 * size:
            active_set = np.where(conv_y >= M_curr * 0.85)[0]

        # LP best response
        b_lp = lp_best_response(y, M_curr, solver, active_set)
        if b_lp is None:
            active_set = np.where(conv_y >= M_curr * 0.8)[0]
            b_lp = lp_best_response(y, M_curr, solver, active_set)
            if b_lp is None:
                continue

        # Normalize LP solution
        b_sum = b_lp.sum()
        g = b_lp * norm / b_sum

        # Verify no major constraint violations
        if len(active_set) < size:
            conv_g = fast_conv(g, y)
            if conv_g.max() > M_curr * 1.05:
                active_set = np.where(conv_y >= M_curr * 0.6)[0]
                b_lp = lp_best_response(y, M_curr, solver, active_set)
                if b_lp is not None:
                    g = b_lp * norm / b_lp.sum()

        # Blend with Nesterov
        alpha = max(0.3, 1.0 - 0.5 * (t / max_iters))
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new * norm / y_new.sum()

        a_new = y_new.copy()
        a = a_new
        y = y_new

        # Track best
        conv_a = fast_conv(a, a)
        score = conv_a.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coord_descent_fast(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a, a).max()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-12:
                continue

            # Try a few candidate values
            current_score = fast_conv(a, a).max()
            best_val = current_score
            best_new_val = ai

            for frac in [0.5, 0.75, 1.25, 1.5, 2.0, 0.1, 0.25]:
                new_val = ai * frac
                a_test = a.copy()
                a_test[i] = new_val
                s_test = a_test.sum()
                if s_test < 1e-10:
                    continue
                a_test = a_test * norm / s_test
                val = fast_conv(a_test, a_test).max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_new_val = new_val

            if best_new_val != ai:
                a[i] = best_new_val
                s = a.sum()
                if s > 1e-10:
                    a = a * norm / s
                    sc = fast_conv(a, a).max()
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_cd(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a, a).max()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    rng = np.random.default_rng(456)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            j = perm[idx + 1]
            ai, aj = a[i], a[j]
            if ai < 1e-12 and aj < 1e-12:
                continue

            best_val = fast_conv(a, a).max()
            best_d = 0.0

            for frac in [0.2, 0.4, 0.6, 0.8]:
                d = ai * frac
                a_try = a.copy()
                a_try[i] = ai - d
                a_try[j] = aj + d
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try * norm / s_try
                val = fast_conv(a_try, a_try).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_d = d

                # Reverse
                d2 = aj * frac
                a_try2 = a.copy()
                a_try2[i] = ai + d2
                a_try2[j] = aj - d2
                s_try2 = a_try2.sum()
                if s_try2 < 1e-10:
                    continue
                a_try2 = a_try2 * norm / s_try2
                val2 = fast_conv(a_try2, a_try2).max()
                if val2 < best_val - 1e-13:
                    best_val = val2
                    best_d = -d2

            if abs(best_d) > 1e-14:
                a[i] += best_d
                a[j] -= best_d
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a * norm / s
                    sc = fast_conv(a, a).max()
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def structured_init(n, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 
None:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
