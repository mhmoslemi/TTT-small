                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                b_sum = b_val.sum()
                if b_sum > 1e-15:
                    g = b_val / b_sum
                else:
                    continue
        
        alpha = 2.0 / (t + 2.0)
        
        if t == 1:
            a_cur = g.copy()
            y = g.copy()
        else:
            beta = (t - 2.0) / (t + 1.0)
            a_new = y + beta * (y - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = g.copy()
            else:
                a_new = a_new / s_new
            a_cur = a_new
            y_prev = y.copy()
            y = a_cur + alpha * (g - a_cur)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a_cur.copy()
            else:
                y = y / s_y
        
        sc = compute_score_fast(a_cur)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_cur.copy()
            if best_score < last_best - 1e-6:
                last_best = best_score
                stall_count = 0
        
        if t % 50 == 0:
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP-Asym t={t} ({elapsed:.0f}s): score={best_score:.8f}")
            
            if last_best - best_score < 1e-9:
                stall_count += 1
            else:
                stall_count = 0
            
            if stall_count > 5:
                y = best_a / best_a.sum()
                y_prev = y.copy()
                a_cur = y.copy()
                stall_count = 0
                t = max(1, t // 2)
    
    return best_a, best_score


def lse_adam_optimize(a0, time_limit, verbose=True):
    """LSE (Log-Sum-Exp) smoothed gradient descent with Adam."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    n_stages = 25
    temps = np.geomspace(50.0, 0.01, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(150, int(remaining * 8)))
        lr = 0.3 / np.sqrt(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / (exp_c.sum() + 1e-30)
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            
            grad = 2.0 * (g1 + g2)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")
    
    return best_a, best_score


def fourier_flat_optimize(a0, time_limit, verbose=True):
    """Try to flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score_fast(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Work in log-magnitude domain
    lr = 0.01
    momentum = 0.9
    vel = np.zeros(n)
    
    n_iterations = 5000
    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        mag = np.abs(A)
        mag_sq = mag ** 2
        
        # Gradient of max(ifft(mag_sq)) w.r.t. a
        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-15:
            break
        
        # LSE smoothed max
        T = max(0.01, M * 0.1)
        shifted = (conv - M) / T
        shifted = np.maximum(shifted, -80.0)
        w = np.exp(shifted)
        w = w / (w.sum() + 1e-30)
        
        W = rfft(w, fft_size)
        A_fft = rfft(a, fft_size)
        
        grad = 2.0 * (irfft(np.conj(A_fft) * W, fft_size)[:n] + irfft(A_fft * W, fft_size)[:n])
        
        vel = momentum * vel + (1 - momentum) * grad
        a = a - lr * vel
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
        
        sc = compute_score_fast(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Adaptive learning rate
        if it % 100 == 0 and it > 0:
            lr *= 0.95
    
    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with gradient-informed moves."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find peak lags
            peak_lags = np.where(conv >= cur_max * 0.8)[0]
            
            # Candidate j values
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                for offset in [-1, 0, 1]:
                    j = int((kl - i + offset + n) % n)
                    j_candidates.add(j)
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gradient of conv w.r.t. moving mass from i to j
                g = np.zeros(size)
                
                # Effect of removing from i
                lo = max(0, i)
                hi = min(size, i + n)
                if hi > lo:
                    g[lo:hi] -= a[:hi-lo]
                lo2 = max(0, i - n + 1)
                hi2 = min(size, i + 1)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[i - np.array(range(lo2, hi2))]
                
                # Effect of adding to j
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[:hi3-lo3]
                lo4 = max(0, j - n + 1)
                hi4 = min(size, j + 1)
                if hi4 > lo4:
                    g[lo4:hi4] += a[j - np.array(range(lo4, hi4))]
                
                d_max = ai * 0.7
                n_samples = 20
                deltas = np.linspace(0.02, d_max, n_samples)
                
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2*i] = conv[2*i] - 2*ai*d + d*d
                    if 2 * j < size:
                        new_conv[2*j] = conv[2*j] + 2*aj*d + d*d
                    if 2*i == 2*j:
                        new_conv[2*i] = conv[2*i] + d*d - d*d  # cancel
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < cur_max - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                n_perturb = max(5, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 12:
                    break
        else:
            no_improve = 0
        
        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-20.0 * (x - 0.3)**2) + np.exp(-20.0 * (x - 0.7)**2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines symmetric LP game, LSE-Adam, Fourier flattening,
    and coordinate descent to minimize the evaluation score.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1601  # odd for symmetry
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
        # Ensure odd length for symmetric optimization
        if n_base % 2 == 0:
            best_seq = np.append(best_seq, best_seq[0])
            n_base = len(best_seq)
            best_score = eval_fn(best_seq)
            print(f"Adjusted to odd n={n_base}, score={best_score:.6f}")
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old_best:.6f}, n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 1: Quick LSE-Adam to get a decent starting point
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{n_base}")
    
    # Phase 2: Symmetric LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 2: Symmetric LP Nesterov n={n_base} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_symmetric(best_seq, t2, solver, verbose=True)
        check_and_update(a_out, f"sylp_n{n_base}")
    
    # Phase 3: LSE-Adam round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 3: LSE-Adam R2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t3, verbose=False)
        check_and_update(a_out, f"lse2_n{len(best_seq)}")
    
    # Phase 4: Symmetric LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: Symmetric LP Nesterov R2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_symmetric(best_seq, t4, solver, verbose=True)
        check_and_update(a_out, f"sylp2_n{len(best_seq)}")
    
    # Phase 5: Asymmetric LP (try to break symmetry)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: Asymmetric LP n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = nesterov_lp_asymmetric(best_seq, t5, solver, verbose=True)
        check_and_update(a_out, f"asylp_n{len(best_seq)}")
    
    # Phase 6: Fourier flattening
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 6: Fourier flatten n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = fourier_flat_optimize(best_seq, t6, verbose=False)
        check_and_update(a_out, f"fourier_n{len(best_seq)}")
    
    # Phase 7: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 7: Pair CD n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t7, rng, verbose=True)
        check_and_update(a_out, f"pcd_n{len(best_seq)}")
    
    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 8: Multi-length exploration ===")
        # Try a few different odd lengths
        candidates_n = []
        base_n = len(best_seq)
        for delta in [-200, -100, 100, 200, 300]:
            n_try = base_n + delta
            if n_try % 2 == 0:
                n_try += 1
            if 500 <= n_try <= 3000:
                candidates_n.append(n_try)
        
        for n_try in candidates_n[:4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_try = min(remaining * 0.30, 50.0)
            print(f"  Trying n={n_try} (budget={t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            # Quick LSE + LP
            a_out, _ = lse_adam_optimize(init_try, t_try * 0.4, verbose=False)
            check_and_update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out, _ = nesterov_lp_symmetric(a_out, min(remaining * 0.4, 30.0), solver, verbose=False)
                check_and_update(a_out, f"ml_lp_n{n_try}")
    
    # Phase 9: Final LSE-Adam + CD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 9: Final polish (remaining={remaining:.0f}s) ===")
        t9a = min(remaining * 0.4, 60.0)
        a_out, _ = lse_adam_optimize(best_seq, t9a, verbose=False)
        check_and_update(a_out, "final_lse")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t9b = min(remaining * 0.5, 50.0)
            a_out, _ = pair_cd_optimize(best_seq, t9b, rng, verbose=False)
            check_and_update(a_out, "final_pcd")
    
    # Final: try symmetrization of best
    a_sym = symmetrize(best_seq)
    check_and_update(a_sym, "final_sym")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
