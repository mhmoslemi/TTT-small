```python
def project_onto_simplex_nonneg(v, target_sum):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    n = len(v)
    # Water-filling
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.arange(1, n + 1)
    mask = (v_sorted - (cumsum - target_sum) / rho) > 0
    if not mask.any():
        return np.full(n, target_sum / n)
    rho = np.max(np.where(mask)[0]) + 1
    theta = (cumsum[rho - 1] - target_sum) / rho
    return np.maximum(v - theta, 0.0)
```

- `y*[k] > 0` implies `(T b*)[k] = M`

The second condition tells us at which lags the convolution is tight. The first tells us which b* variables are positive.

In general, finding the exact primal solution from the dual is non-trivial (it requires solving another LP or using active set methods). But for the purpose of the Nesterov update, we might not need the exact primal solution.

Alternative approach: Instead of using the LP best response directly, use the dual variables as a "weight" in a gradient step.

The subgradient of `max_k conv(a,a)[k]` at `a` is:
`g[i] = sum_k w[k] * (a[k-i] + a[i-k])` (with valid indices)

where `w` is a probability distribution on the "active" lags (those where conv = max).

If we use the dual solution `y*` as the weights (normalized), we get a "soft" subgradient that accounts for all near-active lags, not just the exact argmax. This should give a better optimization direction.

```python
def dual_weighted_gradient(a, y, fft_size):
    """Compute gradient using dual weights."""
    n = len(a)
    size = 2 * n - 1
    # Extend y to full size (y[0:n] is set, y[n:2n-1] is 0)
    y_full = np.zeros(size)
    y_full[:n] = y
    y_sum = y_full.sum()
    if y_sum < 1e-30:
        return np.zeros(n)
    w = y_full / y_sum  # Normalize to probability
    
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g1 = irfft(np.conj(A) * W, fft_size)[:n]
    g2 = irfft(A * W, fft_size)[:n]
    grad = 2.0 * (g1 + g2)
    return grad
```

def fast_conv_self(a):

def compute_dual_fast(a):
    """O(n) analytical solution of the dual covering LP.
    Computes y such that y[i] = max(0, 1/a[i] - sum_{k=i+1}^{n-1} y[k])
    This gives sum_{k=i}^{n-1} y[k] >= 1/a[i] for all i, minimizing sum(y).
    """
    n = len(a)
    a_safe = np.maximum(a, 1e-30)
    y = np.zeros(n)
    running_sum = 0.0
    for i in range(n - 1, -1, -1):
        required = 1.0 / a_safe[i]
        deficit = required - running_sum
        if deficit > 0:
            y[i] = deficit
            running_sum += deficit
    return y

    g2 = irfft(A * W, fft_size)[:n]
