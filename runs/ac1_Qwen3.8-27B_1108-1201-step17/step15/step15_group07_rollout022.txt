def fast_autocorr(a):
    """Compute a * a (full convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v, S):
    """Project v onto {x : x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - S))[0][-1]
    theta = (cv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0)

    # correlate(p, a, 'valid')[i] = sum_k p[i+k] * a[k] for k=0..n-1
