def lse_smooth_optimize(a0, time_limit, verbose=True):
    """Log-sum-exp smoothed optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 15
    temps = np.geomspace(50.0, 0.05, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(50, min(500, int(remaining * 5)))
        
        lr = 1.0 / (T_val * np.sqrt(n))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # Smoothed max via LSE
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient: d/max_LSE / da = 2 * (w * a)_cross
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            # Backtracking line search
            step = lr / (1.0 + 0.01 * it)
            for trial in range(20):
                a_try = a - step * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    step *= 0.5
                    continue
                a_try = a_try * norm / s_try
                sc = fast_conv_self(a_try).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")
    
    return best_a, best_score

def dual_guided_gradient(a0, time_limit, solver, verbose=True):
    """Use LP dual variables for gradient direction."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(300):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break
        
        # Solve LP to get duals
        active = np.where(conv > M * 0.7)[0]
        b, duals = lp_best_response(a, M, solver, active)
        
        if duals is None:
            b2, duals = lp_best_response(a, M, solver, None)
            if duals is None:
                # Fallback
                duals = np.zeros(size)
                duals[np.argmax(conv)] = 1.0
                if b2 is not None:
                    b = b2
        
        # Gradient from duals
        D = rfft(duals, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * D, fft_size)[:n]
        g2 = irfft(A * D, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Line search
        step = 0.5 / np.sqrt(n)
        improved = False
        for trial in range(25):
            a_try = a - step * grad / gn
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try * norm / s_try
            sc = fast_conv_self(a_try).max()
            if sc < M - 1e-12:
                a = a_try
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Small random perturbation
            rng = np.random.default_rng(it + 1000)
            a += 0.01 * rng.standard_normal(n) / np.sqrt(n)
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualGrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

    n = len(a0)

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Process indices in random order
        perm = rng.permutation(n)
        
        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            i = idx
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Try a few random j's and structured ones
            j_candidates = list(rng.integers(n, size=15))

                j_candidates.append(int((kl + i) % n))
            
            best_delta = 0.0
            best_new_max = cur_max
            best_j = -1
            
            for j in set(j_candidates):
                if j == i:
                    continue
                aj = a[j]
                
                # Compute change in convolution when moving delta from i to j
                # delta_conv[k] = delta * (a[k-j] - a[k-i]) approximately
                # More precisely: new_conv[k] = conv[k] + delta*(a[k-j] + a[j-k'] - a[k-i] - a[i-k''])
                # This is complex, so use a simpler approach:

                    a[i] -= d
                    a[j] += d

                    a = a * norm / s_new

                    a[j] -= d
                    a[i] += d
                    a = a * norm / a.sum()

                        best_delta = d
                        best_j = j

                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()

                break

                if s_new > 1e-15:

            no_improve = 0

def structured_init(n, rng):
    """Generate diverse initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 12)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        # Multi-frequency
        a = 0.4 + 0.25 * np.cos(6 * np.pi * x) + 0.15 * np.cos(13 * np.pi * x) + 0.1 * np.cos(21 * np.pi * x)
        a = np.maximum(a, 0.02)
    elif choice == 7:
        # Sinc-like
        a = np.sinc(3.0 * (x - 0.5)) ** 2
        a = np.maximum(a, 0.01)
    elif choice == 8:
        # Two-peak with modulation
        a = 0.5 * np.exp(-6.0 * (x - 0.25) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.75) ** 2)
        a = a * (1.0 + 0.2 * np.cos(15 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 9:
        # Power-law
        a = (x * (1 - x)) ** 0.3 + 0.1
    elif choice == 10:
        # Random smooth
        a = np.ones(n)
        a = np.cumsum(rng.standard_normal(n))
        a = (a - a.min()) / (a.max() - a.min() + 1e-10)
    else:
        # Frank-like
        m = 5
        a = np.abs(np.sin(np.pi * m * x / 2.0)) ** 0.7 + 0.1
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interp_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss
