"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using high-precision LP best-response
with Nesterov acceleration, progressive constraint screening, and multi-length
exploration. Key improvements: full-constraint LP for accuracy, adaptive
screening that tightens over iterations, and a dedicated polish phase.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve_full(a, M, solver):
    """Solve LP with all constraints: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_solve_screened(a, M, solver, active):
    """Solve LP with only active constraints."""
    n = len(a)
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active, :] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated LP best-response.
    Normalizes to sum = sqrt(2n).
    At fixed point, objective = max(conv(a,a)).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Track convergence
    prev_diff = float('inf')
    no_improve_count = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Progressive screening: start with all constraints, gradually screen
        if t < 50:
            # Use full constraints initially for accuracy
            b = lp_solve_full(y, M, solver)
        else:
            # Adaptive screening
            threshold = M * (0.85 + 0.05 * min(1.0, t / 500.0))
            # threshold goes from 0.85M to 0.90M over 500 iterations
            active = np.where(conv_y >= threshold)[0]
            if len(active) < 20:
                active = np.where(conv_y >= M * 0.5)[0]
            
            b = lp_solve_screened(y, M, solver, active)
            
            # Verify constraints
            if b is not None:
                check_conv = fast_conv_cross(y, b)
                violated = np.where(check_conv > M * 1.0001)[0]
                if len(violated) > 0:
                    # Re-solve with violated constraints added
                    full_active = np.union1d(active, violated)
                    b = lp_solve_screened(y, M, solver, full_active)
                    if b is None:
                        b = lp_solve_full(y, M, solver)

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        # Track convergence
        diff = np.linalg.norm(a_new - a) / norm
        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            no_improve_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov t={t} ({elapsed:.0f}s): best_M={best_score:.10f}, diff={diff:.2e}")

        # Early stop if converged
        if t > 100 and diff < 1e-8 and no_improve_count > 50:
            if verbose:
                print(f"    LP-Nesterov: converged at t={t}")
            break

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent for fast initial progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)

def polish_coordinate_descent(a0, time_limit, rng, verbose=True):
    """
    Fine-grained coordinate descent targeting the active set.
    For each pair (i, j), find the optimal transfer amount that minimizes max(conv).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Focus on top peaks
        peak_threshold = cur_max * 0.98
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue

            # Generate target candidates
            j_candidates = set(int(rng.integers(n)) for _ in range(15))
            for kl in peak_lags[:5]:
                j_struct = (kl - i + n) % n
                j_struct2 = (kl + i) % n
                j_candidates.add(j_struct)
                j_candidates.add(j_struct2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            best_conv_update = None

            for j in list(j_candidates)[:20]:
                if j == i:
                    continue
                aj = a[j]

                # Compute the change in convolution when moving delta from i to j
                # conv_new = conv_old + delta * (g_j - g_i) + quadratic terms
                g = np.zeros(size)
                # Contribution of adding to position j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Subtract contribution of removing from position i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Binary search for optimal delta
                lo_d = 0.0
                hi_d = ai * 0.99
                # Check if increasing delta helps
                mid_d = hi_d * 0.5
                new_conv_mid = conv + mid_d * g
                if 2 * i < size:
                    new_conv_mid[2 * i] = conv[2 * i] - 2 * ai * mid_d + mid_d * mid_d
                if 2 * j < size:
                    new_conv_mid[2 * j] = conv[2 * j] + 2 * aj * mid_d + mid_d * mid_d
                new_val = new_conv_mid.max()
                
                if new_val >= best_val:
                    # Try smaller
                    hi_d = mid_d
                    if hi_d < 1e-10:
                        break
                    continue

                # Try larger
                lo_d_old = lo_d
                lo_d = mid_d
                best_val_j = new_val
                best_delta_j = mid_d
                best_conv_update_j = new_conv_mid.copy()

                # Refine with more iterations
                for _ in range(20):
                    mid_d = (lo_d + hi_d) / 2
                    if mid_d <= lo_d or mid_d >= hi_d:
                        break
                    new_conv_mid = conv + mid_d * g
                    if 2 * i < size:
                        new_conv_mid[2 * i] = conv[2 * i] - 2 * ai * mid_d + mid_d * mid_d
                    if 2 * j < size:
                        new_conv_mid[2 * j] = conv[2 * j] + 2 * aj * mid_d + mid_d * mid_d
                    v = new_conv_mid.max()
                    if v < best_val_j:
                        best_val_j = v
                        best_delta_j = mid_d
                        best_conv_update_j = new_conv_mid.copy()
                        lo_d = mid_d
                    else:
                        hi_d = mid_d

                    best_val = best_val_j
                    best_delta = best_delta_j
                    best_j = j

            print(f"    Polish pass ({elapsed:.0f}s): best={best_score:.10f}")

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def symmetric_init(n, rng):
    """Create a symmetric initial guess."""
    half = (n + 1) // 2
    # Various symmetric shapes
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, half)
    if choice == 0:
        h = np.sin(np.pi * x) ** 1.2
    elif choice == 1:
        h = np.cos(np.pi * x / 2) ** 2
    elif choice == 2:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
    elif choice == 3:
        h = np.exp(-3 * (2*x - 1)**2)
    elif choice == 4:
        h = (1 - np.abs(2*x - 1)) ** 1.5 + 0.1
    elif choice == 5:
        h = 0.4 + 0.4*np.sin(np.pi*x)**0.7 + 0.2*np.cos(4*np.pi*x)
    elif choice == 6:
        h = np.exp(-2*(x-0.3)**2/(0.1+0.1*rng.random())) + 0.3*np.exp(-2*(x-0.7)**2/0.15)
    else:
        h = np.sinc(2*(x-0.5))**2 + 0.15
    h = np.maximum(h, 0.01)
    # Make symmetric
    if n % 2 == 0:
        a = np.concatenate([h[::-1], h])
    else:
        a = np.concatenate([h[::-1][:-1], h])
    a = a + 0.005 * rng.random(len(a))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using high-precision LP best-response
with Nesterov acceleration, full-constraint LP for accuracy, progressive
screening, and multi-length exploration with coordinate descent polishing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve_with_constraints(a, M, solver, constraint_rows):
    """Solve LP: max sum(b) s.t. T[constraint_rows,:] @ b <= M, b >= 0."""
    n = len(a)
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if constraint_rows is not None and len(constraint_rows) < 2 * n - 1:
        constraints = [T[constraint_rows, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated LP best-response iteration.
    Normalizes to sum = sqrt(2n). At fixed point, max(conv(a,a)) is the objective.
    Uses full constraints initially, then progressive screening.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Progressive constraint screening
        if t <= 30:
            # Full constraints for accuracy in early iterations
            constraint_rows = None
        elif t <= 100:
            # Keep constraints where conv >= 0.8*M
            threshold = M * 0.80
            constraint_rows = np.where(conv_y >= threshold)[0]
            if len(constraint_rows) < 10:
                constraint_rows = None
        else:
            # Tighter screening as we converge
            frac = min(1.0, (t - 100) / 800.0)
            threshold = M * (0.85 + 0.05 * frac)
            constraint_rows = np.where(conv_y >= threshold)[0]
            if len(constraint_rows) < 10:
                constraint_rows = np.where(conv_y >= M * 0.5)[0]

        b = lp_solve_with_constraints(y, M, solver, constraint_rows)

        # Verify and fix dropped constraints
        if b is not None and constraint_rows is not None and len(constraint_rows) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.0005)[0]
            if len(violated) > 2:
                all_rows = np.unique(np.concatenate([constraint_rows, violated]))
                b = lp_solve_with_constraints(y, M, solver, all_rows)
                if b is None:
                    b = lp_solve_with_constraints(y, M, solver, None)

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score - 1e-13:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-13:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): M={best_score:.10f}")

        # Convergence check
        if t > 200:
            conv_check = fast_conv_self(a)
            M_check = conv_check.max()
            # Check if a is close to a fixed point of best response
            if abs(M_check - M) / max(M, 1e-15) < 1e-7:
                # Additional iterations didn't help, stop
                if verbose:
                    print(f"    LP-Nest: near-converged at t={t}, M={M:.10f}")
                break

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent for fast initial progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(40.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(20, min(150, int(remaining * 4)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            step = norm / (T_val * np.sqrt(n)) * 1.0 / (1.0 + 0.05 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def polish_cd(a0, time_limit, rng, verbose=True):
    """
    Fine-grained coordinate descent with binary search for optimal transfer.
    Targets the active set (near-peak lags) for maximum effect.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_passes = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Focus optimization on coordinates that contribute to peaks
        peak_threshold = cur_max * 0.97
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Candidate targets: random + structured from peak lags
            j_set = set()
            for _ in range(12):
                j_set.add(int(rng.integers(n)))
            for kl in peak_lags[:6]:
                for offset in [-1, 0, 1]:
                    j1 = (kl - i + offset + n) % n
                    j2 = (kl + i + offset) % n
                    j_set.add(j1)
                    j_set.add(j2)
            j_set.discard(i)
            j_list = list(j_set)[:25]

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_list:
                if j == i:
                    continue
                aj = a[j]
                if aj < 1e-15 and ai < 1e-15:
                    continue

                # Compute gain vector: change in conv when moving 1 unit from i to j
                g = np.zeros(size)
                # Add at position j
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # Remove from position i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Check if any improvement possible
                test_d = ai * 0.3
                test_conv = conv + test_d * g
                if 2 * i < size:
                    test_conv[2 * i] = conv[2 * i] - 2 * ai * test_d + test_d * test_d
                if 2 * j < size:
                    test_conv[2 * j] = conv[2 * j] + 2 * aj * test_d + test_d * test_d
                if test_conv.max() >= best_val:
                    # Try small delta
                    test_d = ai * 0.05
                    test_conv = conv + test_d * g
                    if 2 * i < size:
                        test_conv[2 * i] = conv[2 * i] - 2 * ai * test_d + test_d * test_d
