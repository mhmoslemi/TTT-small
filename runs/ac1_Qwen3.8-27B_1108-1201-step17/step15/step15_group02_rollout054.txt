def fast_conv_self(a):
    """Fast self-convolution using FFT"""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Fast cross-convolution using FFT"""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_max_conv(a):
    """Compute max of self-convolution"""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf'), a.copy()
    n = len(a)
    conv = fast_conv_self(a)
    return conv.max(), a

def normalize(a, norm_val=None):
    """Normalize sequence to have given sum (default sqrt(2n))"""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * (norm_val or np.sqrt(2*len(a))) / len(a)
    if norm_val is None:
        norm_val = np.sqrt(2.0 * len(a))
    return a * norm_val / s

def lse_adam_optimize(a0, time_limit, verbose=False):
    """LSE-smoothed Adam optimization"""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0
    
    # Adaptive temperature schedule
    n_stages = 25
    temps = np.geomspace(100.0, 0.01, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(150, int(remaining * 12)))
        lr = 0.3 / np.sqrt(n)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # Soft-max weights
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            # Gradient of sum_k w[k] * conv(a,a)[k] with respect to a
            # = 2 * (a * w) in some sense
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm_val / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")
    
    return best_a, best_score

def build_active_constraint_matrix(a, active_set):
    """Build constraint matrix for LP with given active set of convolution lags"""
    n = len(a)
    m = len(active_set)
    T = np.zeros((m, n))
    for idx, k in enumerate(active_set):
        # conv(a,b)[k] = sum_i a[i] * b[k-i] where 0<=i<n, 0<=k-i<n
        # So b[j] appears with coefficient a[k-j] for j in [k-n+1, k]
        row_start = max(0, k - n + 1)
        row_end = min(n, k + 1)
        for i in range(row_start, row_end):
            T[idx, i] = a[k - i]
    return T

def lp_best_response(a, M, solver, active_set):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for k in active_set, b >= 0"""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and len(active_set) > 0:
        if len(active_set) < 2 * n - 1:
            T = build_active_constraint_matrix(a, active_set)
        else:
            T = build_active_constraint_matrix(a, np.arange(2*n-1))
        constraints = [T @ b_var <= M]
    else:
        # Full constraint set
        size = 2 * n - 1
        T = build_active_constraint_matrix(a, np.arange(size))
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

        if sc_y < best_score:

            best_score = sc_a

        if verbose and t % 20 == 0:

            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}, |AS|={len(active_set) if active_set is not None else size}")

def vegas_optimize(a0, time_limit, verbose=False):
    """Vegas-style multiplicative updates"""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm_val / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2*n-2).bit_length()
    
    n_stages = 20
    etas = np.geomspace(0.05, 0.0005, n_stages)
    
    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(15, min(50, int(remaining * 5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break
            
            # Pressure: where convolution is high
            w = np.maximum(conv - M * 0.85, 0.0)
            ww = w / (w.sum() + 1e-15)
            
            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            
            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-12)
            a = a * norm_val / a.sum()
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and si % 4 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.5f}: best={best_score:.8f}")
    
    return best_a, best_score

def symmetrize(a):
    """Make sequence symmetric"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a_sym = (a + a[::-1]) / 2.0
    s_sym = a_sym.sum()
    if s_sym < 1e-15:
        return a
    return a_sym * s / s_sym

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length"""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    """Generate structured initial sequence"""
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2*np.pi*f1*x) + 0.2 * np.cos(2*np.pi*f2*x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30*(x-c1)**2) + np.exp(-30*(x-c2)**2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0*x - 1.0)) ** 3
    elif choice == 5:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 6:
        n_peaks = rng.integers(3, 8)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.08)
            a += np.exp(-(x-c)**2 / (2*width**2))
    elif choice == 7:
        a = 1.0 + 0.3*np.sin(2*np.pi*3*x) + 0.2*np.sin(2*np.pi*7*x)
    elif choice == 8:
        a = np.exp(-3*x) + 0.3
    else:
        a = 0.5 + 0.3*np.cos(2*np.pi*x) + 0.2*np.cos(4*np.pi*x)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.003 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a
