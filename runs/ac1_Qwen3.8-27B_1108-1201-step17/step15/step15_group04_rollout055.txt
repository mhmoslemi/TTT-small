_solver_cache = None

def get_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.SCS
    print(f"  Using solver (fallback): SCS")
    return _solver_cache


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    # T[k, i] = a[i] if 0 <= k-i < n, else 0
    # Actually T[k, i] should be such that (T @ b)[k] = sum_i T[k,i]*b[i] = conv(a,b)[k]
    # conv(a,b)[k] = sum_i a[max(0,k-n+1):min(n,k+1)] ... 
    # Let's just build it directly:
    T = np.zeros((size, n))
    for i in range(n):
        # b[i] contributes to conv positions i, i+1, ..., i+n-1 with values a[0], a[1], ..., a[n-1]
        lo = i
        hi = i + n
        T[lo:hi, i] = a
    return T


def lp_best_response(a, M, solver, active_idx=None, max_iters=30):
    """Solve the LP best-response with adaptive constraint screening."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if active_idx is not None and len(active_idx) > 0:
        constraints = [T[active_idx, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    # Get dual variables
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active_idx is not None and len(active_idx) < size:
                full_duals = np.zeros(size)
                full_duals[active_idx] = np.maximum(np.asarray(con.dual_value, dtype=np.float64), 0)
                duals = full_duals
            else:
                duals = np.maximum(np.asarray(con.dual_value, dtype=np.float64), 0)
    except Exception:
        pass
    
    # Check for violated constraints
    if active_idx is not None and len(active_idx) < size:
        check = T @ val
        violated = np.where(check > M * (1 + 1e-4))[0]
        if len(violated) > 0 and len(active_idx) < size - 1:
            # Add violated constraints and re-solve
            new_active = np.union1d(active_idx, violated)
            if len(new_active) < size:
                return lp_best_response(a, M, solver, new_active)
            else:
                constraints = [T @ b_var <= M]
                prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
                try:
                    prob.solve(solver=solver, verbose=False)
                except Exception:
                    pass
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                duals = None
        elif len(violated) > 0:
            # Full solve
            constraints = [T @ b_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                pass
            if prob.status in ('optimal', 'optimal_inaccurate'):
                val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            duals = None
    
    return val, duals

def lse_gradient_descent(a0, time_limit, verbose=True, nesterov_momentum=0.9):
    """Nesterov-accelerated LSE smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    prev_v = a.copy()  # For Nesterov
    
    # Adaptive temperature schedule
    T_start = 50.0
    T_end = 0.3
    total_planned = 5000
    
    for it in range(total_planned):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        v = a + nesterov_momentum * (a - prev_v)
        v = np.maximum(v, 0.0)
        sv = v.sum()
        if sv > 1e-15:
            v = v * norm / sv
        
        conv = fast_conv_self(v)
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Adaptive temperature
        progress = it / max(total_planned, 1)
        T_val = T_start * (T_end / T_start) ** progress
        
        # LSE gradient
        shifted = conv / T_val
        shifted = np.maximum(shifted, -50.0)
        exp_c = np.exp(shifted - shifted.max())
        w = exp_c / exp_c.sum()
        
        W = rfft(w, fft_size)
        A = rfft(v, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = g1 + g2
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            prev_v = a.copy()
            continue
        
        # Step size
        step = 1.0 / (grad_norm * (1.0 + 0.1 * np.exp(-it / 100.0)))
        
        # Backtracking line search
        for trial in range(10):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc <= max_c + 0.01 * T_val:
                break
            step *= 0.5
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        prev_v = a.copy()
        a = a_new
        
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it} T={T_val:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent: move mass from high-contribution to low-contribution positions."""

    best_a = a.copy()

        # and TO (low values)

                continue

                j_candidates.append(j1)
                j_candidates.append(j2)
            
            j_candidates = [j for j in j_candidates if j != i and j >= 0 and j < n]
            
            best_delta = 0.0
            best_new_max = cur_max
            best_j = -1
            
            for j in j_candidates[:12]:
                if j == i:
                    continue
                
                # Compute effect of moving delta from i to j
                # conv_new = conv + delta*(contribution_j - contribution_i)
                # contribution_j[k] += a[k-j] for valid k (adding a[j] effect)
                # contribution_i[k] -= a[k-i] for valid k (removing a[i] effect)
                # Plus diagonal corrections at k=2i and k=2j
                
                g = np.zeros(size)
                # Adding mass to position j: adds a[k] * delta to conv[k-j] for k=0..n-1
                for k in range(min(n, size - j)):
                    g[j + k] += a[k] * delta if False else a[k]
                # Actually let me just evaluate a few deltas directly
                # For efficiency, compute the gain vector once
                
                # Gain from moving delta from i to j:
                # new_a = a with a[i]-=delta, a[j]+=delta
                # new_conv = conv - delta*(col_i) + delta*(col_j) + delta^2*(diagonal corrections)
                # where col_i[k] = a[max(0,k-n+1):...] ... 
                # This is complex, let's just try a few values directly
                
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    d = frac * ai
                    if d >= ai:
                        continue
                    a[i] -= d
                    a[j] += d
                    s_new = a.sum()
                    if s_new < 1e-15:
                        a[i] += d
                        a[j] -= d
                        continue
                    a = a * norm / s_new
                    new_conv = fast_conv_self(a)
                    new_max = new_conv.max()
                    
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_delta = d
                        best_j = j
                        a[i] += d  # restore
                        a[j] -= d
                        a = a * norm / a.sum()
                    else:
                        a[i] += d
                        a[j] -= d
                        a = a * norm / a.sum()
                
                # Restore a
                a = np.maximum(a, 0.0)
                s_restore = a.sum()
                if s_restore > 1e-15:
                    a = a * norm / s_restore
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
        
        total_iter += 1
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                # Perturb
                n_perturb = max(10, n // 20)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and total_iter % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD iter={total_iter} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def smooth_migration(a0, time_limit, rng, verbose=True):
    """Smooth migration: gradually redistribute mass using convolution with a small kernel."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Convolve with a small smoothing kernel (biased to reduce peaks)
        k_size = max(3, n // 100)
        if k_size % 2 == 0:
            k_size += 1
        kernel = np.exp(-0.5 * (np.arange(k_size) - k_size // 2) ** 2 / (k_size // 4))
        kernel /= kernel.sum()
        
        a_smooth = np.convolve(a, kernel, mode='same')
        a_smooth = np.maximum(a_smooth, 0.0)
        # Blend
        alpha = 0.1
        a_new = (1 - alpha) * a + alpha * a_smooth
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = compute_score_local(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        elif np.isfinite(sc) and best_seq is None:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  [init] {label}: {sc:.6f} (n={len(a)})")
    
    # === Phase 0: Multi-start initialization ===
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        
        # Try diverse initializations
        inits = []
        for alpha in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
            inits.append(init_ramp(n_base, alpha, rng))
        for power in [0.5, 1.0, 1.5, 2.0]:
            inits.append(init_cosine_taper(n_base, power, rng))
        for sigma in [0.05, 0.1, 0.2]:
            inits.append(init_smoothed_ramp(n_base, sigma, rng))
        for ratio in [0.1, 0.2, 0.5]:
            inits.append(init_two_decay(n_base, ratio, rng))
        for _ in range(3):
            inits.append(init_random_smooth(n_base, rng))
        
        # Add uniform
        inits.append(np.ones(n_base) / n_base)
        
        for ii, init in enumerate(inits):
            sc = compute_score_local(init)
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # === Phase 1: Quick LSE to improve initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 90.0)
    print(f"\n=== Phase 1: LSE init (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=False)
        try_update(a_out, "lse_init")
    
    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 480.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_cur}")
    
    # === Phase 3: LSE refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: LSE refine (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t3, verbose=False)
        try_update(a_out, "lse_refine")
    
    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 360.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_cur}")
    
    # === Phase 5: Pair CD polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.18, 240.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Pair CD n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"pcd_n{n_cur}")
    
    # === Phase 6: Try larger n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 300
    print(f"\n=== Phase 6: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        # Interpolate
        old_n = len(best_seq)
        s = best_seq.sum()
        f_norm = best_seq / s
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, n_long)
        f_interp = np.interp(new_x, old_x, f_norm)
        f_interp = np.maximum(f_interp, 0.0)
        ss = f_interp.sum()
        if ss > 1e-15:
            f_interp = f_interp / ss
        
        t_long = min(remaining * 0.35, 180.0)
        a_out, _ = lp_nesterov_optimize(f_interp, t_long, solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_long:
            t_cd = min(remaining * 0.3, 80.0)
            a_out2, _ = pair_cd_optimize(best_seq, t_cd, rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")
    
    # === Phase 7: Try smaller n (sometimes smaller is better for this objective) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(800, len(best_seq) - 300)
    if remaining > 80 and n_short < len(best_seq):
        print(f"\n=== Phase 7: Shorter n={n_short} (remaining={remaining:.0f}s) ===")
        old_n = len(best_seq)
        s = best_seq.sum()
        f_norm = best_seq / s
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, n_short)
        f_interp = np.interp(new_x, old_x, f_norm)
        f_interp = np.maximum(f_interp, 0.0)
        ss = f_interp.sum()
        if ss > 1e-15:
            f_interp = f_interp / ss
        
        t_short = min(remaining * 0.3, 100.0)
        a_out, _ = lp_nesterov_optimize(f_interp, t_short, solver, verbose=False)
        try_update(a_out, f"lp_short_{n_short}")
    
    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t_f1 = min(remaining * 0.4, 50.0)
        a_out, _ = lse_gradient_descent(best_seq, t_f1, verbose=False)
        try_update(a_out, "lse_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_f2 = min(remaining * 0.5, 40.0)
            a_out, _ = pair_cd_optimize(best_seq, t_f2, rng, verbose=False)
            try_update(a_out, "pcd_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_f3 = min(remaining * 0.6, 30.0)
            a_out, _ = lp_nesterov_optimize(best_seq, t_f3, solver, verbose=False)
            try_update(a_out, "lp_final")
    
    # Final validation
    if best_seq is not None:
        final_score = compute_score_local(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        # Also verify with actual evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(eval_score) and eval_score < best_score:
                best_score = eval_score
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
