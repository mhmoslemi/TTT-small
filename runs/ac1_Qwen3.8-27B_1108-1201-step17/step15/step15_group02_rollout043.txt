**Spectral Flattening:**
1. Compute A = rfft(a, fft_size) (zero-padded FFT)
2. Compute P = |A|^2 (power spectrum)
3. We want P to be as flat as possible (uniform). 
4. Target: P_target = constant = sum(P) / len(P)
5. Update: P_new = P * (P_target / P)^alpha = P^(1-alpha) * P_target^alpha
   (This compresses the dynamic range of the power spectrum)
6. Set |A_new| = sqrt(P_new), keep phase: phase = A / |A|
7. A_new = |A_new| * phase
8. a_new = irfft(A_new, fft_size)[:n]
9. Project: a_new = max(a_new, 0)
10. Renormalize: a_new = a_new * norm / sum(a_new)
11. Repeat with decreasing alpha

The parameter alpha controls aggressiveness:
- alpha close to 1: strong flattening (might create negative values that get clipped)
- alpha close to 0: gentle adjustment

This should be combined with the LSE-Adam approach for best results.

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    eps = 1e-8
    t_adam = 0
    total_steps = 0

    n_stages = 40
    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        frac = si / max(n_stages - 1, 1)
        # Temperature: exponential cooldown from t_start to t_end
        T = t_start * (t_end / t_start) ** frac

        n_steps = max(20, int(remaining * 20 / (n_stages - si)))
        lr = lr_scale / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            total_steps += 1

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T
            shifted = np.clip(shifted, -120.0, 0.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            grad = lse_gradient(a, w, T)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam stage {si}/{n_stages} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, verbose=False):
    """Exact max subgradient with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Subgradient: use all lags within 5% of max
        threshold = M * 0.95
        active = np.where(conv_y >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_y)])

        # Compute subgradient at y
        grad = np.zeros(n)
        for k in active[:50]:  # Limit active set size
            for i in range(n):
                d = abs(k - i)
                if d < n:
                    grad[i] += 2.0 * y[d]
        grad /= len(active[:50])

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        step_size = alpha * norm / (np.linalg.norm(grad) + 1e-15)
        y_new = y - step_size * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_optimize(a0, time_limit, verbose=False):
    """
    Iterative frequency-domain amplitude correction.
    Goal: make the convolution flat by adjusting the power spectrum.
    Strategy: where the convolution has peaks, the corresponding frequency
    content should be reduced; where it's low, increase.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_it = 0
    # Annealing: start with large step, reduce over time
    for outer in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_inner = max(5, min(50, int(remaining * 10 / (50 - outer))))
        alpha = 0.3 * (0.95 ** outer)  # Step size decay

        for inner in range(n_inner):
            if time.monotonic() - t0 > time_limit:
                break
            n_it += 1

            conv = fast_conv_self(a)
            M = conv.max()
            avg = conv.mean()

            # Phase: get current FFT
            A = rfft(a, fft_size)
            mag = np.abs(A)
            phase = A / (mag + 1e-30)

            # Power spectrum
            P = mag * mag

            # Target: we want conv to be flat at value ~ avg
            # In frequency domain, flat conv at avg means P should be
            # proportional to the DTFT of a flat sequence of length 2n-1
            # But we approximate: reduce power where it causes peaks
            # Use the "pressure" from high conv values
            pressure = np.maximum(conv - M * 0.8, 0.0)
            P_pressure = np.abs(rfft(pressure, fft_size)) ** 2

            # Modify power spectrum: reduce where pressure is high
            P_new = P * np.exp(-alpha * P_pressure / (P.mean() + 1e-30))
            P_new = np.maximum(P_new, 0.0)

            # Restrict to positive frequencies (rfft)
            mag_new = np.sqrt(P_new)

            A_new = mag_new * phase
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new * norm / s_new

            sc = fast_conv_self(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            # Otherwise keep old a (rejection)

        if verbose and outer % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter {n_it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
