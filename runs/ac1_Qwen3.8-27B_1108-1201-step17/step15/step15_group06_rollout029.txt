def build_sparse_constraints(a, active_idx):
    """Build sparse constraint matrix for cross-correlation a*b[k] <= M."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                rows.append(row)
                cols.append(j)
                vals.append(a[idx])
    T = coo_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    return T.tocsr()


def lp_max_sum(a, M, solver, active_idx, n):
    """Solve max sum(b) s.t. (a*b)[k] <= M for k in active_idx, b >= 0."""
    size = 2 * n - 1

    T = build_sparse_constraints(a, active_idx)
    if T.shape[0] == 0:
        return np.ones(n) / n
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_min_max(a, solver, active_idx, n, normalize_sum=1.0):
    """Solve min t s.t. (a*b)[k] <= t for k in active_idx, sum(b) = normalize_sum, b >= 0."""
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_sparse_constraints(a, active_idx)
    if T.shape[0] == 0:
        val = np.ones(n) * normalize_sum / n
        return val, 0.0
    cons = [T @ b <= t, cp.sum(b) == normalize_sum, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:

    return val, t_val

def gradient_refine(a0, time_limit, n_peaks=30, verbose=True):
    """Frequency-domain inspired gradient refinement for reducing autocorrelation peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    a_prev = a.copy()
    
    eta = 0.02
    momentum = 0.7
    stall = 0

    for it in range(1, 500000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        
        # Find active peaks
        threshold = M * (0.85 + 0.1 * min(it / 100.0, 1.0))
        active_peaks = np.where(conv >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv)])
        # Limit number of peaks for efficiency
        if len(active_peaks) > n_peaks:
            top_idx = np.argsort(conv[active_peaks])[-n_peaks:]
            active_peaks = active_peaks[top_idx]

        # Compute averaged subgradient
        grad = np.zeros(n)
        for k in active_peaks:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        grad /= len(active_peaks)

        # Nesterov momentum
        a_new = a + momentum * (a - a_prev)
        a_new = a_new - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.02, 0.1)
        else:
            stall += 1
            if stall % 20 == 0:
                eta = max(eta * 0.9, 0.001)
            if stall > 100:
                momentum = max(momentum * 0.95, 0.3)
                stall = 0

        a_prev = a
        a = a_new

        if verbose and it % 500 == 0:
            print(f"    GRAD it={it} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coord_descent(a0, time_limit, rng, verbose=True):

    while time.monotonic() - t0 < time_limit:

                continue

                    f2 = f1

                    a_t1[i] = c1

    while time.monotonic() - t0 < time_limit:
        n_passes += 1

            if time.monotonic() - t0 > time_limit:
                break

            j = perm[idx + 1]
            if i == j:
                continue

                d = max_d * frac
                if d <= 0:

                # i -> j

                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    v = fast_autocorr(a_try).max()
                    if v < best_val - 1e-14:

                        best_d = d
                # j -> i
                a_try2 = a.copy()
                a_try2[j] -= d
                a_try2[i] += d
                s_try2 = a_try2.sum()
                if s_try2 > 1e-15:
                    a_try2 = a_try2 * norm / s_try2
                    v2 = fast_autocorr(a_try2).max()
                    if v2 < best_val - 1e-14:
                        best_val = v2
                        best_d = -d

            if abs(best_d) > 1e-14:
                a_try = a.copy()
                if best_d > 0:
                    a_try[i] -= best_d
                    a_try[j] += best_d
                else:
                    a_try[i] += abs(best_d)
                    a_try[j] -= abs(best_d)
                s_try = a_try.sum()
                if s_try > 1e-15:

                    sc = fast_autocorr(a_try).max()
                    if sc < best_score - 1e-14:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, style=0):

    elif style == 2:

        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:

        # Asymmetric: ramp up then slow decay
        a = np.minimum(x * 3.0, 1.0) * np.exp(-2.0 * x)
        a = a + 0.3 * np.exp(-5.0 * (x - 0.6) ** 2)
        a = np.maximum(a, 0.0)
    elif style == 8:
        # Asymmetric: concentrated in first half
        a = np.exp(-3.0 * (x - 0.25) ** 2) + 0.5 * np.exp(-3.0 * (x - 0.6) ** 2)
        a = np.maximum(a, 0.0)
    elif style == 9:
        # Multi-peak asymmetric
        a = 0.8 * np.exp(-10.0 * (x - 0.15) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.45) ** 2) + 0.4 * np.exp(-6.0 * (x - 0.8) ** 2)
        a = np.maximum(a, 0.0)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def smooth_perturb(a, rng, bandwidth=20, scale=0.03):
    """Smooth perturbation that preserves local structure."""
    n = len(a)
    noise = rng.standard_normal(n)
    # Smooth the noise
    kernel = np.exp(-0.5 * (np.arange(-bandwidth, bandwidth + 1) / (bandwidth / 3.0)) ** 2)
    kernel /= kernel.sum()
    smooth_noise = np.convolve(noise, kernel, mode='same')
    ap = a * (1.0 + scale * smooth_noise)
    ap = np.maximum(ap, 0.0)
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
