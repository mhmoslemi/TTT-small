- If k = 2*j0: the term a[j0]*a[j0] becomes x*x, so delta = x^2 - a[j0]^2
- Otherwise, if k-j0 is valid: the terms a[j0]*a[k-j0] and a[k-j0]*a[j0] each change by (x - a[j0])*a[k-j0], so delta = 2*(x - a[j0])*a[k-j0]
- If k-j0 is not valid: delta = 0

So: `new_conv[k] = old_conv[k] + delta(k)` where:
- If k = 2*j0: `delta = x^2 - a[j0]^2`
- If 0 <= k-j0 < n and k != 2*j0: `delta = 2*(x - a[j0])*a[k-j0]`
- Otherwise: `delta = 0`

This is O(1) per k to compute the delta, so O(n) per coordinate evaluation. With n coordinates and a few x values each, the total per pass is O(n^2) which is manageable.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)

def norm_vec(a, n=None):
    if n is None:
        n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def build_T(a, k_indices, n):
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            mat[idx, lo:hi] = a[k - (np.arange(lo, hi))]
    return mat


def solve_lp(a, M, k_indices, solver, n):
    mat = build_T(a, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dv = constraints[0].dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = norm_vec(a0, n)
    size = 2 * n - 1
    t_start = time.monotonic()

    conv_y = fast_autocorr(a)
    best_score = conv_y.max()
    best_a = a.copy()

    y = a.copy()
    y_prev = a.copy()
    active_set = None
    duals_cache = None
    stall_count = 0
    stall_threshold = 60
    n_iters = 0
    last_improve_iter = 0
    erg_a = np.zeros(n)
    erg_count = 0
    erg_best = None
    erg_best_score = float('inf')

    for t in range(1, 200000):
        if time.monotonic() - t_start > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Active set management
        if active_set is None or t % 30 == 0:
            # Tier 1: always include top peaks
            n_top = max(5, min(20, n // 80))
            top_idx = np.argsort(conv_y)[-n_top:]

            # Tier 2: threshold-based
            tol = max(0.005, 0.04 * np.exp(-t / 400.0))
            thresh_idx = np.where(conv_y >= M_curr * (1.0 - tol))[0]

            # Tier 3: dual-guided (if we have cached duals)
            dual_idx = []
            if duals_cache is not None and active_set is not None:
                # Constraints with significant dual values are "active"
                sig_threshold = np.max(np.abs(duals_cache)) * 0.01
                sig_mask = np.abs(duals_cache) > sig_threshold
                dual_idx = active_set[sig_mask]

            active_set = np.unique(np.concatenate([top_idx, thresh_idx, dual_idx]))
            # Limit size for LP efficiency
            max_active = min(size, max(50, n // 5))
            if len(active_set) > max_active:
                # Keep only the most important ones (highest conv values + significant duals)
                conv_vals = conv_y[active_set]
                order = np.argsort(-conv_vals)
                active_set = active_set[order[:max_active]]

        b, duals = solve_lp(y, M_curr, active_set, solver, n)
        if b is None:
            # Fallback: use all constraints
            if t % 5 == 0:
                active_set = np.arange(size)
                b, duals = solve_lp(y, M_curr, active_set, solver, n)
            if b is None:
                continue
        if duals is not None:
            duals_cache = duals

        # Verify and expand if needed
        if len(active_set) < size - 1:
            comp = np.setdiff1d(np.arange(size), active_set)
            # Sample a subset of complement for speed
            if len(comp) > 200:
                comp_sample = comp[rng.choice(len(comp), size=200, replace=False)]
            else:
                comp_sample = comp
            T_comp = build_T(y, comp_sample, n)
            violations = T_comp @ b
            viol = comp_sample[violations > M_curr * (1 + 1e-5)]
            if len(viol) > 0:
                active_set = np.unique(np.concatenate([active_set, viol]))
                if len(active_set) > 0.95 * size:
                    active_set = np.arange(size)
                b, duals = solve_lp(y, M_curr, active_set, solver, n)
                if b is None:
                    continue
                if duals is not None:
                    duals_cache = duals

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        n_iters += 1

        # Nesterov update with adaptive momentum
        if t == 1:
            y_new = g.copy()
            a_new = g.copy()
        else:
            # Adaptive step: standard Nesterov with cap
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm_val / s_y

            # Momentum for a
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best from y
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
            last_improve_iter = t
        else:
            stall_count += 1

        # Track best from a
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
            last_improve_iter = t

        # Ergodic average
        erg_a += a
        erg_count += 1
        if erg_count % 10 == 0:
            erg_check = erg_a / erg_count
            s_erg = erg_check.sum()
            if s_erg > 1e-15:
                erg_check = erg_check * norm_val / s_erg
                score_erg = fast_autocorr(erg_check).max()
                if score_erg < best_score:
                    best_score = score_erg
                    best_a = erg_check.copy()
                    stall_count = 0
                    last_improve_iter = t
                    erg_best = erg_check.copy()
                    erg_best_score = score_erg

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            duals_cache = None
            stall_count = 0
            # Reset ergodic average for fresh start
            erg_a = np.zeros(n)
            erg_count = 0
            if verbose:
                elapsed = time.monotonic() - t_start
                print(f"    LP restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, active={len(active_set)}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = norm_vec(a0, n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t_start = time.monotonic()
    step = 1.0 / np.sqrt(n)
    patience = 0
    max_patience = 800

    for t in range(1, 5000000):
        if time.monotonic() - t_start > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Multi-peak subgradient
        n_peaks = max(3, min(15, n // 100))
        top_idx = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-8.0 * (M_curr - conv[top_idx]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Vectorized gradient computation
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            # (a*a)[k] derivative w.r.t. a[j]:
            # d/d a[j] sum_i a[i]*a[k-i] = a[k-j] + a[j-k] (with bounds)
            # Vectorized: for fixed k, grad[j] += a[k-j] (if valid) + a[j-k] (if valid)
            k_arr = np.arange(n)
            vals = k - k_arr  # k - j for j = 0..n-1
            mask1 = (vals >= 0) & (vals < n)
            grad[mask1] += w * a[vals[mask1]]
            vals2 = k_arr - k  # j - k
            mask2 = (vals2 >= 0) & (vals2 < n)
            grad[mask2] += w * a[vals2[mask2]]

        # Projected gradient step
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm_val / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.015, 3.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        # Occasional random peak selection for diversity
        if t % 100 == 0:
            threshold = M_curr * 0.9
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 1:
                k_rand = candidates[np.random.randint(len(candidates))]
                grad_r = np.zeros(n)
                k_arr = np.arange(n)
                vals = k_rand - k_arr
                mask1 = (vals >= 0) & (vals < n)
                grad_r[mask1] = a[vals[mask1]]
                vals2 = k_arr - k_rand
                mask2 = (vals2 >= 0) & (vals2 < n)
                grad_r[mask2] += a[vals2[mask2]]
                gn = np.linalg.norm(grad_r)
                if gn > 1e-15:
                    a_trial2 = a - 0.5 * step * grad_r / gn * n ** 0.5
                    a_trial2 = np.maximum(a_trial2, 0.0)
                    s2 = a_trial2.sum()
                    if s2 > 1e-15:
                        a_trial2 = a_trial2 * norm_val / s2
                        score2 = fast_autocorr(a_trial2).max()
                        if score2 < best_score - 1e-13:
                            a = a_trial2
                            best_score = score2
                            best_a = a_trial2.copy()
                            patience = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score

            #              - (x-a[j0])^2 * (1 if k==2*j0) [correction for double counting]
            # Wait, this is getting confusing. Let me just use the direct formula.

            # Direct: new_conv[k] - conv[k] = (x - aj) * (a[k-j] if valid) + (x - aj) * (a[k-j] if valid) - [correction if k=2j]
            # The correction: when k=2j, the term a[j]*a[j] = aj^2 becomes x^2, so:
            # delta[2j] = x^2 - aj^2 (not 2*(x-aj)*a[j])
            # For k != 2j: delta[k] = 2*(x-aj)*a[k-j] (if valid)

            # So: new_conv[k] = conv[k] + 2*(x-aj)*c[k] for k != 2j
            #     new_conv[2j] = conv[2j] + (x^2 - aj^2)

            # For optimization: minimize max_k new_conv[k] over x in [0, some_upper]
            # new_conv[k] = conv[k] + 2*(x-aj)*c[k] for k != 2j  (linear in x)
            # new_conv[2j] = conv[2j] + x^2 - aj^2  (quadratic in x)

            # Build the "base" for k != 2j: base[k] = conv[k] - 2*aj*c[k]
            # Then new_conv[k] = base[k] + 2*x*c[k] for k != 2j
            # new_conv[2j] = conv[2j] - aj^2 + x^2 = (conv[2j] - aj^2) + x^2

            base = conv - 2.0 * aj * c
            base[2 * j] = conv[2 * j] - aj * aj  # This is the constant part for the quadratic

            # Objective: max( base[2j] + x^2, max_{k!=2j}(base[k] + 2*x*c[k]) )
            # Both are convex in x >= 0, so the max is convex.

            # Find minimum using bisection on subgradient
            # Subgradient at x: max( 2*x (if quadratic is active), 2*c[k] (if linear k is active) )

            # Simple approach: ternary search on [0, upper]
            upper = max(aj * 3, norm_val / n * 5)
            lo = 0.0
            hi = upper

            # Quick check at x = 0 and x = aj
            def obj_at(x_val):
                vals = base + 2.0 * x_val * c
                vals[2 * j] = base[2 * j] + x_val * x_val
                return vals.max()

            obj_0 = obj_at(0.0)
            obj_aj = obj_at(aj)
            obj_max = obj_at(upper)

            if obj_0 <= obj_aj and obj_0 <= obj_max:
                best_x = 0.0
                best_obj = obj_0
            elif obj_aj <= obj_0 and obj_aj <= obj_max:
                best_x = aj
                best_obj = obj_aj
            else:
                # Golden section search
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(40):
                    m1 = hi - gr * (hi - lo)
                    m2 = lo + gr * (hi - lo)
                    f1 = obj_at(m1)
                    f2 = obj_at(m2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                    if hi - lo < 1e-10:
                        break
                best_x = (lo + hi) / 2.0
                best_obj = obj_at(best_x)

            if best_obj < current_score - 1e-10:
                a[j] = best_x
                s = a.sum()
                if s > 1e-15:
                    a = a * norm_val / s
                    new_conv = fast_autocorr(a)
                    new_score = new_conv.max()
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                        conv = new_conv
                        current_score = new_score
                    else:
                        a[j] = aj
                        a = a * norm_val / a.sum()

        if not improved:
            no_improve_passes += 1
        else:
            no_improve_passes = 0
            if verbose:
                elapsed = time.monotonic() - t_start
                print(f"    CD pass improved ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def spectral_perturb(a, rng, strength=0.1):
    n = len(a)
    norm_val = np.sqrt(2.0 * n)
    A = rfft(a, 2 * n)
    # Perturb the phase and magnitude slightly
    magnitude = np.abs(A)
    phase = np.angle(A)
    mag_noise = 1.0 + strength * rng.standard_normal(len(magnitude))
    mag_noise = np.maximum(mag_noise, 0.5)
    phase_noise = strength * 0.5 * rng.standard_normal(len(phase))
    new_mag = magnitude * mag_noise
    new_phase = phase + phase_noise
    A_new = new_mag * np.exp(1j * new_phase)
    a_new = irfft(A_new, 2 * n)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * norm_val / s


def time_perturb(a, rng, strength=0.05):
    n = len(a)
    norm_val = np.sqrt(2.0 * n)
    noise = 1.0 + strength * rng.standard_normal(n)
    noise = np.maximum(noise, 0.3)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert * norm_val / s


def symmetrize(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def propose_candidate():
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    # Try symmetrization
    a_sym = symmetrize(best_seq)
    s_sym = a_sym.sum()
    if s_sym > 1e-15:
        a_sym_n = a_sym * np.sqrt(2.0 * len(a_sym)) / s_sym
        sc_sym = evaluate_safe(a_sym_n)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym_n.copy()
            print(f"  *** Symmetrization: {sc_sym:.8f}")

    n_base = len(best_seq)
    print(f"Starting optimization: n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, "lp1")

    # === Phase 2: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 120.0)
    if t2 > 20:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: Subgradient, budget={t2:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t2, verbose=True)
        try_update(a_out, "subg1")

    # === Phase 3: LP Nesterov round 2 (from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: LP Round 2, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True, seed=123)
        try_update(a_out, "lp2")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: Coordinate Descent, budget={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t4, verbose=True)
        try_update(a_out, "cd1")

    # === Phase 5: Spectral perturbation + LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 2
    t5_total = min(remaining * 0.25, 250.0)
    t5_each = t5_total / max(1, n_restarts)
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t5 = min(t5_each, remaining - 20)
        if t5 < 30:
            break

        if r == 0:
            pert = spectral_perturb(best_seq, rng, strength=0.08)
            label = "spec_pert"
        else:
            pert = time_perturb(best_seq, rng, strength=0.03)
            label = "time_pert"

        print(f"\n=== Phase 5.{r+1}: {label} + LP, budget={t5:.0f}s ===")
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=False, seed=300 + r)
        try_update(a_out, f"{label}_lp")

        # Quick subgradient after perturbation
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5s = min(30.0, remaining - 10)
            if t5s > 10:
                a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                a_out2, _ = subgradient_polish(a_norm, t5s, verbose=False)
                try_update(a_out2, f"{label}_subg")

    # === Phase 6: Final subgradient + CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        t6a = min(remaining * 0.5, 80.0)
        t6b = min(remaining * 0.4, 80.0)
        print(f"\n=== Phase 6: Final polish, remaining={remaining:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t6a, verbose=False)
        try_update(a_out, "subg_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = coordinate_descent(a_norm, t6b, verbose=False)
            try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
