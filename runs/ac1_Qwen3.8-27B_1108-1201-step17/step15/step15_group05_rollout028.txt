Hmm wait, I'm overcomplicating this. Let me verify the cross-correlation formula.

Standard cross-correlation: `(x ⋆ y)[n] = sum_m x[m+n] * conj(y[m])`

For real signals: `(x ⋆ y)[n] = sum_m x[m+n] * y[m]`

DFT property: `DFT(x ⋆ y) = conj(DFT(x)) * DFT(y)`

So: `c = ifft(conj(fft(w_padded)) * fft(a_padded))`

where `w_padded` has length `fft_size` with `w` in the first `size` positions and zeros elsewhere, and `a_padded` has `a` in the first `n` positions and zeros elsewhere.

Then `c[i] = sum_m w[m+i] * a[m]` for `i = 0, ..., fft_size-1`, but we only care about `i = 0, ..., n-1`.

Let me verify: `c[i] = sum_m w[m+i] * a[m]`. For `m=0,...,n-1` (where `a[m] != 0`), we need `m+i` to be in `[0, size-1]` (where `w[m+i] != 0`). So `c[i] = sum_{m=0}^{n-1} w[m+i] * a[m]` (with `w[k] = 0` for `k >= size`). This is exactly what we want!

So `g[i] = 2 * c[i]` where `c = ifft(conj(fft(w, fft_size)) * fft(a, fft_size)).real[:n]`.

Great, let me now write the complete algorithm.

Actually, I realize I should also think about whether to use `fft` or `rfft`. Since both `w` and `a` are real, I can use `rfft` and then reconstruct. But for the conjugate operation, it's easier to use full `fft`. The cost is about 2x for `rfft` vs `fft`, but it simplifies the code. For n=2000, `fft_size` is about 8192, which is fast either way.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponentiated gradient (mirror descent) with smoothed-max objective
2. FFT-based gradient computation via cross-correlation
3. Beta annealing for adaptive peak focusing
4. Multi-length exploration with interpolation
5. Symmetrization and adaptive step size
6. LP-based refinement for final polish
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the score."""
    try:

def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def symmetrize(a):
    """Symmetrize and normalize."""
    a_sym = 0.5 * (a + a[::-1].copy())
    return normalize(a_sym)

def compute_smoothed_gradient(a, beta):
    """
    Compute the smoothed gradient using softmax weights over the autocorrelation peaks.
    Returns (gradient, b, max_b, log_weights) where gradient[i] = 2 * sum_k w[k] * a[k-i].
    """
    n = len(a)
    b = fast_autocorr(a)
    size = len(b)
    max_b = b.max()
    
    # Compute softmax weights: w[k] = exp(beta * b[k]) / sum_j exp(beta * b[j])
    # Use log-sum-exp for stability
    shifted = beta * b - beta * max_b  # shift for numerical stability
    log_w = shifted - np.log(np.sum(np.exp(shifted)))
    w = np.exp(log_w)
    
    # Compute cross-correlation: c[i] = sum_k w[k] * a[k-i]
    conv_size = size + n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    
    w_pad = np.zeros(fft_size)
    w_pad[:size] = w
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    
    W = fft(w_pad)
    A = fft(a_pad)
    c = ifft(np.conj(W) * A).real[:n]
    
    grad = 2.0 * c
    return grad, b, max_b, w

def exponentiated_gradient_optimize(a0, time_limit, beta_schedule=None, verbose=True, seed=42):

    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize(a0.copy())
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    best_conv = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    eta = 1.0 / n  # initial step size
    eta_min = 0.01 / n
    eta_max = 10.0 / n
    
    # Ergodic average
    erg_a = np.zeros(n)
    erg_count = 0
    erg_best = best_score
    erg_best_a = a.copy()
    
    stall = 0
    stall_limit = 500
    last_improve_iter = 0
    total_iters = 0
    
    # Beta schedule: start smooth, get sharper
    if beta_schedule is None:
        beta_schedule = 'auto'
    
    while time.monotonic() - t0 < time_limit:
        # Determine beta
        if beta_schedule == 'auto':
            progress = min(1.0, total_iters / 5000.0)
            beta = 1.0 + progress * 50.0  # from 1 to 51
        else:
            beta = float(beta_schedule)
        
        # Compute gradient
        grad, b, max_b, w = compute_smoothed_gradient(a, beta)
        
        # Track best
        if max_b < best_score - 1e-10:
            best_score = max_b
            best_a = a.copy()
            stall = 0
            last_improve_iter = total_iters
        else:
            stall += 1
        
        # Ergodic average (exponential moving average)
        erg_a = 0.95 * erg_a + 0.05 * a
        erg_count += 1
        if erg_count % 100 == 0:
            erg_norm = normalize(erg_a)
            erg_max = fast_autocorr(erg_norm).max()
            if erg_max < erg_best - 1e-10:
                erg_best = erg_max
                erg_best_a = erg_norm.copy()
                if erg_best < best_score:
                    best_score = erg_best
                    best_a = erg_best_a.copy()
                    stall = 0
        
        # Exponentiated gradient update
        # a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        scaled = -eta * grad
        # Clip for numerical stability
        scaled = np.clip(scaled, -50, 50)
        log_a_new = np.log(a + 1e-30) + scaled
        log_a_new -= np.max(log_a_new)  # shift for stability
        a_new = np.exp(log_a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check if update helps (simple backtracking)
        new_max = fast_autocorr(a_new).max()
        if new_max < max_b + eta * max_b * 0.1:  # allow some increase
            a = a_new
            # Increase step size if improving
            if new_max < max_b:
                eta = min(eta * 1.01, eta_max)
            else:
                eta = max(eta * 0.99, eta_min)
        else:
            # Rollback and reduce step size
            eta = max(eta * 0.8, eta_min)
            if eta <= eta_min * 1.01:
                # Try a random perturbation
                noise = rng.standard_normal(n) * 0.01 / n
                a = normalize(a + noise)
                eta = eta_min * 5
        
        # Periodic symmetrization
        if total_iters % 500 == 0:
            a_sym = symmetrize(a)
            sym_max = fast_autocorr(a_sym).max()
            if sym_max < max_b:
                a = a_sym
        
        # Restart if stalled
        if stall > stall_limit:
            a = best_a.copy()
            eta = 5.0 / n
            stall = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    EG restart at iter {total_iters} ({elapsed:.0f}s), best_max={best_score:.10f}")

        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0

    # Final comparison with ergodic

        best_a = erg_best_a.copy()
        best_score = erg_best
    
    return best_a, best_score

def lp_best_response_simple(a_fixed, M, active_set, n):
    """
    Simple LP best response without cvxpy (using scipy.optimize.linprog).
    max sum(b) s.t. T @ b <= M, b >= 0
    where T[i,j] = a_fixed[active_set[i] - j] if valid.
    """
    from scipy.optimize import linprog
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    
    # Build constraint matrix T
    T = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[idx, j] = a_fixed[k - j]
    
    # max sum(b) s.t. T @ b <= M, b >= 0
    # linprog minimizes, so negate: min -sum(b)
    c_obj = -np.ones(n)
    A_ub = T
    b_ub = np.full(size, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    
    return np.maximum(res.x, 0)

def lp_refine(a0, time_limit, verbose=True, seed=0):
    """LP-based best response refinement."""
    import cvxpy as cp
    rng = np.random.default_rng(seed)
    n = len(a0)

    best_a = a.copy()

        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    
    # Iterative best response
    tol_sequence = [0.1, 0.05, 0.02, 0.01, 0.005]
    tol_idx = 0
    
    for round_num in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Active set: peaks above threshold
        tol = tol_sequence[min(tol_idx, len(tol_sequence)-1)]
        active = np.where(conv >= M * (1 - tol))[0]
        
        if len(active) < 3:
            active = np.argsort(conv)[-max(3, n//100):]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active), n))
        for idx, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            T[idx, lo:hi] = a[k - np.arange(lo, hi)]
        
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        
        g = b_val / b_sum  # normalize to sum=1
        
        # Check for constraint violations and expand active set
        conv_check = np.convolve(a, g)
        viol = np.where(conv_check > M * (1 + 1e-5))[0]
        if len(viol) > 0 and len(active) + len(viol) < 2*n-1:
            new_active = np.unique(np.concatenate([active, viol]))
            T2 = np.zeros((len(new_active), n))
            for idx, k in enumerate(new_active):
                lo = max(0, k - n + 1)
                hi = min(n, k + 1)
                T2[idx, lo:hi] = a[k - np.arange(lo, hi)]
            constraints2 = [T2 @ b_var <= M]
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints2)
            prob2.solve(solver=solver, verbose=False)
            if prob2.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                b_sum = b_val.sum()
                if b_sum > 1e-15:
                    g = b_val / b_sum
                active = new_active
        
        # Update: convex combination
        t_blend = 0.3
        a_new = (1 - t_blend) * a + t_blend * g
        a_new = normalize(a_new)
        
        new_max = fast_autocorr(a_new).max()
        if new_max < best_score:
            best_score = new_max
            best_a = a_new.copy()
        a = a_new
        
        # Tighten tolerance after some rounds

            tol_idx += 1
        
        if verbose and round_num % 5 == 0:

            print(f"    LP iter={round_num} ({elapsed:.0f}s): max_b={new_max:.10f}, best={best_score:.10f}, tol={tol}")
    
    return best_a, best_score

def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length, maintaining sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return normalize(a_interp)

def cosine_init(n, freq=1.0, depth=0.3):
    """Initialize with a cosine bump."""
    x = np.linspace(0, 1, n)
    a = 1.0 + depth * np.cos(2 * np.pi * freq * x)
    return normalize(a)

def raised_cosine_init(n):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-6)  # avoid exact zeros
    return normalize(a)

def multi_cosine_init(n, freqs=None, depths=None):
    """Multi-frequency cosine initialization."""
    if freqs is None:
        freqs = [1, 2, 3]
    if depths is None:
        depths = [0.3, 0.2, 0.1]
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for f, d in zip(freqs, depths):
        a = a + d * np.cos(2 * np.pi * f * x)
    return normalize(a)
