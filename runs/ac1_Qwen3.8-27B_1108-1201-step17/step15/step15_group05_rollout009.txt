def fast_autocorr(a):

    n = len(a)

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def symmetrize(a):
    """Symmetrize: a = (a + reverse(a))/2."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1].copy())
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def build_constraint_matrix(a, k_indices, n):
    """Build A matrix where A[idx, j] = a[k-j] if 0<=k-j<n."""
    m = len(k_indices)
    A = np.zeros((m, n), dtype=np.float64)
    for idx in range(m):
        k = k_indices[idx]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            A[idx, lo:hi] = a[k - np.arange(lo, hi)]
    return A


def solve_lp_best_response(a_fixed, M, k_indices, n, solver):
    """
    Solve: max sum(b) s.t. (a_fixed * b)[k] <= M for k in k_indices, b >= 0.
    Returns (b, duals) or (None, None).
    """
    if len(k_indices) == 0:
        return np.ones(n) / n, None

    A = build_constraint_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = b_var.value
    if b_val is None:
        return None, None
    b_val = np.maximum(np.asarray(b_val, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        d_val = constraints[0].dual_value
        if d_val is not None:
            duals = np.asarray(d_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

    active = np.where(conv >= M * (1.0 - tol))[0]
    n_top = max(3, min(top_k_extra, n // 100))
    top_idx = np.argsort(conv)[-n_top:]
    active = np.unique(np.concatenate([active, top_idx]))
    return active


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """
    Nesterov-accelerated LP best-response with progressive active set tightening.
    Key improvements:
    - Adaptive tolerance that tightens geometrically
    - Periodic full-constraint verification
    - Momentum decay near convergence
    - Dual-guided constraint expansion
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    active_set = None
    stall_count = 0
    stall_threshold = 60
    full_check_interval = 30
    tol = 0.15  # Start loose

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        elapsed = time.monotonic() - t0

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Progressive tightening: reduce tolerance over iterations
        if t % 20 == 0:
            tol = max(0.005, 0.15 * np.exp(-t / 200.0))

        # Update active set periodically
        need_new_active = (active_set is None) or (t % 30 == 0)
        if need_new_active:
            active_set = select_active_set(conv_y, M_curr, tol, n)

        # Solve LP
        b, duals = solve_lp_best_response(y, M_curr, active_set, n, solver)

        if b is None:
            # Fallback: try with more constraints
            if len(active_set) < size:
                all_k = np.arange(size)
                b, duals = solve_lp_best_response(y, M_curr, all_k, n, solver)
                if b is not None:
                    active_set = all_k
            if b is None:
                continue

        # Check for constraint violations and expand active set
        if len(active_set) < size:
            complement = np.setdiff1d(np.arange(size), active_set)
            if len(complement) > 0:
                A_comp = build_constraint_matrix(y, complement, n)
                violations = A_comp @ b
                viol_mask = violations > M_curr * (1 + 1e-5)
                if np.any(viol_mask):
                    new_active = np.union1d(active_set, complement[viol_mask])
                    if len(new_active) > 0.85 * size:
                        new_active = np.arange(size)
                    b_re, _ = solve_lp_best_response(y, M_curr, new_active, n, solver)
                    if b_re is not None:
                        active_set = new_active
                        b = b_re

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        g = b * norm / b_sum

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum (decays as we converge)
        if t <= 2:
            a_new = y_new.copy()
        else:
            # Reduce momentum when improvements are small
            improvement = best_score - fast_autocorr(y_new).max()
            if improvement < 1e-6:
                momentum = min((t - 1.0) / (t + 2.0), 0.9)
            else:
                momentum = min((t - 1.0) / (t + 2.0), 0.97)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best (check y, a, and ergodic average)
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Every 50 iterations, check ergodic average
        if t % 50 == 0:
            # Compute a running average from recent iterates (approximation)
            erg_trial = a.copy()
            s_erg = erg_trial.sum()
            if s_erg > 1e-15:
                erg_trial = erg_trial * norm / s_erg
                score_erg = fast_autocorr(erg_trial).max()
                if score_erg < best_score - 1e-14:
                    best_score = score_erg
                    best_a = erg_trial.copy()
                    stall_count = 0

        # Full-constraint verification every full_check_interval
        if t % full_check_interval == 0:
            all_k = np.arange(size)
            b_full, _ = solve_lp_best_response(y, M_curr, all_k, n, solver)
            if b_full is not None:
                b_full_sum = b_full.sum()
                if b_full_sum > 1e-15:
                    g_full = b_full * norm / b_full_sum
                    # Try this as a candidate
                    score_full = fast_autocorr(g_full).max()
                    if score_full < best_score - 1e-12:
                        best_score = score_full
                        best_a = g_full.copy()
                        y = g_full.copy()
                        y_prev = g_full.copy()
                        a = g_full.copy()
                        active_set = all_k
                        stall_count = 0
                        if verbose:
                            print(f"      Full-constraint improvement at t={t}: {best_score:.10f}")

        # Convergence check: if LP solution equals current y (within tolerance)
        if t > 100:
            diff = np.max(np.abs(g - y)) / max(norm, 1e-15)
            if diff < 1e-8:
                if verbose:
                    print(f"      Convergence detected at t={t} ({elapsed:.0f}s), score={best_score:.10f}")
                break

        # Restart if stalled
        if stall_count > stall_threshold:
            if verbose:
                print(f"    LP restart at t={t} ({elapsed:.0f}s), score={best_score:.10f}")
            # Structured restart: small perturbation of best
            pert_scale = 0.01 * np.sqrt(stall_count / stall_threshold)
            noise = 1.0 + pert_scale * rng.standard_normal(n)
            y = np.maximum(best_a * noise, 0.0)
            s = y.sum()
            if s > 1e-15:
                y = y * norm / s
            y_prev = y.copy()
            a = y.copy()
            active_set = None
            stall_count = 0
            tol = max(tol, 0.1)  # Reset tolerance to be looser

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent with bisection line search.
    For each coordinate j, optimize a[j] while keeping others fixed,
    using the fact that the objective in a single coordinate is 1-D.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    n_passes = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        current_score = fast_autocorr(a).max()

        # Process coordinates in random order for diversity
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = a.sum() - aj

            # Compute cross terms: c[k] = sum of a[i]*a[k-i] terms involving a[j]
            # (a*a)[k] = bg[k] + 2*aj*c[k] + aj^2*delta[k,2j]
            # c[k] = sum of a[i] where the term involves a[j]
            c = np.zeros(size)
            # Terms: a[j]*a[k-j] for valid k-j, and a[k-j]*a[j] for valid k-j
            # Actually: (a*a)[k] = sum_i a[i]*a[k-i]
            # Terms involving a[j]: i=j gives a[j]*a[k-j] (if 0<=k-j<n)
            #                    k-i=j gives a[i]*a[j] = a[k-j]*a[j] (if 0<=k-j<n)
            # So contribution of a[j] to C[k] is: a[j]*(a[k-j] + a[k-j]) if both valid
            # Wait, need to be more careful:
            # C[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (with a[m]=0 if m<0 or m>=n)
            # Terms where i=j: a[j] * a[k-j] (if 0<=k-j<n)
            # Terms where k-i=j, i.e., i=k-j: a[k-j] * a[j] (if 0<=k-j<n)
            # If i=j and k-i=j simultaneously, i.e., k=2j: only one term a[j]*a[j]
            # So: contribution of varying a[j] to C[k]:
            #   If k != 2j: a[j] * a[k-j] + a[j] * a[k-j] (if both indices valid)
            #   If k == 2j: a[j]^2
            #   Plus: if k-j valid and k-j != j: a[j] * a[k-j]
            #   And: if j valid (always) and k-j valid: a[k-j] * a[j]
            # Actually let me just compute directly:
            # C[k] = bg[k] + a[j] * (a[k-j] if valid) + a[k-j] * a[j] if valid
            # where bg[k] = C[k] - terms involving a[j]

            # Better approach: just compute C[k] for the current a, then vary a[j]
            conv = fast_autocorr(a)

            # For varying a[j] to x:
            # C_new[k] = conv[k] - contribution_old + contribution_new
            # contribution of a[j] to C[k]:
            #   if 0 <= k-j < n: a[j] * a[k-j]  (term where first index is j)
            #   if 0 <= k-j < n: a[k-j] * a[j]  (term where second index is j, i.e., k-i=j)
            #   These are the same term when i=j and k-i=j, i.e., k=2j
            # So: if 0 <= k-j < n and k-j != j: contribution = 2*a[j]*a[k-j]
            #     if k == 2j (so k-j=j): contribution = a[j]^2
            #     if not (0 <= k-j < n): contribution = 0
            # Wait, I need to be more careful. C[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
            # The terms involving a[j]:
            #   i = j: term is a[j]*a[k-j] if 0<=k-j<n
            #   k-i = j, i.e., i = k-j: term is a[k-j]*a[j] if 0<=k-j<n
            # If j = k-j (i.e., k=2j), these are the same term, so it's just a[j]^2
            # Otherwise, they're different terms and contribution is 2*a[j]*a[k-j]... NO!
            # They're a[j]*a[k-j] and a[k-j]*a[j], which are the same scalar!
            # So total contribution = a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] if k != 2j
            # And = a[j]^2 if k = 2j

            # Let me define: for each k, the "sensitivity" of C[k] to a[j]
            # dC[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n) ... 
            # No, let's just do it numerically:
            # C_new[k] when a[j] changes from aj to x:
            #   delta = x - aj
            #   C_new[k] = C[k] + delta * (a[k-j] if 0<=k-j<n and k-j!=j else 0)
            #                    + delta * (a[k-j] if 0<=k-j<n and k-j!=j else 0)  
            #                    + 2*delta*a[j] if k==2j ... this is getting confusing

            # Simplest approach: just recompute the relevant part
            # C[k] = sum_i a[i]*a[k-i]
            # If I change a[j] from aj to x, the change in C[k] is:
            #   (x-aj) * a[k-j]  (if 0<=k-j<n)  [from the i=j term]
            #   + a[k-j] * (x-aj)  (if 0<=k-j<n)  [from the k-i=j term, i.e., i=k-j]
            #   - if both are the same term (j=k-j, i.e., k=2j), count only once
            # So: delta_C[k] = (x-aj) * (a[k-j] if 0<=k-j<n) + (a[k-j] if 0<=k-j<n) * (x-aj)
            #              = 2*(x-aj)*a[k-j] if 0<=k-j<n and k!=2j
            #              = (x-aj)*a[j] if k==2j  [wait, this isn't right either]

            # Let me just be explicit:
            # C_new[k] = C[k] - aj*a[k-j] (if valid) - a[k-j]*aj (if valid and different term)
            #           + x*a[k-j] (if valid) + a[k-j]*x (if valid and different term)
            # Where "valid" means 0<=k-j<n
            # And "different term" means k-j != j (otherwise it's the same term counted once)

            # OK let me just compute it directly for the bisection:
            def score_for_x(x):
                # Compute max of C_new
                # C_new[k] = C[k] + (x-aj) * sensitivity[k]
                # where sensitivity[k] = a[k-j] (if 0<=k-j<n and k-j!=j) + 
                #                        a[k-j] (if 0<=k-j<n and k-j!=j)
                #                        ... ugh, let me just compute differently
                # C[k] = sum_i a[i]*a[k-i]
                # If I change a[j]:
                #   C_new[k] = C[k] + (x - aj) * [a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n, k-j!=j)]
                #   Wait no. Let me enumerate:
                #   C_new[k] = sum_i a'[i]*a'[k-i] where a'[j]=x, a'[i]=a[i] for i!=j
                #   = sum_{i!=j} a[i]*a'[k-i] + a'[j]*a'[k-j]
                #   If k-j != j and 0<=k-j<n:
                #     = sum_{i!=j, i!=k-j} a[i]*a[k-i] + a[k-j]*x + x*a[k-j]
                #     = C[k] - a[j]*a[k-j] - a[k-j]*a[j] + x*a[k-j] + a[k-j]*x
                #     = C[k] - 2*aj*a[k-j] + 2*x*a[k-j]
                #     = C[k] + 2*(x-aj)*a[k-j]
                #   If k-j == j (i.e., k=2j) and 0<=j<n (always true):
                #     = sum_{i!=j} a[i]*a[k-i] + x*x
                #     = C[k] - aj*aj + x*x
                #     = C[k] - aj^2 + x^2
                #   If not (0<=k-j<n):
                #     C_new[k] = C[k] (no change, a[j] doesn't appear)
                #     Wait, that's not right either. If k-j is out of range, then a[j]
                #     doesn't contribute to C[k] at all, so C_new[k] = C[k].
                #     But wait, a[j] could appear as a[k-i] when i=k-j...
                #     If k-j < 0 or k-j >= n, then a[k-j] = 0, so a[j] doesn't appear.
                #     Correct: if not (0<=k-j<n), C_new[k] = C[k].

                # So:
                # sensitivity[k] = 2*a[k-j] if (0<=k-j<n and k-j!=j)
                #                    2*aj if (k==2j)  [because -aj^2 + x^2 = (x-aj)(x+aj), at x=aj gives 2aj*0... no]
                # Let me just use the formula directly:
                # C_new[k] = C[k] + delta * S[k] + (delta^2 if k==2j)  where delta = x-aj
                # Actually no, for k=2j: C_new[2j] = C[2j] - aj^2 + x^2 = C[2j] + (x^2 - aj^2) = C[2j] + (x-aj)(x+aj)
                # This is NOT linear in x! It's quadratic.
                # For k != 2j: C_new[k] = C[k] + 2*(x-aj)*a[k-j] (linear in x)
                # So the max of C_new[k] over k is the max of:
                #   - linear functions in x (for k != 2j)
                #   - a quadratic in x (for k = 2j)
                # This makes the 1D subproblem a min-max of linear + quadratic functions.
                # The optimum is where the active constraints intersect.

                # For simplicity in bisection, let's just evaluate:
                new_conv = conv.copy()
                delta = x - aj
                # Apply changes
                if 0 <= k_valid_lo and k_valid_lo < size:
                    pass
                # Efficient update:
                # For all k where 0 <= k-j < n:
                #   if k-j != j: C_new[k] += 2*delta*a[k-j]
                #   if k-j == j (k=2j): C_new[k] += delta*(x+aj) = delta*delta + 2*delta*aj
                # Actually C_new[2j] = C[2j] + delta*(x + aj) ... no
                # C_new[2j] = C[2j] - aj*aj + x*x = C[2j] + x^2 - aj^2 = C[2j] + (x-aj)(x+aj)
                # = C[2j] + delta * (2*aj + delta)

                # Let me just build the sensitivity vector:
                # S[k] such that C_new[k] = C[k] + delta*S[k] + (delta^2 if k==2j)
                # For k where 0<=k-j<n and k!=2j: S[k] = 2*a[k-j]
                # For k=2j: C_new[2j] = C[2j] + delta^2 + 2*delta*aj, so S[2j] = 2*aj
                # For other k: S[k] = 0

                # So: C_new[k] = C[k] + delta*S[k] + (delta^2 if k==2j else 0)
                # max_k C_new[k] = max_k (C[k] + delta*S[k] + delta^2 * [k==2j])
                # = max( max_{k!=2j}(C[k] + delta*S[k]), C[2j] + delta*S[2j] + delta^2 )
                # = max( max over linear terms, quadratic term )

                # This is a convex function of delta (max of convex functions).
                # We can find minimum by ternary search or bisection on subgradient.

                # For now, let's just evaluate the score:
                new_conv = conv + delta * S_vec  # S_vec precomputed
                new_conv[2 * j] += delta * delta
                new_sum = s_others + x
                if new_sum < 1e-15:
                    return float('inf')
                return 2.0 * n * new_conv.max() / (new_sum * new_sum)

            # Precompute S_vec (sensitivity)
            S_vec = np.zeros(size)
            lo_k = max(0, j)  # k-j >= 0 => k >= j
            hi_k = min(size, j + n)  # k-j < n => k < j+n
            for k in range(lo_k, hi_k):
                idx = k - j  # = k-j, the index into a
                if idx != j:
                    S_vec[k] = 2.0 * a[idx]
                else:
                    S_vec[k] = 2.0 * aj  # for k=2j

            # Bisection/ternary search for optimal x in [0, 2*norm]
            # The objective is convex in x (it's max of linear/quadratic functions divided by (s_others+x)^2)
            # Actually it's not obviously convex due to the denominator. Let's just do a grid + refinement.
            
            # Quick grid search
            x_candidates = [0.0, aj * 0.3, aj * 0.6, aj * 1.2, aj * 1.5, 2*aj,
                           s_others / (n - 1) if n > 1 else 0, norm / n]
            x_candidates = [x for x in x_candidates if 0 <= x <= 2 * norm]
            if len(x_candidates) == 0:
                x_candidates = [aj]

            best_x = aj
            best_obj = score_for_x(aj)

            for x in x_candidates:
                obj = score_for_x(x)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            # Refine with bisection around best_x
            if best_x != aj and abs(best_x - aj) > 1e-10:
                lo_b = max(0, min(aj, best_x) - abs(best_x - aj) * 0.5)
                hi_b = min(2 * norm, max(aj, best_x) + abs(best_x - aj) * 0.5)
                # Golden section search
                gr = (np.sqrt(5) - 1) / 2
                x1 = hi_b - gr * (hi_b - lo_b)
                x2 = lo_b + gr * (hi_b - lo_b)
                f1 = score_for_x(x1)
                f2 = score_for_x(x2)
                for _ in range(15):
                    if f1 < f2:
                        hi_b = x2
                        x2 = x1
                        f2 = f1
                        x1 = hi_b - gr * (hi_b - lo_b)
                        f1 = score_for_x(x1)
                    else:
                        lo_b = x1
                        x1 = x2
                        f1 = f2
                        x2 = lo_b + gr * (hi_b - lo_b)
                        f2 = score_for_x(x2)
                best_x = (lo_b + hi_b) / 2
                best_obj = score_for_x(best_x)

            if abs(best_x - aj) > 1e-12 and best_obj < best_score - 1e-13:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm / a.sum()

        if not improved:
            no_improve_count += 1
            if no_improve_count >= 3:
                break
        else:
            no_improve_count = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_and_optimize(a0, rng, pert_scale, time_limit, solver, verbose=False):
    """Perturb the sequence and run LP optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    noise = 1.0 + pert_scale * rng.standard_normal(n)
    a_pert = np.maximum(a0 * noise, 0.0)
    s = a_pert.sum()
    if s < 1e-15:
        return a0.copy(), float('inf')
    a_pert = a_pert * norm / s
    return nesterov_lp_optimize(a_pert, time_limit, solver, verbose=verbose)

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"  Could not load parent: {e}")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    # Quick symmetrization check
    a_sym = symmetrize(best_seq)
    s_sym = a_sym.sum()
    if s_sym > 1e-15:
        sc_sym = evaluate_safe(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.copy()
            print(f"  Symmetrization: {sc_sym:.8f}")

    n_base = len(best_seq)
    print(f"Starting with n={n_base}, score={best_score:.8f}")

    # === Phase 1: Main LP optimization on base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multi-length exploration, remaining={remaining:.0f}s ===")
    if remaining > 200:
        # Try a few nearby lengths
        candidate_lengths = []
        for dl in [-200, -100, 100, 200, 300]:
            nl = n_base + dl
            if 500 <= nl <= 3000 and nl != n_base:
                candidate_lengths.append(nl)
        # Limit to 3 lengths
        candidate_lengths = candidate_lengths[:3]

        t_per_len = min(remaining * 0.25 / max(len(candidate_lengths), 1), 100.0)
        for nl in candidate_lengths:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t_alloc = min(t_per_len, remaining - 30)
            if t_alloc < 30:
                break
            print(f"  Trying n={nl}, budget={t_alloc:.0f}s")
            a_interp = interpolate_sequence(best_seq, nl)
            a_norm = a_interp * np.sqrt(2.0 * nl) / a_interp.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t_alloc, solver, verbose=False, seed=nl)
            try_update(a_out, f"len{nl}")

    # === Phase 3: LP with full constraints on best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP full-constraints n={n_cur}, remaining={remaining:.0f}s ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=100)
        try_update(a_out, f"lp3_n{n_cur}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Coordinate descent n={n_cur}, remaining={remaining:.0f}s ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t4, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 5: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Perturbation restarts n={n_cur}, remaining={remaining:.0f}s ===")
    if remaining > 100:
        n_restarts = 3
        t_each = min(remaining * 0.20 / n_restarts, 80.0)
        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t_alloc = min(t_each, remaining - 30)
            if t_alloc < 20:
                break
            pert_scale = 0.015 + 0.01 * r
            a_out, _ = perturb_and_optimize(best_seq, rng, pert_scale, t_alloc, solver,
                                           verbose=False)
            try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 6: Final LP + CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_cur}, remaining={remaining:.0f}s ===")
    if remaining > 60:
        t6_lp = min(remaining * 0.5, 120.0)
        if t6_lp > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t6_lp, solver, verbose=False, seed=999)
            try_update(a_out, f"lp6_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6_cd = min(remaining * 0.6, 80.0)
            a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out, _ = coordinate_descent(a_norm, t6_cd, verbose=False)
            try_update(a_out, f"cd6_n{len(best_seq)}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        fb = np.ones(100) / 100.0
        return [float(x) for x in fb]

    return [float(x) for x in best_seq]
