import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

def lp_solve(a, M, solver, active=None):

    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

    """Build constraint matrix for lags 0..n-1 only (exploiting autocorrelation symmetry)."""
    n = len(a)
    T = np.zeros((n, n))
    for k in range(n):
        for i in range(k + 1):
            T[k, i] = a[k - i]
    return T

    """Solve LP with only n constraints (exploiting symmetry of autocorrelation)."""
    n = len(a)

    if prob.status not in ('optimal', 'optimal_inaccurate'):

            continue
        g = b * norm / b_sum

        # Verify constraints (only check if we used screening)
        if active is not None:
            if use_reduced:
                check = fast_conv_cross(y, g)[:n]
            else:
                check = fast_conv_cross(y, g)
            if check.max() > M * 1.005:
                b_full = lp_fn(y, M, solver, None)
                if b_full is not None:
                    b_sum2 = b_full.sum()
                    if b_sum2 > 1e-15:
                        b = b_full
                        g = b * norm / b_sum2

        # Nesterov update with adaptive restart
        if t % restart_every == 0:
            # Restart momentum periodically to escape oscillation
            y = a.copy()
            alpha = 1.0 / np.sqrt(t)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve_count = 0
            plateau_score = best_score
        else:
            no_improve_count += 1
            # If stuck for too long, try a perturbation
            if no_improve_count > 1000 and t % 500 == 0:
                rng_p = np.random.default_rng(t)
                n_pert = max(10, n // 50)
                idx_pert = rng_p.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.05 * rng_p.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * norm / s_a
                y = a.copy()
                no_improve_count = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def proximal_gradient_lse(a0, time_limit, verbose=True):
    """
    Proximal gradient method on the LSE-smoothed autocorrelation objective.
    Uses adaptive temperature and backtracking step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # Normalize to sum = 1

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Temperature schedule: start high, decrease gradually
    n_stages = 25
    T_start = 30.0
    T_end = 0.3
    temps = np.geomspace(T_start, T_end, n_stages)

    step = 1.0 / (2.0 * np.sqrt(n))

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        # Multiple iterations per temperature
        n_iters = max(20, min(200, int((time_limit - (time.monotonic() - t0)) * 3)))

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()

            # LSE weights
            shifted = (conv - max_c) / T
            shifted = np.maximum(shifted, -100.0)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(size)
                w[np.argmax(conv)] = 1.0
            else:
                w = w / w_sum

            # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
            # This is 2 * corr(a, w) evaluated at appropriate lags
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # cross-correlation: sum_k w[k] * a[k-i] = sum_k w[k] * a[k-i]
            # = (w conv a) evaluated at... let's think.
            # We want g[i] = sum_k w[k] * a[k-i] for i in [0, n-1]
            # This is the cross-correlation of w and a at lag i.
            # In FFT: IFFT(conj(FFT(w)) * FFT(a)) gives... 
            # Actually: (w * a)[m] = sum_k w[k]*a[m-k] (convolution)
            # We want sum_k w[k]*a[k-i] = sum_k w[k]*a[-(i-k)] = (w * a_reversed)[...]
            # Simpler: just compute it directly as a convolution.
            # g[i] = sum_k w[k]*a[k-i] where 0<=k-i<n
            # = sum_j w[i+j]*a[j] where j in [0,n-1] and i+j in [0, size-1]
            # This is the cross-correlation, which is:
            # g = ifft(conj(fft(w)) * fft(a)) but with appropriate indexing.
            
            # Let's just use the direct formula via cross-convolution:
            # g[i] = sum_k w[k] * a[k-i]
            # = (a cross-correlate w)[i]
            # = sum_k a[k] * w[k+i] ... no.
            # 
            # Let's define: g[i] = sum_k w[k] * a[k-i] for k where 0<=k<i+n... 
            # Actually for |k-i| < n: a[k-i] is valid.
            # g[i] = sum_{k: 0<=k<size, 0<=k-i<n} w[k]*a[k-i]
            # Let j = k-i: g[i] = sum_{j: 0<=j<n, 0<=j+i<size} w[j+i]*a[j]
            # = sum_{j=0}^{n-1} w[j+i]*a[j] (since i>=0 and j<n, j+i < n+i <= n+size... )
            # For i in [0, n-1] and j in [0, n-1]: j+i in [0, 2n-2] = [0, size-1]. ✓
            # So g[i] = sum_{j=0}^{n-1} a[j] * w[j+i]
            # This is the cross-correlation of a and w at lag i.
            # In FFT terms: this is IFFT(FFT(a) * conj(FFT(w))) evaluated at... 
            # Actually, cross-correlation r[i] = sum_j a[j]*w[j+i] = (a * w_flipped)[n-1+i]
            # or more simply, we can compute it as:
            # r = ifft(conj(fft(a)) * fft(w)) ... no that's not right either.
            #
            # Let me just use the convolution directly:
            # sum_j a[j]*w[j+i] for i=0,...,n-1
            # = conv(a, w)[n-1+i] ... let's check.
            # conv(a,w)[m] = sum_j a[j]*w[m-j]
            # We want sum_j a[j]*w[j+i] = sum_j a[j]*w[(j+i)]
            # Let m = j+i, j = m-i: sum_m a[m-i]*w[m] = conv(w,a)[i] (if we define conv properly)
            # Actually: (w*a)[i] = sum_j w[j]*a[i-j] = sum_j w[j+i-a... 
            # 
            # Simplest: g[i] = sum_j a[j]*w[j+i]. This is a "strand" of the outer product.
            # We can compute all g[i] for i=0..n-1 via:
            # Create w padded to length size, then for each i, g[i] = dot(a, w[i:i+n])
            # This is O(n^2) total, which for n=1600 is 2.5M ops - fast enough.
            # OR use FFT: g = ifft(fft(a, 2*size) * fft(w, 2*size)) to get cross-correlation.
            
            # Let me use the efficient FFT approach:
            # corr[i] = sum_j a[j]*w[j+i] for i = 0, 1, ..., size-1
            # This equals ifft(fft(a) * conj(fft(w)))... let me verify.
            # ifft(fft(a)*conj(fft(w)))[i] = sum_j a[j]*w[j-i] (with circular indexing)
            # That's not quite what we want. We want w[j+i] not w[j-i].
            # 
            # Alternative: corr[i] = sum_j a[j]*w[j+i]
            # = sum_j a[j]*w_flipped_size[-(j+i) mod 2*size]... this is getting complicated.
            #
            # Let me just compute it the direct way for clarity, using matrix multiply:
            # g[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
            # This is: g = A @ w_vector where A is the n x n matrix with A[i,j] = w[i+j]
            # But that's O(n^2) to build and O(n^2) to multiply.
            #
            # Better: use the convolution trick.
            # Define a_pad = [a, zeros(size)] of length 2*size
            # Define w_pad = [zeros(n-1), w] of length 2*size
            # Then conv(a_pad, w_pad)[i+n-1] = sum_j a_pad[j]*w_pad[i+n-1-j]
            # = sum_{j=0}^{n-1} a[j] * w_pad[i+n-1-j]
            # w_pad[i+n-1-j] = w[i+n-1-j-(n-1)] = w[i-j] if i-j is in [0, size-1]
            # Hmm, this doesn't simplify nicely.
            
            # Let me just use the naive O(n^2) approach since n=1600 means 2.5M ops per gradient eval.
            # Actually, let me use a different trick:
            # g[i] = sum_j a[j]*w[i+j]
            # Consider the (n-1) x (n-1) matrix M where M[i,j] = w[i+1+j] for i,j in [0, n-2].
            # Then g[i] for i in [0, n-2] = sum_j a[j]*w[i+j] = a @ w[i:i+n]
            # And g[n-1] = sum_j a[j]*w[n-1+j] = a @ w[n-1:2*n-1]
            
            # Most efficient: use rfft/matmul approach
            # g[i] = <a, w[i:i+n]> for i = 0, ..., n-1
            # This is the "short cross-correlation" which can be done with one FFT:
            # Take FFT of a (length n) and w (length size), compute cross-spectrum, IFFT.
            
            # Actually the simplest correct approach:
            # g = np.array([np.dot(a, w[i:i+n]) for i in range(n)])
            # For n=1600, this is 1600 dot products of length 1600 = 2.56M multiply-adds.
            # In numpy this should take ~1-2 microseconds per dot, so ~3ms total. Fast enough!
            
            # But even faster: use the Toeplitz structure.
            # g[i] = sum_j a[j]*w[i+j] = (a^T) @ (toeplitz matrix) @ ... 
            # Or just: g = np.array([np.dot(a, w[i:i+n]) for i in range(n)])
            
            # Fastest: use convolution
            # sum_j a[j]*w[i+j] = (a * flip(w))[n-1-i+size-1]... 
            # Let me just use the direct approach with matmul for speed:
            # Build the Hankel matrix H where H[i,j] = w[i+j], shape (n, n)
            # Then g = a @ H  (row vector a @ n x n matrix = n-vector)
            # H can be built with: H = w[0:2*n-1] in a Hankel pattern.
            # Actually: H[i,j] = w[i+j] for i,j in [0,n-1]. This is a Hankel matrix.
            # We can build it as: H = np.array([w[i:i+n] for i in range(n)])
            # Then g = a @ H (or equivalently g[i] = dot(a, w[i:i+n]))
            
            # For n=1600, building H is O(n^2) = 2.5M elements, which is fine.
            # a @ H is O(n^2) = 2.5M multiply-adds.
            # Total: ~5M ops per gradient evaluation. At ~1ns/op, that's ~5ms.
            # Over 5000 iterations, that's 25 seconds. Fine.
            
            # Actually, even simpler: use the fact that g[i] = sum_j a[j]*w[i+j]
            # = sum_j a[j]*w[i+j]
            # If we define the convolution c = conv(a, w_reversed) where w_reversed = w[::-1],
            # then c[k] = sum_j a[j]*w_reversed[k-j] = sum_j a[j]*w[size-1-(k-j)] = sum_j a[j]*w[size-1-k+j]
            # We want g[i] = sum_j a[j]*w[i+j]. Let size-1-k = i, so k = size-1-i.
            # c[size-1-i] = sum_j a[j]*w[i+j]. ✓ (need to check bounds)
            
            # So: g = conv(a, w[::-1])[size-1 : 2*size-n] ... let me just do it directly.
            
            w_rev = w[::-1]
            fft_sz = 1 << (n + size - 1).bit_length()
            A_full = rfft(np.concatenate([a, np.zeros(size)]), fft_sz)
            WR_full = rfft(w_rev, fft_sz)
            corr = irfft(A_full * WR_full, fft_sz)
            # corr[k] = sum_j a[j] * w_rev[k-j] = sum_j a[j] * w[size - 1 - (k-j)] = sum_j a[j] * w[size - 1 - k + j]
            # We want g[i] = sum_j a[j] * w[i + j]
            # So size - 1 - k + j = i + j => size - 1 - k = i => k = size - 1 - i
            # For i in [0, n-1]: k in [size-1-n+1, size-1] = [n-1, size-1]
            # corr[k] for k in [n-1, size-1]: need k-j in [0, len(w_rev)-1] = [0, size-1]
            # k = size-1-i, j in [0, n-1]: k-j = size-1-i-j. For this to be in [0, size-1]:
            # size-1-i-j >= 0 => j <= size-1-i. Since j < n and size-1-i >= size-1-(n-1) = n, 
            # this is always satisfied. And size-1-i-j <= size-1. ✓
            # So g[i] = corr[size-1-i]
            g_vec = corr[n-1 : size][::-1]  # g[i] = corr[size-1-i], i=0..n-1
            # Wait: g[0] = corr[size-1], g[1] = corr[size-2], ..., g[n-1] = corr[n-1]
            # So g_vec = corr[n-1:size][::-1] gives [corr[size-1], ..., corr[n-1]] = [g[0], ..., g[n-1]] ✓
            
            grad = 2.0 * g_vec

            # Projected gradient step
            # a_new = proj_simplex(a - step * grad)
            raw = a - step * grad
            # Projection onto simplex {x >= 0, sum(x) = 1}
            # Use the standard algorithm
            sorted_raw = np.sort(raw)[::-1]
            cumulative = np.cumsum(sorted_raw)
            rho = np.arange(1, n + 1)
            mask = (sorted_raw + (1.0 - cumulative) / rho > 0)
            if not mask.any():
                lambda_val = raw.max() - 1.0 / n
            else:
                rho_star = rho[mask][-1]
                lambda_val = (cumulative[mask][-1] - 1.0) / rho_star
            a_new = np.maximum(raw - lambda_val, 0.0)
            a_new = a_new / a_new.sum()

            # Backtracking: check if LSE objective decreased
            conv_new = fast_conv_self(a_new)
            max_new = conv_new.max()
            # For simplicity, just accept the step (LSE is convex so gradient descent should work)
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ProxGrad LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smoothed gradient descent with Nesterov momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(60.0, 0.3, n_stages)
    v = np.zeros(n)  # momentum
    beta = 0.9

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        n_steps = max(20, min(300, int(remaining * 4)))
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()

            shifted = (conv - max_c) / T
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient via cross-correlation
            w_rev = w[::-1]
            fft_sz = 1 << (n + size - 1).bit_length()
            A_full = rfft(np.concatenate([a, np.zeros(size)]), fft_sz)
            WR_full = rfft(w_rev, fft_sz)
            corr = irfft(A_full * WR_full, fft_sz)
            g_vec = corr[n-1 : size][::-1]
            grad = 2.0 * g_vec

            # Adaptive step size
            step = norm / (T * np.sqrt(n) * (1.0 + 0.02 * it))

            v = beta * v - step * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE GD stage {si} T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def coordinate_refine(a0, time_limit, rng, verbose=True):
    """
    Pair coordinate descent: for each pair (i,j), find optimal mass transfer.
    Uses the autocorrelation structure to guide candidate selection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Identify peak lags
        peak_thresh = cur_max * 0.93
        peak_lags = np.where(conv[:n] >= peak_thresh)[0]

        # Compute coordinate pressure (contribution to peaks)
        pressure = np.zeros(n)
        for kl in peak_lags[:min(30, len(peak_lags))]:
            lo = max(0, kl - n + 1)
            hi = min(kl + 1, n)
            for i in range(lo, hi):
                q = kl - i
                if 0 <= q < n:
                    pressure[i] += a[q]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue

            # Candidate j's: low pressure coordinates + structured candidates from peaks
            j_cands = set()
            # Low pressure candidates
            low_p = np.argsort(pressure)[:min(20, n // 10)]
            for j in low_p:
                if j != i:
                    j_cands.add(int(j))
            # Structured from peaks
            for kl in peak_lags[:5]:
                j1 = (kl - i + n) % n
                j2 = (kl + i) % n
                if j1 != i:
                    j_cands.add(int(j1))
                if j
