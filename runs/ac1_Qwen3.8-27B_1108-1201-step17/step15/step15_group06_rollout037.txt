def lse_gradient(a, tau):
    """
    Compute gradient of log(sum(exp(conv(a,a)/tau))) with respect to a.
    Returns the gradient vector of same length as a.
    """
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1

    max_b = b.max()
    probs = np.exp((b - max_b) / tau)
    sum_probs = probs.sum()
    if sum_probs < 1e-300:

        probs[b.argmax()] = 1.0
    else:
        probs /= sum_probs

    # grad[i] = c[2n-2-i] for i=0..n-1
    grad = c[n-1:2*n-1][::-1]
    return grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp FISTA (Nesterov accelerated proximal gradient) with tau annealing
2. LP best-response with Nesterov acceleration for precision polishing
3. Block coordinate descent with analytic 1D subproblem solver
4. Multi-start, multi-length exploration
5. Remez-style active set refinement
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Local evaluation of the score."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))

def evaluate_safe(a):
    """Safe evaluation using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def lse_objective(a, tau):
    """Compute log-sum-exp of autocorrelation with temperature tau."""
    b = fast_autocorr(a)
    max_b = b.max()
    if not np.isfinite(max_b):
        return float('inf')
    shifted = (b - max_b) / tau
    # Clip to avoid overflow
    shifted = np.clip(shifted, -100, 100)
    return max_b / tau + tau * np.log(np.sum(np.exp(shifted)))

def lse_gradient(a, tau):
    """
    Compute gradient of log(sum(exp(conv(a,a)/tau))) with respect to a.
    Uses FFT for efficiency: O(n log n).
    """
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1

    # Softmax probabilities
    max_b = b.max()
    shifted = (b - max_b) / tau
    shifted = np.clip(shifted, -100, 100)
    probs = np.exp(shifted)
    sum_probs = probs.sum()
    if sum_probs < 1e-300:
        probs = np.zeros(size)
        probs[b.argmax()] = 1.0
    else:
        probs /= sum_probs

    # grad[i] = sum_j a[j] * probs[i+j]
    # Using FFT: grad[i] = conv(a, probs_rev)[2n-2-i]
    p_rev = probs[::-1]
    fft_size = 1 << (size + n - 1).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p_rev, fft_size)
    c = irfft(A * P, fft_size)
    grad = c[n - 1:2 * n - 1][::-1]
    return grad

def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # We need sum(max(v_i - theta, 0)) = target_sum
    # Use bisection
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.max(np.arange(1, n + 1)[v_sorted > 0] - (cssv[v_sorted > 0] - target_sum) / np.arange(1, n + 1)[v_sorted > 0] > 0)
    if rho == 0:
        rho = 1
    theta = (cssv[rho - 1] - target_sum) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    s = result.sum()
    if s > 1e-15:
        result = result * target_sum / s
    return result

def fista_lse(a0, time_limit, seed=0, verbose=True, tau_start=2.0, tau_end=0.05):
    """
    FISTA (Nesterov accelerated proximal gradient) for log-sum-exp objective.
    Projects onto simplex after each step.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_maxb = fast_autocorr(a).max()

    y = a.copy()
    t_fista = 1.0
    eta = 0.1  # step size
    eta_min = 0.001
    eta_max = 1.0
    backtracks = 0
    t0 = time.monotonic()
    stall = 0
    stall_limit = 2000

    f_prev = lse_objective(y, tau_start)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal tau
        progress = min(t / 80000.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute gradient at y
        grad = lse_gradient(y, tau)

        # FISTA update
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        y_new = y - eta * grad
        # Add momentum
        momentum = (t_fista - 1.0) / t_next
        y_new = y_new + momentum * (y - a)

        # Project onto simplex
        a_new = project_simplex(y_new, norm)

        # Check convergence
        maxb_new = fast_autocorr(a_new).max()
        if maxb_new < best_maxb - 1e-12:
            best_maxb = maxb_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive step size
        f_new = lse_objective(a_new, tau)
        f_y = lse_objective(y, tau)
        if f_new > f_y + 0.5 * eta * np.dot(grad, grad):
            eta = max(eta * 0.5, eta_min)
            backtracks += 1
            # Keep old iterate
            a_new = a.copy()
        else:
            if backtracks > 5 and eta < eta_max:
                eta = min(eta * 1.2, eta_max)
                backtracks = 0

        a = a_new
        y = a_new
        t_fista = t_next

        # Restart if stalled
        if stall > stall_limit:
            y = best_a.copy()
            a = best_a.copy()
            t_fista = 1.0
            eta = 0.05
            stall = 0
            y = y + 0.02 * rng.standard_normal(n)
            y = project_simplex(y, norm)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA t={t} ({elapsed:.0f}s): tau={tau:.4f}, eta={eta:.4f}, maxb={best_maxb:.10f}")

    return best_a, best_maxb


def build_active_constraints(a, active_idx):
    """Build constraint matrix T where T[r,j] = a[active_idx[r] - j] if valid."""
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row in range(m):
        k = active_idx[row]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

def lp_best_response(a, M, solver, active_idx=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val

def block_cd(a0, time_limit, rng, block_size=4, verbose=True):
    """Block coordinate descent: update blocks of coordinates at once."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    best_a = a.copy()
    best_maxb = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Random block partition
        perm = rng.permutation(n)
        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            block = perm[start:min(start + block_size, n)]
            if len(block) < 2:
                continue

            # For each block, try several perturbations
            best_block_val = best_maxb
            best_block_a = None

            for trial in range(8):
                a_try = a.copy()
                # Random perturbation within the block
                deltas = rng.standard_normal(len(block)) * a[block].mean() * 0.3
                a_try[block] = a[block] + deltas
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                val = fast_autocorr(a_try).max()
                if val < best_block_val - 1e-12:
                    best_block_val = val
                    best_block_a = a_try.copy()

            if best_block_a is not None:
                a = best_block_a
                if best_block_val < best_maxb:
                    best_maxb = best_block_val
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improve:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD pass={n_passes} ({elapsed:.0f}s): maxb={best_maxb:.10f}")

    return best_a, best_maxb

def coord_descent_fast(a0, time_limit, rng, verbose=True):
    """Fast coordinate descent using analytic 1D subproblem."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-10 * norm:
                continue

            # Try several values for a[i]
            # Keep sum fixed: others sum to norm - ai
            best_val = best_maxb
            best_new_ai = ai

            for factor in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1, 3.0]:
                new_ai = ai * factor

                if s_try < 1e-15:

def structured_init(n, rng, style=0):

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

        pert = best_seq * (1.0 + 0.03 * rng.standard_normal(n_cur))
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum()
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = fista_lse(a_norm, t3, verbose=False, seed=42, tau_start=1.0, tau_end=0.01)
        try_update(a_out, f"fista2_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 4: Block CD (10% time) ===
    print(f"\n=== Phase 4: Block CD n={len(best_seq)} ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = block_cd(a_norm, t4, rng, block_size=4, verbose=True)
        try_update(a_out, f"bcd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Coordinate descent (10% time) ===
    print(f"\n=== Phase 5: Coord Descent n={len(best_seq)} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent_fast(a_norm, t5, rng, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 6: Multi-length ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 60:
        lengths_to_try = []
        n_cur = len(best_seq)
        for dl in [200, -200, 400, -400]:
            nl = n_cur + dl
            if 800 <= nl <= 3000 and nl != n_cur:
                lengths_to_try.append(nl)
        for nl in lengths_to_try[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_each = min(t6 / max(len(lengths_to_try[:2]), 1), remaining - 30)
            if t_each < 30:
                break
            init = interpolate_seq(best_seq, nl)
            a_norm = init * np.sqrt(2.0 * nl) / init.sum()
            # Quick FISTA
            a_out, _ = fista_lse(a_norm, t_each * 0.6, verbose=False, seed=nl % 1000)
            try_update(a_out, f"fista_n{nl}")
            # Quick LP
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out, _ = lp_nesterov(a_norm, min(t_each * 0.4, remaining - 20), solver, verbose=False, seed=nl % 1000)
                try_update(a_out, f"lp_n{nl}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Final LP polish (10% time) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_best} ===")
    t7 = max(remaining * 0.10, 60.0)
    t7 = min(t7, remaining - 30)
    if t7 > 40:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t7, solver, verbose=True, seed=999)
        try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final CD polish ===
    if remaining > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final CD n={n_best} ===")
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent_fast(a_norm, t8, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: Final FISTA with low tau ===
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Final FISTA low-tau n={n_best} ===")
        t9 = min(remaining * 0.6, 80.0)
        if t9 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = fista_lse(a_norm, t9, verbose=False, seed=777, tau_start=0.1, tau_end=0.01)
            try_update(a_out, f"fista_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
