def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_Q_gradient(a, p, cache_fft_p=None):
    """Compute g[i] = sum_j p[i+j] * a[j] for i=0,...,n-1 using FFT."""
    n = len(a)
    p_rev = p[::-1]
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    if cache_fft_p is not None:
        P = cache_fft_p
    else:
        P = np.fft.rfft(p_rev, fft_size)
    conv = np.fft.irfft(P * A, fft_size)
    indices = 2 * n - 2 - np.arange(n)
    return conv[indices], P


def multiplicative_weights_minimax(a0, time_limit, verbose=True, eta_initial=0.8, inner_iters=15, inner_eta=0.5):
    """
    Multiplicative Weights Minimax approach.
    Player 2 (dual) maintains distribution p over lags using exponential weights.
    Player 1 (primal) minimizes expected quadratic via softmax iterations.
    """
    n = len(a0)
    a = norm_sum1(a0)
    K = 2 * n - 1
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    p = np.ones(K) / K
    eta = eta_initial
    rng = np.random.default_rng(777)

    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()

    outer_iter = 0
    while time.monotonic() - t0 < time_limit:
        outer_iter += 1

        # Compute autocorrelation of current a
        if outer_iter == 1:
            ac = fast_autocorr(a)
        else:
            ac = fast_autocorr(a)

        M = ac.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Update dual distribution p using multiplicative weights
        # Normalize ac to keep exponents manageable
        ac_norm = ac / (M + 1e-30)
        p *= np.exp(eta * ac_norm)
        p = np.maximum(p, 1e-30)
        p = p / p.sum()

        # Decay eta slowly
        eta = max(eta * 0.9995, 0.05)

        # Solve inner QP: minimize sum_k p[k] * c_a[k] = a^T Q a
        # using softmax iterations (mirror descent on simplex)
        a_inner = a.copy()
        P_rev_fft = np.fft.rfft(p[::-1], fft_size)

        for s in range(inner_iters):
            g, _ = compute_Q_gradient(a_inner, p, cache_fft_p=P_rev_fft)
            # Softmax update: a_i proportional to exp(-inner_eta * g_i)
            scaled = -inner_eta * g
            max_scaled = scaled.max()
            a_inner = np.exp(scaled - max_scaled)
            a_inner = np.maximum(a_inner, 1e-30)
            a_inner = a_inner / a_inner.sum()

        # Update main iterate with step size
        alpha = 2.0 / (outer_iter + 2.0)
        a_new = (1.0 - alpha) * a + alpha * a_inner
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a_inner.copy()
        a = a_new

        if verbose and outer_iter % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW t={outer_iter} ({elapsed:.1f}s): obj={2*n*best_a.sum()*0 + best_obj:.8f}, M={M:.8f}, eta={eta:.4f}")

    return best_a, best_obj


def compute_soft_subgradient(a, ac, top_k=40, beta=6.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def compute_exact_subgradient(a, ac):
    """Compute subgradient using only the argmax lag(s)."""
    n = len(a)
    M = ac.max()
    threshold = M * 0.999
    active_lags = np.where(ac >= threshold)[0]
    if len(active_lags) == 0:
        active_lags = np.array([np.argmax(ac)])

    grad = np.zeros(n)
    n_active = len(active_lags)
    for k in active_lags:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += (2.0 / n_active) * a[j_idx]
    return grad

def strategic_coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # affect the top lags?
            best_dst = -1
            best_score = float('inf')

                candidates = np.arange(n)

            found = False

def spectral_shaping(a0, time_limit, verbose=True):
    """
    Spectral approach: work in frequency domain to flatten the power spectrum.
    The autocorrelation is the inverse DFT of |A(f)|^2.

        power = np.abs(A) ** 2
        total_power = power.sum()

        # Target: flat power spectrum
        # But we need to maintain non-negativity of a
        # Use a "soft" target: power_target = total_power / (fft_size//2 + 1) * correction
        mean_power = total_power / (fft_size // 2 + 1)

        ac = fast_autocorr(a)
        M = ac.max()
        top_lag = np.argmax(ac)

        # d/d|A(f)| of c[k] is proportional to exp(-2*pi*f*k/fft_size)
        phase_target = np.exp(-2j * np.pi * np.arange(fft_size // 2 + 1) * top_lag / fft_size)

        eta = 0.01 / (1 + 0.001 * t)

        # Update in frequency domain: reduce power at frequencies that
        # contribute to the peak lag
        A_updated = A.copy()
        # Reduce components that align with the peak lag
        reduction = eta * power * np.real(phase_target)
        A_updated -= 0.5 * reduction / (np.abs(A_updated) + 1e-10) * A_updated

        a_new = np.fft.irfft(A_updated, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and try smaller step
            eta *= 0.5

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining multiple methods."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1660.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception as e:
        print(f"  Could not load parent: {e}")

    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0.0)
        s = arr.sum()
        if s < 0.01:
            return
        arr_norm = arr / s
        sc = evaluate_safe(arr_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(arr)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Multiplicative Weights Minimax (primary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MW Minimax n={n0} ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multiplicative_weights_minimax(a_in, t1, verbose=True, eta_initial=0.8)
        try_update(a_out, f"mw1_n{n0}")

    # === Phase 2: Refined MW with momentum ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MW Refined n={len(best_seq)} ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mw_minimax_refined(a_in, t2, verbose=True, eta0=1.0, inner_iters=20, inner_eta=0.3)
        try_update(a_out, f"mw2_n{len(best_seq)}")

    # === Phase 3: Refined AMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: AMD v3 n={len(best_seq)} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_mirror_descent_v3(a_in, t3, top_k=50, beta=5.0, c0=4.0, use_exact=False, verbose=True)
        try_update(a_out, f"amd3_n{len(best_seq)}")

    # === Phase 4: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if solver is not None and n_cur <= 2500:
        print(f"\n=== Phase 4: LP FW n={n_cur} ===")
        t4 = min(remaining * 0.15, 200.0)
        if t4 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True)
            try_update(a_out, f"lp_n{n_cur}")
    else:
        print(f"\n=== Phase 4: Skipping LP (solver={solver}, n={n_cur}) ===")

    # === Phase 5: Strategic coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Strategic Coord n={len(best_seq)} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = strategic_coordinate_polish(a_in, t5, verbose=True, rng_seed=555)
        try_update(a_out, f"strat_coord_n{len(best_seq)}")

    # === Phase 6: AMD polish with exact subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: AMD exact n={len(best_seq)} ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_mirror_descent_v3(a_in, t6, top_k=60, beta=4.0, c0=3.0, use_exact=True, verbose=False)
        try_update(a_out, f"amd_exact_n{len(best_seq)}")

    # === Phase 7: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 7: Length exploration ===")
        for n_try in [n_best - 200, n_best + 200]:
            if n_try < 500 or n_try > 3500:
                continue
            if time.monotonic() - t_global > time_budget - 60:
                break
            a_long = interpolate_sequence(best_seq, n_try)
            a_in = norm_sum1(a_long)
            t7 = min(remaining * 0.15, 80.0)
            if t7 > 20:
                a_out, _ = multiplicative_weights_minimax(a_in, t7, verbose=False, eta_initial=0.6)
                try_update(a_out, f"mw_n{n_try}")

    # === Phase 8: Final MW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final MW polish n={len(best_seq)} ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = mw_minimax_refined(a_in, t8, verbose=True, eta0=0.5, inner_iters=25, inner_eta=0.2)
        try_update(a_out, f"mw_final_n{len(best_seq)}")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final coord n={len(best_seq)} ===")
    t9 = min(remaining * 0.40, 150.0)
    if t9 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = strategic_coordinate_polish(a_in, t9, verbose=True, rng_seed=999)
        try_update(a_out, f"final_coord_n{len(best_seq)}")

    # === Phase 10: Final AMD micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 10: Final AMD micro n={len(best_seq)} ===")
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = accelerated_mirror_descent_v3(a_in, t10, top_k=80, beta=3.0, c0=5.0, use_exact=True, verbose=False)
            try_update(a_out, f"amd_micro_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
