def compute_grad(a, w):
    """Gradient of sum_k w_k * conv(a,a)_k w.r.t. a.
    grad_m = 2 * sum_j w[m+j] * a[j] for valid indices.
    Computed via FFT cross-correlation.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, norm_val):
    """Normalize a to have sum = norm_val, clipping negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * norm_val / n
    return a * norm_val / s


def fista_smooth(a0, time_limit, verbose=True):
    """FISTA with log-sum-exp smoothing and backtracking line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Temperature schedule - more aggressive cooling
    temps = [10.0, 5.0, 3.0, 2.0, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05, 0.03, 0.02, 0.012, 0.008, 0.005, 0.003, 0.002]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Number of steps for this temperature
        n_steps = max(50, min(300, int(remaining * 4)))

        y = a.copy()
        c = 1.0  # Step size for backtracking

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute smooth objective and gradient
            conv_y = fast_autocorr(y)
            c_max = conv_y.max()
            exp_c = np.exp((conv_y - c_max) / T)
            exp_sum = exp_c.sum()
            if exp_sum < 1e-300:
                w = np.zeros(len(conv_y))
                w[np.argmax(conv_y)] = 1.0
            else:
                w = exp_c / exp_sum

            grad = compute_grad(y, w)

            # Backtracking line search
            step = c
            for _ in range(20):
                a_new = y - step * grad
                a_new = project_simplex(a_new, norm)

                conv_new = fast_autocorr(a_new)
                c_new_max = conv_new.max()
                exp_new = np.exp((conv_new - c_new_max) / T)
                obj_new = (c_new_max + T * np.log(exp_new.sum()))

                conv_y_max = conv_y.max()
                exp_y = np.exp((conv_y - conv_y_max) / T)
                obj_y = (conv_y_max + T * np.log(exp_y.sum()))

                # Directional derivative
                grad_obj = compute_grad(y, w)
                lin_bound = obj_y + grad_obj @ (a_new - y) + 0.5 / c * np.sum((a_new - y)**2) \
                    if np.any(a_new != y) else obj_y

                if obj_new <= obj_y + 0.01 * (np.dot(grad, a_new - y)) + 0.5 / c * np.sum((a_new - y)**2):
                    break
                c *= 0.5
                step = c
            else:
                break

            # Nesterov update
            if it == 0:
                y_next = a_new.copy()
                a_next = a_new.copy()
            else:
                t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * (it + 1)**2))
                beta = (t_k - 1.0) / t_k
                y_next = a_next + beta * (a_next - y)
                y_next = project_simplex(y_next, norm)
                a_next = a_new

            y = y_next
            a = a_next

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and (it % 50 == 0 or it == n_steps - 1):
                elapsed = time.monotonic() - t0
                print(f"    FISTA T={T:.3f} it={it}/{n_steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) with subgradient of max(conv).
    Uses adaptive step size based on Polyak-like scheme.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Lower bound estimate (for Polyak step)
    # For sum(a)=sqrt(2n), min possible max(conv) >= 2 (by Cauchy-Schwarz at center lag for symmetric)
    # Use a conservative lower bound
    lb = 1.0  # Conservative lower bound
    ub = best_score

    # Multiplicative weight update
    eta = 1.0 / best_score
    momentum = 0.9

    prev_grad = np.zeros(n)
    avg_a = a.copy()
    n_avg = 1

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fast_autocorr(a)
        peak_val = conv_a.max()
        peak_idx = np.where(conv_a >= peak_val - 1e-10)[0]

        # Subgradient: for each peak position k, grad_m = 2*a[k-m] (valid range)
        # Use the weighted subgradient (average over argmax positions)
        grad = np.zeros(n)
        for k in peak_idx:
            # grad_m += 2 * a[k-m] for 0 <= k-m < n
            valid_start = max(0, k - (n - 1))
            valid_end = min(n - 1, k)
            for m in range(valid_start, valid_end + 1):
                grad[m] += 2.0 * a[k - m]
        grad /= len(peak_idx)

        # Adaptive step size (Polyak-like)
        if peak_val > lb + 1e-6:
            eta = 0.5 / (peak_val - lb + 1e-6)
        eta = min(eta, 2.0 / peak_val)

        # Multiplicative update with clamping
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new * norm / a_new.sum()

        # Smooth with previous (to avoid oscillation)
        a_mixed = momentum * a + (1.0 - momentum) * a_new
        a_mixed = np.maximum(a_mixed, 1e-15)
        a_mixed = a_mixed * norm / a_mixed.sum()

        a = a_mixed

        # Running average
        avg_a = (n_avg * avg_a + a) / (n_avg + 1)
        n_avg += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            ub = score

        score_avg = fast_autocorr(avg_a).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_a.copy()
            ub = score_avg

        if verbose and (it % 100 == 0 or it == 99999):
            elapsed = time.monotonic() - t0
            print(f"    Mirror it={it} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

    return best_a, best_score


def build_conv_rows(a, indices):
    """Build rows of the convolution matrix for given lag indices.
    Returns matrix T of shape (len(indices), n) where T[j,i] gives coefficient of a[i] in conv[a,b][indices[j]].
    """
    n = len(a)
    m = len(indices)
    T = np.zeros((m, n))
    for j, k in enumerate(indices):
        k = int(k)
        # conv[a,b][k] = sum_i a[i]*b[k-i] (valid: 0<=i<n, 0<=k-i<n)
        # So coefficient of b[i] in conv[a,b][k] is a[k-i] if valid
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            T[j, i] = a[k - i]
    return T


def lp_best_response(a, M, solver, active_idx):
    """Solve max sum(b) s.t. conv(a,b) <= M (on active constraints), b >= 0."""
    n = len(a)
    T = build_conv_rows(a, active_idx)

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def mannokeller_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Manno-Keller (weighted average) and constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Manno-Keller: maintain weighted average
    ak_sum = a.copy()  # Accumulated weighted sum
    w_total = 1.0  # Total weight
    n_iters = 0
    stuck = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fast_autocorr(a)
        M = conv_a.max()
        if M < 1e-12:
            break

        # Constraint generation: use top constraints
        threshold = 0.9 * M
        active = np.where(conv_a >= threshold)[0]
        if len(active) < 3:
            active = np.where(conv_a >= 0.7 * M)[0]
        if len(active) < 2:
            active = np.argsort(conv_a)[::-1][:min(20, len(conv_a))]

        b = lp_best_response(a, M, solver, active)
        if b is None:
            # Try with all constraints (slower)

            b = lp_best_response(a, M, solver, all_idx)
            if b is None:
                stuck += 1
                if stuck > 50:
                    break
                continue

        stuck = 0
        b = b * norm / b.sum()
        n_iters += 1

        # Manno-Keller update: weighted average with weight proportional to 1/sqrt(t)
        w_t = 1.0 / np.sqrt(t + 1.0)
        ak_sum = ak_sum + w_t * b
        w_total = w_total + w_t

        # Current iterate is the weighted average
        a = ak_sum / w_total
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = b.copy()
        else:
            a = a * norm / s

        score = fast_autocorr(a).max()

            best_score = score
            best_a = a.copy()

        score_b = fast_autocorr(b).max()
        if score_b < best_score:

def pair_cd(a0, time_limit, rng, window=60, verbose=True):
    """Pair coordinate descent with binary search step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            for offset in range(1, min(window, n) + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i or a[i] < norm * 1e-9:
                        continue

                    # Quick test: move 30% of a[i] to a[j]
                    d = a[i] * 0.3
                    a_test = a.copy()
                    a_test[i] -= d
                    a_test[j] += d
                    a_test = a_test * norm / a_test.sum()
                    sc = fast_autocorr(a_test).max()

                    if sc < best_score - 1e-10:
                        # Binary search for optimal d
                        lo, hi = 0.0, d
                        best_d = d
                        best_sc = sc
                        for _ in range(6):
                            mid = (lo + hi) / 2
                            if mid < a[i] * 0.99:
                                a_t = a.copy()
                                a_t[i] -= mid
                                a_t[j] += mid
                                a_t = a_t * norm / a_t.sum()
                                sc_t = fast_autocorr(a_t).max()
                                if sc_t < best_sc:
                                    best_sc = sc_t
                                    best_d = mid
                                    lo = mid
                                else:
                                    hi = mid

                        a[i] -= best_d
                        a[j] += best_d
                        a = a * norm / a.sum()
                        score = fast_autocorr(a).max()
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                        improved = True
                        no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def gaussian_init(n, rng):
    """Gaussian-shaped initialization."""
    x = np.arange(n)
    center = n / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((x - center) / sigma)**2)
    a = a + 0.01 * rng.random(n)
    return a


def powerlaw_init(n, rng, gamma=0.5):
    """Power-law shaped initialization (higher at edges)."""
    x = np.arange(n)
    center = n / 2.0
    a = 1.0 / (np.abs(x - center) / n + 0.1)**gamma
    a = a + 0.01 * rng.random(n)
    return a


def sine_init(n, rng, p=2.0):
    """Sinusoidal initialization."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x)**p
    a = a + 0.01 * rng.random(n)
    return a


def bimodal_init(n, rng):
    """Two Gaussian bumps."""
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - n*0.25) / (n*0.1))**2) + np.exp(-0.5 * ((x - n*0.75) / (n*0.1))**2)
    a = a + 0.01 * rng.random(n)
    return a


def random_init(n, rng):
    """Random positive initialization."""
    return rng.random(n) + 0.01


def interpolate_renorm(seq, new_n):
    """Interpolate sequence to new length and renormalize."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    s = seq.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    seq_norm = seq / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, seq_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    return f_interp * np.sqrt(2.0 * new_n) / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 0: Diverse initialization exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = remaining * 0.12
    print(f"\n=== Phase 0: Diverse init exploration (budget={t0_phase:.0f}s) ===")

    if t0_phase > 10:
        inits = [
            (gaussian_init(n_base, rng), "gauss"),
            (powerlaw_init(n_base, rng, 0.5), "power05"),
            (powerlaw_init(n_base, rng, 1.0), "power10"),
            (sine_init(n_base, rng, 2.0), "sine2"),
            (sine_init(n_base, rng, 4.0), "sine4"),
            (bimodal_init(n_base, rng), "bimodal"),
            (random_init(n_base, rng), "random"),
        ]
        # Also try the parent
        inits.append((best_seq * np.sqrt(2*n_base) / best_seq.sum(), "parent"))

        per_init_time = t0_phase / len(inits)
        for a_init, name in inits:
            if time.monotonic() - t_global > time_budget:
                break
            a_norm = normalize_seq(a_init, np.sqrt(2.0 * n_base))
            # Quick FISTA on this init
            a_out, _ = fista_smooth(a_norm, per_init_time, verbose=False)
            do_try(a_out, f"init_{name}")

    # === Phase 1: Main FISTA on best so far ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.30
    n_c = len(best_seq)
    print(f"\n=== Phase 1: FISTA n={n_c} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = fista_smooth(a_norm, t1, verbose=True)
        do_try(a_out, f"fista_n{n_c}")

    # === Phase 2: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    n_c = len(best_seq)
    print(f"\n=== Phase 2: Mirror descent n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = mirror_descent(a_norm, t2, verbose=True)
        do_try(a_out, f"mirror_n{n_c}")

    # === Phase 3: LP Manno-Keller ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    n_c = len(best_seq)
    print(f"\n=== Phase 3: MK-LP n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = mannokeller_lp(a_norm, t3, solver, verbose=True)
        do_try(a_out, f"mk_lp_n{n_c}")

    # === Phase 4: FISTA round 2 (on possibly improved best) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    n_c = len(best_seq)
    print(f"\n=== Phase 4: FISTA round 2 n={n_c} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = fista_smooth(a_norm, t4, verbose=True)
        do_try(a_out, f"fista2_n{n_c}")

    # === Phase 5: Mirror descent round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    n_c = len(best_seq)
    print(f"\n=== Phase 5: Mirror round 2 n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = mirror_descent(a_norm, t5, verbose=False)
        do_try(a_out, f"mirror2_n{n_c}")

    # === Phase 6: Pair CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.15
    n_c = len(best_seq)
    print(f"\n=== Phase 6: Pair CD n={n_c} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = pair_cd(a_norm, t6, rng, window=50, verbose=True)
        do_try(a_out, f"pcd_n{n_c}")

    # === Phase 7: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        alts = [n_cur + 50, n_cur - 50, n_cur + 100]
        for n_alt in alts:
            if n_alt < 200:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 40:
                break
            t_alt = min(remaining_sub * 0.3, 120.0)
            print(f"\n=== Phase 7: Alt length n={n_alt} (budget={t_alt:.0f}s) ===")
            init_alt = interpolate_renorm(best_seq, n_alt)
            a_out, _ = fista_smooth(init_alt, t_alt, verbose=False)
            do_try(a_out, f"alt_fista_n{n_alt}")
            remaining_sub2 = time_budget - (time.monotonic() - t_global)
            if remaining_sub2 > 30:
                t_alt2 = min(remaining_sub2 * 0.2, 60.0)
                a_out2, _ = mirror_descent(normalize_seq(best_seq, np.sqrt(2.0*n_alt)) , t_alt2, verbose=False)
                # Interpolate result back
                do_try(a_out2, f"alt_mirror_n{n_alt}")

    # === Final: one more FISTA + PCD on global best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        n_c = len(best_seq)
        t_final = remaining * 0.6
        print(f"\n=== Final polish n={n_c} (budget={t_final:.0f}s) ===")
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = fista_smooth(a_norm, t_final * 0.5, verbose=False)
        do_try(a_out, "final_fista")

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10:
            a_norm2 = normalize_seq(best_seq, np.sqrt(2.0 * len(best_seq)))
            a_out2, _ = pair_cd(a_norm2, remaining2 * 0.8, rng, window=80, verbose=False)
            do_try(a_out2, "final_pcd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in (best_seq / best_seq.sum())]
