# Global solver

def get_solver():

    if _global_solver is not None:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:

    _global_solver = cp.SCS
    print(f"  LP solver (fallback): {_global_solver}")
    return _global_solver

def fast_conv(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    c = fast_autocorr(a)
    s = a.sum()

    return 2.0 * len(a) * c.max() / (s * s)

        k_max = min(size - 1, j + n - 1)  # when k-j = n-1
        T[k_min:k_max+1, j] = a[:k_max-k_min+1]
    return T

import time
from scipy.fft import rfft, irfft

_solver_time = {}

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n), dtype=np.float64)
    for j in range(n):
        k_min = j
        k_max = min(size - 1, j + n - 1)
        T[k_min:k_max + 1, j] = a[:k_max - k_min + 1]
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        t0 = time.monotonic()
        prob.solve(solver=solver, verbose=False)
        _solver_time[solver] = time.monotonic() - t0
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def normalize_to(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def nesterov_lp(a0, time_limit, solver, verbose=True, max_restarts=20):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, 1.0)

    best_a = a.copy()
    c = fast_autocorr(a)
    best_score = c.max()

    y = a.copy()
    a_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    t_k = 1.0
    patience = 0
    max_patience = 60
    restart_count = 0
    rng = np.random.default_rng(7777)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        c_y = fast_autocorr(y)
        M_curr = c_y.max()
        if M_curr < 1e-14:

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            patience += 1
            if patience > 50:
                break
            continue

        patience = 0
        b = normalize_to(b, 1.0)
        n_iters += 1

        # Nesterov extrapolation
        alpha = 2.0 / (t_k + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to(y_new, 1.0)

        # Momentum step
        if t_k < 3.0:
            a_new = b.copy()
        else:
            momentum = (t_k - 1.0) / (t_k + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, 1.0)

        y = y_new
        a_prev = a
        a = a_new
        t_k += 1.0

        # Check improvement
        c_a = fast_autocorr(a)
        score_a = c_a.max()
        c_y = fast_autocorr(y)
        score_y = c_y.max()

            best_a = a.copy()
            patience = 0
        elif score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        # Restart logic
        if patience > max_patience:
            restart_count += 1
            y = best_a.copy()
            a = best_a.copy()
            t_k = 1.0
            patience = 0
            max_patience = min(max_patience + 20, 200)

        M_curr = c[k_star]

        # Subgradient: g[j] = 2*a[k_star - j] if 0 <= k_star-j < n
        g = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_max >= j_min:
            indices = np.arange(j_min, j_max + 1)
            g[indices] = 2.0 * a[k_star - indices]

        # Mirror descent (exponential family update)
        log_a = np.log(a + 1e-30)
        best_trial = a.copy()
        best_trial_score = M_curr

        for factor in [0.2, 0.5, 1.0, 2.0, 3.0]:
            eta_try = eta * factor
            log_a_try = log_a - eta_try * g
            a_try = np.exp(log_a_try - log_a_try.max())
            a_try = normalize_to(a_try, 1.0)
            c_try = fast_autocorr(a_try)
            score_try = c_try.max()
            if score_try < best_trial_score:
                best_trial_score = score_try
                best_trial = a_try
                eta = eta_try

        a = best_trial
        n_iters += 1

        if best_trial_score < best_score - 1e-14:
            best_score = best_trial_score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 200 == 0:
                eta *= 0.85

        if patience > 500:
            # Perturb
            noise = rng.laplace(0, 0.05 / np.sqrt(n), n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = normalize_to(a_pert, 1.0)
            a = a_pert
            patience = 0
            eta = 0.1 / np.sqrt(n)

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, 1.0)

    best_a = a.copy()
    c = fast_autocorr(a)
    best_score = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(3333)
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if no_improve > 30:
            break

        c = fast_autocorr(a)
        k_star = int(np.argmax(c))
        pass_improved = False

        # Try single-coordinate moves
        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-12:
                continue

            # Try moving mass from i to j
            # Focus on positions that contribute to the peak
            offsets = [1, -1, 2, -2, 3, -3, 5, -5, 10, -10]
            for off in offsets:
                j = (i + off) % n
                if j == i:
                    continue

                # Compute new convolution incrementally
                # Moving d from i to j changes conv by:
                # -d * shift(a, i) - d * shift(a, i) reversed + d * shift(a, j) + d * shift(a, j) reversed + d^2 corrections
                # Simpler: just recompute
                for frac in [0.1, 0.3, 0.5]:
                    d = a[i] * frac
                    if d < 1e-15:
                        break
                    a[i] -= d
                    a[j] += d
                    a = normalize_to(a, 1.0)
                    c_new = fast_autocorr(a)
                    score_new = c_new.max()
                    if score_new < best_score - 1e-14:
                        best_score = score_new
                        best_a = a.copy()
                        c = c_new
                        pass_improved = True
                        no_improve = 0
                        break
                    else:
                        a[j] -= d
                        a[i] += d
                        a = normalize_to(a, 1.0)

        if not pass_improved:
            no_improve += 1
            if no_improve % 3 == 0 and no_improve <= 30:
                # Gentle perturbation
                noise = rng.normal(0, 0.01 / np.sqrt(n), n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                a = normalize_to(a_pert, 1.0)
                no_improve = 0

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, verbose=True):
    """Target the top few lags and try to reduce them specifically."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, 1.0)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(8888)
    n_iters = 0

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        top_lags = np.argsort(c)[-5:]
        M_curr = c.max()

        # For each top lag, try to reduce it by adjusting nearby coordinates
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)

            # The contribution to c[k] from a[j] is a[k-j] (for valid j)
            # We want to reduce c[k], so we want to reduce a[j]*a[k-j] for large products
            # Try: reduce the largest a[j] that contributes to lag k, and move mass elsewhere
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_max < j_min:
                continue

            contrib = np.zeros(n)
            for j in range(j_min, j_max + 1):
                contrib[j] = a[j] * a[k - j]

            # Find the j with largest contribution
            j_star = j_min + int(np.argmax(contrib[j_min:j_max + 1]))
            if contrib[j_star] < 1e-15:
                continue

            # Move some mass from j_star to a position that doesn't contribute to lag k
            for _ in range(3):
                j_new = rng.integers(0, n)
                if j_new == j_star:
                    continue
                # Check if j_new contributes to lag k
                if 0 <= k - j_new < n and a[j_new] * a[k - j_new] > 0:
                    continue

                for frac in [0.05, 0.1, 0.2]:
                    d = a[j_star] * frac
                    if d < 1e-15:
                        break
                    a[j_star] -= d
                    a[j_new] += d
                    a = normalize_to(a, 1.0)
                    c_new = fast_autocorr(a)
                    score_new = c_new.max()
                    if score_new < best_score - 1e-14:
                        best_score = score_new
                        best_a = a.copy()
                    else:
                        a[j_new] -= d
                        a[j_star] += d
                        a = normalize_to(a, 1.0)

        n_iters += 1
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        c = fast_autocorr(a)
        sc = 2.0 * len(a) * c.max() / (s * 1.0)  # after normalization, s=1, so sc = 2n*max_c
        # Actually after a = a/s, the sum is 1, so score = 2n * max(c)
        sc = 2.0 * len(a) * c.max()
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    def try_update_real(a, label):
        """Use the actual evaluate_sequence for verification."""
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_list = [float(x) for x in a]
        try:
            sc = float(evaluate_sequence(a_list))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Get parent sequence
    n_base = 1600
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            try_update_real(parent, "parent")
            n_base = len(parent)
    except Exception:
        pass

    if parent is None:
        a_init = np.ones(n_base) / n_base
        parent = a_init
        try_update_real(a_init, "uniform_init")
        print(f"Using uniform init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Multi-start initialization (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.10
    print(f"\n=== Phase 1: Multi-start (budget={t1:.0f}s) ===")

    initializations = [parent.copy()]

    # Structured initialization: smooth wave
    n_w = n_base
    x = np.arange(n_w)
    for freq in [1, 2, 3]:
        a_wave = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x / n_w)
        a_wave = np.maximum(a_wave, 0.01)
        a_wave = a_wave / a_wave.sum()
        initializations.append(a_wave.copy())

    # Random initialization
    a_rand = np.abs(rng.normal(1.0, 0.5, n_base))
    a_rand = a_rand / a_rand.sum()
    initializations.append(a_rand.copy())

    # Exponential decay from center
    center = n_base // 2
    a_decay = np.exp(-0.5 * np.abs(x - center) / (n_base / 4))
    a_decay = a_decay / a_decay.sum()
    initializations.append(a_decay.copy())

    # Two-peak
    a_2peak = np.ones(n_base) * 0.1
    a_2peak[n_base // 4] = 1.0
    a_2peak[3 * n_base // 4] = 1.0
    a_2peak = a_2peak / a_2peak.sum()
    initializations.append(a_2peak.copy())

    t_per_init = t1 / len(initializations)
    for i, a_init in enumerate(initializations):
        if time.monotonic() - t_global > time_budget - 30:
            break
        t_sub = min(t_per_init, time_budget - (time.monotonic() - t_global))
        if t_sub < 5:
            break
        a_out, sc = nesterov_lp(a_init, t_sub, solver, verbose=False)
        try_update(a_out, f"init_{i}")

    # Phase 2: Main Nesterov LP (50% of remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    t2 = remaining * 0.50
    print(f"\n=== Phase 2: Main Nesterov LP n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, sc = nesterov_lp(best_seq, t2, solver, verbose=True)
        try_update(a_out, "main_lp")

    # Phase 3: Subgradient polish (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Subgradient polish n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, sc = subgradient_polish(best_seq, t3, verbose=True)
        try_update(a_out, "sg_polish")

    # Phase 4: Peak shaving (10% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Peak shaving n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_out, sc = peak_shaving(best_seq, t4, verbose=True)
        try_update(a_out, "peak_shave")

    # Phase 5: Second LP round on best so far (10% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: LP Round 2 n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        # Perturb slightly before LP
        noise = rng.normal(0, 0.01 / np.sqrt(n_best), n_best)
        a_pert = best_seq + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_out, sc = nesterov_lp(a_pert, t5, solver, verbose=True)
        try_update(a_out, "lp_round2")

    # Phase 6: Coordinate descent (5% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    t6 = remaining * 0.05
    print(f"\n=== Phase 6: Coordinate Descent n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 5:
        a_out, sc = coordinate_descent(best_seq, t6, verbose=True)
        try_update(a_out, "cd")

    # Phase 7: Final best LP run with most remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq) if best_seq is not None else n_base
        t7 = remaining * 0.8
        print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
        a_out, sc = nesterov_lp(best_seq, t7, solver, verbose=True)
        try_update(a_out, "final_lp")

    # Final verification with actual evaluator
    if best_seq is not None:
        a_list = [float(x) for x in best_seq]
        try:
            final_sc = float(evaluate_sequence(a_list))
            if np.isfinite(final_sc):
                best_score = final_sc
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
