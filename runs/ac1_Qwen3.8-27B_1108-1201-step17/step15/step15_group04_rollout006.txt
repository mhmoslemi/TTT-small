    best_obj = compute_obj(a)
    t0 = time.monotonic()
    step = 0.5 / np.sqrt(n)
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        direction, max_c = subgradient_direction(a, top_k)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        direction = direction / dir_norm

        # Backtracking line search
        improved = False
        for trial in range(20):
            a_new = subgradient_step(a, direction, step, norm_val)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                a = a_new
                best_a = a.copy()
                best_obj = obj_new
                improved = True
                break
            step *= 0.7
        if not improved:
            step *= 0.5
            stagnant += 1
        else:
            step = min(step * 1.05, 2.0 / np.sqrt(n))
            stagnant = 0

        if stagnant > 50:
            # Perturb to escape
            pert_idx = np.random.default_rng(hashed(id(a))).choice(n, size=max(5, n//20), replace=False)
            a[pert_idx] *= (1.0 + 0.1 * np.random.default_rng(int(time.time()*1000) % 100000).standard_normal(len(pert_idx)))
            a = normalize_to(a, norm_val)
            stagnant = 0

        if verbose and (int((time.monotonic() - t0) // 10) % 3 == 0):
            pass  # print every ~30s

    return best_a, best_obj

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_max >= j_min:
            T[k, j_min:j_max+1] = a[k - np.arange(j_min, j_max + 1)]
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b = cp.Variable(n, nonneg=True)

    if active is not None and 0 < len(active) < size:
        T_use = T[active, :]
    else:
        T_use = T

    constraints = [T_use @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=solver, verbose=False)

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    return b_val

def lp_nesterov(a0, time_limit, solver, verbose=True, top_k_constraints=200):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_to(a0.copy(), norm_val)
    best_a = a.copy()
    best_obj = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    prev_y = a.copy()
    size = 2 * n - 1

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        if t < 20:
            threshold = 0.5
        elif t < 100:
            threshold = 0.65
        elif t < 500:
            threshold = 0.75
        else:
            threshold = 0.85
        active = np.where(conv_y > M * threshold)[0]
        if len(active) < 50:
            active = np.where(conv_y > M * 0.4)[0]
        # Cap number of constraints for speed
        if len(active) > top_k_constraints:
            top_idx = np.argsort(conv_y[active])[::-1][:top_k_constraints]
            active = active[top_idx]

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum

        # Verify no major constraint violation
        if active is not None and len(active) < size:
            check = fast_conv_cross(y, g)
            if check.max() > M * (1 + 0.01):
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    g = b_full * norm_val / b_full.sum()

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = normalize_to(y_new, norm_val)

        # Momentum for a
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize_to(a_new, norm_val)

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_obj - 1e-12:
            best_obj = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_obj - 1e-12:
            best_obj = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): peak={best_obj:.8f}, obj={2*n*best_obj/norm_val**2:.6f}")

    return best_a, best_obj

def peak_shave_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent that specifically targets peak reduction."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_to(a0.copy(), norm_val)
    best_a = a.copy()
    best_obj = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Find the top peaks
        n_peaks = min(10, size)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        # For each peak, find the contributing pairs and try to reduce
        pass_improved = False
        perm = rng.permutation(n)

        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = idx
            ai = a[i]
            if ai < norm_val * 1e-8:
                continue

            # Find which peaks i contributes to most
            # i contributes to conv[k] through a[i]*a[k-i] and a[k-i]*a[i]
            # So it contributes to conv[k] for k = i + j where j is valid
            # The main contributions are at k = 2i (self) and k = i+j for other j

            # Try transferring mass from i to a random j
            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            n_trials = 8
            trials_j = list(rng.integers(n, size=n_trials))
            # Add structured candidates from peaks
            for pk in peak_indices[:5]:
                if pk - i >= 0 and pk - i < n and pk - i != i:
                    trials_j.append(pk - i)
                if pk + i >= 0 and pk + i < n and pk + i != i:
                    trials_j.append(pk + i)

            for j in set(trials_j):
                if j == i:
                    continue
                aj = a[j]

                # Compute change in convolution due to a[i] -= d, a[j] += d
                # Delta conv[k] = -d*a[k-i] + d*a[k-j] - d*a[i-k] + d*a[j-k] + d^2*delta terms
                # Simplified: for small d, delta conv[k] ≈ d*(a[k-j] - a[k-i] + a[j-k] - a[i-k])
                # Plus second order: +d^2*(delta_{k,2i}*(-1) + delta_{k,2j}*(1) + delta_{k,i+j}*(-2))
                # ... this is getting complex. Let's just evaluate directly for a few deltas.

                deltas = np.array([0.05, 0.15, 0.3, 0.5, 0.7, 0.9]) * ai
                for d in deltas:
                    if d > ai:
                        continue
                    a[i] -= d
                    a[j] += d
                    new_conv = fast_conv_self(a)
                    new_max = new_conv.max()
                    a[i] += d
                    a[j] -= d

                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize_to(a, norm_val)
                actual_max = fast_conv_self(a).max()
                if actual_max < best_obj - 1e-14:
                    best_obj = actual_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = normalize_to(a, norm_val)

        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                # Perturb
                n_perturb = max(20, n // 15)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = normalize_to(a, norm_val)
                no_improve = 0

        if verbose and (int((time.monotonic() - t0) // 15) % 2 == 0) and no_improve % 50 == 0:
            pass

    return best_a, best_obj

def lse_descent(a0, time_limit, verbose=False):
    """Log-sum-exp smoothed gradient descent."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_to(a0.copy(), norm_val)
    best_a = a.copy()
    best_peak = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(50.0, 0.5, n_stages)

    for si in range(n_stages):
        T_val = temps[si]
        stage_time = time_limit / n_stages
        stage_start = time.monotonic()

        for it in range(200):
            if time.monotonic() - stage_start > stage_time:
                break
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            step = 1.0 / (T_val * np.sqrt(n))
            for trial in range(10):
                a_new = normalize_to(a - step * grad, norm_val)
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5

            a = normalize_to(a - step * grad, norm_val)
            sc = fast_conv_self(a).max()
            if sc < best_peak:
                best_peak = sc
                best_a = a.copy()

    return best_a, best_peak

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = compute_obj(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: Quick LSE warm-up
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    print(f"\n=== Phase 1: LSE warm-up ({t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_descent(best_seq, t1, verbose=False)
        try_update(a_out, "lse_warmup")

    # Phase 2: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} ({t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov(best_seq, t2, solver, verbose=True, top_k_constraints=300)
        try_update(a_out, "lp_main")

    # Phase 3: Subgradient refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Subgradient n={len(best_seq)} ({t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = subgradient_descent(best_seq, t3, top_k=5, verbose=False)
        try_update(a_out, "subgrad")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.2, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 ({t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov(best_seq, t4, solver, verbose=True, top_k_constraints=200)
        try_update(a_out, "lp2")

    # Phase 5: Peak shaving CD
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Peak-shaving CD ({t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = peak_shave_cd(best_seq, t5, rng, verbose=True)
        try_update(a_out, "peak_shave")

    # Phase 6: Try different length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length exploration (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        for n_try in [n_base + 200, n_base - 200]:
            if remaining < 60:
                break
            if n_try < 500 or n_try > 5000:
                continue
            init_long = interpolate_to_length(best_seq, n_try)
            t_try = min(remaining * 0.3, 120.0)
            if t_try < 30:
                break
            it2 = time.monotonic() - t_global
            a_out, _ = lp_nesterov(init_long, t_try, solver, verbose=False, top_k_constraints=200)
            try_update(a_out, f"lp_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        t_f1 = min(remaining * 0.3, 60.0)
        a_out, _ = lse_descent(best_seq, t_f1, verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_f2 = min(remaining * 0.4, 60.0)
            a_out, _ = peak_shave_cd(best_seq, t_f2, rng, verbose=False)
            try_update(a_out, "pcd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_f3 = min(remaining * 0.5, 60.0)
            a_out, _ = lp_nesterov(best_seq, t_f3, solver, verbose=False, top_k_constraints=150)
            try_update(a_out, "lp_final")

    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
