def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, constraints_mask=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if constraints_mask is not None:
        idx = np.where(constraints_mask)[0]
        if len(idx) > 0:
            constraints = [T[idx, :] @ b_var <= M]
        else:
            constraints = [T @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    core_lo = n // 4
    core_hi = size - n // 4

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        core = conv[core_lo:core_hi]
        core_mean = core.mean()
        residual = np.zeros(size)
        residual[core_lo:core_hi] = core - core_mean

        R = rfft(residual, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * R, fft_size)[:n]
        g2 = irfft(A * R, fft_size)[:n]
        grad = 2.0 * (g1 + g2)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        step = S / (np.sqrt(n) * (1.0 + 0.05 * it))
        a_new = a - step * grad
        a_new = normalize_seq(a_new, S)
        a = a_new

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 50 == 0:
            print(f"    VarRed it={it}: max_conv={best_score:.8f}, core_var={((core-core_mean)**2).mean():.6f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_seq(a, S)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < S * 1e-12:
                continue

            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                if 0 <= kl - i < n:
                    j_candidates.add(kl - i)
                if 0 <= i - kl < n:
                    j_candidates.add(i - kl)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:15]:
                if j == i:
                    continue
                aj = a[j]

                h = np.zeros(size)
                lo1 = max(0, j)
                hi1 = min(size, j + n)
                if hi1 > lo1:
                    h[lo1:hi1] += a[0:hi1 - lo1]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    h[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    h[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    h[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 40
                deltas = np.linspace(0.0, ai * 0.98, n_samples)
                for d in deltas:
                    if d < 1e-14:
                        continue
                    new_conv = conv + d * h
                    if 2 * i < size:
                        new_conv[2 * i] += -2.0 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] += 2.0 * aj * d + d * d
                    if (i + j) < size:
                        new_conv[i + j] += -2.0 * d * d
                    v = new_conv.max()
                    if v < best_val - 1e-15:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = normalize_seq(a, S)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_imp += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 2:
                n_pert = max(10, n // 25)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = normalize_seq(a, S)
        else:
            no_improve = 0

        if verbose and total_imp % 100 == 0 and total_imp > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_imp} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-5.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.0
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.3 + 0.2 * np.sin(4 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(5 * np.pi * x) * np.cos(3 * np.pi * x)
    elif choice == 8:
        a = np.sinc(2.5 * (x - 0.5)) ** 2 + 0.05
    elif choice == 9:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.08)
    elif choice == 10:
        a = 1.0 / (1.0 + np.exp(-8.0 * (x - 0.5))) * 0.5 + 0.3
    else:
        a = 0.4 + 0.4 * np.abs(np.sin(np.pi * x * 1.7)) + 0.2 * np.cos(2 * np.pi * 4.3 * x)
    a = np.maximum(a, 0.0)
    a = a + 0.02 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE GD warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_work = best_seq.copy()
        a_out, _ = lse_gd(a_work, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Variance reduction
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: Variance Reduction n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = variance_reduction_step(best_seq, t3, verbose=True)
        try_update(a_out, "varred")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.22, 350.0)
    print(f"\n=== Phase 4: LP Nesterov r2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 5: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Pair CD n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = pair_cd(best_seq, t5, rng, verbose=True)
        try_update(a_out, "pcd1")

    # Phase 6: LSE + CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 6: LSE polish n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = lse_gd(best_seq, t6, verbose=False)
        try_update(a_out, "lse2")

    remaining = time_budget - (time.monotonic() - t_global)
    t6b = min(remaining * 0.08, 80.0)
    if t6b > 10:
        a_out, _ = pair_cd(best_seq, t6b, rng, verbose=False)
        try_update(a_out, "pcd2")

    # Phase 7: Try different length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_alt = n_base + 300
        t7 = min(remaining * 0.30, 250.0)
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={t7:.0f}s) ===")
        if t7 > 30:
            init_alt = interpolate_seq(best_seq, n_alt)
            t7a = min(t7 * 0.5, 100.0)
            a_out, _ = lp_nesterov_optimize(init_alt, t7a, solver, verbose=False)
            try_update(a_out, f"lp_alt{n_alt}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                n_alt2 = n_base - 200
                if n_alt2 >= 800:
                    t7b = min(remaining * 0.3, 80.0)
                    init_alt2 = interpolate_seq(best_seq, n_alt2)
                    a_out2, _ = lp_nesterov_optimize(init_alt2, t7b, solver, verbose=False)
                    try_update(a_out2, f"lp_alt{n_alt2}")

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t8 = remaining * 0.5
        a_out, _ = lse_gd(best_seq, min(t8, 60.0), verbose=False)
        try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = pair_cd(best_seq, min(remaining * 0.6, 50.0), rng, verbose=False)
            try_update(a_out, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
