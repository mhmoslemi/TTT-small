def smooth_obj_and_grad(b, tau, n):
    """Compute smoothed objective and gradient w.r.t. log-parameters b.
    a = softmax(b) * sqrt(2n) (normalized to sum = sqrt(2n))
    Objective: 2n * (1/tau) * log(sum exp(tau * conv_k))
    """
    a = np.exp(b - b.max())
    a_sum = a.sum()
    a = a / a_sum * np.sqrt(2.0 * n)

    conv = fast_autocorr(a)
    max_c = conv.max()
    exp_vals = np.exp(tau * (conv - max_c))
    p_sum = exp_vals.sum()
    if p_sum < 1e-300:
        f = 2.0 * n * max_c
        return f, np.zeros(n)

    p = exp_vals / p_sum
    f = 2.0 * n * ((1.0 / tau) * math.log(p_sum) + max_c)

    # Gradient w.r.t. a: g_a[i] = 2 * (p * rev(a))[n-1+i]
    a_rev = a[::-1]
    size = 2 * n - 1
    fft_size = 1 << (size + n - 2).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a_rev, fft_size)
    conv_pa = irfft(P * A, fft_size)
    g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

    # Gradient w.r.t. b (log-softmax chain rule):
    # a = exp(b - log_sum) * S where S = sqrt(2n)
    # Actually a = exp(b)/sum(exp(b)) * S
    # da[i]/db[j] = a[i]/S * (delta_ij - a[j]/S) ... let me redo.
    # Let e = exp(b), a = e/sum(e) * S
    # da[i]/db[j] = S * (delta_ij * e[i] * sum(e) - e[i] * e[j]) / sum(e)^2
    #              = a[i] * (delta_ij - a[j] / S)
    # df/db[j] = sum_i df/da[i] * da[i]/db[j]
    #          = sum_i g_a[i] * a[i] * (delta_ij - a[j]/S)
    #          = g_a[j]*a[j] - a[j]/S * sum_i g_a[i]*a[i]
    S = np.sqrt(2.0 * n)
    weighted_sum = np.dot(g_a, a)
    g_b = g_a * a - a * (weighted_sum / S)

    return f, g_b


def smooth_lbfgs(a0, time_limit, verbose=True):
    """Smoothed L-BFGS via log-softmax parameterization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10)
    b = np.log(a / a.sum())  # log-softmax parameter
    norm = np.sqrt(2.0 * n)
    b = b - b.mean()  # center for numerical stability

    best_a = normalize_a(a0)
    best_score = fast_autocorr(best_a).max()
    t0 = time.monotonic()

    taus = [50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]
    b_cur = b.copy()

    for tau in taus:
        if time.monotonic() - t0 > time_limit:
            break
        tau_budget = (time_limit - (time.monotonic() - t0)) / (len(taus) - taus.index(tau))
        tau_budget = min(tau_budget, time_limit - (time.monotonic() - t0))
        if tau_budget < 5:
            break

        if verbose:
            print(f"    Smoothed L-BFGS tau={tau:.0f}, budget={tau_budget:.0f}s")

        def obj_wrap(bv):
            f, g = smooth_obj_and_grad(bv, tau, n)
            return f, g

        try:
            res = _minimize_lbfgs(obj_wrap, b_cur, time_limit=tau_budget)
            if res is not None:
                b_cur = res
                a_test = np.exp(b_cur - b_cur.max())
                a_test = a_test / a_test.sum() * norm
                sc = fast_autocorr(a_test).max()
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_test.copy()
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at tau={tau}: {e}")

    return best_a, best_score


def _minimize_lbfgs(func, x0, time_limit):
    """Custom L-BFGS with time limit."""
    from scipy.optimize import minimize
    t0 = time.monotonic()
    state = {'done': False}

    def callback(xk):
        if time.monotonic() - t0 > time_limit * 0.9:
            state['done'] = True
        return state['done']

    res = minimize(func, x0, method='L-BFGS-B', callback=callback,
                   options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-8,
                            'maxfun': 2000})
    return res.x if res.x is not None else x0


def water_filling_cd(a0, time_limit, rng):
    """Coordinate descent with water-filling: adjust a[j] to reduce the max peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 5

    passes = 0
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        passes += 1
        improved = False
        conv = fast_autocorr(a)
        k_star = int(np.argmax(conv))
        M_curr = conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            if aj < 1e-10:
                continue

            # Compute effect of changing a[j]
            # (a*a)[k] involves a[j] in terms: a[j]*a[k-j] and a[k-j]*a[j]
            # Change a[j] by delta: new_conv[k] = conv[k] + 2*delta*a[k-j] + delta^2*(1 if k==2j)
            # (approximately, for the terms involving a[j])
            # More precisely:
            # Let c[k] = sum of a[i]*a[k-i] where i != j and k-i != j
            # Then (a*a)[k] = c[k] + a[j]*a[k-j] + a[k-j]*a[j] - (a[j]^2 if k==2j)
            # = c[k] + 2*a[j]*a[k-j] (if k!=2j) or c[k] + a[j]^2 (if k==2j)
            # After changing a[j] to x:
            # new_conv[k] = c[k] + 2*x*a[k-j] (if k!=2j) or c[k] + x^2 (if k==2j)

            # Compute c[k] = conv[k] - contribution of a[j]
            c = conv.copy()
            for k in range(size):
                j1 = k - j
                if 0 <= j1 < n:
                    c[k] -= a[j] * a[j1]
                j2 = j - k
                if 0 <= j2 < n and j2 != j1:
                    c[k] -= a[j2] * a[j]

            # Now minimize max_k (c[k] + 2*x*a[k-j] + x^2*delta_{k,2j}) subject to
            # sum(a) - aj + x = norm (after renormalization)
            # Since we renormalize, score = 2n * max(new_conv) / (sum)^2
            # Let's just try several x values

            s_others = norm - aj  # after normalization, sum(a)=norm
            candidates = [0.0, aj * 0.3, aj * 0.5, aj * 0.7, aj * 1.0,
                          aj * 1.3, aj * 1.5, s_others * 0.5, s_others * 0.2]
            candidates = [x for x in candidates if 0 <= x <= norm * 1.5]

            best_x = aj
            best_obj = float('inf')
            for x in candidates:
                new_conv = c.copy()
                for k in range(size):
                    j1 = k - j
                    if 0 <= j1 < n:
                        new_conv[k] += x * a[j1]
                    j2 = j - k
                    if 0 <= j2 < n and j2 != j1:
                        new_conv[k] += x * a[j2]
                new_sum = s_others + x
                if new_sum < 1e-10:
                    continue
                obj = 2.0 * n * new_conv.max() / (new_sum * new_sum)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                a = normalize_a(a)
                new_score = fast_autocorr(a).max()
                cur_score = 2.0 * n * new_score / (a.sum() ** 2)
                if cur_score < best_score - 1e-10:
                    best_score = cur_score
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = aj
                    a = normalize_a(a)

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    WF-CD pass={passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def gradient_polish(a0, time_limit):
    """Simple projected gradient with adaptive step size on the smoothed objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.1 / math.sqrt(n)
    patience = 0

    for t in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 300:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        # Find active peaks
        thresh = M * 0.95
        active = np.where(conv >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Combined gradient
        grad = np.zeros(n)
        for k in active:
            w = (conv[k] - thresh) / max(M - thresh, 1e-10) if M > thresh else 1.0
            w = max(w, 0.1)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            indices = np.arange(lo, hi)
            grad[indices] += w * 2.0 * a[k - indices]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s > 1e-15:
            a_trial = a_trial * norm / s
        else:
            a_trial = a.copy()

        sc = fast_autocorr(a_trial).max()
        if sc < best_score - 1e-12:
            a = a_trial
            best_score = sc
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.02, 5.0 / math.sqrt(n))
        else:
            patience += 1
            step *= 0.97
            if step < 1e-12:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_seq(a, rng, scale=0.03):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    a_p = a * noise
    s = a_p.sum()
    if s < 1e-15:
        return a.copy()
    return a_p / s
