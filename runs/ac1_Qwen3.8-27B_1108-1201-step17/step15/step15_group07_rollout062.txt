        conv_new = fast_autocorr(a_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            if verbose:
                print(f"    LinLP iter {it}: score={best_score:.10f}")
        else:
            if verbose:
                print(f"    LinLP iter {it}: no improvement, stopping")
            break

        if abs(score_new - best_score) < tol:
            break

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_patience=60):
    """LP best-response with Nesterov acceleration and adaptive restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    patience = 0
    restart_count = 0
    momentum_t = 1.0
    rng = np.random.default_rng(777)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 20:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive step size: use smaller steps as we get closer to convergence
        if n_iters < 50:
            step_scale = 1.0
        elif n_iters < 200:
            step_scale = 0.7
        else:
            step_scale = 0.4

        alpha = step_scale * 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 3.0:
            a_new = g.copy()
        else:
            momentum = ((momentum_t - 1.0) / (momentum_t + 2.0)) * step_scale
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        # More aggressive restarts
        if patience > max_patience and n_iters % 20 == 0:
            # Try different restart strategies
            strategy = rng.integers(3)
            if strategy == 0:
                y = best_a.copy()
                a = best_a.copy()
            elif strategy == 1:
                # Perturb the best
                noise = rng.laplace(0, norm * 0.02, n)
                y = np.maximum(best_a + noise, 0)
                s = y.sum()
                if s > 1e-15:
                    y = y * norm / s
                a = y.copy()
            else:
                # Mix with uniform
                uniform = np.ones(n) * norm / n
                y = 0.7 * best_a + 0.3 * uniform
                a = y.copy()

            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbose and restart_count % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, restarts={restart_count}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """Polish using subgradient of max convolution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, fast_autocorr(a).max()
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    eta = 0.1
    patience = 0
    rng = np.random.default_rng(55555)

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()

        # Find active lags
        threshold = max_val * 0.9995
        active_mask = conv >= threshold
        active_indices = np.where(active_mask)[0]
        if len(active_indices) == 0:
            active_indices = np.array([np.argmax(conv)])
        if len(active_indices) > n:
            active_indices = np.argsort(conv)[-n:]

        # Compute subgradient: grad = (1/|K|) * sum_{k in K} grad_k
        # grad_k[j] = a[k-j] if 0<=k-j<n else 0 (for the term a[j]*a[k-j])
        # Actually (a*a)[k] = sum_j a[j]*a[k-j], so d/d(a[j]) = a[k-j] + a[j-k] (symmetric)
        # But for linear convolution: d(a*a)/da[j] at lag k = a[k-j] (if valid) + a[j-k] (if valid)
        # Since conv is symmetric, we can just use: grad[j] = sum_{k in K} (a[k-j] if 0<=k-j<n) + (a[j-k] if 0<=j-k<n and j!=k)
        # Simpler: use the convolution matrix row
        grad = np.zeros(n)
        for k in active_indices:
            # (a*a)[k] = sum_j a[j] * a[k-j] for 0<=j<n, 0<=k-j<n
            # d/d(a[j]) = a[k-j] (from the j term) + a[j-k] (from the k-j term, if j != k-j)
            # Actually for the full derivative: d/d(a[j]) sum_i a[i]*a[k-i] = a[k-j] + a[j-k] (if both valid)
            for j in range(n):
                k_minus_j = k - j
                j_minus_k = j - k
                coeff = 0.0
                if 0 <= k_minus_j < n:
                    coeff += a[k_minus_j]
                if 0 <= j_minus_k < n and j_minus_k != k_minus_j:
                    coeff += a[j_minus_k]
                elif 0 <= j_minus_k < n and j_minus_k == k_minus_j:
                    coeff += a[j_minus_k]  # double count when k=2j
                grad[j] += coeff

        grad /= len(active_indices)

        # Project gradient onto the tangent space (sum=0)
        grad -= np.mean(grad)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Take a step
        a_new = a - eta * grad / grad_norm * norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        conv_new = fast_autocorr(a_new)
        score_new = conv_new.max()

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            eta = min(eta * 1.1, 1.0)
        else:
            patience += 1
            if patience % 50 == 0:
                eta *= 0.8
            if patience > 200:
                # Restart with perturbation
                noise = rng.normal(0, norm * 0.01, n)
                a = np.maximum(best_a + noise, 0)
                s = a.sum()
                a = a * norm / s
                patience = 0
                eta = 0.1

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it} ({elapsed:.0f}s): best={best_score:.10f}, eta={eta:.4f}")

    return best_a, best_score


def mirror_descent_polish(a0, time_limit, top_k=30, verbose=True):
    """Mirror descent with multiplicative updates for polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, fast_autocorr(a).max()
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    eta = 0.5 / np.sqrt(n)
    patience = 0
    rng = np.random.default_rng(88888)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()

        k_actual = min(top_k, size)
        active_idx = np.argsort(conv)[-k_actual:]

        # Subgradient
        grad = np.zeros(n)
        for k in active_idx:
            for j in range(n):
                kmj = k - j
                jmk = j - k
                c = 0.0
                if 0 <= kmj < n:
                    c += a[kmj]
                if 0 <= jmk < n:
                    c += a[jmk]
                grad[j] += c
        grad /= len(active_idx)
        grad -= np.mean(grad)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Multiplicative weights update
        best_a_try = a.copy()
        best_conv_max = max_val
        for factor in [0.3, 0.5, 1.0, 2.0]:
            eta_try = eta * factor
            a_try = a * np.exp(-eta_try * grad)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            conv_try = fast_autocorr(a_try).max()
            if conv_try < best_conv_max:
                best_conv_max = conv_try
                best_a_try = a_try
                eta = eta_try

        a = best_a_try

        if best_conv_max < best_score - 1e-13:
            best_score = best_conv_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 200 == 0:
                eta *= 0.85
            if patience > 500:
                noise = rng.laplace(0, norm * 0.005, n)
                a_pert = np.maximum(a + noise, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                patience = 0
                eta = 0.5 / np.sqrt(n)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def make_cosine_init(n, alpha=0.5, freq=1, p=1.0, seed=42):
    """Create a cosine-modulated initialization."""
    i = np.arange(n)
    t = i / n
    base = 1.0 + alpha * np.cos(2 * np.pi * freq * t)
    base = np.maximum(base, 0.01)
    a = base ** p
    s = a.sum()
    return a / s


def make_u_shape_init(n, gamma=2.0):
    """Create a U-shaped initialization (large at ends)."""
    i = np.arange(n)
    t = i / n
    a = 1.0 / (1.0 + gamma * np.sin(np.pi * t) ** 2)
    s = a.sum()
    return a / s


def make_random_init(n, seed=123):
    """Create a random non-negative initialization."""
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s
