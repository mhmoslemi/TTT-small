"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based subgradient descent with multiplicative updates
2. Softmax-weighted active set for smooth subgradients
3. Nesterov acceleration with geometric momentum
4. Adaptive step size with backtracking
5. Coordinate descent polish for local refinement
6. Multi-length exploration with interpolation
7. Asymmetry enforcement (symmetric sequences proven to have score >= 2)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Compute score locally (fast, no clipping)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))

def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def compute_subgradient(a, T):
    """
    Compute the subgradient of max_k (a*a)[k] using softmax weighting.
    Returns the gradient vector (length n) and the max value.
    """
    n = len(a)
    conv = fast_autocorr(a)
    max_conv = conv.max()
    
    if T <= 1e-12:
        # Use only the max lag(s)
        active = np.where(conv >= max_conv * 0.999)[0]
        w = np.zeros(2*n-1)
        w[active] = 1.0 / len(active)
    else:
        # Softmax weighting
        shifted = conv - max_conv  # For numerical stability
        w = np.exp(shifted / T)
        w /= w.sum()
    
    # Compute sg[i] = 2 * sum_m w[i+m] * a[m] for i=0,...,n-1
    # This is a cross-correlation between w (length 2n-1) and a (length n)
    fft_size = 1 << (3*n - 2).bit_length()  # Need at least (2n-1) + n - 1
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    # Actually: h[i] = sum_m w[i+m] * a[m]
    # This is (w correl a)[i] = ifft(FFT(w) * conj(FFT(a)))[i]
    H = irfft(W * np.conj(A), fft_size)[:n]
    sg = 2.0 * H
    
    return sg, max_conv, conv, w

def subgradient_optimize(a0, time_limit, T_decay=0.999, eta_init=1.0, 
                          nesterov=True, verbose=True, seed=0, 
                          T_ratio=0.05):
    """
    Subgradient optimization with multiplicative updates.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * norm / s
    
    conv = fast_autocorr(a)
    best_score = conv.max()  # With normalization sum=norm, score = max_conv
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = T_ratio * best_score  # Initial temperature
    eta = eta_init / (n * max(best_score, 1e-10))
    
    # Nesterov state
    y = a.copy()
    a_prev = a.copy()
    
    iter_count = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Nesterov extrapolation
        if nesterov and iter_count > 1:
            momentum = (iter_count - 1) / (iter_count + 2)
            # Geometric Nesterov: y = a * (a/a_prev)^(momentum/2) 
            # But this can be unstable. Use arithmetic on log scale.
            # Actually, for multiplicative updates, the "natural" Nesterov is:
            # y[i] = a[i] * (a[i]/a_prev[i])^(momentum)
            # But this can create zeros. Let's use a simpler approach:
            # y = a + momentum * (a - a_prev) (arithmetic Nesterov on the original scale)
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 1e-30)
            y = y * norm / y.sum()
        
        # Compute subgradient at y
        sg, max_conv, conv_y, w = compute_subgradient(y, T)
        
        # Multiplicative update
        a_prev = a.copy()
        a_new = a * np.exp(-eta * sg / max_conv)
        a_new = np.maximum(a_new, 1e-30)
        
        # Renormalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        # Check improvement
        conv_new = fast_autocorr(a_new)
        score_new = conv_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        # Adaptive step size
        if score_new > best_score * 1.01:
            eta *= 0.5  # Decrease
        elif score_new < best_score * 0.99:
            eta *= 1.1  # Increase cautiously
        
        # Decay temperature
        T *= T_decay
        T = max(T, 1e-6 * best_score)
        
        # Periodic restart if stalled
        if stall_count > 200:
            a = best_a.copy()
            a_prev = a.copy()
            y = a.copy()
            eta = eta_init / (n * max(best_score, 1e-10))
            T = T_ratio * best_score
            stall_count = 0
        
        if verbose and iter_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={iter_count} ({elapsed:.0f}s): best={best_score:.10f}, T={T:.6f}, eta={eta:.2e}")
    
    return best_a, best_score

def coord_descent(a0, time_limit, rng, verbose=False):
    """Coordinate descent with line search for local refinement."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            idx = perm[i]
            ai = a[idx]
            if ai < 1e-15:
                continue
                
            # Try a few perturbation levels
            best_val = best_score
            best_new = a.copy()
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a_try = a.copy()
                a_try[idx] = ai * factor
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                cv = fast_autocorr(a_try)
                val = cv.max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_new = a_try.copy()
            
            if best_val < best_score - 1e-12:
                a = best_new
                best_score = best_val
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def pair_swap_cd(a0, time_limit, rng, verbose=False, n_trials=6):
    """Pair swap coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(0, n, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx % n]
            j = perm[(idx + 1) % n]
            if i == j:
                continue
            ai = a[i]
            aj = a[j]
            if ai < 1e-15 and aj < 1e-15:
                continue
            
            best_val = best_score
            best_new = a.copy()
            
            # Try moving mass between i and j
            for frac in np.linspace(0.1, 0.9, n_trials):
                # Move frac*ai from i to j
                d = ai * frac
                a_try = a.copy()
                a_try[i] = ai - d
                a_try[j] = aj + d
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    val = fast_autocorr(a_try).max()
                    if val < best_val - 1e-12:
                        best_val = val
                        best_new = a_try.copy()
                
                # Move frac*aj from j to i
                d2 = aj * frac
                a_try2 = a.copy()
                a_try2[i] = ai + d2
                a_try2[j] = aj - d2
                s_try2 = a_try2.sum()
                if s_try2 > 1e-15:
                    a_try2 = a_try2 * norm / s_try2
                    val2 = fast_autocorr(a_try2).max()
                    if val2 < best_val - 1e-12:
                        best_val = val2
                        best_new = a_try2.copy()
            
            if best_val < best_score - 1e-12:
                a = best_new
                best_score = best_val
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
    
    return best_a, best_score
