def fast_autocorr(a):

def normalize_to_sum(a, target):
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a * target / s


def lp_solve_full(a_fixed, M, n, solver):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for all k, b >= 0.
    Uses all 2n-1 constraints."""
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix (sparse for efficiency)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            rows.append(k)
            cols.append(j)
            vals.append(a_fixed[k - j])
    
    T = cp.SparseMatrix((vals, (rows, cols)), shape=(size, n))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def fast_autocorr_cross(a, b):
