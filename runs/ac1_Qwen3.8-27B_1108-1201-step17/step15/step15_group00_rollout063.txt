def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_to(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)

def exact_subgradient(a, ac, tol=0.01):
    """Compute exact subgradient using active set (peaks within tol*max)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    # Active set: peaks within (1-tol) of max
    active = np.where(ac >= M * (1.0 - tol))[0]
    if len(active) == 0:
        active = np.array([np.argmax(ac)])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    g /= len(active)
    return g

def soft_subgradient(a, ac, top_k=30, beta=4.0):
    """Compute soft subgradient using top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    M = top_vals[0]
    w = (top_vals / M) ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-10:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, m_margin=0.0):
    """LP Frank-Wolfe with optional margin on M."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max() * (1.0 + m_margin)
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix sparsely
        b_var = cp.Variable(n, nonneg=True)
        
        # Use vectorized approach for constraints
        # For each lag k, constraint: sum_i y[i] * b[k-i] <= M_curr
        # This is: (T @ b)_k <= M_curr where T[k, i] = y[i] if 0 <= k-i < n
        
        # For efficiency, build T as a banded matrix
        T_rows = []
        T_cols = []
        T_vals = []
        row_idx = []
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                T_rows.append(k)
                T_cols.append(i)
                T_vals.append(y[i])
                row_idx.append(k)
        
        from scipy.sparse import csr_matrix
        T = csr_matrix((T_vals, (T_rows, T_cols)), shape=(2*n-1, n))
        
        constraints = [T @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        # Wait, if sum b = 1, then max sum b = 1 always. That's wrong.
        
        # Actually the LP should be:
        # max sum b
        # subject to b >= 0
        #            T @ b <= M_curr * 1
        # (no normalization constraint on b)
        # Then we normalize b / sum(b) to get the direction.
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_curr])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        
        my = fast_autocorr(y_new).max()
        if my < best_m - 1e-14:
            best_m = my
            best_a = y_new.copy()
        
        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a

def coord_descent(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with multi-fraction exploration."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Find peaks
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            best_dst = -1
            best_score = float('inf')
            
            # Sample candidate destinations
            candidates = rng.integers(0, n, size=min(n, 200))
            for dst in candidates:
                if dst == src:
                    continue
                # Sensitivity estimate
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
            
            if not found:
                pass
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    return best_a

def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_to(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)

def softmax(z):
    z_shifted = z - z.max()
    ez = np.exp(z_shifted)
    return ez / ez.sum()

def power_sum_obj_grad(z, p, n):
    """Objective and gradient for 2n * (sum conv^p)^(1/p) with softmax parameterization."""
    a = softmax(z)
    ac = fast_conv(a)
    ac_clipped = np.maximum(ac, 0.0)
    
    if p == 2:
        obj_val = 2.0 * n * np.sqrt(np.sum(ac_clipped ** 2))
        # grad w.r.t. a_j: 2n * (sum ac^2)^(-1/2) * ac_j * d(ac_j)/da... 
        # Actually: f = 2n * ||ac||_2, df/da_j = 2n * (ac . grad_ac/da_j) / ||ac||_2
        # grad_ac_k / da_j = 2*a[k-j] (if valid)
        # So df/da_j = 2n / ||ac||_2 * sum_k ac_k * 2*a[k-j]
        ac_norm = np.sqrt(np.sum(ac_clipped ** 2))
        if ac_norm < 1e-30:
            return obj_val, np.zeros(n)
        # S2_j = sum_k ac_k * a[k-j]
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A_f = np.fft.rfft(a, fft_size)
        AC_f = np.fft.rfft(ac, fft_size)
        # S2_j = sum_k ac_k * a[k-j] = cross-correlation
        # = ifft(conj(FFT(a)) * FFT(ac)) ... need to verify
        # Actually: S2_j = sum_k ac_k * a[k-j]
        # Let i = k-j: S2_j = sum_i ac_{i+j} * a_i
        # This is (ac_star a)[j] where * is correlation of ac and a
        # FFT: ifft(conj(FFT(ac)) * FFT(a)) [j] = sum_i ac[i+j] * conj(a[i]) = sum_i ac[i+j]*a[i] for real
        S2 = np.fft.irfft(np.conj(AC_f) * A_f, fft_size)[:2*n-1]
        S2 = S2[:n]
        grad_a = (2.0 * n / ac_norm) * 2.0 * S2
    else:
        ac_pow = ac_clipped ** (p - 1)
        sum_pow = np.sum(ac_clipped ** p)
        if sum_pow < 1e-30:
            return 0.0, np.zeros(n)
        obj_val = 2.0 * n * sum_pow ** (1.0 / p)
        # grad_a_j = 2n * sum_pow^(1/p - 1) * sum_k ac_k^(p-1) * 2*a[k-j]
        # = 2n * sum_pow^(1/p-1) * 2 * S2_j where S2_j = sum_k ac_k^(p-1) * a[k-j]
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A_f = np.fft.rfft(a, fft_size)
        WP_f = np.fft.rfft(ac_pow, fft_size)
        S2 = np.fft.irfft(np.conj(WP_f) * A_f, fft_size)[:2*n-1]
        S2 = S2[:n]
        grad_a = 2.0 * n * sum_pow ** (1.0/p - 1.0) * 2.0 * S2
    
    # Chain rule: da_j/dz_j = a_j(1-a_j), da_j/dz_i = -a_j*a_i for i!=j
    # grad_z_j = sum_i grad_a_i * da_i/dz_j = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    return obj_val, grad_z

def lbbfgs_optimize(a0, time_limit, p_start=2, p_end=16, verbose=True):
