        obj, grad, w = smooth_obj_and_grad(a, T, fft_size)

        # Nesterov momentum
        step = norm / (n * T * 0.8)
        a_new = project_simplex(a - step * grad, norm)

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stuck_count = 0

            if verbose and int(T * 100) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SmoothNest T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
        else:
            stuck_count += 1

        a = a_new

        # Adapt temperature
        if stuck_count > 50:
            T = max(T * 0.7, T_min)
            stuck_count = 0
        elif stuck_count > 20:
            T = max(T * 0.85, T_min)
        else:
            T = max(T * T_decay, T_min)

        # Periodic restart: try a perturbed version
        if stuck_count > 100:
            a_pert = a + 0.05 * norm * np.random.randn(n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert * norm / a_pert.sum()
            score_pert = fast_autocorr(a_pert).max()
            if score_pert < best_score * 1.05:
                a = a_pert
            T = max(T * 1.5, 1.0)
            stuck_count = 0

    return best_a, best_score

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T_mat = np.zeros((size, n))
    for i in range(n):
        T_mat[i:i+n, i] = a
    return T_mat

def lp_best_response(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    T_mat = build_conv_matrix(a)
    if active_idx is not None:
        T_use = T_mat[active_idx, :]
    else:
        T_use = T_mat
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lp_nesterov(a0, time_limit, solver, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Constraint generation: top constraints
        thresh = 0.9 * M_curr
        active = np.where(conv_y >= thresh)[0]
        if len(active) < 5:
            active = np.where(conv_y >= 0.7 * M_curr)[0]
        if len(active) < 3:
            active = None

        b = lp_best_response(y, M_curr, solver, active_idx=active)
        if b is None:
            b = lp_best_response(y, M_curr, solver, active_idx=None)
            if b is None:
                stuck += 1
                if stuck > 50:
                    break
                continue

        stuck = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def peak_shaving(a0, time_limit, rng, verbose=False):
    """Targeted peak shaving: move mass from coordinates contributing to the peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        j_star = np.argmax(conv)

        # Subgradient: g_k = 2*a[j_star - k] (clipped to valid range)
        g = np.zeros(n)
        lo = max(0, j_star - (n - 1))
        hi = min(n - 1, j_star)
        for k in range(lo, hi + 1):
            idx = j_star - k
            if 0 <= idx < n:
                g[k] = 2.0 * a[idx]

        # Find coordinate with highest gradient (most contributes to peak)
        # and coordinate with lowest gradient (least contributes)
        g_norm = g / (g.max() + 1e-30)

        # Move mass from high-gradient to low-gradient coordinates
        src_candidates = np.where((g_norm > 0.8) & (a > norm * 1e-6))[0]
        dst_candidates = np.where((g_norm < 0.3) & (a < norm * 0.1 / n))[0]

        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            # Broader search
            src_candidates = np.argsort(g_norm)[-10:]
            dst_candidates = np.argsort(g_norm)[:10]

        improved_this_round = False

        for i in range(min(5, len(src_candidates))):
            src = src_candidates[i]
            if a[src] < norm * 1e-6:
                continue

            for j in range(min(5, len(dst_candidates))):
                dst = dst_candidates[j]
                if dst == src:
                    continue

                # Try several step sizes
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    d = a[src] * frac
                    if d < 1e-15:
                        break
                    a_test = a.copy()
                    a_test[src] -= d
                    a_test[dst] += d
                    a_test = np.maximum(a_test, 0)
                    s_test = a_test.sum()
                    if s_test < 1e-15:
                        break
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < best_score - 1e-12:
                        a = a_test
                        conv = conv_test
                        best_score = conv.max()
                        best_a = a.copy()
                        improved_this_round = True
                        no_improve = 0
                        break

        if not improved_this_round:
            no_improve += 1

            # Try random perturbations
            if no_improve % 10 == 0:
                for _ in range(5):
                    a_pert = a + 0.02 * norm * rng.standard_normal(n)
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert * norm / a_pert.sum()
                    conv_pert = fast_autocorr(a_pert)
                    if conv_pert.max() < best_score:
                        a = a_pert
                        conv = conv_pert
                        best_score = conv_pert.max()
                        best_a = a.copy()
                        no_improve = 0
                        improved_this_round = True
                        break

        if verbose and (time.monotonic() - t0) % 30 < 1:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def generate_structured_init(n, kind, rng):
    if kind == 'uniform':
        return np.ones(n)
    elif kind == 'gaussian':
        x = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        return np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif kind == 'raised_cosine':
        x = np.linspace(0, np.pi, n)
        return np.cos(x - np.pi / 2) ** 2
    elif kind == 'u_shape':
        x = np.linspace(-1, 1, n)
        return 1.0 + x ** 2
    elif kind == 'bimodal':
        x = np.arange(n)
        c1 = n * 0.25
        c2 = n * 0.75
        sigma = n / 8.0
        return np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    elif kind == 'ramp':
        x = np.arange(n)
        return (x + 1.0) / n
    elif kind == 'comb':
        a = np.zeros(n)
        a[::2] = 1.0
        return a
    elif kind == 'random':
        return rng.random(n)
    elif kind == 'truncated_gaussian':
        x = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 3.0
        vals = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        # Zero out edges
        vals[:n // 8] = 0.1 * vals[n // 8]
        vals[-(n // 8):] = 0.1 * vals[-(n // 8) + 1] if n // 8 > 1 else vals[0]
        return vals
    else:
        return np.ones(n)

def main_search(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    safety = 90.0
    time_budget = min(float(budget_s) - safety, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    # === Phase 0: Structured initializations (20s) ===
    print("\n=== Phase 0: Structured inits ===")
    t_phase = 20.0
    kinds = ['gaussian', 'raised_cosine', 'u_shape', 'bimodal', 'truncated_gaussian', 'uniform', 'comb', 'ramp']
    for kind in kinds:
        if time.monotonic() - t_start > time_budget - t_phase:
            break
        a_init = generate_structured_init(n_base, kind, rng)
        a_init = a_init / a_init.sum()
        do_update(a_init, f"init_{kind}")

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: score={best_score:.6f}")

    # === Phase 1: Smooth Nesterov (35% of budget) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: Smooth Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = smooth_nesterov(a_norm, t1, verbose=True)
        do_update(a_out, "smooth1")

    # === Phase 2: LP Nesterov (25% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: LP Nesterov n={n_base} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t2, solver, verbose=True)
        do_update(a_out, "lp1")

    # === Phase 3: Peak shaving (15% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Peak Shaving n={n_base} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = peak_shaving(a_norm, t3, rng, verbose=True)
        do_update(a_out, "peakshave1")

    # === Phase 4: Smooth Nesterov round 2 (15% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Smooth Nesterov R2 n={n_base} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = smooth_nesterov(a_norm, t4, verbose=True)
        do_update(a_out, "smooth2")

    # === Phase 5: LP Nesterov round 2 (10% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: LP Nesterov R2 n={n_base} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t5, solver, verbose=True)
        do_update(a_out, "lp2")

    # === Phase 6: Diversified restarts (10% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: Diversified restarts (budget={t6:.0f}s) ===")
    if t6 > 40:
        n_restarts = 3
        t_per = t6 / n_restarts
        for r in range(n_restarts):
            if time.monotonic() - t_start > time_budget - 15:
                break
            a_pert = best_seq.copy()
            noise_scale = 0.15 + 0.1 * r
            noise = rng.normal(0, noise_scale * a_pert.std(), len(a_pert))
            a_pert = a_pert + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            do_update(a_pert, f"perturb{r}")

            a_norm_p = a_pert * np.sqrt(2.0 * len(a_pert)) / a_pert.sum()
            a_out, _ = smooth_nesterov(a_norm_p, min(t_per * 0.6, 40.0), verbose=False)
            do_update(a_out, f"restart_smooth{r}")

            remaining_sub = time_budget - (time.monotonic() - t_start)
            if remaining_sub > 20:
                a_out2, _ = peak_shaving(a_norm_p, min(t_per * 0.4, 25.0), rng, verbose=False)
                do_update(a_out2, f"restart_peak{r}")

    # === Phase 7: Try alternate lengths ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 60:
        for n_alt in [n_base + 200, n_base - 200, 2048, 1280]:
            if n_alt < 200:
                continue
            if time.monotonic() - t_start > time_budget - 30:
                break
            print(f"\n=== Alt length n={n_alt} ===")
            # Interpolate from best
            old_n = len(best_seq)
            old_x = np.linspace(0, 1, old_n, endpoint=False)
            new_x = np.linspace(0, 1, n_alt, endpoint=False)
            f_interp = np.interp(new_x, old_x, best_seq, period=None)
            f_interp = np.maximum(f_interp, 0)
            s = f_interp.sum()
            if s < 1e-15:
                f_interp = np.ones(n_alt) / n_alt
            else:
                f_interp = f_interp / s
            do_update(f_interp, f"interp_n{n_alt}")

            if time.monotonic() - t_start > time_budget - 40:
                break

            a_norm_alt = f_interp * np.sqrt(2.0 * n_alt) / f_interp.sum()
            t_alt = min(remaining * 0.2, 80.0)
            a_out_alt, _ = smooth_nesterov(a_norm_alt, t_alt, verbose=False)
            do_update(a_out_alt, f"smooth_n{n_alt}")

            remaining_sub = time_budget - (time.monotonic() - t_start)
            if remaining_sub > 30:
                a_out_alt2, _ = lp_nesterov(a_norm_alt, min(remaining_sub * 0.3, 50.0), solver, verbose=False)
                do_update(a_out_alt2, f"lp_n{n_alt}")

    # === Final peak shaving with whatever time remains ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 10 and best_seq is not None:
        print(f"\n=== Final polish (budget={remaining:.0f}s) ===")
        n_final = len(best_seq)
        a_norm_f = best_seq * np.sqrt(2.0 * n_final) / best_seq.sum()
        a_out_f, _ = peak_shaving(a_norm_f, remaining * 0.7, rng, verbose=False)
        do_update(a_out_f, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]

def propose_candidate(seed=42, budget_s=1800.0):
    try:
        result = main_search(seed=seed, budget_s=budget_s)
        return result
    except Exception as e:
        print(f"Error in main search: {e}")
        import traceback
        traceback.print_exc()
        # Fallback
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            p = np.clip(p, 0.0, 1000.0)
            if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
                return [float(x) for x in p / p.sum()]
        except Exception:
            pass
        n = 1600
        return [1.0 / n] * n
