                        new_conv_k2 = conv[k2]
                    obj = max(obj, 2.0 * n * new_conv_k2 / (new_sum * new_sum))
                
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                a = a * norm / a.sum()
                new_score = fast_autocorr(a).max()
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    conv = fast_autocorr(a)
                else:
                    a[j] = aj
                    a = a * norm / a.sum()

        if not improved:
            break

        if verbose:
            elapsed = time.monotonic() - t_start
            print(f"    CD ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def perturb_sequence(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

def interpolated_family(a, alpha):
    """
    Interpolate between a and a uniformly shuffled version.
    Actually, let's interpolate between a and a "smoothed" version.
    """
    n = len(a)
    # Smooth version: moving average
    k = max(1, n // 100)
    kernel = np.ones(k) / k
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s < 1e-15:
        a_smooth = a.copy()
    else:
        a_smooth = a_smooth / a_smooth.sum() * a.sum()
    return (1 - alpha) * a + alpha * a_smooth
