def evaluate_sequence(sequence):
    if not isinstance(sequence, list):
        return np.inf
    if len(sequence) == 0:
        return np.inf
    
    # Convert to a numpy array for easier processing
    a = np.asarray(sequence, dtype=np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
        
    n = len(a)
    size = 2 * n - 1
    # Padding to make length a power of 2 for FFT
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)  # Do the convolution
    conv = np.real(conv)  # Sometimes necessary for real numbers
    conv = conv[:size]  # Keep only the convolution of length 2n-1
    
    max_conv_val = np.max(conv) if np.max(conv) > 0 else 0
    sum_sq = s * s
    score = float(2 * n * max_conv_val / sum_sq)
    
    return score if np.isfinite(score) else float('inf')
