                a_try[important[important != i]] += d / max(1, len(important) - 1) * (len(important) - 1) / n_other * n_other

                    a = a_try

# Full coordinate descent (all coordinates, randomized)

def full_cd(a0, time_limit, rng, n_sub=200):
    """Full coordinate descent with random subset per pass."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        # Random subset of coordinates
        subset = rng.choice(n, size=min(n_sub, n), replace=False)
        rng.shuffle(subset)
        
        for i in subset:
            if time.monotonic() - t0 > time_limit:
                break
            
            ai = a[i]
            if ai < 1e-12:
                continue
            
            # Try several perturbation directions
            best_val = best_score
            best_a_try = None
            
            # Direction 1: reduce a[i], distribute to random others
            for frac in [0.05, 0.1, 0.2, 0.3]:
                d = ai * frac
                a_try = a.copy()
                a_try[i] -= d
                # Add to random coordinates
                others = rng.choice(n, size=min(10, n-1), replace=False)
                others = others[others != i]
                if len(others) == 0:
                    continue
                a_try[others] += d / len(others)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                cv = fast_autocorr(a_try)
                val = cv.max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_a_try = a_try
            
            # Direction 2: increase a[i] (if it's small), take from neighbors
            if ai < norm / n:
                for frac in [0.1, 0.2, 0.5]:
                    d = (norm / n - ai) * frac
                    a_try = a.copy()
                    a_try[i] += d
                    # Take from random large coordinates
                    large_idx = np.argsort(a)[::-1][:10]
                    large_idx = large_idx[large_idx != i]
                    if len(large_idx) == 0:
                        continue
                    a_try[large_idx] -= d / len(large_idx)
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try * norm / s_try
                    cv = fast_autocorr(a_try)
                    val = cv.max()
                    if val < best_val - 1e-12:
                        best_val = val
                        best_a_try = a_try
            
            if best_a_try is not None:
                a = best_a_try
                conv = fast_autocorr(a)
                best_score = conv.max()
                best_a = a.copy()
                pass_improved = True
                no_improve = 0
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
    
    return best_a, best_score

# Perturbation function
def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

# Interpolation
def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Binary search on the threshold
    lo = v.min() - S
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.maximum(v - t, 0)

def simplex_project(v, S):
    n = len(v)
    if n == 0:
        return v
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    rho = np.where(sorted_v > 0)[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    cond = (sorted_v[rho] - (S) / (rho + 1)) > 0
    if not np.any(cond):
        return np.full(n, S / n)
    rho_val = rho[np.where(cond)[0][0]] + 1
    t = (np.sum(sorted_v[:rho_val]) - S) / rho_val
    return np.maximum(v - t, 0.0)

                    no_improve = 0
                    break

            if not pass_improved:
                for frac in [0.05, 0.1, 0.2]:
                    d = ai * frac
                    a_try = a.copy()
                    a_try[i] += d
                    large_idx = np.argsort(a)[::-1][:5]
                    large_idx = large_idx[large_idx != i]
                    if len(large_idx) == 0:
                        continue
                    a_try[large_idx] -= d / len(large_idx)
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try * norm / s_try
                    cv = fast_autocorr(a_try)
                    if cv.max() < best_score - 1e-11:
                        a = a_try
                        conv = fast_autocorr(a)
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                        break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 12:
                break

    return best_a, best_score


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
