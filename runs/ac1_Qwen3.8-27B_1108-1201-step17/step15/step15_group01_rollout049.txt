    # This is (a * b_reversed)[i] where b_reversed[k] = b[2n-2-k]
    # Or: grad = conv(a, reverse(b))[n-1 : n-1+n]
    # Simpler: use the direct FFT formula
    # If we define C[i] = sum_j a[j] * b[i+j], then
    # C is the "cross-correlation" and can be computed as:
    # C = ifft(FFT(a_padded) * conj(FFT(b_padded))) [need to be careful]
    # Actually the cleanest way:
    # C[i] = sum_j a[j] b[i+j] = (a * b_flipped)[i + n - 1] where b_flipped[k] = b[len(b)-1-k]
    # Let's just do it with the definition directly using a shifted FFT
    
    # Simplest correct approach:
    # grad[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for valid i+j < 2n-1
    # Create the Toeplitz-like product. For efficiency, use FFT:
    # Define u = a (length n), v = w (length 2n-1)
    # We want g[i] = sum_j u[j] * v[i+j] for i = 0, ..., n-1
    # This equals (u * v)[n-1+i] where * is linear convolution, IF we extend u to 2n-1
    # No, that's not right either.
    # Let's use: g = ifft(conj(FFT(u_zero_padded)) * FFT(v_zero_padded))[0:n]
    # Where u is zero-padded to length >= 3n-2 and v is zero-padded to length >= 3n-2
    # Hmm, this is getting complicated. Let me just use the convolution approach:
    # g[i] = sum_j u[j] v[i+j]
    # = (u * v_rev)[(2n-2) - i + n - 1] ... this is confusing.
    
    # Let me just compute it directly for correctness, and optimize later if needed.
    # For n ~ 1600, a direct O(n^2) computation is 2.5M ops which is fast enough if done rarely.
    # But we need it every iteration, so let's use FFT properly.
    
    # The correct formula: Define w_ext of length >= 3n-2 (zero-padded)
    # u_ext of length >= 3n-2 (zero-padded with u at the beginning)
    # Then g[i] = sum_j u[j] * w[i+j] = (u_ext * w_ext_flipped)[n-1 + i] where flip is index reversal
    # Actually the simplest correct way:
    # g = cross_correlation(a, w) where cross_correlation(a, w)[i] = sum_j a[j] * w[i+j]
    # Using FFT: g[i] = ifft(FFT(a)_conj * FFT(w))[i] for circular case
    # For linear: pad both to length >= n + (2n-1) - 1 = 3n-2
    fft_size2 = 1 << (3 * n - 2 - 1).bit_length()
    u = np.zeros(fft_size2)
    u[:n] = a
    v = np.zeros(fft_size2)
    v[:size] = np.asarray(w, dtype=np.float64)[:size]
    Uf = rfft(u, fft_size2)
    Vf = rfft(v, fft_size2)
    # cross_corr[i] = sum_j u[j] v[j+i] = ifft(conj(Uf) * Vf)[i]
    # Let's verify: ifft(conj(Uf) * Vf)[i] = sum_j u[j] v[j+i] (circular, but linear if padded enough)
    corr = irfft(np.conj(Uf) * Vf, fft_size2)[:n]
    return 2.0 * corr


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    # Find rho: largest k such that v_sorted[k-1] > (css[k-1] - S) / k
    idx = np.arange(1, n + 1)
    rho_mask = v_sorted * idx > (css - S)
    rho_idx = np.where(rho_mask)[0]
    if len(rho_idx) == 0:
        return np.maximum(v, 0.0) * (S / max(v.sum(), 1e-30))
    rho = rho_idx[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_smooth(a0, time_limit, n_temps=20, verbose=True):
    """Nesterov-accelerated projected gradient on smoothed max-convolution objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    size = 2 * n - 1

    # Temperature schedule: start high (smooth), end low (sharp)
    temps = np.geomspace(50.0, 0.005, n_temps)

    for Ti, T in enumerate(temps):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2.0:
            break

        # Steps proportional to remaining time, but at least a minimum
        n_steps = max(20, min(300, int(remaining * 4)))

        # Reset Nesterov momentum at each temperature change for stability
        y = a.copy()
        momentum = 0.0

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute smoothed gradient
            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = compute_grad_smooth(y, w)

            # Adaptive step size
            # Use Lipschitz-like estimate
            lip = 4.0 * norm  # rough upper bound
            step = 1.0 / (lip * (it + 1.0))
            step = min(step, 0.5 / T)

            # Gradient step with projection
            a_candidate = y - step * grad
            a_candidate = project_simplex(a_candidate, norm)

            # Nesterov / heavy ball momentum
            if it == 0:
                beta = 0.0
            else:
                beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
                momentum = momentum * (1.0 - beta)

            y_new = a_candidate + momentum * (a_candidate - y)
            y_new = project_simplex(y_new, norm)

            # Update
            y = y_new
            a = a_candidate

            # Check improvement
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and (Ti % 4 == 0) and (it % max(1, n_steps // 5) == 0):
                elapsed = time.monotonic() - t0
                print(f"    Smooth N=T={T:.4f} it={it}/{n_steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_T_rows(a, active_idx):
    """Build rows of the convolution matrix for active constraints.
    T[i, j] = a[i-j] if 0 <= i-j < n else 0 (for row i of the convolution matrix)
    Actually: (a * b)[i] = sum_j a[i-j] * b[j] where valid range applies.
    So T[i, j] = a[i-j] if 0 <= i-j < n, else 0.
    """
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(active_idx), n))
    for ri, i in enumerate(active_idx):
        # (a*b)[i] = sum_j a[i-j] * b[j] for j where 0 <= i-j < n
        # i.e., j in [max(0, i-n+1), min(i, n-1)]
        j_lo = max(0, i - n + 1)
        j_hi = min(i, n - 1)
        if j_lo <= j_hi:
            T[ri, j_lo:j_hi+1] = a[j_hi - (j_hi - j_lo):-1 if False else None]  # wrong
            # T[ri, j] = a[i-j] for j in [j_lo, j_hi]
            js = np.arange(j_lo, j_hi + 1)
            T[ri, js] = a[i - js]
    return T


def lp_best_response_fast(a, M, solver, active_idx=None, max_constraints=100):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0, using constraint generation."""
    n = len(a)
    size = 2 * n - 1

    if active_idx is None:
        conv_a = fast_autocorr(a)
        # Select top active constraints
        if len(conv_a) > max_constraints:
            top_idx = np.argsort(conv_a)[::-1][:max_constraints]
            active_idx = top_idx
        else:
            active_idx = np.arange(size)

    T = build_T_rows(a, active_idx)

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, max_active=80, verbose=True):
    """LP best-response with Nesterov acceleration and constraint generation."""
    if solver is None:
        return a0, fast_autocorr(a0).max()

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck = 0
    momentum = 0.0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Select active constraints (top fraction of max)
        conv_y = fast_autocorr(y)
        threshold = 0.8 * M_curr
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 5:
            threshold = 0.5 * M_curr
            active = np.where(conv_y >= threshold)[0]
        if len(active) < 3:
            active = np.where(conv_y >= 0.3 * M_curr)[0]
        if len(active) < 3:
            active = np.argsort(conv_y)[::-1][:max_active]

        b = lp_best_response_fast(y, M_curr, solver, active_idx=active, max_constraints=max_active)
        if b is None:
            stuck += 1
            if stuck > 20:
                break
            continue

        stuck = 0
        b_sum = b.sum()
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
            momentum = momentum * (1.0 - beta)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, window=30, verbose=True):
    """Pair coordinate descent with binary search line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            # Try moving mass from i to nearby positions
            offsets = np.arange(1, window + 1)
            for off in offsets:
                for j in [(i + off) % n, (i - off) % n]:
                    if j == i or a[i] < norm * 1e-9:
                        continue

                    # Binary search for optimal transfer amount
                    lo, hi = 0.0, a[i] * 0.95
                    best_d = 0.0
                    best_val = conv.max()

                    for _ in range(7):
                        mid = (lo + hi) / 2
                        if mid <= 0:
                            lo = mid
                            continue
                        a_t = a.copy()
                        a_t[i] -= mid
                        a_t[j] += mid
                        a_t = a_t * norm / a_t.sum()
                        conv_t = fast_autocorr(a_t)
                        if conv_t.max() < best_val - 1e-13:
                            best_val = conv_t.max()
                            best_d = mid
                            lo = mid
                        else:
                            hi = mid

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                        if conv.max() < best_score - 1e-13:
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_cd_single(a0, time_limit, verbose=True):
    """Single-coordinate descent: try shifting each coordinate's value."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    for _ in range(50):  # max 50 full passes
        if time.monotonic() - t0 > time_limit:
            break
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            # Try scaling a[i] by various factors
            for factor in [0.5, 0.8, 1.2, 1.5, 2.0, 0.3, 0.7]:
                a_t = a.copy()
                delta = a_t[i] * (factor - 1.0)
                a_t[i] *= factor
                # Redistribute the change uniformly
                if abs(delta) > 1e-15:
                    a_t[:] = a_t - delta / (n - 1) if n > 1 else a_t
                    a_t = np.maximum(a_t, 0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        continue
                    a_t = a_t * norm / s_t
                    conv_t = fast_autocorr(a_t)
                    if conv_t.max() < best_score - 1e-13:
                        a = a_t
                        conv = conv_t
                        best_score = conv.max()
                        best_a = a.copy()
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_a, best_score


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    scale = np.sqrt(2.0 * n)

    # Uniform
    a = np.ones(n) * scale / n
    inits.append(('uniform', a))

    # Gaussian window
    x = np.linspace(-1, 1, n)
    a = np.exp(-2.0 * x**2)
    a = a * scale / a.sum()
    inits.append(('gaussian', a))

    # Cosine window
    a = np.sin(np.pi * np.arange(n) / n) ** 2
    a = a * scale / a.sum()
    inits.append(('cosine', a))

    # Ramp
    a = np.linspace(1, 2, n)
    a = a * scale / a.sum()
    inits.append(('ramp', a))

    # Half concentrated
    a = np.zeros(n)
    half = n // 2
    a[:half] = 1.0
    a = a * scale / a.sum()
    inits.append(('half', a))

    # Random
    a = rng.random(n)
    a = a * scale / a.sum()
    inits.append(('random', a))

    # Sparse (few nonzero)
    k = max(4, n // 50)
    d = np.zeros(n)
    indices = rng.choice(n, size=k, replace=False)
    d[indices] = 1.0
    d = d * scale / d.sum()
    inits.append(('sparse', d))

    # Triangular (peaked in middle)
    x = np.linspace(-1, 1, n)
    a = np.maximum(1.0 - np.abs(x), 0)
    a = a * scale / a.sum()
    inits.append(('triangular', a))

    return inits


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    return f_new * np.sqrt(2.0 * new_n) / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-phase optimization to minimize the sequence score."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    parent_n = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            parent_n = len(best_seq)
            print(f"Parent: n={parent_n}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={parent_n}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_current = len(best_seq)

    # === Phase 1: Fast Nesterov smooth optimization on parent (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.30
    print(f"\n=== Phase 1: Nesterov Smooth n={n_current} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_current) / best_seq.sum()
        a_out, _ = nesterov_smooth(a_norm, t1, n_temps=20, verbose=True)
        try_update(a_out, "nesterov_smooth")

    # === Phase 2: LP Nesterov polishing (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: LP Nesterov n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 30 and solver is not None:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t2, solver, max_active=80, verbose=True)
        try_update(a_out, "lp_nesterov")

    # === Phase 3: Coordinate descent polishing (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Pair CD n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t3, rng, window=25, verbose=True)
        try_update(a_out, "pair_cd")

    # === Phase 4: Nesterov smooth on diverse initializations (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Multi-start Nesterov (budget={t4:.0f}s) ===")
    if t4 > 60:
        inits = generate_initializations(n_c, rng)
        per_init = t4 / len(inits)
        for name, a_init in inits:
            sub_time = time_budget - (time.monotonic() - t_global)
            if sub_time < 10:
                break
            sub_budget = min(per_init, sub_time - 5.0)
            if sub_budget < 15:
                break
            a_out, _ = nesterov_smooth(a_init, sub_budget, n_temps=12, verbose=False)
            try_update(a_out, f"multi_{name}")
            # If this init beats the best, also do LP on it
            remaining_after = time_budget - (time.monotonic() - t_global)
            if remaining_after > 60 and solver is not None:
                a_lp, _ = lp_nesterov(a_out, min(remaining_after * 0.3, 60.0), solver, max_active=60, verbose=False)
                try_update(a_lp, f"multi_{name}_lp")

    # === Phase 5: Second LP round with more constraints (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: LP round 2 n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 30 and solver is not None:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t5, solver, max_active=120, verbose=True)
        try_update(a_out, "lp_round2")

    # === Phase 6: Pair CD with larger window (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t6 = remaining * 0.05
    print(f"\n=== Phase 6: Pair CD wide n={n_c} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t6, rng, window=50, verbose=False)
        try_update(a_out, "pair_cd_wide")

    # === Phase 7: Try alternate lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_base = len(best_seq)
        alt_lengths = [n_base + 50, n_base - 50, n_base + 100, n_base - 100]
        for n_alt in alt_lengths:
            if n_alt < 100:
                continue
            sub_time = time_budget - (time.monotonic() - t_global)
            if sub_time < 40:
                break
            sub_budget = min(sub_time * 0.4, 120.0)
            if sub_budget < 20:
                break
            print(f"\n=== Alt length n={n_alt} (budget={sub_budget:.0f}s) ===")
            a_interp = interpolate_to_length(best_seq, n_alt)
            a_out, _ = nesterov_smooth(a_interp, sub_budget * 0.6, n_temps=10, verbose=False)
            try_update(a_out, f"alt_n{n_alt}")
            if time_budget - (time.monotonic() - t_global) > 30 and solver is not None:
                a_lp, _ = lp_nesterov(a_out, min(time_budget - (time.monotonic() - t_global) * 0.3, 60.0), solver, max_active=60, verbose=False)
                try_update(a_lp, f"alt_n{n_alt}_lp")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
