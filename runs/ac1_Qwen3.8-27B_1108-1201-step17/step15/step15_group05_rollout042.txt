```python
    A_rev = rfft(a_rev, N)
    W = rfft(w, N)
    conv_result = irfft(A_rev * W, N)
    grad = 2.0 * conv_result[n-1:n-1+n]
    
    return grad, c
```

Wait, I realize there's an issue with the gradient computation. Let me double-check.

We want to minimize max_k c_k where c = autocorr(a). The log-sum-exp approximation is:
L(a) = (1/τ) * log(sum_k exp(τ * c_k))

The gradient is:
∂L/∂a[j] = (1/τ) * sum_k τ * (∂c_k/∂a[j]) * exp(τ*c_k) / sum_k exp(τ*c_k)
          = sum_k (∂c_k/∂a[j]) * w_k

where w_k = exp(τ*c_k) / sum_k exp(τ*c_k)

And ∂c_k/∂a[j] = 2*a[k-j] if 0 <= k-j < n, else 0.

So: ∂L/∂a[j] = sum_k 2*a[k-j] * w_k (where k-j is in [0, n-1])
             = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]  (substituting m = k-j)

Yes, that's what I had. Good.

Now, the constraint is that sum(a) = 1 (we normalize). So we need to project onto the simplex {a >= 0, sum a = 1}. The projection is: a_proj = max(a - λ, 0) where λ is chosen so that sum(a_proj) = 1.

Actually, since we're just rescaling after the step (a / sum(a)), we're effectively doing a "radial" projection, not the proper simplex projection. For better behavior, we should use the proper simplex projection.

Let me implement the simplex projection:


Actually, for our purposes, the simple rescaling a / sum(a) should work fine since we're always keeping a >= 0 and just adjusting the scale. The proper simplex projection would be better but the rescaling is a good approximation.

Let me also think about a different approach: **direct optimization using the KKT conditions**.

At the optimum of minimize max_k c_k s.t. sum(a) = 1, a >= 0:
- There exists λ > 0 (for the sum constraint) and μ_j >= 0 (for a_j >= 0) such that:
  - 2 * sum_k w_k * a[k-j] = λ + μ_j for all j
  - w_k >= 0, sum_k w_k = 1, w_k > 0 only if c_k = max(c)
  - μ_j * a_j = 0 (complementary slackness)
  - sum(a) = 1

This is a system of equations that could potentially be solved. But it's complicated.

Let me focus on the practical approach: a fast gradient method + LP polish.

Here's my refined plan:

1. **Phase 1 (fast gradient descent, ~60% of time)**:
   - Log-sum-exp gradient descent with Nesterov acceleration
   - Multiple restarts with different initializations
   - Adaptive temperature and learning rate
   - Try multiple lengths (1600, 2000, 2400)

2. **Phase 2 (LP polish, ~30% of time)**:
   - Take the best candidate from Phase 1
   - Run the LP best-response method with all constraints (or a large active set)
   - This should give a small improvement

3. **Phase 3 (coordinate descent, ~10% of time)**:
   - Fine-tune individual coordinates

Let me also think about the initialization. Good initializations:
- Uniform: a_j = 1/n (score = 2)
- Cosine: a_j = (1 + cos(πj/(n-1))) / 2 (triangular shape)
- Gaussian: a_j = exp(-j^2 / (2σ^2))
- The parent sequence (perturbed)
- Random with Dirichlet distribution

For the gradient method, I need to be careful about the learning rate. The problem is scale-invariant (we normalize after each step), so the effective "size" of the problem is the shape of a, not its magnitude.

Let me also consider: what if I use a **different smoothing** than log-sum-exp? For example:
- Power approximation: (sum_k c_k^p)^(1/p) for large p
- This is also smooth and convex in c (and the composition with the quadratic is... not necessarily convex)

Actually, the power approximation might be better because:
- (sum c_k^p)^(1/p) → max(c_k) as p → ∞
- The gradient is: p * c_k^(p-1) / (sum c_k^p) * ∂c_k/∂a[j]
- This gives more weight to the peak than log-sum-exp does for moderate τ

But log-sum-exp is theoretically cleaner and I'll stick with it.

One more idea: **entropic mirror descent**. Instead of Euclidean projection, use the KL divergence. The update would be:
a_new[j] ∝ a[j] * exp(-η * grad[j])

This is more natural for the simplex constraint and might give better behavior. Let me include this as an option.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated gradient descent on log-sum-exp approximation
2. LP best-response polishing (Frank-Wolfe for non-convex problem)
3. Coordinate descent for final refinement
4. Multi-length exploration and multi-start strategy
Key improvements: FFT-based fast gradients (no LP in main loop),
adaptive temperature, entropic mirror descent, and better time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_grad_lse(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*c_k)) with respect to a."""
    n = len(a)
    c = fast_autocorr(a)
    # Softmax weights
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    w = exp_c / exp_c.sum()
    # grad_j = 2 * sum_m a[m] * w[m+j] for m=0..n-1
    # = 2 * conv(a[::-1], w)[n-1 + j] for j=0..n-1
    a_rev = a[::-1]
    conv_len = n + len(w) - 1
    N = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a_rev, N)
    W = rfft(w, N)
    conv = irfft(A_rev * W, N)
    grad = 2.0 * conv[n-1:n-1+n]
    return grad, c

def grad_descent_lse(a0, time_limit, seed=0, verbose=False, tau_init=100.0, tau_min=20.0):
    """
    Nesterov-accelerated gradient descent on log-sum-exp of autocorrelation.
    Minimizes max_k c_k subject to sum(a)=1, a>=0.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_c = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    
    tau = tau_init
    tau_decay = 0.9995
    lr = 0.005
    lr_decay = 0.999
    
    t0 = time.monotonic()
    stall = 0
    stall_threshold = 300
    n_restarts = 0
    max_restarts = 20
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad, c = compute_grad_lse(y, tau)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y - alpha * lr * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = y_new
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        
        # Track best
        c_max_a = fast_autocorr(a).max()
        if c_max_a < best_c - 1e-14:
            best_c = c_max_a
            best_a = a.copy()
            stall = 0
        
        c_max_y = fast_autocorr(y).max()
        if c_max_y < best_c - 1e-14:
            best_c = c_max_y
            best_a = y.copy()
            stall = 0
        
        if stall > 0:
            stall += 1
        
        # Decay parameters
        tau = max(tau * tau_decay, tau_min)
        lr = max(lr * lr_decay, 1e-5)
        
        # Restart if stalled
        if stall > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall = 0
            n_restarts += 1
            lr = 0.005 / (1.0 + 0.5 * n_restarts)
            tau = min(tau * 1.5, 500.0)
            if n_restarts > max_restarts:
                break
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      GD restart {n_restarts} at t={t} ({elapsed:.0f}s), best_c={best_c:.10f}")
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      GD t={t} ({elapsed:.0f}s): best_c={best_c:.10f}, tau={tau:.1f}, lr={lr:.6f}, stall={stall}")
    
    return best_a, best_c

def mirror_descent(a0, time_limit, seed=0, verbose=False):
    """
    Entropic mirror descent for the smoothed max problem.
    Update: a_new[j] ∝ a[j] * exp(-eta * grad[j])
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_c = fast_autocorr(a).max()
    
    tau = 50.0
    eta = 0.01
    t0 = time.monotonic()
    stall = 0
    stall_threshold = 200
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad, c = compute_grad_lse(a, tau)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Mirror descent step
        a_new = a * np.exp(-eta * grad)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_max = fast_autocorr(a_new).max()
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 0.1)
        else:
            stall += 1
            eta *= 0.95
            if eta < 1e-6:
                break
        
        tau = max(tau * 0.999, 20.0)
        
        if stall > stall_threshold:
            a = best_a.copy()
            stall = 0
            eta = 0.01
            if verbose:
                print(f"      MD restart at t={t}, best_c={best_c:.10f}")
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD t={t} ({elapsed:.0f}s): best_c={best_c:.10f}, eta={eta:.6f}")
    
    return best_a, best_c

def lp_best_response_full(a_fixed, M, solver):
    """
    Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for ALL k, b >= 0.
    Uses all 2n-1 constraints.
    """
    import cvxpy as cp
    n = len(a_fixed)
    size = 2 * n - 1
    # Build constraint matrix
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n
    T = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[k, j] = a_fixed[k - j]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_polish(a0, time_limit, solver, verbose=False):
    """LP best-response polishing (Frank-Wolfe)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_c = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    t = 0
    stall = 0
    
    y = a.copy()
    y_prev = a.copy()
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(y)
        M = c.max()
        
        b = lp_best_response_full(y, M, solver)
        if b is None:
            break
        
        b_sum = b.sum()
        g = b / b_sum  # normalize to sum = 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = y_new
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        
        c_max = fast_autocorr(a).max()
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 100:
            break
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best_c={best_c:.10f}")
    
    return best_a, best_c

def coordinate_descent_fast(a0, time_limit, verbose=False):
    """Fast coordinate descent with adaptive step sizes."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_c = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    step_sizes = np.ones(n) * 0.1
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            break
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            c = fast_autocorr(a)
            
            # Try modifying a[j]
            best_x = aj
            best_val = c.max()
            
            # Try a few perturbation sizes
            for frac in [-0.5, -0.3, -0.1, 0.1, 0.3, 0.5]:
                x = max(0.0, aj * (1 + frac))
                if x == aj:
                    continue
                a[j] = x
                a = a / a.sum()
                c_new = fast_autocorr(a)
                if c_new.max() < best_val:
                    best_val = c_new.max()
                    best_x = x
            a[j] = aj
            a = a / a.sum()
            
            if best_x != aj:
                a[j] = best_x
                a = a / a.sum()
                c = fast_autocorr(a)
                if c.max() < best_c - 1e-14:
                    best_c = c.max()
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = aj
                    a = a / a.sum()
                step_sizes[j] *= 0.5
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if verbose and (improved or no_improve % 10 == 0):
            elapsed = time.monotonic() - t0
            print(f"      CD ({elapsed:.0f}s): best_c={best_c:.10f}, no_improve={no_improve}")
    
    return best_a, best_c

def make_initialization(n, kind, rng):
    """Create various initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
        return a / a.sum()
    elif kind == 'gaussian':
        x = np.linspace(-1, 1, n)
        sigma = 0.3
        a = np.exp(-x**2 / (2 * sigma**2))
        return a / a.sum()
    elif kind == 'random':
        a = rng.dirichlet(np.ones(n))
        return a
    elif kind == 'random_sparse':
        a = rng.exponential(1.0, size=n)
        a[a < np.median(a)] = 0
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        return a
    elif kind == 'bump':
        x = np.linspace(0, 1, n)
        center = 0.5
        width = 0.2
        a = np.exp(-((x - center) / width)**2)
        return a / a.sum()
    elif kind == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) / 0.15)**2) + np.exp(-((x - 0.7) / 0.15)**2)
        return a / a.sum()
    else:
        return np.ones(n) / n
```

```python
def lp_best_response_active(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_set, b >= 0."""
    import cvxpy as cp
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None
    # Build constraint matrix
    T = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[idx, j] = a_fixed[k - j]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def make_init(n, kind, rng):
    """Create various initializations of length n, sum=1, non-negative."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(np.pi * x))
        return a / a.sum()
    elif kind == 'sinusoidal':
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == 'gaussian':
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 / (2 * 0.3 ** 2))
        return a / a.sum()
    elif kind == 'random':
        return rng.dirichlet(np.ones(n))
    elif kind == 'bimodal':
        x = np.linspace(0, 1, n)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated gradient descent on log-sum-exp approximation of max autocorr
2. Fast FFT-based gradient computation (no LP in main loop)
3. LP best-response polishing for final refinement
4. Multi-length exploration (1600, 2000, 2400) with interpolation
5. Adaptive temperature, multiple restarts, and perturbation-based diversity
Key improvements: 10-100x faster iterations via pure FFT gradients, better
convergence through adaptive smoothing, and smarter time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_lse(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*c_k)) w.r.t. a via FFT.
    Returns (grad, c) where grad_j = sum_k w_k * dc_k/da_j.
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    w = exp_c / exp_c.sum()
    # grad_j = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    a_rev = a[::-1]
    conv_len = n + len(w) - 1
    N = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a_rev, N)
    W = rfft(w, N)
    conv = irfft(A_rev * W, N)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, c


def grad_descent_smooth(a0, time_limit, seed=0, verbose=False,

    """Nesterov-accelerated gradient descent on smoothed max autocorrelation."""

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_c = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()

    tau = tau_init
    tau_decay = 0.9997

    t0 = time.monotonic()
    stall = 0
    stall_threshold = 250
    n_restarts = 0
    max_restarts = 80

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        grad, c = compute_grad_lse(y, tau)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Adaptive learning rate
        lr = 0.008 / (1.0 + 0.03 * np.sqrt(t))

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y - alpha * lr * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.92)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        c_max_a = fast_autocorr(a).max()
        if c_max_a < best_c - 1e-14:

            best_a = a.copy()

        if c_max_y < best_c - 1e-14:
            best_c = c_max_y
            best_a = y.copy()
            stall = 0

        if stall > 0:
            stall += 1

        tau = max(tau * tau_decay, tau_min)

        # Restart if stalled
        if stall > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall = 0
            n_restarts += 1
            if n_restarts % 4 == 1:

                a = a * (1.0 + scale * rng.standard_normal(n))
                a = np.maximum(a, 0.0)
                s2 = a.sum()
                if s2 > 1e-15:
                    a = a / s2
                y = a.copy()
                y_prev = a.copy()
            tau = min(tau * 1.5, 800.0)
            if n_restarts > max_restarts:
                break
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      GD restart {n_restarts} t={t} ({elapsed:.0f}s) c={best_c:.10f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      GD t={t} ({elapsed:.0f}s): c={best_c:.10f}, tau={tau:.0f}")

    return best_a, best_c


def lp_best_response_active(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_set, b >= 0."""
    import cvxpy as cp
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None
    T = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[idx, j] = a_fixed[k - j]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def make_init(n, kind, rng):
    """Create various initializations of length n, sum=1, non-negative."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(np.pi * x))
        return a / a.sum()
    elif kind == 'sinusoidal':
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == 'gaussian':
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 / (2 * 0.35 ** 2))
        return a / a.sum()
    elif kind == 'random':
        return rng.dirichlet(np.ones(n))
    elif kind == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) / 0.15) ** 2) + np.exp(-((x - 0.7) / 0.15) ** 2)
        return a / a.sum()
    elif kind == 'flat_cos':
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    else:
        return np.ones(n) / n


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(x_new, x_old, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s
```
