import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(40, min(400, int(remaining * 8)))
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        # Adaptive constraint screening with tighter threshold
        threshold_1 = M * 0.92
        active = np.where(conv_y >= threshold_1)[0]
        if len(active) < 20:
            threshold_2 = M * 0.65
            active = np.where(conv_y >= threshold_2)[0]
        b = lp_solve_fast(y, M, solver, active)
        if b is None:
            b = lp_solve_fast(y, M, solver, None)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        # Verify dropped constraints
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * 1.002)[0]
            if len(violated) > 5:
                b_full = lp_solve_fast(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a
        y = y_new
        a = a_new
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def exact_pair_cd(a0, time_limit, rng, verbose=True):
    """Exact pair coordinate descent using 1D min-max bisection."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        # Compute pressure: how much each coordinate contributes to top peaks
        peak_threshold = cur_max * 0.90
        peak_lags = np.where(conv >= peak_threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(conv)])
        pressure = np.zeros(n)
        for kl in peak_lags[:min(60, len(peak_lags))]:
            # Coordinate i contributes to lag kl if 0 <= kl-i < n
            lo_i = max(0, kl - n + 1)
            hi_i = min(n, kl + 1)
            for i in range(lo_i, hi_i):
                j = kl - i
                if 0 <= j < n:
                    if kl == 2 * i:
                        pressure[i] += a[i] * a[i]
                    else:
                        pressure[i] += 2.0 * a[i] * a[j]

        # Select pairs: high pressure source, low pressure target
        n_pairs = min(30, n)
        pass_improved = False

        # Weight for sampling sources (high pressure)
        p_source = pressure + 1e-10
        p_source = p_source / p_source.sum()
        # Weight for targets (inversely proportional to pressure)
        p_target = 1.0 / (pressure + cur_max * 0.1)
        p_target = p_target / p_target.sum()

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.choice(n, p=p_source))
            j = int(rng.choice(n, p=p_target))
            if i == j:
                continue
            ai = a[i]
            aj = a[j]
            if ai < 1e-15 and aj < 1e-15:
                continue
            # Compute W(m) = G_i(m) - G_j(m)
            # G_p(m) = 2*a[m-p] if 0 <= m-p < n, else 0
            W = np.zeros(size)
            # G_i contribution: 2*a[m-i] for m in [i, i+n-1]
            lo_i = i
            hi_i = min(size, i + n)
            W[lo_i:hi_i] += 2.0 * a[0:hi_i - lo_i]
            # G_j contribution: 2*a[m-j] for m in [j, j+n-1]
            lo_j = j
            hi_j = min(size, j + n)
            W[lo_j:hi_j] -= 2.0 * a[0:hi_j - lo_j]
            # Q(m): +1 at m=2i, +1 at m=2j, -2 at m=i+j
            mi1 = 2 * i
            mj1 = 2 * j
            mij = i + j
            Q_nonzero = {}
            if mi1 < size:
                Q_nonzero[mi1] = Q_nonzero.get(mi1, 0.0) + 1.0
            if mj1 < size:
                Q_nonzero[mj1] = Q_nonzero.get(mj1, 0.0) + 1.0
            if mij < size:
                Q_nonzero[mij] = Q_nonzero.get(mij, 0.0) - 2.0
            # Bisection on t to find optimal d
            # f(d) = max_m (conv[m] + d*W[m] + d^2*Q[m])
            # We want min_d f(d)
            # Bisection: find smallest t such that exists d with f(d) <= t
            t_lo = 0.0
            t_hi = cur_max
            # Ensure t_hi is feasible (d=0 gives f(0)=cur_max)
            for _bisect in range(50):
                t_mid = (t_lo + t_hi) / 2.0
                # Check feasibility: exists d such that all constraints satisfied
                d_lo = -np.inf
                d_hi = np.inf
                feasible = True
                # Linear constraints: conv[m] + d*W[m] <= t_mid
                pos_mask = W > 1e-15
                neg_mask = W < -1e-15
                if np.any(pos_mask):
                    ub = (t_mid - conv) / W
                    ub_valid = ub[pos_mask]
                    d_hi = min(d_hi, np.min(ub_valid))
                if np.any(neg_mask):
                    lb = (t_mid - conv) / W
                    lb_valid = lb[neg_mask]
                    d_lo = max(d_lo, np.max(lb_valid))
                # Quadratic constraints at nonzero Q positions
                for m_q, q_val in Q_nonzero.items():
                    c_m = conv[m_q]
                    w_m = W[m_q]
                    if q_val > 0:
                        # d^2 * q_val + d * w_m + c_m - t_mid <= 0
                        # q_val*d^2 + w_m*d + (c_m - t_mid) <= 0
                        disc = w_m * w_m - 4.0 * q_val * (c_m - t_mid)
                        if disc < 0:
                            feasible = False
                            break
                        sqrt_disc = np.sqrt(disc)
                        root1 = (-w_m - sqrt_disc) / (2.0 * q_val)
                        root2 = (-w_m + sqrt_disc) / (2.0 * q_val)
                        d_lo = max(d_lo, root1)
                        d_hi = min(d_hi, root2)
                    else:
                        # q_val = -2: -2*d^2 + w_m*d + c_m - t_mid <= 0
                        # 2*d^2 - w_m*d + (t_mid - c_m) >= 0
                        disc = w_m * w_m - 8.0 * (t_mid - c_m)
                        if disc >= 0:
                            sqrt_disc = np.sqrt(disc)
                            root1 = (w_m - sqrt_disc) / 4.0
                            root2 = (w_m + sqrt_disc) / 4.0
                            # d must be outside [root1, root2]
                            # This means either d <= root1 or d >= root2
                            # This makes the feasible set potentially two intervals
                            # For simplicity, we handle this by checking both sides
                            # Actually, let's just note that for the downward parabola,
                            # the constraint is satisfied for large |d|, so it usually
                            # doesn't bind. We'll be conservative and skip it.
                            pass
                if not feasible or d_lo > d_hi + 1e-12:
                    t_lo = t_mid
                else:
                    t_hi = t_mid
            # Optimal t is t_hi, find d in [d_lo, d_hi]
            # Recompute d bounds at t_hi
            d_lo = -np.inf
            d_hi = np.inf
            pos_mask = W > 1e-15
            neg_mask = W < -1e-15
            if np.any(pos_mask):
                ub = (t_hi - conv) / W
                d_hi = min(d_hi, np.min(ub[pos_mask]))
            if np.any(neg_mask):
                lb = (t_hi - conv) / W
                d_lo = max(d_lo, np.max(lb[neg_mask]))
            for m_q, q_val in Q_nonzero.items():
                c_m = conv[m_q]
                w_m = W[m_q]
                if q_val > 0:
                    disc = w_m * w_m - 4.0 * q_val * (c_m - t_hi)
                    if disc >= 0:
                        sqrt_disc = np.sqrt(disc)
                        root1 = (-w_m - sqrt_disc) / (2.0 * q_val)
                        root2 = (-w_m + sqrt_disc) / (2.0 * q_val)
                        d_lo = max(d_lo, root1)
                        d_hi = min(d_hi, root2)
            if d_lo > d_hi:
                d_opt = 0.0
            else:
                d_opt = (d_lo + d_hi) / 2.0
            # Apply update if it improves
            if abs(d_opt) > 1e-15:
                new_conv = conv + d_opt * W
                for m_q, q_val in Q_nonzero.items():
                    new_conv[m_q] += d_opt * d_opt * q_val
                new_max = new_conv.max()
                if new_max < cur_max - 1e-10:
                    a[i] += d_opt
                    a[j] -= d_opt
                    a = np.maximum(a, 0.0)
                    s_new = a.sum()
                    if s_new > 1e-15:
                        a = a * norm_val / s_new
                    new_score = fast_conv_self(a).max()
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve_count += 1
            if no_improve_count >= 5:
                break
            if no_improve_count >= 2:
                n_perturb = max(5, n // 50)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_val / s_new
        else:
            no_improve_count = 0
        if verbose and (no_improve_count == 0):
            elapsed = time.monotonic() - t0
            if elapsed % 10 < 1:
                print(f"    PairCD ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def mirror_descent(a0, time_limit, rng, verbose=True):
    """Multiplicative weight / mirror descent with KL-like updates."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    size = 2 * n - 1

    eta = 1.0 / np.sqrt(n)
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_conv_self(a)
        max_c = conv.max()
        # Soft assignment to peaks
        temp = max(0.05, max_c * 0.03 / (1.0 + 0.005 * it))
        shifted = (conv - max_c) / temp
        shifted = np.maximum(shifted, -80.0)
        w = np.exp(shifted)
        w = w / w.sum()
        # Gradient via FFT
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        # Mirror descent update: a_new = a * exp(-eta * grad)
        eta_adapt = eta / (1.0 + 0.001 * it)
        log_update = -eta_adapt * grad / max_c
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm_val / s_new
        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDec it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.0
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(3.0 * (x - 0.5)) ** 2 + 0.1
    elif choice == 9:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.1)
    elif choice == 10:
        # Welch-like: sum of shifted sincs
        a = np.zeros(n)
        for k in range(5):
            center = (k + 0.5) / 5.0
            a += np.exp(-8.0 * (x - center) ** 2)
    else:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x + rng.random() * np.pi)
    a = np.maximum(a, 0.0)
    a = a + 0.01 * rng.random(n)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 40:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Exact pair CD (NEW)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 3: Exact Pair CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = exact_pair_cd(best_seq, t3, rng, verbose=True)
        try_update(a_out, "paircd1")

    # Phase 4: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Mirror Descent n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = mirror_descent(best_seq, t4, rng, verbose=True)
        try_update(a_out, "mirror1")

    # Phase 5: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 5: LP Nesterov r2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 6: Exact pair CD round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: Exact Pair CD r2 n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = exact_pair_cd(best_seq, t6, rng, verbose=True)
        try_update(a_out, "paircd2")

    # Phase 7: Try slightly different length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_alt = len(best_seq) + 200
        t7 = min(remaining * 0.3, 180.0)
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={t7:.0f}s) ===")
        if t7 > 30:
            init_alt = interpolate_seq(best_seq, n_alt)
            t7a = t7 * 0.6
            a_out, _ = nesterov_lp_optimize(init_alt, t7a, solver, verbose=False)
            try_update(a_out, f"lp_alt{n_alt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t7b = min(remaining * 0.3, 60.0)
                if t7b > 15:
                    a_out2, _ = exact_pair_cd(init_alt if a_out is not None else best_seq, t7b, rng, verbose=False)
                    try_update(a_out2, f"paircd_alt")

    # Phase 8: Final LSE + pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        t_half = remaining * 0.4
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 60.0), verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = exact_pair_cd(best_seq, min(remaining * 0.6, 50.0), rng, verbose=False)
            try_update(a_out, "paircd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
