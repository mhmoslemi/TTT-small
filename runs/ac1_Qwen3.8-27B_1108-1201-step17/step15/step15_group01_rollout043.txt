def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def hankel_matvec(p, x):
    """Compute Q @ x where Q[i,j] = p[i+j] (Hankel matrix). O(n log n) via FFT."""
    n = len(x)
    N = 1 << (n + len(p) - 1).bit_length()
    X_fft = rfft(x, N)
    P_fft = rfft(p, N)
    return irfft(np.conj(X_fft) * P_fft, N)[:n]


def cross_corr_trunc(a, p, n_out):
    """Compute v[i] = sum_k p[k] * a[k-i] for i=0..n_out-1 via FFT."""
    n = len(a)
    N = 1 << (n + len(p) - 1).bit_length()
    A_fft = rfft(a, N)
    P_fft = rfft(p, N)
    v = irfft(np.conj(A_fft) * P_fft, N)
    return v[:n_out]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_qp_inexact(p, a_init, n_inner_steps=200):
    """Approximately solve min a^T Q a s.t. a >= 0, sum(a)=1 using projected gradient with restarts.
    Q[i,j] = p[i+j] is Hankel, so Q@x is O(n log n)."""
    n = len(p) // 2 + len(p) % 2  # n = (len(p)+1)//2
    n = (len(p) + 1) // 2
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = a @ hankel_matvec(p, a)
    
    # Estimate step size from first gradient
    grad = 2.0 * hankel_matvec(p, a)
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a
    
    # Use Lipschitz estimate: L ~ max eigenvalue of Q
    # Bounded by the L1 norm of a row of Q
    row_sum = np.array([p[i:i+n].sum() for i in range(n)])
    L_est = 2.0 * row_sum.max()
    if L_est < 1e-10:
        L_est = 2.0
    step = 1.0 / L_est
    
    for it in range(n_inner_steps):
        grad = 2.0 * hankel_matvec(p, a)
        a_new = a - step * grad
        a_new = project_simplex(a_new, 1.0)
        
        val = a_new @ hankel_matvec(p, a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        a = a_new
        # Adaptive step: if oscillating, reduce step
        if it % 50 == 49 and it > 0:
            step *= 0.95
    
    return best_a


def dual_mirror_descent(n, time_limit, p_init=None, verbose=True):
    """Dual mirror descent: max_p min_a a^T Q(p) a s.t. a>=0, sum(a)=1."""
    t0 = time.monotonic()
    n_dual = 2 * n - 1  # number of dual variables (lags)
    
    if p_init is None:
        p = np.ones(n_dual) / n_dual
    else:
        p = p_init.copy()
        p = np.maximum(p, 1e-30)
        p = p / p.sum()
    
    a = np.ones(n) / n
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    # Total iterations estimate
    T_total = 2000
    it = 0
    
    for it in range(1, T_total + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve QP approximately
        K_inner = max(50, min(200, n // 8))
        a = solve_qp_inexact(p, a, n_inner_steps=K_inner)
        
        # Compute dual gradient
        d = fast_autocorr(a)  # d[k] = (a*a)[k]
        d_max = d.max()
        
        # Track best primal
        if d_max < best_score:
            best_score = d_max
            best_a = a.copy()
        
        # Mirror descent update on p (multiplicative weights)
        eta = np.sqrt(2.0 * np.log(n_dual) / it)
        eta = min(eta, 2.0)  # cap
        
        # Numerically stable update
        d_shifted = d - d.max()
        log_weights = np.log(p) + eta * d_shifted
        log_weights -= log_weights.max()
        p = np.exp(log_weights)
        p = p / p.sum()
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            dual_val = a @ hankel_matvec(p, a)
            print(f"    Dual MD it={it} ({elapsed:.0f}s): primal={d_max:.6f}, dual_bound={dual_val:.6f}, gap={d_max-dual_val:.6f}")
    
    return best_a, best_score


def smooth_gradient_optimize(a0, time_limit, verbose=True):
    """Minimize max(conv(a,a)) using log-sum-exp smooth gradient with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(30, min(300, int(remaining * 2)))
        
        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(y)
            c_max = conv.max()
            exp_c = np.exp((conv - c_max) / T)
            p = exp_c / exp_c.sum()
            
            # Gradient = 2 * (a ⋆ p) truncated
            grad = 2.0 * cross_corr_trunc(y, p, n)
            
            # Adaptive step size
            step = 1.0 / (n * T * (1.0 + 0.02 * it))
            step = min(step, 4.0 / n)
            
            a_new = y - step * grad
            a_new = project_simplex(a_new, 1.0)
            
            # Nesterov
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, 1.0)
            
            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, 1.0)
            
            y = y_new
            a = a_curr
            
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Smooth T={T:.3f} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def targeted_cd(a0, time_limit, rng, verbose=True):
    """Targeted coordinate descent: move mass from peak-contributing to non-contributing positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 5:
            break
        n_passes += 1
        pass_improved = False
        
        # Identify peak lag
        k_star = np.argmax(conv)
        
        # Gradient at peak: grad[i] = 2*a[k_star - i] (valid indices)
        # Positions contributing most to peak: those where a[k_star - i] is large AND a[i] is large
        # Contribution of a[i] to conv[k_star] is: a[i] * a[k_star - i] (if valid)
        
        # Compute contribution vector
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Hot positions: high contribution to peak
        # Cold positions: low a value (good targets)
        
        # Sort by contribution (hot to cold)
        hot_order = np.argsort(contrib)[::-1]
        
        for idx in range(min(200, n)):
            if time.monotonic() - t0 > time_limit:
                break
            i = hot_order[idx]
            if a[i] < 1e-10:
                continue
            
            # Try moving to several candidate positions
            # Candidates: positions with low a value, spread across the sequence
            n_targets = min(10, n // 2)
            # Pick targets with lowest a values (excluding i)
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            target_candidates = np.argsort(a)[mask]
            targets = target_candidates[:n_targets]
            
            # Also try some random targets
            random_targets = rng.choice(n, size=min(5, n//2), replace=False)
            all_targets = np.unique(np.concatenate([targets, random_targets]))
            
            for j in all_targets:
                if j == i or a[i] < 1e-10:
                    continue
                
                # Try moving 20% of a[i] to a[j]
                for frac in [0.3, 0.15, 0.05]:
                    d = a[i] * frac
                    a_test = a.copy()
                    a_test[i] -= d
                    a_test[j] += d
                    s_test = a_test.sum()
                    if s_test < 1e-15:
                        continue
                    a_test = a_test / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < conv.max() - 1e-12:
                        a = a_test
                        conv = conv_test
                        if conv.max() < best_score:
                            best_score = conv.max()
                            best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                        break  # Try next i
        
        if not pass_improved:
            no_improve += 1
        
        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.6f}, no_improve={no_improve}")
    
    return best_a, best_score


def pair_cd_standard(a0, time_limit, rng, window=50, verbose=True):
    """Standard pair coordinate descent."""
    n = len(a0)

            break

        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-12:
                continue
            
            for offset in range(1, min(window, n) + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    if a[i] < 1e-12:
                        break
                    
                    # Binary search for best transfer
                    lo, hi = 0.0, a[i] * 0.99
                    best_d = 0.0
                    best_val = conv.max()
                    for _ in range(6):
                        mid = (lo + hi) / 2
                        if mid <= 1e-15:
                            lo = mid
                            continue
                        a_t = a.copy()
                        a_t[i] -= mid
                        a_t[j] += mid
                        a_t = a_t / a_t.sum()
                        conv_t = fast_autocorr(a_t)
                        if conv_t.max() < best_val - 1e-13:
                            best_val = conv_t.max()
                            best_d = mid
                            lo = mid
                        else:
                            hi = mid
                    
                    if best_d > 1e-15:
                        a[i] -= best_d
                        a[j] += best_d
                        a = a / a.sum()
                        conv = fast_autocorr(a)
                        if conv.max() < best_score - 1e-13:
                            best_score = conv.max()
                            best_a = a.copy()
                        pass_improved = True

        if verbose and n_passes % 5 == 0:

def lp_best_response(a, M, solver, active_only=None):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    
    if active_only is not None:
        T_use = T[active_only, :]
    else:
        T_use = T
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def structured_init(n, kind='linear', rng=None):
    """Generate structured initializations."""
    if kind == 'linear':
        a = np.arange(1, n + 1, dtype=np.float64)
    elif kind == 'quadratic':
        a = np.arange(1, n + 1, dtype=np.float64) ** 2
    elif kind == 'exp_decay':
        a = np.exp(-0.5 * np.arange(n) / (n * 0.3))
    elif kind == 'exp_rise':
        a = np.exp(0.5 * np.arange(n) / (n * 0.3))
    elif kind == 'triangular':
        mid = n // 2
        a = 1.0 - np.abs(np.arange(n) - mid) / (mid + 1)
        a = np.maximum(a, 0.01)
    elif kind == 'chirp_abs':
        freq = np.linspace(0, 1, n)
        phase = 2 * np.pi * np.cumsum(freq) * 0.1
        a = np.abs(np.cos(phase)) + 0.1
    elif kind == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    except Exception:
        return float('inf')

        n_base = len(best_seq)
    
    # === Phase 1: Dual Mirror Descent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    n1 = n_base
    print(f"\n=== Phase 1: Dual Mirror Descent n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_init = best_seq / best_seq.sum()
        a_out, _ = dual_mirror_descent(n1, t1, p_init=None, verbose=True)
        do_try_update(a_out, f"dual_n{n1}")
        
        # Also try with non-uniform p init (concentrated on off-center lags)
        remaining_sub = time_budget - (time.monotonic() - t_global)
        if remaining_sub > 30 and best_seq is not None:
            conv_best = fast_autocorr(best_seq / best_seq.sum())
            p_init = conv_best.copy()
            p_init = np.maximum(p_init, 1e-10)
            p_init = p_init / p_init.sum()
            a_out2, _ = dual_mirror_descent(n1, min(remaining_sub * 0.3, 120), p_init=p_init, verbose=False)
            do_try_update(a_out2, f"dual2_n{n1}")
    
    # === Phase 2: Smooth Gradient (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Smooth Gradient n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq / best_seq.sum()
        a_out, _ = smooth_gradient_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"smooth_n{n_c}")
    
    # === Phase 3: Targeted CD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Targeted CD n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = best_seq / best_seq.sum()
        a_out, _ = targeted_cd(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"tcd_n{n_c}")
    
    # === Phase 4: LP Nesterov (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: LP Nesterov n={n_c} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_c}")
    
    # === Phase 5: Pair CD (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Pair CD n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq / best_seq.sum()
        a_out, _ = pair_cd_standard(a_norm, t5, rng, window=60, verbose=True)
        do_try_update(a_out, f"pcd_n{n_c}")
    
    # === Phase 6: Diversified restarts (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t6 = remaining * 0.05
    print(f"\n=== Phase 6: Diversified restarts n={n_c} (budget={t6:.0f}s) ===")
    if t6 > 20:
        # Perturb and smooth-optimize
        a_pert = best_seq / best_seq.sum()
        noise = rng.normal(0, 0.05, n_c)
        a_pert = a_pert + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        t6a = min(t6 * 0.5, 60)
        a_out, _ = smooth_gradient_optimize(a_pert, t6a, verbose=False)
        do_try_update(a_out, "restart_smooth")
        
        remaining6 = time_budget - (time.monotonic() - t_global)
        if remaining6 > 10:
            a_norm_r = best_seq / best_seq.sum()
            a_out_r, _ = targeted_cd(a_norm_r, min(remaining6 * 0.5, 30), rng, verbose=False)
            do_try_update(a_out_r, "restart_tcd")
    
    # === Phase 7: Try alternate lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        n_base_final = len(best_seq)
        for n_alt in [n_base_final + 200, n_base_final - 200]:
            if n_alt < 200:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 30:
                break
            print(f"\n=== Alternate length n={n_alt} (budget={min(remaining_sub * 0.3, 80):.0f}s) ===")
            t_alt = min(remaining_sub * 0.3, 80.0)
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_out_alt, _ = smooth_gradient_optimize(init_alt, t_alt, verbose=False)
            do_try_update(a_out_alt, f"alt_smooth_n{n_alt}")
            
            remaining_sub2 = time_budget - (time.monotonic() - t_global)
            if remaining_sub2 > 20 and best_seq is not None:
                init_alt2 = interpolate_sequence(best_seq, n_alt)
                a_out_alt2, _ = targeted_cd(init_alt2, min(remaining_sub2 * 0.3, 40), rng, verbose=False)
                do_try_update(a_out_alt2, f"alt_tcd_n{n_alt}")
    
    # === Final polish: one more round of targeted CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        n_c = len(best_seq)
        print(f"\n=== Final Polish n={n_c} (budget={min(remaining * 0.5, 60):.0f}s) ===")
        t_final = min(remaining * 0.5, 60.0)
        a_norm = best_seq / best_seq.sum()
        a_out, _ = targeted_cd(a_norm, t_final, rng, verbose=True)
        do_try_update(a_out, "final_polish")
        
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 15 and best_seq is not None:
            a_norm2 = best_seq / best_seq.sum()
            a_out2, _ = pair_cd_standard(a_norm2, min(remaining2 * 0.5, 30), rng, window=80, verbose=False)
            do_try_update(a_out2, "final_pcd")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
