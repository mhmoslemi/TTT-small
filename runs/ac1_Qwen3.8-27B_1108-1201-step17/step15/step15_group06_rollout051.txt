The "Barker" bound suggests that for binary sequences, the aperiodic autocorrelation can't be too flat. For non-negative real sequences, we have more freedom.

a[j] = (j+1) * (n-j) for j = 0, ..., n-1 (a "parabola" or "tent" shape)

For this, sum a = sum (j+1)(n-j) = n(n+1)(n+2)/6 (as computed).
The convolution would be more "flat" because the shape is symmetric and smooth.

Actually, for a symmetric shape, conv(a,a)[n-1] = sum a[j]^2, which by Cauchy-Schwarz is >= (sum a)^2/n. So the score is >= 2. Not helpful.

For an asymmetric shape (like the ramp), we can get below 2.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Nesterov optimization
with ramp-based initialization, multi-length exploration, and coordinate descent polishing.
Key improvements over previous:
- Ramp-based initialization exploiting the 4/3 asymptotic bound
- Adaptive active set with tighter tolerance
- Multi-length exploration (n=1600 to n=3200)
- Warm-started LP solves
- Extended coordinate descent with L-BFGS-like memory
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver_cache = None

def get_best_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    candidates = [cp.CLARABEL, cp.PDLP, cp.SCS]
    for s in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                return s
        except Exception:
            continue
    _solver_cache = cp.SCS
    return cp.SCS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))

def build_T(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

def solve_lp_max_sum(a, M, solver, active_idx, b_prev=None):
    n = len(a)
    T = build_T(a, active_idx)
    b = cp.Variable(n, nonneg=True)
    if b_prev is not None:
        b.value = b_prev
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val

def solve_lp_min_max(a, solver, active_idx, normalize_sum=1.0):
    n = len(a)
    T = build_T(a, active_idx)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    prob = cp.Problem(cp.Minimize(t), [T @ b <= t, cp.sum(b) == normalize_sum, b >= 0])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val

def nesterov_lp(a0, time_limit, solver, seed=0, verbose=True):
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    stall_count = 0
    b_prev = None

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set - tighter tolerance
        tol = 0.015 + 0.01 * np.exp(-t / 500.0)
        threshold = M_curr * (1.0 - tol)
        active_set = np.where(conv_y >= threshold)[0]
        if len(active_set) < 10:
            active_set = np.where(conv_y >= M_curr * 0.8)[0]
        if len(active_set) > 0.7 * size:
            active_set = np.where(conv_y >= M_curr * 0.92)[0]

        # Max-sum LP
        b_max = solve_lp_max_sum(y, M_curr, solver, active_set, b_prev)

        # Min-max LP every 5 iterations
        b_min = None
        if t % 5 == 0:
            b_min, _ = solve_lp_min_max(y, solver, active_set, normalize_sum=norm)

        if b_max is None:
            b_max = solve_lp_max_sum(y, M_curr * 1.01, solver, active_set, b_prev)
        if b_max is None:
            continue

        # Blend max-sum and min-max
        g = b_max * norm / b_max.sum()
        if b_min is not None and b_min.sum() > 1e-10:
            g_min = b_min * norm / b_min.sum()
            blend = 0.25
            g = (1.0 - blend) * g + blend * g_min

        # Nesterov update
        alpha_t = max(2.0 / (t + 2.0), 0.05)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new
        b_prev = b_max

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > 100:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            b_prev = None
            stall_count = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def ramp_init(n, style=0):
    x = np.linspace(0, 1, n)
    if style == 0:
        a = np.arange(1, n + 1, dtype=np.float64)
    elif style == 1:
        a = np.arange(1, n + 1, dtype=np.float64) ** 0.8
    elif style == 2:
        a = np.arange(1, n + 1, dtype=np.float64) ** 1.2
    elif style == 3:
        a = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** 0.3)
    elif style == 4:
        a = np.sin(np.pi * x * 0.5) ** 0.5
    elif style == 5:
        a = x ** 0.7 * (1 - x) ** 0.2
    else:
        a = np.arange(1, n + 1, dtype=np.float64) * (1.0 + 0.1 * np.sin(2 * np.pi * x * 3))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def coord_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-10 * norm:
                continue

            # Try a range of values for a[i]
            best_vi = fast_autocorr(a).max()
            best_val = ai
            for mult in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]:
                a_try = a.copy()
                a_try[i] = ai * mult
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                sc = fast_autocorr(a_try).max()
                if sc < best_vi - 1e-13:
                    best_vi = sc
                    best_val = ai * mult

            if abs(best_val - ai) > 1e-12:
                a_try = a.copy()
                a_try[i] = best_val
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    sc = fast_autocorr(a_try).max()
                    if sc < best_score - 1e-13:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss
