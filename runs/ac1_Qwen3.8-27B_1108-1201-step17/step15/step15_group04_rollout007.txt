def fast_conv_self(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    A = rfft(a, fft_size)

    """Build the convolution matrix T where (T @ b)[k] = (a*b)[k]."""

Minimizes 2*n*max(autocorr(a))/sum(a)^2 using accelerated Mirror Descent
with LP best-response, adaptive constraint screening, Nesterov acceleration,
Polyak averaging, symmetry enforcement, and gradient-based final polish.

from scipy.fft import rfft, irfft

def fast_conv_self(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    """Build T where (T @ b)[k] = (a conv b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def solve_best_response_lp(a, M, solver, active_idx=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    if active_idx is not None and 0 < len(active_idx) < size:
        constraints = [T[active_idx, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active_idx is not None and len(active_idx) < size:
                duals = np.zeros(size)
                duals[active_idx] = np.asarray(con.dual_value, dtype=np.float64)
            else:

        pass

def compute_score(a):

def dual_gradient(a, duals):
    """Compute gradient using LP dual variables."""
    n = len(a)

    D = rfft(duals, fft_size)
    g1 = irfft(np.conj(A) * D, fft_size)[:n]
    g2 = irfft(A * D, fft_size)[:n]
