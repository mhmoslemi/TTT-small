- But we need to maintain `a = IDFT(A) >= 0`

This is essentially a "peak reduction" problem in the frequency domain, similar to PAPR reduction in OFDM.

- Identify the "worst" frequency bins (those contributing most to the peak)
- Reduce their magnitudes
- Project back to the non-negative constraint (set negative `a[i]` to 0 and re-FFT)

Now, let me think about the frequency-domain peak shaving. The idea is:
1. Compute `A = rfft(a, fft_size)`
2. Compute `|A|^2` (power spectrum)
3. The autocorrelation is `conv = irfft(|A|^2, fft_size)[:2n-1]`
4. Find the peak lag `k*`
5. To reduce `conv[k*]`, we need to reduce the contribution of frequencies that constructively interfere at lag `k*`
6. Specifically, `conv[k*] = (1/fft_size) * sum_j |A[j]|^2 * exp(2πi*j*k*/fft_size)`
7. Frequencies with `exp(2πi*j*k*/fft_size)` having positive real part contribute positively
8. Reduce those frequencies, increase others (maintaining non-negativity of `a`)

This is similar to the "proj" algorithm for PAPR reduction. Let me implement a simplified version:

```python
def freq_peak_shaving(a, time_limit, n_iters=100, verbose=True):
    """Reduce autocorrelation peak via frequency-domain shaping."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    fft_size = 1 << (2*n-2).bit_length()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        power = np.abs(A)**2
        conv = irfft(power, fft_size)[:2*n-1]
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]
        
        # Compute the "phase alignment" at the peak lag
        # conv[peak_idx] = (1/fft_size) * sum_j |A[j]|^2 * exp(2πi*j*peak_idx/fft_size)
        # The real part of exp(2πi*j*peak_idx/fft_size) tells us the sign of contribution
        j = np.arange(len(power))
        phase = np.cos(2 * np.pi * j * peak_idx / fft_size)
        
        # Reduce power at frequencies with positive phase alignment
        # Increase at negative (maintain total power)
        max_reduction = 0.3
        power_new = power.copy()
        power_new[phase > 0.5] *= (1 - max_reduction)
        power_new[phase < -0.5] *= (1 + max_reduction)
        
        # Reconstruct A with modified power (keep phases)
        A_new = A.copy()
        A_new = A_new * np.sqrt(power_new / (power + 1e-30))
        
        # Reconstruct a
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new
        
        sc = fast_autocorr(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_score
```

Actually, a better frequency-domain approach would be:
1. Compute the autocorrelation `conv`

3. We want `conv[k] <= M' < M` for the peak lags

Let me instead focus on a better "coordinate descent" approach:

**Peak-targeted coordinate descent:**
1. Find the peak lag `k*` and value `M`
2. For each pair `(i, j)` with `i < j`:
   - Try moving mass from `i` to `j` (or vice versa)
   - Compute the effect on `conv[k*]` (and nearby lags)
   - Accept if the peak decreases

The effect of moving mass `d` from position `p` to position `q`:
- `conv[k]` changes by: `d * (a[k-p] - a[k-q]) + d * (a[k-(n-1-p)] - a[k-(n-1-q)])` (approximately, for the cross terms)
- Plus second-order terms: `-d^2` at lag `2p` and `+d^2` at lag `2q` (approximately)

```python
def build_conv_row(a, k):
    n = len(a)
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    row = np.zeros(n)
    if i_hi >= i_lo:
        row[i_lo:i_hi + 1] = a[k - i_hi:k - i_lo + 1][::-1]
    return row


def solve_lp_with_constraints(a, M, active_indices, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_indices:
        row = build_conv_row(a, int(k))
        constraints.append(row @ b_var <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(b, 0.0)

            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n0 = len(best_seq)

    # Phase 1: LP Nesterov on parent size (45% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.45
    print(f"\n=== Phase 1: LP Nesterov n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n0) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, top_k=40, verbose=True)
        try_update(a_out, f"lp1_n{n0}")

    # Phase 2: Peak-targeted CD (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.18
    print(f"\n=== Phase 2: Peak CD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
        a_out, _ = peak_cd(a_norm, t2, verbose=True)
        try_update(a_out, f"cd1_n{len(best_seq)}")

    # Phase 3: LP round 2 (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LP R2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, top_k=50, verbose=True)
        try_update(a_out, f"lp2_n{n_cur}")

    # Phase 4: Frequency-domain shave (8% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Freq shave n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = freq_shave(a_norm, t4, verbose=True)
        try_update(a_out, f"freq_n{n_cur}")

    # Phase 5: Subgradient descent (7% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.09
    print(f"\n=== Phase 5: Subgrad n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgrad_descent(a_norm, t5, verbose=True)
        try_update(a_out, f"sg_n{n_cur}")

    # Phase 6: Try larger n (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.12
    print(f"\n=== Phase 6: Larger n exploration (budget={t6:.0f}s) ===")
    if t6 > 30:
        n_try = n0 + 400
        # Interpolate parent to new size
        old = best_seq
        new = np.zeros(n_try)
        for i in range(n_try):
            pos = i * (len(old) - 1) / (n_try - 1)
            i0 = int(pos)
            i1 = min(i0 + 1, len(old) - 1)
            frac = pos - i0
            new[i] = old[i0] * (1 - frac) + old[i1] * frac
        new = np.maximum(new, 0)
        s_new = new.sum()
        if s_new > 1e-10:
            new = new / s_new
            a_norm = new * np.sqrt(2.0 * n_try) / new.sum()
            t_half = t6 * 0.5
            a_out, _ = nesterov_lp_optimize(a_norm, t_half, solver, top_k=35, verbose=False)
            try_update(a_out, f"lp_n{n_try}")

    # Phase 7: Final LP polish (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_cur = len(best_seq)
        t7 = remaining * 0.7
        print(f"\n=== Phase 7: Final LP n={n_cur} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, top_k=45, verbose=False)
        try_update(a_out, f"lp_final_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_cur = len(best_seq)
        print(f"\n=== Phase 8: Final CD n={n_cur} (budget={remaining*0.6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_cd(a_norm, remaining * 0.5, verbose=False)
        try_update(a_out, f"cd_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
