def fast_autocorr(a):

    size = 2 * n - 1

def normalize(a, target_sum=None):
    """Normalize a to have the target sum (default: 1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    # Build constraint matrix T where T @ b = conv(a, b)
    # T is (2n-1) x n
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_subgradient(a0, time_limit, top_k=20, verbose=True):
    """
    Nesterov-accelerated subgradient method on f(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Works with sum(a) = 1 normalization, minimizing 2n*max(conv(a,a)).
    """
    n = len(a0)
    a = normalize(a0, 1.0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    y = a.copy()  # Nesterov lookahead
    a_prev = a.copy()
    t_param = 1.0
    eta0 = 1.0 / n
    patience = 0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        conv_y = fast_autocorr(y)
        max_val = conv_y.max()
        
        # Find top-k active lags
        k_actual = min(top_k, 2 * n - 1)
        active_idx = np.argsort(conv_y)[-k_actual:]
        
        # Compute subgradient: average of gradients at active lags
        # ∂c_k/∂a_i = 2*a_{k-i} (with bounds)
        grad = np.zeros(n)
        for k in active_idx:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            for i in range(i_lo, i_hi + 1):
                grad[i] += 2.0 * y[k - i]
        grad /= k_actual
        
        # Nesterov update
        alpha = 2.0 / (t_param + 2.0)
        
        # Gradient step
        y_step = y - eta0 * alpha * grad
        y_step = project_simplex(y_step, 1.0)
        
        # Nesterov extrapolation
        momentum = (t_param - 1.0) / (t_param + 2.0)
        a_new = y_step + momentum * (y_step - y)
        a_new = project_simplex(a_new, 1.0)
        
        y = y_step
        a = a_new
        t_param += 1.0
        
        # Evaluate
        score = 2.0 * n * fast_autocorr(a).max()
        score_y = 2.0 * n * conv_y.max()
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 100 and it % 50 == 0:
            # Reset with perturbation
            rng = np.random.default_rng(it)
            noise = rng.normal(0, 0.01, n)
            a_pert = a + noise
            a = project_simplex(a_pert, 1.0)
            y = a.copy()
            t_param = 1.0
            eta0 = 1.0 / n
            patience = 0
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG iter={it} ({elapsed:.0f}s): best={best_score:.8f}, t={t_param:.0f}")
    
    return best_a, best_score


def active_set_cd(a0, time_limit, verbose=True):
    """
    Coordinate descent that directly targets the active lags.
    For each active lag k, try to reduce c_k by adjusting pairs (i, k-i).
    """
    n = len(a0)
    a = normalize(a0, 1.0)
    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    pass_count = 0
    no_improve = 0
    
    top_lags = 15
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 50:
            break
        
        pass_count += 1
        conv = fast_autocorr(a)
        current_max = conv.max()
        active_idx = np.argsort(conv)[-top_lags:]
        
        improved = False
        
        # For each active lag, try to reduce it
        for k in active_idx:
            k = int(k)
            if current_max < best_score * (1 - 1e-6):
                break
            
            # The lag k involves pairs (i, k-i) for valid i
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            
            # Try moving mass within the pairs that contribute to c_k
            # c_k = sum_{i=i_lo}^{i_hi} a_i * a_{k-i}
            # To reduce c_k, we want to reduce the products a_i * a_{k-i}
            # Strategy: find the largest product and try to unbalance it
            
            products = a[i_lo:i_hi+1] * a[k-i_hi:k-i_lo+1][::-1]
            max_prod_idx = np.argmax(products)
            
            i_max = i_lo + max_prod_idx
            j_max = k - i_max
            
            if i_max == j_max:
                # Can't easily fix self-product
                continue
            
            # Try to move mass from the larger of a[i_max], a[j_max] to elsewhere
            if a[i_max] >= a[j_max]:
                src = i_max
                dst = j_max
            else:
                src = j_max
                dst = i_max
            
            # Try moving some mass from src to a random other position
            for _ in range(5):
                target = rng.integers(0, n)
                if target == src or target == dst:
                    continue
                
                # Try different fractions
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    delta = a[src] * frac
                    a[src] -= delta
                    a[target] += delta
                    s = a.sum()
                    if s < 1e-15:
                        a[src] += delta
                        a[target] -= delta
                        continue
                    a = a / s
                    
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    new_score = 2.0 * n * new_max
                    
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        a = a / a.sum()
                
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            # Perturb
            if no_improve % 10 == 0:
                noise = rng.laplace(0, 0.005, n)
                a_pert = a + noise
                a = normalize(a_pert, 1.0)
        else:
            no_improve = 0
        
        if verbose and pass_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ASCD pass={pass_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
