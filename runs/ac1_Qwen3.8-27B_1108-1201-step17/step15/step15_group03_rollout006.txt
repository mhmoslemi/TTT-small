import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a) * target_sum
    return a * target_sum / s


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def fista_smooth(a0, time_limit, verbose=True):
    n = len(a0)

    x = np.linspace(0.0, 1.0, n)

        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4 + 0.2 * np.sin(3 * np.pi * x)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sinc(3.0 * (x - 0.5)) ** 2 + 0.1
    elif choice == 9:
        a = np.exp(-3.0 * (x - 0.4) ** 2) * (1 + 0.3 * np.cos(10 * np.pi * x))
    elif choice == 10:
        a = 0.2 + 0.8 * np.exp(-5.0 * (x - 0.5) ** 2)
    else:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / (0.15 + 0.1 * rng.random()))
    a = np.maximum(a, 0.0)
    a = a + 0.01 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: FISTA (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: FISTA n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 25:
        a_out, _ = fista_smooth(best_seq, t1, verbose=True)
        try_update(a_out, "fista1")

    # Phase 2: LP with adaptive heavy ball
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 400.0)
    print(f"\n=== Phase 2: LP-HB n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 40:
        a_out, _ = lp_adaptive_heavy_ball(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp_hb1")

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 3: CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = coord_descent_fast(best_seq, t3, rng, verbose=True)
        try_update(a_out, "cd1")

    # Phase 4: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: LP-HB round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_adaptive_heavy_ball(best_seq, t4, solver, verbose=True)
        try_update(a_out, "lp_hb2")

    # Phase 5: Coord descent round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: CD round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = coord_descent_fast(best_seq, t5, rng, verbose=True)
        try_update(a_out, "cd2")

    # Phase 6: Gradient perturb + refine
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 6: GradPerturb n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = gradient_perturb_and_refine(best_seq, t6, rng, verbose=True)
        try_update(a_out, "grad_perturb")

    # Phase 7: Final CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final CD polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 15:
        a_out, _ = coord_descent_fast(best_seq, remaining * 0.9, rng, verbose=False)
        try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
