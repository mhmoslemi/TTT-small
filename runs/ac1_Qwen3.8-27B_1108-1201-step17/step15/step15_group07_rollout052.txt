            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_beta, 'ftol': 1e-12, 'gtol': 1e-8, 'disp': False}
        )
        x = res.x
        a_new = softmax(x)
        conv_max = fast_conv(a_new).max()
        if conv_max < best_obj:
            best_obj = conv_max
            best_a = a_new.copy()
    
    return best_a, best_obj

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbosity=1):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    patience = 0
    restart_count = 0
    momentum_t = 1.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        if patience > 100 and n_iters % 50 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            restart_count += 1
            if verbosity > 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbosity > 0 and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score

def peak_targeted_search(a0, time_limit, top_k=20, verbosity=1):
    """
    Directly target the peak lags and try to reduce them.
    Strategy: for each peak lag k, try to redistribute mass to reduce conv[k].
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    conv = fast_conv(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(12345)
    n_iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        max_val = conv.max()
        active_idx = np.argsort(conv)[-top_k:]
        
        improved = False
        # For each active lag, try to reduce it by adjusting the pairs (i, k-i)
        for k in active_idx:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            
            # conv[k] = sum_{i=i_lo}^{i_hi} a[i] * a[k-i]
            # To reduce this, we can try to reduce products a[i]*a[k-i]
            # by moving mass from larger to smaller factors
            
            for i in range(i_lo, i_hi + 1):
                j = k - i
                if i == j:
                    # a[i]^2 term: reduce by moving mass from i
                    if a[i] > norm * 0.01:
                        # Move some mass from i to a random other position
                        target = rng.integers(0, n)
                        while target == i:
                            target = rng.integers(0, n)
                        d = a[i] * 0.1
                        a[i] -= d
                        a[target] += d
                        new_conv_max = fast_conv(a).max()
                        if new_conv_max < best_score - 1e-14:
                            best_score = new_conv_max
                            best_a = a.copy()
                            improved = True
                        else:
                            a[i] += d
                            a[target] -= d
                else:
                    # a[i]*a[j] term: reduce by making them more unequal
                    # (product is maximized when they're equal)
                    # Actually, for fixed sum, product is max at equality
                    # So to reduce, we want to increase inequality
                    # Move mass from the smaller to the larger (counterintuitively!)
                    # Wait: if we increase the difference, the product decreases
                    # So move mass from min(a[i],a[j]) to max(a[i],a[j])
                    if a[i] < a[j]:
                        src, dst = i, j
                    else:
                        src, dst = j, i
                    if a[src] > norm * 0.01:
                        d = a[src] * 0.1
                        a[src] -= d
                        a[dst] += d
                        new_conv_max = fast_conv(a).max()
                        if new_conv_max < best_score - 1e-14:
                            best_score = new_conv_max
                            best_a = a.copy()
                            improved = True
                        else:
                            a[src] += d
                            a[dst] -= d
        
        n_iters += 1
        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Perturb
                noise = rng.normal(0, norm * 0.01, n)
                a = np.maximum(a + noise, 0)
                s = a.sum()
                a = a * norm / s
                no_improve = 0
        else:
            no_improve = 0

        if verbosity > 0 and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PTS iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def spectrum_flatten(a0, time_limit, verbosity=1):
    """
    Try to flatten the power spectrum |A(m)|^2 which is related to the
    circular autocorrelation. This can help reduce peaks in the linear convolution.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    a = a * norm / max(s0, 1e-15)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for iteration in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute DTFT (using FFT as approximation for circular convolution)
        A = rfft(a, 2*n)
        power = A * np.conj(A)
        power = np.real(power)
        
        # Try to flatten: a_new = IDFT(phase(A) * sqrt(target_power))
        # but this may give negative values...
        # Instead, use iterative amplitude-phase separation (Gerchberg-Saxton-like)
        target_power = np.full_like(power, power.mean())
        # Smooth the target slightly
        target_power = target_power * (1 + 0.1 * np.random.randn(len(target_power)))
        target_power = np.maximum(target_power, 0)
        
        new_A = A / np.sqrt(np.maximum(np.abs(A)**2, 1e-30)) * np.sqrt(target_power)
        a_new = irfft(new_A, 2*n)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new * norm / s
            sc = fast_conv(a_new).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
        
        if verbosity > 0 and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SF iter={iteration} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
