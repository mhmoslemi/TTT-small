            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                step *= 0.5
                continue
            a_trial = a_trial * S / s_trial
            conv_trial = fast_autocorr(a_trial)
            if conv_trial.max() < best_conv_max * (1 + 1e-10):
                best_conv_max = conv_trial.max()
                best_a = a_trial.copy()
                a = a_trial
                found_improvement = True
                patience = 0
                break
            step *= 0.5

        if not found_improvement:
            patience += 1
            # Try a slightly different step
            a_trial = a - lr * 0.1 * direction
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 1e-15:
                a_trial = a_trial * S / s_trial
                conv_trial = fast_autocorr(a_trial)
                if conv_trial.max() < best_conv_max:
                    best_conv_max = conv_trial.max()
                    best_a = a_trial.copy()
                    a = a_trial
                    patience = 0
                    stall_count = 0
                else:
                    stopatience = patience
                    if patience % 500 == 0:
                        # Restart from best
                        a = best_a.copy()
                        m = np.zeros(n)
                        v = np.zeros(n)
                        patience = 0
                        lr = max(0.001, lr * 0.8)

        # Periodically vary the temperature for exploration
        if t % 2000 == 0:
            # Try a perturbation
            rng_step = np.random.default_rng(t)
            a_pert = a * (1.0 + 0.01 * rng_step.standard_normal(n))
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert * S / sp
                cp_pert = fast_autocorr(a_pert)
                if cp_pert.max() < best_conv_max:
                    best_conv_max = cp_pert.max()
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    patience = 0

        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothGrad iter={n_iters} ({elapsed:.0f}s): best_max={best_conv_max:.10f}, K={K}, tau={tau:.1f}, lr={lr:.4f}")

    return best_a, best_conv_max


def spectral_shaping(a0, time_limit, verbose=False):
    """
    Frequency-domain iterative approach: alternately flatten the power spectrum
    and enforce time-domain non-negativity.
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * S / a.sum()

    best_a = a.copy()
    best_conv_max = fast_autocorr(a).max()

    # FFT size for linear convolution
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()

    t0 = time.monotonic()
    t = 0
    alpha = 0.3  # peak reduction strength
    patience = 0

    while time.monotonic() - t0 < time_limit:
        if patience > 5000:
            break
        t += 1

        # Pad and compute DFT
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = rfft(a_pad, L)
        power = A.real ** 2 + A.imag ** 2  # |A|^2

        # Compute autocorrelation from power spectrum
        corr = irfft(power, L)
        # corr[0:n-1] should match fast_autocorr(a)[:n-1] approximately
        # The full linear autocorrelation is in corr[0:size]
        conv_from_fft = np.concatenate([corr[:n], corr[L - n + 1:]])
        # Actually, the linear autocorrelation from circular autocorrelation of zero-padded:
        # conv[k] = corr[k] for k=0,...,n-1 and conv[k] = corr[k-L] for k=n,...,2n-2
        # But since we zero-padded, corr[k] = 0 for k > n-1 and k < L-n+1
        # So conv = corr[0:size] should work if L >= 2n-1

        conv_check = fast_autocorr(a)

        # Flatten the power spectrum: reduce peaks
        power_mean = power.mean()
        # Additive peak reduction
        excess = np.maximum(0, power - power_mean * 1.5)
        power_new = power - alpha * excess
        power_new = np.maximum(power_new, 0.0)

        # Reconstruct: keep phase, modify amplitude
        amplitude_new = np.sqrt(power_new)
        phase = np.angle(A)
        A_new = amplitude_new * np.exp(1j * phase)

        # IFFT to time domain
        a_pad_new = np.real(irfft(A_new, L))
        a_new = a_pad_new[:n]

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            # Failed, try softer reduction
            alpha *= 0.5
            patience += 1
            continue
        a_new = a_new * S / s_new

        conv_new = fast_autocorr(a_new)
        if conv_new.max() < best_conv_max:
            best_conv_max = conv_new.max()
            best_a = a_new.copy()
            a = a_new.copy()
            patience = 0
            # Gradually increase aggressiveness
            alpha = min(alpha * 1.01, 0.8)
        else:
            patience += 1
            # Try different alpha
            alpha = max(0.05, alpha * 0.98)
            a = best_a.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpectralShape iter={t} ({elapsed:.0f}s): best_max={best_conv_max:.10f}, alpha={alpha:.3f}")

    return best_a, best_conv_max


def build_constraint_matrix(a, k_indices, n):
    """Build constraint matrix for LP: T @ b <= M where T[k,j] = a[k-j] if valid."""
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a[int(k) - j]
    return mat


def lp_best_response(a_fixed, M, active_set, solver, n):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M, b >= 0."""
    k_indices = np.asarray(active_set)
    T = build_constraint_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, None


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response iteration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    size = 2 * n - 1

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 30
    n_iters = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if active_set is None or t % 30 == 0:
            tol = max(0.005, 0.04 * np.exp(-t / 200.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            top_k = np.argsort(conv_y)[-max(3, n // 150):]
            active_set = np.unique(np.concatenate([active_set, top_k]))

        b, _ = lp_best_response(y, M_curr, active_set, solver, n)

        if b is None:
            if t % 20 == 0:
                active_set = np.arange(size)
                b, _ = lp_best_response(y, M_curr, active_set, solver, n)
            if b is None:
                continue

        # Verify and expand if needed
        conv_ab = fast_autocorr(np.concatenate([y, np.zeros(n)]))  # placeholder
        # Quick check: compute y*b at all lags
        T_all = build_constraint_matrix(y, np.arange(size), n)
        viol = T_all @ b
        viol_idx = np.where(viol > M_curr * (1 + 1e-6))[0]
        if len(viol_idx) > 0:
            new_active = np.union1d(active_set, viol_idx)
            if len(new_active) > 0.85 * size:
                new_active = np.arange(size)
            b, _ = lp_best_response(y, M_curr, new_active, solver, n)
            if b is None:
                continue
            active_set = new_active

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha_n = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_n) * y + alpha_n * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.97)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average check
        if n_iters % 30 == 0:
            erg = (y + a + best_a) / 3.0
            s_e = erg.sum()
            if s_e > 1e-15:
                erg = erg * norm / s_e
                score_e = fast_autocorr(erg).max()
                if score_e < best_score:
                    best_score = score_e
                    best_a = erg.copy()
                    stall_count = 0

        # Restart
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False):
    """
    Fast coordinate descent: for each coordinate, try several candidate values
    and pick the best. Uses precomputed background for efficiency.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    size = 2 * n - 1
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve_passes >= 3:
            break
        improved = False
        conv = fast_autocorr(a)

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = a.sum() - aj

            # Try several candidate values
            candidates = [0.0, aj * 0.5, aj * 0.8, aj, aj * 1.2, aj * 1.5,
                          s_others / n * 0.5, s_others / n * 1.5]
            candidates = [x for x in candidates if 0 <= x <= norm * 2]

            best_x = aj
            best_obj = 2.0 * n * conv.max() / (a.sum() ** 2)

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                # Compute new conv efficiently
                # Delta = 2*x*(cross with a) + x^2 at lag 2j - contribution of old aj
                # Simpler: just compute full conv (expensive but correct)
                a[j] = x
                a_scaled = a * norm / a.sum()
                conv_new = fast_autocorr(a_scaled)
                obj = conv_new.max()
                if obj < best_obj:
                    best_obj = obj
                    best_x = x
            a[j] = best_x
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
                new_score = fast_autocorr(a).max()
                if new_score < best_score * (1 - 1e-12):
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                conv = fast_autocorr(a)

        if not improved:
            no_improve_passes += 1
        else:
            no_improve_passes = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return a_interp / ss


def perturb_sequence(a, rng, scale=0.03):
    """Add multiplicative log-normal noise."""
    n = len(a)
    noise = np.exp(scale * rng.standard_normal(n))
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
