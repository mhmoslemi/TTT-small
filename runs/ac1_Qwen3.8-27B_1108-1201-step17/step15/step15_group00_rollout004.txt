def adam_optimizer(n, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Simple Adam optimizer."""
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    return m, v, t

def adam_step(m, v, t, grad, lr, beta1, beta2, eps):
    t += 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v, t, step

def peak_shaving(a0, time_limit, verbose=True):

    by moving mass from contributing positions to non-contributing ones."""

    n = len(a0)
    a = norm_simplex(a0)

        M_curr = aac.max()

        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_peak - idx
        contrib = a[idx] * a[j_idx]
        
        if contrib.max() < 1e-20:
            break
        
        # The position with maximum contribution
        src_pos = idx[np.argmax(contrib)]
        
        # Find a good destination: a position that minimizes the increase in autocorrelation
        # For each candidate destination d, the change in aac is approximately:
        # delta_aac[k] += a[k-d]*(-delta) + a[d*(k)]*(+delta) ... 
        # More precisely, if we move delta from position s to position d:
        # a' = a - delta*e_s + delta*e_d
        # a'*a' = a*a - delta*(a*s + a*d... ) 
        # Actually: (a + delta*(e_d - e_s)) * (a + delta*(e_d - e_s))
        # = a*a + delta*(a*(e_d-e_s) + (e_d-e_s)*a) + delta^2*(e_d-e_s)*(e_d-e_s)
        # The first-order change in b[k]: delta * (a[k-d] - a[k-s] + a[d-k] - a[s-k])
        # (with appropriate bounds)
        
        # For efficiency, only check a subset of candidates
        n_candidates = min(n, 200)
        # Prefer positions far from the peak
        # Compute "peak sensitivity" for each position
        peak_sens = np.zeros(n)
        # For the top few autocorrelation peaks
        top_k_idx = np.argsort(aac)[::-1][:min(10, 2*n-1)]
        for k2 in top_k_idx:

            for i in range(max(0, k2_int - n + 1), min(n, k2_int + 1)):
                j = k2_int - i
                if 0 <= j < n:
                    peak_sens[i] += a[j]
        
        # Sort by peak sensitivity (ascending - we want low sensitivity positions)
        candidates = np.argsort(peak_sens)[:n_candidates]
        
        best_dst = -1
        best_delta_M = float('inf')
        
        for d in candidates:
            d = int(d)
            if d == src_pos:
                continue
            # Estimate change in max autocorrelation
            delta = a[src_pos] * 0.01  # small move
            if delta < 1e-18:
                continue
            
            # First-order change in each b[k]:
            # delta_b[k] = delta * (a[k-d] - a[k-s]) where a[x]=0 if x not in [0,n-1]
            # We only care about the max, so compute for top peaks
            max_increase = 0.0
            for k2 in top_k_idx[:5]:
                k2_int = int(k2)
                def a_at(pos):
                    return a[pos] if 0 <= pos < n else 0.0
                change = delta * (a_at(k2_int - d) - a_at(k2_int - src_pos))
                # Also the other direction (since conv is symmetric in a sense)
                change += delta * (a_at(d - k2_int) - a_at(src_pos - k2_int))
                if change > max_increase:
                    max_increase = change
            
            if max_increase < best_delta_M:
                best_delta_M = max_increase
                best_dst = d
        
        if best_dst < 0:
            stall += 1
            continue
        
        # Try the move
        dst = best_dst
        found = False
        for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
            amt = a[src_pos] * frac
            if amt < 1e-18 or a[src_pos] - amt < -1e-15:
                continue
            a_trial = a.copy()
            a_trial[src_pos] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                stall = 0
                found = True
                break

            stall += 1

    if score < _best_score:
        _best_score = score
        _best_seq = best_a.copy()
        print(f"  *** Peak shaving new best: {score:.6f} (n={n})")
    
    return best_a
