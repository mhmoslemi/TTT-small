                active = new_active

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            no_improve_count = 0

        # If stuck, perturb
        if no_improve_count > patience:
            n_pert = max(5, n // 50)
            idx_p = np.random.choice(n, size=n_pert, replace=False)
            a[idx_p] *= (1.0 + 0.1 * np.random.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a
            y = a.copy()
            no_improve_count = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def dual_guided_step(a, y, M, solver, eta=0.5):
    """Use dual variables from LP to compute a guided gradient step."""
    n = len(a)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)

    active = np.where(fast_conv_self(y) >= M * 0.7)[0]
    if len(active) < 100:
        active = np.argsort(fast_conv_self(y))[-300:]

    b, dual_vals = lp_solve_restricted(y, M, solver, active)
    if dual_vals is None or b is None:
        return a

    # Compute gradient using dual weights
    T = build_T(y, active)
    grad = T.T @ dual_vals
    grad = 2.0 * grad  # factor of 2 for the convolution

    a_new = a - eta * grad / (np.linalg.norm(grad) + 1e-15)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    return a_new * norm / s_new

def peak_cd(a0, time_limit, rng, verbose=True):
    """Peak-targeted pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        if cur_max < 1e-15:
            break

        perm = rng.permutation(n)
        pass_improved = False

        # Identify significant peaks
        peak_thresh = cur_max * 0.90
        peak_lags = np.where(conv >= peak_thresh)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(conv)])

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Compute pressure on coordinate i from peaks
            pressure = 0.0
            for kl in peak_lags[:20]:
                j_idx = int(kl) - i
                if 0 <= j_idx < n:
                    pressure += a[j_idx]
                j_idx2 = i - int(kl)
                if 0 <= j_idx2 < n:
                    pressure += a[j_idx2]

            if pressure < cur_max * 0.1:
                continue

            # Find transfer targets (low pressure, or structured)
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j1 = (int(kl) - i + n) % n
                if j1 != i:
                    j_candidates.add(j1)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                if j == i or j < 0 or j >= n:
                    continue
                aj = a[j]

                # Compute gain vector for transfer i -> j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 15
                deltas = np.linspace(0.0, ai * 0.9, n_samples)
                for d in deltas:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-14:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                pass_improved = True
                total_iters += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                n_perturb = max(8, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_iters % 500 == 0 and total_iters > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD iter={total_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def gs_init(n, rng):
    """Gerchberg-Saxton inspired initialization for flat autocorrelation."""
    a = rng.random(n)
    a = a / a.sum() * np.sqrt(2 * n)
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(50):
        A = rfft(a, fft_size)
        phase = np.angle(A)
        target_mag = np.sqrt(fft_size / (2 * n)) * np.sqrt(2 * n / n)
        A_new = target_mag * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a = np.maximum(a_new, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a * np.sqrt(2 * n) / s

    return a

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.3 + 0.2 * np.sin(4 * np.pi * x)
    elif choice == 7:
        a = gs_init(n, rng) / np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    a = a + 0.01 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    solver = get_solver()
    print(f"Budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick LSE smoothing
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 60.0)
    print(f"\n=== Phase 1: LSE smoothing n={n_base} ({t1:.0f}s) ===")
    if t1 > 15:
        a_work = best_seq.copy()
        a_out, _ = lse_smooth(a_work, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.40, 650.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} ({t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Peak CD
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 3: Peak CD n={n3} ({t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_cd(best_seq, t3, rng, verbose=True)
        try_update(a_out, "pcd1")

    # Phase 4: LP Nesterov round 2 (on the polished sequence)
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.22, 300.0)
    print(f"\n=== Phase 4: LP Nesterov r2 n={n4} ({t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov(best_seq, t4, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 5: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        for n_try in [n_base + 200, n_base - 200]:
            if n_try < 500 or remaining < 60:
                break
            t5 = min(remaining * 0.4, 150.0)
            print(f"\n=== Phase 5: Length {n_try} ({t5:.0f}s) ===")
            if t5 > 20:
                init_try = interpolate_seq(best_seq, n_try)
                a_out, _ = lp_nesterov(init_try, t5 * 0.6, solver, verbose=False)
                try_update(a_out, f"lp_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 20:
                    a_out2, _ = peak_cd(a_out if a_out is not None else init_try,
                                         min(remaining * 0.4, 40.0), rng, verbose=False)
                    try_update(a_out2, f"pcd_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)

    # Phase 6: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_final} ({remaining:.0f}s) ===")
    if remaining > 20:
        t6a = min(remaining * 0.5, 60.0)
        a_out, _ = peak_cd(best_seq, t6a, rng, verbose=False)
        try_update(a_out, "pcd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.7, 80.0)
            a_out, _ = lp_nesterov(best_seq, t6b, solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
