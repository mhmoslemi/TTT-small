def fast_autocorr(a):
    """Compute full linear autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_seq(a, norm_val=None):
    """Normalize sequence to have sum = norm_val (default sqrt(2n))."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        if norm_val is None:
            norm_val = np.sqrt(2.0 * n)
        return np.ones(n) * norm_val / n
    if norm_val is None:
        norm_val = np.sqrt(2.0 * n)
    return a * norm_val / s


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    if S < 1e-15:
        return np.zeros(n)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho_mask = v_sorted * np.arange(1, n + 1) > (css - S)
    rho_idx = np.where(rho_mask)[0]
    if len(rho_idx) == 0:
        return np.full(n, S / n)
    rho = rho_idx[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_rows(a, row_indices):
    """Build rows of the convolution matrix T where T[j,i] = a[j-i] if valid."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros((len(row_indices), n))
    for r, j in enumerate(row_indices):
        i_lo = max(0, j - (n - 1))
        i_hi = min(n - 1, j)
        if i_lo <= i_hi:
            rows[r, i_lo:i_hi + 1] = a[i_lo:i_hi + 1]
    return rows


def compute_smooth_grad(a, w):
    """Gradient of T*log(sum exp(c/T)): grad_i = 2 * cross_corr(w, a)[i]."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross


def lp_best_response(a, M, solver, active_rows=None, max_cg_iters=20):
    """
    Solve max sum(b) s.t. T*b <= M, b >= 0 using constraint generation.
    Starts with active_rows and adds violated constraints iteratively.
    """
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    conv_a = fast_autocorr(a)

    if active_rows is None:
        threshold = 0.9 * M
        active_rows = np.where(conv_a >= threshold)[0]
        if len(active_rows) < 5:
            active_rows = np.where(conv_a >= 0.5 * M)[0]
        if len(active_rows) < 3:
            active_rows = np.arange(size)

    b_var = cp.Variable(n, nonneg=True)
    current_rows = active_rows.copy()

    for cg_iter in range(max_cg_iters):
        T_rows = build_conv_rows(a, current_rows)
        constraints = [T_rows @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)

        # Check for constraint violations
        if len(current_rows) < size:
            conv_b = fast_autocorr(np.concatenate([a, b_val]))  # wrong, need a*b not b*b
            # Actually need (a*b)[k] for all k
            # a*b: full convolution of a and b
            fft_size = 1 << (2 * n - 1).bit_length()
            A_f = rfft(a, fft_size)
            B_f = rfft(b_val, fft_size)
            conv_ab = irfft(A_f * B_f, fft_size)[:2 * n - 1]

            # Find constraints violated by more than tol
            tol = 1e-6 * M
            violated = conv_ab > M + tol
            if np.any(violated):
                # Add the most violated constraints
                viol_idx = np.where(violated)[0]
                # Add up to 20 most violated
                viol_conv = conv_ab[viol_idx]
                order = np.argsort(viol_conv)[::-1][:20]
                new_rows = viol_idx[order]
                current_rows = np.unique(np.concatenate([current_rows, new_rows]))
                continue
            else:
                break
        else:
            break

    if b_val is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, max_iters=10000):
    """LP best-response with Nesterov acceleration and constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    best_avg_score = best_score
    best_avg_a = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best individual iterate
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Track running average (often better than individual iterates)
        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Also track a "tail average" (average of last K iterates)
        if n_avg % 10 == 0 and n_avg >= 20:
            # We'd need to store all iterates for this, so just use the running average
            pass

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def smooth_adam_optimize(a0, time_limit, verbose=True):
    """Minimize smooth approximation using Adam optimizer with projected updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_step = 0

    t0 = time.monotonic()

    # Temperature schedule - more aggressive annealing
    temps = [10.0, 5.0, 3.0, 2.0, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05, 0.03, 0.02, 0.01, 0.006, 0.003, 0.001]

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(20, min(300, int(remaining * 4)))
        lr_base = norm / (n * np.sqrt(T))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            t_step += 1
            c = fast_autocorr(a)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = compute_smooth_grad(a, w)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)

            lr = lr_base / (1.0 + 0.01 * np.sqrt(t_step))
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Project to simplex
            a_new = project_simplex(a_new, norm)
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth-Adam T={T:.3f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            a_new = project_simplex(a_new, norm)

            # Nesterov
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        a = y  # Continue from y

    return best_a, best_score


def targeted_pair_cd(a0, time_limit, rng, verbose=True, max_window=60):
    """
    Pair coordinate descent that targets peak-contributing pairs.
    For each peak position k*, the pairs (i, k*-i) contribute to c[k*].
    We try to redistribute mass within those pairs to reduce the peak.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        no_improve += 1
        if no_improve > 5:
            break
        n_passes += 1
        pass_improved = False

        conv = fast_autocorr(a)
        c_max = conv.max()

        # Find top peak positions
        n_peaks = min(10, conv.size)
        peak_threshold = 0.9 * c_max
        peak_positions = np.where(conv >= peak_threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-10:
                continue

            # Find which peaks this index contributes to
            # a[i] contributes to c[k] for k in [i, i+n-1] (as a[i]*a[k-i])
            # and also c[k] for k in [i-n+1, i] (as a[k-i+n-1]... actually for full conv)
            # For full conv c[k] = sum_j a[j]*a[k-j], a[i] appears when j=i (giving a[i]*a[k-i])
            # or when k-j=i (j=k-i, giving a[k-i]*a[i])
            # So a[i] contributes to c[k] for k-i in [0,n-1], i.e., k in [i, i+n-1]
            # AND for k-i in [-(n-1), -1+... actually just k in [i, i+n-1] covers it since
            # the other direction is k-i < 0 which means j = k-i < 0 which is invalid.
            # Wait, for full convolution c[k] = sum_j a[j]*a[k-j] where 0<=j<n and 0<=k-j<n
            # So j = i contributes when 0 <= k-i < n, i.e., i <= k <= i+n-1.
            # And j = k-i contributes when 0 <= k-i < n, same condition.
            # So a[i] appears in c[k] for k in [i, i+n-1].

            i_low = i
            i_high = min(i + n - 1, 2 * n - 2)
            relevant_peaks = peak_positions[(peak_positions >= i_low) & (peak_positions <= i_high)]

            if len(relevant_peaks) == 0:
                continue

            # For each relevant peak, try moving mass from i to j where j is paired
            # c[k] includes a[i]*a[k-i], so the "partner" of i for peak k is j = k-i
            for k in relevant_peaks[:3]:  # Limit to top 3 peaks
                j = k - i
                if j < 0 or j >= n or j == i:
                    continue
                if a[j] < norm * 1e-12:
                    continue

                # Try moving a small amount from i to j and vice versa
                # a[i] -> a[j]: reduces a[i]*a[j] in c[k] (if i!=j)
                # Actually c[k] = ... + a[i]*a[j] + ..., so reducing a[i] or a[j] reduces c[k]
                # But we also need to check effect on other c[k'] values

                # Try moving delta from i to j (or j to i)
                delta = min(a[i], a[j]) * 0.2
                for sign in [1, -1]:
                    a_test = a.copy()
                    src = i if sign > 0 else j
                    dst = j if sign > 0 else i
                    move = delta * sign
                    if a_test[src] + move < 0:
                        continue
                    a_test[src] += move
                    a_test[dst] -= move
                    a_test = np.maximum(a_test, 0)
                    s_test = a_test.sum()
                    if s_test < 1e-15:
                        continue
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < best_score - 1e-12:
                        a = a_test
                        best_score = conv_test.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

            # Also try moving to nearby positions (not just peak partners)
            for offset in [1, 2, 3, 5, 10, 20, 40]:
                j1 = (i + offset) % n
                j2 = (i - offset) % n
                for j in [j1, j2]:
                    if j == i or time.monotonic() - t0 > time_limit:
                        continue
                    delta = a[i] * 0.1
                    if delta < norm * 1e-12:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    s_test = a_test.sum()
                    if s_test < 1e-15:
                        continue
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < best_score - 1e-12:
                        a = a_test
                        best_score = conv_test.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Targeted PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def make_raised_cosine(n):
    """Raised cosine window: (1 + cos(2*pi*i/n))/2."""
    i = np.arange(n)
    return (1.0 + np.cos(2.0 * np.pi * i / n)) / 2.0


def make_two_bump(n, width=0.15):
    """Two Gaussian bumps at n/4 and 3n/4."""
    i = np.arange(n, dtype=np.float64)
    center1 = n * 0.25
    center2 = n * 0.75
    sigma = width * n
    a = np.exp(-0.5 * ((i - center1) / sigma) ** 2) + np.exp(-0.5 * ((i - center2) / sigma) ** 2)
    return a


def make_edge_weighted(n, p=2.0):
    """Weighted toward edges: (1 - |2*i/n - 1|)^p."""
    i = np.arange(n, dtype=np.float64)
    x = np.abs(2.0 * i / n - 1.0)
    return (1.0 - x) ** p


def make_gaussian(n, width=0.2):
    """Centered Gaussian."""
    i = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = width * n
    return np.exp(-0.5 * ((i - center) / sigma) ** 2)


def make_power_law(n, alpha=1.5):
    """Power-law decay from center: (1 + |i - n/2|/n)^(-alpha)."""
    i = np.arange(n, dtype=np.float64)
    d = np.abs(i - n / 2.0) / n
    return (1.0 + d) ** (-alpha)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LP-Nesterov on parent (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.30
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n1))
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        try_update(a_out, f"lp_n{n1}")

    # === Phase 2: Smooth Nesterov on best (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = remaining * 0.15
    print(f"\n=== Phase 2: Smooth Nesterov n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = smooth_nesterov_optimize(a_norm, t2, verbose=True)
        try_update(a_out, f"smooth_n{n_c}")

    # === Phase 3: LP-Nesterov on diverse init: raised cosine (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    n3 = n_base
    print(f"\n=== Phase 3: LP on raised cosine n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        init3 = make_raised_cosine(n3)
        init3 = normalize_seq(init3, np.sqrt(2.0 * n3))
        # Blend with parent for continuity
        blend = 0.3
        init3 = (1 - blend) * init3 + blend * normalize_seq(best_seq, np.sqrt(2.0 * n3))
        init3 = normalize_seq(init3, np.sqrt(2.0 * n3))
        a_out, _ = nesterov_lp(init3, t3, solver, verbose=True)
        try_update(a_out, f"lp_rc_n{n3}")

    # === Phase 4: LP-Nesterov on diverse init: two-bump (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    n4 = n_base
    print(f"\n=== Phase 4: LP on two-bump n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        init4 = make_two_bump(n4, width=0.12)
        init4 = normalize_seq(init4, np.sqrt(2.0 * n4))
        blend = 0.3
        init4 = (1 - blend) * init4 + blend * normalize_seq(best_seq, np.sqrt(2.0 * n4))
        init4 = normalize_seq(init4, np.sqrt(2.0 * n4))
        a_out, _ = nesterov_lp(init4, t4, solver, verbose=True)
        try_update(a_out, f"lp_2bump_n{n4}")

    # === Phase 5: Smooth-Adam on best (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Smooth-Adam n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = smooth_adam_optimize(a_norm, t5, verbose=True)
        try_update(a_out, f"adam_n{n_c}")

    # === Phase 6: Targeted pair CD on best (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: Targeted PCD n={n_c} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = targeted_pair_cd(a_norm, t6, rng, verbose=True)
        try_update(a_out, f"tpcd_n{n_c}")

    # === Phase 7: LP-Nesterov round 2 on edge-weighted (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = remaining * 0.05
    n7 = n_base
    print(f"\n=== Phase 7: LP edge-weighted n={n7} (budget={t7:.0f}s) ===")
    if t7 > 20:
        init7 = make_edge_weighted(n7, p=1.5)
        init7 = normalize_seq(init7, np.sqrt(2.0 * n7))
        blend = 0.2
        init7 = (1 - blend) * init7 + blend * normalize_seq(best_seq, np.sqrt(2.0 * n7))
        init7 = normalize_seq(init7, np.sqrt(2.0 * n7))
        a_out, _ = nesterov_lp(init7, t7, solver, verbose=False)
        try_update(a_out, f"lp_edge_n{n7}")

    # === Phase 8: Final smooth polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t8 = remaining * 0.05
    print(f"\n=== Phase 8: Final smooth polish n={n_c} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_c))
        a_out, _ = smooth_nesterov_optimize(a_norm, t8, verbose=False)
        try_update(a_out, f"final_smooth_n{n_c}")

    # === Final: try alternate lengths if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_base_final = len(best_seq)
        for n_alt in [n_base_final + 200, n_base_final - 200, n_base_final + 500]:
            if n_alt < 200:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 40:
                break
            print(f"\n=== Alt length n={n_alt} (budget={min(remaining_sub*0.3, 120):.0f}s) ===")
            t_alt = min(remaining_sub * 0.3, 120.0)
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_out_alt, _ = nesterov_lp(init_alt, t_alt, solver, verbose=False)
            try_update(a_out_alt, f"alt_n{n_alt}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
