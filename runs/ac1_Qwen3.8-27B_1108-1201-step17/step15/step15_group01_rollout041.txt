To decrease c[k*], we need a[k*-j] < a[k*-i], i.e., a[i] should be at a position where a[k*-i] is large (i.e., i is "close" to k* in the right sense), and a[j] should be at a position where a[k*-j] is small.

But we also need to make sure other c[k] don't increase too much. This is the constraint.

A greedy approach: always move from the position i that most increases c[k*] (i.e., where a[k*-i] is largest) to the position j that most decreases it (i.e., where a[k*-j] is smallest), while checking that no other c[k] increases above the current max.

This is a form of "accelerated coordinate descent" that focuses on the bottleneck constraint.

Let me implement this. Here's the algorithm:

```
repeat:
    k* = argmax c (current bottleneck)
    Find i that maximizes a[k*-i] (the position that most contributes to c[k*])
    Find j that minimizes a[k*-j] (the position that least contributes to c[k*])
    Try moving delta from i to j:
        new_c[k] = c[k] + 2*delta*(a[k-j] - a[k-i]) for all k
        if max(new_c) < c[k*]:
            accept the move
        else:
            try smaller delta (binary search)
            if no positive delta works:
                try different i, j pairs
    If no move improves: stuck
```

The issue is that this greedy approach might get stuck. To escape, we can:
- Allow temporary increases in the max (simulated annealing)
- Try random moves
- Use the LP to find a better direction

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast smooth (log-sum-exp) Nesterov gradient with adaptive temperature
2. Analytical pair coordinate descent targeting the bottleneck lag

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:

def compute_smooth_grad(a, w):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross

def project_simplex(v, S):
    n = len(v)
    v = np.array(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_nesterov(a0, time_limit, T_start=5.0, T_end=0.05, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    ntemps = 40
    log_temps = np.linspace(np.log(T_start), np.log(T_end), ntemps)
    for ti in range(ntemps):
        if time.monotonic() - t0 > time_limit:
            break
        T = log_temps[ti]
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_steps = max(50, min(5000, int(remaining * 100)))
        y = a.copy()
        y_prev = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            c = fast_autocorr(y)
            c_max = c.max()
            shift = c_max
            exp_c = np.exp((c - shift) / T)
            w = exp_c / exp_c.sum()
            grad = compute_smooth_grad(y, w)
            step = norm / (n * (1.0 + 0.1 * it))
            step = min(step, norm / (n * 0.3))
            x_new = y - step * grad
            x_new = project_simplex(x_new, norm)
            if it == 0:
                y_new = x_new.copy()
            else:
                alpha = 2.0 / (it + 2.0)
                y_new = (1.0 - alpha) * y + alpha * x_new
                y_new = project_simplex(y_new, norm)
            y_prev = y.copy()
            y = y_new
            a = x_new
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
        a = best_a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Smooth T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def fast_pair_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 200
    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            break
        k_star = int(np.argmax(conv))
        a_k = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            a_k[lo:hi+1] = np.concatenate([a[hi:lo-1:-1], a[0:lo]])[:(hi-lo+1)]
        # Actually let me compute a[k_star - i] for i = 0..n-1
        contrib = np.zeros(n)
        for i in range(n):
            k_minus_i = k_star - i
            if 0 <= k_minus_i < n:
                contrib[i] = a[k_minus_i]
            else:
                km = k_star - i
                if km >= 2*n - 1:
                    pass
                else:
                    # k_star can be 0..2n-2, i can be 0..n-1
                    # k_star - i can be negative or >= n
                    pass
        # For positions i where 0 <= k_star-i < n, moving from i reduces c[k_star]
        # For positions j where 0 <= k_star-j < n, moving to j increases c[k_star]
        # We want: move from i (high contrib) to j (low contrib)
        valid_mask = np.zeros(n, dtype=bool)
        for i in range(n):
            if 0 <= k_star - i < n:
                valid_mask[i] = True
        if valid_mask.sum() < 2:
            no_improve += 1
            continue
        valid_idx = np.where(valid_mask)[0]
        contrib_valid = contrib[valid_idx]
        # Sort by contribution: move from highest to lowest
        order = valid_idx[np.argsort(-contrib_valid)]
        improved_this_round = False
        n_tried = 0
        max_tries = 200
        for attempt in range(max_tries):
            if time.monotonic() - t0 > time_limit:
                break
            if n_tried > 50:
                break
            # Pick source i (high contribution) and target j (low contribution)
            i_src = order[attempt % len(order)]
            j_dst = order[-1 - (attempt % len(order))]
            if i_src == j_dst:
                continue
            if a[i_src] < norm * 1e-10:
                continue
            n_tried += 1
            # Try moving delta from i_src to j_dst
            delta = a[i_src] * 0.5
            # Quick check: does this reduce c[k_star]?
            dck = 2.0 * delta * (a[k_star - j_dst] if 0 <= k_star - j_dst < n else 0.0 - (a[k_star - i_src] if 0 <= k_star - i_src < n else 0.0))
            # Actually: change in c[k] = 2*delta*(a[k-j_dst] - a[k-i_src]) for the k_star component
            a_kj = a[k_star - j_dst] if 0 <= k_star - j_dst < n else 0.0
            a_ki = a[k_star - i_src] if 0 <= k_star - i_src < n else 0.0
            change_at_ks = 2.0 * delta * (a_kj - a_ki)
            if change_at_ks >= 0:
                continue
            # Full check
            a_test = a.copy()
            a_test[i_src] -= delta
            a_test[j_dst] += delta
            if a_test[i_src] < 0:
                delta = a[i_src]
                a_test = a.copy()
                a_test[i_src] -= delta
                a_test[j_dst] += delta
            a_test = a_test * norm / a_test.sum()
            conv_test = fast_autocorr(a_test)
            new_max = conv_test.max()
            if new_max < best_score - 1e-12:
                a = a_test
                conv = conv_test
                best_score = new_max
                best_a = a.copy()
                improved_this_round = True
                no_improve = 0
        if not improved_this_round:
            no_improve += 1
            # Try random perturbation to escape
            if no_improve % 50 == 0:
                noise = rng.normal(0, 0.02 * norm / n, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert * norm / a_pert.sum()
                conv_pert = fast_autocorr(a_pert)
                if conv_pert.max() < best_score:
                    a = a_pert
                    conv = conv_pert
                    best_score = conv_pert.max()
                    best_a = a.copy()
                    no_improve = 0
        if verbose and no_improve % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")
    return best_a, best_score

def block_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    block_sizes = [5, 10, 20, 50]
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        if no_improve > 100:
            break
        k_star = int(np.argmax(conv))
        improved = False
        bs = block_sizes[no_improve % len(block_sizes)]
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            # Pick a random block to move
            i_start = rng.integers(0, max(1, n - bs))
            j_start = rng.integers(0, max(1, n - bs))
            if abs(i_start - j_start) < bs:
                continue
            block_a = a[i_start:i_start+bs].sum()
            if block_a < norm * 1e-8:
                continue
            delta_frac = 0.3
            delta = block_a * delta_frac
            a_test = a.copy()
            a_test[i_start:i_start+bs] -= delta / bs
            a_test[j_start:j_start+bs] += delta / bs
            if (a_test < -1e-12).any():
                continue
            a_test = np.maximum(a_test, 0)
            a_test = a_test * norm / a_test.sum()
            conv_test = fast_autocorr(a_test)
            if conv_test.max() < best_score - 1e-12:
                a = a_test
                conv = conv_test
                best_score = conv_test.max()
                best_a = a.copy()
                improved = True
                no_improve = 0
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                noise = rng.normal(0, 0.05 * norm / n, n)
                a_pert = np.maximum(a + noise, 0)
                a_pert = a_pert * norm / a_pert.sum()
                conv_pert = fast_autocorr(a_pert)
                if conv_pert.max() < best_score:
                    a = a_pert
                    conv = conv_pert
                    best_score = conv_pert.max()
                    best_a = a.copy()
                    no_improve = 0
        if verbose and no_improve % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BlockCD ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def generate_initializations(n, rng):
    inits = {}
    inits['uniform'] = np.ones(n) / n
    x = np.arange(n)
    inits['gaussian'] = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    inits['cos2'] = np.cos(np.pi * x / (n - 1) + np.pi/2) ** 2
    inits['ramp'] = (x + 1).astype(float)
    inits['ramp_rev'] = (n - x).astype(float)
    inits['triangle'] = np.minimum(x + 1, n - x).astype(float)
    inits['w_shape'] = np.cos(2 * np.pi * x / n) ** 2
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['bimodal'] = np.exp(-0.5 * ((x - n*0.25) / (n*0.1))**2) + np.exp(-0.5 * ((x - n*0.75) / (n*0.1))**2)
    for k in inits:
        inits[k] = np.maximum(inits[k], 0)
        s = inits[k].sum()
        if s > 1e-15:
            inits[k] = inits[k] / s
    return inits

def multiplicative_weights_direction(a, n_iter=500):
    """Use multiplicative weights to approximate the LP best-response direction."""
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    if M < 1e-15:
        return a.copy()
    # Game: row player (min) picks distribution over lags 0..2n-2
    # Column player (max) picks distribution over positions 0..n-1
    # Payoff matrix T[k,i] = a[k-i] if 0<=k-i<n else 0
    # We want: min_p max_i sum_k p[k]*T[k,i]
    # MW update for p: p[k] *= exp(eta * T[k,i_col])
    p = np.ones(2 * n - 1) / (2 * n - 1)
    v = np.ones(n) / n
    eta = 1.0
    for it in range(n_iter):
        # Column player best response: i* = argmax_i sum_k p[k]*T[k,i]
        # sum_k p[k]*a[k-i] = cross_corr(p, a)[i]
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        P = rfft(p, fft_size)
        A = rfft(a, fft_size)
        payoff = irfft(P * np.conj(A), fft_size)[:n]
        i_star = int(np.argmax(payoff))
        # Update p: p[k] *= exp(eta * T[k, i_star] / (2n-1))
        # T[k, i_star] = a[k - i_star] if 0 <= k - i_star < n
        T_col = np.zeros(2 * n - 1)
        lo = i_star
        hi = min(i_star + n - 1, 2 * n - 2)
        if lo <= hi:
            T_col[lo:hi+1] = a[:hi-lo+1]
        p = p * np.exp(eta * T_col / (2 * n - 1))
        p = p / p.sum()
        # Update v: v[i] *= exp(eta * T[i*, i] / n) ... actually v is the column player's strategy
        # For our purpose, we just need p (the "certifying" distribution)
    # The value of the game approximates max(c)/M... 
    # Actually, we want the best-response direction: b = argmax sum(b) s.t. T@b <= M
    # The dual gives: min M*sum(p) s.t. sum_k p[k]*T[k,i] >= 1/n for all i
    # So b direction is related to the slack: b[i] is large where the constraint is tight
    # For now, just return a smoothed version of a weighted by the "sensitivity"
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    sensitivity = irfft(P * np.conj(A), fft_size)[:n]
    # The direction to move: decrease where sensitivity is high, increase where low
    b = np.maximum(sensitivity.min() + 0.1 - sensitivity, 0)
    b_sum = b.sum()
    if b_sum < 1e-15:
        return a.copy()
    b = b * a.sum() / b_sum
    return b

            do_update(a_out, f"alt_n{n_alt}")
            # Also try CD on this
            remaining6 = time_budget - (time.monotonic() - t_global)
            if remaining6 > 30:
                a_out2, _ = fast_pair_cd(best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum(), min(remaining6 * 0.3, 100.0), rng, verbose=False)
                do_update(a_out2, "pcd_final")
    # Final: one more short CD pass
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        n_final = len(best_seq)
        a_final = best_seq * np.sqrt(2.0 * n_final) / best_seq.sum()
        a_out, _ = fast_pair_cd(a_final, min(remaining * 0.8, 200.0), rng, verbose=False)
        do_update(a_out, "pcd_ultimate")
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    return [float(max(0, min(1000.0, x))) for x in best_seq]
