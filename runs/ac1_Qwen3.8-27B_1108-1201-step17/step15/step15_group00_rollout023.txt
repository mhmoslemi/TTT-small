            if prob.status in ('optimal', 'optimal_inaccurate'):

                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_subgradient(a, c, top_k=20):
    """Exact subgradient of max(c) using uniform weights on top-k lag values."""
    n = len(a)
    k_vals = np.argsort(c)[::-1][:min(top_k, len(c))]
    g = np.zeros(n)
    for k in k_vals:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        # c[k] = sum a[j]*a[k-j], d/d a[i] = a[k-i] + a[i-k] (if in range)
        contrib = np.zeros(len(idx))
        for p, i in enumerate(idx):
            j1 = k_int - i
            if 0 <= j1 < n:
                contrib[p] += a[j1]
            j2 = i - k_int
            if 0 <= j2 < n:
                contrib[p] += a[j2]
        g[idx] += contrib
    g /= len(k_vals)
    return g


def mirror_descent_exact(a0, time_limit, top_k=20, verbose=True, rng_seed=42):
    """Mirror descent with exact subgradient and adaptive step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    eta = 1.0 / np.sqrt(n)
    stall = 0
    max_stall = 2000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        c = fast_conv(y)
        M_curr = c.max()
        g = get_subgradient(y, c, top_k=top_k)

        # Adaptive step size: decrease if M increased, increase if decreased
        a_new = y * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            M_new = fast_conv(a_new).max()
        else:
            a_new = y.copy()
            M_new = M_curr

        if M_new <= M_curr:
            x_prev = x
            x = a_new
            eta = min(eta * 1.05, 2.0 / np.sqrt(n))
            stall = 0
        else:
            # Reject step, shrink eta
            eta = eta * 0.7
            if eta < 1e-8 / np.sqrt(n):
                # Reset with perturbation
                x = best_a + rng.normal(0, 0.005, n)
                x = np.maximum(x, 1e-10)
                x = x / x.sum()
                x_prev = x.copy()
                eta = 1.0 / np.sqrt(n)
                stall += 500

        M_x = fast_conv(x).max()
        if M_x < best_m:
            best_m = M_x
            best_a = x.copy()
        else:
            stall += 1

        if stall > max_stall:
            # Restart from best with larger perturbation
            x = best_a * (1 + rng.normal(0, 0.02, n))
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            eta = 1.0 / np.sqrt(n)
            stall = 0
            max_stall = int(max_stall * 1.5)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}")

    return best_a, best_m


def coordinate_swaps(a0, time_limit, verbose=True, rng_seed=123):
    """Two-point coordinate swaps to reduce max convolution value."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        # Find hot lag positions
        hot_k = np.argsort(c)[::-1][:10]

        # Find positions contributing most to hot lags
        contrib = np.zeros(n)
        for k in hot_k[:5]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            for i in idx:
                j = k_int - i
                if 0 <= j < n:
                    contrib[i] += a[j]
                    contrib[j] += a[i]

        # Source: positions with high contribution AND high value
        src_candidates = np.argsort(contrib * a)[::-1][:20]
        # Destination: positions with low contribution
        dst_candidates = np.argsort(contrib)[:min(n // 4, 100)]

        improved = False
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord swaps ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe: find direction maximizing sum subject to conv constraint."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    iter_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(y)
        M_curr = c.max()
        if M_curr < 1e-14:
            break

        # Solve LP: max sum(b) s.t. b>=0, (y*b)[k] <= M_curr for all k
        b_var = cp.Variable(n, nonneg=True)
        # Build constraint matrix: (y*b)[k] = sum_j y[j]*b[k-j]
        # For efficiency, use dense matrix
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new
        iter_count += 1

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_to_new_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_asymmetric_init(n, alpha=0.3):
    """Create an asymmetric initialization: uniform tilted by a linear ramp."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2 * x - 1)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_two_lobe_init(n, center1=0.25, center2=0.75, width=0.15):
    """Two-lobed initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center1) / width) ** 2) + np.exp(-((x - center2) / width) ** 2)
    return a / a.sum()


def make_quadratic_init(n, skew=0.2):
    """Quadratic profile with optional skew."""
    x = np.linspace(0, 1, n)
    y = x * (1 - x)
    a = 1.0 + skew * (2 * x - 1)
    a = y * a
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def multi_start_mirror_descent(a0, time_limit, n_starts=3, verbose=True):
    """Run mirror descent from multiple perturbations of a0, keep best."""
    n = len(a0)
    rng = np.random.default_rng(777)
    best_a = a0.copy()
    best_m = fast_conv(a0).max()
    t0 = time.monotonic()

    time_per = (time_limit - 10) / n_starts

    for s in range(n_starts):
        if time.monotonic() - t0 > time_limit - 5:
            break

        if s == 0:
            init = norm_sum1(a0)
        else:
            # Perturb the best so far
            init = best_a * (1 + rng.normal(0, 0.03 * (s + 1), n))
            init = np.maximum(init, 1e-10)
            init = init / init.sum()

        remain = time_limit - (time.monotonic() - t0)
        t_alloc = min(time_per, remain)
        if t_alloc < 5:
            break

        a_out, m_out = mirror_descent_exact(init, t_alloc, top_k=15, verbose=False)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Multi-start {s}: J={2.0*n*best_m:.6f} ({elapsed:.0f}s)")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = 1.0 + 0.2 * (2 * x - 1)
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror descent at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror descent n={n0} ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_exact(a_in, t1, top_k=20, verbose=True)
        try_update(a_out, f"md_n{n0}")

    # === Phase 2: Coordinate swaps polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coord swaps n={len(best_seq)} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_swaps(a_in, t2, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 3: Scale to larger length + optimize ===
    n1 = min(n0 + 600, 3600)
    remaining = time_budget - (time.monotonic() - t_global)
    if n1 > n0 and remaining > 100:
        print(f"\n=== Phase 3: Scale to n={n1} + MD ===")
        a_long = interpolate_to_new_length(best_seq, n1)
        a_in = norm_sum1(a_long)
        t3 = min(remaining * 0.20, 300.0)
        if t3 > 30:
            a_out, _ = multi_start_mirror_descent(a_in, t3, n_starts=2, verbose=True)
            try_update(a_out, f"md_n{n1}")
            # Coord polish at new length
            remaining = time_budget - (time.monotonic() - t_global)
            t3b = min(remaining * 0.08, 100.0)
            if t3b > 15:
                a_in2 = norm_sum1(best_seq)
                a_out2, _ = coordinate_swaps(a_in2, t3b, verbose=True, rng_seed=99)
                try_update(a_out2, f"coord_n{len(best_seq)}")

    # === Phase 4: Try even larger length ===
    n2 = min(n1 + 800, 4800)
    remaining = time_budget - (time.monotonic() - t_global)
    if n2 > n1 and remaining > 120:
        print(f"\n=== Phase 4: Scale to n={n2} + MD ===")
        a_long2 = interpolate_to_new_length(best_seq, n2)
        a_in = norm_sum1(a_long2)
        t4 = min(remaining * 0.25, 350.0)
        if t4 > 30:
            a_out, _ = multi_start_mirror_descent(a_in, t4, n_starts=2, verbose=True)
            try_update(a_out, f"md_n{n2}")

    # === Phase 5: LP Frank-Wolfe at best length (if small enough) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 3000 and remaining > 60:
        print(f"\n=== Phase 5: LP n={n_best} ===")
        t5 = min(remaining * 0.20, 200.0)
        if t5 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True)
            try_update(a_out, f"lp_n{n_best}")

    # === Phase 6: Diverse initializations at current best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 6: Diverse inits n={n_best} ===")
        inits = []
        inits.append(make_asymmetric_init(n_best, alpha=0.4))
        inits.append(make_asymmetric_init(n_best, alpha=-0.4))
        inits.append(make_quadratic_init(n_best, skew=0.3))
        inits.append(make_two_lobe_init(n_best, 0.2, 0.8, 0.2))

        # Blend with current best
        for idx in range(min(3, len(inits))):
            blend = 0.5 * norm_sum1(best_seq) + 0.5 * inits[idx]
            blend = norm_sum1(blend)
            inits[idx] = blend

        t_per = min(remaining * 0.15, 60.0)
        for idx, init in enumerate(inits[:3]):
            if time.monotonic() - t_global > time_budget - 30:
                break
            t6 = min(t_per, time_budget - 30 - (time.monotonic() - t_global))
            if t6 > 10:
                a_out, _ = mirror_descent_exact(init, t6, top_k=15, verbose=False)
                try_update(a_out, f"diverse{idx}_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 7: Final coord n={n_best} ===")
        t7 = min(remaining * 0.5, 150.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_swaps(a_in, t7, verbose=True, rng_seed=777)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Final mirror descent refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final MD n={n_best} ===")
        t8 = min(remaining * 0.7, 200.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent_exact(a_in, t8, top_k=30, verbose=False)
            try_update(a_out, f"md_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
