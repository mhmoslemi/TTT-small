def logsumexp_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*C_k)) where C = conv(a,a)."""
    n = len(a)
    C = fast_autocorr(a)
    # Stable softmax
    tau_C = tau * C
    tau_C -= tau_C.max()
    p = np.exp(tau_C)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(C)) / len(C)
    else:
        p /= p_sum
    # grad[m] = 2 * sum_j a[j] * p[j+m]
    # Compute via cross-correlation using FFT
    # a_rev * p gives (a ⊛ p)[m] at index n-1+m
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(A * P, fft_size)[:size]
    grad = 2.0 * conv_result[n-1:2*n-1]
    return grad

```python
def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def logsumexp_grad(a, tau):
    n = len(a)
    C = fast_autocorr(a)
    tau_C = tau * C
    tau_C -= tau_C.max()

def topk_grad(a, top_k, beta):
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        constraints = [cp.sum(b_var) == 1, T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.1f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full log-sum-exp smoothed gradient (via FFT cross-correlation) with Nesterov mirror descent
2. Adaptive temperature annealing for smooth-to-sharp transition
3. Top-k soft subgradient with adaptive parameters (diversity)
4. Projected gradient descent with backtracking (fast local polish)
5. Aggressive coordinate descent (final polish)
6. LP-based Frank-Wolfe (structural improvement)
7. Multi-length exploration and multi-start
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def logsumexp_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*C_k)) where C=conv(a,a).
    Uses FFT-based cross-correlation for O(n log n) computation."""
    n = len(a)
    C = fast_autocorr(a)
    tau_C = tau * C
    tau_C -= tau_C.max()
    p = np.exp(tau_C)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(C)) / len(C)
    else:
        p /= p_sum
    # grad[m] = 2 * sum_j a[j] * p[j+m]  (cross-correlation)
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(A * P, fft_size)[:size]
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def topk_grad(a, top_k, beta):
    """Compute weighted subgradient focusing on top-k peaks."""
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_logsumexp(a0, time_limit, tau_start=5.0, tau_end=150.0, c0=3.5, momentum_cap=0.92, verbose=True):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothed gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 6000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature: linearly increase from tau_start to tau_end
        progress = min(t / 40000.0, 1.0)
        tau = tau_start + (tau_end - tau_start) * progress

        # Nesterov extrapolation with momentum cap
        beta_t = min((t - 1.0) / (t + 1.5), momentum_cap)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Full log-sum-exp gradient
        g = logsumexp_grad(y, tau)

        # Mirror descent step (entropy-normalized)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_new = np.log(y) - eta * g
        max_log = log_new.max()
        x_new = np.exp(log_new - max_log)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Restart with perturbation around best
            pert = rng.normal(0, 0.02, n) * best_a
            x_re = best_a + pert
            x_re = np.maximum(x_re, 1e-10)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            stall = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-AMD t={t} ({elapsed:.1f}s): J={2*n*best_m:.8f}, tau={tau:.1f}")

    return best_a, best_m


def amd_topk(a0, time_limit, top_k=60, beta=4.0, c0=4.5, verbose=True):
    """Nesterov mirror descent with top-k soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(55555)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = min((t - 1.0) / (t + 1.5), 0.93)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        g = topk_grad(y, cur_top_k, cur_beta)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_new = np.log(y) - eta * g
        max_log = log_new.max()
        x_new = np.exp(log_new - max_log)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            pert = rng.normal(0, 0.015, n) * best_a
            x_re = best_a + pert
            x_re = np.maximum(x_re, 1e-10)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            stall = 0
            cur_top_k = min(cur_top_k + 15, 100)
            cur_beta = max(cur_beta - 1.0, 2.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TopK-AMD t={t} ({elapsed:.1f}s): J={2*n*best_m:.8f}, k={cur_top_k}")

    return best_a, best_m


def projected_gradient_polish(a0, time_limit, tau=80.0, verbose=True):
    """Fast local polish: projected gradient steps with backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    step = 1.0
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        g = logsumexp_grad(a, tau)
        g_proj = g - g.mean()
        g_norm = np.sqrt(np.sum(g_proj ** 2))
        if g_norm < 1e-12:
            break

        # Try step with backtracking
        found = False
        for _ in range(20):
            direction = -g_proj / g_norm
            a_trial = a + step * direction
            # Entropy-like projection: clip and normalize
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                found = True
                # Slightly increase step
                step = min(step * 1.05, 2.0)
                break
            else:
                step *= 0.7
        if not found:
            step = max(step, 0.01)
            # Try a random nudge to escape
            rng = np.random.default_rng(it)
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.01
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    step = 0.5

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ProjGrad it={it} ({elapsed:.1f}s): J={2*n*best_m:.8f}, step={step:.4f}")

    return best_a, best_m


def coord_descent_aggressive(a0, time_limit, verbose=True, seed=42):
    """Aggressive coordinate descent using gradient-informed direction selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        g = logsumexp_grad(a, 60.0)
        g_proj = g - g.mean()

        improved = False
        # Source: indices with high positive g_proj (want to decrease)
        # Dest: indices with high negative g_proj (want to increase)
        src_order = np.argsort(g_proj)[::-1]
        dst_order = np.argsort(g_proj)

        n_tried = 0
        max_tries = min(500, n)

        for si in src_order[:40]:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(si)
            if a[i] < 1e-12:
                continue
            for di in dst_order[:40]:
                n_tried += 1
                if n_tried > max_tries:
                    break
                j = int(di)
                if j == i:
                    continue
                for frac in [0.002, 0.006, 0.012, 0.025, 0.05, 0.10, 0.18]:
                    amt = a[i] * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < -1e-15:
                        break
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.06)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.1f}s): J={2*n*best_m:.8f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        constraints = [cp.sum(b_var) == 1, T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.1f}s): J={2*n*best_m:.8f}")

    return best_a, best_m


def try_update(state, a, label):
    """Try to update the best sequence in the state dict."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < state['score']:
        state['score'] = sc
        state['seq'] = a_norm.copy()
        elapsed = time.monotonic() - state['t0']
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies:
    - Log-sum-exp gradient mirror descent (primary, full gradient via FFT)
    - Top-k subgradient mirror descent (diversity)
    - Projected gradient polish (fast local improvement)
    - Coordinate descent (fine-tuning)
    - LP Frank-Wolfe (structural moves)
    - Multi-length exploration
    """
    t0 = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    state = {'seq': None, 'score': float('inf'), 't0': t0}

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            state['seq'] = parent / parent.sum()
            state['score'] = p_score
    except Exception:
        pass

    if state['seq'] is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        init = np.sin(np.pi * x) ** 2
        state['seq'] = init / init.sum()
        state['score'] = evaluate_safe(state['seq'])
        print(f"Default init: n={n_base}, score={state['score']:.8f}")

    n0 = len(state['seq'])

    # === Phase 1: LSE-AMD (main workhorse, ~35%) ===
    remaining = time_budget - (time.monotonic() - t0)
    print(f"\n=== Phase 1: LSE-AMD n={n0} ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 30:
        a_in = norm_sum1(state['seq'])
        a_out, _ = amd_logsumexp(a_in, t1, tau_start=5.0, tau_end=200.0, c0=3.5, momentum_cap=0.92, verbose=True)
        try_update(state, a_out, f"lse1_n{n0}")

    # === Phase 2: LSE-AMD with different params (refinement, ~15%) ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(state['seq'])
    print(f"\n=== Phase 2: LSE-AMD refine n={n_best} ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(state['seq'])
        a_out, _ = amd_logsumexp(a_in, t2, tau_start=30.0, tau_end=300.0, c0=2.5, momentum_cap=0.95, verbose=True)
        try_update(state, a_out, f"lse2_n{n_best}")

    # === Phase 3: Top-k AMD (diversity, ~15%) ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(state['seq'])
    print(f"\n=== Phase 3: TopK-AMD n={n_best} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(state['seq'])
        a_out, _ = amd_topk(a_in, t3, top_k=70, beta=3.5, c0=4.0, verbose=True)
        try_update(state, a_out, f"topk_n{n_best}")

    # === Phase 4: Projected gradient polish (~10%) ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(state['seq'])
    print(f"\n=== Phase 4: ProjGrad n={n_best} ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(state['seq'])
        a_out, _ = projected_gradient_polish(a_in, t4, tau=100.0, verbose=True)
        try_update(state, a_out, f"projgrad_n{n_best}")

    # === Phase 5: Coordinate descent (~10%) ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(state['seq'])
    print(f"\n=== Phase 5: Coord n={n_best} ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(state['seq'])
        a_out, _ = coord_descent_aggressive(a_in, t5, verbose=True, seed=42)
        try_update(state, a_out, f"coord1_n{n_best}")

    # === Phase 6: LP Frank-Wolfe (if size allows, ~10%) ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(state['seq'])
    if n_best <= 2500 and remaining > 60:
        print(f"\n=== Phase 6: LP FW n={n_best} ===")
        t6 = min(remaining * 0.10, 150.0)
        if t6 > 30:
            a_in = norm_sum1(state['seq'])
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True)
            try_update(state, a_out, f"lp_n{n_best}")

    # === Phase 7: Scale to larger length (~10%) ===
    n1 = min(n0 + 800, 3200)
    remaining = time_budget - (time.monotonic() - t0)
    if n1 > len(state['seq']) and remaining > 80:
        print(f"\n=== Phase 7: Scale to n={n1} ===")
        a_long = interpolate_sequence(state['seq'], n1)
        a_in = norm_sum1(a_long)
        t7 = min(remaining * 0.10, 150.0)
        if t7 > 30:
            a_out, _ = amd_logsumexp(a_in, t7, tau_start=5.0, tau_end=150.0, c0=3.0, momentum_cap=0.92, verbose=True)
            try_update(state, a_out, f"lse_long_n{n1}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(state['seq'])
    if remaining > 40:
        print(f"\n=== Phase 8: Final Coord n={n_best} ===")
        t8 = min(remaining * 0.30, 120.0)
        if t8 > 15:
            a_in = norm_sum1(state['seq'])
            a_out, _ = coord_descent_aggressive(a_in, t8, verbose=True, seed=999)
            try_update(state, a_out, f"coord_final_n{n_best}")

    # === Phase 9: Final LSE-AMD with high tau ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(state['seq'])
    if remaining > 50:
        print(f"\n=== Phase 9: Final LSE n={n_best} ===")
        t9 = min(remaining * 0.50, 200.0)
        if t9 > 20:
            a_in = norm_sum1(state['seq'])
            a_out, _ = amd_logsumexp(a_in, t9, tau_start=50.0, tau_end=400.0, c0=2.0, momentum_cap=0.96, verbose=True)
            try_update(state, a_out, f"lse_final_n{n_best}")

    # Final validation
    if state['seq'] is not None:
        fs = evaluate_safe(state['seq'])
        if np.isfinite(fs):
            state['score'] = fs
        print(f"\nFINAL: score={state['score']:.8f}, n={len(state['seq'])}, time={time.monotonic()-t0:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in state['seq']]
```
