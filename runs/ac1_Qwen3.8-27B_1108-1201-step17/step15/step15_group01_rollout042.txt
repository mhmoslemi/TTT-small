**Why is the non-negativity constraint so costly?**

For general (signed) sequences, we can achieve `(a*a)[k] ≈ 0` for all `k ≠ 0` (like a delta function), giving a score close to 1. But with non-negativity, we can't cancel out the sidelobes.

The best non-negative sequences would be those that are "as flat as possible" (since a flat sequence has autocorrelation that's a triangle, with peak = sum = sqrt(2n) and the max autocorrelation is at lag 0).

For a flat sequence `a_i = 1/n` (normalized to sum=1):
`(a*a)[k] = (n-|k|)/n^2` for `|k| < n`
Max is at k=0: `(a*a)[0] = n/n^2 = 1/n`

Score = `2n * (1/n) / 1 = 2`. That's the baseline.

To do better than 2, we want to "spread out" the energy in the autocorrelation. The peak at lag 0 is `(a*a)[0] = sum a_i^2`, which we want to minimize (subject to `sum a_i = 1`, `a_i >= 0`). The minimum of `sum a_i^2` is `1/n` (achieved at the uniform sequence).

But we also want the sidelobes to be small. The sidelobes are `(a*a)[k] = sum_i a_i * a_{i-k}` for `k ≠ 0`.

So the trade-off is:
- Making the sequence uniform minimizes `(a*a)[0]` but the sidelobes are also large (triangle shape).
- Making the sequence "spread out" in some other way might reduce the peak sidelobe.

**Key insight**: The peak of the autocorrelation is `max((a*a)[0], max_{k≠0} (a*a)[k])`. 

`(a*a)[0] = sum a_i^2 >= 1/n` (with equality at uniform).

def fast_autocorr(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    g[mask] = 2.0 * a[idx[mask]]
    return g


def smooth_optimize(a0, time_limit, verbose=True):
    """Smoothed Nesterov with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    c = fast_autocorr(a)
    best_score = c.max()

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Adaptive temperature: start high, decrease geometrically
    T_current = best_score / max(1, n) * 10
    T_min = best_score / (n * 100)
    T_factor = 0.75
    patience = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Current smoothed objective and gradient
        c = fast_autocorr(a)
        c_max = c.max()
        exp_c = np.exp((c - c_max) / T_current)
        w = exp_c / exp_c.sum()

        # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (cross correlation)
        # Use FFT: g = cross_corr(w, a)
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        cross = irfft(W * np.conj(A), fft_size)[:n]
        grad = 2.0 * cross

        # Nesterov steps
        n_steps = max(20, min(200, int(rem * 0.3)))
        y = a.copy()
        improve_at_start = best_score

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            # Recompute gradient at y
            c_y = fast_autocorr(y)
            c_ymax = c_y.max()
            exp_y = np.exp((c_y - c_ymax) / T_current)
            w_y = exp_y / exp_y.sum()
            W_y = rfft(w_y, fft_size)
            A_y = rfft(y, fft_size)
            cross_y = irfft(W_y * np.conj(A_y), fft_size)[:n]
            grad_y = 2.0 * cross_y

            # Step size: based on Lipschitz constant estimate
            # L ~ 2 * ||y||_2^2 (for the quadratic part)
            L_est = 4.0 * np.sum(y * y) / n
            step = 1.0 / L_est

            a_new = y - step * grad_y
            a_new = project_simplex(a_new, norm)

            # Nesterov momentum
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new * norm / s_y
            else:
                y_new = y.copy()

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = np.maximum(a_curr, 0.0)
                s_a = a_curr.sum()
                if s_a > 1e-15:
                    a_curr = a_curr * norm / s_a
                else:
                    a_curr = y_new.copy()

            y = y_new
            a = a_curr

            # Check improvement
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        # Check convergence for temperature reduction
        if best_score < improve_at_start - 1e-10 * best_score:
            patience = 0
        else:
            patience += 1

        if patience >= 3 or T_current < T_min:
            T_current *= T_factor
            patience = 0
        # Also aggressively reduce T if it's too high relative to the spread
        c_now = fast_autocorr(a)
        top2 = np.sort(c_now)[::-1]
        if len(top2) > 1:
            spread = top2[0] - top2[min(10, len(top2)-1)]
            if T_current > 2 * spread and spread > 1e-10:
                T_current = max(T_current * 0.5, T_min)

        if verbose and T_current <= best_score / n:
            elapsed = time.monotonic() - t0
            print(f"    Smooth T={T_current:.6f}: best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def subgrad_nesterov(a0, time_limit, verbose=True):
    """Subgradient Nesterov with exact max and Polyak step + heavy averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    # Lower bound on max autocorrelation: (sum a)^2 / (2n-1) = 2n/(2n-1)
    lb = 2.0 * n / (2.0 * n - 1)

    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    stuck = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(y)
        c_max = c.max()
        k_star = np.argmax(c)
        g = subgradient_fft(y, k_star)

        # Polyak step
        g_norm_sq = np.sum(g * g)
        if g_norm_sq < 1e-20:
            break
        eta = (c_max - lb) / g_norm_sq

        # Cap the step
        eta = min(eta, 1.0 / (np.linalg.norm(g, ord=np.inf) + 1e-10))

        a_new = y - eta * g
        a_new = project_simplex(a_new, norm)

        # Nesterov extrapolation
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new * norm / s_y
        else:
            y_new = y.copy()

        if t == 1:
            a_curr = a_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_curr = y_new + momentum * (y_new - y)
            a_curr = np.maximum(a_curr, 0.0)
            s_a = a_curr.sum()
            if s_a > 1e-15:
                a_curr = a_curr * norm / s_a
            else:
                a_curr = y_new.copy()

        y = y_new
        a = a_curr

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stuck = 0

        # Heavy averaging
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}")

        if stuck > 5000:
            # Perturb to escape
            noise = np.random.default_rng(t).normal(0, 0.01 * best_score / n, n)
            y = y + noise
            y = project_simplex(y, norm)
            stuck = 0

    return best_a, best_score


def pairwise_cd(a0, time_limit, rng, verbose=True):
    """Pairwise coordinate descent using O(n) correction formula."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    c = fast_autocorr(a)
    best_score = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Identify peak lags for prioritization
        c_max = c.max()
        peak_lags = np.where(c >= 0.9 * c_max)[0]

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            # Find a good j: prefer j where a[peak - j] is small
            # For each peak lag k*, we want j such that a[k*-j] is small
            best_j = -1
            best_j_score = 0
            for _ in range(min(5, n // 10)):
                j = rng.integers(0, n)
                if j == i:
                    continue
                # Score: how much this pair affects the peak
                j_score = 0
                for k_p in peak_lags[:5]:
                    mi = k_p - i
                    mj = k_p - j
                    vi = a[mi] if 0 <= mi < n else 0
                    vj = a[mj] if 0 <= mj < n else 0
                    j_score += vi - vj  # positive means j is better than i for this lag
                if j_score > best_j_score:
                    best_j_score = j_score
                    best_j = j

            if best_j < 0:
                continue
            j = best_j

            # Binary search for optimal delta (move from i to j)
            s_pair = a[i] + a[j]
            lo, hi = 0.0, s_pair * 0.95
            best_d = 0.0
            best_val = c_max

            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                mid = (lo + hi) / 2.0
                if mid <= 0:
                    lo = mid
                    continue
                # Compute new autocorrelation at affected lags
                # Δc[k] = -2*mid*a[k-i] + 2*mid*a[k-j] + mid^2*(I[k=2i]+I[k=2j]-2*I[k=i+j])
                # Affected lags: k where 0<=k-i<n or 0<=k-j<n or k=2i or k=2j or k=i+j
                delta = mid
                new_ai = a[i] - delta
                new_aj = a[j] + delta

                # Compute new c at affected positions
                affected_max = 0
                k_lo = max(0, min(i, j) - n + 1)
                k_hi = min(2 * n - 2, max(i, j) + n - 1)
                # Just check a window around affected lags
                for k in range(max(0, min(i, j)), min(2*n-1, max(i, j) + n)):
                    # Original c[k]
                    orig_ck = c[k]
                    # Correction
                    mi = k - i
                    mj = k - j
                    corr = 0.0
                    if 0 <= mi < n:
                        corr += -2.0 * delta * a[mi]
                    if 0 <= mj < n:
                        corr += 2.0 * delta * a[mj]
                    if k == 2 * i:
                        corr += delta * delta
                    if k == 2 * j:
                        corr += delta * delta
                    if k == i + j:
                        corr += -2.0 * delta * delta
                    new_ck = orig_ck + corr
                    if new_ck > affected_max:
                        affected_max = new_ck

                # The overall max is max(unaffected_max, affected_max)
                # For speed, just use affected_max as an upper bound check
                # (if affected_max < best_val, it's likely an improvement)
                if affected_max < best_val - 1e-12:
                    best_val = affected_max
                    best_d = mid
                    lo = mid
                else:
                    hi = mid

            if best_d > 0:
                a[i] -= best_d
                a[j] += best_d
                a = a * norm / a.sum()
                c = fast_autocorr(a)
                new_max = c.max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pairwise_cd_fast(a0, time_limit, rng, verbose=True):
    """Faster pairwise CD: only evaluate at peak lags for speed."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    c = fast_autocorr(a)
    best_score = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        c_max = c.max()

        # Get top peak lags
        top_k = min(20, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:top_k]

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            # Find best j by checking a few random candidates
            cands = rng.integers(0, n, size=min(3, n))
            best_j = -1
            best_score_j = -np.inf
            for j_c in cands:
                j = int(j_c)
                if j == i:
                    continue
                score_j = 0
                for kp in peak_idx[:5]:
                    mi = kp - i
                    mj = kp - j
                    vi = a[mi] if 0 <= mi < n else 0.0
                    vj = a[mj] if 0 <= mj < n else 0.0
                    score_j += a[i] * vi - a[j] * vj
                if score_j > best_score_j:
                    best_score_j = score_j
                    best_j = j

            if best_j < 0 or best_score_j <= 0:
                continue
            j = best_j

            # Ternary search for optimal split
            s_pair = a[i] + a[j]
            lo, hi = 0.0, s_pair * 0.99
            best_t = a[i]  # current value (no change)
            best_cmax = c_max

            for _ in range(12):
                if time.monotonic() - t0 > time_limit:
                    break
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                def eval_t(t_val):
                    delta = t_val - a[i]
                    if abs(delta) < 1e-14:
                        return c_max
                    # Check only at peak lags + a few neighbors
                    max_val = 0.0
                    for kp in peak_idx:
                        mi = kp - i
                        mj = kp - j
                        corr = 0.0
                        if 0 <= mi < n:
                            corr += -2.0 * delta * a[mi]
                        if 0 <= mj < n:
                            corr += 2.0 * delta * a[mj]
                        if kp == 2 * i:
                            corr += delta * delta
                        if kp == 2 * j:
                            corr += delta * delta
                        if kp == i + j:
                            corr += -2.0 * delta * delta
                        new_ck = c[kp] + corr
                        if new_ck > max_val:
                            max_val = new_ck
                    # Also check lag 0 and a few others
                    for kp in [0, n - 1, i, j, i + j]:
                        if 0 <= kp < 2 * n - 1:
                            mi = kp - i
                            mj = kp - j
                            corr = 0.0
                            if 0 <= mi < n:
                                corr += -2.0 * delta * a[mi]
                            if 0 <= mj < n:
                                corr += 2.0 * delta * a[mj]
                            if kp == 2 * i:
                                corr += delta * delta
                            if kp == 2 * j:
                                corr += delta * delta
                            if kp == i + j:
                                corr += -2.0 * delta * delta
                            new_ck = c[kp] + corr
                            if new_ck > max_val:
                                max_val = new_ck
                    return max_val

                v1 = eval_t(m1)
                v2 = eval_t(m2)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1

                if hi - lo < s_pair * 1e-6:
                    break

            t_opt = (lo + hi) / 2.0
            delta = t_opt - a[i]
            if abs(delta) > 1e-12:
                a[i] = t_opt
                a[j] = s_pair - t_opt
                a = a * norm / a.sum()
                c = fast_autocorr(a)
                new_max = c.max()
                if new_max < best_score - 1e-10:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD-Fast pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def chirp_init(n, rng):
    """Generate a chirp-based initialization."""
    i = np.arange(n)
    # Quadratic phase chirp
    phase = np.pi * i * (i + 1) / (2.0 * n)
    a = 1.0 + 0.5 * np.cos(phase)
    a = np.maximum(a, 0.0)
    a = a + rng.normal(0, 0.1, n)
    a = np.maximum(a, 0.0)
    a = a * np.sqrt(2.0 * n) / a.sum()
    return a


def smooth_init(n, rng):
    """Generate a random smooth (low-pass) initialization."""
    a = np.ones(n)
    # Add smoothed noise
    freq = rfft(rng.normal(0, 1, n))
    # Low-pass filter
    mask = np.ones(len(freq))
    cutoff = max(2, n // 20)
    mask[cutoff:] = 0
    freq = freq * mask
    noise = np.abs(irfft(freq, n))
    a = a + 0.3 * noise
    a = np.maximum(a, 0.0)
    a = a * np.sqrt(2.0 * n) / a.sum()
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent_a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_a = p / p.sum()
            parent_score = evaluate_safe(parent_a)
            if np.isfinite(parent_score):
                best_seq = parent_a.copy()
                best_score = parent_score
                print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialization
    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Try alternative initializations
    init_alternatives = []
    for init_func, name in [(chirp_init, 'chirp'), (smooth_init, 'smooth')]:
        a_alt = init_func(n_base, rng)
        sc = evaluate_safe(a_alt / a_alt.sum())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_alt / a_alt.sum()
            print(f"Better init [{name}]: score={sc:.6f}")

    print(f"Starting best: score={best_score:.6f}, n={n_base}")

    # === Phase 1: Subgradient Nesterov (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.25
    n1 = n_base
    print(f"\n=== Phase 1: SubGrad Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = subgrad_nesterov(a_norm, t1, verbose=True)
        do_try_update(a_out, f"subgrad_n{n1}")

    # === Phase 2: Smoothed Nesterov (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = remaining * 0.30
    print(f"\n=== Phase 2: Smooth Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n2) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"smooth_n{n2}")

    # === Phase 3: Fast Pairwise CD (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = remaining * 0.25
    print(f"\n=== Phase 3: PCD-Fast n={n3} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n3) / best_seq.sum()
        a_out, _ = pairwise_cd_fast(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"pcd_fast_n{n3}")

    # === Phase 4: Smooth again (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Smooth round 2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n4) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t4, verbose=False)
        do_try_update(a_out, f"smooth2_n{n4}")

    # === Phase 5: PCD-Fast round 2 (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: PCD-Fast round 2 n={n5} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n5) / best_seq.sum()
        a_out, _ = pairwise_cd_fast(a_norm, t5, rng, verbose=True)
        do_try_update(a_out, f"pcd2_n{n5}")

    # === Phase 6: Length exploration (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_base_final = len(best_seq)
        for n_alt in [n_base_final + 200, n_base_final - 200, n_base_final + 400]:
            if n_alt < 50:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 30:
                break
            print(f"\n=== Alt length n={n_alt} (budget={min(remaining_sub*0.4, 90):.0f}s) ===")
            t_alt = min(remaining_sub * 0.4, 90.0)
            init_alt = interpolate_sequence(best_seq, n_alt)
            # Quick smooth optimization
            a_out_alt, _ = smooth_optimize(init_alt, t_alt * 0.6, verbose=False)
            do_try_update(a_out_alt, f"alt_smooth_n{n_alt}")
            remaining_sub2 = time_budget - (time.monotonic() - t_global)
            if remaining_sub2 > 20 and best_seq is not None:
                n_r = len(best_seq)
                a_norm_r = best_seq * np.sqrt(2.0 * n_r) / best_seq.sum()
                a_out_r, _ = pairwise_cd_fast(a_norm_r, min(remaining_sub2 * 0.3, 40.0), rng, verbose=False)
                do_try_update(a_out_r, f"alt_pcd_n{n_r}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
