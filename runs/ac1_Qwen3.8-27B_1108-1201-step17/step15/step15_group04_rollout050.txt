The key insight for improvement: **The rate of convergence depends on the "spectral gap" of the iteration matrix.** If the iteration has a small spectral gap (i.e., it converges slowly), we need better acceleration.

Nesterov acceleration helps, but it's tuned for smooth convex optimization, not for this specific problem. Let me try a different acceleration:

**Idea: Use the LP dual to get a better update direction.**

The LP dual is:
Minimize `sum_k λ_k * M` (or equivalently, minimize `M * sum_k λ_k`)
Subject to: `sum_k λ_k * T_k^T 1 ≥ 1` (for each component of b)
`λ_k ≥ 0`

Wait, let me set up the LP properly.

Primal:
Maximize: `sum_i b_i`
Subject to: For each k in {0,...,2n-2}: `sum_i T[k,i] * b_i ≤ M`
           `b_i ≥ 0`

where `T[k,i]` is the Toeplitz matrix (shift of a).

Dual:
Minimize: `sum_k λ_k * M`
Subject to: For each i: `sum_k T[k,i] * λ_k ≥ 1`
           `λ_k ≥ 0`

The dual variable `λ_k` tells us the "price" of the k-th constraint. If `λ_k > 0`, the k-th constraint is binding.

At the dual optimum, `sum_k λ_k * M` is the maximum value of `sum(b)`.

Now, here's a key observation: **The dual variables λ_k define a "weighting" of the convolution lags.** The dual constraint `sum_k T[k,i] * λ_k ≥ 1` says that the weighted sum of `a`'s shifted values (shifted by k, weighted by λ_k) must be at least 1 at every position i.

This is exactly the condition that I derived earlier from KKT! The optimal `a` should satisfy:
`sum_k λ_k * (a shifted by k) ≥ 1` at every position (with equality at positions where `a[i] > 0`).

So the dual variables give us a "certificate" of optimality. If we can find `a` and `λ` that satisfy:
- `a ≥ 0`, `sum(a) = S`
- `conv(a,a)[k] ≤ M` for all k
- `sum_k λ_k * (a shifted by k) = 1` where `a[i] > 0`
- `λ_k ≥ 0`, `sum_k λ_k * M = S` (complementary slackness)

Then `M = S^2/(2n-1)` and the solution is optimal!

This gives us a way to measure the "gap" to optimality:
- Primal value: `M = max conv(a,a)`
- Dual value: `M_dual = S^2 / sum(b)` where `b` is the LP solution
- Gap: `M - M_dual`

When the gap is small, we're close to optimal.

# Global solver cache
_solver = None

def get_solver():
    """Try to get the fastest available LP solver."""
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    return _solver

def fast_conv_self(a):
    """Compute linear self-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    """Compute linear cross-convolution a*b using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):

def build_T(a):
    """Build the Toeplitz matrix where T[k,i] gives contribution of a[i] to conv position k."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T

def lp_solve_best_response(a, M, solver, active_indices=None):
    """
    Solve the LP: max sum(b) s.t. T(b) <= M, b >= 0.
    Returns (b_optimal, dual_variables) or (None, None) on failure.
    """
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)  # T[k, i] = a contribution to position k from component i
    
    b_var = cp.Variable(n, nonneg=True)
    
    if active_indices is not None and len(active_indices) < size:
        constraints = [T[active_indices, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    # Get dual variables
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active_indices is not None and len(active_indices) < size:
                full_duals = np.zeros(size)
                full_duals[active_indices] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    
    return val, duals

def lse_smooth_descent(a0, time_limit, verbose=False):
    """
    Gradient descent on the log-sum-exp smoothed objective.
    Minimizes t*logsumexp(conv(a,a)/t) which approximates max(conv(a,a)).
    Uses multiple temperature stages.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Temperature stages from high to low
    n_stages = 15
    temps = np.geomspace(50.0, 0.05, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(200, int(remaining * 5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # Compute softmax weights
            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -300, 0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient of t*logsumexp(conv(a,a)/t) = 2 * sum_k w_k * (T_k @ a)
            # T_k @ a gives the k-th "shifted" version of a
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            # Adaptive step size with backtracking
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            step = 1.0 / (T_val * np.sqrt(n))
            for trial in range(20):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c * 1.001:  # Allow tiny increase
                    break
                step *= 0.5
            
            a = a - step * grad
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a * norm / s
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score

def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """
    LP best-response with Nesterov acceleration.
    This is the main optimization workhorse.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()  # Nesterov lookahead
    
    no_improve = 0
    max_no_improve = 150
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive constraint selection
        # Start loose, tighten over time
        if t < 20:
            threshold = 0.4
        elif t < 100:
            threshold = 0.6
        elif t < 500:
            threshold = 0.75
        else:
            threshold = 0.85
            
        active = np.where(conv_y > M * threshold)[0]
        if len(active) < 20:
            active = np.where(conv_y > M * 0.3)[0]
        
        # Solve LP
        b, duals = lp_solve_best_response(y, M, solver, active)
        if b is None:
            b, duals = lp_solve_best_response(y, M, solver, None)
        if b is None or b.sum() < 1e-15:
            continue
        
        b_sum = b.sum()
        g = b * norm / b_sum
        
        # Verify dropped constraints (if we used screening)
        if active is not None and len(active) < size:
            check = fast_conv_cross(y, g)
            if check.max() > M * 1.001:
                b_full, _ = lp_solve_best_response(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    g = b_full * norm / b_full.sum()
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Extrapolated point
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        # Perturbation to escape local optima
        if no_improve == max_no_improve:
            rng_local = np.random.default_rng(t)
            n_pert = max(20, n // 15)
            idx_pert = rng_local.choice(n, size=n_pert, replace=False)
            y[idx_pert] *= (1.0 + 0.3 * rng_local.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0
        
        # Log
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f} no_improve={no_improve}")
    
    return best_a, best_score

def peak_targeted_cd(a0, time_limit, verbose=True):
    """
    Coordinate descent that specifically targets the peak autocorrelation values.
    For each peak, tries to reduce it by redistributing mass.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find peak positions
        peak_threshold = cur_max * 0.95
        peak_lags = np.where(conv >= peak_threshold)[0]
        
        pass_improved = False
        
        for k in peak_lags[:20]:  # Focus on top 20 peaks
            if time.monotonic() - t0 > time_limit:
                break
            
            # At lag k, conv[k] = sum_i a[i] * a[k-i] (valid indices)
            # To reduce conv[k], we want to reduce the products a[i]*a[k-i]
            # Strategy: find the pair (i, k-i) with the largest product and reduce one of them
            
            # Identify contributing pairs
            if k < n:
                # conv[k] = sum_{i=0}^{n-1-k} a[i]*a[i+k] (one direction)
                # But actually conv[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i]*a[k-i]
                i_lo = max(0, k - n + 1)
                i_hi = min(k, n - 1)
                pairs = [(i, k - i) for i in range(i_lo, i_hi + 1)]
            else:
                # k >= n: conv[k] = sum_{i=k-n+1}^{n-1} a[i]*a[k-i]
                i_lo = k - n + 1
                i_hi = n - 1
                pairs = [(i, k - i) for i in range(i_lo, i_hi + 1)]
            
            if not pairs:
                continue
            
            # Find the pair with the largest product
            max_prod = 0
            max_pair = None
            for (i, j) in pairs:
                prod = a[i] * a[j]
                if prod > max_prod:
                    max_prod = prod
                    max_pair = (i, j)
            
            if max_pair is None:
                continue
            
            i, j = max_pair
            ai, aj = a[i], a[j]
            
            # Try to reduce the product by moving mass from the larger to a random position
            if ai >= aj:
                src, dst_val = i, aj
            else:
                src, dst_val = j, ai
            
            # Try several target positions
            for trial in range(5):
                if trial == 0:
                    tgt = int(rng.integers(n))
                else:
                    tgt = int(rng.integers(n))
                
                if tgt == src:
                    continue
                
                # Try moving a fraction of a[src] to tgt
                for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    delta = frac * a[src]
                    if delta < norm * 1e-10:
                        continue
                    
                    a[src] -= delta
                    a[tgt] += delta
                    sc = fast_conv_self(a).max()
                    
                    if sc < cur_max - 1e-12:
                        cur_max = sc
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                        pass_improved = True
                        break
                    else:
                        a[src] += delta
                        a[tgt] -= delta
            
            if time.monotonic() - t0 > time_limit:
                break
        
        # Rescale
        s = a.sum()
        if s > 1e-15:
            a = a * norm / s
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                # Perturb
                n_pert = max(10, n // 25)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        # Rapid oscillation
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 20 * x)
        a = np.maximum(a, 0.01)
    elif choice == 6:
        # Chirp-like
        a = 0.5 + 0.4 * np.cos(2 * np.pi * (10 * x**2 + 5 * x))
        a = np.maximum(a, 0.01)
    elif choice == 7:
        # Multi-frequency
        a = 0.4 + 0.2 * np.cos(2 * np.pi * 5 * x) + 0.15 * np.cos(2 * np.pi * 13 * x) + 0.1 * np.cos(2 * np.pi * 30 * x)
        a = np.maximum(a, 0.02)
    elif choice == 8:
        # Random-like (Poisson)
        a = rng.poisson(1, size=n).astype(float) + 0.1
    elif choice == 9:
        # Bernoulli-like
        a = rng.random(n)
        a = (a > 0.5).astype(float) * 0.8 + 0.2 * (1 - (a > 0.5).astype(float))
    elif choice == 10:
        # Serrated with noise
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8 + 0.05 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
    else:
        # Pure noise
        a = rng.random(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
