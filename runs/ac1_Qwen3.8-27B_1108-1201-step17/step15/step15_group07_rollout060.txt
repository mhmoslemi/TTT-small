    a = np.asarray(a, dtype=np.float64)
    size = 2 * n - 1
    
    c = fast_autocorr(a)

    c_scaled = c / epsilon
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    p = exp_c / exp_c.sum()
    LSE = epsilon * (c_max + np.log(exp_c.sum()))
    
    obj = 2.0 * n * LSE / (s * s)
    
    # Gradient of LSE w.r.t. a_i
    # dLSE/da_i = sum_k p_k * dc_k/da_i
    # dc_k/da_i = 2*a_{k-i} (for 0<=k-i<n, k≠2i) + a_i (for k=2i)
    # Approximation: dLSE/da_i ≈ 2 * sum_k p_k * a_{k-i} [0<=k-i<n]
    #               = 2 * (p * a_rev)[n-1+i]
    
    a_rev = a[::-1]
    fft_len = 1 << (size - 1).bit_length()
    P = rfft(p, fft_len)
    AR = rfft(a_rev, fft_len)
    cross = irfft(P * AR, fft_len)[:size]
    
    dLSE = 2.0 * cross[n - 1:n - 1 + n].copy()
    
    # Diagonal correction: for each i where 2i < size, subtract p[2i]*a[i]
    diag_idx = np.arange(n) * 2
    valid_diag = diag_idx < size
    dLSE[valid_diag] -= p[diag_idx[valid_diag]] * a[valid_diag]
    
    # Full gradient
    # f = 2n * LSE / s^2
    # df/da_i = 2n/s^2 * dLSE/da_i - 4n*LSE/s^3

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def smooth_obj_and_grad(a, epsilon):
    n = len(a)
    size = 2 * n - 1
    c = fast_autocorr(a)
    s = a.sum()
    
    c_scaled = c / epsilon
    c_max_val = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max_val)
    exp_sum = exp_c.sum()
    p = exp_c / exp_sum
    LSE = epsilon * (c_max_val + np.log(exp_sum))
    
    obj = 2.0 * n * LSE / (s * s)
    
    a_rev = a[::-1]
    fft_len = 1 << (size - 1).bit_length()
    P = rfft(p, fft_len)
    AR = rfft(a_rev, fft_len)
    cross = irfft(P * AR, fft_len)[:size]
    
    dLSE = 2.0 * cross[n - 1:n]
    
    # Diagonal correction
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < size
    dLSE[valid] -= p[diag_idx[valid]] * a[valid]
    
    grad = (2.0 * n / (s * s)) * dLSE - (4.0 * n * LSE) / (s ** 3)
    
    return obj, grad

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_cross_corr(a, p):
    """Compute R[i] = sum_j a[j] * p[j+i] for i=0..n-1.
    a has length n, p has length 2n-1.
    Uses FFT: R = full_conv(p, a_rev)[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    # Full conv of p (len 2n-1) and a_rev (len n) has length 3n-2
    need = (2 * n - 1) + n - 1
    fft_len = 1 << (need - 1).bit_length()
    P = rfft(p, fft_len)
    AR = rfft(a_rev, fft_len)
    conv = irfft(P * AR, fft_len)
    return conv[n - 1: 2 * n - 1]


def smooth_obj_grad(a, epsilon):
    """Compute smoothed objective and gradient.
    f(a) = 2n * LSE_eps(conv(a,a)) / (sum(a))^2
    """
    n = len(a)
    size = 2 * n - 1
    c = fast_autocorr(a)
    s = a.sum()

    c_scaled = c / epsilon
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    exp_sum = exp_c.sum()
    p = exp_c / exp_sum
    LSE = epsilon * (c_max + np.log(exp_sum))

    obj = 2.0 * n * LSE / (s * s)

    # Gradient of LSE w.r.t. a_i:
    # dLSE/da_i = 2 * R[i] - a_i * p[2i] (if 2i < size)
    R = compute_cross_corr(a, p)
    dLSE = 2.0 * R.copy()

    # Diagonal correction
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < size
    dLSE[valid] -= p[diag_idx[valid]] * a[valid]

    # Full gradient of f = 2n*LSE/s^2
    grad = (2.0 * n / (s * s)) * dLSE - (4.0 * n * LSE) / (s ** 3)

    return obj, grad


def smooth_obj_grad_deflated(a, epsilon, peak_lag, mu):
    """Smoothed objective with deflation penalty on specific lag."""
    n = len(a)
    size = 2 * n - 1
    c = fast_autocorr(a)
    s = a.sum()

    # Modified "autocorrelation" with deflation
    c_mod = c.copy()
    # We add mu * c[peak_lag] as a penalty, equivalent to modifying p
    # Actually, let's just modify the objective directly:
    # f = 2n * (LSE(c) + mu * c[peak_lag]) / s^2
    # Gradient: 2n/s^2 * (dLSE/da + mu * dc[peak_lag]/da) - 4n*(LSE+mu*c[peak_lag])/s^3

    c_scaled = c / epsilon
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    exp_sum = exp_c.sum()
    p = exp_c / exp_sum
    LSE = epsilon * (c_max + np.log(exp_sum))

    obj = 2.0 * n * (LSE + mu * c[peak_lag]) / (s * s)

    R = compute_cross_corr(a, p)
    dLSE = 2.0 * R.copy()
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < size
    dLSE[valid] -= p[diag_idx[valid]] * a[valid]

    # Gradient of c[peak_lag] w.r.t. a_i:
    # dc_k/da_i = 2*a_{k-i} (if 0<=k-i<n) - a_i (if k=2i)
    k = peak_lag
    dc_da = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    if i_hi >= i_lo:
        j_lo = k - i_hi
        j_hi = k - i_lo
        dc_da[i_lo:i_hi + 1] = 2.0 * a[j_lo:j_hi + 1][::-1]
        # Diagonal correction
        if 2 * np.arange(i_lo, i_hi + 1) == k:
            diag_pos = np.arange(i_lo, i_hi + 1)[2 * np.arange(i_lo, i_hi + 1) == k]
            for dp in diag_pos:
                dc_da[dp] -= a[dp]

    total_obj_inner = LSE + mu * c[peak_lag]
    grad = (2.0 * n / (s * s)) * (dLSE + mu * dc_da) - (4.0 * n * total_obj_inner) / (s ** 3)

    return obj, grad


def lbfgs_optimize(a0, epsilon_schedule, time_limit, max_iter_per_eps=500):
    """Run L-BFGS with epsilon annealing."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a.copy()

    best_a = a.copy()
    best_score = evaluate_sequence([float(x) for x in a])
    t0 = time.monotonic()
    n = len(a)

    for eps in epsilon_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        def fun(x):
            o, _ = smooth_obj_grad(x, eps)
            return o

        def grad_fun(x):
            _, g = smooth_obj_grad(x, eps)
            return g

        try:
            res = minimize(fun, a, jac=grad_fun, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': max_iter_per_eps,
                                    'ftol': 1e-15, 'gtol': 1e-10,
                                    'maxls': 100, 'maxfun': max_iter_per_eps * 3})
            a_new = np.maximum(res.x, 0.0)
            s_new = a_new.sum()
            if s_new >= 0.01:
                a = a_new
                sc = evaluate_sequence([float(x) for x in a])
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        except Exception as e:
            pass

    return best_a, best_score

def targeted_cd(a0, time_limit, n_sample=500):
    """Targeted coordinate descent: try mass moves that reduce peak autocorrelation."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return a, float('inf')
    a = a.copy()

    best_a = a.copy()
    best_score = evaluate_sequence([float(x) for x in a])
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        c = fast_autocorr(a)
        max_val = c.max()
        peak_idx = c.argmax()

        # Identify coordinates that contribute to the peak
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(peak_idx, n - 1)
        contributing = list(range(i_lo, i_hi + 1))

        improved = False
        for _ in range(n_sample):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick source from contributing set, target from anywhere
            if contributing:
                i = contributing[rng.integers(0, len(contributing))]
            else:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-15:
                continue

            for frac in [0.05, 0.1, 0.2, 0.4, 0.7]:
                d = a[i] * frac
                if d < 1e-15:
                    continue
                a[i] -= d
                a[j] += d
                s_new = a.sum()
                if s_new < 0.01:
                    a[i] += d
                    a[j] -= d
                    continue
                sc = 2.0 * n * fast_autocorr(a).max() / (s_new * s_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
                else:
                    a[i] += d
                    a[j] -= d

        if not improved:
            no_improve += 1
            # Perturb to escape
            if no_improve % 20 == 0:
                noise = rng.normal(0, 0.02, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 0.01:
                    a = a_pert
                    no_improve = 0

    return best_a, best_score


def pair_cd_fine(a0, time_limit, window=30):
    """Fine pair coordinate descent with delta computation."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return a, float('inf')
    a = a.copy()

    best_a = a.copy()
    best_score = evaluate_sequence([float(x) for x in a])
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 10:
        c_curr = fast_autocorr(a)
        max_curr = c_curr.max()
        improved = False

        # Try moves from high-value to low-value positions
        order = np.argsort(a)
        # Source: high values, Target: low values
        sources = order[-n // 4:]
        targets = order[:n // 4]

        for i in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            for j in targets:
                if time.monotonic() - t0 > time_limit:
                    break
                if i == j:
                    continue
                if abs(i - j) > window and abs(i - j) < n - window:
                    continue  # skip far pairs for efficiency

                # Compute delta vector for moving d from i to j
                # delta[k] = change in c[k] per unit d moved from i to j
                delta = np.zeros(size)
                # Removing d from i: c[k] changes by -d*(a[k-i] + a[i-k]) for valid indices
                # Adding d to j: c[k] changes by +d*(a[k-j] + a[j-k]) for valid indices
                for k in range(max(0, i - n + 1), min(size, i + n)):
                    if 0 <= k - i < n:
                        delta[k] -= a[k - i]
                    if 0 <= i - k < n:
                        delta[k] -= a[i - k]
                for k in range(max(0, j - n + 1), min(size, j + n)):
                    if 0 <= k - j < n:
                        delta[k] += a[k - j]
                    if 0 <= j - k < n:
                        delta[k] += a[j - k]
                # Diagonal corrections
                if 2 * i < size:
                    delta[2 * i] += a[i]  # was -2a[i], now -a[i], so +a[i] correction
                    delta[2 * i] -= 0  # actually let's just use the 2x approximation
                if 2 * j < size:
                    delta[2 * j] -= a[j]

                # Try a few values of d
                best_d = 0
                best_new_max = max_curr
                d_max = a[i] * 0.8
                for d in np.linspace(0.01, d_max, 8):
                    new_c = c_curr + d * delta
                    new_c = np.maximum(new_c, 0)
                    nm = new_c.max()
                    if nm < best_new_max:
                        best_new_max = nm
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s_new = a.sum()
                    if s_new < 0.01:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    sc = 2.0 * n * fast_autocorr(a).max() / (s_new * s_new)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        c_curr = fast_autocorr(a)
                        max_curr = c_curr.max()

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return best_a, best_score


def gradient_check(a, epsilon, n_test=5):
    """Verify gradient correctness with finite differences."""
    obj, grad = smooth_obj_grad(a, epsilon)
    a_copy = a.copy()
    max_rel_err = 0
    for i in range(min(n_test, len(a))):
        h = 1e-6 * max(1.0, abs(a[i]))
        a_copy[i] += h
        obj_p, _ = smooth_obj_grad(a_copy, epsilon)
        a_copy[i] -= 2 * h
        obj_m, _ = smooth_obj_grad(a_copy, epsilon)
        a_copy[i] += h
        fd = (obj_p - obj_m) / (2 * h)
        rel_err = abs(fd - grad[i]) / max(1e-15, abs(fd))
        max_rel_err = max(max_rel_err, rel_err)
    return max_rel_err


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1600.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best_seq = p / p.sum()
            best_score = evaluate_sequence([float(x) for x in best_seq])
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n = 1600
        best_seq = np.ones(n) / n
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Default init: n={n}, score={best_score:.8f}")

    n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_sequence([float(x) for x in a])
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Verify gradient once
    print("Checking gradient...")
    a_test = best_seq.copy()
    err = gradient_check(a_test, 0.001, n_test=3)
    print(f"  Max relative gradient error: {err:.2e}")

    # Phase 1: L-BFGS with epsilon annealing (35% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: L-BFGS annealing (budget={t1:.0f}s) ===")
    if t1 > 60:
        # Epsilon schedule: start moderate, decrease
        # Typical autocorrelation values ~ 1/n to 1/sqrt(n)
        eps1 = 1.0 / n_base
        schedule1 = [eps1 * (10 ** k) for k in range(2, -4, -1)]  # 100x to 0.001x
        schedule1 = [max(e, 1e-8) for e in schedule1]
        print(f"  Epsilon schedule: {[f'{e:.2e}' for e in schedule1]}")
        a_out, sc = lbfgs_optimize(best_seq, schedule1, t1, max_iter_per_eps=300)
        try_update(a_out, "lbfgs1")

    # Phase 2: Deflated L-BFGS (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Deflated L-BFGS (budget={t2:.0f}s) ===")
    if t2 > 40:
        eps2 = 1.0 / n_base
        schedule2 = [eps2 * (10 ** k) for k in range(1, -4, -1)]
        schedule2 = [max(e, 1e-8) for e in schedule2]
        a_out, sc = lbfgs_deflated(best_seq, schedule2, t2, n_peaks=3, mu=0.3)
        try_update(a_out, "deflated")

    # Phase 3: Targeted coordinate descent (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: Targeted CD (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, sc = targeted_cd(best_seq, t3, n_sample=300)
        try_update(a_out, "tcd")

    # Phase 4: L-BFGS from perturbed starts (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Perturbed L-BFGS (budget={t4:.0f}s) ===")
    if t4 > 60:
        eps4 = 1.0 / n_base
        schedule4 = [eps4 * 10, eps4, eps4 * 0.1, eps4 * 0.01]
        n_restarts = 3
        t_per = t4 / n_restarts
        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget:
                break
            scale = 0.03 * (r + 1)
            noise = rng.laplace(0, scale, n_base)
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 0.01:
                a_pert = a_pert / s_pert
                t_sub = min(t_per, time_budget - (time.monotonic() - t_global))
                if t_sub > 30:
                    a_out, sc = lbfgs_optimize(a_pert, schedule4, t_sub, max_iter_per_eps=200)
                    try_update(a_out, f"pert{r}")

    # Phase 5: Fine pair CD (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Fine Pair CD (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, sc = pair_cd_fine(best_seq, t5, window=50)
        try_update(a_out, "pair_cd")

    # Final phase: last L-BFGS polish with tight epsilon
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final: L-BFGS polish (budget={remaining:.0f}s) ===")
        eps_final = 1.0 / n_base
        schedule_f = [eps_final * 0.1, eps_final * 0.01, eps_final * 0.001]
        schedule_f = [max(e, 1e-10) for e in schedule_f]
        a_out, sc = lbfgs_optimize(best_seq, schedule_f, remaining * 0.7, max_iter_per_eps=500)
        try_update(a_out, "final_lbfgs")

    # Absolute final check
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        a_out, sc = targeted_cd(best_seq, remaining * 0.8, n_sample=200)
        try_update(a_out, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.1f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
