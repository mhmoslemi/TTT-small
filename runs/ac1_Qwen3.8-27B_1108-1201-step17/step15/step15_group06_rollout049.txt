def fast_autocorr(a):

    fft_size = 1 << (size - 1).bit_length()

def build_T_matrix(a, indices):
    n = len(a)
    m = len(indices)
    T = np.zeros((m, n))
    for row, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def solve_minmax_lp(a, solver, active_idx, normalize_sum):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_T_matrix(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == normalize_sum]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, float('inf')
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, float('inf')
    return val, t_val


def solve_maxsum_lp(a, M, solver, active_idx):

    b = cp.Variable(n, nonneg=True)
    T = build_T_matrix(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def subgradient_step(a, c, n_active, eps=0.01):
    """Smoothed subgradient of max(auto-conv) using softmax weights."""
    n = len(a)
    size = 2 * n - 1
    # Softmax weights over the auto-convolution
    c_shifted = c - c.max()
    w = np.exp(c_shifted / eps)
    w /= w.sum()
    # Gradient: sum_k w[k] * 2*a[k-i] (valid where 0<=k-i<n)
    grad = np.zeros(n)
    for i in range(n):
        k_lo = max(0, i)
        k_hi = min(size - 1, i + n - 1)
        if k_lo <= k_hi:
            indices = np.arange(k_lo, k_hi + 1)
            grad[i] = 2.0 * np.dot(w[k_lo:k_hi+1], a[indices - i])
    return grad

def coord_descent_analytical(a0, time_limit, rng, verbose=True):
    """Coordinate descent using analytical gradient information."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv = fast_autocorr(a)
    best_peak = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * best_peak
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        conv = fast_autocorr(a)
        k_star = int(np.argmax(conv))

        # Get subgradient direction
        grad = np.zeros(n)
        for i in range(n):
            ki = k_star - i
            if 0 <= ki < n:
                grad[i] = 2.0 * a[ki]

        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            # Try small perturbations in the subgradient direction
            best_val = best_peak
            best_delta = 0.0
            scale = max(a[i], 0.01 / n)

            for frac in [-0.5, -0.3, -0.1, 0.1, 0.3, 0.5, 0.8]:
                delta = frac * scale
                if delta < 0 and delta < -a[i]:
                    continue
                a_try = a.copy()
                a_try[i] = max(0.0, a[i] + delta)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try / st
                cv = fast_autocorr(a_try)
                val = cv.max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_delta = delta

            if abs(best_delta) > 1e-14:
                a_new = a.copy()
                a_new[i] = max(0.0, a[i] + best_delta)
                sn = a_new.sum()
                a_new = a_new / sn
                pc = fast_autocorr(a_new).max()
                if pc < best_peak - 1e-14:
                    a = a_new
                    best_peak = pc
                    best_a = a.copy()
                    best_score = 2.0 * n * best_peak
                    pass_improved = True
                    no_improve = 0

        if not pass_improve:
            no_improve += 1
            if no_improve >= 8:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def smooth_subgradient_descent(a0, time_limit, solver, rng, verbose=True):
    """Combined smoothed subgradient + LP method."""
    n = len(a0)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    best_score = 2.0 * n * best_peak
    t0 = time.monotonic()
    eta = 0.05  # Learning rate for subgradient
    eps_smooth = 0.005  # Smoothing parameter

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M = conv.max()

        # Smoothing parameter annealing
        eps_curr = eps_smooth * (1.0 + 0.5 * np.exp(-t / 200.0))

        # Compute smoothed gradient
        c_shifted = conv - M
        w = np.exp(c_shifted / eps_curr)
        w /= w.sum()

        # Gradient computation using FFT trick
        # grad[i] = 2 * sum_k w[k] * a[k-i] (valid where 0<=k-i<n)
        # This is 2 * (w * a_reversed) evaluated appropriately
        # Actually, let's compute it directly using convolution
        # grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * (w * a)[i] where * is "cross-correlation-like"
        # (w * a)[i] = sum_k w[k] * a[i-k] ... no that's not right
        # Let's use: grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
        # = 2 * (w_conv_a)[i+n-1] where w is zero-padded

        # Simpler: use the fact that grad = 2 * ifft(FFT(w_full) * FFT(a_full))
        # Actually let me just do it with the convolution we already have
        # w_full has length 2n-1, a has length n
        # We need: for each i, sum_k w[k]*a[k-i] where 0<=k<n_ext and 0<=k-i<n
        # This is (w * rev_a)[i + n - 1] ... let me just use a direct FFT approach

        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * (w * a)_correlation[i]
        # correlation[i] = sum_k w[k+i] * a[k] = sum_k w[k] * a[k-i]
        # This is ifft(FFT(w) * conj(FFT(a))) ... no
        # sum_k w[k] * a[k-i] is (w rev-convolve a) or equivalently:
        # Let b = a[::-1], then sum_k w[k]*a[k-i] = sum_k w[k]*b[n-1-(k-i)] = (w*b)[n-1-i]... 
        # This is getting complicated. Let me just use a direct approach for the gradient.
        
        # Actually, the simplest correct formula:
        # (a*a)[k] = sum_m a[m]*a[k-m]
        # d(a*a)[k]/da[i] = 2*a[k-i] if 0<=k-i<n, else 0
        # So grad[i] = 2 * sum_k w[k] * a[k-i] * (0<=k-i<n)
        # = 2 * (w convolved with a, but only the valid parts)
        # Let's compute c_wa = convolve(w, a) which has length (2n-1)+(n-1) = 3n-2
        # c_wa[m] = sum_k w[k]*a[m-k]
        # We want: for each i, 2 * sum_k w[k]*a[k-i] = 2 * c_wa[i]... 
        # Wait: c_wa[i] = sum_k w[k]*a[i-k]. We want sum_k w[k]*a[k-i].
        # Let j = k-i, then sum_j w[j+i]*a[j] = (w_shifted * a) where w_shifted[j]=w[j+i]
        # This is (w * a) at position i in the "correlation" sense.
        # 
        # Actually, sum_k w[k]*a[k-i] = cross-correlation of w and a at lag i
        # = (w * a_flipped)[i + n - 1] where a_flipped = a[::-1]
        
        a_flipped = a[::-1]
        fft_size2 = 1 << (size + n - 2 - 1).bit_length()
        Wf = rfft(w, fft_size2)
        Af = rfft(a_flipped, fft_size2)
        corr = irfft(Wf * Af, fft_size2)
        grad = 2.0 * corr[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0.0)  # For non-negative sequences, gradient should be non-negative

        # Projected gradient step
        lr = eta / (1.0 + 0.1 * np.sqrt(t))
        a_new = a - lr * grad / max(grad.max(), 1e-15)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        # Occasionally use LP for a better direction
        if t % 5 == 0:
            tol = 0.05
            threshold = M * (1.0 - tol)
            active_set = np.where(conv >= threshold)[0]
            if len(active_set) < 10:
                active_set = np.argsort(conv)[-max(10, int(0.03 * size)):]
            max_active = min(size, 300)
            if len(active_set) > max_active:
                active_set = np.argsort(conv[active_set])[-max_active:]
            b_lp, _ = solve_minmax_lp(a, solver, active_set, normalize_sum=1.0)
            if b_lp is not None:
                # Blend subgradient step with LP solution
                blend = 0.5
                a_new = (1.0 - blend) * a_new + blend * b_lp
                a_new = np.maximum(a_new, 0.0)
                sn = a_new.sum()
                if sn > 1e-15:
                    a_new = a_new / sn

        a = a_new
        pk = fast_autocorr(a).max()
        if pk < best_peak:
            best_peak = pk
            best_a = a.copy()
            best_score = 2.0 * n * best_peak

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SSD t={t} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.3 + 0.5 * np.sin(np.pi * x) ** 0.7 + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.9 * np.exp(-6.0 * (x - 0.75) ** 2) + 0.3 * np.exp(-4.0 * (x - 0.5) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif style == 5:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.2 * np.cos(8 * np.pi * x))
        a = np.maximum(a, 0.0)
    elif style == 6:
        # Gaussian-like centered
        a = np.exp(-3.0 * ((x - 0.5) * 4.0) ** 2)
    elif style == 7:
        # Flat top with smooth edges
        a = np.ones(n)
        edge = int(0.2 * n)
        for i in range(edge):
            frac = i / edge
            a[i] = 0.3 + 0.7 * frac ** 2
            a[n - 1 - i] = 0.3 + 0.7 * frac ** 2
    elif style == 8:
        # Raised cosine
        a = 0.5 * (1.0 + np.cos(2 * np.pi * x)) ** 0.5
    else:
        a = np.sin(np.pi * x) ** 1.0 * (1.0 + 0.1 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def propose_candidate():
    rng = np.random.default_rng(2024)
    t_global_start = time.monotonic()
    time_budget = 1750.0

    solver = get_best_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, delta={old_score - sc:.8f})")

    def time_left():
        return time_budget - (time.monotonic() - t_global_start)

    # === Phase 0: Multi-start initialization ===
    print(f"\n=== Phase 0: Multi-start ===")
    if best_seq is None:
        n_init = 1600
        for sty in range(9):
            cand = structured_init(n_init, rng, sty)
            sc = evaluate_safe(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # Quick LP warmup on best initial
    if time_left() > 100:
        n_cur = len(best_seq)
        s_cur = best_seq.sum()
        a_norm = best_seq / s_cur if s_cur > 0 else np.ones(n_cur) / n_cur
        a_out, sc_out = nesterov_minmax_lp(a_norm, min(60.0, time_left() * 0.1), solver, verbose=False, seed=1)
        try_update(a_out, f"warmup_n{n_cur}")

    # === Phase 1: Main LP optimization (40% of remaining time) ===
    print(f"\n=== Phase 1: LP Nesterov n={len(best_seq)} ===")
    t1 = min(time_left() * 0.40, 500.0)
    if t1 > 60:
        n_cur = len(best_seq)
        a_norm = best_seq / best_seq.sum()
        a_out, _ = nesterov_minmax_lp(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"lp_main_n{n_cur}")

    # === Phase 2: Smoothed subgradient (15%) ===
    print(f"\n=== Phase 2: Smoothed Subgradient n={len(best_seq)} ===")
    t2 = min(time_left() * 0.15, 200.0)
    if t2 > 60:
        n_cur = len(best_seq)
        a_norm = best_seq / best_seq.sum()
        a_out, _ = smooth_subgradient_descent(a_norm, t2, solver, rng, verbose=True)
        try_update(a_out, f"ssd_n{n_cur}")

    # === Phase 3: Coordinate descent polish (10%) ===
    print(f"\n=== Phase 3: Coord Descent n={len(best_seq)} ===")
    t3 = min(time_left() * 0.10, 150.0)
    if t3 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq / best_seq.sum()
        a_out, _ = coord_descent_analytical(a_norm, t3, rng, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 4: Perturbation + LP restarts (20%) ===
    print(f"\n=== Phase 4: Perturbation restarts ===")
    t4_total = min(time_left() * 0.20, 300.0)
    n_restarts = 4
    t4_each = t4_total / n_restarts
    for r in range(n_restarts):
        if time_left() < 80:
            break
        t4 = min(t4_each, time_left() - 50)
        if t4 < 30:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.03 * r
        pert = perturb_seq(best_seq, rng, scale)
        a_out, _ = nesterov_minmax_lp(pert, t4, solver, verbose=False, seed=100 + r * 13)
        try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 5: Multi-length exploration (10%) ===
    print(f"\n=== Phase 5: Multi-length ===")
    t5_total = min(time_left() * 0.10, 150.0)
    n_base = len(best_seq)
    lengths_to_try = [n_base + 300, n_base + 600, max(1000, n_base - 300), n_base + 1000]
    t5_each = t5_total / max(1, len([l for l in lengths_to_try if time_left() > 40]))
    for nl in lengths_to_try:
        if time_left() < 50:
            break
        t5 = min(t5_each, time_left() - 30)
        if t5 < 25:
            break
        if nl <= 0:
            continue
        print(f"  Trying n={nl}...")
        init_long = interpolate_seq(best_seq, nl)
        a_out, _ = nesterov_minmax_lp(init_long, t5, solver, verbose=False, seed=42)
        try_update(a_out, f"len{nl}")

    # === Phase 6: Final LP on best (remaining time) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP n={n_best} ===")
    t6 = max(time_left() * 0.5, 60.0)
    t6 = min(t6, time_left() - 20)
    if t6 > 30:
        a_norm = best_seq / best_seq.sum()
        a_out, _ = nesterov_minmax_lp(a_norm, t6, solver, verbose=True, seed=999)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final CD polish ===
    n_best = len(best_seq)
    if time_left() > 40:
        print(f"\n=== Phase 7: Final CD n={n_best} ===")
        t7 = min(time_left() * 0.6, 100.0)
        if t7 > 20:
            a_norm = best_seq / best_seq.sum()
            a_out, _ = coord_descent_analytical(a_norm, t7, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global_start:.0f}s")
        return [float(x) for x in best_seq]
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
