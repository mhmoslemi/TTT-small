"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based conditional gradients
(Frank-Wolfe) with constraint generation, adaptive step sizes, symmetry exploitation,
and coordinate descent polishing. Explores multiple sequence lengths and restarts.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def get_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    conv = fast_conv_self(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (S * S))

def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S

def build_T_values_full(a, rows, cols, n, size):
    vals = np.empty(len(rows), dtype=np.float64)
    for idx in range(len(rows)):
        k = rows[idx]
        i = cols[idx]
        vals[idx] = a[i]
    return vals

def lp_solve(a, M, active_k, n, size):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for k in active_k."""
    nn = len(active_k)
    if nn == 0:
        return None
    
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_k):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[i])
    
    if len(rows) == 0:
        return np.ones(n) / n
    
    T = cp.SparseMatrix((np.array(vals, dtype=np.float64), (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))), (nn, n))
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, eps=1e-8, max_iters=10000)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    val = np.maximum(val, 0.0)
    return val

    n = len(a0)

        M = conv.max()

        if solve_full:

        else:
            # Select top active constraints
            sorted_idx = np.argsort(conv)[::-1]
            active_k = sorted_idx[:max_active].tolist()

        if b is None:
            # Fallback: try with fewer constraints
            if len(active_k) > 50:
                b = lp_solve(a, M, active_k[:100], n, size)
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:

            # Actually just check conv(a, b) directly
            check_conv = fast_conv_cross(a, b)
            violated = np.where(check_conv > M * 1.001)[0]
            if len(violated) > 0:
                new_active = list(set(active_k + violated.tolist()))
                if len(new_active) > max_active:
                    # Keep top max_active
                    conv_vals = np.array([check_conv[k] for k in new_active])
                    top_idx = np.argsort(conv_vals)[::-1][:max_active]

def coord_descent_polish(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent to polish the solution."""
    n = len(a0)
    a = normalize(np.maximum(np.asarray(a0, dtype=np.float64), 0.0))
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        # Find peak locations
        peak_thresh = cur_max * 0.9
        peak_lags = np.where(conv >= peak_thresh)[0]
        
        # For each peak, try to reduce it by adjusting pairs
        improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue
            
            # Candidate j: locations that would reduce peak
            j_candidates = set()
            for kl in peak_lags[:15]:
                j1 = int(kl - i)
                j2 = int(kl + i)
                if 0 <= j1 < n and j1 != i:
                    j_candidates.add(j1)
                if 0 <= j2 < n and j2 != i:
                    j_candidates.add(j2)
            # Also random
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_j = -1
            best_new_max = cur_max
            
            for j in list(j_candidates)[:12]:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute change in convolution when moving delta from i to j
                # conv_new = conv + delta*(contribution from j - contribution from i)
                delta_max = ai * 0.9
                n_samples = 15
                deltas = np.linspace(0.02, delta_max, n_samples)
                
                for d in deltas:
                    # Approximate new peak
                    new_conv = conv.copy()
                    # Subtract contribution of a[i]
                    for k in range(max(0, i - n + 1), min(size, i + n)):
                        pass  # Too slow, use FFT
                    
                    # Use FFT for exact computation
                    a_test = a.copy()
                    a_test[i] = ai - d
                    a_test[j] = aj + d
                    new_conv = fast_conv_self(a_test)
                    v = new_conv.max()
                    if v < best_new_max:
                        best_new_max = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] = ai - best_delta
                a[best_j] += best_delta
                a = normalize(a)
                new_max = fast_conv_self(a).max()
                if new_max < cur_max * (1.0 - 1e-8):
                    improved = True
                    sc = get_score(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                else:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                    a = normalize(a)
        
        if not improved:
            no_improve += 1
            # Perturb to escape
            n_pert = max(3, n // 100)
            idx_p = rng.choice(n, size=n_pert, replace=False)
            a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
            a = normalize(a)
            if no_improve >= 8:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    return normalize(f_interp)

            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)
            a = a_new
            
            sc = get_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    
    print(f"Starting optimization. Budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p.copy())
            best_score = get_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = get_score(best_seq)
        print(f"Random init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 0.01:
            return
        a = normalize(a)
        sc = get_score(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old:.6f}, n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 1: Quick LSE-Adam to improve initial point
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 1: LSE-Adam init (t={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam_quick(best_seq, t1)
        check_update(a_out, "lse_init")
    
    # Phase 2: Main LP-FW optimization
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 2: LP-FW main (t={t2:.0f}s, n={len(best_seq)}) ===")
    if t2 > 30:
        # Start with moderate active set, grow it
        a_out, _ = fw_lp_optimize(best_seq, t2, max_active=200, verbose=True)
        check_update(a_out, "lpfw_200")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t2b = min(remaining * 0.4, 300.0)
            print(f"  Continuing LP-FW with more constraints (t={t2b:.0f}s)...")
            a_out, _ = fw_lp_optimize(best_seq, t2b, max_active=500, verbose=True)
            check_update(a_out, "lpfw_500")
    
    # Phase 3: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.2, 250.0)
    print(f"\n=== Phase 3: Coord descent polish (t={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = coord_descent_polish(best_seq, t3, rng)
        check_update(a_out, "cd_polish")
    
    # Phase 4: Second LP-FW round from polished point
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.2, 200.0)
    print(f"\n=== Phase 4: LP-FW round 2 (t={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = fw_lp_optimize(best_seq, t4, max_active=300, verbose=False)
        check_update(a_out, "lpfw_round2")
    
    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        n_values = [n_base + 100, n_base - 100, n_base + 300, n_base + 200]
        for n_try in n_values:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            if n_try < 200 or n_try > 4000:
                continue
            t_try = min(remaining * 0.4, 40.0)
            print(f"  Trying n={n_try} (t={t_try:.0f}s)...")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam_quick(init_try, min(t_try, 15.0))
            check_update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 25:
                a_out, _ = fw_lp_optimize(a_out, min(remaining * 0.5, 20.0), max_active=150, verbose=False)
                check_update(a_out, f"ml_lp_n{n_try}")
    
    # Phase 6: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 6: Final polish (t={remaining:.0f}s) ===")
        t_pol = min(remaining * 0.6, 60.0)
        a_out, _ = coord_descent_polish(best_seq, t_pol, rng, verbose=False)
        check_update(a_out, "final_cd")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = fw_lp_optimize(best_seq, min(remaining * 0.7, 40.0), max_active=400, verbose=False)
            check_update(a_out, "final_lp")
    
    # Final evaluation
    if best_seq is not None:
        final_score = get_score(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
