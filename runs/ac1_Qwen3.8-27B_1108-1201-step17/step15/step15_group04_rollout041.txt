def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    from scipy.fft import rfft, irfft
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    from scipy.fft import rfft, irfft
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def solve_dual_lp(a, M):
    """
    Solve the dual LP in O(n) time.
    Dual: min M * sum(lam) s.t. a[j] * window_sum(lam, j, n) >= 1 for all j, lam >= 0.
    
    Returns (lam, dual_value) where dual_value = M * sum(lam).
    """
    n = len(a)
    size = 2 * n - 1
    
    # P[k] = prefix sum of lam up to position k (P[-1] = 0)
    # P[k] = max(P[k-1], P[k-n] + 1/a[k-n+1]) for k >= n-1
    # P[k] = P[k-1] for k < n-1 (but P[n-1] >= 1/a[0])
    
    # Using 0-indexed: P_pos[i] for position i (i = -1, 0, 1, ..., 2n-2)
    # P_pos[-1] = 0
    # P_pos[k] = max(P_pos[k-1], ...)
    
    P = np.zeros(size + 1)  # P[0] = P_pos[-1] = 0, P[i+1] = P_pos[i]
    
    for k in range(size):  # k = 0, 1, ..., 2n-2
        # P_pos[k] = P[k+1]
        # Monotonicity: P_pos[k] >= P_pos[k-1], i.e., P[k+1] >= P[k]
        P[k+1] = P[k]
        
        if k == n - 1:
            # Constraint j=0: P_pos[n-1] >= 1/a[0]
            P[k+1] = max(P[k+1], 1.0 / max(a[0], 1e-30))
        elif k >= n:
            # Constraint j = k - n + 1: P_pos[k] >= P_pos[j-1] + 1/a[j]
            j = k - n + 1
            if j < n:
                # P_pos[j-1] = P[j]
                P[k+1] = max(P[k+1], P[j] + 1.0 / max(a[j], 1e-30))
    
    # sum(lam) = P_pos[2n-2] = P[size] = P[2n-1]
    sum_lam = P[size]
    
    # Recover lam
    lam = np.diff(P)  # lam[i] = P[i+1] - P[i], which is lam at position i
    # lam has length size = 2n-1, corresponding to positions 0, 1, ..., 2n-2
    # Wait, P has length size+1 = 2n, so np.diff gives length 2n-1 = size. ✓
    
    return lam, M * sum_lam

def approximate_primal_from_dual(a, lam, M):
    """
    Given the dual solution lam, construct an approximate primal solution b.
    b[j] = 1 / (T^T lam)[j], normalized.
    """
    n = len(a)
    size = 2 * n - 1
    
    # (T^T lam)[j] = a[j] * window_sum(lam, j, n)
    # window_sum(lam, j, n) = sum_{k=max(0,j)}^{min(j+n-1, 2n-2)} lam[k]
    
    P = np.zeros(size + 1)
    P[1:] = np.cumsum(lam)
    
    # window_sum for each j
    window_sums = np.zeros(n)
    for j in range(n):
        lo = max(0, j)
        hi = min(j + n, size)  # exclusive
        window_sums[j] = P[hi] - P[lo]
    
    # (T^T lam)[j] = a[j] * window_sums[j]
    Tt_lam = a * window_sums
    
    # b[j] = 1 / Tt_lam[j]
    Tt_lam_safe = np.maximum(Tt_lam, 1e-30)
    b = 1.0 / Tt_lam_safe
    b_sum = b.sum()
    if b_sum < 1e-30:
        return np.ones(n) / n
    b = b / b_sum
    
    return b

def nesterov_lp_fast(a0, time_limit, verbose=True, use_approx=True):
    """
    Nesterov LP iteration using the O(n) dual solver.
    Much faster than the original LP-based approach.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_maxc = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov auxiliary variable
    no_improve = 0
    momentum = 0.0
    prev_y = None
    prev_a = None
    iter_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute M = max(y*y)
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-15:
            break
        
        # Solve dual LP in O(n)
        lam, dual_val = solve_dual_lp(y, M)
        
        # Get approximate primal
        if use_approx:
            b = approximate_primal_from_dual(y, lam, M)
        else:
            b = np.ones(n) / n  # fallback
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) / n
            b_sum = 1.0
        g = b * norm / b_sum
        
        # Nesterov update with adaptive momentum
        if t == 1:
            y_new = g.copy()
            a_new = g.copy()
        else:
            # Heavy-ball momentum
            momentum = min(0.95, 0.9 + 0.05 * t / (t + 100))
            y_new = (1.0 - momentum) * y + momentum * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            
            # Nesterov lookahead
            if prev_y is not None:
                beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
                a_new = y_new + (momentum / beta) * (y_new - prev_y)
            else:
                a_new = y_new.copy()
            
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        iter_count += 1
        
        # Check improvement
        sc_y = fast_conv_self(y).max()
        if sc_y < best_maxc - 1e-12:
            best_maxc = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_maxc - 1e-12:
            best_maxc = sc_a
            best_a = a.copy()
            no_improve = 0
        
        # Restart if stalled
        if no_improve == 200:
            y = best_a.copy()
            prev_y = None
            momentum = 0.0
            no_improve = 0
        
        # Periodic perturbation
        if no_improve == 500:
            rng = np.random.default_rng(t)
            n_perturb = max(10, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            y[idx] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            prev_y = None
            no_improve = 0
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_maxc / (norm * norm)
            print(f"  NesterovFast it={t} ({elapsed:.0f}s): maxc={best_maxc:.8f}, score={score:.6f}")
    
    return best_a, best_maxc

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. O(n) dual LP solver for ultra-fast best-response in Nesterov iteration
2. FISTA with Log-Sum-Exp approximation and adaptive temperature
3. Subgradient method with Polyak averaging
4. Informed pair coordinate descent
5. Multi-start with diverse initializations and length extension
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def solve_dual_lp(a, M):
    """Solve dual LP in O(n): min M*sum(lam) s.t. a[j]*window_sum(lam,j,n)>=1, lam>=0."""
    n = len(a)
    size = 2 * n - 1
    P = np.zeros(size + 1)
    for k in range(size):
        P[k + 1] = P[k]
        if k == n - 1:
            val = 1.0 / max(a[0], 1e-30)
            if val > P[k + 1]:
                P[k + 1] = val
        elif k >= n:
            j = k - n + 1
            if j < n:
                val = P[j] + 1.0 / max(a[j], 1e-30)
                if val > P[k + 1]:
                    P[k + 1] = val
    sum_lam = P[size]
    lam = np.diff(P)
    return lam, M * sum_lam


def get_approx_primal(a, lam):
    """Construct approximate primal from dual solution."""
    n = len(a)
    size = 2 * n - 1
    P = np.zeros(size + 1)
    P[1:] = np.cumsum(lam)
    Tt_lam = np.zeros(n)
    for j in range(n):
        hi = min(j + n, size)
        ws = P[hi] - P[j]
        Tt_lam[j] = a[j] * ws
    Tt_lam_safe = np.maximum(Tt_lam, 1e-30)
    b = 1.0 / Tt_lam_safe
    b_sum = b.sum()
    if b_sum < 1e-30:
        return np.ones(n) / n
    return b / b_sum


def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def nesterov_fast(a0, time_limit, verbose=True):
    """Nesterov iteration with O(n) dual LP solver."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_maxc = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    prev_y = a.copy()
    no_improve = 0
    iter_count = 0
    rng = np.random.default_rng(123)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-15:
            break
        lam, _ = solve_dual_lp(y, M)
        b = get_approx_primal(y, lam)
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) / n
            b_sum = 1.0
        g = b * norm / b_sum
        if t == 1:
            y_new = g.copy()
            a_new = g.copy()
        else:
            momentum = min(0.9, 0.5 + 0.4 * t / (t + 50))
            y_new = (1.0 - momentum) * y + momentum * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
            a_new = y_new + (momentum / beta) * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        prev_y = y.copy()
        y = y_new
        a = a_new
        iter_count += 1
        sc_y = fast_conv_self(y).max()
        if sc_y < best_maxc - 1e-12:
            best_maxc = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        sc_a = fast_conv_self(a).max()
        if sc_a < best_maxc - 1e-12:
            best_maxc = sc_a
            best_a = a.copy()
            no_improve = 0
        if no_improve == 300:
            y = best_a.copy()
            prev_y = best_a.copy()
            no_improve = 0
        if no_improve == 800:
            n_pert = max(10, n // 15)
            idx = rng.choice(n, size=n_pert, replace=False)
            y[idx] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            prev_y = y.copy()
            no_improve = 0
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_maxc / (norm * norm)
            print(f"    NF it={t} ({elapsed:.0f}s): score={score:.6f}")
    return best_a, best_maxc


def lse_fista(a0, time_limit, verbose=True):
    """FISTA with Log-Sum-Exp smooth approximation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_maxc = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(80.0, 0.3, n_stages)
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        y = a.copy()
        k_fista = 1.0
        prev_y = a.copy()
        n_steps = max(30, int(remaining * 2.5))
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(y)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            Y = rfft(y, fft_size)
            conv_part = irfft(Y * W, fft_size)[:n]
            W_rev = rfft(np.flip(w), fft_size)
            cross_part = irfft(Y * W_rev, fft_size)[:n]
            grad = 2.0 * (conv_part + cross_part)
            mu = 1.0 / T_val
            step = 1.0 / (mu * np.max(grad) + 1e-15)
            step = min(step, norm / (np.sqrt(n) * 2.0))
            y_new = y - step * grad
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k_fista * k_fista))
            z = y_new + ((k_fista - 1.0) / k_new) * (y_new - prev_y)
            z = np.maximum(z, 0.0)
            s_z = z.sum()
            if s_z < 1e-15:
                z = y_new.copy()
            else:
                z = z * norm / s_z
            prev_y = y.copy()
            y = z
            k_fista = k_new
            a = y_new
            sc = fast_conv_self(a).max()
            if sc < best_maxc - 1e-12:
                best_maxc = sc
                best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_maxc / (norm * norm)
            print(f"    FISTA stage {si} T={T_val:.1f} ({elapsed:.0f}s): score={score:.6f}")
    return best_a, best_maxc


def subgradient_polyak(a0, time_limit, verbose=True):
    """Subgradient method with Polyak step size and averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_maxc = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    avg_a = np.zeros(n)
    n_avg = 0
    f_star_est = best_maxc * 0.999
    no_improve = 0
    rng = np.random.default_rng(456)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-15:
            break
        peak_idx = np.where(conv >= M * 0.95)[0]
        indicator = np.zeros(size)
        indicator[peak_idx] = 1.0
        indicator = indicator / indicator.sum()
        IND = rfft(indicator, fft_size)
        A = rfft(a, fft_size)
        conv_part = irfft(A * IND, fft_size)[:n]
        IND_rev = rfft(np.flip(indicator), fft_size)
        cross_part = irfft(A * IND_rev, fft_size)[:n]
        grad = 2.0 * (conv_part + cross_part)
        gap = M - f_star_est
        if gap < 1e-12:
            gap = 1e-12
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-30:
            break
        step = gap / g_norm_sq
        step = min(step, norm / np.sqrt(n) / 5.0)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_maxc - 1e-12:
            best_maxc = sc
            best_a = a.copy()
            no_improve = 0
            f_star_est = min(f_star_est, sc * 0.999)
        else:
            no_improve += 1
        avg_a += a
        n_avg += 1
        if n_avg % 5000 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_maxc:
                    best_maxc = sc_avg
                    best_a = avg_check.copy()
        if no_improve == 500:
            a = best_a.copy()
            f_star_est = best_maxc * 0.999
            no_improve = 0
        if no_improve == 1500:
            n_pert = max(10, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
            no_improve = 0
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_maxc / (norm * norm)
            print(f"    SubGrad t={t} ({elapsed:.0f}s): score={score:.6f}")
    return best_a, best_maxc


def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with peak-informed candidate selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_maxc = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_passes = 0
    total_imps = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        peak_lags = np.where(conv >= cur_max * 0.85)[0]
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:10]:
                j1 = int(kl - i)
                j2 = int(kl + i)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            j_candidates.discard(i)
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                n_samples = 15
                deltas = np.linspace(0.05, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d
                if best_d > 0 and best_v < best_val - 1e-13:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_maxc - 1e-13:
                        best_maxc = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_imps += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 6:
                break
            if no_improve_passes == 3:
                n_pert = max(10, n // 25)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve_passes = 0
        if verbose and total_imps % 500 == 0 and total_imps > 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_maxc / (norm * norm)
            print(f"    PCD imps={total_imps} ({elapsed:.0f}s): score={score:.6f}")
    return best_a, best_maxc


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x) + 0.1 * np.cos(2 * np.pi * 13 * x)
        a = np.maximum(a, 0.05)
    elif choice == 7:
        a = np.abs(np.sin(4 * np.pi * x)) ** 0.6
    elif choice == 8:
        a = np.exp(-0.5 * ((x - 0.5) / 0.15) ** 2) + 0.3 * np.ones(n)
    elif choice == 9:
        a = 0.4 + 0.3 * np.sin(2 * np.pi * 2 * x + 0.5) + 0.2 * np.sin(2 * np.pi * 11 * x)
        a = np.maximum(a, 0.03)
    elif choice == 10:
        a = np.ones(n) * (0.5 + 0.5 * np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.01)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
