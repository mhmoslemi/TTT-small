def fast_conv_self(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):

    n = len(a)

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_grad_penalty(y, b, M, fft_size_large):

    for t in range(1, 100000):

        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        # Find all peaks (for smoothing)
        # Subgradient: for the argmax lag k*, grad_i = a[(i+k*)] + a[(i-k*)] (cyclic-ish)
        # For full convolution: (a*a)_k = sum_i a[i]*a[i+k] (valid range)
        # d/db_i (a*a)_k* = a[k*-i] + a[i+k*] (with boundary handling)
        
        k_star = int(np.argmax(conv))
        
        # Compute subgradient
        grad = np.zeros(n)
        # (a*a)_k* = sum_{i: 0<=i<n, 0<=i+k*<2n} a[i] * a_{i+k*}
        # Actually for full conv: (a*a)_k = sum_{i=max(0,k-(n-1))}^{min(n-1,k)} a[i] * a[k-i]
        # d/d a[j] (a*a)_k = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, j≠k-j case)
        # More carefully: (a*a)_k = sum_i a[i]*a[k-i]
        # d/d a[j] = a[k-j] + a[j-k] if j ≠ k-j, else a[k-j] (when j = k/2)
        
        # For 0 <= j < n:
        # Term 1: a[k-j] contributes if 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)
        # Term 2: a[j-k] contributes if 0 <= j-k < n, i.e., k <= j <= min(n-1, k+n-1) = k+n-1
        #         but j < n, so k <= j < n, i.e., j in [k, n-1] if k < n
        
        # So:
        # For j in [0, min(n-1, k)]: grad[j] += a[k-j]
        # For j in [k, n-1] (if k < n): grad[j] += a[j-k]
        # If j = k and k is integer: both terms give a[0], so grad[k] += 2*a[0] if k < n
        # Wait, that's only if k < n. If k >= n, then j-k >= 0 means j >= k >= n, impossible.

            # Term 1: j in [0, k_star], a[k_star - j]
            for j in range(k_star + 1):
                grad[j] += a[k_star - j]
            # Term 2: j in [k_star, n-1], a[j - k_star]
            for j in range(k_star, n):
                grad[j] += a[j - k_star]
            # At j = k_star: a[0] is added twice (once from each term)
            # But that's correct: d/d a[k_star] sum_i a[i]*a[k_star-i] 
            # = sum_i [delta(i,k_star)*a[k_star-i] + a[i]*delta(k_star-i, k_star)]
            # = a[0] + a[0] = 2*a[0]
            # So the above is correct.
        else:
            # k_star >= n: only term 1 is valid
            # j in [k_star - n + 1, n-1], a[k_star - j]
            for j in range(k_star - n + 1, n):
                grad[j] += a[k_star - j]
            # Term 2: j-k_star >= 0 means j >= k_star >= n, impossible for j < n
            # So no term 2
        
        # Normalize gradient (it's a subgradient of the max, not the LSE)
        # Frank-Wolfe: find min of <grad, v> over v in {v >= 0, sum(v) = S}
        # = S * min(grad) (achieved at v = S * e_{argmin grad})
        
        s_min = grad.min()
        i_min = int(np.argmin(grad))
        
        # FW step size
        t_fw = 2.0 / (t + 2.0)
        
        # Update
        v = np.zeros(n)
        v[i_min] = S
        a_new = (1.0 - t_fw) * a + t_fw * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * S / s_new
        a = a_new

            elapsed = time.monotonic() - t0

def lse_nesterov(a0, time_limit, verbose=True):
    """LSE gradient descent with Nesterov acceleration."""
    n = len(a0)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()

    a = a * S / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Nesterov state
    y = a.copy()
    
    temps = np.geomspace(50.0, 0.5, 15)
    it = 0
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        n_steps = max(20, min(200, int(remaining * 5)))
        
        for _ in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            
            # Compute smoothed objective gradient
            conv_y = fast_conv_self(y)
            max_c = conv_y.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv_y - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            w = np.exp(shifted)
            w = w / w.sum()
            
            # Gradient of sum w[k] * (y*y)_k
            W = rfft(w, fft_size)
            Y = rfft(y, fft_size)
            g1 = irfft(np.conj(Y) * W, fft_size)[:n]
            g2 = irfft(Y * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            # Nesterov step
            lr = S / (T_val * np.sqrt(n) * (1.0 + 0.01 * it % 1000))
            alpha = 2.0 / (it + 2.0)
            
            y_new = y - lr * grad
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new * S / s_y
            
            mom = (it - 1.0) / (it + 2.0) if it > 1 else 0.0
            a_new = y_new + mom * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * S / s_a
            
            y = y_new
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")
    
    return best_a, best_score

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    size = 2 * n - 1

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

    """Build the matrix T where (T @ b)[k] = (conv(a,b))[k]"""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

    """

    else:
        T = build_conv_matrix(a)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)

def interpolate_seq(f, new_n):

    if old_n == new_n:
        return f.copy()

    old_x = np.linspace(0.0, 1.0, old_n)

    f_interp = np.interp(new_x, old_x, f_norm)
