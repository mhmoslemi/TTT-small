import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print(f"  LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to_sum(a, target_sum):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_relaxed(a, M, solver, slack=1.05):
    """LP with slightly relaxed constraint to allow more flexible solutions."""
    return lp_best_response(a, M * slack, solver)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.asarray(a0, dtype=np.float64), norm)
    y = a.copy()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    n_iters = 0
    momentum_t = 1.0
    patience = 0
    restart_count = 0
    stuck_count = 0
    damping = 1.0
    osc_counter = 0
    prev_score = best_score
    score_history = []

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 50:
                # Try relaxed version
                b = lp_best_response(y, M_curr * 1.1, solver)
                if b is None:
                    break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update with adaptive damping
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new * norm / max(y_new.sum(), 1e-15)

        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        # Oscillation detection
        score_y = fast_autocorr(y_new).max()
        score_a = fast_autocorr(a_new).max()
        cur_score = min(score_y, score_a)

        score_history.append(cur_score)
        if len(score_history) > 20:
            score_history.pop(0)

        if len(score_history) >= 10:
            recent = score_history[-10:]
            oscillation = sum(1 for i in range(1, len(recent)) if (recent[i] - recent[i-1]) * (recent[i-1] - recent[i-2] if i >= 2 else 1) < 0)
            if oscillation >= 5:
                osc_counter += 1
                if osc_counter > 3:
                    damping = max(0.3, damping * 0.7)
                    momentum_t = max(1.0, momentum_t * 0.5)
                    osc_counter = 0
            else:
                osc_counter = 0
                damping = min(1.0, damping * 1.1)
                momentum_t += 1.0
        else:
            momentum_t += 1.0

        # Apply damping to the update
        y = y + damping * (y_new - y)
        y = np.maximum(y, 0.0)
        y = y * norm / max(y.sum(), 1e-15)

        a_candidate = a_new if score_a <= score_y else y_new
        a = a + damping * (a_candidate - a)
        a = np.maximum(a, 0.0)
        a = a * norm / max(a.sum(), 1e-15)

        # Track best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy() if score_a <= score_y else y.copy()
            patience = 0
        else:
            patience += 1

        # Restart logic based on patience and score trajectory
        if patience > 60 and n_iters % 30 == 0:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 0
            damping = 1.0
            restart_count += 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart {restart_count} at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        # Strong restart: if we've made no progress in 200 iterations
        if patience > 200:
            y = best_a.copy()
            a = best_a.copy()
            momentum_t = 1.0
            patience = 100
            damping = 0.5
            restart_count += 1

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}, damp={damping:.2f}")

    return best_a, best_score


def smooth_max_optimize(a0, time_limit, verbose=True):
    """Optimize using logsumexp smooth approximation of max."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    tau = norm * 0.05
    eta = 1.0
    patience = 0
    rng = np.random.default_rng(7777)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()

        # Compute smooth max gradient
        scaled = conv / tau
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()

        # Gradient of smooth max
        grad = np.zeros(n)
        for k in range(size):
            if weights[k] < 1e-10:
                continue
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_hi >= i_lo:
                j_lo = k - i_hi
                j_hi = k - i_lo
                grad[i_lo:i_hi + 1] += weights[k] * a[j_lo:j_hi + 1][::-1]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Adaptive step size
        best_a_try = a.copy()
        best_obj = max_val
        for factor in [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            eta_try = eta * factor
            a_try = a - eta_try * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            obj_try = fast_autocorr(a_try).max()
            if obj_try < best_obj:
                best_obj = obj_try
                best_a_try = a_try
                eta = eta_try

        a = best_a_try

        # Adaptive tau
        if it % 50 == 0 and it > 0:
            tau *= 0.95

        if best_obj < best_score - 1e-12:
            best_score = best_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 200 == 0:
                eta *= 0.8

        if patience > 500:
            noise = rng.normal(0, norm * 0.003, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert * norm / s_pert
            patience = 0
            eta = 1.0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SM iter={it} ({elapsed:.0f}s): best={best_score:.8f}, tau={tau:.4f}, eta={eta:.4f}")

    return best_a, best_score


def active_set_polish(a0, time_limit, verbose=True):
    """Final polish using exact line search on the dominant active constraint."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(31415)

    for outer in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()
        # Find active indices (within 1% of max)
        threshold = max_val * 0.98
        active_mask = conv >= threshold
        active_idx = np.where(active_mask)[0]

        if len(active_idx) == 0:
            break

        # Gradient of max active constraint (averaged)
        grad = np.zeros(n)
        for k in active_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_hi >= i_lo:
                j_lo = k - i_hi
                j_hi = k - i_lo
                grad[i_lo:i_hi + 1] += a[j_lo:j_hi + 1][::-1]
        grad /= len(active_idx)

        # Negative gradient descent with exact line search
        direction = -grad
        # Project direction onto simplex tangent (sum = 0)
        direction = direction - direction.mean()

        if np.linalg.norm(direction) < 1e-15:
            break

        direction = direction / np.linalg.norm(direction)

        # Line search: find best alpha
        best_alpha = 0.0
        best_val = max_val
        alphas = np.logspace(-6, 0, 30) * norm / np.linalg.norm(grad)
        for alpha in alphas:
            a_try = a + alpha * direction
            if (a_try < -1e-12).any():
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            val_try = fast_autocorr(a_try).max()
            if val_try < best_val:
                best_val = val_try
                best_alpha = alpha

        if best_alpha > 0:
            a = a + best_alpha * direction
            a = np.maximum(a, 0.0)
            a = a * norm / max(a.sum(), 1e-15)
            cur_conv = fast_autocorr(a)
            cur_max = cur_conv.max()
            if cur_max < best_score - 1e-13:
                best_score = cur_max
                best_a = a.copy()

        # Occasionally perturb to escape
        if outer % 200 == 199:
            noise = rng.normal(0, norm * 0.001, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert * norm / s_pert

        if verbose and outer % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS iter={outer} ({elapsed:.0f}s): best={best_score:.8f}, n_active={len(active_idx)}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Pair coordinate descent with exact convolution update."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(27182)
    no_improve = 0

    max_passes = 500
    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # Try moving mass from i to j
            for offset in range(1, 30):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    # Compute delta in convolution if we move d from i to j
                    # conv_new = conv_old + d*(delta_j - delta_i)
                    # where delta_i is the contribution of position i
                    delta_i = np.zeros(size)
                    lo = i
                    hi = min(size, i + n)
                    delta_i[lo:hi] = a[:hi - lo]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n + n - 1 - i + 1)
                    if hi2 > lo2:
                        idx2 = np.arange(lo2, hi2)
                        src2 = n - 1 - idx2
                        valid = (src2 >= 0) & (src2 < n)
                        if valid.any():
                            delta_i[lo2:hi2][valid] += a[src2[valid]]
                    # Note: position i contributes a[i] to conv[2i], handled specially

                    delta_j = np.zeros(size)
                    lo = j
                    hi = min(size, j + n)
                    delta_j[lo:hi] = a[:hi - lo]
                    lo2 = max(0, n - 1 - j)
                    hi2 = min(size, n + n - 1 - j + 1)
                    if hi2 > lo2:
                        idx2 = np.arange(lo2, hi2)
                        src2 = n - 1 - idx2
                        valid = (src2 >= 0) & (src2 < n)
                        if valid.any():
                            delta_j[lo2:hi2][valid] += a[src2[valid]]

                    delta_vec = 2.0 * (delta_j - delta_i)

                    # Self-terms: a[i]^2 at conv[2i], a[j]^2 at conv[2j]
                    # When we move d from i to j:
                    # conv[2i] changes by -2*a[i]*d + d^2
                    # conv[2j] changes by +2*a[j]*d + d^2

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 8
                    deltas = np.linspace(0.01, 0.9, n_samples) * a[i]
                    for d in deltas:
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] -= 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] += 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-13:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        if a[i] < norm * 1e-10:
                            break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                # Small perturbation
                noise = rng.normal(0, norm * 0.002, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                    conv = fast_autocorr(a)
                no_improve = 0
            elif no_improve >= 15:
                break

        if verbose and p % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={p} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_start_explore(a0, time_limit, solver, n_starts=4, verbose=True):
    """Try multiple structured perturbations and re-optimize."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = normalize_to_sum(np.asarray(a0, dtype=np.float64), norm)
    best_score = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(5555)
    time_per_start = time_limit / max(1, n_starts)

    # Different perturbation strategies
    strategies = []

    # Strategy 1: Small Gaussian noise
    strategies.append(('gauss_small', lambda arr: arr + rng.normal(0, norm * 0.02, n)))

    # Strategy 2: Large Laplace noise
    strategies.append(('laplace_large', lambda arr: arr + rng.laplace(0, norm * 0.08, n)))

    # Strategy 3: Frequency-domain perturbation
    def freq_perturb(arr):
        A = np.fft.fft(arr)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        # Perturb the magnitude
        noise_mag = rng.normal(0, A_mag.mean() * 0.1, len(A_mag))
        A_mag_pert = A_mag + noise_mag
        A_mag_pert = np.maximum(A_mag_pert, 0)
        A_pert = A_mag_pert * np.exp(1j * A_phase)
        a_pert = np.fft.ifft(A_pert).real
        return a_pert
    strategies.append(('freq', freq_perturb))

    # Strategy 4: Random binary-like structure blended with current
    def binary_blend(arr):
        binary = (rng.random(n) > 0.5).astype(float)
        return 0.7 * arr + 0.3 * binary * norm / n
    strategies.append(('binary', binary_blend))

    for si, (name, perturb_fn) in enumerate(strategies):
        if time.monotonic() - t0 > time_limit:
            break
        try:
            a_pert = perturb_fn(best_a.copy())
        except Exception:
            continue
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert * norm / s

        # Quick LP optimization
        t_rem = time_limit - (time.monotonic() - t0)
        t_sub = min(time_per_start * 0.8, t_rem * 0.85)
        if t_sub < 10:
            break
        a_out, sc = nesterov_lp_optimize(a_pert, t_sub, solver, verbose=False)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    MS [{name}] improved to {best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            # Evaluate with target normalization
            norm_parent = np.sqrt(2.0 * n_parent)
            p_norm = parent * norm_parent
            p_max = fast_autocorr(p_norm).max()
            p_score = 2.0 * n_parent * p_max / (norm_parent ** 2)
            # Simplifies to p_max since sum^2 = 2n
            p_score = p_max
            print(f"Parent: n={n_parent}, score={p_score:.8f}")
            best_seq = p_norm.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        n = len(a)
        a_norm = a * np.sqrt(2.0 * n) / s
        sc = fast_autocorr(a_norm).max()
        # Verify with evaluate_sequence
        sc_check = evaluate_safe([float(x) for x in a_norm])
        if not np.isfinite(sc_check):
            sc_check = sc
        if sc_check < best_score - 1e-10:
            best_score = sc_check
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc_check:.8f} (n={n}, t={elapsed:.0f}s)")
        elif np.isfinite(sc_check) and sc_check < best_score:
            # Small improvement, still update
            best_score = sc_check
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"    Improved [{label}]: {sc_check:.8f} ({elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) * np.sqrt(2.0 * n_base) / n_base
        best_seq = a_init.copy()
        best_score = fast_autocorr(a_init).max()
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov - main optimization (40% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n_base}")

    # Phase 2: Smooth max optimization (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.15
    print(f"\n=== Phase 2: Smooth Max n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = smooth_max_optimize(best_seq, t2, verbose=True)
        do_try_update(a_out, f"sm_n{len(best_seq)}")

    # Phase 3: LP Nesterov round 2 from updated point (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LP Nesterov R2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 4: Multi-start exploration (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Multi-Start n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = multi_start_explore(best_seq, t4, solver, n_starts=3, verbose=True)
        do_try_update(a_out, f"ms_n{len(best_seq)}")

    # Phase 5: Active-set polish (7% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.07
    print(f"\n=== Phase 5: Active-Set Polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = active_set_polish(best_seq, t5, verbose=True)
        do_try_update(a_out, f"as_n{len(best_seq)}")

    # Phase 6: Coordinate descent polish (5% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.05
    print(f"\n=== Phase 6: CD Polish n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = coordinate_descent_polish(best_seq, t6, verbose=True)
        do_try_update(a_out, f"cd_n{len(best_seq)}")

    # Final phase: one more LP + active-set
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7a = min(remaining * 0.5, 60.0)
        t7b = remaining - t7a - 5
        if t7a > 15:
            print(f"\n=== Phase 7a: Final LP n={len(best_seq)} (budget={t7a:.0f}s) ===")
            a_out, _ = nesterov_lp_optimize(best_seq, t7a, solver, verbose=False)
            do_try_update(a_out, f"lp3_n{len(best_seq)}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10:
            print(f"\n=== Phase 7b: Final AS+CD n={len(best_seq)} (budget={remaining2:.0f}s) ===")
            t_as = remaining2 * 0.5
            t_cd = remaining2 * 0.45
            if t_as > 5:
                a_out, _ = active_set_polish(best_seq, t_as, verbose=False)
                do_try_update(a_out, f"as_final")
            remaining3 = time_budget - (time.monotonic() - t_global)
            if remaining3 > 5:
                a_out, _ = coordinate_descent_polish(best_seq, remaining3 * 0.8, verbose=False)
                do_try_update(a_out, f"cd_final")

    # Final verification
    if best_seq is not None:
        final_score = evaluate_safe([float(x) for x in best_seq])
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
