For the coordinate descent, the key operation is: given the current `a`, try moving a small amount of mass from position `i` to position `j`, and check if the max of the convolution decreases. The change in the convolution from moving mass `d` from `i` to `j` is:

`Δb[k] = d * (a[k-j] - a[k-i])` (approximately, for small d, ignoring second-order terms and boundary effects)

More precisely:
- `a[i] -= d` changes `b[k]` by `-d * (a[k-i] + a[k-i]) = -2d*a[k-i]` (for valid k)
  - Except at `k=2i`: change is `-2d*a[i] + d^2`
- `a[j] += d` changes `b[k]` by `+d * (a[k-j] + a[k-j]) = +2d*a[k-j]` (for valid k)
  - Except at `k=2j`: change is `+2d*a[j] + d^2`

So: `Δb[k] ≈ 2d*(a[k-j] - a[k-i])` (for most k)

The new max is approximately `max_k (b[k] + 2d*(a[k-j] - a[k-i]))`.

To minimize the new max, we want to choose `d` (and the pair (i,j)) to reduce the peak.

This is a sublinear program for each pair (i,j), which can be solved by bisection on d.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization,
precise pair coordinate descent, and multi-phase search with restarts.
Key innovations: subgradient computation via vectorized operations,
exact line search for pair moves, and adaptive learning rate scheduling.

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_grad(a, active):
    """Compute subgradient of max(conv(a,a)) at positions 'active'."""
    n = len(a)
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        i_range = np.arange(i_min, i_max + 1)
        grad[i_min:i_max+1] += 2.0 * a[k - i_range]

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    if s < 1e-15:

    a = a * norm / s
    
    conv = fast_conv(a)
    best_score = conv.max()
    best_a = a.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)

    t = 0

                a[j] -= d
                s_new2 = a.sum()
                a = a * norm / s_new2
        
        if not improved:
            no_improve += 1
            # Small perturbation
            noise = rng.normal(0, norm * 0.002, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a = a_pert * norm / s_pert
            no_improve = max(0, no_improve - 1)
    
    return best_a, best_score

# --- LP best response ---
def lp_best_response(a, M, solver):
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    # Build constraint matrix efficiently
    # (a*b)[k] = sum_j a[j]*b[k-j] <= M for all k
    # This is: for each k, sum over valid j of a[j]*b[k-j] <= M
    # The constraint matrix has rows indexed by k (0..2n-2) and columns by j (0..n-1)
    # Entry (k,j) = a[j] if 0 <= k-j < n, else 0
    
    # For efficiency, build as sparse
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    for k in range(size):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    T = sp.csc_matrix((vals, (rows, cols)), shape=(size, n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_optimize(a0, time_limit, solver, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    y = a.copy()
    t0 = time.monotonic()
    momentum_t = 1.0
    patience = 0
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_conv(y).max()
        if M < 1e-12:
            break
        
        b = lp_best_response(y, M, solver)
        if b is None:
            if patience > 30:
                break
            patience += 1
            continue
        patience = 0
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        momentum_t += 1.0
        
        sc = fast_conv(y).max()
        if sc < best_score:
            best_score = sc
            best_a = y.copy()
    
    return best_a, best_score
