def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the score without calling evaluate_sequence (for speed)."""
    try:
        a = np.asarray(a, dtype=np.float64)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        n = len(a)
        conv = fast_autocorr(a)
        score = 2.0 * n * conv.max() / (s * s)
        return float(score)
    except Exception:
        return float('inf')


def build_constraint_rows(a, k_indices, n):
    """
    For fixed a, the constraint (a * b)[k] <= M is linear in b.
    (a*b)[k] = sum_j a[k-j] * b[j] for 0 <= k-j < n.
    Row k has: row[j] = a[k-j] if 0 <= k-j < n, else 0.
    Returns sparse-ish matrix (dense for simplicity).
    """
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            mat[idx, lo:hi] = a[k - hi + 1:k - lo + 1][::-1]
    return mat


def lp_best_response(a_fixed, M, active_indices, solver, n):
    """
    Solve: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_indices, b >= 0.
    """
    k_arr = np.asarray(active_indices)
    T = build_constraint_rows(a_fixed, k_arr, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None, None
    # Get duals
    duals = None
    T @ b_val  # cache
    try:
        dv = constraints[0].dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals, T

def mirror_descent_optimize(a0, time_limit, verbose=False, seed=0):

    minimizes max_k (a*a)[k] s.t. sum(a) = sqrt(2n), a >= 0.

        a = np.ones(n) / n
    else:
        a = a / s  # Normalize to sum=1 for mirror descent

    best_score = evaluate_safe(a)  # This gives the actual score
    
    t0 = time.monotonic()

    max_patience = 2000

        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        # Compute subgradient at the worst lag
        conv = fast_autocorr(a)
        M = conv.max()

        top_frac = 0.1
        threshold = conv.max() * (1 - top_frac)
        candidates = np.where(conv >= threshold)[0]
        k_star = candidates[rng.integers(len(candidates))]
        
        # Subgradient of (a*a)[k_star] w.r.t. a[j]
        # d/da[j] (a*a)[k] = a[k-j] + a[j-k] (with bounds)
        # For the score function 2*n*max/(sum)^2, the gradient involves more terms
        # but for the normalized problem (sum=1), we just minimize max_k conv(k)
        grad = np.zeros(n)
        for j in range(n):
            v = 0.0
            i1 = k_star - j
            if 0 <= i1 < n:
                v += a[i1]
            i2 = j - k_star
            if 0 <= i2 < n and i2 != i1:  # avoid double counting when i1==i2
                v += a[i2]
            # Actually for k=0: d/da[j] conv(0) = 2*a[j]
            # For k!=0: d/da[j] conv(k) = a[k-j] (if valid) + a[j-k] (if valid and j-k != k-j)
            # Let's be more careful:
            # conv(k) = sum_i a[i] * a[k-i]
            # d/da[j] conv(k) = a[k-j] (from the a[i] term when i=j) 
            #                   + a[j-k] (from the a[k-i] term when k-i=j, i.e., i=k-j)
            # If k-j == j (i.e., k=2j), these are the same term, giving 2*a[j]
            # So for k != 0: grad[j] = a[k-j]*1[k-j in [0,n)) + a[j-k]*1[j-k in [0,n))
            # This is what I wrote above.
            
            if i1 == i2 and i1 >= 0 and i1 < n:
                # k-j = j-k means k=0 (handled separately) or j=k/2
                # For k=0: grad[j] = 2*a[j]
                # For k=2j (j=k/2): grad[j] = 2*a[j]
                pass  # The formula already handles this since we add both terms
            
            grad[j] = v
        
        # For k=0, grad[j] = 2*a[j] (special case)
        if k_star == 0:
            grad = 2.0 * a
        
        # Mirror descent update (exponential weights)
        # a_new[j] = a[j] * exp(-eta * grad[j]) / sum(...)
        eta = step
        logits = np.log(a + 1e-30) - eta * grad
        logits -= logits.max()  # numerical stability
        weights = np.exp(logits)
        s_w = weights.sum()
        if s_w < 1e-30:
            a_trial = a.copy()
        else:
            a_trial = weights / s_w
        
        score_trial = evaluate_safe(a_trial)

            patience = 0
            step = min(step * 1.01, 5.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break
    
    # Convert back to proper normalization

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_constraint_rows(a, k_indices, n):

    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):

        hi = min(n, k + 1)
        if lo < hi:

        return None, None

        return None, None

    active_set = None
    iteration = 0
    erg_a = np.zeros(n)
    erg_count = 0
    last_improve_iter = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set: anneal tolerance from loose to tight
        if active_set is None or t % 25 == 0:
            # Anneal: start with 5% of max, tighten to 0.5%
            anneal_frac = max(0.005, 0.05 * np.exp(-t / 200.0))
            tol = M_curr * (1.0 - anneal_frac)
            active_set = np.where(conv_y >= tol)[0]
            # Always include top peaks
            n_top = max(5, n // 80)
            top_idx = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_idx]))
            # Limit for LP efficiency
            max_c = min(size, 3 * n)
            if len(active_set) > max_c:
                vals = conv_y[active_set]
                keep = np.argsort(vals)[-max_c:]
                active_set = active_set[keep]

        b, duals = lp_solve(y, M_curr, active_set, solver, n)

        if b is None:
            if t % 10 == 0:
                b, duals = lp_solve(y, M_curr, np.arange(size), solver, n)
            if b is None:
                continue

        # Periodic verification
        if t % 15 == 0:
            complement = np.setdiff1d(np.arange(size), active_set)
            if len(complement) > 0 and len(complement) < 4000:
                T_comp = build_constraint_rows(y, complement, n)
                viol = T_comp @ b
                viol_mask = viol > M_curr * (1 + 1e-5)
                if viol_mask.any():
                    new_active = np.union1d(active_set, complement[viol_mask])
                    max_c = min(size, 4 * n)
                    if len(new_active) > max_c:
                        vals = conv_y[new_active]
                        keep = np.argsort(vals)[-max_c:]
                        new_active = new_active[keep]
                    b2, _ = lp_solve(y, M_curr, new_active, solver, n)
                    if b2 is not None:
                        active_set = new_active
                        b = b2

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        iteration += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t == 1:
            a_mom = g.copy()
        else:
            beta = min((t - 1.0) / (t + 2.0), 0.99)
            a_mom = y_new + beta * (y_new - y_prev)
            a_mom = np.maximum(a_mom, 0.0)
            s_a = a_mom.sum()
            if s_a < 1e-15:
                a_mom = y_new.copy()
            else:
                a_mom = a_mom * norm / s_a

        y_prev = y
        y = y_new

        # Track best
        for cand in [y, a_mom]:
            cs = fast_autocorr(cand).max()
            if cs < best_score:
                best_score = cs
                best_a = cand.copy()
                last_improve_iter = iteration

        # Ergodic average
        erg_a += y
        erg_count += 1
        if erg_count % 20 == 0:
            erg_c = erg_a / erg_count
            se = erg_c.sum()
            if se > 1e-15:
                erg_c = erg_c * norm / se
                cs_e = fast_autocorr(erg_c).max()
                if cs_e < best_score:
                    best_score = cs_e
                    best_a = erg_c.copy()
                    last_improve_iter = iteration

        # Blend restart when stalled
        if iteration - last_improve_iter > 100 and iteration > 50:
            blend = 0.4
            y_r = (1 - blend) * y + blend * best_a
            y_r = np.maximum(y_r, 0.0)
            sr = y_r.sum()
            if sr > 1e-15:
                y_r = y_r * norm / sr
                cs_r = fast_autocorr(y_r).max()
                if cs_r < best_score:
                    best_score = cs_r
                    best_a = y_r.copy()
                y = y_r
                y_prev = y_r
                a_mom = y_r
                erg_a = np.zeros(n)
                erg_count = 0
                active_set = None
                last_improve_iter = iteration
                if verbose:
                    print(f"    Blend restart at iter {iteration}, M={best_score:.10f}")

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iteration} ({elapsed:.0f}s): M={M_curr:.8f}, best={best_score:.10f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=False, seed=7):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    step = 2.0 / np.sqrt(n)
    patience = 0
    max_patience = 3000

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M = conv.max()

        # Random lag from top for diversity
        top_frac = max(0.05, 0.2 * np.exp(-t / 1000.0))
        threshold = M * (1.0 - top_frac)
        candidates = np.where(conv >= threshold)[0]
        k_star = int(candidates[rng.integers(len(candidates))])

        # Subgradient
        grad = np.zeros(n)
        if k_star == 0:
            grad = 2.0 * a
        else:
            for j in range(n):
                v = 0.0
                i1 = k_star - j
                if 0 <= i1 < n:
                    v += a[i1]
                i2 = j - k_star
                if 0 <= i2 < n:
                    if i2 != i1:
                        v += a[i2]
                    else:
                        v += a[i2]
                grad[j] = v

        # Exponential weight update
        logits = np.log(a + 1e-30) - step * grad
        logits -= logits.max()
        w = np.exp(logits)
        sw = w.sum()
        if sw < 1e-30:
            a_trial = a.copy()
        else:
            a_trial = w / sw

        sc = evaluate_safe(a_trial)
        if sc < best_score - 1e-14:
            a = a_trial
            best_score = sc
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 10.0)
        else:
            patience += 1
            step *= 0.99
            if step < 1e-12:
                break

        if verbose and t % 5000 == 0:
            print(f"    Mirror t={t}: score={best_score:.10f}, step={step:.6f}")

    best_a_norm = best_a * norm / best_a.sum()
    return best_a_norm, evaluate_safe(best_a_norm)


def coord_descent(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_autocorr(a)
        tol = conv.max() * 0.9
        active_lags = np.where(conv >= tol)[0]

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            s_others = a.sum() - aj

            # Try several candidate values
            cands = [0.0, aj * 0.3, aj * 0.6, aj, aj * 1.4, aj * 2.0, s_others * 0.5]
            cands = [x for x in cands if 0 <= x <= s_others * 3]

            best_x = aj
            cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

            for x in cands:
                a[j] = x
                ns = a.sum()
                if ns < 1e-15:
                    a[j] = aj
                    continue
                nc = fast_autocorr(a)
                obj = 2.0 * n * nc.max() / (ns * ns)
                if obj < 2.0 * n * conv.max() / (a.sum() ** 2) - 1e-14:
                    best_x = x

            a[j] = best_x
            ns = a.sum()
            if ns > 1e-15:
                a = a * norm / ns
                ns_score = fast_autocorr(a).max()
                if ns_score < best_score - 1e-13:
                    best_score = ns_score
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose:
            print(f"    CD: best={best_score:.10f}, no_improve={no_improve}")

    return best_a, best_score


def propose_candidate():
    rng = np.random.default_rng(31415)
    t_global = time.monotonic()
    time_budget = 1740.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No parent, using default.")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n = 1600
        x = np.linspace(0, 1, n)
        best_seq = 0.5 + 0.35 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # Phase 1: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.28, 480.0)
    if t1 > 60:
        print(f"\n=== P1: LP Nesterov n={n_base} t={t1:.0f}s ===")
        a_n = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_o, _ = nesterov_lp(a_n, t1, solver, verbose=True, seed=0)
        try_update(a_o, f"lp1_n{n_base}")

    # Phase 2: LP with perturbation
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 40:
        print(f"\n=== P2: LP perturbed n={n_c} t={t2:.0f}s ===")
        pert = best_seq * (1.0 + 0.015 * rng.standard_normal(n_c))
        pert = np.maximum(pert, 0.0)
        if pert.sum() > 0.01:
            pert = pert / pert.sum()
            a_n = pert * np.sqrt(2.0 * n_c) / pert.sum()
            a_o, _ = nesterov_lp(a_n, t2, solver, verbose=True, seed=99)
            try_update(a_o, f"lp2_n{n_c}")

    # Phase 3: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 25:
        print(f"\n=== P3: Mirror n={n_c} t={t3:.0f}s ===")
        a_o, sc = mirror_descent(best_seq, t3, verbose=False, seed=7)
        try_update(a_o, f"mirror_n{n_c}")

    # Phase 4: Larger sequence
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 180:
        n_big = min(2400, n_c + 400)
        print(f"\n=== P4: LP large n={n_big} remaining={remaining:.0f}s ===")
        old_x = np.linspace(0, 1, n_c)
        new_x = np.linspace(0, 1, n_big)
        a_interp = np.interp(new_x, old_x, best_seq)
        a_interp = np.maximum(a_interp, 0.0)
        if a_interp.sum() > 0.01:
            a_interp = a_interp / a_interp.sum()
            t4 = min(remaining * 0.25, 300.0)
            a_n = a_interp * np.sqrt(2.0 * n_big) / a_interp.sum()
            a_o, _ = nesterov_lp(a_n, t4, solver, verbose=False, seed=456)
            try_update(a_o, f"lp_big_n{n_big}")

    # Phase 5: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t5 = min(remaining * 0.12, 120.0)
    if t5 > 20:
        print(f"\n=== P5: CD n={n_c} t={t5:.0f}s ===")
        a_n = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_o, _ = coord_descent(a_n, t5, verbose=True)
        try_update(a_o, f"cd_n{n_c}")

    # Phase 6: Final LP boost
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    if remaining > 80:
        t6 = min(remaining * 0.5, 200.0)

    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    if remaining > 40:

        print(f"\n=== P7: CD final n={n_c} t={t7:.0f}s ===")
        a_n = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_o, _ = coord_descent(a_n, t7, verbose=False)
        try_update(a_o, f"cd_fin_n{n_c}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):

        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
