def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_sparse(a):
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        ai = a[i]
        if ai > 1e-15:
            rows.extend(range(i, i + n))
            cols.extend([i] * n)
            vals.extend([ai] * n)
    T = sparse.csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def build_T_dense(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

    s = a.sum()

    return a * target_sum / s

def lp_solve_with_screening(y, M, solver, active_frac=0.7, min_active=50):
    n = len(y)
    size = 2 * n - 1
    conv_y = fast_conv_self(y)

    active = np.where(conv_y >= threshold)[0]

    T = build_T_dense(y)

        prob.solve(solver=solver, verbose=False)

        return None, active

        return None, active

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a)
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    y = a.copy()
    no_improve = 0
    patience = 300
    
    # Adaptive active fraction: start permissive, tighten over time
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-15:
            break
        
        # Adaptive screening fraction
        frac = min(0.95, 0.5 + 0.0005 * t)
        b, active = lp_solve_with_screening(y, M, solver, active_frac=frac)
        
        if b is None:
            b = lp_solve_full(y, M, solver)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new)
        
        if t > 1:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
        else:
            a_new = y_new.copy()
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        y = y_new
        a = a_new
        
        sc_a = fast_conv_self(a).max()
        sc_y = fast_conv_self(y).max()
        
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0
        elif sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic restart
        if no_improve >= patience:
            rng_restart = np.random.default_rng(t)
            n_pert = max(5, n // 20)
            idx_pert = rng_restart.choice(n, size=n_pert, replace=False)
            a[idx_pert] *= (1.0 + 0.2 * rng_restart.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            a = normalize(a)
            y = a.copy()
            no_improve = 0
            patience = max(100, patience // 2)
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

        w = np.exp(shifted)
        w = w / w.sum()
        
        # Gradient: d/d a_i [sum_k w[k] * conv[k]]
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g1 = irfft(np.conj(A) * W, fft_size)[:n]
        g2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (g1 + g2)
        
        # Multiplicative update in log space
        eta_t = eta / (1.0 + 0.01 * t)
        log_a = log_a - eta_t * grad
        
        # Project to simplex-like constraint (normalize)
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = normalize(a_new)
        
        sc = fast_conv_self(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            log_a = np.log(np.maximum(a_new, eps))
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiDesc t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lse_gd(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent for fast warmup."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    size = 2 * n - 1
    
    n_stages = 8
    temps = np.geomspace(50.0, 0.3, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 4)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            
            step = norm_val / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.02 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)
            a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.3 + 0.2 * np.sin(3 * np.pi * x)
    else:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * rng.random() * x)
    a = np.maximum(a, 0.0)
    a = a + 0.01 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}, solver={solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize if no parent
    if not parent_loaded:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n_base = len(best_seq)
    
    # === Phase 1: LSE warmup (fast) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gd(best_seq.copy(), t1, verbose=True)
        try_update(a_out, "lse1")
    
    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t2, solver, verbose=True)
        try_update(a_out, "lp1")
    
    # === Phase 3: Peak shaving CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 220.0)
    print(f"\n=== Phase 3: Peak Shave n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = peak_shave_cd(best_seq.copy(), t3, rng, verbose=True)
        try_update(a_out, "peak1")
    
    # === Phase 4: Second LP Nesterov round ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 380.0)
    print(f"\n=== Phase 4: LP Nesterov r2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t4, solver, verbose=True)
        try_update(a_out, "lp2")
    
    # === Phase 5: Multiplicative descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: MultiDesc n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = multiplicative_descent(best_seq.copy(), t5, verbose=True)
        try_update(a_out, "multidesc")
    
    # === Phase 6: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 6: Final Peak Shave n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = peak_shave_cd(best_seq.copy(), t6, rng, verbose=True)
        try_update(a_out, "peak2")
    
    # === Phase 7: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 7: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        t7 = min(remaining * 0.4, 200.0)
        init_long = interpolate_seq(best_seq, n_long)
        init_long = init_long / init_long.sum()
        
        # Quick LSE on longer
        t7a = min(t7 * 0.3, 50.0)
        a_out, _ = lse_gd(init_long, t7a, verbose=False)
        a_long = a_out / a_out.sum() if a_out.sum() > 1e-10 else init_long
        
        # LP Nesterov on longer
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.5, 150.0)
        if t7b > 30:
            a_out2, _ = nesterov_lp_optimize(a_long, t7b, solver, verbose=False)
            try_update(a_out2, f"lp_long{n_long}")
            # Compare with interpolated version of best
            # The longer sequence should have lower score due to 2n/(2n-1) factor
            # But evaluation uses actual n, so it's fair
    
    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final LP polish n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t8 = min(remaining * 0.6, 100.0)
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t8, solver, verbose=False)
        try_update(a_out, "lp_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out2, _ = peak_shave_cd(best_seq.copy(), min(remaining * 0.5, 40.0), rng, verbose=False)
            try_update(a_out2, "peak_final")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
