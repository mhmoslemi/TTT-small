def compute_subgradient(a, K):
    """Compute exact subgradient of max(conv(a,a)) at indices K."""
    n = len(a)
    SG = np.zeros(n)
    for k in K:
        if k < n:
            SG[0:k+1] += a[k::-1]
            SG[k:n] += a[0:n-k]
        else:
            lo = k - n + 1
            SG[lo:n] += a[lo:n][::-1]
    return SG

def compute_smoothed_subgradient(a, conv, tau):
    """Compute smoothed subgradient using log-sum-exp weighting."""
    n = len(a)
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    shifted = np.maximum(shifted, -300.0)

def subgradient_mirror_descent(a0, time_limit, verbose=True):

    no_improve = 0

            a[pert_idx] *= (1.0 + 0.3 * np.random.standard_normal(len(pert_idx)))

def lse_gradient_descent(a0, time_limit, verbose=True):

    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Adam-like state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    t_step = 0
    
    n_stages = 10
    temps = np.geomspace(200.0, 1.0, n_stages)

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(50, min(500, int(remaining * 6)))

            if time.monotonic() - t0 > time_limit:
                break
            t_step += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            
            grad = compute_smoothed_subgradient(a, conv, T_val)

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t_step)
            v_hat = v / (1 - beta2**t_step)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            sc = fast_conv_self(a_new).max()
            if sc <= max_c + 0.01 * T_val:
                a = a_new
            else:
                lr *= 0.8
            
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")
    
    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    no_improve_count = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it_count += 1
        if it_count > 100000:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive constraint screening
        if it_count < 30:
            active = np.where(conv_y > M * 0.6)[0]
        elif it_count < 100:
            active = np.where(conv_y > M * 0.75)[0]
        else:
            active = np.where(conv_y > M * 0.85)[0]
        if len(active) < 20:
            active = np.where(conv_y > M * 0.5)[0]
        
        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Verify dropped constraints
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * (1 + 1e-3))[0]
            if len(violated) > 0:
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None and b_full.sum() > 1e-15:
                    b = b_full
                    g = b * norm / b.sum()
        
        # Nesterov update
        alpha = 2.0 / (it_count + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        momentum = min(1.0, (it_count - 1.0) / (it_count + 2.0))
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        
        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0
        
        if no_improve_count == 150:
            n_perturb = max(10, n // 25)
            rng_p = np.random.default_rng(it_count)
            idx_perturb = rng_p.choice(n, size=n_perturb, replace=False)
            y[idx_perturb] *= (1.0 + 0.25 * rng_p.standard_normal(n_perturb))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve_count = 0
        
        if verbose and it_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={it_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def spectral_perturb(a, rng, strength=0.3):
    """Perturb in frequency domain to escape local optima."""
    n = len(a)
    s = a.sum()
    a_norm = a / s
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_norm, fft_size)
    
    # Random phase perturbation
    phases = rng.uniform(-strength, strength, len(A))
    A_pert = A * np.exp(1j * phases)
    
    a_new = irfft(A_pert, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    else:
        a_new = a_norm
    return a_new

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent for polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            
            peak_threshold = cur_max * 0.92
            peak_lags = np.where(conv >= peak_threshold)[0]
            for kl in peak_lags[:6]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                j_candidates.add(j1)
                j_candidates.add(j2)
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                n_samples = 15
                deltas = np.linspace(0.05, 0.9, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d
                
                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(5, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and total_improvements % 100 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Multi-peak
        peaks = rng.uniform(0.1, 0.9, size=4)
        a = np.zeros(n)
        for p in peaks:
            a += np.exp(-8.0 * (x - p) ** 2)
    elif choice == 7:
        # Periodic with decay
        a = (0.5 + 0.5 * np.cos(2 * np.pi * 5 * x)) * np.exp(-2.0 * (x - 0.5) ** 2)
        a = np.maximum(a, 0.01)
    elif choice == 8:
        # Random non-negative
        a = rng.uniform(0, 1, n)
        a += 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif choice == 9:
        # Two separated Gaussian clusters
        a = np.exp(-4.0*(x-0.2)**2) + np.exp(-4.0*(x-0.8)**2)
        a *= (1.0 + 0.2*np.cos(15.0*np.pi*x))
        a = np.maximum(a, 0.01)
    elif choice == 10:
        # Sawtooth-like
        a = np.abs(x - np.round(x + 0.5)) ** 0.5
        a = np.maximum(a, 0.05)
    else:
        # High-frequency modulation
        a = 0.4 + 0.3*np.cos(2*np.pi*3*x) + 0.2*np.cos(2*np.pi*11*x) + 0.1*np.cos(2*np.pi*23*x)
        a = np.maximum(a, 0.05)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # Phase 0: Quick subgradient mirror descent to improve init
    remaining = time_budget - (time.monotonic() - t_global)
    t0_sgd = min(remaining * 0.1, 120.0)
    if t0_sgd > 20:
        print(f"\n=== Phase 0: Subgrad MD init (budget={t0_sgd:.0f}s) ===")
        a_out, _ = subgradient_mirror_descent(best_seq, t0_sgd, verbose=True)
        try_update(a_out, "sgd_init")
    
    # Phase 1: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True)
        try_update(a_out, f"lp1_n{n1}")
    
    # Phase 2: Subgradient MD round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Subgrad MD round 2 n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = subgradient_mirror_descent(best_seq, t2, verbose=True)
        try_update(a_out, f"sgd2_n{n2}")
    
    # Phase 3: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LP Nesterov round 2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        try_update(a_out, f"lp2_n{n3}")
    
    # Phase 4: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 4: LSE GD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t4, verbose=True)
        try_update(a_out, f"lse_n{n4}")
    
    # Phase 5: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Pair CD n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"pcd_n{n5}")
    
    # Phase 6: Spectral perturbation + re-optimize
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Spectral perturbation (remaining={remaining:.0f}s) ===")
    if remaining > 100:
        for attempt in range(3):
            if remaining < 60:
                break
            a_pert = spectral_perturb(best_seq, rng, strength=0.2)
            t_sp = min(remaining * 0.25, 80.0)
            if t_sp > 20:
                a_out, _ = subgradient_mirror_descent(a_pert, min(t_sp, 60.0), verbose=False)
                try_update(a_out, f"spectral_{attempt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                a_out2, _ = nesterov_lp_optimize(a_pert, min(remaining * 0.3, 60.0), solver, verbose=False)
                try_update(a_out2, f"spectral_lp_{attempt}")
            remaining = time_budget - (time.monotonic() - t_global)
    
    # Phase 7: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 7: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.5, 120.0)
        a_out, _ = nesterov_lp_optimize(init_long, t_long, solver, verbose=True)
        try_update(a_out, f"lp_long_{n_long}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_long:
            t_long2 = min(remaining * 0.5, 60.0)
            a_out2, _ = subgradient_mirror_descent(best_seq, t_long2, verbose=False)
            try_update(a_out2, f"sgd_long_{n_long}")
    
    # Phase 8: Final polish loop
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    polish_rounds = 0
    while remaining > 40:
        polish_rounds += 1
        t_polish = min(remaining * 0.3, 60.0)
        
        a_out, _ = subgradient_mirror_descent(best_seq, t_polish, verbose=False)
        try_update(a_out, f"polish_sgd_{polish_rounds}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        
        a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.3, 40.0), rng, verbose=False)
        try_update(a_out, f"polish_pcd_{polish_rounds}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        
        t_lp = min(remaining * 0.3, 50.0)
        a_out, _ = nesterov_lp_optimize(best_seq, t_lp, solver, verbose=False)
        try_update(a_out, f"polish_lp_{polish_rounds}")
        remaining = time_budget - (time.monotonic() - t_global)
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
