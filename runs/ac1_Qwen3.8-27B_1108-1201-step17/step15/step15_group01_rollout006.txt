        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Check average
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Early stopping if stuck
        if n_iters > 50 and best_score >= prev_best - 1e-10:
            stuck_count += 1
            if stuck_count > 100:
                break
        else:
            stuck_count = 0
        prev_best = best_score

        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Full iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def smooth_optimize_v2(a0, time_limit, verbose=True):
    """Improved smooth optimization with adaptive temperature and multiple restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    
    # More aggressive temperature schedule
    temps = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(20, min(300, int(remaining * 2)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            
            # Use top-k peaks for the weighting (more focused)
            k_top = max(5, n // 10)
            top_k = np.argsort(c)[-k_top:]
            w = np.zeros(len(c))
            w[top_k] = np.exp((c[top_k] - c_max) / T)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.exp((c - c_max) / T)
                w_sum = w.sum()
            w = w / w_sum

            # Gradient: 2 * cross_corr(w, a)
            fft_size = 1 << (2 * n - 1 + 1).bit_length()
            W = rfft(w[:n], fft_size)  # Only need first n of w for cross corr
            A = rfft(a, fft_size)
            # cross_corr(w, a)[i] = sum_k w[k] * a[k+i] (circular, but we want linear)
            # Actually: (w * a)[i] = sum_k w[k] * a[i-k]
            # We want: sum_k w[k+i] * a[k] for each i
            # = cross_corr
            cross = irfft(W * np.conj(A), fft_size)[:n]
            grad = 2.0 * cross

            # Adaptive step size
            step = norm / (n * T * (1.0 + 0.01 * it))
            step = min(step, norm / (n * 0.3))

            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            # Nesterov
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score

def simulated_annealing_block(a0, time_limit, rng, block_size=10, verbose=True):
    """Simulated annealing with block moves."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    current_score = best_score

    t0 = time.monotonic()
    initial_temp = 0.05
    cooling = 0.9999
    temp = initial_temp
    accepted = 0
    total = 0

    while time.monotonic() - t0 < time_limit:
        # Pick a random block of positions
        start = rng.integers(0, n - block_size)
        block_indices = list(range(start, start + block_size))
        
        # Randomly redistribute mass within the block
        old_block = a[block_indices].copy()
        old_sum = old_block.sum()
        if old_sum < 1e-15:
            continue
        
        # Create a new random distribution of the mass
        # Mix: keep some of the original, add random perturbation
        new_block = old_block + rng.normal(0, 0.1 * old_sum / block_size, block_size)
        new_block = np.maximum(new_block, 0)
        ns = new_block.sum()
        if ns < 1e-15:
            continue
        new_block = new_block * old_sum / ns

        a_new = a.copy()
        a_new[block_indices] = new_block
        a_new = np.maximum(a_new, 0)
        a_new = a_new * norm / a_new.sum()

        new_score = fast_autocorr(a_new).max()
        total += 1
        
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / temp):
            a = a_new
            current_score = new_score
            accepted += 1
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

        temp *= cooling

        if verbose and total % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA Block iter={total} ({elapsed:.0f}s): best={best_score:.8f}, temp={temp:.4f}, accept={accepted}/{total}")

    return best_a, best_score

def peak_targeted_cd(a0, time_limit, verbose=True):
    """Coordinate descent that targets the top peak positions specifically."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        conv = fast_autocorr(a)
        c_max = conv.max()
        
        # Get top peak positions
        top_k = min(20, len(conv))
        peak_positions = np.argsort(conv)[-top_k:]
        
        improved = False
        for k in peak_positions:
            # Direction to reduce conv[k]: d[i] = -a[k-i] (centered)
            d = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    d[i] = -a[j]
            
            d -= d.mean()  # Center to maintain sum
            
            # Line search
            best_step = 0
            best_val = c_max
            for step_scale in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                step = step_scale * norm / np.sqrt(np.sum(d**2) + 1e-15)
                a_test = a + step * d
                a_test = np.maximum(a_test, 0)
                s = a_test.sum()
                if s < 0.01:
                    continue
                a_test = a_test * norm / s
                val = fast_autocorr(a_test).max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_step = step
                else:
                    break
            
            if best_step > 0:
                a = a + best_step * d
                a = np.maximum(a, 0)
                a = a * norm / a.sum()
                new_max = fast_autocorr(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                improved = True

        if not improved and n_iters > 100:
            break

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Peak CD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss
