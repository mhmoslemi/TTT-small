Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive active-set LP best-response with Frank-Wolfe + momentum
2. Softmax L-BFGS polishing
3. Peak-directed coordinate descent

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T_subset(a, rows):
    """Build rows of the convolution Toeplitz matrix."""
    n = len(a)
    size = 2 * n - 1
    K = len(rows)
    T = np.zeros((K, n))
    for idx, k in enumerate(rows):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_end > i_start:
            T[idx, i_start:i_end] = a[k - np.arange(i_start, i_end)]
    return T

def solve_lp_best_response(a, M, active_rows, solver):
    """Solve: max sum(b) s.t. T_active @ b <= M, b >= 0."""
    n = len(a)
    T = build_T_subset(a, active_rows)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def adaptive_lp_best_response(a, M, solver, initial_K=50, max_expand=200, verbose=False):
    """LP with adaptive active set: start small, expand if violations found."""
    n = len(a)
    size = 2 * n - 1

    K = min(initial_K, size)
    active = np.argsort(conv)[-K:]
    active = np.sort(active)
    
    b = None
    for expand_iter in range(5):
        b = solve_lp_best_response(a, M, active, solver)
        if b is None:
            return None
        
        # Check for violations outside active set
        cross_conv = fast_conv_cross(a, b)
        if len(active) < size:
            inactive = np.setdiff1d(np.arange(size), active, assume_unique=False)
            violations = cross_conv[inactive] > M * 1.001
            if np.any(violations):
                # Add violated constraints
                violated_idx = inactive[violations]
                # Add top violators
                worst = violated_idx[np.argsort(cross_conv[violated_idx])[-10:]]
                active = np.sort(np.concatenate([active, worst]))
                if verbose:
                    print(f"      LP expand: {len(active)} constraints (iter {expand_iter})")
                continue
        return b
    
    return b

def softmax_objective_and_grad(a, temperature):
    """Compute softmax-smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    conv = fast_conv_self(a)
    shifted = conv / temperature

    exp_c = np.exp(shifted)

    obj = temperature * np.log(exp_c.sum()) + conv.max()  # = temperature * logsumexp(conv/T)

    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    grad = 2.0 * irfft(np.conj(A) * W, fft_size)[:n]
    
    return obj, grad, conv.max()

def softmax_lbfgs_polish(a0, time_limit, verbose=True):
    """Polish using L-BFGS on softmax-smoothed objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, fast_conv_self(a0).max()
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Try multiple temperatures
    temperatures = [2.0, 1.0, 0.5, 0.2, 0.1]
    
    for T_val in temperatures:
        if time.monotonic() - t0 > time_limit - 5:
            break
        if time.monotonic() - t0 > time_limit:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        
        def fun_and_grad(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10, np.zeros_like(x)
            x_norm = x * norm / sx
            obj, grad, hard_max = softmax_objective_and_grad(x_norm, T_val)

            # The gradient of soft_max wrt x (before normalization) needs chain rule

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive active-set LP best-response with Frank-Wolfe step + momentum
2. Restricted step-size line search for accelerated convergence
3. Softmax L-BFGS polishing via smooth max approximation
4. Peak-directed pair coordinate descent for final refinement
5. Multi-length exploration to avoid local optima
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# ========== Core computation ==========

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T_subset(a, rows):
    n = len(a)
    K = len(rows)
    T = np.zeros((K, n))
    for idx in range(K):
        k = rows[idx]
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_end > i_start:
            idxs = np.arange(i_start, i_end)
            T[idx, i_start:i_end] = a[k - idxs]
    return T

def build_T_full(a):
    n = len(a)
    size = 2 * n - 1
    k_arr = np.arange(size)[:, None]
    i_arr = np.arange(n)[None, :]
    diff = k_arr - i_arr
    valid = (diff >= 0) & (diff < n)
    T = np.zeros((size, n))
    T[valid] = a[diff[valid]]
    return T

# ========== LP Best Response ==========

def solve_lp(a, M, rows, solver):
    n = len(a)
    T = build_T_subset(a, rows)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def adaptive_lp(a, M, solver, init_K=60, max_iter=8):
    n = len(a)
    size = 2 * n - 1
    conv = fast_conv_self(a)
    K = min(init_K, size)
    active = np.sort(np.argsort(conv)[-K:])
    b = None
    for _ in range(max_iter):
        b = solve_lp(a, M, active, solver)
        if b is None:
            return None
        if len(active) >= size:
            break
        cross = fast_conv_cross(a, b)
        inactive_mask = np.ones(size, dtype=bool)
        inactive_mask[active] = False
        viol = cross[inactive_mask] > M * 1.0005
        if np.any(viol):
            inactive_idx = np.where(inactive_mask)[0]
            violated = inactive_idx[viol]
            worst = violated[np.argsort(cross[violated])[-20:]]
            active = np.sort(np.concatenate([active, worst]))
        else:
            break
    return b

# ========== FW + LP Optimizer ==========

def fw_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    y = a.copy()  # Nesterov variable
    size = 2 * n - 1

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive active set LP
        K = min(60 + t // 50, size)
        b = adaptive_lp(y, M, solver, init_K=K)
        if b is None:
            active_all = np.arange(size)
            b = solve_lp(y, M, active_all, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum

        # Frank-Wolfe step size: gamma = 2/(t+2)
        gamma = 2.0 / (t + 2.0)

        # Nesterov momentum
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

# ========== Softmax Polish ==========

def softmax_grad_fn(a, temperature):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    conv = fast_conv_self(a)
    shifted = conv / temperature
    shifted = shifted - shifted.max()
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    grad = 2.0 * irfft(np.conj(A) * W, fft_size)[:n]
    return grad

def softmax_lbfgs(a0, time_limit, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, fast_conv_self(a0).max()
    a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    temperatures = [3.0, 1.5, 0.7, 0.3, 0.15, 0.08]
    for T_val in temperatures:
        if time.monotonic() - t0 > time_limit - 3:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        cur_a = a.copy()

        def obj_grad(x):
            x_pos = np.maximum(x, 0.0)
            sx = x_pos.sum()
            if sx < 1e-15:
                return 1e10, np.zeros(n)
            x_norm = x_pos * norm_val / sx
            grad = softmax_grad_fn(x_norm, T_val)
            conv = fast_conv_self(x_norm)
            shifted = conv / T_val
            shifted = shifted - shifted.max()
            obj = T_val * np.log(np.exp(shifted).sum()) + conv.max()
            return obj, grad * (norm_val / sx)

        try:
            res = minimize(obj_grad, cur_a, method='L-BFGS-B',
                          options={'maxiter': 300, 'maxfun': 800,
                                   'ftol': 1e-14, 'gtol': 1e-11,
                                   'max_time': remaining - 2, 'disp': False})
            if res.x is not None:
                a_new = np.maximum(res.x, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm_val / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                    a = a_new
                    if verbose:
                        print(f"    Softmax T={T_val}: score={best_score:.8f}")
        except Exception:
            pass

    return best_a, best_score

def peak_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, fast_conv_self(a0).max()
    a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        # Focus on top peaks
        peak_threshold = cur_max * 0.93
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm_val * 1e-12:
                continue

            # Find candidate j's that reduce the peaks
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                for offset in [-1, 0, 1]:
                    j_s = (int(kl) - i + offset + n) % n
                    if j_s != i:
                        j_candidates.add(j_s)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:12]:
                aj = a[j]

                # Compute gain vector for transferring delta from i to j
                # conv change: -delta*(a[k-i]) + delta*(a[k-j]) - delta^2*delta_{k,2i} + delta^2*delta_{k,2j} + 2*delta*(a[j]-a[i])... 
                # Simplified: new_conv[k] = conv[k] - delta*a[k-i] + delta*a[k-j]
                # (ignoring second-order terms for candidate selection)
                g_vec = np.zeros(size)
                # -delta * a[k-i] for k where 0<=k-i<n
                ki = np.arange(max(0, i), min(size, i + n))
                g_vec[ki] -= a[ki - i]
                # +delta * a[k-j] for k where 0<=k-j<n
                kj = np.arange(max(0, j), min(size, j + n))
                g_vec[kj] += a[kj - j]

                # Sample deltas
                max_d = ai * 0.9
                n_samples = 15
                deltas = np.linspace(0.0, max_d, n_samples)
                for d in deltas[1:]:
                    new_conv = conv + d * g_vec
                    # Second-order corrections for self-terms
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross term correction
                    cross_idx = i + j
                    if cross_idx < size:
                        new_conv[cross_idx] = conv[cross_idx] - ai * d + aj * d + d * (aj - ai)
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a * norm_val / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 2:
                n_pert = max(10, n // 25)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_val / s_new
        else:
            no_improve = 0

        if verbose and (idx + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(50.0, 0.3, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 1:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm_val / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.03 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm_val / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.35 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.3
    elif choice == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x * (1 - x) * 4) * (1 + 0.3 * np.sin(5 * np.pi * x))
    else:
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 0.8 + 0.2 * np.sin(4 * np.pi * x)
    a = np.maximum(a, 0.0)
    a = a + 0.005 * rng.random(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    if t3 > 15:
        a_out, _ = softmax_lbfgs(best_seq, t3, verbose=True)
        try_update(a_out, "softmax1")

    # Phase 4: Peak CD
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Peak CD n={len(best_seq)} ({t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = peak_cd(best_seq, t4, rng, verbose=True)
        try_update(a_out, "peakcd1")

    # Phase 5: Second round FW-LP (with improved starting point)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: FW-LP round 2 n={len(best_seq)} ({t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = fw_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "fwlp2")

    # Phase 6: Try a different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_base + 200
    t6 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 6: Alt length n={n_alt} ({t6:.0f}s) ===")
    if t6 > 30 and remaining > 80:
        init_alt = interpolate_seq(best_seq, n_alt)
        t6a = min(t6 * 0.6, 100.0)
        a_out, _ = fw_lp_optimize(init_alt, t6a, solver, verbose=False)
        if try_update(a_out, f"alt_n{n_alt}"):
            n_base = n_alt
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out2, _ = peak_cd(best_seq, min(remaining * 0.5, 50.0), rng, verbose=False)
            try_update(a_out2, "alt_cd")

    # Phase 7: Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} ({remaining:.0f}s) ===")
    if remaining > 30:
        t7a = min(remaining * 0.35, 100.0)
        a_out, _ = fw_lp_optimize(best_seq, t7a, solver, verbose=False)
        try_update(a_out, "final_lp")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.35, 80.0)
            a_out, _ = softmax_lbfgs(best_seq, t7b, verbose=True)
            try_update(a_out, "final_softmax")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t7c = min(remaining * 0.6, 60.0)
            a_out, _ = peak_cd(best_seq, t7c, rng, verbose=False)
            try_update(a_out, "final_cd")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
