3. Adaptive constraint screening with validation
4. Projected subgradient polishing
5. Multi-length exploration (n=1600 primary, n=2000 secondary)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    import cvxpy as cp
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver: ECOS (fallback)")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def build_T(a, active):
    n = len(a)
    T = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if hi > lo:
            T[idx, lo:hi] = a[k - np.arange(lo, hi)]
    return T

def solve_lp(a, M, active, solver):
    import cvxpy as cp
    n = len(a)
    T = build_T(a, active)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def project_to_simplex(v, target_sum):
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0.0)
    if v.sum() < 1e-15:
        return np.full(n, target_sum / n)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return v * (target_sum / v.sum())
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_subgrad_at_maxlag(a, k):
    n = len(a)
    sg = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(k + 1, n)
    if hi1 > lo1:
        sg[lo1:hi1] += a[k - np.arange(lo1, hi1)]
    lo2 = max(k, 0)
    hi2 = min(k + n, n)
    if hi2 > lo2:
        sg[lo2:hi2] += a[np.arange(lo2, hi2) - k]
    return sg

def lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    a = a * norm / S

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        if t < 50:
            frac = 0.3
        elif t < 200:
            frac = 0.5
        elif t < 1000:
            frac = 0.65
        else:
            frac = 0.8

        active = np.where(conv > M * frac)[0]
        if len(active) < 15:
            active = np.where(conv > M * 0.2)[0]

        b = solve_lp(a, M, active, solver)
        if b is None:
            b = solve_lp(a, M, np.arange(size), solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Validate
        if len(active) < size:
            check = fast_conv_cross(a, g)
            if check.max() > M * 1.005:
                more_active = np.where(conv > M * 0.45)[0]
                b2 = solve_lp(a, M, more_active, solver)
                if b2 is not None and b2.sum() > 1e-15:
                    g = b2 * norm / b2.sum()

        # Launching heuristic: try multiple step sizes
        alpha_ns = 2.0 / (t + 2.0)
        candidates = [alpha_ns, 1.0 / (t + 1.0), 0.25, 0.4, 0.6, 0.8]
        # Deduplicate
        seen = set()
        unique_candidates = []
        for c in candidates:
            c = max(0.01, min(1.0, c))
            key = round(c, 4)
            if key not in seen:
                seen.add(key)
                unique_candidates.append(c)

        best_candidate = None
        best_candidate_score = compute_score(a)

        for gamma in unique_candidates:
            a_new = (1.0 - gamma) * a + gamma * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new * norm / s_new
            sc = compute_score(a_new)
            if sc < best_candidate_score:
                best_candidate_score = sc
                best_candidate = a_new

        # Also try geometric interpolation
        eps = 1e-12
        for gamma in [0.3, 0.5, 0.7]:
            a_geom = a * (g + eps) ** gamma
            a_geom = a_geom / (a_geom ** (1.0 / n))
            a_geom = np.maximum(a_geom, 0.0)
            s_g = a_geom.sum()
            if s_g < 1e-15:
                continue
            a_geom = a_geom * norm / s_g
            sc = compute_score(a_geom)
            if sc < best_candidate_score:
                best_candidate_score = sc
                best_candidate = a_geom

        if best_candidate is not None:
            a = best_candidate
            if best_candidate_score < best_score - 1e-9:
                best_score = best_candidate_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve == 80:
            rng_p = np.random.default_rng(t)
            n_pert = max(5, n // 40)
            idx_p = rng_p.choice(n, size=n_pert, replace=False)
            a[idx_p] *= (1.0 + 0.25 * rng_p.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a
            no_improve = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def subgradient_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a0, compute_score(a0)
    a = a * norm / S

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    eta0 = norm * 0.3

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()
        tol = M * 1e-6
        max_lags = np.where(conv >= M - tol)[0]
        if len(max_lags) == 0:
            max_lags = np.array([np.argmax(conv)])

        # Compute subgradient (average over max lags)
        sg = np.zeros(n)
        for k in max_lags[:20]:
            sg += compute_subgrad_at_maxlag(a, int(k))
        sg /= min(len(max_lags), 20)

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Adaptive step size
        eta = eta0 / (1.0 + 0.005 * it)
        a_new = a - eta * sg / sg_norm * 2.0
        a_new = project_to_simplex(a_new, norm)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        # Always move (even if not improving) to explore
        a = a_new

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def pair_cd_polish(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent: move mass from one position to another."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a0, compute_score(a0)
    a = a * norm / S

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        n_evals = 0

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-8:
                continue

            # Find j to move mass to
            # Try positions that would reduce the peak
            peak_lags = np.where(conv >= cur_max * 0.85)[0]
            j_candidates = set()
            for kl in peak_lags[:10]:
                j_candidates.add(int((kl - i) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.add(int(rng.integers(n)))
            j_candidates.discard(i)

            best_delta = 0.0
            best_j = -1
            best_val = cur_max

            for j in list(j_candidates)[:8]:
                if j == i or a[j] > norm:
                    continue
                # Compute gain vector for moving delta from i to j
                # conv(new)[k] = conv(old)[k] + delta*(g_j[k] - g_i[k]) + delta^2*(corr terms)
                # For small delta, we can approximate
                n_samples = 15
                deltas = np.linspace(0.05, 0.8, n_samples) * a[i]
                for d in deltas:
                    # Fast approximate: just check if it helps
                    a_trial = a.copy()
                    a_trial[i] -= d
                    a_trial[j] += d
                    sc = compute_score(a_trial)
                    n_evals += 1
                    if sc < best_val:
                        best_val = sc
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score * best_score * a.sum() / (2.0 * n) + 1e-10:
                        sc_actual = compute_score(a)
                        if sc_actual < best_score:
                            best_score = sc_actual
                            best_a = a.copy()
                            pass_improved = True
                    else:
                        # Revert
                        a[best_j] -= best_delta
                        a[i] += best_delta

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break
            if no_improve_passes == 3:
                n_pert = max(10, n // 30)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * norm / s_a
        else:
            no_improve_passes = 0

        if verbose and no_improve_passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD passes_no_imp={no_improve_passes} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.exp(-4.0 * x) * (1.0 + 0.3 * np.cos(6.0 * np.pi * x))
    elif choice == 1:
        a = np.exp(-3.0 * x) * (1.0 + 0.2 * np.sin(10.0 * np.pi * x))
    elif choice == 2:
        a = (1.0 - x) ** 2 * (1.0 + 0.4 * np.cos(8.0 * np.pi * x))
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = np.sin(np.pi * x) ** 0.6 * np.exp(-2.0 * x)
    elif choice == 5:
        a = 0.4 * np.exp(-3.0 * x) + 0.6 * np.exp(-6.0 * x)
    elif choice == 6:
        a = np.maximum(0.0, np.cos(np.pi * x) ** 3 + 0.2 * np.cos(12.0 * np.pi * x))
    else:
        a = np.exp(-2.0 * x) * (0.5 + 0.5 * np.cos(14.0 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LP optimization on parent
    n1 = len(best_seq) if best_seq is not None else 1600
    if best_seq is None:
        # Try several inits and pick best
        best_init = None
        best_init_sc = float('inf')
        for trial in range(8):
            cand = structured_init(n1, rng)
            sc = compute_score(cand)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_sc
        print(f"Default init: n={n1}, score={best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_start)
    t1 = remaining * 0.50
    print(f"\n=== Phase 1: LP optimization n={n1} (budget={t1:.0f}s) ===")
    a_out, _ = lp_optimize(best_seq, t1, solver, verbose=True)
    try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Subgradient polish
    remaining = time_budget - (time.monotonic() - t_start)
    n2 = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Subgrad polish n={n2} (budget={t2:.0f}s) ===")
    a_out, _ = subgradient_polish(best_seq, t2, verbose=True)
    try_update(a_out, f"subgrad_n{n2}")

    # Phase 3: LP round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_start)
    n3 = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LP round 2 n={n3} (budget={t3:.0f}s) ===")
    a_out, _ = lp_optimize(best_seq, t3, solver, verbose=True)
    try_update(a_out, f"lp2_n{n3}")

    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_start)
    n4 = len(best_seq)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: PCD n={n4} (budget={t4:.0f}s) ===")
    a_out, _ = pair_cd_polish(best_seq, t4, rng, verbose=True)
    try_update(a_out, f"pcd_n{n4}")

    # Phase 5: Length extension to n=2000
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 200:
        n_long = 2000
        init_long = interpolate_seq(best_seq, n_long)
        t5 = min(remaining * 0.35, 300.0)
        print(f"\n=== Phase 5: LP on n={n_long} (budget={t5:.0f}s) ===")
        a_out, _ = lp_optimize(init_long, t5, solver, verbose=True)
        try_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 60 and len(best_seq) == n_long:
            t5b = min(remaining * 0.4, 120.0)
            a_out, _ = subgradient_polish(best_seq, t5b, verbose=False)
            try_update(a_out, f"subgrad_long_{n_long}")

    # Phase 6: Final LP polish on best
    remaining = time_budget - (time.monotonic() - t_start)
    n6 = len(best_seq)
    if remaining > 60:
        t6 = min(remaining * 0.6, 200.0)
        print(f"\n=== Phase 6: Final LP n={n6} (budget={t6:.0f}s) ===")
        a_out, _ = lp_optimize(best_seq, t6, solver, verbose=False)
        try_update(a_out, f"lp_final_n{n6}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
