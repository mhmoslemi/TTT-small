def fast_conv_self(a):

def fast_conv_cross(a, b):

def build_T_column(a, j):
    """Return column j of the Toeplitz matrix T where (T@b)_k = (a*b)_k"""
    n = len(a)
    size = 2 * n - 1
    col = np.zeros(size)
    # (a*b)_k = sum_i a_i * b_{k-i}
    # For fixed j = b's index: contribution to k is a_i * b_j where k-i = j, i.e., k = i+j
    # So col[k] = a_{k-j} for valid i (where i = k-j is in [0, n-1])
    for i in range(n):
        k = i + j
        if 0 <= k < size:
            col[k] += a[i]
    return col

def lp_active_set(a, M, active_set, solver, warm_start=None):
    """
    Solve LP: max sum(b) s.t. T(a) @ b <= M (on active_set), b >= 0
    Returns (b, duals_on_active)
    """
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        b_var.value = warm_start
    
    # Build constraints only on active set
    # T @ b at position k = sum_i a_i * b_{k-i} (with bounds)
    # We need: for each k in active_set, sum_i a_i * b_{|k-i|} <= M
    
    constraints = [b_var >= 0]
    
    # Efficient construction: for each k, the constraint is sum over valid i of a[i] * b[k-i] <= M
    # Actually let me use the dense matrix approach for the active set
    if len(active_set) == 0:
        active_set = np.arange(size)
    
    # Build submatrix
    n_active = len(active_set)
    T_sub = np.zeros((n_active, n))
    for idx, k in enumerate(active_set):
        # (a*b)_k = sum_i a[i] * b[k-i] where 0<=i<n and 0<=k-i<n
        # i ranges from max(0, k-(n-1)) to min(k, n-1)
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            j = k - i  # index into b
            T_sub[idx, j] = a[i]
    
    constraints = [T_sub @ b_var <= M * np.ones(n_active)]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    # Get duals
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            d = np.asarray(con.dual_value, dtype=np.float64)
            duals = d
    except Exception:
        pass
    
    return val, duals

def lp_cg_optimize(a, M, solver, max_iters=20, verbose=False):
    """
    Constraint generation LP: start with few constraints, add violated ones.
    Returns best b found.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Start with peak constraints
    conv_a = fast_conv_self(a)
    threshold = M * 0.9
    active = np.where(conv_a >= threshold)[0]
    if len(active) < 5:
        active = np.where(conv_a >= M * 0.5)[0]
    if len(active) < 3:
        active = np.argsort(conv_a)[-10:]
    
    b = None
    warm_start = None
    
    for it in range(max_iters):
        b, duals = lp_active_set(a, M, active, solver, warm_start=b)
        if b is None:
            # Try with more constraints
            if len(active) < size:
                active = np.where(conv_a >= M * 0.3)[0]
                continue
            return None
        
        # Check all constraints
        cross = fast_conv_cross(a, b)
        violated = np.where(cross > M * (1 + 1e-6))[0]
        
        if len(violated) == 0:
            return b
        
        # Add violated constraints to active set
        active = np.unique(np.concatenate([active, violated]))
        warm_start = b
        
        if verbose:
            print(f"      CG LP iter {it}: {len(active)} active, {len(violated)} violated")
    
    # Final check
    cross = fast_conv_cross(a, b)
    if cross.max() <= M * (1 + 1e-4):
        return b
    # If still violated, solve with all constraints
    active = np.arange(size)
    b, _ = lp_active_set(a, M, active, solver, warm_start=b)
    return b

def nesterov_lp_main(a0, time_limit, solver, verbose=True, momentum_adaptive=True):
    """
    Main Nesterov LP optimization loop.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    
    # Track for Adam-like adaptation
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2 = 0.9, 0.999
    
    no_improve = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Constraint generation LP
        b = lp_cg_optimize(y, M, solver, max_iters=10)
        if b is None:
            # Fallback: try with relaxed M
            b = lp_cg_optimize(y, M * 1.01, solver, max_iters=5)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Adaptive momentum
        if momentum_adaptive and t > 10:
            # Use Adam-like update
            direction = g - y
            m = beta1 * m + (1 - beta1) * direction
            v = beta2 * v + (1 - beta2) * direction ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 1.0 / (np.sqrt(t) + 1.0)
            y_new = y + lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        else:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
        
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Nesterov lookahead
        momentum = min(1.0, t / (t + 2.0))
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Check improvement
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        # Perturbation to escape local optima
        if no_improve == 80:
            rng_p = np.random.default_rng(t)
            n_perturb = max(10, n // 25)
            idx_p = rng_p.choice(n, size=n_perturb, replace=False)
            scale = 1.0 + 0.25 * rng_p.standard_normal(n_perturb)

            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0
        
        # Averaging trick
        if t % 200 == 0:
            avg_a = a.copy()
            s_avg = avg_a.sum()
            if s_avg > 1e-15:

                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_a.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP t={t} ({elapsed:.0f}s): M={M:.6f}, best={best_score:.8f}")
    
    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True):
    """Log-sum-exp gradient descent with annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Multi-stage annealing
    temps = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(200, int(remaining * 3)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # LSE approximation of max
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient: d/d a_i [sum_k w_k * (a*a)_k]
            # (a*a)_k = sum_i a_i * a_{k-i}
            # d/d a_j = sum_k w_k * (a_{k-j} + a_{j-k}) where indices valid
            # = 2 * sum_k w_k * a_{|k-j|} (by symmetry)
            # In FFT: grad_j = 2 * (IFFT(conj(FFT(a)) * FFT(w)))_j
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)
            
            # Normalized gradient descent
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            grad = grad / grad_norm
            
            # Backtracking line search
            step = 1.0 / (T_val * 0.1 + 1.0)
            for trial in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.01 * T_val:
                    break
                step *= 0.5
            
            a = a - step * grad
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = a.copy()
            else:
                a = a * norm / s_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE T={T_val}: best={best_score:.8f}")
    
    return best_a, best_score

def freq_equalize_init(n, rng):
    """Initialize using frequency domain equalization."""
    # Start with a random phase spectrum, equalize power
    A = (rng.standard_normal(n) + 1j * rng.standard_normal(n))
    # Equalize power spectrum
    P = np.abs(A) ** 2
    P_mean = P.mean()
    P_flat = np.ones(n) * P_mean
    # Reconstruct with flat power but random phase
    A_eq = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
    a = np.fft.ifft(A_eq).real
    a = np.abs(a)  # ensure non-negative
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def multiplicative_weights_refine(a0, time_limit, solver, verbose=True):
    """
    Pure multiplicative weights (exponentiated gradient) with LP oracle.
    This is equivalent to solving the dual of the LP gap.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Maintain dual variables (price for each lag)
    dual = np.zeros(2 * n - 1)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        M = conv.max()
        if M < 1e-12:
            break
        
        # Update dual variables: add weight at peak
        peak_lag = np.argmax(conv)
        lr = 1.0 / (np.sqrt(t) + 1.0)
        dual[peak_lag] += lr
        # Also add to near-peaks
        near_peaks = np.where(conv > M * 0.85)[0]
        dual[near_peaks] += lr * 0.5
        dual = np.maximum(dual, 0)
        
        # Solve LP with current dual guidance
        # The LP gives the best response
        b = lp_cg_optimize(a, M, solver, max_iters=8)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Multiplicative update (exponentiated gradient)
        eta = 1.0 / (np.sqrt(t) + 1.0)
        ratio = g / np.maximum(a, 1e-30)
        a_new = a * np.exp(eta * np.log(np.clip(ratio, 1e-10, 1e10)))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW t={t} ({elapsed:.0f}s): best={best_score:.8f}")

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver_cache

def lp_with_constraints(a, M, constraint_indices, solver, warm_start=None):
    """Solve max sum(b) s.t. (a*b)[k] <= M for k in constraint_indices, b >= 0."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    n_con = len(constraint_indices)
    
    if n_con == 0:
        constraint_indices = np.array([np.argmax(fast_conv_self(a))])
        n_con = 1
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: for each lag k, (a*b)_k = sum_i a[i]*b[k-i]
    # T_sub[row, col] where row corresponds to lag, col to b index
    T_sub = np.zeros((n_con, n))
    for r, k in enumerate(constraint_indices):
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            T_sub[r, j] = a[i]
    
    constraints = [T_sub @ b_var <= M * np.ones(n_con)]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    if val.sum() < 1e-15:
        return val, None
    
    # Get dual variables
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    
    return val, duals

def lp_oracle(a, M, solver, max_rounds=15):
    """
    Constraint generation LP oracle.
    Returns best feasible b (or best found).
    """
    n = len(a)
    size = 2 * n - 1
    conv_a = fast_conv_self(a)
    
    # Start with top constraints
    top_k = max(20, min(100, size // 10))
    active = np.argsort(conv_a)[-top_k:]
    
    b = None
    for round_idx in range(max_rounds):
        b, duals = lp_with_constraints(a, M, active, solver, warm_start=b)
        if b is None:
            # Relax slightly
            b, duals = lp_with_constraints(a, M * 1.005, active, solver, warm_start=b)
            if b is None:
                return None
        
        # Check all constraints
        cross = fast_conv_cross(a, b)
        violated = np.where(cross > M * (1 + 1e-5))[0]
        
        if len(violated) == 0:
            return b
        
        # Add violated to active set
        active = np.unique(np.concatenate([active, violated]))
        # Limit active set size for speed
        if len(active) > 300:
            # Keep most violated
            cross_on_active = cross[active]
            order = np.argsort(-cross_on_active)
            active = active[order[:300]]
    
    # Final: if still violated, try full
    cross = fast_conv_cross(a, b)
    if cross.max() > M * (1 + 1e-3):
        # Try with all violated + top
        violated = np.where(cross > M)[0]
        active = np.unique(np.concatenate([np.argsort(conv_a)[-50:], violated]))
        b2, _ = lp_with_constraints(a, M, active, solver, warm_start=b)
        if b2 is not None:
            b = b2
    
    return b

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_adaptive_momentum=False):
    """Main Nesterov LP optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    
    # Dual averaging tracking
    dual_avg = np.zeros(2 * n - 1)
    
    no_improve = 0
    perturbation_count = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # LP oracle
        b = lp_oracle(y, M, solver, max_rounds=10)
        if b is None:
            b = lp_oracle(y, M * 1.01, solver, max_rounds=5)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Update
        if use_adaptive_momentum and t > 20:
            # Exponentiated gradient with decaying rate
            eta = 1.0 / (np.sqrt(t) * 0.5 + 1.0)
            ratio = g / np.maximum(y, 1e-20)
            y_new = y * np.exp(eta * np.log(np.clip(ratio, 1e-8, 1e8)))
        else:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
        
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Nesterov lookahead
        if t > 5:
            beta = (t - 1.0) / (t + 1.0)
            a_new = y_new + beta * np.maximum(y_new - y, 0)
        else:
            a_new = y_new.copy()
        
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track dual info
        peak_idx = np.argmax(conv_y)
        dual_avg[peak_idx] += 1.0
        
        # Check improvement
        sc_y = fast_conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        # Perturbation to escape local optima
        if no_improve == 60 + perturbation_count * 20:
            perturbation_count += 1
            rng_p = np.random.default_rng(t + perturbation_count * 1000)
            n_pert = max(10, n // (20 + perturbation_count * 5))
            idx_p = rng_p.choice(n, size=n_pert, replace=False)
            scale = 1.0 + 0.3 * rng_p.standard_normal(n_pert)
            y[idx_p] *= np.clip(scale, 0.2, 3.0)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            no_improve = 0
        
        # Periodic: try dual-guided gradient step
        if t > 50 and t % 100 == 0:
            # Gradient from dual averaging
            dual_norm = dual_avg / (dual_avg.sum() + 1e-10)
            fft_size = 1 << (2 * n - 2).bit_length()
            A_f = rfft(a, fft_size)
            D_f = rfft(dual_norm, fft_size)
            g1 = irfft(np.conj(A_f) * D_f, fft_size)[:n]
            g2 = irfft(A_f * D_f, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-10:
                step = 0.1 / (1.0 + 0.01 * t)
                a_test = a - step * grad / grad_norm
                a_test = np.maximum(a_test, 0.0)
                s_test = a_test.sum()
                if s_test > 1e-15:
                    a_test = a_test * norm / s_test
                    sc_test = fast_conv_self(a_test).max()
                    if sc_test < best_score - 1e-10:
                        best_score = sc_test
                        best_a = a_test.copy()
                        a = a_test.copy()
                        improved = True
        
        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): M={M:.6f}, best={best_score:.8f}, noimp={no_improve}")
    
    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True):
    """Annealed LSE gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    temps = np.geomspace(80.0, 0.05, 15)
    
    for T_val in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(15, min(150, int(remaining * 2.5)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -300.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            grad = grad / grad_norm
            
            # Line search
            step = 2.0 / (T_val * 0.5 + 1.0)
            for trial in range(12):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.02 * T_val:
                    break
                step *= 0.5
            
            a = a - step * grad
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = a.copy()
            else:
                a = a * norm / s_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE T={T_val:.2f}: best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    no_improve_passes = 0
    total_moves = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_lag = int(np.argmax(conv))
        
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Generate candidate j's
            candidates = set()
            for _ in range(8):
                candidates.add(int(rng.integers(n)))
            # Structured from peak
            for off in range(-3, 4):
                j1 = (peak_lag - i + off) % n
                j2 = (peak_lag + i + off) % n
                candidates.add(j1)
                candidates.add(j2)
            candidates.discard(i)
            
            best_move = None
            best_v = cur_max
            
            for j in candidates:
                if j < 0 or j >= n or j == i:
                    continue
                aj = a[j]
                
                # Try transfer fractions
                for frac in [0.2, 0.4, 0.6, 0.8]:
                    d = frac * ai
                    if d >= ai:
                        continue
                    # Quick check at peak lag
                    # (a*a)[peak] involves a[i]*a[peak-i] and a[peak-i]*a[i]
                    # Moving d from i to j changes terms involving i and j
                    di = a[(peak_lag - i) % n] if 0 <= peak_lag - i < n else 0.0
                    dj = a[(peak_lag - j) % n] if 0 <= peak_lag - j < n else 0.0
            
            
            # Just do a simple: move small amount from i to best random j
            j = int(rng.integers(n))
            d = 0.3 * ai
            a_test = a.copy()
            a_test[i] -= d
            a_test[j] += d
            a_test = np.maximum(a_test, 0.0)
            s_test = a_test.sum()
            if s_test > 1e-15:
                a_test = a_test * norm / s_test
                v = fast_conv_self(a_test).max()
                if v < best_v - 1e-10:
                    a = a_test
                    pass_improved = True
                    total_moves += 1
        
        if pass_improved:
            new_max = fast_conv_self(a).max()
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 8:
                break
            if no_improve_passes == 4:
                n_pert = max(5, n // 40)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_test = a.sum()
                if s_test > 1e-15:
                    a = a * norm / s_test
        
        if verbose and total_moves % 500 == 0 and total_moves > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD moves={total_moves} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 14)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1.0 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3.0 * np.pi * x)) ** 0.8
    elif choice == 9:
        # Frequency equalized
        A = rng.standard_normal(n) + 1j * rng.standard_normal(n)
        P = np.abs(A) ** 2
        A_eq = np.sqrt(P.mean()) * np.exp(1j * np.angle(A))
        a = np.fft.ifft(A_eq).real
        a = np.abs(a)
    elif choice == 10:
        # Lowpass random
        a = rng.standard_normal(n)
        A_f = rfft(a, 2 * n)
        A_f[int(0.4 * n):] = 0
        a = irfft(A_f, 2 * n)[:n]
        a = np.abs(a)
    elif choice == 11:
        # Asymmetric ramp with oscillation
        a = (1.0 - x) * (1.0 + 0.2 * np.cos(15 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 12:
        # Two-scale
        a = 0.6 * np.sin(np.pi * x) + 0.4 * np.sin(4 * np.pi * x) ** 2
        a = np.maximum(a, 0.01)
    else:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining Nesterov LP, LSE, and local CD with
    constraint-generation LP oracle for speed.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(10):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # Phase 0: Quick LSE warmup to smooth initial point
    remaining = time_budget - (time.monotonic() - t_global)
    t0 = min(remaining * 0.05, 80.0)
    if t0 > 10:
        a_out, _ = lse_gradient_descent(best_seq, t0, verbose=False)
        try_update(a_out, "lse_warm")
    
    # Phase 1: Main Nesterov LP (fixed momentum)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)

    print(f"\n=== Phase 1: NesterovLP n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True, use_adaptive_momentum=False)
        try_update(a_out, f"nlp1_n{n1}")
    
    # Phase 2: Nesterov LP with adaptive (exponentiated) momentum
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: NesterovLP-Exp n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True, use_adaptive_momentum=True)
        try_update(a_out, f"nlp2_n{n2}")
    
    # Phase 3: LSE gradient (different landscape perception)
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)

    # Phase 6: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 6: Length extension to n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 150:
        init_long = interpolate_seq(best_seq, n_long)
        t_long = min(remaining * 0.40, 180.0)
        a_out, _ = nesterov_lp_optimize(init_long, t_long, solver, verbose=True, use_adaptive_momentum=False)
        try_update(a_out, f"nlp_long_{n_long}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and len(best_seq) == n_long:
            t_long2 = min(remaining * 0.35, 100.0)
            a_out2, _ = lse_gradient_descent(best_seq, t_long2, verbose=False)
            try_update(a_out2, f"lse_long_{n_long}")
    
    # Phase 7: Final quick polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t_f1 = min(remaining * 0.35, 50.0)
        a_out, _ = lse_gradient_descent(best_seq, t_f1, verbose=False)
        try_update(a_out, "lse_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            t_f2 = min(remaining * 0.4, 35.0)
            a_out, _ = pair_cd_optimize(best_seq, t_f2, rng, verbose=False)
            try_update(a_out, "cd_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_f3 = min(remaining * 0.5, 25.0)
            a_out, _ = nesterov_lp_optimize(best_seq, t_f3, solver, verbose=False, use_adaptive_momentum=False)
            try_update(a_out, "nlp_final")
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
