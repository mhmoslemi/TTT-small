def get_solver():

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve_fast(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        con = constraints[0]
        if con.dual_value is not None:
            if active is not None and 0 < len(active) < size:
                full_duals = np.zeros(size)
                full_duals[active] = np.asarray(con.dual_value, dtype=np.float64)
                duals = full_duals
            else:
                duals = np.asarray(con.dual_value, dtype=np.float64)
    except Exception:
        pass
    return val, duals

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(200.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)

                break

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)
                for trial in range(12):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)

                    partner = k - i

                # This is getting complex; use the FFT approach for exact computation

                        # Contribution from a[j] increasing by d
                        # conv[k] += d * (a[k-j] + a[j-k]) for valid k
                        # minus contribution from a[i] decreasing by d
                        # conv[k] -= d * (a[k-i] + a[i-k]) for valid k
                        # plus second order terms
                        0
                    )
                    # This is too complex to do analytically; let's just evaluate directly
                    a[i] -= d
                    a[j] += d
                    s_new = a.sum()
                    if s_new > 1e-15:
                        a = a * norm / s_new
                        new_max = fast_conv_self(a).max()
                    else:
                        new_max = cur_max
                    a[i] += d
                    a[j] -= d
                    a = a * norm / a.sum()

                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_pert = max(10, n // 25)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and no_improve % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def basin_hopping_lp(a0, time_limit, solver, rng, verbose=True):
    """Basin hopping with LP-based re-optimization after each perturbation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    for hop in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        # Perturb
        n_pert = max(20, n // 15)
        idx_p = rng.choice(n, size=n_pert, replace=False)
        scale = 0.15 + 0.1 * np.sqrt(hop) / max(1, np.sqrt(len(idx_p)))
        a[idx_p] *= (1.0 + scale * rng.standard_normal(n_pert))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * norm / s

        # Short LP Nesterov re-optimization
        t_sub = min(remaining * 0.6, 30.0)
        a_out, _ = adaptive_lp_nesterov(a, t_sub, solver, verbose=False)
        sc = fast_conv_self(a_out).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_out.copy()
            a = a_out.copy()
            if verbose:
                print(f"    BasinHop {hop} ({time.monotonic()-t0:.0f}s): NEW BEST {best_score:.8f}")
        else:
            # Keep the perturbed version for next hop (allows exploration)
            # but also try to snap back towards best
            a = 0.7 * best_a + 0.3 * a
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s

    return best_a, best_score

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        # Chirp-like
        a = 0.5 + 0.4 * np.cos(2 * np.pi * (5 * x**2 - 2 * x))
        a = np.maximum(a, 0.05)
    elif choice == 10:
        # Multi-scale oscillation
        a = 0.4 + 0.25 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 13 * x) + 0.1 * np.cos(2 * np.pi * 29 * x)
        a = np.maximum(a, 0.05)
    else:
        # Random with smoothing
        a = rng.random(n)
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(8):
            cand = structured_init(n_base, rng)
            sc = eval_fn(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick LSE to sharpen init
    remaining = time_budget - (time.monotonic() - t_global)
    t_init = min(remaining * 0.06, 90.0)
    if t_init > 15:
        a_out, _ = lse_gradient_descent(best_seq, t_init, verbose=False)
        try_update(a_out, "lse_init")

    # Phase 1: Main LP Nesterov (long run)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = adaptive_lp_nesterov(best_seq, t1, solver, verbose=True)
        try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Dual gradient refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Dual Gradient n={n2} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = dual_gradient_refine(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"dualgrad_n{n2}")

    # Phase 3: Basin hopping
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: Basin Hopping n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = basin_hopping_lp(best_seq, t3, solver, rng, verbose=True)
        try_update(a_out, f"basinhop_n{n3}")

    # Phase 4: Peak shaving CD
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: Peak Shaving CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = peak_shaving_cd(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"peakcd_n{n4}")

    # Phase 5: Final LP Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Final LP Nesterov n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = adaptive_lp_nesterov(best_seq, t5, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n5}")

    # Phase 6: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 6: Length extension n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120 and len(best_seq) < n_long:
        init_long = interpolate_seq(best_seq, n_long)
        t_long1 = min(remaining * 0.4, 150.0)
        a_out, _ = adaptive_lp_nesterov(init_long, t_long1, solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == n_long:
            t_long2 = min(remaining * 0.4, 80.0)
            a_out2, _ = peak_shaving_cd(best_seq, t_long2, rng, verbose=False)
            try_update(a_out2, f"peakcd_long_{n_long}")

    # Phase 7: Try a different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = 2000
    print(f"\n=== Phase 7: Alt length n={n_alt} (remaining={remaining:.0f}s) ===")
    if remaining > 100 and n_alt != len(best_seq):
        init_alt = interpolate_seq(best_seq, n_alt)
        t_alt = min(remaining * 0.5, 150.0)
        a_out, _ = adaptive_lp_nesterov(init_alt, t_alt, solver, verbose=False)
        try_update(a_out, f"lp_alt_{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t_alt2 = min(remaining * 0.5, 60.0)
            a_out2, _ = lse_gradient_descent(best_seq, t_alt2, verbose=False)
            try_update(a_out2, f"lse_alt_{n_alt}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
