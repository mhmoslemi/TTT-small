            a2 = a1
            fa2 = fa1
            a1 = hi - gr * (hi - lo)
            fa1 = obj(a1)
        else:
            lo = a1
            a1 = a2
            fa1 = fa2
            a2 = lo + gr * (hi - lo)
            fa2 = obj(a2)
    
    return (lo + hi) / 2.0, obj((lo+hi)/2)

def build_constraint_matrix(a, k_indices, n):
    size = len(k_indices)
    mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            mat[idx, j] = a[k - j]
    return mat


def lp_solve(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    if len(k_indices) == 0:
        return None, None
    T = build_constraint_matrix(a_fixed, k_indices, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp(a0, time_limit, solver, seed=0, verbose=False):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 50

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if active_set is None or t % 30 == 0:
            tol = max(0.005, 0.04 * np.exp(-t / 200.0))
            active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
            top_k = np.argsort(conv_y)[-max(5, n // 100):]
            active_set = np.unique(np.concatenate([active_set, top_k]))

        b, duals = lp_solve(y, M_curr, active_set, solver)

        if b is None:
            active_set = np.arange(2 * n - 1)
            b, duals = lp_solve(y, M_curr, active_set, solver)
            if b is None:
                if t % 5 == 0:
                    active_set = None
                continue

        # Verify and expand
        size = 2 * n - 1
        complement = np.setdiff1d(np.arange(size), active_set)
        if len(complement) > 0 and t % 10 == 0:
            T_comp = build_constraint_matrix(y, complement, n)
            violations = T_comp @ b
            viol_idx = np.where(violations > M_curr * (1 + 1e-5))[0]
            if len(viol_idx) > 0:
                new_active = np.union1d(active_set, complement[viol_idx])
                if len(new_active) > 0.85 * size:
                    new_active = np.arange(size)
                b_new, _ = lp_solve(y, M_curr, new_active, solver)
                if b_new is not None:
                    active_set = new_active
                    b = b_new

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.97)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                print(f"    LP restart at iter {t}, score={best_score:.10f}")

        if verbose and t % 500 == 0:
            print(f"    LP iter={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False, seed=42):
    """
    Aggressive coordinate descent with golden-section search.
    Highly vectorized inner loop; many passes possible.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    t0 = time.monotonic()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    S = a.sum()
    pass_count = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        S = a.sum()
        conv = fast_autocorr(a)
        curr_score = 2.0 * n * conv.max() / (S * S)
        improved = False

        # Randomize coordinate order
        order = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            j = order[idx]
            aj = a[j]

            # Compute cross-correlation c[k] = a[k-j] for valid k-j
            c = np.zeros(size)
            c[j:j + n] = a

            # Background: remove j's contribution
            bg = conv - 2.0 * aj * c
            if 2 * j < size:
                bg[2 * j] -= aj * aj

            # Golden section search on x in [0, S]
            # Objective: f(x) = max_k (bg[k] + 2*x*c[k] + x^2*delta_{k,2j}) / (S-aj+x)^2
            lo = 0.0
            hi = S
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)

            # Evaluate objective at a1
            nc1 = bg + 2.0 * a1 * c
            if 2 * j < size:
                nc1[2 * j] += a1 * a1
            sum1 = S - aj + a1
            fa1 = 2.0 * n * nc1.max() / (sum1 * sum1) if sum1 > 1e-15 else 1e18

            # Evaluate at a2
            nc2 = bg + 2.0 * a2 * c
            if 2 * j < size:
                nc2[2 * j] += a2 * a2
            sum2 = S - aj + a2
            fa2 = 2.0 * n * nc2.max() / (sum2 * sum2) if sum2 > 1e-15 else 1e18

            for _ in range(25):
                if fa1 < fa2:
                    hi = a2
                    a2 = a1
                    fa2 = fa1
                    a1 = hi - gr * (hi - lo)
                    nc1 = bg + 2.0 * a1 * c
                    if 2 * j < size:
                        nc1[2 * j] += a1 * a1
                    sum1 = S - aj + a1
                    fa1 = 2.0 * n * nc1.max() / (sum1 * sum1) if sum1 > 1e-15 else 1e18
                else:
                    lo = a1
                    a1 = a2
                    fa1 = fa2
                    a2 = lo + gr * (hi - lo)
                    nc2 = bg + 2.0 * a2 * c
                    if 2 * j < size:
                        nc2[2 * j] += a2 * a2
                    sum2 = S - aj + a2
                    fa2 = 2.0 * n * nc2.max() / (sum2 * sum2) if sum2 > 1e-15 else 1e18

            x_opt = (lo + hi) / 2.0
            if abs(x_opt - aj) < 1e-10:
                continue

            # Apply update
            a[j] = x_opt
            S_new = S - aj + x_opt
            if S_new < 1e-15:
                a[j] = aj
                continue
            a = a * norm / S_new
            S = S_new
            # Update conv incrementally: conv = bg + 2*x*c + x^2*delta
            conv = bg + 2.0 * x_opt * c
            if 2 * j < size:
                conv[2 * j] += x_opt * x_opt
            # Note: conv is now approximate (missing the renormalization effect on other coords)
            # We'll recompute at start of next pass, but for score check use approximate
            new_score_approx = 2.0 * n * conv.max() / (S * S)
            if new_score_approx < curr_score - 1e-12:
                improved = True
                if new_score_approx < best_score:
                    best_score = new_score_approx
                    best_a = a.copy()

        # End of pass: recompute exact score
        conv = fast_autocorr(a)
        exact_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if exact_score < best_score - 1e-13:
            best_score = exact_score
            best_a = a.copy()
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break

        if verbose and pass_count % 20 == 0:
            print(f"    CD pass={pass_count} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=False, seed=7):
    """
    Subgradient polishing using top active peaks with adaptive step size.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    max_patience = 1000
    step = 2.0 / n ** 0.5

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get top peaks with exponential weighting
        n_peaks = max(3, min(20, n // 80))
        top_indices = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Combined subgradient (vectorized)
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            # d(a*a)[k]/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
            idx1 = np.arange(max(0, k - n + 1), min(n, k + 1))
            vals1 = a[k - idx1]
            for j in idx1:
                grad[j] += w * vals1[j - idx1[0]]
            # Second part: a[j-k] for valid j-k
            idx2 = np.arange(max(0, k), min(n, k + n))
            # j-k ranges from k-n to k, valid when 0 <= j-k < n
            # j ranges from k to k+n-1, clipped to [0, n-1]
            j_start = max(0, k)
            j_end = min(n, k + n)
            if j_start < j_end:
                offset = j_start - k
                vals2 = a[offset:offset + (j_end - j_start)]
                grad[j_start:j_end] += w * vals2

        # Normalized step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 5.0)
        else:
            patience += 1
            step *= 0.99
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            print(f"    Subgrad t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate():
    """
    Main search: LP Nesterov + aggressive CD + subgradient + length exploration.
    """
    rng = np.random.default_rng(12345)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting with n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (quick warmup) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, seed=0, verbose=True)
        try_update(a_out, "lp1")

    # === Phase 2: Aggressive Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 20:
        print(f"\n=== Phase 2: CD n={n_cur}, budget={t2:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t2, verbose=True, seed=42)
        try_update(a_out, "cd1")

    # === Phase 3: LP Nesterov from CD result ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP Nesterov round 2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, seed=99, verbose=True)
        try_update(a_out, "lp2")

    # === Phase 4: More CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 20:
        print(f"\n=== Phase 4: CD round 2 n={n_cur}, budget={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t4, verbose=True, seed=777)
        try_update(a_out, "cd2")

    # === Phase 5: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Subgradient n={n_cur}, budget={t5:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t5, verbose=True, seed=55)
        try_update(a_out, "subg")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Length exploration, remaining={remaining:.0f}s ===")

    # Try nearby lengths
    for n_try in [n_cur - 100, n_cur + 100, n_cur - 200, n_cur + 200]:
        if n_try < 500 or n_try > 4000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        t_try = min(remaining * 0.25, 120.0)
        if t_try < 30:
            break

        a_interp = interpolate_to_length(best_seq, n_try)
        a_interp_norm = a_interp * np.sqrt(2.0 * n_try) / a_interp.sum()
        a_out, _ = coordinate_descent_fast(a_interp_norm, t_try, verbose=False, seed=n_try)
        try_update(a_out, f"len{n_try}")

    # === Phase 7: Perturbation + CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Perturbation restarts, remaining={remaining:.0f}s ===")
    for r in range(3):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_r = min(remaining * 0.2, 100.0)
        if t_r < 20:
            break
        scale = 0.015 + 0.01 * r
        pert = perturb_sequence(best_seq, rng, scale=scale)
        pert_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = coordinate_descent_fast(pert_norm, t_r, verbose=False, seed=500 + r)
        try_update(a_out, f"pert_{r}")

    # === Phase 8: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final CD n={n_cur}, remaining={remaining:.0f}s ===")
        t8 = remaining - 10
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_fast(a_norm, t8, verbose=True, seed=2024)
        try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
