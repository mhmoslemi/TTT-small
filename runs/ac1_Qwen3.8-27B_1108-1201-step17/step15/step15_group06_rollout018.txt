def build_active_matrix(a, active_idx):

def cross_corr_wa(w, a):
    """Compute cross-correlation d[i] = sum_k w[k]*a[k-i] using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_sz = 1 << (size - 1).bit_length()
    Wf = rfft(w, fft_sz)
    Af = rfft(a, fft_sz)
    # Cross-correlation: xcorr[i] = sum_k w[k] * a[k-i]
    # = irfft(conj(FFT(w)) * FFT(a)) with appropriate shifting
    # Actually: cross_corr = irfft(FFT(w) * conj(FFT(a))) gives sum_k w[k]*a[k+i-n]
    # Let me just use: the linear convolution of w and flip(a)
    a_flip = a[::-1]
    Wf = rfft(w, fft_sz)
    Afl = rfft(a_flip, fft_sz)
    result = irfft(Wf * Afl, fft_sz)
    # result[i] = sum_k w[k] * a_flip[i-k] = sum_k w[k] * a[n-1-(i-k)]
    # = sum_k w[k] * a[n-1-i+k]
    # We want d[i] = sum_k w[k] * a[k-i]
    # Let j = k-i, then k = i+j
    # d[i] = sum_j w[i+j] * a[j]
    # This is the (i)-th element of the auto-cross correlation
    # Using the relation: d = ifft(conj(FFT(a)) * FFT(w)) shifted
    # Let me just use a simpler formula:
    # d[i] = (w * a)[i] where * denotes cross-correlation
    # In FFT: cross_corr = ifft(FFT(w) * conj(FFT(a)))
    # This gives: result[m] = sum_n w[n] * a[n-m] (circular)
    # For linear: result[m] = sum_n w[n] * a[n-m] for valid n, n-m
    
    # OK this is getting complicated. Let me just do it the simple way for d:
    # d[i] = sum_{k=max(i,0)}^{min(len(w)-1, i+n-1)} w[k] * a[k-i]
    # = sum_{j=0}^{n-1} w[i+j] * a[j]  (where i+j < len(w))
    
    # This is the (i)-th element of the "valid" cross-correlation
    # scipy.signal.correlate(w, a, mode='full') gives this
    # But we want to use FFT for speed.
    
    # The full cross-correlation of w (length 2n-1) and a (length n) has length 3n-2
    # The "valid" part (what we want) is the central n elements.
    
    # Using FFT: cross_full = ifft(FFT(w_padded) * conj(FFT(a_padded)))
    # where we pad to avoid circular convolution artifacts
    total_len = len(w) + n - 1
    fft_total = 1 << (total_len - 1).bit_length()
    Wp = rfft(w, fft_total)
    Ap = rfft(a, fft_total)
    cross_full = irfft(Wp * np.conj(Ap), fft_total)
    # cross_full[m] = sum_n w[n] * a[n-m] (with zero padding)
    # We want d[i] = sum_k w[k] * a[k-i] = sum_j w[i+j] * a[j]
    # cross_full[m] = sum_n w[n] * a[n-m]
    # Let j = n-m, n = m+j: cross_full[m] = sum_j w[m+j] * a[j]
    # So cross_full[i] = sum_j w[i+j] * a[j] = d[i]!
    
    return cross_full[:n]


def mw_nesterov(a0, time_limit, verbose=True, seed=0):
    """Multiplicative weights with Nesterov acceleration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        w = conv_y / M
        d = cross_corr_wa(w, y)
        d = np.maximum(d, 0.0)
        max_d = d.max()
        if max_d < 1e-15:
            break

        eta = 0.7 / max_d
        a_new = y * (1.0 - eta * d)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new * norm / sn

        # Nesterov update
        alpha_t = max(2.0 / (t + 2.0), 0.15)
        y_new = (1.0 - alpha_t) * y + alpha_t * a_new
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy

        prev_y = y
        y = y_new

        # Nesterov momentum for tracking
        if t > 1:
            momentum = min((t - 1.0) / (t + 2.0), 0.90)
            a_momentum = y + momentum * (y - prev_y)
            a_momentum = np.maximum(a_momentum, 0.0)
            sm = a_momentum.sum()
            if sm < 1e-15:
                a_momentum = y.copy()
            else:
                a_momentum = a_momentum * norm / sm
        else:
            a_momentum = y.copy()

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_m = fast_autocorr(a_momentum).max()
        if score_m < best_score:
            best_score = score_m
            best_a = a_momentum.copy()
            stall_count = 0

        # Restart if stalled
        if stall_count > 150:
            y = best_a.copy()
            prev_y = best_a.copy()
            stall_count = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_peak_reduction(a0, time_limit, verbose=True, seed=0):
    """Direct subgradient method targeting the peak lag."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    prev_y = a.copy()
    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(y)
        M = conv.max()
        if M < 1e-12:
            break

        # Find all peak lags
        tol = 0.05
        peak_idx = np.where(conv >= M * (1.0 - tol))[0]
        k_star = int(peak_idx[np.argmax(conv[peak_idx])])

        # Subgradient of c_{k*}: g_j = 2*a_{k*-j} (for valid j)
        g = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                if j == k_j:
                    g[j] = 2.0 * a[j]
                else:
                    g[j] += a[k_j]
                    # Also from the other side: when k*-i = j, i = k*-j
                    # d(c)/da_j gets contribution from a_{k*-j} * da_j/da_j = a_{k*-j}
                    # Wait, let me recompute.
                    # c_k = sum_i a_i * a_{k-i}
                    # d(c_k)/da_j = a_{k-j} (from i=j term) + a_{k-j} (from k-i=j term, i=k-j)
                    # = 2*a_{k-j} (if j != k-j) or 2*a_j (if j = k-j)
                    pass
        
        # Redo gradient properly
        g = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                g[j] = 2.0 * y[kj] if j != kj else 2.0 * y[j]
        
        # Gradient step with projection
        # Project onto {x: sum(x) = norm, x >= 0}
        # Simple: a_new = y - eta * g, then clip and renormalize
        eta_base = 0.1 * norm / (np.max(np.abs(g)) + 1e-15)
        eta_t = eta_base * 2.0 / (2.0 + t)
        
        a_new = y - eta_t * g
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new * norm / sn

        # Nesterov
        alpha_t = max(2.0 / (t + 2.0), 0.2)
        y_new = (1.0 - alpha_t) * y + alpha_t * a_new
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy

        prev_y = y
        y = y_new

        if t > 1:
            momentum = min((t - 1.0) / (t + 2.0), 0.85)
            a_mom = y + momentum * (y - prev_y)
            a_mom = np.maximum(a_mom, 0.0)
            sm = a_mom.sum()
            if sm < 1e-15:
                a_mom = y.copy()
            else:
                a_mom = a_mom * norm / sm
        else:
            a_mom = y.copy()

        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        sc_m = fast_autocorr(a_mom).max()
        if sc_m < best_score:
            best_score = sc_m
            best_a = a_mom.copy()
            stall_count = 0

        if stall_count > 200:
            y = best_a.copy()
            prev_y = best_a.copy()
            stall_count = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SUB t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0, alpha_min=0.1, beta_blend=0.25):
    """Nesterov-accelerated LP iteration with dual LP blending."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 60

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set - tighter tolerance for better LP direction
        if active_set is None or t % 25 == 0:
            tol = 0.02 + 0.03 * np.exp(-t / 200.0)
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 10:
                active_set = np.where(conv_y >= M_curr * 0.6)[0]
            if len(active_set) > 0.7 * size:
                active_set = np.where(conv_y >= M_curr * 0.92)[0]

        # Max-sum LP (primary direction)
        b_max = lp_max_sum(y, M_curr, solver, active_set)

        # Min-max LP (secondary, for blending)
        b_min = None
        if t % 5 == 0:
            b_min, _ = lp_min_max(y, solver, active_set, normalize_sum=norm)

        if b_max is None:
            if t % 30 == 0:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
            if b_max is None:
                # Fallback: use MW step
                b_max = multiplicative_weights_step(y, normalize_target=norm)
                if b_max is None:
                    continue

        b_sum = b_max.sum()
        if b_sum < 1e-15:
            continue
        g = b_max * norm / b_sum

        if b_min is not None and b_min.sum() > 1e-10:
            g_min_norm = b_min * norm / b_min.sum()
            g = (1.0 - beta_blend) * g + beta_blend * g_min_norm

        # Verify constraint violations
        if active_set is not None and len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                T_check = build_active_matrix(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 1e-6))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        # Nesterov update with improved momentum
        alpha_t = max(2.0 / (t + 2.0), alpha_min)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.91)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_reset = max(40, stall_reset - 5)
            stall_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coord_descent(a0, time_limit, rng, verbose=True):
    """Coordinate descent with golden section search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    phi = 0.6180339887498949

    def obj_val(a_vec):
        c = fast_autocorr(a_vec)
        s = a_vec.sum()
        return 2.0 * n * c.max() / (s * s)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            lo = 0.0
            hi = ai * 3.5 + norm * 0.05

            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            a_test1 = a.copy()
            a_test1[i] = c1
            s1 = a_test1.sum()
            if s1 > 1e-15:
                a_test1 = a_test1 * norm / s1
                f1 = obj_val(a_test1)
            else:
                f1 = float('inf')

            a_test2 = a.copy()
            a_test2[i] = c2
            s2 = a_test2.sum()
            if s2 > 1e-15:
                a_test2 = a_test2 * norm / s2
                f2 = obj_val(a_test2)
            else:
                f2 = float('inf')

            for _ in range(25):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a_test1 = a.copy()
                    a_test1[i] = c1
                    s1 = a_test1.sum()
                    if s1 > 1e-15:
                        a_test1 = a_test1 * norm / s1
                        f1 = obj_val(a_test1)
                    else:
                        f1 = float('inf')
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a_test2 = a.copy()
                    a_test2[i] = c2
                    s2 = a_test2.sum()
                    if s2 > 1e-15:
                        a_test2 = a_test2 * norm / s2
                        f2 = obj_val(a_test2)
                    else:
                        f2 = float('inf')

            optimal_val = min(f1, f2)
            a_test_opt = a.copy()
            a_test_opt[i] = c1 if f1 < f2 else c2
            s_opt = a_test_opt.sum()
            if s_opt > 1e-15:
                a_test_opt = a_test_opt * norm / s_opt
                sc = obj_val(a_test_opt)
                if sc < best_score - 1e-12:
                    a = a_test_opt
                    conv = fast_autocorr(a)
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s
