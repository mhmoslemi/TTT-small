    Compute c[j] = sum_k w[k] * a[k-j] for j=0..n-1
    where w has length 2n-1 and a has length n.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s == 0:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad(z, T):
    """
    Compute the soft-max smoothed objective and gradient w.r.t. z.
    Objective: (2n/T) * log(sum_k exp(T * b_k)) where b = conv(a,a), a = softmax(z)
    This equals 2n * soft_max_T(b), which upper-bounds 2n * max(b).
    """
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Weights
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a = 2n * cross_corr(w, a)
    grad_a = 2.0 * n * cross_corr_valid(w, a)
    
    # Chain rule for softmax: grad_z = a * (grad_a - a * (a . grad_a))
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def run_lbgfs(z0, T, n, time_limit, maxiter=5000):
    """Run L-BFGS with time limit."""
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_obj_grad(z, T)[0]
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_obj_grad(z, T)[1]
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        return res.x, res.fun, res
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return z0, func(z0), None


def subgradient_polish(a0, time_limit, lr_init=0.002, momentum=0.9, top_k=20):
    """
    Subgradient method using the exact max.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Find top-k peaks
        k = min(top_k, 2*n-1)
        peak_idx = np.argpartition(b, -k)[-k:]
        peak_vals = b[peak_idx]
        
        # Soft weights (temperature based on spread)
        spread = peak_vals.max() - peak_vals.min()
        if spread < 1e-15:
            w = np.ones(k) / k
        else:
            temp = spread * 0.1
            scaled = (peak_vals - peak_vals.max()) / temp
            w = np.exp(scaled)
            w /= w.sum()
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(k):
            if w[i] < 1e-14:
                continue
            pk = int(peak_idx[i])
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * n * w[i] * a[pk - idx]
        
        # Update with momentum and simplex projection
        v = momentum * v - lr * sg
        a_new = a + v
        
        # Project to simplex
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        
        # Evaluate
        b_new = fast_convolve(a_new)
        obj_new = 2.0 * n * b_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
        
        lr *= 0.998
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a /= a.sum()
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def active_set_newton(a0, max_iter=100, tol=1e-10):
    """
    Newton's method on the active set of peaks.
    Given the active peaks, solve the KKT system.
    """
    a = a0.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    
    for it in range(max_iter):
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active peaks (within tolerance of max)
        tol_act = max_b * 0.001 + 1e-15
        active = np.where(b >= max_b - tol_act)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > 50:
            active = np.where(b >= max_b - max_b * 0.01)[0]
        
        m = len(active)
        
        # Build the system: sum_i lambda_i * a_{k_i - j} = mu for j in support
        # This is a (m+1) x (m+1) system
        # Variables: lambda_1, ..., lambda_m, mu
        # Equations: for each active peak k_i: sum_j a_j * a_{k_i - j} = t = max_b
        #   (but this is automatically satisfied if we're at the max)
        # Actually, let me use a different formulation.
        
        # The KKT conditions for minimizing max(b) subject to sum(a)=1:
        # At the optimum, for active peaks k_i with multiplier lambda_i > 0:
        #   sum_i lambda_i * grad_{a_j}(b_{k_i}) = mu for all j with a_j > 0
        #   sum_i lambda_i = 1
        #   b_{k_i} = max_b for active i
        #   b_k <= max_b for inactive k
        
        # grad_{a_j}(b_k) = a_{k-j} (for valid j)
        # So: sum_i lambda_i * a_{k_i - j} = mu for all j in support
        
        # This is a linear system. Let's build it:
        # For j in support: sum_i lambda_i * a[max(0, min(n-1, k_i - j))] = mu
        # Plus: sum_i lambda_i = 1
        
        # Build matrix A_sys:
        # A_sys[i, j] = a[k_i - j] for i=0..m-1, j=0..n-1 (for valid indices)
        # A_sys[m, j] = 1 for j=0..m (for the sum constraint)
        
        # Actually this is an (m+1) x (m+1) system if we have m active peaks.
        # Wait, we have n equations (one per j in support) plus 1 (sum of lambdas = 1).
        # But we have m+1 unknowns (lambda_1..lambda_m, mu).
        # If the support is all n elements, we have n+1 equations and m+1 unknowns.
        # This is overdetermined unless m+1 = n+1, i.e., m = n.
        
        # So this approach doesn't directly work unless m = n. Let me use a least-squares approach.
        
        # Build the system: for all j (0 to n-1):
        # sum_i lambda_i * a_{k_i - j} - mu = 0
        # And: sum_i lambda_i = 1
        
        # This is (n+1) equations, (m+1) unknowns.
        # Use least squares.
        
        k_active = active.astype(int)
        
        # A is (n+1) x (m+1)
        A_sys = np.zeros((n + 1, m + 1))
        b_sys = np.zeros(n + 1)
        b_sys[m] = 1.0  # sum of lambdas = 1... wait, b_sys should be the RHS
        
        # Actually: sum_i lambda_i * a_{k_i - j} = mu for each j
        # => sum_i lambda_i * a_{k_i - j} - mu = 0
        # So the system is: [A_block | -1] * [lambda; mu] = 0
        # And: [1...1 | 0] * [lambda; mu] = 1
        
        for i in range(m):
            ki = k_active[i]
            for j in range(n):
                k_minus_j = ki - j
                if 0 <= k_minus_j < n:
                    A_sys[j, i] = a[k_minus_j]
        
        A_sys[:n, m] = -1.0  # coefficient of mu
        A_sys[n, :m] = 1.0   # sum of lambdas
        A_sys[n, m] = 0.0
        
        b_sys[:n] = 0.0
        b_sys[n] = 1.0
        
        # Solve least squares (since n+1 > m+1 in general)
        try:
            sol, residuals, rank, sv = np.linalg.lstsq(A_sys, b_sys, rcond=None)
        except:
            break
        
        lambda_vals = sol[:m]
        mu_val = sol[m]
        
        # Check if solution is valid (lambdas should be non-negative)
        if np.any(lambda_vals < -1e-10):
            # Negative lambda means this peak shouldn't be active
            # Remove the peak with the most negative lambda
            worst = np.argmin(lambda_vals)
            active = np.delete(active, worst)
            continue
        
        # Now use this to compute a Newton step
        # The idea: we want to change a to reduce max_b while maintaining optimality
        # Use the second-order info from the active constraints
        
        # Actually, let me use a simpler approach: use the lambda values as weights
        # and do a gradient-like step
        # grad = sum_i lambda_i * grad(b_{k_i}) = sum_i lambda_i * a_{k_i - j}
        # But this should equal mu (by KKT), so it's uniform.
        # This means we're at a stationary point of the Lagrangian.
        # To make progress, we need to move in a direction that reduces the max.
        
        # Alternative: directly minimize a quadratic model of max_b
        # Around the current point, max_b ≈ max_b + sum_j grad_j * delta_j + 0.5 * delta^T H delta
        # where grad_j = a_{k* - j} for the peak k*
        # This is getting too complex. Let me just use the subgradient approach.
        
        break
    
    return best_a, best_m

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_valid(w, a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()

    return exp_z / s
