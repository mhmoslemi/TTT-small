def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        ws = w.sum()
        if ws < 1e-300:
            w_arr = np.zeros(len(conv_y))
            w_arr[np.argmax(conv_y)] = 1.0
        else:
            w_arr = w / ws
        
        if top_k is not None and top_k < len(w_arr):
            idx = np.argpartition(w_arr, -top_k)[-top_k:]
            w_masked = np.zeros(len(w_arr))
            w_masked[idx] = w_arr[idx]
            w_masked /= w_masked.sum()
            w_arr = w_masked
        
        sg = compute_weighted_subgradient(y, w_arr)
        
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adam step
        step = cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = f - step
        f = project_simplex(f_new)
        
        # Update Nesterov look-ahead
        y = f + nesterov * (f - (f + step))  # y = f - nesterov * step
        # Actually standard Nesterov: y = f + momentum*(f - f_prev)
        # Simplified: y = f + nesterov * (f_old - f) where f_old was before update
        # Let me just use: y = f (no look-ahead) as fallback if it causes issues
        # Better: y = f + nesterov * (f - f_prev)
        
        temp = max(temp * temp_decay, temp_min)
        cur_lr = lr * (0.9999 ** t)
        cur_lr = max(cur_lr, lr * 0.1)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 200 == 0:
                print(f"    Adam-N iter {t}: obj={best_obj:.10f}, T={temp:.6f}")
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.002 + 0.002 * min(no_improve // 100, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            y = f.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * (0.5 + 0.3 * np.random.random()), 0.002)
            cur_lr = lr
            no_improve = 0
    
    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.02, lr_decay=0.9995,
                            temp_init=0.008, temp_decay=0.9995,
                            temp_min=1e-5, top_k=None, verbose=False):
    """Mirror descent (multiplicative weights update)."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    temp = temp_init
    cur_lr = lr_init
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w_arr = np.zeros(len(conv))
            w_arr[np.argmax(conv)] = 1.0
        else:
            w_arr = w / ws
        
        if top_k is not None and top_k < len(w_arr):
            idx = np.argpartition(w_arr, -top_k)[-top_k:]
            w_masked = np.zeros(len(w_arr))
            w_masked[idx] = w_arr[idx]
            w_masked /= w_masked.sum()
            w_arr = w_masked
        
        sg = compute_weighted_subgradient(f, w_arr)
        
        # Multiplicative update: f_new = f * exp(-lr * sg), then normalize
        log_update = -cur_lr * sg
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new
        
        temp = max(temp * temp_decay, temp_min)
        cur_lr *= lr_decay
        cur_lr = max(cur_lr, lr_init * 0.05)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 200 == 0:
                print(f"    Mirror iter {t}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        if no_improve > 150:
            rng = np.random.default_rng(t * 23 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            temp = max(temp_init * 0.6, 0.003)
            cur_lr = lr_init
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            multipliers = [0.4, 0.6, 0.8, 1.2, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_u_shape_init(n, depth=1.0):
    """U-shaped initialization (more mass at edges)."""
    x = np.linspace(0, 1, n)
    f = 1.0 + depth * (2 * x - 1) ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_w_shape_init(n, depth=1.0):
    """W-shaped initialization."""
    x = np.linspace(0, 1, n)
    f = 1.0 + depth * (2 * x - 1) ** 4
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'flat':
        f = np.ones(n)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Smooth random initialization as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()
