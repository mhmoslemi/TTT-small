import numpy as np
import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def flat_spectrum_init(n, rng):
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

        a0 = a  # use current a for linearization

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using ADMM optimization with
QP sub-solves, multiple initializations, and diverse sequence lengths.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    a = rng.exponential(1.0, n)
    return a / a.sum()


def auto_correlation(a):
    """Compute linear auto-correlation r[m] = sum_l a[l]*a[l+m] for m=0..n-1."""
    n = len(a)
    N = 2 * n - 1
    A = np.fft.rfft(a, N)
    ac = np.fft.irfft(A * np.conj(A), N)
    return ac[:n]


def solve_qp_simplex(P, q, n, fallback=None):
    """Solve min a^T P a - q^T a s.t. a >= 0, sum(a) = 1 using CVXPY."""
    a_var = cp.Variable(n)
    obj = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(P)) - q @ a_var)
    constraints = [a_var >= 0, cp.sum(a_var) == 1]
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            if fallback is not None:
                return fallback
            return np.ones(n) / n
    if a_var.value is not None:
        result = np.maximum(a_var.value, 0)
        s = result.sum()
        if s > 1e-10:
            return result / s
    if fallback is not None:
        return fallback
    return np.ones(n) / n

def fista_qp_update(P, q, a_init, n_inner=50):
    """FISTA for min a^T P a - q^T a s.t. a >= 0, sum(a) = 1."""
    n = len(a_init)
    a = a_init.copy()
    # Compute Lipschitz constant (use power iteration for safety)
    # L = 2*lambda_max(P)
    # For speed, use Gershgorin bound or just a conservative estimate
    max_row_sum = np.max(np.abs(P).sum(axis=1))
    L = 2.0 * max_row_sum + 1e-6
    eta = 1.0 / L

    y = a.copy()
    t_k = 1.0
    for k in range(n_inner):
        # Gradient
        grad = 2.0 * (P @ y) - q
        # Gradient step
        z = y - eta * grad
        # Project onto simplex
        a_new = project_simplex(z, 1.0)
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_k = t_new

    return a
