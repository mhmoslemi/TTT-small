"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth convex optimization
(log-sum-exp + L-BFGS-B with softmax parameterization), combined with
mirror descent polishing, across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_sequence(a_list):
    # This is provided externally, but we also compute locally for speed
    a = np.asarray(a_list, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return float(2 * n * conv.max() / (s * s))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_objective(x, beta, n):
    """Log-sum-exp of auto-convolution, via softmax parameterization."""
    a = softmax(x)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    return np.log(np.exp(scaled).sum()) / beta


def lse_gradient(x, beta, n):
    """Gradient of log-sum-exp objective w.r.t. x (softmax params)."""
    a = softmax(x)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # grad_a[i] = 2 * sum_j a[j] * w[i+j] = 2 * correlate(w, a, 'valid')[i]
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    
    # Chain rule through softmax: dx = a * (grad_a - <a, grad_a>)
    dot = np.dot(a, grad_a)
    dx = a * (grad_a - dot)
    return dx


def lse_optimize(a0, time_limit, betas=None, verbose=False):
    """
    Optimize using log-sum-exp approximation with L-BFGS-B.
    Uses softmax parameterization to handle constraints.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    if betas is None:
        betas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
    
    x = np.log(np.maximum(a, 1e-30))
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Limit iterations based on remaining time
        remaining = time_limit - (time.monotonic() - t0)
        maxiter = min(500, max(50, int(remaining * 20)))
        
        try:
            res = minimize(
                lse_objective, x,
                args=(beta, n),
                jac=lambda x_: lse_gradient(x_, beta, n),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-14}
            )
            x = res.x
        except Exception:
            break
        
        a_new = softmax(x)
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    LSE beta={beta}: obj={best_obj:.10f}")
    
    return best_a, best_obj


def mirror_polish(a0, time_limit, verbose=False):
    """
    Mirror descent with negative entropy for polishing.
    Uses adaptive step size and multiple peak tracking.
    """
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 0.5 / np.sqrt(n)  # Initial step size
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Track top peaks with softmax weighting
        k_top = min(30, len(conv))
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peak_idx]
        # Use temperature that adapts
        temp = (pv.max() - pv.min()) / max(1, k_top * 0.3)
        if temp < 1e-15:
            temp = pv.max() * 0.01 + 1e-15
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            # grad[j] += 2 * w[i] * a[k-j] (if 0 <= k-j < n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]
        
        # Multiplicative (mirror) update
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay step size
        lr *= 0.9995
        
        if no_improve > 200:
            break
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit, verbose=False):
    """
    Projected subgradient method with Polyak step size.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Lower bound: 1/(2n-1) (uniform auto-convolution)
    lower_bound = 1.0 / (2 * n - 1)
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Subgradient at the peak
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_max - idx]
        
        # Polyak step size
        gap = M - lower_bound
        if gap < 1e-12:
            break
        # ||grad||^2
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-30:
            break
        step = gap / g_norm_sq
        step = min(step, 0.5)  # Cap step size
        
        # Projected subgradient step
        a_new = a - step * grad
        a_new = project_simplex(a_new)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose and it % 100 == 0:
                print(f"    SubGrad iter {it}: obj={best_obj:.10f}")
    
    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.35):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_optimal_shape(n):
    """
    Try to create a sequence whose auto-convolution is close to uniform.
    The ideal is a sequence whose DFT has constant magnitude (chirp-like).
    But for non-negative sequences, we approximate with a raised cosine window.
    """
    x = np.arange(n) / n
    # Raised cosine (Hann window)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, frac=0.7):
    """Flat-top window shape."""
    x = np.arange(n) / n
    a = np.ones(n)
    # Apply gentle rolloff at edges
    half = n // 2
    for i in range(half):
        frac_edge = i / half
        if frac_edge < (1 - frac):
            a[i] = 0.5 * (1 + np.cos(np.pi * frac_edge / (1 - frac)))
            a[n-1-i] = a[i]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

Let me also add a different technique: **conditional gradient (Frank-Wolfe) without LP**. Instead of solving an LP at each step, we can compute the "most violated constraint" direction greedily.

The Frank-Wolfe step requires finding:
`g = argmin_{x in simplex} <s, x>` where s is the subgradient.

The solution is `g = e_{argmin_j s_j}` (a vertex of the simplex). But this gives a very coarse direction. We can improve by using a "shrinkage" step:
`g_shrunk = project_simplex(a - alpha * s)`

Or we can use the "strongly separable oracle" which for the simplex gives a vertex.

Actually, the standard FW on the simplex is to pick `g = e_{min}` (the basis vector at the minimum component of s). Then:
`a_new = (1 - gamma) * a + gamma * g`

This is quite simple but can be slow to converge. The "accelerated" version (Starcraft) uses momentum.

Let me add a simple FW without LP as an additional method:

        je = min(n - 1, k_max)

        grad[idx] = 2.0 * a[k_max - idx]
        
        # Find best pairwise swap: move mass from j (high grad) to k (low grad)
        j_max = int(np.argmax(grad))
        j_min = int(np.argmin(grad))

        best_obj_swap = M

            if alpha >= a[j_max]:
                continue
            a_trial = a.copy()
            a_trial[j_max] -= alpha
            a_trial[j_min] += alpha
            a_trial = a_trial / a_trial.sum()
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_obj_swap:
                best_obj_swap = obj_trial

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth convex optimization
(log-sum-exp + L-BFGS-B with softmax parameterization), combined with
mirror descent polishing, subgradient methods, and multi-length exploration.
Key innovation: uses differentiable log-sum-exp surrogate optimized via
L-BFGS-B in a softmax parameterization, avoiding expensive LP solves.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_objective(x, beta, n):
    """Log-sum-exp of auto-convolution coefficients, via softmax params."""
    a = softmax_fn(x)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    return np.log(np.exp(scaled).sum()) / beta


def lse_gradient(x, beta, n):
    """Gradient of log-sum-exp objective w.r.t. softmax parameters x."""
    a = softmax_fn(x)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # grad_a[i] = 2 * sum_j a[j] * w[i+j]
    # This is 2 * correlate(w, a, 'valid')
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    # Chain rule through softmax: dx = a * (grad_a - <a, grad_a>)
    dot = np.dot(a, grad_a)
    dx = a * (grad_a - dot)
    return dx


def hard_max_objective(x, beta, n):
    """Hard max of auto-convolution (non-smooth, used with subgradient)."""
    a = softmax_fn(x)
    conv = fast_convolve(a)
    return conv.max()

def mirror_polish(a0, time_limit, verbose=False):
    """
    Mirror descent with negative entropy (multiplicative weights) for polishing.
    Tracks multiple peaks with adaptive softmax weighting.
    """
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = 0.8 / np.sqrt(n)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Track top peaks with softmax weighting
        k_top = min(40, len(conv))
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peak_idx]
        temp = (pv.max() - pv.min()) / max(1, k_top * 0.25)
        if temp < 1e-15:
            temp = pv.max() * 0.01 + 1e-15
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]

        # Multiplicative (mirror) update
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay step size
        lr *= 0.9992

        if no_improve > 500 or lr < 1e-6:
            break

    return best_a, best_obj


def subgradient_polish(a0, time_limit, verbose=False):
    """
    Projected subgradient method with Polyak step size.
    Uses the known lower bound 1/(2n-1) for gap estimation.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # Lower bound: uniform auto-convolution gives 1/(2n-1)
    lower_bound = 1.0 / (2 * n - 1)

    t0 = time.monotonic()
    it = 0
    avg_gap = best_obj - lower_bound
    alpha = 0.1

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Subgradient at the peak
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_max - idx]

        # Polyak step size with adaptive alpha
        gap = M - lower_bound
        if gap < 1e-13:
            break
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-30:
            break
        step = alpha * gap / g_norm_sq
        step = min(step, 0.3)

        # Projected subgradient step
        a_new = a - step * grad
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 200 == 0:
                print(f"    SubGrad iter {it}: obj={best_obj:.10f}")

        # Adapt alpha
        alpha = max(0.05, alpha * 0.9999)

    return best_a, best_obj


def fw_pairwise_swap(a0, time_limit, verbose=False):
    """
    Frank-Wolfe with pairwise coordinate swaps.
    Moves mass from high-gradient to low-gradient coordinates.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        k_max = int(np.argmax(conv))

        # Subgradient
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_max - idx]

        # Find best swap pair
        j_out = int(np.argmax(grad))
        j_in = int(np.argmin(grad))

        if j_out == j_in:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Try multiple swap amounts
        best_alpha = 0
        best_swap_obj = M
        max_alpha = min(a[j_out] * 0.9, 0.1)

        for alpha_frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            alpha = max_alpha * alpha_frac
            if alpha < 1e-12:
                continue
            a_trial = a.copy()
            a_trial[j_out] -= alpha
            a_trial[j_in] += alpha
            a_trial = a_trial / a_trial.sum()
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_swap_obj:
                best_swap_obj = obj_trial
                best_alpha = alpha

        if best_alpha > 1e-12:
            a[j_out] -= best_alpha
            a[j_in] += best_alpha
            a = a / a.sum()
            if best_swap_obj < best_obj:
                best_obj = best_swap_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Try random swaps
            improved = False
            for _ in range(20):
                j1 = rng.integers(n)
                j2 = rng.integers(n)
                if j1 == j2 or a[j1] < 1e-10:
                    continue
                direction = 1 if a[j1] > a[j2] else -1
                alpha = min(a[j1] * 0.1, 0.02)
                a_trial = a.copy()
                a_trial[j1] -= direction * alpha
                a_trial[j2] += direction * alpha
                if a_trial[j1] < -1e-12 or a_trial[j2] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_obj - 1e-13:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        if no_improve > 100:
            break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.35):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    """Hann window (raised cosine) shape."""
    x = np.arange(n) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, rolloff_frac=0.15):
    """Flat-top with gentle cosine rolloff at edges."""
    a = np.ones(n)
    edge_len = int(n * rolloff_frac)
    for i in range(edge_len):
        t = i / max(1, edge_len)
        val = 0.5 * (1 + np.cos(np.pi * t))
        a[i] = val
        a[n - 1 - i] = val
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_optimal_guess(n):
    """
    Initial guess based on the insight that we want the auto-convolution
    to be as flat as possible. The ideal sequence has a DFT with constant
    magnitude (like a chirp), but for non-negative sequences, a broad
    smooth window is a good approximation.
    """
    x = np.arange(n) / n
    # Use a super-Gaussian (flatter top)
    a = np.exp(-4.0 * np.abs(x - 0.5) ** 4)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
