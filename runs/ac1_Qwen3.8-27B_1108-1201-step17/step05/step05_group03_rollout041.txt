def conv_grad(a, weights):
    """Compute grad_i = 2 * sum_k weights[k] * a[k-i] using correlate."""
    # weights has length 2n-1, a has length n
    # result[i] = sum_j weights[i+j] * a[j], length n

    return 2.0 * C


def smooth_obj_and_grad_x(x, temp):

    a = softmax(x)
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()

    tb -= tb.max()
    exp_tb = np.exp(tb)
    sum_exp = exp_tb.sum()
    w = exp_tb / sum_exp
    smooth_val = b_max + np.log(sum_exp) / temp
    grad_a = conv_grad(a, w)
    mean_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_grad)

def mirror_descent(a, time_limit, temp=20.0, eta0=0.3, decay=0.9995,
                   restart_interval=2000, verbose=False):
    """Mirror descent with multiplicative updates for flattening."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = compute_score(a)
    eta = eta0
    t0 = time.monotonic()
    step = 0
    best_step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        b_max = b.max()
        # Softmax weights on peaks

        tb -= tb.max()

        a = a_new

            best_a = a.copy()

def kkt_iterate(a, time_limit, eps=0.05, max_iter=500, verbose=False):
    """KKT-inspired iteration: solve (F*a)_i = mu for active peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for iteration in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_max = b.max()

        # Active set: peaks above (1-eps) * max
        threshold = b_max * (1.0 - eps)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Uniform weights on active set
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        # Solve: (w * a)_i = mu for all i (approximately)
        # (w * a)_i = sum_j w[i+j] * a[j] = C_i (from conv_grad / 2)
        # We want C_i to be as flat as possible.
        # Use multiplicative update: a_i <- a_i * (mu / C_i)
        C = np.correlate(w, a, mode='valid')
        C = np.maximum(C, 1e-15)
        mu = np.mean(C)
        # Damping
        damping = 0.5
        a_new = a * (mu / C) ** damping
        a_new = normalize(a_new)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try smaller step
            a_try = a * (mu / C) ** (damping * 0.1)
            a_try = normalize(a_try)
            sc_try = compute_score(a_try)
            if sc_try < best_score:
                best_score = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
            else:
                # Shrink eps (tighten active set)
                eps *= 0.9
                if eps < 1e-6:
                    break

        if verbose and iteration % 50 == 0:
            print(f"      KKT iter {iteration}: score={best_score:.10f}, |S|={len(active)}, eps={eps:.4f}")

    return best_a, best_score


def lbfgs_optimize(a_init, time_limit, temp_start=2.0, temp_end=80.0,

    t0 = time.monotonic()

    phase_temps = np.logspace(np.log10(temp_start), np.log10(temp_end), n_phases)
    time_per_phase = time_limit / n_phases

    for phase, temp in enumerate(phase_temps):
        phase_start = time.monotonic()

def coord_descent_local(a, time_limit, n_rounds=50, verbose=False):

            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            # Try several transfer amounts
            for f in [0.05, 0.1, 0.2, 0.33, 0.5]:
                for direction in [1, -1]:

                    new_a = a.copy()
                    new_a[j] += delta
                    new_a[i] -= delta

                        best_a = new_a.copy()
                        a = new_a.copy()

    c2 = center2 * (n - 1)
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    return normalize(a)


def make_u_shaped(n, sharpness=2.0):
    """U-shaped: high at edges, low in middle."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return normalize(a)


def make_inverted_u(n, sharpness=2.0):
    """Inverted U: high in middle, low at edges."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * (1.0 - x ** 2)
    return normalize(a)


def make_sin_mod(n, freq, amp, phase=0.0):
    """Uniform with sinusoidal modulation: (1 + amp*cos(2*pi*freq*i/n + phase))."""
    i = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.01)  # ensure positivity
    return normalize(a)


def make_power_law(n, alpha):
    """Power law decay: (i+1)^(-alpha)."""
    i = np.arange(1, n + 1)
    a = i ** (-alpha)
    return normalize(a)


def make_two_block(n, frac1, frac2, gap_frac):
    """Two blocks of uniform mass with a gap."""
    a = np.zeros(n)
    w1 = int(n * frac1)
    w2 = int(n * frac2)
    gap = int(n * gap_frac)
    a[:w1] = 1.0
    a[n - w2:] = 1.0
    return normalize(a)


def generate_initializations(n, rng):
    """Generate diverse initializations for given length n."""
    inits = []
    labels = []

    # Uniform
    inits.append(make_uniform(n))
    labels.append(f"uniform_n{n}")

    # Geometric
    for r in [0.99, 0.995, 0.999, 0.9995]:
        inits.append(make_geometric(n, r))
        labels.append(f"geom_r{r}_n{n}")

    # Gaussian (centered)
    for sf in [0.2, 0.3, 0.4, 0.5, 0.7]:
        inits.append(make_gaussian(n, 0.5, sf))
        labels.append(f"gauss_sf{sf}_n{n}")

    # Bimodal
    for c1, c2 in [(0.25, 0.75), (0.2, 0.8), (0.3, 0.7)]:
        for sf in [0.1, 0.2, 0.3]:
            inits.append(make_bimodal(n, c1, c2, sf))
            labels.append(f"bi_c{c1}_{c2}_sf{sf}_n{n}")

    # U-shaped
    for s in [0.5, 1.0, 2.0, 4.0]:
        inits.append(make_u_shaped(n, s))
        labels.append(f"ushape_s{s}_n{n}")

    # Inverted U
    for s in [1.0, 2.0, 4.0]:
        inits.append(make_inverted_u(n, s))
        labels.append(f"invu_s{s}_n{n}")

    # Sinusoidal modulation (KEY NEW IDEA)
    for freq in [1, 2, 3, 4, 5, 6, 8, 12]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            inits.append(make_sin_mod(n, freq, amp))
            labels.append(f"sin_f{freq}_a{amp}_n{n}")

    # Power law
    for alpha in [0.3, 0.5, 0.7, 1.0]:
        inits.append(make_power_law(n, alpha))
        labels.append(f"pow_a{alpha}_n{n}")

    # Two block
    for f1, f2, gap in [(0.2, 0.2, 0.2), (0.3, 0.3, 0.1), (0.25, 0.25, 0.25)]:
        inits.append(make_two_block(n, f1, f2, gap))
        labels.append(f"2block_{f1}_{f2}_{gap}_n{n}")

    # Random smooth
    for _ in range(3):
        a = rng.random(n)
        # Smooth with a kernel
        k = max(3, n // 30)
        kernel = np.exp(-0.5 * (np.arange(-k, k+1) / (k*0.5))**2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        inits.append(normalize(a))
        labels.append(f"rand_smooth_n{n}")

    return inits, labels


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining diverse initialization, mirror descent,
    KKT iteration, L-BFGS-B with softmax parameterization, and coordinate descent.
    Explores multiple sequence lengths and structured starting points.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    # === Phase 0: Quick evaluation of diverse inits at multiple lengths ===
    print("=== Phase 0: Evaluating diverse initializations ===")
    lengths = [512, 768, 1024, 1536]
    all_candidates = []  # (score, a, label)

    for n in lengths:
        rem = deadline - time.monotonic() - 120
        if rem < 30:
            break
        inits, labels = generate_initializations(n, rng)
        for a, label in zip(inits, labels):
            sc = compute_score(a)
            all_candidates.append((sc, a, label))
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  NEW BEST [{label}]: {score:.8f} (n={n})" if False else
                      f"  NEW BEST [{label}]: {sc:.8f} (n={n})")

    # Add parent if available
    if parent is not None:
        all_candidates.append((compute_score(parent), parent, "parent"))

    # Sort by score
    all_candidates.sort(key=lambda x: x[0])
    print(f"  Top 5 initializations:")
    for i, (sc, a, label) in enumerate(all_candidates[:5]):
        print(f"    #{i+1} [{label}]: {sc:.8f}")

    # === Phase 1: Optimize top diverse candidates with mirror descent ===
    print(f"\n=== Phase 1: Mirror descent optimization ===")

    # Select diverse top candidates
    selected = []
    used_base = set()
    for sc, a, label in all_candidates:
        if len(selected) >= 8:
            break
        # Extract base type for diversity
        parts = label.split('_')
        base = parts[0] if parts else label
        if base not in used_base:
            selected.append((a, label, sc))
            used_base.add(base)

    # Ensure we include the overall best
    if all_candidates[0][1] not in [s[0] for s in selected]:
        selected.append((all_candidates[0][1], all_candidates[0][2], all_candidates[0][0]))

    n_sel = len(selected)
    t_per_md = min((deadline - time.monotonic() - 600) / max(1, n_sel), 120)

    for a_init, label, sc_init in selected:
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_alloc = min(t_per_md, rem)
        print(f"  MD [{label}]: t={t_alloc:.0f}s, init_score={sc_init:.8f}")

        # Run mirror descent with different parameters
        a_md, sc_md = mirror_descent(a_init, time_limit=t_alloc * 0.6,
                                     temp=20.0, eta0=0.3, decay=0.9998)
        try_seq(a_md, label + "_md")

        # Also try with different temperature
        rem2 = deadline - time.monotonic() - 350
        if rem2 > 10:
            a_md2, sc_md2 = mirror_descent(a_init, time_limit=min(rem2 * 0.5, 60),
                                           temp=5.0, eta0=0.1, decay=0.999)
            try_seq(a_md2, label + "_md_soft")

    # === Phase 2: KKT iteration on best ===
    print(f"\n=== Phase 2: KKT iteration (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            a_kkt, sc_kkt = kkt_iterate(best_a, time_limit=min(rem * 0.5, 120),
                                        eps=0.05, verbose=True)
            try_seq(a_kkt, "kkt")

            # Also try with tighter eps
            rem2 = deadline - time.monotonic() - 250
            if rem2 > 15:
                a_kkt2, sc_kkt2 = kkt_iterate(best_a, time_limit=min(rem2 * 0.5, 80),
                                              eps=0.02, verbose=True)
                try_seq(a_kkt2, "kkt_tight")

    # === Phase 3: L-BFGS-B with softmax parameterization ===
    print(f"\n=== Phase 3: L-BFGS-B optimization (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            n_best = len(best_a)
            a_lb, sc_lb = lbfgs_optimize(best_a, time_limit=min(rem * 0.5, 180),
                                         temp_start=1.0, temp_end=60.0,
                                         verbose=True)
            try_seq(a_lb, "lbfgs")

            # Also try from a different start
            rem2 = deadline - time.monotonic() - 150
            if rem2 > 20 and len(all_candidates) > 1:
                a_alt = all_candidates[1][1]
                if len(a_alt) == n_best:
                    a_lb2, sc_lb2 = lbfgs_optimize(a_alt, time_limit=min(rem2 * 0.5, 100),
                                                   temp_start=1.0, temp_end=40.0)
                    try_seq(a_lb2, "lbfgs_alt")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration (best={best_score:.8f}, n={len(best_a) if best_a is not None else '?'}) ===")
    if best_a is not None:
        best_n = len(best_a)
        target_lengths = [l for l in [384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]
                          if l != best_n]
        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            # Interpolate
            old_x = np.linspace(0, 1, best_n)
            new_x = np.linspace(0, 1, target_n)
            a_interp = np.interp(new_x, old_x, best_a)
            a_interp = normalize(a_interp)

            t_alloc = min(rem * 0.3, 80)
            a_opt, sc_opt = mirror_descent(a_interp, time_limit=t_alloc,
                                           temp=20.0, eta0=0.2, decay=0.9995)
            try_seq(a_opt, f"interp_n{target_n}_md")

            rem2 = deadline - time.monotonic() - 120
            if rem2 > 15:
                a_lbo, sc_lbo = lbfgs_optimize(a_opt, time_limit=min(rem2 * 0.5, 60),
                                               temp_start=2.0, temp_end=30.0)
                try_seq(a_lbo, f"interp_n{target_n}_lbfgs")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate descent polish (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            a_cd, sc_cd = coord_descent_local(best_a, time_limit=min(rem * 0.5, 120))
            try_seq(a_cd, "cd_polish")

            # Final mirror descent with very small eta
            rem2 = deadline - time.monotonic() - 30
            if rem2 > 10:
                a_final, sc_final = mirror_descent(best_a, time_limit=min(rem2, 60),
                                                   temp=50.0, eta0=0.05, decay=0.9999,
                                                   restart_interval=5000)
                try_seq(a_final, "final_md")

    # === Phase 6: Fresh starts from best with perturbation ===
    print(f"\n=== Phase 6: Perturbation restarts (best={best_score:.8f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        for trial in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            noise_scale = 0.01 * (trial + 1)
            a_pert = best_a + rng.standard_normal(n_best) * noise_scale
            a_pert = normalize(a_pert)
            t_alloc = min(rem * 0.4, 60)
            a_opt, sc_opt = mirror_descent(a_pert, time_limit=t_alloc,
                                           temp=20.0, eta0=0.2, decay=0.9995)
            try_seq(a_opt, f"perturb_{trial}")

            # L-BFGS on perturbed
            rem2 = deadline - time.monotonic() - 20
            if rem2 > 10:
                a_lbp, sc_lbp = lbfgs_optimize(a_opt, time_limit=min(rem2 * 0.6, 40),
                                               temp_start=5.0, temp_end=50.0)
                try_seq(a_lbp, f"perturb_{trial}_lbfgs")

    # === Final verification ===
    if best_a is None:
        best_a = make_sin_mod(1024, 3, 0.5)
        best_score = compute_score(best_a)

    # Clip to valid range for evaluate_sequence
    final_a = np.clip(best_a, 0.0, 1000.0)
    final_score = float(evaluate_sequence([float(x) for x in final_a]))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Fallback: scale to ensure sum > 0.01
        s = final_a.sum()
        if s < 0.01:
            final_a = final_a * (1.0 / max(s, 1e-10))
        final_score = float(evaluate_sequence([float(x) for x in final_a]))
        best_score = final_score if np.isfinite(final_score) else best_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(final_a)})")
    return [float(x) for x in final_a]
