        N = 1 << (3 * n - 1).bit_length()
        F_fft = np.fft.rfft(f, N)
        K_fft = np.fft.rfft(K, N)
        grad = np.fft.irfft(F_fft * np.conj(K_fft), N)[:n] / top_k

        # Project to simplex tangent space
        grad -= grad.mean()

        f_new = project_simplex(f - lr * grad)
        cur_obj = fast_convolve(f_new).max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        if lr < 1e-5:
            break
        if no_improve > 5000:
            break

        if verbose and it % 1000 == 0:
            print(f"    PeakShave iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(14):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 150:
            rng_p = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min(no_improve // 100, 5)
            noise = rng_p.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def exact_cd(f0, time_limit, verbose=False):
    """Coordinate descent with exact max evaluation and golden section."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    ORDER = 0.6180339887
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Randomize coordinate order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 5e-6:
                continue

            # Try setting f[j] to different values
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_local_val = old_val
            best_local_obj = best_obj

            # Golden section on the range [0, min(0.999, 2*old_val + 0.001)]
            hi = min(0.999, old_val * 3 + 0.01)
            lo = max(0.0, old_val * 0.3)
            if hi - lo < 1e-4:
                hi = old_val + 0.01
                lo = max(0.0, old_val - 0.01)

            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)

            # Evaluate at c
            f_trial = f.copy()
            f_trial[j] = c
            f_trial[~(np.arange(n) == j)] *= (1.0 - c) / denom
            fc = fast_convolve(f_trial).max()

            # Evaluate at d
            f_trial[j] = d
            f_trial[~(np.arange(n) == j)] *= (1.0 - d) / denom
            fd = fast_convolve(f_trial).max()

            for _ in range(16):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    f_trial[j] = c
                    f_trial[~(np.arange(n) == j)] *= (1.0 - c) / denom
                    fc = fast_convolve(f_trial).max()
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    f_trial[j] = d
                    f_trial[~(np.arange(n) == j)] *= (1.0 - d) / denom
                    fd = fast_convolve(f_trial).max()

            opt_val = (lo + hi) / 2.0
            f_trial[j] = opt_val
            f_trial[~(np.arange(n) == j)] *= (1.0 - opt_val) / denom
            opt_obj = fast_convolve(f_trial).max()

            if opt_obj < best_obj - 1e-13:
                f = f_trial.copy()
                f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def make_geometric(n, alpha):
    """Geometric sequence with decay parameter alpha = -n*ln(r)."""
    r = np.exp(-alpha / n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_geometric_tapered(n, alpha, taper):
    """Geometric with edge taper: a[j] = r^j * (1 - taper * (j/n)^2)."""
    r = np.exp(-alpha / n)
    j = np.arange(n, dtype=np.float64)
    a = r ** j * (1.0 - taper * (j / n) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimizer: geometric init, smooth Adam, peak shaving,
    LP-FW, and exact CD. Focuses on asymmetric front-loaded sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception as e:
        print(f"Parent loading failed: {e}")

    n_parent = len(parent) if parent is not None else 768

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) [was {old_score:.8f}]")

    # === Phase 0: Initialization ===
    print("=== Phase 0: Initialization ===")
    for alpha in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0, 4.0, 5.0]:
        g = make_geometric(n_parent, alpha)
        _try(g, f"geo_a{alpha}")

    for alpha in [1.0, 1.5, 2.0]:
        for taper in [0.1, 0.3, 0.5]:
            g = make_geometric_tapered(n_parent, alpha, taper)
            _try(g, f"geotap_a{alpha}_t{taper}")

    # Mixed init: blend parent with uniform
    f_mix = 0.6 * best_seq + 0.4 * (np.ones(n_parent) / n_parent)
    f_mix = project_simplex(f_mix)
    _try(f_mix, "mixed_init")

    print(f"  After Phase 0: best={best_score:.8f} (n={len(best_seq)})")

    # === Phase 1: Main smooth optimization ===
    print(f"\n=== Phase 1: Smooth Optimization (best={best_score:.8f}) ===")
    base = best_seq.copy()

    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 500)
        print(f"  Smooth opt n={len(base)}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize(base, time_limit=t_alloc, T_start=0.25, T_end=0.003, verbose=True)
        _try(a_opt, "smooth_p1")

    # Second smooth run from a perturbed start
    rem = deadline - time.monotonic() - 400
    if rem > 40:
        noise = rng.standard_normal(len(best_seq)) * 0.005
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem, 200)
        a_opt, _ = smooth_optimize(f_pert, time_limit=t_alloc, T_start=0.15, T_end=0.002)
        _try(a_opt, "smooth_pert")

    # === Phase 2: Peak shaving ===
    print(f"\n=== Phase 2: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem, 250)
        print(f"  PeakShave n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = peak_shave(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "peakshave_p2")

    # === Phase 3: LP-FW ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t_alloc = min(rem, 150)
        print(f"  LP-FW n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "lpfw_p3")

    # === Phase 4: Exact coordinate descent ===
    print(f"\n=== Phase 4: Exact CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_alloc = min(rem, 120)
        print(f"  Exact CD n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = exact_cd(best_seq, time_limit=t_alloc)
        _try(a_opt, "cd_p4")

    # === Phase 5: Longer sequences ===
    print(f"\n=== Phase 5: Longer Sequences (best={best_score:.8f}) ===")
    for target_n in [1024, 1536]:
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        f_interp = interpolate_seq(best_seq, target_n)

        t1 = min(rem * 0.4, 80)
        a_opt, _ = smooth_optimize(f_interp, time_limit=t1, T_start=0.2, T_end=0.02)
        _try(a_opt, f"smooth_n{target_n}")

        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t2 = min(rem * 0.5, 50)
            a_opt, _ = peak_shave(a_opt, time_limit=t2)
            _try(a_opt, f"peakshave_n{target_n}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Smooth with very low T
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_alloc = min(rem, 60)
            a_opt, _ = smooth_optimize(best_seq, time_limit=t_alloc, T_start=0.03, T_end=0.002)
            _try(a_opt, "smooth_fine")

        # Peak shave
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 40)
            a_opt, _ = peak_shave(best_seq, time_limit=t_alloc)
            _try(a_opt, "peakshave_final")

        # Quick CD
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            t_alloc = min(rem, 30)
            a_opt, _ = exact_cd(best_seq, time_limit=t_alloc)
            _try(a_opt, "cd_final")

        # Small perturbations
        for pert_idx in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 6:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 15)
            a_opt, _ = smooth_optimize(f_pert, time_limit=t_alloc, T_start=0.02, T_end=0.001)
            _try(a_opt, f"perturb_{pert_idx}")

        # Reversed
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            f_rev = best_seq[::-1].copy()
            t_alloc = min(rem, 10)
            a_opt, _ = smooth_optimize(f_rev, time_limit=t_alloc, T_start=0.02, T_end=0.001)
            _try(a_opt, "reversed")

    # Fallback
    if best_seq is None:
        best_seq = make_geometric(768, 1.5)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
