import numpy as np

def solve_lp_direction(a, M):
    n = len(a)
    # c = -1 (we maximize, linprog minimizes)
    c = -np.ones(n)
    
    # Inequality constraints: A_ub @ g <= b_ub
    # conv(a,g)[k] = sum_i a[i] * g[k-i] <= M
    # For k=0: a[0]*g[0] <= M
    # For k=1: a[0]*g[1] + a[1]*g[0] <= M
    # ...
    # For k=n-1: a[0]*g[n-1] + a[1]*g[n-2] + ... + a[n-1]*g[0] <= M
    # ...
    # For k=2n-2: a[n-1]*g[n-1] <= M
    
    n_constraints = 2*n - 1
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.full(n_constraints, M)
    
    for k in range(n_constraints):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, j] = a[i]
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = -result.x  # negate because we maximized
        return g
    return None

    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided direction finding,
exact subgradient optimization, coordinate descent polishing, and multi-length
exploration. The LP finds a direction g that minimizes max(conv(a,g)) subject
to g >= 0 and sum(g) = 1, providing informed descent directions.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(a):
    """Build sparse matrix B such that B @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(k - i)
            vals.append(a[i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_find_direction(a, M=None):
    """
    Solve LP: min t s.t. conv(a,g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t) where g is the optimal direction.
    """
    n = len(a)
    B = build_conv_matrix_sparse(a)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality: B @ g - t <= 0
    # A_ub = [B, -e_{n}] (where e_n is the unit vector for t)
    A_t = -np.ones((2 * n - 1, 1))
    A_ub = csr_matrix.hstack([B, csr_matrix(A_t)]).tocsr()
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, t free (but t >= 0 since conv >= 0 for non-negative a,g)
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        t = result.x[n]
        return g, t
    return None, None


def lp_find_direction_v2(a, M):

    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0, sum(g) <= 1.
    This is the original formulation from the strategy.
    """

    B = build_conv_matrix_sparse(a)
    
    c = -np.ones(n)  # minimize -sum(g)
    
    A_ub = B.copy()
    b_ub = np.full(2 * n - 1, M)

    A_ub = csr_matrix.vstack([A_ub, csr_matrix(np.ones((1, n)))]).tocsr()
    b_ub = np.append(b_ub, 1.0)

def exact_subgradient(a, top_k=1):
    """Compute exact subgradient of max(conv(a,a)) at the top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            grad[i] += 2.0 * a[k - i]
    
    return grad / top_k, conv


def coordinate_descent(a, n_iter=50, multipliers=None):
    """Coordinate descent with local search."""
    if multipliers is None:
        multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
    
    n = len(a)
    a = a.copy()
    obj = fast_convolve(a).max()
    
    for _ in range(n_iter):
        improved = False
        for j in range(n):
            if a[j] < 1e-12:
                continue
            old_val = a[j]
            remaining = 1.0 - old_val
            if remaining < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = obj
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / remaining
                trial = a * scale
                trial[j] = new_val
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-12:
                    continue
                trial /= ts
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / remaining
                a = a * scale
                a[j] = best_j_val
                a = np.maximum(a, 0)
                a /= a.sum()
                obj = best_j_obj
                improved = True
        
        if not improved:
            break
    
    return a, obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided direction finding
(solving min max(conv(a,g)) s.t. g in simplex at each step), exact subgradient
fallback, coordinate descent polishing, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def lp_direction(a):
    """Solve LP: min t s.t. conv(a,g)[k] <= t forall k, sum(g)=1, g>=0."""
    n = len(a)
    
    # Build B matrix: B[k,j] such that (B@g)[k] = conv(a,g)[k]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(k - i)
            vals.append(a[i])
    
    B = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: min t
    # s.t. B@g - t <= 0  (2n-1 constraints)
    #      sum(g) = 1     (1 equality)
    #      g >= 0, t >= 0 (bounds)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # A_ub: [B | -1]
    neg_one_col = csr_matrix(-np.ones((2 * n - 1, 1)))
    A_ub = csr_matrix.hstack([B, neg_one_col]).tocsr()
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq: [1...1 | 0]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = np.maximum(result.x[:n], 0)
        if g.sum() > 1e-15:
            g = g / g.sum()
        return g, result.x[n]
    return None, None


def exact_subgradient(a, top_k=1):
    n = len(a)
    conv = fast_convolve(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            grad[i] += 2.0 * a[k - i]
    return grad / top_k, conv
