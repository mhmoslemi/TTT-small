```python
        max_prefix = b[:j].max()
    else:
        max_prefix = 0.0
    if j+n < 2*n-1:
        max_suffix = b[j+n:].max()
    else:
        max_suffix = 0.0
    new_max = max(max_aff, max_prefix, max_suffix)
    return 2.0 * n * new_max / (new_s * new_s)
```

```python
def coordinate_descent(a_init, time_limit):
    a = a_init.astype(np.float64).copy()
    # Normalize to sum=1
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    s = 1.0
    n = len(a)
    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max() / (s*s)
    best_a = a.copy()
    
    t0 = time.monotonic()
    ORDER = 0.6180339887498949
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            if aj < 1e-15:
                # Try setting to a small positive value
                lo, hi = 0.0, s * 0.01
            else:
                lo = -aj * 0.999
                hi = aj * 2.0
            
            if hi - lo < 1e-15:
                continue
            
            # Golden section
            gr = hi - ORDER * (hi - lo)
            gl = lo + ORDER * (hi - lo)
            fg = fast_eval(b, a, j, gr, n, s)
            fl = fast_eval(b, a, j, gl, n, s)
            
            for _ in range(25):
                if fl < fg:
                    hi = gr; gr = gl; fg = fl
                    gl = lo + ORDER * (hi - lo)
                    fl = fast_eval(b, a, j, gl, n, s)
                else:
                    lo = gl; gl = gr; fl = fg
                    gr = hi - ORDER * (hi - lo)
                    fg = fast_eval(b, a, j, gr, n, s)
            
            opt_d = (lo + hi) / 2.0
            opt_obj = fast_eval(b, a, j, opt_d, n, s)
            
            if opt_obj < best_obj - 1e-15:
                # Apply
                b[j:j+n] += 2.0 * opt_d * a
                b[2*j] += opt_d * opt_d
                a[j] += opt_d
                s += opt_d
                best_obj = opt_obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj
```
