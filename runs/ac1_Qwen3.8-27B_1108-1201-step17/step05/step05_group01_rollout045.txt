Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced multi-strategy approach:

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

                        history.pop(0)
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > max_no_improve:
            # Choose restart source: best or from history
            if history and rng.random() < 0.4:
                src = history[rng.integers(0, len(history))]
            else:
                src = best_f
            restart_type = no_improve % 5
            f = structured_restart(src, rng, restart_type)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def lbfqg_optimize(f0, time_limit, verbose=False):
    """Optimize using L-BFGS on smooth approximation with annealed beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)

    betas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    t0_time = time.monotonic()
    total_time = time_limit

    for bi, beta in enumerate(betas):
        remaining = total_time - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)
        max_iter = max(50, int(sub_time * 8))

        try:
            res = minimize(
                smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-9, 'maxls': 50,
                         'maxfun': max_iter * 5}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            break

    return best_f, best_obj


def peak_targeted_sg(f0, time_limit, top_k=3, verbose=False):
    """Subgradient method focused on the top-k convolution peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    v = np.zeros(n)
    momentum = 0.92
    lr = 0.004
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(54321)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        
        # Focus on top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        # Weight peaks by how much above median they are
        median_val = np.median(conv)
        weights = np.maximum(conv[peaks] - median_val, 0.001)
        weights = weights / weights.sum()

        # Compute subgradient from top peaks
        sg = np.zeros(n)
        for i in range(len(peaks)):
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * f[k - idx]
        
        sg *= 2.0

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and best_obj < 1.51:
                print(f"    Peak-SG: obj={best_obj:.10f}")
        else:
            f = f_new  # still move, just don't update best
            no_improve += 1

        lr *= 0.9995
        if no_improve > 200:
            f = structured_restart(best_f, rng, restart_type=rng.integers(0, 5))
            v = np.zeros(n)
            lr = 0.003
            no_improve = 0
            momentum = 0.92

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.4, 0.6, 0.8, 1.2, 1.4, 1.8]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def crossover(a, b):
    """Create a crossover of two sequences (requires same length)."""
    n = len(a)
    if len(b) != n:
        b = interpolate_sequence(b, n)
    # Convex combination
    t = 0.5
    c = (1 - t) * a + t * b
    c = np.maximum(c, 0)
    s = c.sum()
    if s > 1e-15:
        c = c / s
    return c


def propose_candidate(seed=42, budget_s=1800):

    Phase 2: L-BFGS smoothed optimization (250s)

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    population = []  # Keep top diverse solutions

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score

            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")

            if len(population) < 6:
                population.append(a.copy())
            else:
                # Replace the worst if this is better
                pop_scores = [float(evaluate_sequence(p.tolist())) for p in population]
                worst_idx = np.argmax(pop_scores)
                if score < pop_scores[worst_idx]:
                    population[worst_idx] = a.copy()
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))

            best_seq = parent.copy()
            best_score = p_score
            population.append(parent.copy())
    except Exception:
        pass

    if parent is None:

            try_seq(a_opt, "lpfw_reversed")

            # Bimodal start

            x = np.arange(n)
            center = n / 2.0
            width = n / 6.0
            a_bimodal = np.exp(-((x - center * 0.3) ** 2) / (2 * width ** 2)) + \
                        np.exp(-((x - center * 1.7 - n * 0.1) ** 2) / (2 * width ** 2))
            a_bimodal = np.maximum(a_bimodal, 0)
            a_bimodal /= a_bimodal.sum()
            t1b = min(60, rem - 700)
            if t1b > 15:
                a_opt, _ = lp_fw_optimize(a_bimodal, time_limit=t1b)

    # === Phase 2: L-BFGS smoothed optimization (250s) ===

    return [float(x) for x in best_seq]
