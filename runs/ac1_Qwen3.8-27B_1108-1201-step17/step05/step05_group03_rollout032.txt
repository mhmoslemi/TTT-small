"""
Minimize 2*n*max(auto_conv)/(sum)^2 using Newton-based peak equalization via FFT,
subgradient mirror descent, and fast coordinate descent.
Key innovation: Newton step solves the linearized peak-equalization system in O(n log n)
using FFT division, enabling rapid quadratic convergence near the KKT conditions.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_autoconv(a):
    """Compute linear auto-convolution using FFT."""
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:conv_len]


def compute_score(a):
    """Compute 2*n*max_conv/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_autoconv(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def nnewton_delta(a, c, active_peaks, T, reg=1e-8):
    """
    Compute Newton step delta such that (a*delta)[k] = (T - c[k])/2 for active peaks.
    Uses FFT division with Tikhonov regularization.
    """
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    # Build right-hand side
    r = np.zeros(conv_len)
    for k in active_peaks:
        r[k] = (T - c[k]) / 2.0

    # Pad and FFT
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)

    r_pad = np.zeros(fft_size)
    r_pad[:conv_len] = r
    R = np.fft.rfft(r_pad)

    # Regularized inverse: D = R * conj(A) / (|A|^2 + reg*mean(|A|^2))
    abs_A_sq = np.abs(A) ** 2
    mean_abs_A_sq = np.mean(abs_A_sq[abs_A_sq > 0]) if np.any(abs_A_sq > 0) else 1.0
    epsilon = reg * mean_abs_A_sq

    D = np.zeros_like(A, dtype=complex)
    denom = abs_A_sq + epsilon
    D = R * np.conj(A) / denom
    D[0] = 0.0  # Ensure sum(delta) = 0

    delta = np.fft.irfft(D, fft_size)[:n]
    return delta

def subgradient_optimize(a, time_limit, verbose=True):
    """Mirror descent with adaptive step size on the peak of auto-convolution."""
    n = len(a)
    a = normalize(a.copy())
    c = fast_autoconv(a)
    best_a = a.copy()
    best_score = 2.0 * n * c.max()
    t0 = time.monotonic()

    eta = 0.05
    top_k = max(10, min(50, n // 8))
    consecutive_fail = 0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Find top peaks with softmax weighting
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        pv = c[peaks]

        temp = max(1e-10, (pv.max() - pv.min()) * 0.4)
        w = (pv - pv.min()) / temp
        w = np.exp(w)
        w /= w.sum()

        # Compute subgradient
        g = np.zeros(n)
        for idx in range(k_actual):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            lo_i = max(0, k - n + 1)
            hi_i = min(n - 1, k)
            if lo_i <= hi_i:
                ii = np.arange(lo_i, hi_i + 1)
                jj = k - ii
                g[ii] += 2.0 * w[idx] * a[jj]

        # Multiplicative update (mirror descent for simplex)
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = fast_autoconv(a_new)
        new_score = 2.0 * n * c_new.max()

        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            c = c_new
            eta = min(eta * 1.01, 0.5)
            consecutive_fail = 0
        else:
            consecutive_fail += 1
            eta *= 0.95
            if eta < 1e-7:
                # Restart with perturbation
                a = normalize(a + np.random.standard_normal(n) * 0.005)
                c = fast_autoconv(a)
                eta = 0.02
                if 2.0 * n * c.max() < best_score:
                    best_score = 2.0 * n * c.max()
                    best_a = a.copy()
                consecutive_fail = 0

        if verbose and it % 5000 == 0:
            el = time.monotonic() - t0
            print(f"      SG iter {it} (t={el:.1f}s): score={best_score:.10f}")

    return best_a, best_score

def geometric_sequence(n, r):
    a = (1.0 - r) * r ** np.arange(n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def triangular_sequence(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 90)
            a_interp = interpolate_sequence(best_a, target_n)
            a_opt, sc = newton_optimize(a_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"dim_{target_n}")

            # Quick SG polish
            rem = deadline - time.monotonic() - 100
            if rem > 15:
                a_sg, sc_sg = subgradient_optimize(a_opt, time_limit=min(rem * 0.3, 30), verbose=False)
                try_seq(a_sg, f"dim_{target_n}_sg")

    # === Phase 3: Aggressive local search on best ===
    print(f"\n=== Phase 3: Local search (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            # Newton with tighter tolerance
            t_n2 = min(rem * 0.4, 120)
            a_opt, sc = newton_optimize(best_a, time_limit=t_n2, verbose=True)
            try_seq(a_opt, "final_newton")

            rem = deadline - time.monotonic() - 50
            if rem > 15:
                # Coordinate descent polish
                t_cd = min(rem * 0.5, 60)
                a_cd, sc_cd = coord_descent_fast(a_opt, time_limit=t_cd, verbose=False)
                try_seq(a_cd, "final_cd")

            rem = deadline - time.monotonic() - 30
            if rem > 10:
                # Multiple perturbations
                n_best = len(best_a)
                for trial in range(4):
                    rem = deadline - time.monotonic() - 20
                    if rem < 10:
                        break
                    noise_scale = 0.003 * (1.0 + trial * 0.5)
                    noise = rng.standard_normal(n_best) * noise_scale
                    a_pert = project_simplex(best_a + noise)
                    a_opt2, sc2 = newton_optimize(a_pert, time_limit=min(rem, 30), verbose=False)
                    try_seq(a_opt2, f"perturb_{trial}")

    # === Final verification ===
    if best_a is None:
        best_a = cosine_sequence(768)
        best_score = compute_score(best_a)

    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
