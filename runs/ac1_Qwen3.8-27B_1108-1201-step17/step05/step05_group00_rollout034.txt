"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent on the simplex.
Key insight: for non-symmetric sequences, the auto-convolution peak can be significantly

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def score_unnormalized(a):
    """Compute the score for a non-negative sequence (handles its own normalization)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    f = a / s
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def make_ramp(n, p=1.0):
    """Power-law decreasing sequence: f[j] ∝ (n-1-j)^p."""
    x = np.arange(n, 0, -1, dtype=np.float64)
    f = x ** p
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def make_exponential(n, alpha):
    """Exponentially decaying sequence."""
    j = np.arange(n, dtype=np.float64)
    f = np.exp(-alpha * j / n)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def make_shifted_gaussian(n, center_frac=0.3, sigma_frac=0.2):
    """Gaussian centered at a non-central position."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * n
    sigma = sigma_frac * n
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def make_cosine_weighted(n, phase=0.0):
    """Cosine-weighted sequence for spectral shaping."""
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1) if n > 1 else 0
    f = np.abs(np.cos(np.pi * t + phase)) + 0.1
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def coordinate_descent_pass(f, conv, n, order, tol=1e-13):
    """
    One full pass of coordinate descent. Returns updated f, conv, and whether improved.
    order: array of coordinate indices to visit.
    """
    current_max = conv.max()
    any_improved = False

    for j in order:
        fj = f[j]
        if fj < 1e-15:
            continue

        # Affected range: k in [j, j+n-1] intersected with [0, 2n-2]
        lo_k = j
        hi_k = min(j + n - 1, 2 * n - 2)

        # Background max (conv values NOT affected by coordinate j)
        bg_max = 0.0
        if lo_k > 0:
            bg_max = max(bg_max, float(conv[:lo_k].max()))
        if hi_k < 2 * n - 2:
            bg_max = max(bg_max, float(conv[hi_k + 1:].max()))

        # Precompute for the affected range
        k_range = np.arange(lo_k, hi_k + 1)
        k_minus_j = k_range - j  # indices into f, range [0, n-1]

        # Q_k = contribution to conv[k] involving f[j]
        # For k != 2j: Q_k = 2*f[j]*f[k-j]
        # For k == 2j: Q_k = f[j]^2
        Q = 2.0 * fj * f[k_minus_j]
        if 2 * j >= lo_k and 2 * j <= hi_k:
            idx_2j = 2 * j - lo_k
            Q[idx_2j] = fj * fj

        # P_k = conv[k] - Q_k (part not involving f[j])
        P = conv[k_range] - Q
        # Clamp small negatives from numerical error
        P = np.maximum(P, 0.0)

        # Ternary search for optimal x = f[j]
        # phi(x) = max(bg_max, max_k [s^2 * P_k + cross_k(x)])
        # where s = (1-x)/(1-fj), cross = 2*x*s*f[k-j] (or x^2 for k=2j)

        denom = 1.0 - fj
        if denom < 1e-15:
            continue

        # Precompute cross base: f[k_minus_j] for cross terms
        f_kmj = f[k_minus_j]
        has_2j = (2 * j >= lo_k and 2 * j <= hi_k)
        idx_2j = 2 * j - lo_k if has_2j else -1

        def phi(x):
            s = (1.0 - x) / denom
            s2 = s * s
            cross = 2.0 * x * s * f_kmj
            if has_2j:
                cross[idx_2j] = x * x
            conv_new = s2 * P + cross
            m = conv_new.max()
            return m if m > bg_max else bg_max

        # Coarse ternary search
        lo_x = 0.0
        hi_x = 1.0 - 1e-9
        for _ in range(35):
            m1 = lo_x + (hi_x - lo_x) / 3.0
            m2 = hi_x - (hi_x - lo_x) / 3.0
            if phi(m1) <= phi(m2):
                hi_x = m2
            else:
                lo_x = m1

        x_new = (lo_x + hi_x) / 2.0
        phi_xnew = phi(x_new)

        if phi_xnew < current_max - tol:
            # Apply update
            s = (1.0 - x_new) / denom
            f *= s
            f[j] = x_new
            conv = fast_convolve(f)
            current_max = conv.max()
            any_improved = True

    return f, conv, any_improved


def coordinate_descent(f_init, time_limit, rng, verbose=False):
    """
    Full coordinate descent with random restarts and perturbations.
    """
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f /= s
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve_passes = 0
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        # Random coordinate order
        order = rng.permutation(n)

        f_prev_max = conv.max()
        f, conv, improved = coordinate_descent_pass(f, conv, n, order)
        pass_count += 1

        current_max = conv.max()

        if current_max < best_obj - 1e-14:
            best_obj = current_max
            best_f = f.copy()
            no_improve_passes = 0
            if verbose and pass_count % 20 == 0:
                print(f"    CD pass {pass_count}: obj={best_obj:.10f}, score={2*n*best_obj:.6f}")
        else:
            no_improve_passes += 1

        if no_improve_passes >= 5:
            # Perturb to escape local minimum
            noise_scale = min(0.01, 0.001 * no_improve_passes)
            f = best_f + rng.standard_normal(n) * noise_scale
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f /= fs
            conv = fast_convolve(f)
            no_improve_passes = 0

        if no_improve_passes > 20:
            break

    return best_f, 2.0 * n * best_obj


def multiplicative_descent(f_init, time_limit, rng, verbose=False):
    """
    Mirror descent (multiplicative weight) for minimizing max auto-convolution.
    """
    n = len(f_init)
    f = np.maximum(f_init.copy().astype(np.float64), 1e-12)
    f = f / f.sum()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.01
    lr_decay = 0.9997
    temperature = 0.001
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Subgradient: weighted combination of active peak gradients
        k_actual = min(15, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        conv_new = fast_convolve(f)
        co = conv_new.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, score={2*n*best_obj:.6f}")
        else:
            no_improve += 1

        lr *= lr_decay
        temperature *= 1.003

        if no_improve > 200:
            rng_local = np.random.default_rng(it * 31 + 7)
            f = np.maximum(best_f + rng_local.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.005
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, 2.0 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses exact coordinate descent (solving convex 1D subproblems), multiplicative
    descent, and diverse asymmetric initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse asymmetric initializations ===
    print("=== Phase 1: Diverse Initializations ===", flush=True)
    init_candidates = []

    # Try various lengths and shapes
    for n_try in [384, 512, 768, 1024]:
        # Power-law ramps (different exponents)
        for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            init_candidates.append((make_ramp(n_try, p), f"ramp_{p}_n{n_try}"))
        # Exponentials
        for alpha in [1.0, 2.0, 4.0, 8.0]:
            init_candidates.append((make_exponential(n_try, alpha), f"exp_{alpha}_n{n_try}"))
        # Shifted Gaussians
        for cf in [0.2, 0.3, 0.4]:
            for sf in [0.15, 0.25, 0.35]:
                init_candidates.append((make_shifted_gaussian(n_try, cf, sf),
                                        f"gauss_c{cf}_s{sf}_n{n_try}"))
        # Cosine weighted
        for ph in [0.0, 0.3, 0.7, 1.0]:
            init_candidates.append((make_cosine_weighted(n_try, ph),
                                    f"cos_ph{ph}_n{n_try}"))
        # Uniform (baseline)
        init_candidates.append((np.ones(n_try) / n_try, f"uniform_n{n_try}"))

    # Add parent if available
    if parent is not None:
        init_candidates.append((parent.copy(), "parent"))

    # Quick evaluation of all initializations
    best_init = None
    best_init_obj = float('inf')
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        obj = score_unnormalized(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        sc = try_seq(a_norm, label)

    if best_init is None:
        best_init = make_ramp(512, 1.0)
        best_init_obj = score_unnormalized(best_init)

    print(f"  Best init obj: {best_init_obj:.6f} (score={2*len(best_init)*best_init_obj:.6f})", flush=True)
    if best_seq is not None:
        print(f"  Current best score: {best_score:.6f}", flush=True)

    # === Phase 2: Coordinate descent on best initializations ===
    print("\n=== Phase 2: Coordinate Descent ===", flush=True)

    # Run CD on top few initializations
    sorted_inits = sorted(init_candidates, key=lambda x: score_unnormalized(x[0]))
    top_inits = sorted_inits[:5]

    if parent is not None:
        for idx, (a, label) in enumerate(top_inits):
            if a is parent:
                break
        else:
            top_inits.append((parent.copy(), "parent"))

    for a_init, label in top_inits[:4]:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(rem, 200)
        print(f"  CD on [{label}]: t={t_alloc:.0f}s", flush=True)
        a_opt, obj = coordinate_descent(a_init, t_alloc, rng, verbose=True)
        try_seq(a_opt, f"cd_{label}")

        # Also try CD on the best found so far
        if best_seq is not None and best_seq is not a_opt:
            rem = deadline - time.monotonic() - 200
            if rem > 30:
                t_alloc = min(rem, 120)
                a_opt2, obj2 = coordinate_descent(best_seq, t_alloc, rng, verbose=True)
                try_seq(a_opt2, f"cd_best_{label}")

    # === Phase 3: Multiplicative descent on best ===
    print("\n=== Phase 3: Multiplicative Descent ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  MD: t={t_alloc:.0f}s", flush=True)
            a_opt, obj = multiplicative_descent(best_seq, t_alloc, rng, verbose=True)
            try_seq(a_opt, "md")

    # === Phase 4: Try different lengths via interpolation ===
    print("\n=== Phase 4: Length Exploration ===", flush=True)
    base = best_seq if best_seq is not None else best_init
    for target_n in [640, 896, 1280, 1536]:
        rem = deadline - time.monotonic() - 150
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        sc = try_seq(f_interp, f"interp_n{target_n}")

        # Quick CD on interpolated
        t_alloc = min(rem, 100)
        a_opt, obj = coordinate_descent(f_interp, t_alloc, rng, verbose=True)
        try_seq(a_opt, f"cd_interp_n{target_n}")

    # === Phase 5: Random restarts with CD ===
    print("\n=== Phase 5: Random Restarts ===", flush=True)
    base_n = len(best_seq) if best_seq is not None else 768

    for restart in range(8):
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        t_alloc = min(rem / max(1, 8 - restart), 80)

        # Generate perturbed starting point
        base_f = best_seq if best_seq is not None else make_ramp(base_n, 1.0)
        base_f = base_f / base_f.sum()

        noise_scale = 0.02 * (restart + 1)
        f_init = np.maximum(base_f + rng.standard_normal(base_n) * noise_scale, 0.0)
        f_init = f_init / f_init.sum()

        print(f"  Restart {restart}: noise={noise_scale:.3f}, t={t_alloc:.0f}s", flush=True)
        a_opt, obj = coordinate_descent(f_init, t_alloc, rng)
        try_seq(a_opt, f"restart_{restart}")

    # === Phase 6: Fine-tune best with multiple CD passes ===
    print("\n=== Phase 6: Fine-tuning ===", flush=True)
    if best_seq is not None:
        for ft in range(5):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            t_alloc = min(rem, 60)
            a_opt, obj = coordinate_descent(best_seq, t_alloc, rng, verbose=(ft == 0))
            try_seq(a_opt, f"fine_{ft}")

    # === Phase 7: MD polish ===
    print("\n=== Phase 7: MD Polish ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_alloc = min(rem, 50)
            a_opt, obj = multiplicative_descent(best_seq, t_alloc, rng, verbose=True)
            try_seq(a_opt, "md_polish")
            # CD after MD
            rem = deadline - time.monotonic() - 15
            if rem > 15:
                t_alloc = min(rem, 40)
                a_opt2, obj2 = coordinate_descent(best_seq, t_alloc, rng)
                try_seq(a_opt2, "cd_after_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
