def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    if not cond.any():
        return np.full(n, target_sum / n)
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def smooth_obj_and_grad_x(x, tau):
    """
    Objective and gradient in softmax parameterization.
    x is the log-parameter, a = softmax(x).
    Returns (objective, gradient_wrt_x).

    # Compute a = softmax(x)
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()

    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)

def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian initialization with given sigma fraction."""
    x = np.arange(n) / n - 0.5
    a = np.exp(-x**2 / (2 * sigma_frac**2))
    return a / a.sum()


def raised_cosine_init(n):
    """Raised cosine window."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def smooth_obj_and_grad_x(x, tau):
    """
    Smooth objective and gradient in softmax parameterization.
    objective = (1/tau) * log(sum(exp(tau * c_k))) where c = conv(softmax(x), softmax(x))
    Returns (objective, gradient_wrt_x) as a tuple.
    """
    n = len(x)
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    c = fast_convolve(a)
    c_max = c.max()
    c_shifted = c - c_max
    exp_tau_c = np.exp(tau * c_shifted)
    w_sum = exp_tau_c.sum()
    w = exp_tau_c / w_sum
    
    obj = c_max + np.log(w_sum) / tau
    
    # Gradient wrt a
    grad_a = np.zeros(n)
    threshold = 1e-14
    sig = np.where(w > threshold)[0]
    for ki in sig:
        k = int(ki)
        wk = 2.0 * w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            grad_a[js:je+1] += wk * a[k-je : k-js+1][::-1]
    
    # Gradient wrt x: df/dx_j = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x


def optimize_lbfgs_anneal(a0, tau_init, tau_final, time_limit, verbose=False):
    """
    L-BFGS-B optimization with temperature annealing on softmax parameterization.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    t0 = time.monotonic()
    
    x = np.log(a)
    x = x - x.max()
    
    # Determine number of annealing steps
    min_step_time = 10.0  # minimum seconds per step
    max_steps = max(3, int(time_limit / min_step_time))
    min_steps = 3
    n_steps = min(max_steps, max(min_steps, 8))
    taus = np.geomspace(tau_init, tau_final, n_steps)
    
    for si in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        tau = taus[si]
        remaining = time_limit - (time.monotonic() - t0)
        
        # Estimate max iterations: ~5ms per iteration for n=768
        max_iter = max(30, int(remaining * 0.7 / 0.005))
        max_iter = min(max_iter, 2000)
        
        try:
            result = minimize(
                smooth_obj_and_grad_x, x, args=(tau,),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxfun': max_iter * 3}
            )
            x = result.x
        except Exception:
            break
        
        x_shifted = x - x.max()
        a_new = np.exp(x_shifted)
        a_new = a_new / a_new.sum()
        
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_conv_max:
            best_conv_max = obj_new
            best_a = a_new.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Anneal step {si+1}/{n_steps}: tau={tau:.1f}, conv_max={best_conv_max:.12f} ({elapsed:.0f}s)")
    
    return best_a, best_conv_max


def lp_water_filling(a, M, max_iter=5, blend_rate=0.3):
    """
    LP-based water-filling refinement.
    At each step, solve LP to find b maximizing sum(b) s.t. b>=0, conv(a,b)<=M.
    Then blend a with b.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_conv = fast_convolve(a).max()
    
    for it in range(max_iter):
        cur_M = fast_convolve(a).max()
        
        # Build constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0
        # A has shape (2n-1, n)
        # For efficiency, construct via the convolution structure
        A = np.zeros((2*n-1, n))
        for j in range(n):
            # A[k,j] = a[k-j] for k in [j, j+n-1]
            A[j:j+n, j] = a
        
        # LP: minimize -sum(b) s.t. A @ b <= cur_M, b >= 0
        c = -np.ones(n)
        b_ub = cur_M * np.ones(2*n-1)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0)
                b_sum = b.sum()
                if b_sum < 1e-10:
                    continue
                
                # Normalize b to have same sum as a (which is 1)
                b = b / b_sum
                
                # Blend
                a_new = (1 - blend_rate) * a + blend_rate * b
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                
                conv_new = fast_convolve(a_new).max()
                if conv_new < best_conv:
                    best_conv = conv_new
                    best_a = a_new.copy()
                
                a = a_new
            else:
                break
        except Exception:
            break
    
    return best_a, best_conv


def coordinate_descent_v2(a0, time_limit, verbose=False):
    """Improved coordinate descent with adaptive multipliers."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            # Try multiple multipliers, including finer grid near 1.0
            multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 1:
            break
    
    return best_a, best_obj


def local_search(a0, time_limit, seed=0):
    """Randomized local search with various move types."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    support = np.where(a > 1e-8)[0]
    if len(support) == 0:
        support = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            move_type = rng.integers(0, 3)
            
            if move_type == 0:
                # Pair swap
                if len(support) < 2:
                    continue
                idx = rng.integers(0, len(support), size=2)
                i, j = int(support[idx[0]]), int(support[idx[1]])
                if i == j or a[i] < 1e-10 or a[j] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    for direction in [1, -1]:
                        transfer = direction * frac * min(a[i], a[j])
                        new_i = a[i] + transfer
                        new_j = a[j] - transfer
                        if new_i < 0 or new_j < 0:
                            continue
                        a_trial = a.copy()
                        a_trial[i] = new_i
                        a_trial[j] = new_j
                        val = fast_convolve(a_trial).max()
                        if val < best_obj - 1e-14:
                            best_obj = val
                            best_a = a_trial.copy()
                            a = a_trial.copy()
            
            elif move_type == 1:
                # Spread: reduce a peak coordinate and distribute to neighbors
                if len(support) < 2:
                    continue
                i = int(support[rng.integers(0, len(support))])
                if a[i] < 1e-10:
                    continue
                # Reduce i by a fraction
                for frac in [0.05, 0.1, 0.15, 0.2]:
                    reduction = frac * a[i]
                    # Distribute to 2-4 neighbors
                    n_neighbors = min(4, n - 1)
                    neighbors = rng.choice(n - 1, size=n_neighbors, replace=False)
                    neighbors = neighbors[neighbors != i]
                    if len(neighbors) == 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= reduction
                    a_trial[neighbors] += reduction / len(neighbors)
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial /= ss
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
            
            else:
                # Aversh: increase a coordinate and scale down others
                i = rng.integers(0, n)
                if a[i] > 0.9:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    old_remaining = 1.0 - a[i]
                    if old_remaining < 1e-10:
                        continue
                    new_val = a[i] + frac
                    scale = (1.0 - new_val) / old_remaining
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a_trial[mask] *= scale
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial /= ss
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
    
    return best_a, best_obj


def subgradient_step(a, tau, lr, v, momentum):
    """Single subgradient step with momentum on the simplex."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    c_shifted = c - c_max
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    w = w / w_sum
    
    # Gradient
    grad = np.zeros(n)
    threshold = 1e-12
    sig = np.where(w > threshold)[0]
    for ki in sig:
        k = int(ki)
        wk = 2.0 * w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            grad[js:je+1] += wk * a[k-je : k-js+1][::-1]
    
    # Center gradient
    grad = grad - grad.mean()
    
    # Update
    v = momentum * v - lr * grad
    a_new = project_simplex(a + v, target_sum=1.0)
    
    return a_new, v, c_max


def optimize_subgrad_v2(a0, time_limit, verbose=False):
    """Improved subgradient optimization with better scheduling."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    v = np.zeros(n)
    tau = 50.0
    lr = 0.05
    momentum = 0.85
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Adaptive scheduling
        progress = (time.monotonic() - t0) / time_limit
        # Decrease lr over time
        lr_base = 0.05 * (1.0 - 0.7 * progress)
        # Increase tau over time (sharper)
        tau_base = 50.0 * (1.0 + 5.0 * progress)
        
        if no_improve > 300:
            tau = min(500.0, tau * 1.02)
            lr = max(0.001, lr * 0.95)
        elif no_improve > 100:
            tau = tau * 1.005
            lr = min(lr_base * 1.5, lr * 1.01)
        else:
            tau = min(300.0, max(30.0, tau + (tau_base - tau) * 0.1))
            lr = min(0.1, max(0.005, lr + (lr_base - lr) * 0.1))
        
        a, v, c_max = subgradient_step(a, tau, lr, v, momentum)
        
        if c_max < best_obj - 1e-13:
            best_obj = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 500:
            rng_r = np.random.default_rng(iter_count * 17 + 3)
            noise_scale = 0.003 + 0.003 * min(no_improve // 500, 5)
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.04
            tau = 80.0
            momentum = 0.85
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining L-BFGS-B, LP water-filling,
    subgradient methods, and local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 50.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a = a / a.sum()
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    # === Phase 1: Generate and evaluate initializations ===
    print("\n=== Phase 1: Initializations ===")
    n_parent = len(parent) if parent is not None else 768
    
    candidate_n = [n_parent]
    for n_try in [384, 512, 640, 768, 1024, 1280]:
        if n_try not in candidate_n:
            candidate_n.append(n_try)
    candidate_n.sort(key=lambda x: (abs(x - n_parent), x))
    
    inits = {}
    for n in candidate_n[:4]:
        best_init = None
        best_init_obj = float('inf')
        
        methods = [
            ("flat_spec", flat_spectrum_init(n, rng)),
            ("random", random_init(n, rng)),
            ("gauss", gaussian_init(n, 0.25)),
            ("gauss_wide", gaussian_init(n, 0.4)),
            ("raised_cos", raised_cosine_init(n)),
        ]
        
        if parent is not None:
            methods.append(("parent_interp", interpolate_sequence(parent, n)))
        
        for name, init in methods:
            init = np.maximum(init, 0)
            ss = init.sum()
            if ss < 1e-10:
                continue
            init = init / ss
            obj = fast_convolve(init).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = init
        
        if best_init is not None:
            inits[n] = (best_init, best_init_obj)
            print(f"  n={n}: best init conv_max={best_init_obj:.10f}")
    
    # === Phase 2: L-BFGS-B with annealing ===
    print(f"\n=== Phase 2: L-BFGS-B Annealing (best={best_score:.6f}) ===")
    sorted_inits = sorted(inits.items(), key=lambda x: x[1][1])
    
    # Allocate time: 40% of remaining for this phase
    rem = deadline - time.monotonic()
    phase2_time = min(rem * 0.35, max(60, rem * 0.4))
    
    n_optimize = min(3, len(sorted_inits))
    time_per_opt = phase2_time / n_optimize
    
    for rank, (n_len, (init, init_obj)) in enumerate(sorted_inits[:n_optimize]):
        rem = deadline - time.monotonic() - (phase2_time - time_per_opt * (n_optimize - 1 - rank))
        if rem < 20:
            break
        t_alloc = min(time_per_opt * 1.2, rem)
        if t_alloc < 15:
            break
        
        print(f"  L-BFGS-B n={n_len} (init_obj={init_obj:.10f}), t={t_alloc:.0f}s")
        # Two-stage: high tau then low tau
        tau_init = 200.0
        tau_final = 20.0
        
        a_opt, obj = optimize_lbfgs_anneal(init, tau_init, tau_final, time_limit=t_alloc, verbose=(rank == 0))
        try_seq(a_opt, f"lbfgs_n{n_len}_r{rank}")
    
    # === Phase 3: LP Water-filling ===
    print(f"\n=== Phase 3: LP Water-Filling (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_lp = min(rem * 0.3, 120)
            print(f"  LP t={t_lp:.0f}s")
            # Run water-filling with different blend rates
            for bl in [0.2, 0.4, 0.6]:
                rem = deadline - time.monotonic() - 200
                if rem < 20:
                    break
                a_opt, obj = lp_water_filling(best_seq, M=None, max_iter=3, blend_rate=bl)
                try_seq(a_opt, f"lp_blend{bl}")
    
    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_alloc = min(rem / (3 - p), 100)
            
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            else:
                f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init, target_sum=1.0)
            
            print(f"  SG variant {p}, t={t_alloc:.0f}s")
            a_opt, obj = optimize_subgrad_v2(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"sg_v2_{p}")
    
    # === Phase 5: Coordinate descent + local search ===
    print(f"\n=== Phase 5: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_cd = min(rem * 0.5, 80)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent_v2(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_v2")
        
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_ls = min(rem * 0.5, 60)
            print(f"  LS t={t_ls:.0f}s")
            a
