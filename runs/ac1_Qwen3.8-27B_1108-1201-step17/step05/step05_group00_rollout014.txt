def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Softmax-smoothed gradient descent with backtracking line search
2. LP-based Frank-Wolfe for high-precision polishing
3. Multi-length exploration with diverse initializations
4. Coordinate descent for final local refinement

Key novelty: Uses log-sum-exp smoothed max for stable gradients at high
sharpness levels, combined with adaptive step sizes and Nesterov-like
momentum for faster convergence than plain subgradient methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_softmax(f, c, beta):
    """
    Compute gradient of the log-sum-exp smoothed max:
    LSE_beta(c) = (1/beta) * log(sum(exp(beta * c_k)))
    Gradient w.r.t. f_j: sum_k softmax(beta*c)_k * 2*f_{k-j} * [0<=k-j<n]
    """
    n = len(f)
    # Softmax weights over convolution coefficients
    scaled = beta * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # g_j = 2 * sum_i f_i * w_{j+i}
    # This is a cross-correlation: g = 2 * correlate(w, f)
    # Using FFT: correlate(w, f)_j = sum_i f_i * w_{j+i}
    # = convolve(f, w_rev)[len(w)-1 + j]
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    
    # g_j = 2 * conv[len(w_rev)-1 + j] = 2 * conv[2n-2+j]... 
    # Let me verify: conv[k] = sum_i f[i] * w_rev[k-i] = sum_i f[i] * w[2n-2-(k-i)] = sum_i f[i] * w[2n-2-k+i]
    # We want g_j/2 = sum_i f[i] * w[j+i], so 2n-2-k = j => k = 2n-2-j
    # So g_j = 2 * conv[2n-2-j] for j=0..n-1
    # But wait, conv has length n + (2n-1) - 1 = 3n-2, and index 2n-2-j ranges from 2n-2 down to n-1
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g


def compute_gradient_pnorm(f, c, p):
    """Compute gradient of (1/p)*sum(c^p) where c = f*f."""
    n = len(f)
    if p <= 1:
        w = np.ones(len(c))
    else:
        max_c = np.max(c)
        if max_c < 1e-15:
            return np.zeros(n)
        # For numerical stability, normalize c before taking power
        c_norm = c / max_c
        w = c_norm ** (p - 1) * max_c ** (p - 1)
    
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g


def gradient_descent_smoothed(f0, time_limit, verbose=False):
    """
    Main gradient-based optimizer using log-sum-exp smoothing.
    Progressively increases sharpness (decreases 1/beta).
    Uses backtracking line search with momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    
    t0 = time.monotonic()
    
    # Beta schedule: start smooth (low beta), end sharp (high beta)
    # beta = 1/temperature; higher beta = sharper approximation to max
    beta_schedule = [2, 4, 8, 16, 32, 64, 128, 256]
    
    # Momentum
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.85:
            break
        
        it = 0
        no_improve = 0
        lr = 1.0 / (beta * np.max(fast_convolve(f)) ** 0.5 * n ** 0.5)
        
        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.08, (time_limit - (time.monotonic() - t0)) * 0.15)
        
        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.85:
            it += 1
            c = fast_convolve(f)
            g = compute_gradient_softmax(f, c, beta)
            
            # Normalize gradient
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            # Adaptive step size with backtracking
            lr = min(lr, 1.0 / (g_norm + 1e-10))
            
            # Momentum update
            momentum = mu * momentum - lr * g
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)
            
            # Backtracking: ensure objective doesn't increase too much
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            
            max_allowed_increase = 1e-6
            if obj_trial > best_obj + max_allowed_increase:
                # Reduce step
                lr *= 0.5
                momentum = mu * momentum - lr * g
                f_trial = f + momentum
                f_trial = project_simplex(f_trial)
                c_trial = fast_convolve(f_trial)
                obj_trial = c_trial.max()
            
            if obj_trial <= c.max() + 1e-10:
                f = f_trial
                if obj_trial < best_obj - 1e-12:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    # Slightly increase lr on improvement
                    lr = min(lr * 1.05, 1.0 / (g_norm * 0.5 + 1e-10))
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.7
            
            if no_improve > 80:
                # Restart with perturbation
                rng = np.random.default_rng(it * 13 + int(beta) * 7)
                noise_scale = 0.002 / np.sqrt(beta / 8.0)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                momentum = np.zeros(n)
                lr = 1.0 / (beta * 2.0 * n ** 0.5)
                no_improve = 0
            
            if verbose and it % 100 == 0:
                print(f"    GD beta={beta} iter {it}: obj={c.max():.10f}, best={best_obj:.10f}")
        
        if verbose:
            print(f"  GD beta={beta} done: best={best_obj:.10f}, time={time.monotonic()-t0:.1f}s")
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def multiplicative_peak_reduction(f0, time_limit, verbose=False):
    """
    Multiplicative weight update that specifically targets peaks.
    Uses the convolution structure to compute per-element peak contributions.
    """
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        
        # Find the top peaks
        k_peaks = min(30, len(c))
        peak_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        
        # Weight peaks by their excess above median
        median_c = np.median(c)
        peak_vals = c[peak_idx]
        excess = np.maximum(peak_vals - median_c, 0)
        if excess.sum() < 1e-15:
            excess = np.ones(len(peak_idx))
        w_peaks = excess / excess.sum()
        
        # Compute subgradient: for each peak k, the contributing f elements are
        # f[j] for j in [max(0,k-n+1), min(n-1,k)]
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w_peaks[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w_peaks[i] * f[k - idx]
        
        # Multiplicative update
        f = f * np.exp(-lr * 2.0 * sg)
        f = np.maximum(f, 1e-12)
        f = f / f.sum()
        
        c_new = fast_convolve(f)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            lr *= 0.95
        
        if verbose and it % 50 == 0:
            print(f"    MPR iter {it}: obj={c_new.max():.10f}, best={best_obj:.10f}")
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 17)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 1e-10)
            f = f / f.sum()
            lr = 0.008
            no_improve = 0
        
        if lr < 1e-6:
            break
    
    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """
    Fine coordinate descent: adjust individual coordinates to reduce max convolution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try different multiplier values
            for mult in [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.3:
                    continue
                
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Light smoothing
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6, power=4):
    """Create a bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_optimal_shape(n):
    """
    Create a sequence based on the theoretical optimal shape for minimizing
    max auto-convolution. Uses a raised-cosine-like profile.
    """
    x = np.linspace(-1, 1, n)
    # Raised cosine (Hann window)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization and quick screening
    Phase 2: Smoothed gradient descent (main workhorse)
    Phase 3: LP-FW high-precision polish
    Phase 4: Multi-length exploration
    Phase 5: Final coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = make_gaussian(512, 0.3)
    
    # === Phase 1: Quick screening of initializations ===
    print("=== Phase 1: Initialization Screening ===")
    t1_start = time.monotonic()
    
    init_candidates = []
    n_parent = len(parent)
    
    # Parent-based
    init_candidates.append(("parent", parent))
    
    # Various shapes at parent length
    for name, shape_fn in [("gauss", lambda: make_gaussian(n_parent, 0.25)),
                           ("gauss_wide", lambda: make_gaussian(n_parent, 0.35)),
                           ("bell", lambda: make_bell(n_parent, 0.6, 4)),
                           ("bell_flat", lambda: make_bell(n_parent, 0.7, 6)),
                           ("raised_cos", lambda: make_optimal_shape(n_parent))]:
        init_candidates.append((name, shape_fn()))
    
    # Evaluate all
    best_init = None
    best_init_obj = float('inf')
    for name, a in init_candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        c = fast_convolve(a)
        obj = 2 * len(a) * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a
        try_seq(a, f"init_{name}")
    
    # Add small random perturbations to parent
    for trial in range(5):
        noise = rng.standard_normal(n_parent) * 0.01
        a_pert = np.maximum(parent + noise, 0)
        a_pert = a_pert / a_pert.sum()
        c = fast_convolve(a_pert)
        obj = 2 * n_parent * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_pert
        try_seq(a_pert, f"init_pert{trial}")
    
    print(f"  Best init: obj={best_init_obj:.8f}")
    print(f"  Phase 1 time: {time.monotonic()-t1_start:.1f}s")
    
    # === Phase 2: Smoothed gradient descent ===
    print(f"\n=== Phase 2: Gradient Descent (n={len(best_init)}) ===")
    t2_start = time.monotonic()
    rem = deadline - time.monotonic() - 600
    t_alloc = max(30, min(rem, 500))
    print(f"  Allocation: {t_alloc:.0f}s")
    
    a_opt, obj = gradient_descent_smoothed(best_init, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, "gd_smooth")
    
    # Also try from parent directly if different
    if not np.allclose(best_init, parent):
        rem = deadline - time.monotonic() - 500
        if rem > 100:
            t_alloc2 = min(rem * 0.3, 150)
            a_opt2, obj2 = gradient_descent_smoothed(parent, time_limit=t_alloc2)
            try_seq(a_opt2, "gd_parent")
    
    print(f"  Phase 2 time: {time.monotonic()-t2_start:.1f}s")
    
    # === Phase 3: Multiplicative peak reduction ===
    print(f"\n=== Phase 3: Multiplicative Peak Reduction ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 50:
            t_alloc = min(rem * 0.4, 200)
            print(f"  Allocation: {t_alloc:.0f}s")
            a_opt, obj = multiplicative_peak_reduction(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "mpr")
    
    # === Phase 4: LP-FW high-precision polish ===
    print(f"\n=== Phase 4: LP-FW Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 60:
            t_alloc = min(rem * 0.5, 300)
            print(f"  Allocation: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    base = best_seq if best_seq is not None else best_init
    target_lengths = [1024, 896, 1152]
    
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 150
        if rem < 80:
            break
        
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")
        
        # Quick gradient descent on interpolated
        t_gd = min(rem * 0.3, 100)
        if t_gd > 30:
            a_opt, obj = gradient_descent_smoothed(f_interp, time_limit=t_gd)
            try_seq(a_opt, f"gd_n{target_n}")
        
        # LP-FW on the result
        rem2 = deadline - time.monotonic() - 100
        if rem2 > 60:
            t_lpfw = min(rem2 * 0.5, 120)
            a_opt, obj = lp_fw_optimize(a_opt, time_limit=t_lpfw)
            try_seq(a_opt, f"lpfw_n{target_n}")
    
    # === Phase 6: Deep LP-FW with diverse restarts ===
    print(f"\n=== Phase 6: Deep LP-FW ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 40:
                break
            t_alloc = min(rem / (4 - p), 100)
            
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                # Reverse blend
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Mix with raised cosine
                rc = make_optimal_shape(n_best)
                f_init = 0.8 * best_seq + 0.2 * rc
                f_init = f_init / f_init.sum()
            
            print(f"  LP-FW variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_seq(a_opt, f"deep_lpfw_{p}")
    
    # === Phase 7: Final coordinate polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            t_alloc = min(rem, 60)
            print(f"  Coord descent: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_final")
        
        # One more LP-FW squeeze
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t_alloc = min(rem, 40)
            a_opt, obj = lp_fw_optimize(a_opt, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
