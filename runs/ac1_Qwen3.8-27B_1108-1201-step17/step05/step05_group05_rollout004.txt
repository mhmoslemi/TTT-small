    """Build matrix T such that T @ b = convolve(a, b)."""
    n = len(a)

    objective = cp.Maximize(cp.sum(b))

        except:

    if b.value is None:

    b_val = np.maximum(b.value, 0)
    s = b_val.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
as the primary method, combined with subgradient optimization, coordinate
descent, and multi-length exploration. The LP at each step finds the best
direction that maintains the autocorrelation bound, enabling reliable
monotone improvement.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix T such that T @ b = linear_convolve(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, tau, top_k=100):
    """Compute smooth approximation of subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    return grad, conv


def lp_solve(a, M):
    """Solve LP: maximize sum(b) s.t. b>=0, (a*b)[k] <= M for all k.
    Returns normalized b or None if infeasible/failed."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    solvers = [cp.GLOP, cp.CBC, cp.GLPK, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if b.value is not None:
                b_val = np.maximum(b.value, 0.0)
                s = b_val.sum()
                if s > 1e-10:
                    return b_val / s
            return None
        except Exception:
            continue
    return None


def line_search(a, g, n_iter=12):
    """Golden section search for best t in [0,1] minimizing max(conv((1-t)a+tg, (1-t)a+tg))."""
    best_t = 0.0
    best_obj = fast_convolve(a).max()
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    f_c = fast_convolve((1.0 - c) * a + c * g).max()
    f_d = fast_convolve((1.0 - d) * a + d * g).max()
    for _ in range(n_iter):
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - inv_phi * (hi - lo)
            f_c = fast_convolve((1.0 - c) * a + c * g).max()
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + inv_phi * (hi - lo)
            f_d = fast_convolve((1.0 - d) * a + d * g).max()
    t_mid = (lo + hi) / 2.0
    f_mid = fast_convolve((1.0 - t_mid) * a + t_mid * g).max()
    if f_mid < best_obj:
        best_t = t_mid
        best_obj = f_mid
    # Also check t=1
    f_1 = fast_convolve(g).max()
    if f_1 < best_obj:
        best_t = 1.0
        best_obj = f_1
    return best_t, best_obj

def coordinate_descent(a0, time_limit):
    """Coordinate descent with local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        idx_order = rng.permutation(n)

        for j in idx_order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break

    return best_a, best_obj


def pair_swap_search(a0, time_limit, seed=0):
    """Randomized pair-swap local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)

    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    support = np.where(a > 1e-6)[0]
    if len(support) == 0:
        support = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < -1e-15 or new_j < -1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = max(0.0, new_i)
                    a_trial[j] = max(0.0, new_j)
                    ss = a_trial.sum()
                    if ss < 1e-10:
                        continue
                    a_trial /= ss
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-10:
        return np.ones(n) / n
    return a / ss


def random_init(n, rng):
    a = rng.exponential(1.0, n)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print(f"\n=== Phase 4: Different Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        lengths_to_try = []
        for mult in [0.75, 0.5, 1.5, 2.0]:
            target_n = int(n_best * mult)
            if 256 <= target_n <= 2048 and target_n != n_best:
                lengths_to_try.append(target_n)
        # Also try some fixed lengths
        for fixed_n in [384, 512, 1024, 1280, 1536]:
            if fixed_n not in lengths_to_try:
                lengths_to_try.append(fixed_n)
        lengths_to_try = sorted(set(lengths_to_try), key=lambda x: abs(x - n_best))[:4]

        for target_n in lengths_to_try:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem * 0.15, 120)
            f_interp = interpolate_sequence(best_seq, target_n)
            # Also try fresh init for this length
            f_fresh = flat_spectrum_init(target_n, rng)
            # Pick the better initial
            obj_interp = fast_convolve(f_interp).max()
            obj_fresh = fast_convolve(f_fresh).max()
            if obj_fresh < obj_interp:
                f_interp = f_fresh

            a_opt, obj = lp_iteration(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"lp_n{target_n}")

    # === Phase 5: LP iteration on best (longer) ===
    print(f"\n=== Phase 5: Extended LP on Best (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_lp2 = min(rem * 0.3, 300)
        if t_lp2 > 30:
            print(f"  Time: {t_lp2:.0f}s")
            a_opt, obj = lp_iteration(best_seq, time_limit=t_lp2, verbose=True)
            try_seq(a_opt, "lp_extended")

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.3, 200)
        if t_cd > 15:
            print(f"  Time: {t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 7: Pair Swap ===
    print(f"\n=== Phase 7: Pair Swap (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_ps = min(rem * 0.3, 150)
        if t_ps > 10:
            print(f"  Time: {t_ps:.0f}s")
            a_opt, obj = pair_swap_search(best_seq, time_limit=t_ps, seed=seed)
            try_seq(a_opt, "pair_swap")

    # === Phase 8: Final polish - combined ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for _ in range(5):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            # Short LP burst
            t_short = min(rem * 0.2, 40)
            if t_short > 10:
                a_opt, obj = lp_iteration(best_seq, time_limit=t_short)
                try_seq(a_opt, "lp_polish")
            # Short CD
            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem * 0.2, 30))
                try_seq(a_opt, "cd_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
