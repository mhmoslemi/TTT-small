Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA-accelerated subgradient,

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def smoothed_max(conv, tau, top_k=200):
    """Smoothed max of convolution values using softmax."""
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        return pv.max()
    return pv.max() + np.log(ws) / tau

def golden_section_1d(a, j, time_check=None, deadline=None):
    """Optimize coordinate j using golden section search, keeping sum=1."""
    n = len(a)
    old_val = a[j]
    if old_val < 1e-15:
        return a, fast_convolve(a).max(), False
    
    # When we change a[j], we scale all other coordinates to maintain sum=1
    # a_trial[j] = x, a_trial[k] = a[k] * (1-x)/(1-a[j]) for k != j
    # x in [0, 1-eps]
    
    best_obj = fast_convolve(a).max()
    best_x = old_val
    
    # Golden section search
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0 - 1e-10
    
    # First evaluate at golden points
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    
    a_c = a.copy()
    a_c[j] = c
    rem = 1.0 - c
    old_rem = 1.0 - old_val
    if old_rem > 1e-10:
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a_c[mask] = a[mask] * rem / old_rem
    obj_c = fast_convolve(np.maximum(a_c, 0)).max()
    
    a_d = a.copy()
    a_d[j] = d
    rem_d = 1.0 - d
    if old_rem > 1e-10:
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a_d[mask] = a[mask] * rem_d / old_rem
    obj_d = fast_convolve(np.maximum(a_d, 0)).max()
    
    # Also check original
    if best_obj < obj_c and best_obj < obj_d:
        # Original is best, just return
        return a, best_obj, False
    
    # Golden section
    for _ in range(40):
        if time_check and time.monotonic() > deadline:
            break
        if obj_c < obj_d:
            hi = d
            d = c
            obj_d = obj_c
            c = hi - inv_phi * (hi - lo)
            a_c = a.copy()

def pair_local_search(a0, time_limit, seed=0):
    """Local search on pairs of coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    support = np.where(a > 1e-10)[0]
    if len(support) == 0:
        support = np.arange(n)
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        i, j = int(rng.integers(0, len(support))), int(rng.integers(0, len(support)))
        if i == j:
            continue
        i, j = int(support[i]), int(support[j])
        if a[i] < 1e-12 or a[j] < 1e-12:
            continue
        
        # Try transferring mass between i and j
        # a_trial[i] = a[i] + t, a_trial[j] = a[j] - t, others scaled
        for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
            for direction in [1, -1]:
                transfer = direction * frac * min(a[i], a[j])
                new_i = a[i] + transfer
                new_j = a[j] - transfer
                if new_i < -1e-12 or new_j < -1e-12:
                    continue
                new_i = max(0.0, new_i)
                new_j = max(0.0, new_j)
                a_trial = a.copy()
                a_trial[i] = new_i
                a_trial[j] = new_j

                    continue

                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_a = a_trial.copy()
                    a = a_trial.copy()
    
    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def simulated_annealing_polish(a0, time_limit, seed=0):
    """Simulated annealing with small perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)

        temp = max(min_temp, 0.001 * (1.0 - elapsed_frac) ** 2)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, tau, top_k=150):
    """Compute gradient of smoothed max of conv(a,a) and the value."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    # Numerically stable softmax
    max_s = scaled.max()
    w = np.exp(scaled - max_s)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    smooth_val = pv.max() + (max_s + np.log(ws)) / tau
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]

    return grad, conv, smooth_val

    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    y = a.copy()
    v = a.copy()
    t_k = 1.0
    tau = 80.0
    step = 0.05
    L_estimate = 20.0
    
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Adaptive tau scheduling
        if no_improve > 500:
            tau = min(400.0, tau * 1.03)
        elif no_improve > 200:
            tau = min(300.0, tau * 1.01)
        elif no_improve < 50:
            tau = max(30.0, tau * 0.995)
        
        grad, conv, smooth_val = compute_smooth_grad(y, tau=tau, top_k=min(120, n))
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum_coeff = (t_k - 1.0) / t_new
        y_next = v + momentum_coeff * (v - a)
        y_next = np.maximum(y_next, 0.0)
        
        # Gradient step with backtracking
        trial = project_simplex(y_next - step * grad, target_sum=1.0)
        
        # Check if step is good enough (simplified descent condition)
        obj_trial = fast_convolve(trial).max()
        if obj_trial > smooth_val + 2.0 * step * L_estimate * step:
            # Step too large, backtrack
            step *= 0.5
            trial = project_simplex(y_next - step * grad, target_sum=1.0)
            obj_trial = fast_convolve(trial).max()
        elif obj_trial < smooth_val - 0.5 * step * L_estimate * step:
            # Step could be larger
            L_estimate = max(5.0, L_estimate * 0.9)
            step = min(0.2, step * 1.1)
        
        v = trial
        a = trial
        t_k = t_new
        
        # Track best
        curr_obj = obj_trial
        if curr_obj < best_obj - 1e-12:
            best_obj = curr_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA iter {iter_count}: obj={best_obj:.10f}, step={step:.4f}, tau={tau:.1f}, no_imp={no_improve} ({elapsed:.0f}s)", flush=True)
        
        # Restart if stuck
        if no_improve > 1000:
            rng_r = np.random.default_rng(iter_count * 31 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve // 1000, 5)
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, target_sum=1.0)
            v = a.copy()
            y = a.copy()
            t_k = 1.0
            no_improve = 0
            step = 0.04
            L_estimate = 25.0
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with adaptive multi-point search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    no_improve_passes = 0
    
    # Precompute mask template
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved_this_pass = 0
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-13:
                # Try setting it to small positive value
                candidates = [0.001 / n, 0.002 / n]
            else:
                candidates = [old_val * m for m in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 1.8]]
                candidates.append(0.0)
                candidates = [c for c in candidates if 0 <= c < 1.0]
            
            best_j_val = old_val
            best_j_obj = best_obj
            best_trial = None
            
            old_remaining = 1.0 - old_val
            
            for new_val in candidates:
                if new_val < 0 or new_val >= 1.0 - 1e-10:
                    continue
                rem = 1.0 - new_val
                if old_remaining < 1e-12:
                    # All mass was on j, redistribute
                    a_trial = np.full(n, rem / (n - 1))
                    a_trial[j] = new_val
                else:
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] = a[mask] * rem / old_remaining
                    a_trial = np.maximum(a_trial, 0)
                
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
                    best_trial = a_trial
            
            if best_trial is not None:
                a = best_trial
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_this_pass += 1
        
        if improved_this_pass > 0:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
        
        if verbose and pass_count % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}, improved={improved_this_pass} ({elapsed:.0f}s)", flush=True)
        
        if no_improve_passes >= 2:
            break
    
    return best_a, best_obj


def pair_search(a0, time_limit, seed=0):
    """Randomized pair transfer local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    support = np.where(a > 1e-10)[0]
    if len(support) < 2:
        support = np.arange(n)
    
    evals = 0
    while time.monotonic() - t0 < time_limit:
        # Batch of random pairs
        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            if a[i] < 1e-12 or a[j] < 1e-12:
                continue
            
            min_val = min(a[i], a[j])
            for frac in [0.05, 0.1, 0.2, 0.35]:
                for direction in [1, -1]:
                    transfer = direction * frac * min_val
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < -1e-12 or new_j < -1e-12:
                        continue
                    new_i = max(0.0, new_i)
                    new_j = max(0.0, new_j)
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial /= ss
                    val = fast_convolve(a_trial).max()
                    evals += 1
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
    
    if verbose and evals > 0:
        print(f"    Pair search: {evals} evals, obj={best_obj:.10f}", flush=True)
    
    return best_a, best_obj


def iterative_spectral_init(n, rng, n_iter=30):
    """Iteratively alternate between flat-spectrum (freq) and non-negative (time)."""
    a = np.ones(n) / n
    
    for it in range(n_iter):
        # Frequency domain
        A = np.fft.fft(a)
        
        # Soft equalization of magnitude
        mags = np.abs(A)
        mags[mags < 1e-15] = 1e-15
        target_mag = mags.mean()
        alpha = 0.3 + 0.4 * (it / n_iter)  # gradually increase equalization strength
        
        A_new = A * (target_mag / mags) ** alpha
        
        # Ensure Hermitian symmetry for real output
        A_new[0] = A_new[0].real
        for k in range(1, n // 2 + 1):
            if k < n - k:
                A_new[n - k] = np.conj(A_new[k])
        if n % 2 == 0:
            A_new[n // 2] = A_new[n // 2].real
        
        # Back to time domain
        a = np.fft.ifft(A_new).real
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_sequence(a, best_state):
    """Try a sequence and update best state."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[1]:
        best_state[1] = score
        best_state[0] = a.copy()
        print(f"  NEW BEST: {score:.6f} (n={len(a)})", flush=True)
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: iterative spectral init, FISTA, coordinate descent,
    pair search, multi-length exploration, and final polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    # === Phase 1: Smart Initializations ===
    print("\n=== Phase 1: Initializations ===", flush=True)
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    for n_try in [384, 512, 640, 768, 896, 1024]:
        if n_try not in target_lengths:
            target_lengths.append(n_try)
    target_lengths.sort(key=lambda x: abs(x - (len(parent) if parent is not None else 768)))
    
    inits = {}
    for n in target_lengths[:4]:
        best_init = None
        best_init_obj = float('inf')
        
        methods = [
            ("spectral_iter", iterative_spectral_init(n, rng, n_iter=40)),
            ("flat_spec", flat_spectrum_init(n, rng)),
            ("random", random_init(n, rng)),
        ]
        
        if parent is not None:
            methods.append(("parent_interp", interpolate_sequence(parent, n)))
            # Perturbed parent
            if n == len(parent):
                methods.append(("parent_pert", project_simplex(parent + rng.standard_normal(n) * 0.01, target_sum=1.0)))
        
        for name, init in methods:
            try:
                init = np.maximum(init, 0)
                ss = init.sum()
                if ss < 1e-10:
                    continue
                init = init / ss
                obj = fast_convolve(init).max()
                if obj < best_init_obj:
                    best_init_obj = obj
                    best_init = init
            except Exception:
                pass
        
        if best_init is not None:
            inits[n] = (best_init, best_init_obj)
            print(f"  n={n}: best init obj={best_init_obj:.6f}", flush=True)
    
    # Include current best
    if best_seq is not None:
        n_b = len(best_seq)
        bn = best_seq / best_seq.sum()
        ob = fast_convolve(bn).max()
        if n_b not in inits or ob < inits[n_b][1]:
            inits[n_b] = (bn, ob)
    
    sorted_inits = sorted(inits.items(), key=lambda x: x[1][1])
    
    # === Phase 2: FISTA on top initializations ===
    print(f"\n=== Phase 2: FISTA (best={best_score:.6f}) ===", flush=True)
    
    total_budget = deadline - time.monotonic() - 100
    n_opt = min(3, len(sorted_inits))
    time_fracs = [0.40, 0.25, 0.20]
    
    for rank in range(n_opt):
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        n_len, (init, init_obj) = sorted_inits[rank]
        t_alloc = min(total_budget * time_fracs[rank], rem)
        print(f"  FISTA n={n_len} (obj={init_obj:.6f}), t={t_alloc:.0f}s", flush=True)
        a_opt, obj = fista_optimize(init, time_limit=t_alloc, verbose=(rank == 0))
        try_seq(a_opt, f"fista_n{n_len}")
    
    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_cd = min(rem * 0.5, 180)
            print(f"  CD t={t_cd:.0f}s", flush=True)
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_main")
    
    # === Phase 4: Pair Search ===
    print(f"\n=== Phase 4: Pair Search (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_ps = min(rem * 0.4, 100)
            print(f"  PS t={t_ps:.0f}s", flush=True)
            a_opt, obj = pair_search(best_seq, time_limit=t_ps, seed=seed)

    # === Phase 5: Iterative Refinement ===
    print(f"\n=== Phase 5: Iterative Refinement (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        for rnd in range(4):
            rem = deadline - time.monotonic() - 50
            if rem < 25:
                break
            
            # FISTA with different perturbation
            noise_scale = 0.002 * (rnd + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale, target_sum=1.0)
            t_f = min(rem * 0.35, 80)
            a_opt, obj = fista_optimize(f_pert, time_limit=t_f)
            try_seq(a_opt, f"fista_r{rnd}")
            
            # CD
            rem = deadline - time.monotonic() - 30
            if rem > 15:
                t_c = min(rem * 0.35, 60)
                a_opt, obj = coordinate_descent(best_seq, time_limit=t_c)
                try_seq(a_opt, f"cd_r{rnd}")
            
            # PS
            rem = deadline - time.monotonic() - 20
            if rem > 10:
                t_p = min(rem * 0.3, 30)
                a_opt, obj = pair_search(best_seq, time_limit=t_p, seed=seed + rnd * 100)
                try_seq(a_opt, f"ps_r{rnd}")
    
    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        candidates = [n_best - 32, n_best + 32, n_best + 64, int(n_best * 1.25)]
        seen = set()
        for target_n in candidates:
            if target_n in seen or target_n <= 128 or target_n > 2048 or target_n == n_best:

            seen.add(target_n)

            rem = deadline - time.monotonic() - 20
            if rem > 15:
                a_opt, obj = coordinate_descent(a_opt, time_limit=min(rem, 30))
                try_seq(a_opt, f"cd_{target_n}")
    
    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        # Final CD
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_c = min(rem, 50)
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_c)
            try_seq(a_opt, "final_cd")
        
        # Quick pair search
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_p = min(rem, 25)
            a_opt, obj = pair_search(best_seq, time_limit=t_p, seed=seed + 999)
            try_seq(a_opt, "final_ps")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
