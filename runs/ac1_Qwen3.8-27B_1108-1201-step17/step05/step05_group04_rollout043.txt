def softmax_weights(c, temp, top_k=None):
    c = np.asarray(c, dtype=np.float64)
    n = c.size
    if n == 0: return c.astype(float)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return np.ones(n)/n
    r = c / cmax
    if top_k is not None and 0 < top_k < n:
        idx = np.argpartition(r, -top_k)[-top_k:]
        mask = np.zeros(n, dtype=bool); mask[idx]=True
        mask = mask | mask[::-1]
        r = np.where(mask, r, -1e9)
    shifted = (r - 1.0) / float(temp)
    # r can be -1e9, shifted very negative; subtract max (0)
    shifted = np.where(np.isfinite(shifted), shifted, -1e9)
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = float(w.sum())
    if not np.isfinite(ws) or ws < 1e-300:
        # fallback uniform over top r
        k = max(1, int(0.1*n))
        idx = np.argpartition(c, -k)[-k:]
        mask = np.zeros(n, bool); mask[idx]=True; mask = mask | mask[::-1]
        w = mask.astype(float); w/=w.sum()
    else:
        w /= ws
    w = 0.5*(w + w[::-1])
    ws = float(w.sum())
    if ws > 0: w /= ws
    else: w = np.ones(n)/n
    return w

```
def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0: return v
    if n == 1:
        if v[0] > 0: return np.array([1.0])
        return np.array([0.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    s = w.sum()
    if s <= 0: return np.ones(n)/n
    return w/s
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for a nonnegative sequence.

The program searches by:
- Starting from the provided parent sequence when available.
- Generating diverse candidates by interpolation, smooth shapes, and inverse
  stationarity solves H_p a = 1, where H_p is the Hankel matrix with
  H_p[i,j] = p[i+j].  These equations come from first-order stationarity
  conditions for minimax self-convolution.
- Running a fixed-point active-set / softmax optimizer on those conditions.
- Using Adam with correct subgradients T_p a.
- Polishing with a fast exact coordinate-descent algorithm on the full
  convolution scores.
"""

import numpy as np
import time


def normalize_nonneg(v):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    if v.size == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s > 1e-300:
        v /= s
    else:
        v[:] = 1.0 / v.size
    return v


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf")
    c = fast_convolve(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(c.max()) / (s * s))


def symmetrize_if_better(a):
    a = normalize_nonneg(a)
    sc = score_normalized(a)
    if a.size < 2:
        return a, sc
    sym = normalize_nonneg(0.5 * (a + a[::-1]))
    sc_sym = score_normalized(sym)
    if sc_sym < sc:
        return sym, sc_sym
    return a, sc


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0], dtype=np.float64) if v[0] > 0 else np.array([0.0], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = v - theta
    w[w < 0.0] = 0.0
    s = float(w.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return w / s


def softmax_weights(c, temp, top_k=None):
    c = np.asarray(c, dtype=np.float64).ravel()
    n = c.size
    if n == 0:
        return c.astype(np.float64)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    r = c / cmax
    if top_k is not None and 0 < top_k < n:
        idx = np.argpartition(r, -top_k)[-top_k:]
        mask = np.zeros(n, dtype=bool)
        mask[idx] = True
        mask = mask | mask[::-1]
        r = np.where(mask, r, -1e9)
    shifted = (r - 1.0) / float(temp)
    shifted = np.where(np.isfinite(shifted), shifted, -1e9)
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = float(w.sum())
    if not np.isfinite(ws) or ws < 1e-300:
        k = max(1, int(0.1 * n))
        idx = np.argpartition(c, -k)[-k:]
        mask = np.zeros(n, dtype=bool)
        mask[idx] = True
        mask = mask | mask[::-1]
        w = mask.astype(np.float64)
        ws = float(w.sum())
        w = w / ws if ws > 0 else np.ones(n) / n
    else:
        w /= ws
    w = 0.5 * (w + w[::-1])
    ws = float(w.sum())
    if ws > 0:
        w /= ws
    else:
        w = np.ones(n, dtype=np.float64) / n
    return w


def softmin_weights(values, temp):
    v = np.asarray(values, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v.astype(np.float64)
    if temp <= 1e-300:
        temp = 1e-300
    shifted = -v / float(temp)
    shifted = np.where(np.isfinite(shifted), shifted, -1e9)
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = float(w.sum())
    if not np.isfinite(ws) or ws < 1e-300:
        w = np.ones(n, dtype=np.float64) / n
    else:
        w /= ws
    w = 0.5 * (w + w[::-1])
    ws = float(w.sum())
    if ws > 0:
        w /= ws
    else:
        w = np.ones(n, dtype=np.float64) / n
    return w


def active_uniform_weights(c, delta):
    c = np.asarray(c, dtype=np.float64).ravel()
    n = c.size
    if n == 0:
        return c.astype(np.float64)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    active = c >= cmax - max(0.0, float(delta)) * cmax
    if not np.any(active):
        active = np.zeros(n, dtype=bool)
        active[int(np.argmax(c))] = True
    active = active | active[::-1]
    p = active.astype(np.float64)
    s = float(p.sum())
    if s <= 0:
        return np.ones(n, dtype=np.float64) / n
    return p / s


def regularize_weights(p, alpha=0.01):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p
    u = np.ones(n, dtype=np.float64) / n
    return (1.0 - alpha) * p + alpha * u


def subgradient_term(a, p):
    """
    Compute H_p(a) where (H_p a)_j = sum_i p[i+j] * a[i].
    This is the subgradient term for active lag weights p.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a.copy()
    rev = a[::-1]
    size = p.size + n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    B = np.fft.rfft(rev, fft_size)
    conv = np.fft.irfft(P * B, fft_size)[:size]
    term = conv[n - 1:2 * n - 1]
    term = np.where(np.isfinite(term), term, 0.0)
    return np.maximum(term, 0.0)


def solve_for_p(p, idx, n, diag, ridge=1e-6):
    """
    Solve H_p x = 1, with H_p[i,j] = p[i+j].  The solution is normalized.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size != 2 * n - 1:
        return None
    H = p[idx].astype(np.float64, copy=False)
    b = np.ones(n, dtype=np.float64)
    x = None
    last_Hs = None

    ridges = (ridge, ridge * 100.0, ridge * 10000.0, 1e-1)
    for rg in ridges:
        Hs = H.copy()
        Hs[diag] += rg
        try:
            x = np.linalg.solve(Hs, b)
        except Exception:
            x = None
        if x is not None and np.all(np.isfinite(x)) and float(np.max(np.abs(x))) < 1e12:
            last_Hs = Hs
            break

    if x is None:
        try:
            x = np.linalg.lstq(H, b, rcond=None)[0]
        except Exception:
            return None
        if x is None or not np.all(np.isfinite(x)):
            return None

    # Remove antisymmetric noise by symmetrizing if it helps.
    xs = 0.5 * (x + x[::-1])
    neg0 = float(np.mean(x < -1e-7))
    negs = float(np.mean(xs < -1e-7))
    if negs <= neg0:
        x = xs

    neg = float(np.mean(x < -1e-7))
    if neg > 0.15 and last_Hs is not None:
        order = np.argsort(x)[::-1]
        m = int(max(1, np.floor(0.8 * n)))
        support = np.zeros(n, dtype=bool)
        support[order[:m]] = True
        Hs2 = last_Hs[np.ix_(support, support)]
        b2 = np.ones(m, dtype=np.float64)
        xsol = None
        d2 = np.diag_indices(m)
        for rg2 in (1e-5, 1e-3, 1e-1):
            Hs3 = Hs2.copy()
            Hs3[d2] += rg2
            try:
                xsol = np.linalg.solve(Hs3, b2)
            except Exception:
                xsol = None
            if xsol is not None and np.all(np.isfinite(xsol)) and float(np.max(np.abs(xsol))) < 1e12:
                break
        if xsol is None:
            try:
                xsol = np.linalg.lstq(Hs2, b2, rcond=None)[0]
            except Exception:
                xsol = None
        if xsol is not None and np.all(np.isfinite(xsol)):
            xsol = np.maximum(xsol, 0.0)
            s = float(xsol.sum())
            if s > 1e-12:
                xf = np.zeros(n, dtype=np.float64)
                xf[support] = xsol
                return xf / s

    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-12:
        return None
    return x / s


def make_gaussian_weight(n, sigma):
    m = 2 * n - 1
    pos = np.linspace(0.0, 1.0, m, dtype=np.float64)
    sigma = max(float(sigma), 1e-3)
    p = np.exp(-0.5 * ((pos - 0.5) / sigma) ** 2)
    s = float(p.sum())
    if s <= 0:
        return np.ones(m, dtype=np.float64) / m
    return p / s


def make_window_weight(n, low, high):
    m = 2 * n - 1
    pos = np.linspace(0.0, 1.0, m, dtype=np.float64)
    p = np.zeros(m, dtype=np.float64)
    mask = (pos >= low) & (pos <= high)
    if not np.any(mask):
        mask = np.zeros(m, dtype=bool)
        mask[int(np.argmin(np.abs(pos - 0.5)))] = True
    p[mask] = 1.0
    s = float(p.sum())
    if s <= 0:
        return np.ones(m, dtype=np.float64) / m
    return p / s


def candidate_from_p(n, p, idx, diag, ridge=1e-5):
    b = solve_for_p(p, idx, n, diag, ridge)
    if b is None:
        return None
    b = normalize_nonneg(b)
    bs = normalize_nonneg(0.5 * (b + b[::-1]))
    if score_normalized(bs) < score_normalized(b):
        return bs
    return b


def shape_vector(n, kind, param=0.0):
    if n <= 0:
        return np.array([], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    if kind == "uniform":
        v = np.ones(n, dtype=np.float64)
    elif kind == "arcsine":
        v = 1.0 / np.sqrt(np.maximum(x * (1.0 - x), 1e-12))
    elif kind == "sin":
        v = np.sin(np.pi * x) ** 2
    elif kind == "tri":
        v = 1.0 - np.abs(2.0 * x - 1.0)
    elif kind == "gauss":
        c = 0.5
        s = float(param) if param > 0 else 0.2
        v = np.exp(-0.5 * ((x - c) / max(s, 1e-3)) ** 2)
    elif kind == "edge":
        v = 1.0 / np.sqrt(np.maximum(x, 1e-8)) + 1.0 / np.sqrt(np.maximum(1.0 - x, 1e-8))
    else:
        v = np.ones(n, dtype=np.float64)
    v = np.maximum(v, 1e-12)
    s = float(v.sum())
    return v / s if s > 0 else v


def smooth_random_vector(n, rng, n_components=6):
    if n <= 0:
        return np.array([], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    for _ in range(max(1, int(n_components))):
        c = float(rng.uniform(0.15, 0.85))
        s = float(rng.uniform(0.07, 0.30))
        a = float(rng.uniform(0.3, 2.0))
        v += a * np.exp(-0.5 * ((x - c) / max(s, 1e-3)) ** 2)
    v = np.maximum(v, 1e-10)
    s = float(v.sum())
    return v / s if s > 0 else v


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64).ravel()
    if f.size == 0:
        return np.ones(new_n, dtype=np.float64) / max(1, new_n)
    if f.size == new_n:
        return normalize_nonneg(f)
    x_old = np.linspace(0.0, 1.0, f.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    g = np.interp(x_new, x_old, f)
    return normalize_nonneg(np.maximum(g, 0.0))


def adam_optimize(a_init, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.10, temp_decay=0.9995, top_k=None,
                  verbose=False):
    a_init = normalize_nonneg(a_init)
    n = a_init.size
    if n == 0 or time_limit <= 0:
        return a_init, float("inf")

    f = a_init.copy()
    conv = fast_convolve(f)
    best_f = f.copy()
    best_score = score_normalized(f)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    temp = float(temp_init)
    cur_lr = float(lr)
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if no_improve > 50:
            rng = np.random.default_rng(t * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            temp = max(0.02, float(temp_init) * 0.5)
            cur_lr = float(lr)
            no_improve = 0

        cmax = float(conv.max())
        if cmax <= 1e-300:
            break

        p = softmax_weights(conv, temp, top_k)
        sg = subgradient_term(f, p)
        sg = np.where(np.isfinite(sg), sg, 0.0)

        t += 1
        m = beta1 * m + (1.0 - beta1) * sg
        v = beta2 * v + (1.0 - beta2) * sg * sg
        mh = m / (1.0 - beta1 ** t)
        vh = v / (1.0 - beta2 ** t)
        f_new = f - cur_lr * mh / (np.sqrt(vh) + eps)
        f = project_simplex(f_new)

        conv = fast_convolve(f)
        cur_score = 2.0 * n * float(conv.max())

        if cur_score < best_score * (1.0 - 1e-12):
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp = max(0.01, temp * float(temp_decay))
        cur_lr = max(float(lr) * (0.9997 ** t), float(lr) * 0.1)

        if verbose and t % 100 == 0:
            print(f"      Adam iter {t}, best={best_score:.8f}")

    best_f, best_score = symmetrize_if_better(best_f)
    return best_f, best_score


def self_consistent_optimize(a_init, time_limit, max_dense_n=1024, ridge=1e-5, verbose=False):
    a_init = normalize_nonneg(a_init)
    n = a_init.size
    if n == 0 or time_limit <= 0 or n > max_dense_n:
        return None, float("inf")

    a, cur_score = symmetrize_if_better(a_init)
    best_a = a.copy()
    best_score = cur_score

    idx = np.add.outer(np.arange(n), np.arange(n))
    diag = np.diag_indices(n)

    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(a)
        s = float(a.sum())
        cmax = float(conv.max())
        cur_score = 2.0 * n * cmax / (s * s)

        if cur_score > best_score * (1.0 + 1e-6):
            a = best_a.copy()
            conv = fast_convolve(a)
            s = float(a.sum())
            cur_score = best_score

        best_move_a = None
        best_move_score = cur_score

        if no_improve < 15:
            specs = [("s", 0.05), ("s", 0.15), ("a", 0.05), ("a", 0.15)]
        elif no_improve < 45:
            specs = [("s", 0.10), ("s", 0.30), ("a", 0.10), ("a", 0.30)]
        else:
            specs = [("s", 0.20), ("s", 0.40), ("a", 0.20), ("a", 0.40)]

        for kind, par in specs:
            if time.monotonic() - t0 > time_limit:
                break

            if kind == "s":
                p = softmax_weights(conv, par, top_k=None)
            else:
                p = active_uniform_weights(conv, par)
            p = regularize_weights(p, alpha=0.01)

            base_candidates = []

            b = solve_for_p(p, idx, n, diag, ridge)
            if b is not None:
                b = normalize_nonneg(b)
                base_candidates.append(b)

            h = subgradient_term(a, p)
            hr = float(h.max()) - float(h.min())
            if hr <= 1e-300:
                g = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                g = softmin_weights(h, 0.05 * hr + 1e-12)
            g = normalize_nonneg(g)
            base_candidates.append(g)

            for base in base_candidates:
                base = normalize_nonneg(base)
                base_sym = normalize_nonneg(0.5 * (base + base[::-1]))

                sc1 = score_normalized(base)
                sc2 = score_normalized(base_sym)
                if sc1 <= sc2:
                    pure = base
                    sc_pure = sc1
                else:
                    pure = base_sym
                    sc_pure = sc2

                if sc_pure < best_move_score * (1.0 - 1e-12):
                    best_move_score = sc_pure
                    best_move_a = pure

                if sc_pure < cur_score * 1.02:
                    mix = normalize_nonneg(0.5 * a + 0.5 * pure)
                    sc_mix = score_normalized(mix)
                    if sc_mix < best_move_score * (1.0 - 1e-12):
                        best_move_score = sc_mix
                        best_move_a = mix

        if best_move_a is not None and best_move_score < cur_score * (1.0 - 1e-12):
            a = best_move_a
            if best_move_score < best_score:
                best_score = best_move_score
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 15 or no_improve % 30 == 0:
                rng = np.random.default_rng(it * 37 + 11)
                a = normalize_nonneg(best_a + rng.standard_normal(n) * 0.004)
                a, sc_pert = symmetrize_if_better(a)
                if sc_pert > best_score * 1.05:
                    a = best_a.copy()

        if it % 20 == 0:
            asy = normalize_nonneg(0.5 * (a + a[::-1]))
            sc_asy = score_normalized(asy)
            if sc_asy < cur_score * (1.0 - 1e-12):
                a = asy
                cur_score = sc_asy
                if sc_asy < best_score:
                    best_score = sc_asy
                    best_a = a.copy()

        if verbose and it % 50 == 0:
            print(f"      SCAS iter {it}, best={best_score:.8f}")

    best_a, best_score = symmetrize_if_better(best_a)
    return best_a, best_score


def coordinate_descent_raw(a_init, time_limit, multipliers=None):
    a_init = normalize_nonneg(a_init)
    n = a_init.size
    if n == 0 or time_limit <= 0:
        return a_init, float("inf")

    if multipliers is None:
        multipliers = (0.25, 0.5, 0.75, 0.9, 1.1, 1.25, 1.5, 2.0, 4.0)

    a = a_init.copy()
    c = fast_convolve(a)
    S = float(a.sum())
    cur_score = 2.0 * n * float(c.max()) / (S * S)
    best_a = a.copy()
    best_c = c.copy()
    best_S = S
    best_score = cur_score

    t0 = time.monotonic()
    passes = 0
    Lc = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        coeff = 2.0 * a

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            v = float(a[j])
            start = j
            end = j + n

            mo = -1.0
            if start > 0:
                mo = float(c[:start].max())
            if end < Lc:
                mo2 = float(c[end:].max())
                if mo2 > mo:
                    mo = mo2

            if v <= 1e-15:
                base = max(float(S) / float(n), 1e-6)
                cands = [0.1 * base, 0.5 * base, base, 5.0 * base, 20.0 * base, 100.0 * base]
            else:
                cands = [v * float(mm) for mm in multipliers]

            inside = c[start:end]
            central_idx = j
            best_w = None
            best_local_score = cur_score

            for w in cands:
                w = float(w)
                if not np.isfinite(w) or w < 0.0:
                    continue
                if w > 100.0 * max(float(S), 1e-12):
                    continue
                newS = S - v + w
                if newS <= 1e-15:
                    continue

                new_inside = inside + coeff * (w - v)
                new_inside[central_idx] = inside[central_idx] + w * w - v * v
                mi = float(new_inside.max())
                new_max = mi if mi > mo else mo
                new_score = 2.0 * n * new_max / (newS * newS)

                if new_score < best_local_score * (1.0 - 1e-12):
                    best_local_score = new_score
                    best_w = w

            if best_w is not None:
                w = float(best_w)
                delta = coeff * (w - v)
                delta[central_idx] = w * w - v * v
                c[start:end] += delta
                a[j] = w
                coeff[j] = 2.0 * w
                S = S - v + w
                cur_score = best_local_score
                improved = True

                if cur_score < best_score * (1.0 - 1e-12):
                    best_score = cur_score
                    best_a = a.copy()
                    best_c = c.copy()
                    best_S = S

        if not improved:
            break

        if S < 0.5 or S > 2.0 or passes % 5 == 0:
            a = normalize_nonneg(a)
            c = fast_convolve(a)
            S = float(a.sum())
            cur_score = score_normalized(a)
            if cur_score < best_score * (1.0 - 1e-12):
                best_score = cur_score
                best_a = a.copy()
                best_c = c.copy()
                best_S = S

    best_a, best_score = symmetrize_if_better(best_a)
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            pp = normalize_nonneg(p)
            if np.sum(pp) > 1e-12:
                parent = pp
    except Exception:
        parent = None

    if parent is not None:
        best_seq = parent.copy()
    else:
        best_seq = shape_vector(768, "arcsine", 0.0)

    best_seq, best_score = symmetrize_if_better(best_seq)
    print(f"Start: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    # Build a pool of diverse starting candidates.
    cand_arrays = [best_seq]
    cand_arrays.append(normalize_nonneg(0.5 * (best_seq + best_seq[::-1])))

    for tn in (512, 640, 896, 960, 1024, 1152):
        if 128 <= tn <= 1280 and tn != n_best:
            cand_arrays.append(interpolate_sequence(best_seq, tn))

    for kind in ("arcsine", "sin", "tri", "gauss", "edge"):
        cand_arrays.append(shape_vector(n_best, kind, 0.0))

    for _ in range(2):
        cand_arrays.append(smooth_random_vector(n_best, rng, 6))

    if n_best <= 1024:
        idx_b = np.add.outer(np.arange(n_best), np.arange(n_best))
        diag_b = np.diag_indices(n_best)

        for sigma in (0.08, 0.15, 0.25):
            p = make_gaussian_weight(n_best, sigma)
            cand = candidate_from_p(n_best, p, idx_b, diag_b, 1e-5)
            if cand is not None:
                cand_arrays.append(cand)

        for low, high in ((0.25, 0.75), (0.35, 0.65), (0.45, 0.55)):
            p = make_window_weight(n_best, low, high)
            cand = candidate_from_p(n_best, p, idx_b, diag_b, 1e-5)
            if cand is not None:
                cand_arrays.append(cand)

    scored = []
    for i, arr in enumerate(cand_arrays):
        if arr is None:
            continue
        arr = normalize_nonneg(arr)
        sc = score_normalized(arr)
        if np.isfinite(sc):
            scored.append((sc, i, arr))
    scored.sort()

    # Phase 1: strong fixed-point optimizer on current best.
    rem = deadline - time.monotonic()
    if best_seq is not None and len(best_seq) <= 1024 and rem > 60:
        t_alloc = min(float(rem) * 0.35, 420.0)
        print(f"Phase 1 SCAS best: budget {t_alloc:.1f}s")
        a_opt, sc_opt = self_consistent_optimize(best_seq, t_alloc, max_dense_n=1024)
        if a_opt is not None:
            a_opt, sc_opt = symmetrize_if_better(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                n_best = len(best_seq)
                print(f"  improved by SCAS: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 60:
        t_alloc = min(float(rem) * 0.20, 200.0)
        print(f"Phase 1 Adam best: budget {t_alloc:.1f}s")
        a_opt, sc_opt = adam_optimize(best_seq, t_alloc,
                                      lr=0.005, temp_init=0.08, temp_decay=0.9995)
        a_opt, sc_opt = symmetrize_if_better(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            n_best = len(best_seq)
            print(f"  improved by Adam: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 60:
        t_alloc = min(float(rem) * 0.25, 200.0)
        print(f"Phase 1 coordinate best: budget {t_alloc:.1f}s")
        a_opt, sc_opt = coordinate_descent_raw(best_seq, t_alloc)
        a_opt, sc_opt = symmetrize_if_better(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            n_best = len(best_seq)
            print(f"  improved by coordinate: {best_score:.8f}")

    # Phase 2: Try a few diverse alternatives.
    alt_count = 0
    for sc, i, arr in scored[1:7]:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        if alt_count >= 3:
            break
        alt_count += 1
        arr = normalize_nonneg(arr)
        n_arr = len(arr)
        print(f"Phase 2 alternative {alt_count}: n={n_arr}, init={sc:.8f}")

        if n_arr <= 1024:
            t_alloc = min(float(rem) * 0.25, 120.0)
            a_opt, sc_opt = self_consistent_optimize(arr, t_alloc, max_dense_n=1024)
        else:
            t_alloc = min(float(rem) * 0.30, 120.0)
            a_opt, sc_opt = adam_optimize(arr, t_alloc,
                                          lr=0.005, temp_init=0.08, temp_decay=0.9995)

        if a_opt is not None:
            a_opt, sc_opt = symmetrize_if_better(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                n_best = len(best_seq)
                print(f"  alternative improved: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem > 50:
                t_cd = min(float(rem) * 0.20, 60.0)
                a_cd, sc_cd = coordinate_descent_raw(a_opt, t_cd)
                a_cd, sc_cd = symmetrize_if_better(a_cd)
                if sc_cd < best_score:
                    best_seq = a_cd.copy()
                    best_score = sc_cd
                    n_best = len(best_seq)
                    print(f"  alternative+CD improved: {best_score:.8f}")

    # Phase 3: Explore nearby lengths.
    for tn in (1024, 1152, 960, 896, 640, 512):
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        if best_seq is None or len(best_seq) == tn:
            continue

        f_interp = interpolate_sequence(best_seq, tn)
        f_interp, sc_interp = symmetrize_if_better(f_interp)
        if sc_interp < best_score:
            best_seq = f_interp.copy()
            best_score = sc_interp
            n_best = tn
            print(f"Length {tn} interpolation improved: {best_score:.8f}")

        if tn <= 1024:
            t_alloc = min(float(rem) * 0.35, 180.0)
            print(f"Phase 3 SCAS length {tn}: budget {t_alloc:.1f}s")
            a_opt, sc_opt = self_consistent_optimize(f_interp, t_alloc, max_dense_n=1024)
        else:
            t_alloc = min(float(rem) * 0.35, 180.0)
            print(f"Phase 3 Adam length {tn}: budget {t_alloc:.1f}s")
            a_opt, sc_opt = adam_optimize(f_interp, t_alloc,
                                          lr=0.004, temp_init=0.08, temp_decay=0.9995)

        if a_opt is not None:
            a_opt, sc_opt = symmetrize_if_better(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                n_best = tn
                print(f"  length {tn} improved: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem > 60:
                t_cd = min(float(rem) * 0.30, 90.0)
                a_cd, sc_cd = coordinate_descent_raw(a_opt, t_cd)
                a_cd, sc_cd = symmetrize_if_better(a_cd)
                if sc_cd < best_score:
                    best_seq = a_cd.copy()
                    best_score = sc_cd
                    n_best = tn
                    print(f"  length {tn}+CD improved: {best_score:.8f}")

    # Phase 4: Final polishing.
    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 60:
        t_alloc = min(float(rem) * 0.30, 150.0)
        print(f"Phase 4 fine Adam: budget {t_alloc:.1f}s")
        a_opt, sc_opt = adam_optimize(best_seq, t_alloc,
                                      lr=0.002, temp_init=0.05, temp_decay=0.9998)
        a_opt, sc_opt = symmetrize_if_better(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            n_best = len(best_seq)
            print(f"  fine Adam improved: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 60:
        t_cd = min(float(rem) * 0.40, 180.0)
        print(f"Phase 4 final coordinate: budget {t_cd:.1f}s")
        a_cd, sc_cd = coordinate_descent_raw(best_seq, t_cd)
        a_cd, sc_cd = symmetrize_if_better(a_cd)
        if sc_cd < best_score:
            best_seq = a_cd.copy()
            best_score = sc_cd
            n_best = len(best_seq)
            print(f"  final CD improved: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 50:
        pert = normalize_nonneg(best_seq + rng.standard_normal(len(best_seq)) * 0.002)
        t_cd = min(float(rem) * 0.5, 80.0)
        a_pert, sc_pert = coordinate_descent_raw(pert, t_cd)
        a_pert, sc_pert = symmetrize_if_better(a_pert)
        if sc_pert < best_score:
            best_seq = a_pert.copy()
            best_score = sc_pert
            n_best = len(best_seq)
            print(f"  perturbed CD improved: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if best_seq is not None and rem > 60 and len(best_seq) <= 1024:
        t_sc = min(float(rem) * 0.6, 120.0)
        print(f"Phase 4 final SCAS: budget {t_sc:.1f}s")
        a_opt, sc_opt = self_consistent_optimize(best_seq, t_sc, max_dense_n=1024)
        if a_opt is not None:
            a_opt, sc_opt = symmetrize_if_better(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                n_best = len(best_seq)
                print(f"  final SCAS improved: {best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(768, dtype=np.float64) / 768.0

    best_seq = normalize_nonneg(best_seq)
    best_seq, final_score = symmetrize_if_better(best_seq)

    try:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            final_score = fs
    except Exception:
        pass

    if not np.isfinite(final_score):
        best_seq = np.ones(768, dtype=np.float64) / 768.0
        try:
            final_score = evaluate_sequence([float(x) for x in best_seq])
        except Exception:
            final_score = float("inf")

    print(f"FINAL: n={len(best_seq)}, score={final_score:.8f}")
    return [float(x) for x in best_seq]
