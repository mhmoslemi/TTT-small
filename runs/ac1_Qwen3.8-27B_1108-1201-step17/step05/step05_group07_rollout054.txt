Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating iteration
(the state-of-the-art approach), combined with log-softmax L-BFGS refinement,
multi-length exploration, and peak-shaving heuristics. The LP iteration
provides a fundamentally different optimization path than pure gradient methods,
exploiting the convex structure when one factor is fixed.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def cross_corr(w, a):
    """Cross-correlation for gradient computation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]

def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_convolution_lp(f, M):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for all k.
    Returns the optimal g (or None if infeasible).
    """
    n = len(f)
    import cvxpy as cp
    
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0]
    
    # Build convolution constraints efficiently
    # (f*g)[k] = sum_{i} f[i] * g[k-i] for valid i
    # We can vectorize this using the structure
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        g_idx = k - idx  # indices of g that participate
        constraints.append(f[idx] @ g[g_idx] <= M)
    
    prob = cp.Problem(objective, constraints)
    
    # Try solvers in order of preference
    for solver in [cp.CBC, cp.GLPK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                return np.maximum(g.value, 0.0)
        except Exception:
            continue
    
    return None

    """
    LP-based alternating iteration:
    1. M = max(f*f)
    2. Solve LP to get g maximizing sum(g) s.t. (f*g) <= M, g >= 0
    3. Rescale g to match sum(f)
    4. Update f <- (1-t)*f + t*g
    """
    t0 = time.monotonic()
    f = f.copy()
    current_sum = f.sum()
    f_norm = f / current_sum  # work with normalized (sum=1)
    
    best_f = f_norm.copy()
    best_max = fast_convolve(f_norm).max()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(f_norm)
        M = b.max()
        
        # Solve LP
        g = solve_convolution_lp(f_norm, M)
        if g is None or g.sum() < 1e-10:
            break
        
        # Rescale g to have sum = 1
        g = g / g.sum()
        
        # Check if g*g has a smaller max
        g_max = fast_convolve(g).max()
        
        # Update with step size
        t = step_size
        f_new = (1 - t) * f_norm + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        
        new_max = fast_convolve(f_new).max()
        
        if new_max < best_max:
            best_max = new_max
            best_f = f_new.copy()
            f_norm = f_new
        else:
            # Try smaller step
            t = t * 0.5
            if t < 0.01:
                break
            f_new = (1 - t) * f_norm + t * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-10:
                break
            f_new = f_new / s
            new_max = fast_convolve(f_new).max()
            if new_max < best_max:
                best_max = new_max
                best_f = f_new.copy()
                f_norm = f_new
            else:
                break
        
        # Gradually increase step size
        step_size = min(step_size * 1.1, 0.9)
    
    return best_f * current_sum, best_max

def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def peak_shave(a, n_peaks=5, n_steps=200, time_limit=30.0):
    """Peak shaving heuristic."""
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        
        improved = False
        for k in peak_indices:
            if b[k] <= best_obj * 0.98:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            
            # Try reducing the largest contributors
            top_local = np.argsort(contributions)[::-1][:3]
            for tl in top_local:
                i = idx[tl]
                j = k - i
                for target in [i, j]:
                    old_val = a[target]
                    if old_val < 1e-10:
                        continue
                    for factor in [0.9, 0.8, 0.7]:
                        new_val = old_val * factor
                        diff = old_val - new_val
                        mask = np.ones(n, dtype=bool)
                        mask[target] = False
                        others = a[mask]
                        if others.sum() < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[target] = new_val
                        a_trial[mask] = others + diff * others / others.sum()
                        a_trial = a_trial / a_trial.sum()
                        val = fast_convolve(a_trial).max()
                        if val < best_obj - 1e-12:
                            a = a_trial
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit=30.0):
    """Coordinate descent with multiple multipliers."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(0, n, max(1, n // 500)):  # subsample for speed
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj

def adam_simplex(a0, time_limit=60.0, lr=0.01, top_k=50):
    """
    Adam optimizer on the simplex with subgradient of the smoothed max.
    Different from L-BFGS: uses adaptive per-coordinate learning rates.
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_temp = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_convolve(a)
        
        # Soft-max weights on top peaks
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        # Adam update
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg**2
        update = m / (np.sqrt(v) + eps)
        
        a = project_simplex(a - lr_t * update)
        co = fast_convolve(a).max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_temp *= 1.005
        lr *= 0.9995
        
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            cur_temp = 0.01
            lr = 0.01
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating iteration
(state-of-the-art approach), Adam-on-simplex refinement, log-softmax L-BFGS
with temperature annealing, multi-length exploration, and peak-shaving.
The LP iteration exploits the convex structure when one factor is fixed,
providing a fundamentally different optimization path than pure gradients.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def cross_corr(a, b):
    """(A^T v)[i] = sum_k a[k-i]*v[k] for valid k."""
    n = len(a)
    result = np.zeros(n)
    for i in range(n):
        k_start = max(0, i)
        k_end = min(2 * n - 2, i + n - 1)
        if k_start > k_end:
            continue
        k_range = np.arange(k_start, k_end + 1)
        result[i] = np.dot(a[k_range - i], b[k_range])
    return result


def cross_corr_fft(w, a):
    """Cross-correlation via FFT: result[i] = sum_j w[j]*a[j+i] (valid range)."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the convolution constraint matrix A where (A@g)[k] = conv(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(2 * n - 1):
        j_start = max(0, i - n + 1)
        j_end = min(n - 1, i)
        for j in range(j_start, j_end + 1):
            A[i, j] = f[i - j]
    return A


def solve_lp_conv(f, M, timeout=30.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for all k.
    Uses scipy linprog with HiGHS solver. Returns g or None.
    """
    t_start = time.monotonic()
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_qp_flat(f, max_iter=5000, time_limit=30.0):
    """
    Solve QP: min ||conv(f,g)||_2^2 subject to g >= 0, sum(g) = 1.
    Uses projected gradient with backtracking line search.
    """
    t0 = time.monotonic()
    n = len(f)
    rng = np.random.default_rng(42)
    g = project_simplex(0.5 + rng.standard_normal(n) * 0.5)
    
    # Compute smoothness constant (upper bound)
    f_self_max = fast_convolve(f).max()
    L = 4.0 * f_self_max + 1.0
    lr = 1.0 / L
    
    prev_obj = np.inf
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        fg = fast_convolve(f, g)
        obj = np.dot(fg, fg)
        grad = 2.0 * cross_corr(f, fg)
        
        # Backtracking line search
        step = lr
        g_trial = project_simplex(g - step * grad)
        fg_trial = fast_convolve(f, g_trial)
        obj_trial = np.dot(fg_trial, fg_trial)
        
        while obj_trial > obj + 0.5 * step * np.dot(grad, g_trial - g) and step > 1e-12:
            step *= 0.5
            g_trial = project_simplex(g - step * grad)
            fg_trial = fast_convolve(f, g_trial)
            obj_trial = np.dot(fg_trial, fg_trial)
        
        g = g_trial
        
        if abs(prev_obj - obj_trial) < 1e-14:
            break
        prev_obj = obj_trial
    
    return g


def lp_alternating_iteration(f0, n_iter=30, time_limit=300.0):
    """
    LP-based alternating iteration:
    1. M = max(f*f)
    2. Solve LP: max sum(g) s.t. (f*g) <= M, g >= 0
    3. Rescale g to sum=1
    4. Update f <- (1-t)*f + t*g
    """
    t0 = time.monotonic()
    n = len(f0)
    f = f0.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_max = fast_convolve(f).max()
    
    step_size = 0.5
    stagnant = 0
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(f)
        M = b.max()
        
        # Try LP first
        g = solve_lp_conv(f, M, timeout=20.0)
        source = "LP"
        
        if g is None or g.sum() < 1e-10:
            # Fallback to QP
            g = solve_qp_flat(f, max_iter=2000, time_limit=15.0)
            source = "QP"
        
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        g_max = fast_convolve(g).max()
        
        # Try different step sizes
        best_step = None
        best_new_max = best_max
        for t in [step_size, step_size * 0.5, step_size * 0.25, 0.8, 0.9]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_max = fast_convolve(f_new).max()
            if new_max < best_new_max:
                best_new_max = new_max
                best_step = t
        
        if best_step is not None:
            f = (1 - best_step) * f + best_step * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            if best_new_max < best_max:
                best_max = best_new_max
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            step_size = min(step_size * 1.05, 0.95)
        else:
            stagnant += 1
            if stagnant > 5:
                # Perturb and try again
                rng = np.random.default_rng(it)
                f = project_simplex(f + rng.standard_normal(n) * 0.01)
                stagnant = 0
            else:
                step_size *= 0.7
    
    return best_f, best_max


def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr_fft(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):

            return np.zeros_like(z)

            fun=func, x0=z0, jac=jac,

                     'maxfun': maxiter * 25, 'disp': False}

        return res.x, res.fun

def adam_simplex(a0, time_limit=60.0, lr_init=0.01, top_k=50, temp_init=0.01):

    lr = lr_init
    no_improve = 0

        t += 1
        b = fast_convolve(a)
        
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        lr_t = lr * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg ** 2
        update = m / (np.sqrt(v) + eps)
        
        a = project_simplex(a - lr_t * update)
        co = fast_convolve(a).max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_temp *= 1.005
        lr *= 0.9995
        
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 10**6)

            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            cur_temp = temp_init
            lr = lr_init
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalized sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                        if others.sum() < 1e-15:
                            continue

                        a_trial = a_trial / a_trial.sum()
                        val = fast_convolve(a_trial).max()
                        if val < best_obj - 1e-13:
                            a = a_trial
                            best_obj = val

                        break

                    break

                break

def coordinate_descent(a0, time_limit=45.0):
    """Coordinate descent cycling through all positions with multipliers."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = np.array([0.6, 0.8, 0.9, 1.1, 1.25, 1.5])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-13:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj


def make_structured_init(n, kind="uniform"):
    """Create structured initializations."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "semicircle":
        x = np.linspace(-1, 1, n)
        f = np.sqrt(np.maximum(1 - x ** 2, 0))
        return f / f.sum()
    elif kind == "gaussian":
        x = np.linspace(-3, 3, n)
        f = np.exp(-x ** 2 / 2)
        return f / f.sum()
    elif kind == "bimodal":
        x = np.linspace(0, 1, n)
        f = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01)
        return f / f.sum()
    elif kind == "triangle":
        x = np.linspace(0, 1, n)
        f = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
        return f / f.sum()
    elif kind == "edge_conc":
        x = np.linspace(0, 1, n)
        f = 1.0 / (0.1 + np.abs(x - 0.5))
        return f / f.sum()
    else:
        return np.ones(n) / n

        per_T = total_budget / len(temps)
        
        for T in temps:
            rem = deadline - time.monotonic() - 200
            if rem < 10:
                break
            t_alloc = min(per_T, 80.0)
            print(f"  L-BFGS n={n_b}, T={T}, t={t_alloc:.0f}s")
            z_curr, _ = lbgfs_optimize(z_curr, T, n_b, t_alloc, maxiter=3000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, f"lbgfs_T{T}")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = []
        n_best = len(best_seq)
        for tn in [256, 512, 768, 1024, 1536, 2048, 3072]:
            if tn != n_best:
                target_lengths.append(tn)
        
        # Sort by proximity to current best (try nearby sizes first)
        target_lengths.sort(key=lambda x: abs(x - n_best))
        
        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            
            f_interp = interpolate_sequence(best_seq, target_n)
            z0 = z_from_a(f_interp)
            
            # Quick LP iteration at new length
            t_lp = min(40.0, rem * 0.2)
            if t_lp > 10 and target_n <= 1536:
                a_lp, _ = lp_alternating_iteration(f_interp, n_iter=10, time_limit=t_lp)
                try_seq(a_lp, f"lp_n{target_n}")
                if best_seq is not None and len(best_seq) == target_n:
                    f_interp = best_seq
                    z0 = z_from_a(f_interp)
            
            # L-BFGS refinement
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                continue
            t_lbg = min(60.0, rem * 0.3)
            z_curr, _ = lbgfs_optimize(z0, 20, target_n, t_lbg, maxiter=2000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, f"lbgfs_n{target_n}_T20")
            
            # Higher T refinement
            rem = deadline - time.monotonic() - 80
            if rem < 10:
                continue
            t_lbg2 = min(45.0, rem * 0.3)
            z_curr, _ = lbgfs_optimize(z_curr, 100, target_n, t_lbg2, maxiter=2000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, f"lbgfs_n{target_n}_T100")
    
    # === Phase 5: Peak shaving + coordinate descent ===
    print(f"\n=== Phase 5: Peak Shaving + CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_ps = min(50.0, rem * 0.35)
            a_ps, _ = peak_shave(best_seq, n_peaks=10, n_steps=400, time_limit=t_ps)
            try_seq(a_ps, "peak_shave")
            
            if best_seq is not None:
                rem = deadline - time.monotonic() - 50
                if rem > 15:
                    t_cd = min(45.0, rem * 0.5)
                    a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd)
                    try_seq(a_cd, "cd")
    
    # === Phase 6: Final deep refinement ===
    print(f"\n=== Phase 6: Final Deep Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        
        # Very high T L-BFGS (near-exact max)
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            z0 = z_from_a(best_seq)
            t_alloc = min(40.0, rem * 0.4)
            z_final, _ = lbgfs_optimize(z0, 500, n_b, t_alloc, maxiter=3000)
            a_final = softmax_from_z(z_final)
            try_seq(a_final, "final_T500")
        
        # Adam refinement
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_adam = min(30.0, rem * 0.5)
            a_final2, _ = adam_simplex(best_seq, time_limit=t_adam, lr_init=0.003,
                                       top_k=max(50, n_b // 6), temp_init=0.002)
            try_seq(a_final2, "final_adam")
        
        # Random restarts + quick optimization
        rem = deadline - time.monotonic() - 10
        for attempt in range(5):
            if rem < 8:
                break
            noise_scale = 0.003 * (1 + attempt * 0.5)
            noise = rng.standard_normal(n_b) * noise_scale
            a_pert = project_simplex(best_seq + noise)
            z_pert = z_from_a(a_pert)
            t_alloc = min(12.0, rem * 0.4)
            z_p, _ = lbgfs_optimize(z_pert, 200, n_b, t_alloc, maxiter=1500)
            a_p = softmax_from_z(z_p)
            try_seq(a_p, f"restart_{attempt}")
            rem = deadline - time.monotonic() - 10
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
