def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_with(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(f)

    best_f = f.copy()
    
    t0 = time.monotonic()
    eps = eps_init
    lr = lr_init
    no_improve = 0
    it = 0
    
    # Polyak step size estimate

        if M < best_obj - 1e-14:

            best_f = f.copy()
            no_improve = 0

        else:

        # Compute subgradient

            lr = lr_init * 0.5
            eps = eps_init * 0.1
        
        if lr < 1e-8:

    return best_f, best_obj

        # So A_ub[j+j, j] = f[0], A_ub[j+1, j] = f[1], ..., A_ub[j+n-1, j] = f[n-1]
        # Wait: constraint k involves g[j] when k-j is in [0, n-1], i.e., k in [j, j+n-1]
        # Coefficient is f[k-j]
        start_k = j
        end_k = j + n  # exclusive
        A_ub[start_k:end_k, j] = f[:n]
    
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    
    options = {'presolve': True}
    if previous_g is not None:
        # Use as initial guess (HiGHS doesn't support warm start directly in scipy)
        pass
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):

def mirror_descent_smooth(f0, time_limit, verbose=False, eps_init=0.005):
    """Mirror descent using smooth max of convolution."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    eps = eps_init
    lr = 0.002
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, eps={eps:.6f}")
        else:
            no_improve += 1
        
        # Compute gradient of smooth max
        sg = compute_subgrad(f, conv, eps, top_k=60)
        
        # Multiplicative (mirror) update
        log_f = np.log(f)
        log_f = log_f - lr * sg
        f = np.exp(log_f - log_f.max())  # For numerical stability
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        lr *= 0.9998
        eps *= 0.998
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 23 + 17)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.001
            eps = eps_init * 0.5
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_f, best_obj


def coordinate_descent_fast(f0, time_limit, verbose=False):
    """Fast coordinate descent: try adjusting pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        
        # Find the peak lag
        k_peak = int(np.argmax(conv))
        
        # For coordinate descent, try adjusting individual coordinates
        # Focus on coordinates that contribute to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        if je >= js:
            # Try reducing the largest contributor to the peak
            idx = np.arange(js, je + 1)
            contributions = f[idx] * f[k_peak - idx]
            
            # Sort by contribution
            n_try = min(5, len(idx))
            top_contrib = idx[np.argsort(contributions)[-n_try:]]
            
            for j in top_contrib:
                j = int(j)
                old_val = f[j]
                if old_val < 1e-10:
                    continue
                
                # Try reducing f[j] and adding to f at a position that doesn't contribute to peak
                for frac in [0.7, 0.85, 0.5, 0.9]:
                    new_val = old_val * frac
                    diff = old_val - new_val
                    if diff < 1e-12:
                        continue
                    
                    trial = f.copy()
                    trial[j] = new_val
                    # Add diff to the position furthest from k_peak
                    # Find position with minimum contribution to peak
                    if k_peak - j >= 0 and k_peak - j < n:
                        # The partner position
                        pass
                    # Add to the minimum element
                    min_idx = np.argmin(f)
                    if min_idx == j:
                        min_idx = (min_idx + 1) % n
                    trial[min_idx] += diff
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()
                    
                    val = fast_convolve(trial).max()
                    if val < best_obj - 1e-14:
                        f = trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
        
        # Check time
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Create a bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def weighted_averaging_subgrad(f0, time_limit, verbose=False):
    """
    Subgradient method with weighted averaging (like in Nesterov's book).
    Uses a decreasing step size scheme and averages the iterates.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    # Phase 1: Find a good initial M estimate
    conv0 = fast_convolve(f)
    M0 = conv0.max()
    
    # Lower bound: by Cauchy-Schwarz, max conv >= (sum f)^2 / (2n-1) = 1/(2n-1)
    Mlb = 1.0 / (2 * n - 1)
    
    t0 = time.monotonic()
    f_avg = f.copy() * 0.0
    weight_sum = 0.0
    best_obj = M0
    best_f = f.copy()
    it = 0
    
    # Use Polyak step size: t_k = (M_k - M_lb) / ||g_k||^2
    eps = 0.01
    eps_min = 0.0001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute subgradient
        sg = compute_subgrad(f, conv, eps, top_k=50)
        sg_norm = np.linalg.norm(sg)
        
        if sg_norm < 1e-15:
            break
        
        # Polyak step size
        gap = M - Mlb
        if gap < 1e-10:
            break
        t_step = gap / (sg_norm ** 2)
        t_step = min(t_step, 0.1)  # Cap the step
        
        # Gradient step with projection
        f_new = f - t_step * sg
        f_new = project_simplex(f_new)
        
        # Update average
        w_k = 1.0 / (it + 1)
        f_avg = (1 - w_k) * f_avg + w_k * f_new
        weight_sum += w_k
        
        f = f_new
        eps = max(eps * 0.999, eps_min)
        
        # Evaluate
        conv_new = fast_convolve(f)
        M_new = conv_new.max()
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f.copy()
            if verbose and it % 100 == 0:
                print(f"    WSG iter {it}: obj={best_obj:.10f}, step={t_step:.6f}")
        
        # Also check the average
        if it % 50 == 0:
            conv_avg = fast_convolve(f_avg)
            M_avg = conv_avg.max()
            if M_avg < best_obj - 1e-14:
                best_obj = M_avg
                best_f = f_avg.copy()
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient descent with smooth max (fast warm-up)
2. LP-based Frank-Wolfe (main optimization engine)
3. Mirror descent refinement
4. Multi-length exploration with interpolation
5. Perturbation-based restarts to escape local minima
6. Coordinate descent polish
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_weights(v, eps):
    """Compute normalized weights for smooth max approximation."""
    v_max = v.max()
    w = np.exp((v - v_max) / eps)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(v))
        w[int(np.argmax(v))] = 1.0
    else:
        w /= w_sum
    return w


def compute_subgrad(f, conv, eps, top_k=60):
    """Compute subgradient of smooth max of auto-convolution w.r.t. f."""
    n = len(f)
    w_all = smooth_max_weights(conv, eps)

    if top_k < len(conv):
        top_indices = np.argpartition(w_all, -top_k)[-top_k:]
        w_top = w_all[top_indices]
    else:
        top_indices = np.arange(len(conv))
        w_top = w_all

    sg = np.zeros(n)
    for i, k in enumerate(top_indices):
        wi = w_top[i]
        if wi < 1e-15:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        sg[idx] += wi * f[k - idx]

    return 2.0 * sg


def subgradient_descent(f0, time_limit, verbose=False, eps_init=0.008):
    """Subgradient descent with smooth max approximation."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    eps = eps_init
    lr = 0.0008
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}, eps={eps:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1

        sg = compute_subgrad(f, conv, eps, top_k=80)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Polyak-like step size
        gap = M - 1.0 / (2 * n - 1)
        t_step = min(gap / (sg_norm ** 2 + 1e-15), lr)
        t_step = max(t_step, lr * 0.1)

        f_new = f - t_step * sg
        f_new = project_simplex(f_new)
        f = f_new

        lr *= 0.99995
        eps = max(eps * 0.9995, 0.001)

        if no_improve > 500:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_improve = 0
            lr = 0.0005
            eps = eps_init * 0.3

        if lr < 1e-9:
            break

    return best_f, best_obj


def build_lp_matrix(n):
    """Prebuild the constraint matrix structure for the LP."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = 1.0
    return A


def solve_lp_direction(f, M, n, A_ub_template=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)

    if A_ub_template is None:
        A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A_ub[j:j + n, j] = f
    else:
        # Fill in the f values using the template structure
        A_ub = A_ub_template.copy()
        for j in range(n):
            A_ub[j:j + n, j] = f

    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success and res.fun is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with fast line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Solve LP
        lp_t0 = time.monotonic()
        g = solve_lp_direction(f, M * m_exp, n)
        lp_time = time.monotonic() - lp_t0

        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        # Quick gap check
        gap = np.max(np.abs(g - f))
        if gap < 1e-13:
            no_improve += 1
            if no_improve > 40:
                m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 120:
                rng = np.random.default_rng(it * 19 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0015)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Fast line search: try full step first, then halve
        best_alpha = 0.0
        best_val = M

        # Try alpha = 1.0 first
        ft = g.copy()
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = 1.0

        if best_alpha < 1.0:
            # Try decreasing alphas
            for alpha in [0.7, 0.4, 0.2]:
                ft = (1 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 0 and best_alpha < 1.0 - 1e-6:
            # Fine binary search
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = (1 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            M = conv_new.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.3f}, "
                          f"lp={lp_time:.3f}s, elapsed={elapsed:.0f}s")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if no_improve > 150:
            break

    return best_f, best_obj


def mirror_descent(f0, time_limit, verbose=False, eps_init=0.003):
    """Mirror descent (multiplicative weights) with smooth max."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    eps = eps_init
    lr = 0.0015
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, eps={eps:.6f}")
        else:
            no_improve += 1

        sg = compute_subgrad(f, conv, eps, top_k=50)

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        lr *= 0.9998
        eps = max(eps * 0.999, 0.0005)

        if no_improve > 400:
            rng = np.random.default_rng(it * 23 + 17)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.001
            eps = eps_init * 0.5
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Targeted coordinate descent on peak-contributing elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k_peak = int(np.argmax(conv))
        improved = False

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        if je >= js:
            idx = np.arange(js, je + 1)
            contributions = f[idx] * f[k_peak - idx]
            n_try = min(4, len(idx))
            top_contrib = idx[np.argsort(contributions)[-n_try:]]

            for j in top_contrib:
                j = int(j)
                old_val = f[j]
                if old_val < 1e-10:
                    continue

                for frac in [0.75, 0.6, 0.85, 0.5]:
                    new_val = old_val * frac
                    diff = old_val - new_val
                    if diff < 1e-12:
                        continue

                    trial = f.copy()
                    trial[j] = new_val
                    # Redistribute to minimize peak
                    # Find position with lowest contribution to top peaks
                    min_pos = np.argmin(f)
                    if min_pos == j:
                        min_pos = (min_pos + 1) % n
                    trial[min_pos] += diff
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()

                    val = fast_convolve(trial).max()
                    if val < best_obj - 1e-14:
                        f = trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Create a bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

    # === Phase 8: Final LP-FW squeeze ===
    print(f"\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
