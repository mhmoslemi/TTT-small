```python
def smoothed_gradient(f, T, top_k=50):
    """Compute the smoothed gradient of the log-sum-exp objective."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Get top k values for efficiency
    if top_k < len(conv):
        # Use log-sum-exp trick
        top_k_idx = np.argpartition(conv, -top_k)[-top_k:]
        # Compute weights
        vals = conv[top_k_idx]
        # Include the rest as a lump
        max_val = vals.max()
        exp_vals = np.exp((vals - max_val) / T)
        # For the rest, use exp((conv_rest - max_val)/T) which is small
        # Actually for accuracy, let's just use all values if T is small enough
        # For large T, we can use all values
        
    # Better: just compute all weights directly
    vals = conv - conv.max()  # for numerical stability
    exp_vals = np.exp(vals / T)
    w = exp_vals / exp_vals.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    # For each k with significant weight, add contribution
    significant = w > 1e-10
    for k in np.where(significant)[0]:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[k] * f[k - idx]
    
    return grad
```

This is O(n * num_significant_peaks) per iteration. If num_significant_peaks is small (say 50), this is O(50n) which is fine for n up to a few thousand.

```python
def smoothed_multiplicative_descent(f0, T_init, T_final, lr_init, lr_final, num_steps, time_limit):
    f = f0 / f0.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for step in range(num_steps):
        # Anneal temperature
        progress = step / num_steps
        T = T_init * (T_final / T_init) ** progress
        lr = lr_init * (lr_final / lr_init) ** progress
        
        # Compute convolution
        conv = fast_convolve(f)
        
        # Compute log-sum-exp gradient
        vals = conv - conv.max()
        exp_vals = np.exp(vals / T)
        w = exp_vals / exp_vals.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                idx = np.arange(js, je + 1)
                grad[idx] += 2.0 * w[k] * f[k - idx]
        
        # Multiplicative update
        f = f * np.exp(-lr * grad)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        # Track best
        obj = fast_convolve(f).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
```

```python
def compute_smoothed_gradient(f, conv, T):
    """Compute the smoothed gradient using FFT-based cross-correlation."""
    n = len(f)
    
    # Compute weights
    vals = conv - conv.max()
    exp_vals = np.exp(vals / T)
    w = exp_vals / exp_vals.sum()
    
    # F_rev = reversed f
    F_rev = f[::-1]
    
    # Convolve w (length 2n-1) with F_rev (length n) -> length 3n-2
    size = len(w) + n - 1  # = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(F_rev, fft_size)
    conv_result = np.fft.irfft(W * F, fft_size)[:size]
    
    # Extract grad: grad_j = 2 * conv_result[j + n - 1] for j = 0..n-1
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    
    return grad
```

```python
import numpy as np
from scipy.optimize import linprog
```

```python
def fast_convolve(a):
    """Full linear auto-convolution via real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_grad(f, T, conv):
    """
    Compute smoothed gradient of log-sum-exp of auto-convolution.
    grad_j = 2 * sum_m w_{j+m} * f[m] computed via FFT cross-correlation.
    O(n log n) complexity.
    """
    n = len(f)
    # Softmax weights over peaks
    c = conv - conv.max()
    exp_c = np.exp(c / T)
    w = exp_c / exp_c.sum()

    # Cross-correlation via FFT: grad_j = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]
    F_rev = f[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(F_rev, fft_size)
    result = np.fft.irfft(W * F, fft_size)[:size]
    grad = 2.0 * result[n - 1: 2 * n - 1]
    return grad

def gs_initialize(n, num_iters=200, seed=42):
    """
    Gerchberg-Saxton initialization: alternate between time-domain
    (non-negativity + normalization) and frequency-domain (flat spectrum)
    projections to find a distribution with low auto-convolution peak.
    """
    rng = np.random.default_rng(seed)
    p = np.abs(rng.standard_normal(n)) + 0.1
    p = p / p.sum()

    L = 1 << (2 * n - 1).bit_length()

    for _ in range(num_iters):
        # Frequency domain: flatten |FFT|
        P = np.fft.rfft(p, L)
        # Target: constant magnitude (keeping DC)
        P[0] = 1.0  # sum(p) = 1
        mags = np.abs(P)
        mags[0] = 1.0
        # Use geometric mean as target
        target = np.median(mags[1:]) if len(mags) > 1 else mags[0]
        # Update: keep phase, set magnitude
        for k in range(1, len(P)):
            if mags[k] > 1e-15:
                P[k] = P[k] / mags[k] * target
            else:
                P[k] = target * np.exp(1j * rng.standard_normal() * 0.01)
        # Time domain: project onto non-negative simplex
        p = np.fft.irfft(P, L)[:n]
        p = np.maximum(p, 0.0)
        ps = p.sum()
        if ps < 1e-15:
            p = np.ones(n) / n
        else:
            p = p / ps

    return p

                    f[j_src] -= best_delta
                    f[j_dst] += best_delta
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    best_obj = best_trial_obj
                    best_f = f.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Try a different peak
                conv = fast_convolve(f)
                peak_vals = np.sort(conv)[::-1]
                if len(peak_vals) > 1 and abs(peak_vals[0] - peak_vals[1]) < 1e-8 * peak_vals[0]:
                    # Multiple similar peaks: try reducing all simultaneously
                    top_peaks = np.argsort(conv)[::-1][:3]
                    for k_p in top_peaks:
                        k_p = int(k_p)
                        js2 = max(0, k_p - n + 1)
                        je2 = min(n - 1, k_p)
                        idx2 = np.arange(js2, je2 + 1)
                        prods2 = f[idx2] * f[k_p - idx2]
                        j_max2 = idx2[np.argmax(prods2)]
                        if f[j_max2] < 1e-10:
                            continue
                        # Reduce this position slightly
                        delta = f[j_max2] * 0.005
                        trial = f.copy()
                        trial[j_max2] -= delta
                        # Redistribute to minimum positions
                        min_idx = np.argsort(f)[:5]
                        redistribution = delta / len(min_idx)
                        for mi in min_idx:
                            if mi != j_max2:
                                trial[mi] += redistribution
                        trial = np.maximum(trial, 0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-13:
                            f = trial
                            best_obj = obj
                            best_f = f.copy()
                            improved = True
                            break
                if not improved:
                    break
        else:
            no_improve = 0

    return best_f, best_obj


def lp_fw_step(f, M, n):
    """Single LP-Frank-Wolfe direction: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = _time.monotonic()
    m_exp = 1.0
    no_improve = 0
    it = 0

    while _time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_fw_step(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 80:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            M = conv_new.max()
            f = f_new
            m_exp = 1.0
            no_improve = 0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    """Bell-shaped initial with flat top."""
    x = np.arange(n) - n / 2.0
    half_w = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = 1.0 - t ** 6
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length via linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: FFT-accelerated smoothed descent (primary),
    peak-targeted reduction, LP-Frank-Wolfe, GS initialization, and
    multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = _time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    # === Phase 1: Quick evaluation of diverse initializations ===
    print("=== Phase 1: Initializations ===")
    init_best = None
    init_best_score = float('inf')

    init_list = []
    if parent is not None:
        init_list.append(parent)

    for n_try in [n_parent, 512, 1024]:
        init_list.append(make_gaussian(n_try, 0.25))
        init_list.append(make_bell(n_try, 0.5))

    # GS initializations
    for n_try in [n_parent, 768]:
        gs_p = gs_initialize(n_try, num_iters=100, seed=seed + n_try)
        init_list.append(gs_p)

    for a_init in init_list:
        a_norm = a_init / a_init.sum()
        sc = try_seq(a_norm, "init")
        obj_local = 2 * len(a_norm) * fast_convolve(a_norm).max()
        if np.isfinite(obj_local) and obj_local < init_best_score:
            init_best_score = obj_local
            init_best = a_norm

    if init_best is None:
        init_best = parent if parent is not None else make_gaussian(768)

    print(f"  Best init local obj: {init_best_score:.8f} (n={len(init_best)})")

    # === Phase 2: Main smoothed descent (primary method) ===
    print(f"\n=== Phase 2: Smoothed Descent (n={len(init_best)}) ===")
    rem = deadline - _time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 400)
        print(f"  Alloc: {t_alloc:.0f}s")
        a_opt, obj = smoothed_descent(init_best, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "sd_main")

    # === Phase 3: Peak-targeted reduction ===
    print(f"\n=== Phase 3: Peak Targeted Reduction ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, obj = peak_targeted_reduction(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "peak_reduce")

    # === Phase 4: Second smoothed descent round (different T) ===
    print(f"\n=== Phase 4: Second SD Round ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  Alloc: {t_alloc:.0f}s")
            # Perturb slightly before second round
            noise = rng.standard_normal(len(best_seq)) * 0.001
            f_pert = np.maximum(best_seq + noise, 0)
            f_pert = f_pert / f_pert.sum()
            a_opt, obj = smoothed_descent(f_pert, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "sd_round2")

    # === Phase 5: Try longer sequence ===
    print(f"\n=== Phase 5: Longer Sequence (n=1024) ===")
    base = best_seq if best_seq is not None else init_best
    rem = deadline - _time.monotonic() - 250
    if rem > 60:
        f_1024 = interpolate_to_length(base, 1024)
        try_seq(f_1024, "interp_1024")
        t_alloc = min(rem * 0.4, 120)
        print(f"  SD n=1024: {t_alloc:.0f}s")
        a_opt, obj = smoothed_descent(f_1024, time_limit=t_alloc)
        try_seq(a_opt, "sd_1024")

    # === Phase 6: LP-FW polish ===
    print(f"\n=== Phase 6: LP-FW Polish ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 7: Final peak reduction ===
    print(f"\n=== Phase 7: Final Peak Reduction ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, obj = peak_targeted_reduction(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_final")

    # === Phase 8: Final smoothed descent squeeze ===
    print(f"\n=== Phase 8: Final SD Squeeze ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem, 60)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, obj = smoothed_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "sd_final")

    # === Phase 9: Try n=512 variant ===
    print(f"\n=== Phase 9: n=512 Variant ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 40
        if rem > 30:
            f_512 = interpolate_to_length(best_seq, 512)
            try_seq(f_512, "interp_512")
            t_alloc = min(rem * 0.5, 40)
            a_opt, obj = smoothed_descent(f_512, time_limit=t_alloc)
            try_seq(a_opt, "sd_512")

    # === Phase 10: Final LP-FW ===
    print(f"\n=== Phase 10: Final LP-FW ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 20)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "lpfw_final")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(384, 0.3)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
```
