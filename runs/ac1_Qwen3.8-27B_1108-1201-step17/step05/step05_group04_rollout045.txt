```python
def lp_step(f, M):
    """Given f (length n) and bound M, find g that maximizes sum(g)
    subject to (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_i f_i * g_{k-i} where 0 <= i < n and 0 <= k-i < n
    
    # Build constraint matrix A (2n-1 x n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    b = np.full(2*n-1, M)
    
    # g >= 0 handled by bounds
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return result.x
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response iteration,
diverse initializations, adaptive step sizes, length exploration, and 
coordinate descent polishing.
"""

import numpy as np
from scipy import optimize, sparse
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n, f):
    """Build sparse constraint matrix for LP: (f*g)_k <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            if f[i] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A


def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    A = build_lp_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, 
                           bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return np.ones(n) / n  # fallback


def lp_iteration(f, t, use_lp=True):
    """One iteration of the LP-based best response."""
    n = len(f)
    f = f / f.sum()
    M = fast_convolve(f).max()
    
    if M < 1e-15:
        return f
    
    if use_lp:
        g = lp_solve(f, M)
        gs = g.sum()
        if gs < 1e-15:
            g = f.copy()
            gs = 1.0
        g = g / gs
    else:
        g = f.copy()
    
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    fs = f_new.sum()
    if fs < 1e-15:
        f_new = np.ones(n) / n
    else:
        f_new = f_new / fs
    
    return f_new


def compute_subgradient(f, temp, top_k=None):
    """Compute weighted subgradient using soft-max."""
    n = len(f)
    conv = fast_convolve(f)
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_m = np.zeros(len(conv))
        w_m[idx] = w[idx]
        w = w_m / w_m.sum()
    
    # Subgradient: sg[j] = sum_i w[i+j] * f[i]
    # This is cross-correlation of w and f
    sg = np.correlate(w, f, 'valid')
    return sg, conv.max()


def adam_optimize(f0, time_limit, lr=0.003, temp_init=0.005, 
                  temp_decay=0.9994, top_k=None, beta1=0.9, beta2=0.999):
    """Adam optimizer with subgradient method."""
    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    temp = temp_init
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t_count += 1
        sg, cur_obj = compute_subgradient(f, temp, top_k)
        
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f_new)
        
        temp = max(temp * temp_decay, 1e-7)
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng = np.random.default_rng(t_count * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent polishing."""
    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            # Try several multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response iteration,
Adam with subgradients, coordinate descent polishing, length exploration,
and diverse initializations including polynomial shapes.
"""

import numpy as np
from scipy import optimize, sparse
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n, f):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            if f[i] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(f, M):
    n = len(f)
    A = build_lp_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = optimize.linprog(c, A_ub=A, b_ub=b,
                           bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return np.ones(n) / n


def lp_iteration(f, t):
    n = len(f)
    f = f / f.sum()
    conv = fast_convolve(f)
    M = conv.max()
    if M < 1e-15:
        return f
    try:
        g = lp_solve(f, M)
        gs = g.sum()
        if gs > 1e-10:
            g = g / gs
        else:
            return f
    except Exception:
        return f
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    fs = f_new.sum()
    if fs < 1e-15:
        f_new = np.ones(n) / n
    else:
        f_new = f_new / fs
    return f_new


def compute_subgradient(f, temp, top_k=None):
    n = len(f)
    conv = fast_convolve(f)
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_m = np.zeros(len(conv))
        w_m[idx] = w[idx]
        w = w_m / w_m.sum()
    sg = np.correlate(w, f, 'valid')
    return sg, conv.max()


def adam_optimize(f0, time_limit, lr=0.003, temp_init=0.005,
                  temp_decay=0.9994, top_k=None, beta1=0.9, beta2=0.999):

    f = f / f.sum()
    n = len(f)
    m = np.zeros(n)
    v = np.zeros(n)

    temp = temp_init
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

        t_count += 1
        sg, _ = compute_subgradient(f, temp, top_k)
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f_new)

def coordinate_descent(f0, time_limit):
    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            multipliers = [0.4, 0.6, 0.8, 0.9, 1.1, 1.2, 1.4, 1.6]

                new_val = old_val * mult

                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                val = fast_convolve(f_trial).max()

                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved and passes > 3:
            break
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    x = np.arange(n, dtype=np.float64) + 1
    f = x ** power
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_sine_init(n):
    x = np.linspace(0, 1, n)
    f = np.sin(np.pi * x) ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_gaussian_mix(n, rng, n_comp=4):

    x = np.linspace(0, 1, n)

        c = rng.uniform(0.15, 0.85)
        s = rng.uniform(0.08, 0.3)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def lp_optimize(f0, time_limit, t_init=0.3, t_min=0.05, t_decay=0.97):

            break
        t_alloc = min(rem * 0.4, 50)
        print(f"  LP+Adam on '{name}' (t={t_alloc:.0f}s)...")
        try:
            f_lp, _, _ = lp_optimize(init, time_limit=t_alloc * 0.6)
            try_seq(f_lp, f"lp_{name}")
        except Exception:
            pass
        rem = t_phase1_end - time.monotonic()
        if rem < 10:
            break
        start = best_seq if best_seq is not None else init
        a_opt, _ = adam_optimize(start, time_limit=min(rem * 0.5, 30),
                                 lr=0.003, temp_init=0.004,
                                 temp_decay=0.9995, top_k=max(15, n_parent // 14))
        try_seq(a_opt, f"adam_{name}")

    # === Phase 2: Deep LP on best ===
    print(f"\n=== Phase 2: Deep LP optimization (best={best_score:.8f}) ===")
    t_phase2_end = time.monotonic() + 300

    if best_seq is not None:
        f_cur = best_seq.copy()
        n_cur = len(f_cur)
        t_mix = 0.25
        while time.monotonic() < t_phase2_end:
            rem = t_phase2_end - time.monotonic()
            if rem < 2:
                break
            conv = fast_convolve(f_cur)
            M = conv.max()
            try:
                g = lp_solve(f_cur, M)
                gs = g.sum()
                if gs < 1e-10:
                    break
                g = g / gs
            except Exception:
                break
            f_new = (1 - t_mix) * f_cur + t_mix * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            new_obj = fast_convolve(f_new).max()
            f_cur = f_new
            try_seq(f_cur, "lp_deep")
            t_mix = max(t_mix * 0.98, 0.03)

    # === Phase 3: Adam with multiple configs ===
    print(f"\n=== Phase 3: Adam multi-config (best={best_score:.8f}) ===")
    t_phase3_end = time.monotonic() + 300

    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            (0.003, 0.004, 0.9995, max(15, n_best // 14)),
            (0.005, 0.006, 0.9993, max(25, n_best // 10)),
            (0.002, 0.003, 0.9996, max(10, n_best // 18)),
            (0.004, 0.005, 0.9994, None),
        ]
        for ci, (lr, ti, td, tk) in enumerate(configs):
            rem = t_phase3_end - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.55, 70)
            a_opt, _ = adam_optimize(best_seq, time_limit=t_alloc,
                                     lr=lr, temp_init=ti,
                                     temp_decay=td, top_k=tk)
            try_seq(a_opt, f"adam_c{ci}")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + 250

    if best_seq is not None:
        n_best = len(best_seq)
        targets = [512, 640, 768, 896, 1024, 1152, 1280, 1536]
        targets = [t for t in targets if t != n_best and 256 <= t <= 2048]
        targets.sort(key=lambda x: abs(x - n_best))
        targets = targets[:4]

        for tn in targets:
            rem = t_phase4_end - time.monotonic()
            if rem < 25:
                break
            f_int = interpolate_sequence(best_seq, tn)
            t_lp = min(rem * 0.5, 60)
            try:
                f_res, _, _ = lp_optimize(f_int, time_limit=t_lp)
                try_seq(f_res, f"lp_n{tn}")
            except Exception:
                try_seq(f_int, f"interp_n{tn}")
            rem = t_phase4_end - time.monotonic()
            if rem < 15:
                continue
            a_opt, _ = adam_optimize(best_seq if best_seq is not None else f_int,
                                     time_limit=min(rem * 0.5, 40),
                                     lr=0.003, temp_init=0.005,
                                     temp_decay=0.9994, top_k=max(15, tn // 12))
            try_seq(a_opt, f"adam_n{tn}")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: CD polishing (best={best_score:.8f}) ===")
    t_phase5_end = time.monotonic() + 200

    if best_seq is not None:
        rem = t_phase5_end - time.monotonic()
        if rem > 20:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.7, 140))
            try_seq(a_opt, "cd_main")

        rem = t_phase5_end - time.monotonic()
        if rem > 15 and best_seq is not None:
            n_best = len(best_seq)
            for pi in range(2):
                if time.monotonic() > t_phase5_end:
                    break
                pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.004)
                a_opt, _ = coordinate_descent(pert, time_limit=min(rem * 0.4, 40))
                try_seq(a_opt, f"cd_pert{pi}")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            n_best = len(best_seq)
            a_opt, _ = adam_optimize(best_seq, time_limit=min(rem * 0.5, 60),
                                     lr=0.001, temp_init=0.002,
                                     temp_decay=0.9998, top_k=max(8, n_best // 25))
            try_seq(a_opt, "final_adam")

        rem = deadline - time.monotonic() - 10
        if rem > 10 and best_seq is not None:
            n_best = len(best_seq)
            for fi in range(3):
                rem = deadline - time.monotonic() - 8
                if rem < 8:
                    break
                pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
                a_opt, _ = adam_optimize(pert, time_limit=min(rem * 0.4, 15),
                                         lr=0.002, temp_init=0.003,
                                         temp_decay=0.9997, top_k=max(8, n_best // 20))
                try_seq(a_opt, f"final_pert{fi}")

        # One more CD pass
        rem = deadline - time.monotonic() - 5
        if rem > 10 and best_seq is not None:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.7, 30))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
