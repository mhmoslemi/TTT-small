import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]

def build_conv_matrix(a):

    Then C @ g = conv(a, g)."""

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        C[k, idx] = a[k - idx]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def solve_lp_for_direction(a, M):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns the optimal g (normalized to non-negative).
    """
    n = len(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    # Build constraint matrix C where C[k,j] = a[k-j] if 0<=k-j<n else 0
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je+1)
            C[k, js:je+1] = a[k - idx]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5.0, 'presolve': True})
    
    if res.success:
        return np.maximum(res.x, 0)
    return None

    return best_a, best_M


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving relative shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: LP-based alternating optimization with multi-phase strategy."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or parent.sum() < 1e-10:
            raise ValueError
    except Exception:
        parent = np.ones(512)

    n_parent = len(parent)
    best_overall = None
    best_score = float('inf')

    def record(a, label):
        nonlocal best_overall, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_overall = a.copy()
            print(f"  NEW BEST [{label}]: score={score:.6f} (n={len(a)})")

    # === Phase 1: LP alternating on parent ===
    print(f"=== Phase 1: LP alternating, n={n_parent} ===")
    S = np.sqrt(2.0 * n_parent)
    a0 = (parent / parent.sum()) * S
    t1_limit = min(500.0, (deadline - time.monotonic()) * 0.45)
    if t1_limit > 20:
        a1, m1 = lp_alternate(a0, S, t1_limit, rng, verbose=True)
        record(a1, "lp_parent")

    # === Phase 2: More LP iterations from current best ===
    print(f"\n=== Phase 2: Deeper LP refinement ===")
    if best_overall is not None:
        n_best = len(best_overall)
        S_best = np.sqrt(2.0 * n_best)
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t2_limit = min(rem * 0.5, 300.0)
            a2, m2 = lp_alternate(best_overall, S_best, t2_limit, rng, verbose=True)
            record(a2, "lp_deeper")

    # === Phase 3: Random restarts ===
    print(f"\n=== Phase 3: Random restarts ===")
    if best_overall is not None:
        n_best = len(best_overall)
        S_best = np.sqrt(2.0 * n_best)
        rem = deadline - time.monotonic() - 200
        if rem > 60:
            n_restarts = min(4, int(rem / 60))
            for r in range(n_restarts):
                rem_c = deadline - time.monotonic() - 100
                if rem_c < 40:
                    break
                # Different random initializations
                if r % 3 == 0:
                    a_rand = rng.dirichlet(np.ones(n_best) * 0.3)
                elif r % 3 == 1:
                    a_rand = rng.dirichlet(np.ones(n_best) * 2.0)
                else:
                    a_rand = np.ones(n_best) / n_best
                    # Add structured perturbation
                    a_rand += 0.3 * np.sin(np.linspace(0, 4 * np.pi, n_best))
                    a_rand = np.maximum(a_rand, 0)
                    a_rand = a_rand / a_rand.sum()
                a_rand = a_rand * S_best
                t_alloc = min(rem_c * 0.35, 120.0)
                a_r, _ = lp_alternate(a_rand, S_best, t_alloc, rng, verbose=False)
                record(a_r, f"restart_{r}")

    # === Phase 4: Try larger length ===
    print(f"\n=== Phase 4: Larger length exploration ===")
    if best_overall is not None:
        for mult in [2, 3]:
            new_n = n_parent * mult
            if new_n > 3072:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 60:
                break
            S_new = np.sqrt(2.0 * new_n)
            a_interp = interpolate_to_length(best_overall, new_n)
            a_interp = a_interp * S_new
            t_alloc = min(rem * 0.4, 250.0)
            a_new, _ = lp_alternate(a_interp, S_new, t_alloc, rng, verbose=False)
            record(a_new, f"n{new_n}")

    # === Phase 5: Final polish on overall best ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_overall is not None:
        n_best = len(best_overall)
        S_best = np.sqrt(2.0 * n_best)
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t5 = min(rem, 100.0)
            a5, _ = lp_alternate(best_overall, S_best, t5, rng, verbose=True)
            record(a5, "polish")

    # Fallback
    if best_overall is None:
        best_overall = parent.copy()
        best_score = float(evaluate_sequence(best_overall.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_overall.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.6f} (n={len(best_overall)})")
    return [float(x) for x in best_overall]
