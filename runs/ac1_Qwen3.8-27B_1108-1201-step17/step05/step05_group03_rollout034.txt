"""
Minimizes 2*n*max(auto_conv)/sum^2 using alternating LP (linear programming
to find optimal direction), combined with coordinate descent polishing and
multi-start exploration. The LP at each step finds the best sequence b in
the simplex that minimizes max_k (a*b)_k given the current a, providing a
globally informed direction for improvement.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build the convolution matrix M where M[k,j] = a[k-j] if valid."""
    n = len(a)
    idx = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    valid = (idx >= 0) & (idx < n)
    M = np.zeros((2 * n - 1, n))
    M[valid] = a[idx[valid]]
    return M


def solve_lp_for_b(a):
    """Given a (in simplex), find b in simplex minimizing max_k (a*b)_k.
    Returns (b, t_opt)."""
    n = len(a)
    
    # Variables: b_0..b_{n-1}, t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Build constraint matrix
    M = build_conv_matrix(a)
    A_ub = np.hstack([M, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def alternating_lp_optimize(a, max_iters=500, damping=0.5, time_limit=300, verbose=False):
    """Alternating LP optimization with damping."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b, t_val = solve_lp_for_b(a)
        if b is None:
            break
        
        b = normalize(b)
        a_new = (1 - damping) * a + damping * b
        a = normalize(a_new)
        
        sc = compute_score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 50 == 0:
            print(f"    ALT-LP iter {it}: score={best_score:.10f}, t={t_val:.8f}")
    
    return best_a, best_score


def coord_descent_polish(a, time_limit=60, verbose=False):
    """Precise coordinate descent with golden-section search per coordinate."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            lo = max(0.0, old_val * 0.3)
            hi = min(0.99, old_val * 3.0)
            if hi - lo < 1e-6:
                lo = max(0.0, old_val - 0.02)
                hi = min(0.99, old_val + 0.02)
            
            def obj_v(v):
                v = max(0.0, min(1.0, v))
                new_a = a.copy()
                new_a[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                scale = (1.0 - v) / other_sum
                new_a[mask] *= scale
                new_a = np.maximum(new_a, 0)
                s = new_a.sum()
                if s < 1e-12:
                    return float('inf')
                new_a /= s
                conv = fast_convolve(new_a)
                return conv.max()
            
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-7, 'maxiter': 20})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        scale = (1.0 - v_new) / other_sum
                        new_a[mask] *= scale
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-12:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if verbose and pass_count % 3 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")
        
        if not improved and pass_count > 2:
            break
    
    return best_a, best_score


def geometric_sequence(n, r):
    a = (1.0 - r) * r ** np.arange(n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    x = np.linspace(0, np.pi, n)

def gaussian_sequence(n, sigma_frac=0.3):

def triangular_sequence(n):
    x = np.linspace(0, 1, n)

def interpolate_sequence(f, new_n):

    if old_n == new_n:

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

Actually, let me reconsider the alternating method. The standard approach is:
1. Given `a`, find `b` minimizing `max_k (a*b)_k` over `b` in simplex.
2. Given `b`, find `a` minimizing `max_k (a*b)_k` over `a` in simplex.

At step 1, the LP is as I described. At step 2, it's the same LP with roles reversed:
- Variables: `a_0, ..., a_{n-1}, t`

- Constraints: `(a*b)_k <= t` for all k, where `(a*b)_k = sum_i a_i * b_{k-i}`

But this is the same LP! The constraint `(a*b)_k = sum_i a_i * b_{k-i}` is the same whether we view it as "a convolved with b" or "b convolved with a" (for linear convolution, `a*b = b*a`).

So the alternating method is:
1. Solve LP with "fixed" coefficients = current `a`, find optimal `b`.

"""
Minimizes 2*n*max(auto_conv)/sum^2 using alternating LP (linear programming
to find globally optimal direction at each step), combined with precise
coordinate descent polishing, diverse initializations, and dimension
exploration. The key innovation over previous approaches is using LP subproblems
to find the best sequence in the simplex that minimizes cross-convolution peak,
providing a globally informed direction rather than a subgradient estimate.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2 (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix M where M[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    idx = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    valid = (idx >= 0) & (idx < n)
    M = np.zeros((2 * n - 1, n))
    M[valid] = a[idx[valid]]
    return M


def solve_lp_for_b(a):
    """Given a in simplex, find b in simplex minimizing max_k (a*b)_k.
    Returns (b, t_opt) or (None, None) on failure."""
    n = len(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    M = build_conv_matrix(a)
    A_ub = np.hstack([M, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        return res.x[:n], float(res.x[-1])
    return None, None


def alternating_lp_optimize(a, max_iters=500, damping_start=0.5, damping_end=0.1,
                            time_limit=300, verbose=False):
    """Alternating LP optimization with annealed damping."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Annealed damping
        frac = it / max(1, max_iters)
        damping = damping_start + (damping_end - damping_start) * frac

        b, t_val = solve_lp_for_b(a)
        if b is None:
            break

        b = normalize(b)
        a_new = (1.0 - damping) * a + damping * b
        a = normalize(a_new)

        sc = compute_score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > 200:
            # Perturb to escape
            a = normalize(a + np.random.standard_normal(n) * 0.001)
            no_improve_count = 0

        if verbose and it % 100 == 0:
            print(f"    ALT-LP iter {it}: score={best_score:.10f}, t={t_val:.8f}, damp={damping:.3f}")

    return best_a, best_score


def subgradient_step(a, lr, top_k=50):
    """One subgradient step on the peak of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]

    temp = max(1e-6, pv.max() - pv.min()) * 0.5
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    sg = np.zeros(n)
    for i in range(k_actual):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]

    a = np.maximum(a * np.exp(-lr * sg), 1e-15)
    a = a / a.sum()
    return a


def subgradient_optimize(a, time_limit=60, lr0=0.01, verbose=False):
    """Averaged subgradient optimization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    cur_lr = lr0
    avg_a = np.zeros(n)
    avg_count = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        a = subgradient_step(a, cur_lr, top_k=max(20, n // 10))
        avg_a += a
        avg_count += 1
        
        sc = compute_score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
        
        avg_candidate = avg_a / avg_count
        sc_avg = compute_score(avg_candidate)
        if sc_avg < best_score - 1e-13:
            best_score = sc_avg
            best_a = avg_candidate.copy()

        cur_lr *= 0.9995
        if step % 3000 == 0:
            cur_lr = max(cur_lr, lr0 * 0.05)

        if verbose and step % 2000 == 0:
            print(f"    SG step {step}: score={best_score:.10f}")

    return best_a, best_score


def coord_descent_polish(a, time_limit=60, verbose=False):
    """Precise coordinate descent with golden-section search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            lo = max(0.0, old_val * 0.4)
            hi = min(0.99, old_val * 2.5)
            if hi - lo < 1e-6:
                lo = max(0.0, old_val - 0.01)
                hi = min(0.99, old_val + 0.01)

            def obj_v(v):
                v = max(0.0, min(1.0, v))
                new_a = a.copy()
                new_a[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                scale = (1.0 - v) / other_sum
                new_a[mask] *= scale
                new_a = np.maximum(new_a, 0)
                s = new_a.sum()
                if s < 1e-12:
                    return float('inf')
                new_a /= s
                conv = fast_convolve(new_a)
                return conv.max()

            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-8, 'maxiter': 25})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        scale = (1.0 - v_new) / other_sum
                        new_a[mask] *= scale
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-13:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

        if not improved and pass_count > 2:
            break
        if pass_count > 100:
            break

    return best_a, best_score


def geometric_sequence(n, r):
    a = (1.0 - r) * r ** np.arange(n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def asymmetric_sequence(n, beta=1.5):
    """Asymmetric shape: one side decays faster."""
    x = np.arange(n) / n
    a = np.exp(-beta * x) * np.exp(-2.0 * (1.0 - x))
    return normalize(a)


def bimodal_sequence(n, pos1=0.3, pos2=0.7, width=0.15):
    """Bimodal distribution."""
    x = np.arange(n) / n
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    return normalize(a)


def triangular_sequence(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

            selected_keys.add(key)

    # Ensure parent included
    if not any(l == "parent" for _, l in selected):
        for sc, a, label in start_scores:
            if label == "parent" and sc < best_score * 1.02:
                selected.append((a, label))
                break

    n_selected = len(selected)
    avail_time = deadline - time.monotonic() - 400
    t_per = max(60, min(avail_time / max(1, n_selected), 350))

    for a_init, label in selected:
        rem = deadline - time.monotonic() - 250
        if rem < 60:
            break
        t_alloc = min(t_per, rem)
        print(f"  Optimizing [{label}]: t={t_alloc:.0f}s")
        a_opt, sc_opt = full_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_opt")

    # === Phase 2: Dimension exploration ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        candidate_ns = [384, 512, 640, 768, 896, 1024, 1280, 1536]
        for target_n in candidate_ns:
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 80:
                break
            t_alloc = min(rem * 0.5, 150)
            a_interp = interpolate_sequence(best_a, target_n)
            a_opt, sc = full_optimize(a_interp, time_limit=t_alloc, verbose=False)
            if sc < best_score:
                try_seq(a_opt, f"dim_{target_n}")

    # === Phase 3: Aggressive local search on best ===
    print(f"\n=== Phase 3: Aggressive local search (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            # Fine LP with very small damping
            t_fine = min(rem * 0.35, 80)
            print(f"  Fine LP: t={t_fine:.0f}s")
            a_fine, sc = alternating_lp_optimize(best_a, max_iters=300,
                                                  damping_start=0.15, damping_end=0.03,
                                                  time_limit=t_fine, verbose=True)
            if sc < best_score:
                best_score = sc
                best_a = a_fine.copy()
                print(f"  After fine LP: {best_score:.10f}")

        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_cd = min(rem * 0.5, 60)
            print(f"  Final CD: t={t_cd:.0f}s")
            a_cd, sc = coord_descent_polish(best_a, time_limit=t_cd, verbose=True)
            if sc < best_score:
                best_score = sc
                best_a = a_cd.copy()

        # Perturbation + re-optimize
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            n_best = len(best_a)
            for trial in range(4):
                rem = deadline - time.monotonic() - 20
                if rem < 12:
                    break
                noise_scale = 0.001 * (trial + 1) * 0.5
                noise = rng.standard_normal(n_best) * noise_scale
                a_pert = project_simplex(best_a + noise)
                t_pert = min(rem, 35)
                a_opt, sc = full_optimize(a_pert, time_limit=t_pert, verbose=False)
                if sc < best_score:
                    try_seq(a_opt, f"perturb_{trial}")

    # === Final: one more LP+CD cycle on absolute best ===
    if best_a is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            a_final, sc = full_optimize(best_a, time_limit=min(rem, 60), verbose=False)
            if sc < best_score:
                best_score = sc
                best_a = a_final.copy()

    # Fallback
    if best_a is None:
        best_a = cosine_sequence(768)
        best_score = compute_score(best_a)

    # Final verification with actual evaluate_sequence
    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_a)})")
    return [float(x) for x in best_a]
