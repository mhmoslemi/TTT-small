def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_averaged(a, time_limit, lr0=0.01, decay=0.9995, top_k=30, verbose=False):

    conv = fast_convolve(a)

    best_a = a.copy()

    avg_a = np.zeros(n)
    avg_count = 0
    cur_lr = lr0

    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1

        # Find top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        # Softmax weights on peaks
        temp = max(1e-6, pv.max() - pv.min()) * 0.5
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])

            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()

        avg_count += 1

def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def geometric_sequence(n, r):
    """Geometric sequence normalized to sum=1."""
    a = (1.0 - r) * r ** np.arange(n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def cosine_sequence(n):
    """Cardinal sine shaped sequence."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    """Gaussian-shaped sequence centered."""
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def triangular_sequence(n):
    """Triangular (tent) sequence."""
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)
