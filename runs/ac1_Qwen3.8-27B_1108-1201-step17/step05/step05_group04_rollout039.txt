```python
def lp_direction(f, M):
    """Find direction g using LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    from scipy.optimize import linprog
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    # Convolution constraint: conv(f, g)[k] <= M for k = 0, ..., 2n-2
    # conv(f,g)[k] = sum_i f[i] * g[k-i] (valid indices)
    # This is a matrix-vector product: A @ g <= M
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            A[k, j] = f[i]
    
    A_ub = A
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-10:
            g = g / s
        return g
    return None
```

```python
def lp_minimax_direction(f):
    """Find g that minimizes max_k conv(f,g)[k] s.t. g>=0, sum(g)=1."""
    from scipy.optimize import linprog
    n = len(f)
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1  # minimize t
    
    # conv(f,g)[k] - t <= 0 for k = 0, ..., 2n-2
    # conv(f,g)[k] = sum_i f[i] * g[k-i]
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            A_ub[k, j] = f[i]
        A_ub[k, -1] = -1
    
    # sum(g) = 1 -> equality constraint
    A_eq = np.zeros((1, n+1))
    b_eq = np.array([1.0])
    A_eq[0, :n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        t_opt = res.x[-1]
        return g, t_opt
    return None, None
```

The algorithm would be:
1. Start with current `f` (normalized to sum 1)
2. Solve the LP to get optimal `g` and optimal `t_opt = max_k conv(f,g)[k]`
3. Blend: `f_new = (1-alpha)*f + alpha*g` for some step size alpha
4. If `alpha = 1`, we jump to `g`. If `alpha < 1`, we take a partial step.
5. The peak of `conv(f_new, f_new)` should decrease if `t_opt < max(conv(f,f))`.

The key insight: if `t_opt < M` (current peak), then for small enough alpha, the peak will decrease. As alpha → 1, we approach `g` whose cross-correlation with `f` has peak `t_opt < M`.

But `conv(g,g)` might have a larger peak than `t_opt`. So we need to be careful about the step size.

A safer approach: use alpha = 0.5 or less, and check that the actual peak decreased.

Let me also think about a potential issue: the LP gives a direction `g` that minimizes `max_k conv(f,g)[k]`. But the actual objective we want to minimize is `max_k conv(f_new, f_new)[k]` where `f_new = (1-alpha)f + alpha*g`. 

`conv(f_new, f_new) = (1-alpha)^2 conv(f,f) + 2(1-alpha)alpha conv(f,g) + alpha^2 conv(g,g)`

The peak of this is at most:
`(1-alpha)^2 M_f + 2(1-alpha)alpha t_opt + alpha^2 M_g`

where `M_f = max(conv(f,f))`, `t_opt = max(conv(f,g))`, `M_g = max(conv(g,g))`.

We want this to be < M_f. This gives:
`(1-alpha)^2 M_f + 2(1-alpha)alpha t_opt + alpha^2 M_g < M_f`
`M_f - 2(1-alpha)alpha M_f + 2(1-alpha)alpha t_opt + alpha^2 M_g < M_f`
`2(1-alpha)alpha (t_opt - M_f) + alpha^2 M_g < 0`
`alpha [2(1-alpha)(t_opt - M_f) + alpha M_g] < 0`

Since `t_opt < M_f` (otherwise the LP doesn't help), the first term is negative. For small alpha, the `alpha M_g` term is negligible, so we're fine. For larger alpha, we need `2(1-alpha)(M_f - t_opt) > alpha M_g`.

So the safe step size is: `alpha < 2(M_f - t_opt) / (M_g + 2(M_f - t_opt))`.

If `M_g` is not too large (say `M_g < M_f`), then `alpha < 2(M_f - t_opt) / (M_g + 2(M_f - t_opt))` is reasonably large.

In practice, I'll use a line search: try alpha = 1, 0.5, 0.25, 0.1, ... and pick the first one that decreases the peak.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided minimax direction finding,
Adam with soft-max subgradients, and coordinate descent polishing.

The key novel approach is solving a minimax LP at each iteration to find the
optimal direction that most reduces the peak of the auto-convolution.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_minimax_direction(f):
    """
    Solve: minimize t s.t. conv(f,g)[k] <= t for all k, g >= 0, sum(g) = 1.
    Returns (g, t_opt) or (None, None) on failure.
    """
    n = len(f)
    num_lags = 2 * n - 1
    
    # Build constraint matrix A where A[k, j] = f[i] if i + j = k (i.e., j = k-i)
    # conv(f, g)[k] = sum_{i+j=k} f[i] * g[j]
    # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((num_lags, n + 1))
    for k in range(num_lags):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            i = k - j
            A[k, j] = f[i]
        A[k, n] = -1  # -t
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A, b_ub=np.zeros(num_lags), A_eq=A_eq, b_eq=np.ones(1),
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        t_opt = res.x[n]
        return g, t_opt
    return None, None

def lp_step(f, max_line_search=8):
    """Take one LP-guided step with line search."""
    n = len(f)
    M_f = fast_convolve(f).max()
    
    g, t_opt = lp_minimax_direction(f)
    if g is None:
        return f, M_f, False
    
    # Line search
    best_f = f.copy()
    best_obj = M_f
    improved = False
    
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new /= s
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            improved = True
            if obj < M_f * 0.99:  # Good enough improvement
                break
    
    return best_f, best_obj, improved

def adam_softmax(f0, time_limit, lr=0.005, temp_init=0.008, temp_decay=0.9995,
                 top_k=None):
    """Adam with soft-max subgradient."""
    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Soft-max weights
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked
        
        # Subgradient
        sg = np.correlate(w, f, 'valid')
        
        # Adam update
        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f_new)
        
        temp = max(temp * temp_decay, 1e-7)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(t * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = temp_init * 0.8
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit):
    """Coordinate descent polishing."""
    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
    
        if not improved:
            break
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided minimax direction finding,
Adam with soft-max subgradients, and coordinate descent polishing.

Key novel approach: At each iteration, solve a minimax LP to find the optimal
direction g that most reduces the peak of conv(f,g), then blend with line search.
This is more principled than soft-max subgradient approximation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the matrix A such that A @ g = conv(f, g)."""
    n = len(f)
    num_lags = 2 * n - 1
    A = np.zeros((num_lags, n))
    for j in range(n):
        end = min(j + n, num_lags)
        if end > j:
            A[j:end, j] = f[:(end - j)]
    return A


def lp_minimax_direction(f):
    """
    Solve: minimize t s.t. conv(f,g)[k] <= t for all k, g >= 0, sum(g) = 1.
    Returns (g, t_opt) or (None, None) on failure.
    """
    n = len(f)
    num_lags = 2 * n - 1

    # Build A matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((num_lags, n + 1))
    for j in range(n):
        end = min(j + n, num_lags)
        if end > j:
            A[j:end, j] = f[:(end - j)]
    A[:, n] = -1.0

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A, b_ub=np.zeros(num_lags), A_eq=A_eq, b_eq=np.ones(1),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            g = res.x[:n]
            t_opt = res.x[n]
            return g, t_opt
    except Exception:
        pass
    return None, None


def lp_step(f, max_alphas=None):
    """Take one LP-guided step with line search. Returns (f_new, obj, improved)."""
    n = len(f)
    M_f = fast_convolve(f).max()

    g, t_opt = lp_minimax_direction(f)
    if g is None:
        return f, M_f, False

    if max_alphas is None:
        max_alphas = [1.0, 0.75, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    best_f = f.copy()
    best_obj = M_f
    improved = False

    for alpha in max_alphas:
        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new /= s
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            improved = True
            if obj < M_f * 0.995:
                break

    return best_f, best_obj, improved


def adam_softmax(f0, time_limit, lr=0.005, temp_init=0.008, temp_decay=0.9995,
                 top_k=None, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with simplex projection and soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    cur_lr = lr

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        sg = np.correlate(w, f, 'valid')

        m = beta1 * m + (1.0 - beta1) * sg
        v = beta2 * v + (1.0 - beta2) * sg * sg
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-8)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, multipliers=None):
    """Coordinate descent with local search per coordinate."""
    if multipliers is None:
        multipliers = [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0]

    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.2):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_cosine_init(n):
    """Create a raised-cosine initialization."""
    x = np.linspace(0, 1, n)
    f = np.cos(np.pi * (x - 0.5)) ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_binomial_init(n):
    """Create a binomial-like (bell-shaped) initialization."""
    k = np.arange(n)
    mean = (n - 1) / 2.0
    sigma = np.sqrt(n) / 4.0
    f = np.exp(-0.5 * ((k - mean) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

                print(f"    LP iter {lp_iter}: obj={f_obj:.10f}, perturb")

        # Every 50 successful LP iterations, refresh with Adam
        if lp_iter % 80 == 0 and time.monotonic() < t2_end - 30:
            a_opt, _ = adam_softmax(f, time_limit=20, lr=0.003,
                                    temp_init=0.004, temp_decay=0.9997,
                                    top_k=max(15, n_base // 15))
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"p2_adam_{lp_iter}")
            if best_seq is not None and base is not None:
                if len(best_seq) == n_base:
                    base = best_seq.copy()

    # Final LP check
    best_seq, best_score = try_and_update(f, best_seq, best_score, "p2_final")

    base = best_seq.copy()
    n_base = len(base)

    # === Phase 3: Extended LP with smaller steps (400s) ===
    print(f"\n=== Phase 3: Extended LP fine steps (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + 400

    f = base.copy()
    f = f / f.sum()
    small_alphas = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]
    lp_no_improve = 0
    lp_iter = 0

    while time.monotonic() < t3_end:
        lp_iter += 1
        f_new, new_obj, improved = lp_step(f, max_alphas=small_alphas)

        if improved:
            f = f_new
            lp_no_improve = 0
            if lp_iter % 10 == 0:
                best_seq, best_score = try_and_update(f, best_seq, best_score, f"p3_lp_{lp_iter}")
        else:
            lp_no_improve += 1
            if lp_no_improve > 50:
                noise = rng.standard_normal(n_base) * 0.001
                f = project_simplex(base + noise)
                lp_no_improve = 0

        # Alternate with Adam
        if lp_iter % 100 == 0 and time.monotonic() < t3_end - 20:
            a_opt, _ = adam_softmax(f, time_limit=15, lr=0.001,
                                    temp_init=0.002, temp_decay=0.9998,
                                    top_k=max(10, n_base // 20))
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"p3_adam_{lp_iter}")

    best_seq, best_score = try_and_update(f, best_seq, best_score, "p3_final")

    base = best_seq.copy()
    n_base = len(base)

    # === Phase 4: Coordinate descent polishing (300s) ===
    print(f"\n=== Phase 4: Coordinate descent (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + 300

    a_opt, _ = coordinate_descent(base, time_limit=180)
    best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "p4_cd")

    rem = t4_end - time.monotonic()
    if rem > 30 and best_seq is not None:
        n_b = len(best_seq)
        noise = rng.standard_normal(n_b) * 0.002
        perturbed = project_simplex(best_seq + noise)
        a_opt, _ = coordinate_descent(perturbed, time_limit=min(rem * 0.6, 80))
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "p4_cd_pert")

    base = best_seq.copy()
    n_base = len(base)

    # === Phase 5: Length exploration (150s) ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===")
    t5_end = time.monotonic() + 150

    target_lengths = sorted(set([n_base - 64, n_base + 64, n_base + 128, n_base + 256]),
                           key=lambda x: abs(x - n_base))
    target_lengths = [t for t in target_lengths if 128 <= t <= 2560 and t != n_base][:3]

    for target_n in target_lengths:
        rem = t5_end - time.monotonic()
        if rem < 25:
            break
        f_interp = interpolate_sequence(base, target_n)
        a_opt, _ = adam_softmax(f_interp, time_limit=min(rem * 0.5, 45),
                                lr=0.004, temp_init=0.005,
                                temp_decay=0.9995, top_k=max(15, target_n // 12))
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"p5_n{target_n}")

        # Quick LP on interpolated
        rem = t5_end - time.monotonic()
        if rem > 15:
            f_lp = f_interp
            for _ in range(10):
                if time.monotonic() > t5_end - 5:
                    break
                f_lp, _, imp = lp_step(f_lp)
                if not imp:
                    break
            best_seq, best_score = try_and_update(f_lp, best_seq, best_score, f"p5_lp_n{target_n}")

    base = best_seq.copy()
    n_base = len(base)

    # === Phase 6: Final combined polish (remaining time) ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

    if rem > 20 and best_seq is not None:
        # Fine Adam
        t_adam = min(rem * 0.4, 80)
        a_opt, _ = adam_softmax(best_seq, time_limit=t_adam, lr=0.001,
                                temp_init=0.002, temp_decay=0.9998,
                                top_k=max(10, n_base // 20))
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "p6_fine_adam")

    # LP steps
    rem = deadline - time.monotonic() - 10
    if rem > 10 and best_seq is not None:
        f = best_seq.copy()
        f = f / f.sum()
        for _ in range(50):
            if time.monotonic() > deadline - 10:
                break
            f, _, imp = lp_step(f)
            if imp:
                best_seq, best_score = try_and_update(f, best_seq, best_score, "p6_lp")
            else:
                break

    # Tiny perturbations
    rem = deadline - time.monotonic() - 10
    if rem > 15 and best_seq is not None:
        n_b = len(best_seq)
        for pi in range(5):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            noise = rng.standard_normal(n_b) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = adam_softmax(f_pert, time_limit=min(rem * 0.4, 15),
                                    lr=0.002, temp_init=0.003,
                                    temp_decay=0.9997, top_k=max(10, n_b // 20))
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"p6_pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
