"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Fast projected subgradient descent with momentum and adaptive step sizes
2. LP-based Frank-Wolfe for global moves
3. Peak-targeted coordinate descent
4. Multi-length exploration and restarts
Key innovation: exponential peak weighting, RMSProp-style adaptation,
and a "peak peeling" strategy that focuses on the dominant peak.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=100, temp=0.001):
    """Compute weighted subgradient using top-k peaks with exponential weighting."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    if k_actual <= 0:
        return np.zeros(n)

    sorted_idx = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[sorted_idx]

    # Exponential weighting based on excess over the median of top-k
    med = np.median(pv)
    excess = np.maximum(pv - med, 0)
    if excess.max() < 1e-15:
        w = np.ones(len(pv)) / len(pv)
    else:
        scaled = excess / max(temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(pv)) / len(pv)
        else:
            w /= ws

    sg = np.zeros(n)
    for i in range(len(pv)):
        if w[i] < 1e-14:
            continue
        k = int(sorted_idx[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        # d(conv[k])/d(f[j]) = 2*f[k-j] (or f[j] if k=2j)
        contrib = 2.0 * f[k - idx]
        # Handle the diagonal (k=2j) case
        diag = k - idx == idx
        if diag.any():
            contrib[diag] = f[idx[diag]]
        sg[idx] += w[i] * contrib

    return sg


def subgradient_optimize(f0, time_limit, verbose=False):
    """
    Projected subgradient descent with momentum, RMSProp adaptation,
    and peak peeling.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    v = np.zeros(n)  # momentum
    sq = np.ones(n)  # RMSProp accumulated squared gradients
    beta1 = 0.92  # momentum
    beta2 = 0.99  # RMSProp
    lr_base = 0.008
    top_k = max(30, n // 8)
    temp = 0.0005
    no_improve = 0
    it = 0
    best_temp = temp
    best_lr = lr_base

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        sg = compute_subgradient(f, conv, top_k=top_k, temp=temp)
        sg_norm = np.linalg.norm(sg)

        if sg_norm < 1e-15:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            # RMSProp update
            sq = beta2 * sq + (1 - beta2) * sg * sg
            effective_lr = lr_base / (np.sqrt(sq) + 1e-8)
            v = beta1 * v - effective_lr * sg
            f_new = project_simplex(f + v)

            if np.any(f_new > 0):
                conv = fast_convolve(f_new)
                co = conv.max()

                if co < best_obj - 1e-16:
                    best_obj = co
                    best_f = f_new.copy()
                    no_improve = 0
                    f = f_new
                    if verbose and it % 200 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"    SG iter {it} ({elapsed:.0f}s): obj={best_obj:.12f}")
                else:
                    f = f_new
                    no_improve += 1

        # Adaptive temperature and learning rate
        if no_improve > 100:
            temp *= 1.5
            lr_base *= 0.95
            if no_improve > 300:
                # Restart with perturbation
                scale = 0.001 + 0.003 * min((no_improve - 300) // 100, 5)
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                v = np.zeros(n)
                sq = np.ones(n)
                lr_base = max(lr_base * 2, 0.003)
                temp = best_temp
                no_improve = 0

        # Increase top_k over time to capture more peaks
        top_k = min(n, int(top_k * (1 + 0.0005)))
        # Decrease temperature for finer peeling
        if it % 500 == 0:
            temp *= 0.85

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_step(f):
    """Single LP-FW iteration: find direction and take best step."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()

    g = solve_lp_direction(f, M * 1.001, n)
    if g is None:
        g = solve_lp_direction(f, M * 1.01, n)
    if g is None:
        g = solve_lp_direction(f, M * 1.05, n)
    if g is None:
        return f, M

    g_sum = g.sum()
    if g_sum < 1e-15:
        return f, M
    g = g / g_sum

    if np.max(np.abs(g - f)) < 1e-12:
        return f, M

    # Line search
    best_alpha = 0
    best_val = M
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha > 1e-13:
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        return f_new, fast_convolve(f_new).max()
    return f, M

                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss

def two_phase_optimize(f0, time_limit, verbose=False):
    """
    Two-phase optimizer:
    Phase A: Fast subgradient descent (majority of time)
    Phase B: LP-FW + coordinate descent polish (remaining time)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    t_start = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    # Phase A: Subgradient
    t_sg = time_limit * 0.6
    if verbose:
        print(f"  Phase A (SG): {t_sg:.0f}s")
    a_opt, obj = subgradient_optimize(f, time_limit=t_sg, verbose=verbose)
    if obj < best_obj:
        best_obj = obj
        best_f = a_opt.copy()

    f = a_opt

    # Phase B: LP-FW
    rem = time_limit - (time.monotonic() - t_start)
    t_lp = rem * 0.4
    if verbose and t_lp > 5:
        print(f"  Phase B (LP-FW): {t_lp:.0f}s")
    if t_lp > 5:
        it_count = 0
        while time.monotonic() - t_start < t_start + time_limit - 2:
            f_new, obj_new = lp_fw_step(f)
            it_count += 1
            if obj_new < best_obj - 1e-16:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                if verbose:
                    print(f"    LP-FW {it_count}: {best_obj:.12f}")
            else:
                f = f_new
                if it_count > 50:
                    break

    # Phase C: Coordinate descent
    rem = time_limit - (time.monotonic() - t_start)
    t_cd = rem * 0.5
    if verbose and t_cd > 5:
        print(f"  Phase C (CD): {t_cd:.0f}s")
    if t_cd > 5:
        a_opt, obj = coordinate_descent(best_f, time_limit=t_cd)
        if obj < best_obj - 1e-16:
            best_obj = obj
            best_f = a_opt.copy()

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_init(n, phase=0.0, amplitude=0.3):
    """Generate a cosine-modulated initialization."""
    j = np.arange(n)
    base = np.ones(n) / n
    modulation = 1.0 + amplitude * np.cos(2 * np.pi * j / n + phase)
    f = base * modulation
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f


def ramp_init(n, power=1.0):
    """Generate a power-law ramp initialization."""
    j = np.arange(n, dtype=np.float64)
    f = (j + 1) ** power
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f

        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Deep optimization on parent ===
    print(f"\n=== Phase 1: Two-phase on parent (n={n_parent}) ===")
    t1 = min(600, (deadline - time.monotonic()) * 0.35)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_opt, obj = two_phase_optimize(parent, time_limit=t1, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f} (n={len(a_opt)})")

    # === Phase 2: Diverse initializations on same length ===
    print(f"\n=== Phase 2: Diverse inits (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    inits = []

    # Cosine modulations
    for amp in [0.2, 0.4, 0.6]:
        for ph in [0, np.pi / 2, np.pi, np.pi * 3 / 4]:
            inits.append(cosine_init(n_best, phase=ph, amplitude=amp))

    # Ramps
    for pw in [0.5, 1.0, 2.0, 0.3]:
        inits.append(ramp_init(n_best, power=pw))

    # Perturbed parent
    for scale in [0.005, 0.01, 0.02, 0.05]:
        inits.append(project_simplex(best_seq + rng.standard_normal(n_best) * scale))

    # Mirror + blend
    f_rev = best_seq[::-1]
    for alpha in [0.3, 0.5, 0.7]:
        inits.append(project_simplex(alpha * best_seq + (1 - alpha) * f_rev))

    # Evaluate initializations cheaply
    scored_inits = []
    for fi in inits:
        sc = float(evaluate_sequence(fi.tolist()))
        if np.isfinite(sc):
            scored_inits.append((sc, fi))
    scored_inits.sort(key=lambda x: x[0])

    # Try top diverse initializations
    tried = [(sc, fi) for sc, fi in scored_inits if sc < best_score + 0.1]
    # Ensure diversity: limit to 8
    tested = set()
    selected = []
    for sc, fi in tried:
        key = tuple(np.round(fi[:10], 3))
        if key not in tested:
            tested.add(key)
            selected.append((sc, fi))
        if len(selected) >= 8:
            break

    for idx, (sc_init, f_init) in enumerate(selected):
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        t_alloc = min(120, rem * 0.15)
        if t_alloc < 15:
            continue
        print(f"  Init {idx} (score={sc_init:.6f}): t={t_alloc:.0f}s")
        a_opt, obj = two_phase_optimize(f_init, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f} (n={len(a_opt)})")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = sorted(set([
        n_best - 128, n_best - 64, n_best,
        n_best + 64, n_best + 128, n_best + 256,
    ]))
    candidate_lengths = [nl for nl in candidate_lengths if 128 <= nl <= 3072]

    for target_n in candidate_lengths:
        if target_n == n_best:
            continue
        rem = deadline - time.monotonic() - 90
        if rem < 40:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(100, rem * 0.2)
        if t_alloc < 20:
            continue
        print(f"  n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = two_phase_optimize(f_interp, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f} (n={target_n})")

    # === Phase 4: Final deep optimization on best ===
    print(f"\n=== Phase 4: Final deep opt (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 30
    if rem > 60:
        t_final = min(rem * 0.7, 300)
        print(f"  Budget: {t_final:.0f}s")
        a_opt, obj = two_phase_optimize(best_seq, time_limit=t_final, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 5: Micro perturbations ===
    print(f"\n=== Phase 5: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(15):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0002 + trial * 0.0001
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            # Quick polish
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = subgradient_optimize(f_pert, time_limit=min(rem2 * 0.3, 10))
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_sg]: {sc2:.6f}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
