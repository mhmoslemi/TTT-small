Actually, let me reconsider. The subgradient of max_k (a*a)[k] is tricky because:

    # Descent step

"""
Optimized step function minimizer using subgradient methods, smooth
log-sum-exp optimization with L-BFGS, active-set polishing, and
multi-length exploration. Key innovations: exact subgradient descent
on the max-convolution objective, active-set identification for
targeted lag reduction, and improved temperature scheduling.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.optimize import project
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_objective(f, T):
    """Log-sum-exp smooth approximation of max(conv(f,f))."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    return T * (scaled.max() + np.log(e.sum()))


def smooth_grad(f, T):
    """Gradient of log-sum-exp smooth objective."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w


def subgradient_direction(f, k_star, active_lags=None):
    """Compute subgradient of max conv at given lag(s)."""
    n = len(f)
    if active_lags is None:
        active_lags = [k_star]
    grad = np.zeros(n)
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += f[k - j]
    grad /= len(active_lags)
    return grad

    f = np.maximum(f0.astype(np.float64), 0.0)

        it += 1

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    best_obj = M

    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 6.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:

        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)

            if val < best_val:

                if st < 1e-15:

                ft = ft / st

                    best_alpha = mid

            M = conv.max()
            m_exp = 1.0

                best_obj = M
                best_f = f.copy()
                no_improve = 0

            conv = fast_convolve(f)

def smooth_optimize_adam(f0, time_limit, T_start=0.5, T_end=0.005, verbose=False):
    """Smooth optimization with Adam and cosine temperature annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0

    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        # Cosine annealing for temperature
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        T = max(T, T_end)
        lr = alpha * (1.0 - 0.7 * progress)

        grad, lse_val, conv, w = smooth_grad(f, T)

        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)

        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)

        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Smooth iter {it}: obj={best_obj:.10f}, T={T:.5f}")
        else:
            no_improve += 1

        if no_improve > 2000:
            seed_val = int(elapsed * 1000) % 1000000
            rng_p = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.005)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0

    return best_f, best_obj


def lbfgs_smooth(f0, time_limit, T_start=0.3, T_end=0.02, verbose=False):
    """Use L-BFGS-B on the smoothed objective for faster convergence."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    current_T = [T_start]
    t0 = time.monotonic()
    n_evals = [0]

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        current_T[0] = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        current_T[0] = max(current_T[0], T_end)

        def obj(x):
            n_evals[0] += 1
            x = np.maximum(x, 0.0)
            ss = x.sum()
            if ss < 1e-15:
                return 1e10
            x = x / ss
            return smooth_objective(x, current_T[0])

        def obj_grad(x):
            n_evals[0] += 1
            x = np.maximum(x, 0.0)
            ss = x.sum()
            if ss < 1e-15:
                return 1e10, np.zeros(len(x))
            x = x / ss
            g, _, _, _ = smooth_grad(x, current_T[0])
            return smooth_objective(x, current_T[0]), g

        maxfun = max(50, min(500, int((time_limit - elapsed) * 2)))
        try:
            res = minimize(obj_grad, f.copy(), method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxfun': maxfun, 'maxiter': 200})
            if res.success or res.fun < 1e10:
                f_new = np.maximum(res.x, 0.0)
                ss = f_new.sum()
                if ss > 1e-15:
                    f_new = f_new / ss
                    cur_obj = fast_convolve(f_new).max()
                    if cur_obj < best_obj - 1e-14:
                        best_obj = cur_obj
                        best_f = f_new.copy()
                        if verbose:
                            print(f"    L-BFGS: obj={best_obj:.10f}, T={current_T[0]:.5f}")
                        f = f_new
        except Exception:
            pass

        elapsed2 = time.monotonic() - t0
        if elapsed2 > time_limit * 0.95:
            break

    return best_f, best_obj


def active_set_optimize(f0, time_limit, verbose=False):
    """Optimize by focusing on the active set of lags achieving near-max."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit * 0.95:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 20 == 0:
                print(f"    ActiveSet iter {it}: obj={best_obj:.10f}")

        # Find active lags
        threshold = M * (0.98 - 0.01 * min(it // 50, 5))
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute average gradient over active lags
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j]
        grad /= len(active)

        # Normalized gradient step with line search
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        grad = grad / grad_norm
        lr_base = 0.02 * (1.0 + it / 100)
        lr_base = min(lr_base, 0.1)

        best_alpha = 0.0
        best_val = M
        for alpha in [lr_base * a for a in [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]]:
            f_trial = f - alpha * grad
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            # Perturb and retry
            rng_p = np.random.default_rng(it * 17)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.003)
            continue

        f = f - best_alpha * grad
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        if it > 200 and (time.monotonic() - t0) > time_limit * 0.7:
            break

    return best_f, best_obj


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.001
    a = np.maximum(a, 0)
    return a / a.sum()


def make_tanh_window(n, sharpness=3.0):
    """Tanh-based window (smooth step)."""
    x = np.linspace(-1, 1, n)
    a = np.tanh(sharpness * (1 - np.abs(x))) + 0.01
    a = np.maximum(a, 0)
    return a / a.sum()


def make_flat_top(n, alpha=3.0):
    """Flat-top window design (minimizes peak of auto-correlation)."""
    x = np.linspace(-1, 1, n)
    a = np.cos(np.pi * x * 0.5) ** alpha + 0.005
    a = np.maximum(a, 0)
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimizer combining:
    - Diverse window initializations (cosine, tanh, flat-top)
    - Smooth log-sum-exp Adam optimization
    - L-BFGS on smoothed objective
    - Exact subgradient polishing
    - Active-set targeted optimization
    - LP Frank-Wolfe with adaptive relaxation
    - Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        a = np.clip(a, 0.0, 1000.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old_score:.6f}]")

    # === Phase 0: Diverse initializations ===
    print("=== Phase 0: Diverse Initializations ===")
    lengths_to_try = [n_parent, 512, 768]
    if n_parent not in lengths_to_try:
        lengths_to_try.append(n_parent)

    for n_try in lengths_to_try:
        for pw in [1.0, 1.5, 2.0, 3.0, 4.0, 6.0]:
            g = make_cosine(n_try, power=pw)
            _try(g, f"cos_p{pw}_n{n_try}")

        for sh in [2.0, 3.0, 5.0, 8.0]:
            g = make_tanh_window(n_try, sharpness=sh)
            _try(g, f"tanh_s{sh}_n{n_try}")

        for al in [2.0, 3.0, 4.0, 6.0]:
            g = make_flat_top(n_try, alpha=al)
            _try(g, f"flat_a{al}_n{n_try}")

        # Geometric
        for r in [0.95, 0.97, 0.98, 0.99, 0.995]:
            g = r ** np.arange(n_try, dtype=np.float64)
            g = g / g.sum()
            _try(g, f"geo_r{r}_n{n_try}")

    print(f"  After Phase 0: best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    # === Phase 1: Smooth optimization (Adam) on best ===
    print(f"\n=== Phase 1: Smooth Adam (best={best_score:.6f}) ===")
    base = best_seq.copy()
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 300)
        print(f"  Smooth Adam n={len(base)}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize_adam(base, time_limit=t_alloc, T_start=0.5, T_end=0.005, verbose=True)
        _try(a_opt, "smooth_adam_p1")

    # === Phase 2: L-BFGS on smoothed ===
    print(f"\n=== Phase 2: L-BFGS Smooth (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        t_alloc = min(rem, 200)
        print(f"  L-BFGS n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lbfgs_smooth(best_seq, time_limit=t_alloc, T_start=0.25, T_end=0.01, verbose=True)
        _try(a_opt, "lbfgs_p2")

    # === Phase 3: Subgradient polishing ===
    print(f"\n=== Phase 3: Subgradient Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 15:
        t_alloc = min(rem, 150)
        print(f"  SubGrad n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_polish(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "subgrad_p3")

    # === Phase 4: Active-set optimization ===
    print(f"\n=== Phase 4: Active Set (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 15:
        t_alloc = min(rem, 120)
        print(f"  ActiveSet n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = active_set_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "active_p4")

    # === Phase 5: LP-FW ===
    print(f"\n=== Phase 5: LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 15:
        t_alloc = min(rem, 120)
        print(f"  LP-FW n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "lpfw_p5")

        # LP-FW with perturbation
        rem = deadline - time.monotonic() - 150
        if rem > 10:
            noise = rng.standard_normal(len(best_seq)) * 0.005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lpfw_optimize(f_pert, time_limit=min(rem, 60))
            _try(a_opt, "lpfw_pert")

    # === Phase 6: Longer sequences ===
    print(f"\n=== Phase 6: Longer Sequences (best={best_score:.6f}) ===")
    for target_n in [1024, 1536]:
        rem = deadline - time.monotonic() - 80
        if rem < 30:
            break
        f_interp = interpolate_seq(best_seq, target_n)

        t1 = min(rem * 0.4, 60)
        a_opt, _ = smooth_optimize_adam(f_interp, time_limit=t1, T_start=0.3, T_end=0.03)
        _try(a_opt, f"smooth_n{target_n}")

        rem = deadline - time.monotonic() - 70
        if rem > 15:
            t2 = min(rem * 0.5, 50)
            a_opt, _ = subgradient_polish(f_interp, time_limit=t2)
            _try(a_opt, f"subgrad_n{target_n}")

    # === Phase 7: Final polishing cycle ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Tight smooth optimization
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            a_opt, _ = smooth_optimize_adam(best_seq, time_limit=min(rem, 60),
                                            T_start=0.03, T_end=0.003, verbose=True)
            _try(a_opt, "smooth_fine")

        # Subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = subgradient_polish(best_seq, time_limit=min(rem, 40), verbose=True)
            _try(a_opt, "subgrad_fine")

        # Active set
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_opt, _ = active_set_optimize(best_seq, time_limit=min(rem, 25))
            _try(a_opt, "active_fine")

        # LP-FW final
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 20))
            _try(a_opt, "lpfw_final")

        # Multiple perturbations
        for perturb_i in range(5):
            rem = deadline - time.monotonic() - 5
            if rem < 5:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_optimize_adam(f_pert, time_limit=min(rem, 15),
                                            T_start=0.02, T_end=0.002)
            _try(a_opt, f"perturb_{perturb_i}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
