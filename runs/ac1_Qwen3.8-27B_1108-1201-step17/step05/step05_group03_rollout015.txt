    return np.fft.irfft(A * A, fft_size)[:size]

def conv_subgradient(a, c, peaks, weights):
    """
    Compute subgradient of max_k c[k] at point a.
    c[k] = sum_i a[i] * a[k-i]
    dc[k]/da[i] = a[k-i] + a[i-k] (with a[j]=0 for j<0 or j>=n)
    """
    n = len(a)
    sg = np.zeros(n)
    for k, w in zip(peaks, weights):
        k = int(k)
        # c[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
        # dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        for i in range(n):
            val = 0.0
            ki = k - i
            if 0 <= ki < n:
                val += a[ki]
            ik = i - k
            if 0 <= ik < n:
                val += a[ik]
            sg[i] += w * val
    return sg


def conv_subgradient_fast(a, c, peaks, weights):
    """Fast subgradient computation using FFT."""
    n = len(a)
    sg = np.zeros(n)
    # For each peak k, the subgradient contribution at position i is:
    # a[k-i] + a[i-k] (with bounds)
    # This is equivalent to: (a reversed and shifted) + (a shifted)
    # Specifically, for peak k:
    #   contribution[i] = a[min(k-i, n-1) if k-i>=0 else ...] 
    # Actually, let's just do it directly but vectorized
    
    for k, w in zip(peaks, weights):
        k = int(k)
        # a[k-i] for i in range(n): this is a reversed and shifted
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
        
        # Vectorized: for a[k-i], we can use:
        #   indices = k - np.arange(n), clip to [0, n-1], but only valid where 0<=k-i<n
        idx1 = k - np.arange(n)
        mask1 = (idx1 >= 0) & (idx1 < n)
        contrib1 = np.zeros(n)
        contrib1[mask1] = a[idx1[mask1]]
        
        idx2 = np.arange(n) - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        contrib2 = np.zeros(n)
        contrib2[mask2] = a[idx2[mask2]]
        
        sg += w * (contrib1 + contrib2)
    
    return sg

Minimizes 2*n*max(auto_conv)/(sum)^2 using a combination of:
1. Multi-peak multiplicative weight descent (entropy mirror) with averaging
2. Frank-Wolfe (conditional gradient) with acceleration
3. Fast coordinate descent polish
4. Structured diverse initializations (U-shaped, bilinear, edge-concentrated)
5. Adaptive sequence length exploration
Key insight: for non-negative sequences, the convolution peak can be below
the uniform level (2) by creating U-shaped profiles that flatten the
auto-convolution. The score 2n*max_c can dip below 2 when max_c < sum_a^2/n.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_max_conv(a):
    """Compute max of auto-convolution."""
    return fast_convolve(a).max()


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)

    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)

    if s < 1e-15:

    return a / s


def conv_subgradient_fast(a, peaks, weights):
    """
    Compute (weighted) subgradient of max_k c[k] at point a.
    c[k] = sum_i a[i]*a[k-i], dc[k]/da[i] = a[k-i] + a[i-k] (with bounds).
    """
    n = len(a)
    sg = np.zeros(n)
    arange = np.arange(n)
    for k, w in zip(peaks, weights):
        k = int(k)
        idx1 = k - arange
        mask1 = (idx1 >= 0) & (idx1 < n)
        contrib1 = np.zeros(n)
        contrib1[mask1] = a[idx1[mask1]]
        idx2 = arange - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        contrib2 = np.zeros(n)
        contrib2[mask2] = a[idx2[mask2]]
        sg += w * (contrib1 + contrib2)

def multiplicative_weight_descend(a, time_limit, lr_init=0.04, decay=0.997,

    """

    best_a = a.copy()

    t0 = time.monotonic()
    step = 0
    lr = lr_init
    k_peaks = max(10, int(n * top_k_frac))
    temp_frac = 0.4

    avg = a.copy()
    count = 0

    while time.monotonic() - t0 < time_limit:
        step += 1

        c = fast_convolve(a)

        # Top-k peaks with softmax weighting
        k_act = min(k_peaks, len(c))
        peaks = np.argpartition(c, -k_act)[-k_act:]
        pv = c[peaks]

def frank_wolfe_descend(a, time_limit, verbose=False):
    """
    Accelerated Frank-Wolfe (conditional gradient) for min max_k c[k] over simplex.
    Each step is O(n) for the linear minimization oracle.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_maxc = compute_max_conv(a)
    t0 = time.monotonic()
    step = 0
    k_peaks = max(5, n // 40)

    # Acceleration: maintain virtual iterate y
    y = a.copy()

    while time.monotonic() - t0 < time_limit:
        step += 1

        c_y = fast_convolve(y)
        k_act = min(k_peaks, len(c_y))
        peaks = np.argpartition(c_y, -k_act)[-k_act:]
        pv = c_y[peaks]

        spread = pv.max() - pv.min()
        temp = max(1e-8, spread * 0.3)
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = conv_subgradient_fast(y, peaks, w)

        # FW oracle: min_{v in simplex} <sg, v> = sg[argmin sg]
        i_min = int(np.argmin(sg))

        # Accelerated step: gamma = 2/(t+2)
        gamma = 2.0 / (step + 2.0)

        e_min = np.zeros(n)
        e_min[i_min] = 1.0

        # Update virtual and actual iterates
        y = (1.0 - gamma) * y + gamma * e_min
        a = (1.0 - gamma) * a + gamma * e_min
        a = normalize(a)

        mc_a = compute_max_conv(a)
        if mc_a < best_maxc - 1e-14:
            best_maxc = mc_a
            best_a = a.copy()

        mc_y = c_y.max()
        if mc_y < best_maxc - 1e-14:
            best_maxc = mc_y
            best_a = y.copy()

        if verbose and step % 10000 == 0:
            sc = 2.0 * n * best_maxc
            print(f"      FW step {step}: maxc={best_maxc:.8f}, score~{sc:.6f}")

    return best_a, best_maxc

                new_a[int(j)] = v
                mask = np.ones(n, dtype=bool)
                mask[int(j)] = False
                other_sum = 1.0 - v
                if other_sum < 1e-12:
                    continue
                new_a[mask] *= (1.0 - v) / other_sum
                new_a = normalize(new_a)
                mc = compute_max_conv(new_a)
                if mc < best_cv - 1e-13:
                    best_cv = mc
                    best_v = v

            if best_v != old_val and best_cv < best_maxc - 1e-14:
                a = a.copy()
                a[int(j)] = best_v
                mask = np.ones(n, dtype=bool)
                mask[int(j)] = False
                other_sum = 1.0 - best_v
                if other_sum > 1e-12:
                    a[mask] *= (1.0 - best_v) / other_sum
                a = normalize(a)
                best_maxc = best_cv
                best_a = a.copy()
                improved = True

        if verbose and it % 10 == 0:
            sc = 2.0 * n * best_maxc
            print(f"      TCD iter {it}: maxc={best_maxc:.8f}, score~{sc:.6f}")

    return best_a, best_maxc


def generate_u_shaped(n, edge_width_frac=0.3):
    """U-shaped sequence: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    # V-shape
    a = np.abs(2 * x - 1)
    # Sharpen edges
    a = a ** 0.5
    return normalize(a)


def generate_bilinear(n, asymmetry=1.0):
    """Bilinear (triangle) sequence with optional asymmetry."""
    x = np.linspace(0, 1, n)
    peak = asymmetry  # peak position in [0,1]
    a = np.where(x <= peak,
                 x / peak if peak > 0 else 0,
                 (1 - x) / (1 - peak) if peak < 1 else 0)
    return normalize(a)


def generate_edge_concentrated(n, concentration=2.0):
    """Sequence concentrated at both edges."""
    x = np.linspace(0, 1, n)
    # Two Gaussian bumps at edges
    sigma = 0.15
    a = np.exp(-((x) / sigma) ** 2) + np.exp(-((1 - x) / sigma) ** 2)
    # Add small uniform floor
    a += 0.01
    return normalize(a)


def generate_geometric(n, r):
    """Geometric sequence."""
    a = r ** np.arange(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def generate_cosine(n):
    """Cosine-shaped sequence."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def generate_kaiser_like(n, beta=2.0):
    """Kaiser window-like sequence (bell shaped)."""
    r = np.linspace(-1, 1, n)
    from scipy.special import i0
    a = i0(beta * np.sqrt(1 - r ** 2))
    return normalize(a)


def generate_random_smooth(n, rng, smoothness=0.8):
    """Random sequence smoothed by IIR filter."""
    a = rng.random(n)
    # Apply exponential moving average smoothing
    kernel_size = max(3, int(n * (1 - smoothness)))
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def optimize_sequence(a, time_limit, verbose=False):
    """Multi-stage optimization pipeline."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_maxc = compute_max_conv(a)
    t0 = time.monotonic()

    # Stage 1: Multiplicative weight descent (exploratory, fast)
    rem = time_limit - (time.monotonic() - t0)
    t1 = min(rem * 0.25, 120)
    if rem > t1 + 30:
        a1, mc1 = multiplicative_weight_descend(a, time_limit=t1, verbose=verbose)
        if mc1 < best_maxc:
            best_maxc = mc1
            best_a = a1.copy()
            a = a1.copy()
        if verbose:
            print(f"      After MW: score~{2*n*best_maxc:.6f}")

    # Stage 2: Frank-Wolfe
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.2, 90)
    if rem > t2 + 20:
        a2, mc2 = frank_wolfe_descend(a, time_limit=t2, verbose=verbose)
        if mc2 < best_maxc:
            best_maxc = mc2
            best_a = a2.copy()
            a = a2.copy()
        if verbose:
            print(f"      After FW: score~{2*n*best_maxc:.6f}")

    # Stage 3: Targeted coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.3, 150)
    if rem > t3 + 15:
        a3, mc3 = targeted_coordinate_descent(a, time_limit=t3, verbose=verbose)
        if mc3 < best_maxc:
            best_maxc = mc3
            best_a = a3.copy()
            a = a3.copy()
        if verbose:
            print(f"      After TCD: score~{2*n*best_maxc:.6f}")

    # Stage 4: Final polynomial CD polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a4, mc4 = coordinate_descent(a, time_limit=min(rem * 0.6, 90), verbose=verbose)
        if mc4 < best_maxc:
            best_maxc = mc4
            best_a = a4.copy()
        if verbose:
            print(f"      After CD: score~{2*n*best_maxc:.6f}")

    return best_a, best_maxc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining diverse initializations,
    multiplicative weight descent, Frank-Wolfe, and coordinate descent.
    Focuses on U-shaped and edge-concentrated profiles that flatten
    the auto-convolution below the uniform level.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = compute_score(a_norm)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    print("=== Generating diverse starting points ===")
    starts = []

    # U-shaped sequences (key innovative family)
    for n in [512, 640, 768, 896, 1024, 1280]:
        starts.append((generate_u_shaped(n), f"u_n{n}"))
        starts.append((generate_edge_concentrated(n), f"edge_n{n}"))

    # Bilinear with various peak positions
    for n in [640, 768, 1024]:
        for asym in [0.3, 0.5, 0.7]:
            starts.append((generate_bilinear(n, asym), f"bil_{asym}_n{n}"))

    # Geometric
    for n in [640, 768, 1024]:
        for r in [0.998, 0.999, 0.9995]:
            starts.append((generate_geometric(n, r), f"geom_{r}_n{n}"))

    # Kaiser
    for n in [768, 1024]:
        for beta in [1.0, 3.0, 5.0]:
            starts.append((generate_kaiser_like(n, beta), f"kaiser_{beta}_n{n}"))

    # Cosine
    for n in [640, 768, 1024]:
        starts.append((generate_cosine(n), f"cos_n{n}"))

    # Random smooth
    for n in [768, 1024]:
        starts.append((generate_random_smooth(n, rng), f"randsmooth_n{n}"))

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Evaluate all starts quickly
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label in starts:
        try:
            sc = compute_score(a)
            start_scores.append((sc, a, label))
        except Exception:
            pass

    start_scores.sort(key=lambda x: x[0])

    # Print top starts
    for i, (sc, a, label) in enumerate(start_scores[:8]):
        print(f"    Start #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # === Phase 1: Optimize top diverse starts ===
    print(f"\n=== Phase 1: Optimize top starts (best={best_score:.8f}) ===")

    # Select diverse starts (different types/lengths)
    selected = []
    selected_types = set()
    for sc, a, label in start_scores:
        if len(selected) >= 8:
            break
        # Extract type (everything before the last _n)
        base = label.replace('_n512','').replace('_n640','').replace('_n768','') \
                    .replace('_n896','').replace('_n1024','').replace('_n1280','')
        key = base.split('_')[0] if '_' in base else base
        if key not in selected_types:
            selected.append((a, label))
            selected_types.add(key)

    # Always include parent if good
    if not any(l == "parent" for _, l in selected):
        for sc, a, label in start_scores:
            if label == "parent" and sc < best_score * 1.1:
                selected.append((a, label))
                break

    # Also ensure we have some U-shaped / edge starts
    if not any('u_n' in l or 'edge_n' in l for _, l in selected):
        for sc, a, label in start_scores:
            if 'u_n' in label or 'edge_n' in label:
                if len(selected) < 10:
                    selected.append((a, label))
                break

    n_selected = len(selected)
    time_per_start = (deadline - time.monotonic() - 300) / max(1, n_selected)
    time_per_start = max(30, min(time_per_start, 200))

    for a_init, label in selected:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(time_per_start, rem)
        print(f"  Optimizing [{label}]: t={t_alloc:.0f}s, current best={best_score:.8f}")
        try:
            a_opt, mc_opt = optimize_sequence(a_init, time_limit=t_alloc, verbose=True)
            sc_opt = 2.0 * len(a_opt) * mc_opt
            if sc_opt < best_score:
                try_seq(a_opt, label + "_opt")
        except Exception as e:
            print(f"    Error optimizing [{label}]: {e}")

    # === Phase 2: Dimension exploration on best ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        target_ns = [384, 512, 640, 768, 896, 1024, 1280, 1536]
        for target_n in target_ns:
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 40:
                break
            t_alloc = min(rem * 0.3, 100)
            a_interp = interpolate_sequence(best_a, target_n)
            try:
                a_opt, sc = optimize_sequence(a_interp, time_limit=t_alloc)
                sc_val = 2.0 * target_n * sc
                if sc_val < best_score:
                    try_seq(a_opt, f"dim_{target_n}")
            except Exception:
                pass

    # === Phase 3: Aggressive final polish ===
    print(f"\n=== Phase 3: Final polish (best={best_score:.8f}) ===")
    if best_a is not None:
        # Extended MW from best
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_mw = min(rem * 0.4, 150)
            a_mw, mc_mw = multiplicative_weight_descend(
                best_a, time_limit=t_mw, lr_init=0.02, decay=0.998, verbose=True)
            if 2.0 * len(a_mw) * mc_mw < best_score:
                try_seq(a_mw, "final_mw")

        # Targeted CD
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_tcd = min(rem * 0.5, 120)
            a_tcd, mc_tcd = targeted_coordinate_descent(
                best_a, time_limit=t_tcd, verbose=True)
            if 2.0 * len(a_tcd) * mc_tcd < best_score:
                try_seq(a_tcd, "final_tcd")

        # Standard CD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_cd, mc_cd = coordinate_descent(
                best_a, time_limit=min(rem, 90), verbose=True)
            if 2.0 * len(a_cd) * mc_cd < best_score:
                try_seq(a_cd, "final_cd")

        # Perturbation + re-optimize
        for trial in range(3):
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            n_best = len(best_a)
            noise = rng.standard_normal(n_best) * 0.005
            a_pert = best_a + noise
            a_pert = normalize(a_pert)
            t_pert = min(rem * 0.4, 60)
            try:
                a_opt, sc = optimize_sequence(a_pert, time_limit=t_pert)
                sc_val = 2.0 * n_best * sc
                if sc_val < best_score:
                    try_seq(a_opt, f"perturb_{trial}")
            except Exception:
                pass

    # Fallback
    if best_a is None:
        best_a = generate_cosine(768)
        best_score = compute_score(best_a)

    # Final verification with evaluate_sequence
    final_list = [float(x) for x in best_a]
    try:
        final_score = float(evaluate_sequence(final_list))
        if np.isfinite(final_score):
            best_score = final_score
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return final_list
