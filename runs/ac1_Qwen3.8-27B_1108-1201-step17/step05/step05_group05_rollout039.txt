    """Solve LP to find best direction."""
    n = len(f)
    b = cp.Variable(n)
    constraints = [b >= 0]

    # (f*b)[k] = sum_j f[j] * b[k-j] where b[k-j] is 0 if k-j not in [0,n-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Remez algorithm
with coordinate descent polishing. The LP-based approach solves a linear
program at each step to find the optimal direction that reduces the peak
of the auto-convolution, providing a more informed step than subgradient
methods.
"""

import numpy as np
import time
from scipy.linalg import toeplitz

# Try to import cvxpy
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, tau, top_k=100):
    """Compute smooth approximation of subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    return grad, conv


def build_conv_matrix(f):
    """Build matrix C such that C @ b = conv(f, b)."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        C[k, js:je + 1] = f[js:je + 1]
    return C


def solve_lp_direction(f, M, solver=None):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    if not HAS_CVXPY:
        return None
    
    C = build_conv_matrix(f)
    b = cp.Variable(n)
    constraints = [C @ b <= M, b >= 0]
    
    if solver is None:
        solver = cp.CLARABEL
    elif solver == 'CLARABEL' and not cp.installed_solvers().issuperset([cp.CLARABEL]):
        solver = cp.ECOS
    
    try:
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        prob.solve(solver=solver)
        if b.value is not None and np.all(np.isfinite(b.value)):
            return np.maximum(b.value, 0.0)
    except Exception:
        pass
    return None


def lp_remez(f_init, time_limit, max_iters=500, verbose=True):
    """LP-based Remez algorithm."""
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f_init)) / len(f_init)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    # Determine solver
    solver = None
    if HAS_CVXPY:
        installed = cp.installed_solvers()
        if cp.CLARABEL in installed:
            solver = cp.CLARABEL
        elif cp.MOSEK in installed:
            solver = cp.MOSEK
        elif cp.ECOS in installed:
            solver = cp.ECOS
        elif cp.SCS in installed:
            solver = cp.SCS
        else:
            solver = None
    
    if solver is None:
        print("  No LP solver available, skipping LP phase.")
        return best_f, best_M
    
    while time.monotonic() - t0 < time_limit and no_improve < 60:
        iter_count += 1
        M = fast_convolve(f).max()
        
        # Solve LP
        b = solve_lp_direction(f, M, solver=solver)
        if b is None or b.sum() < 1e-10:
            # Try with slightly larger M
            b = solve_lp_direction(f, M * 1.001, solver=solver)
            if b is None or b.sum() < 1e-10:
                no_improve += 1
                continue
        
        b = b / b.sum()
        
        # Line search
        best_t = 0.1
        best_M_new = M
        for t in [0.05, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1.0 - t) * f + t * b
            f_trial = np.maximum(f_trial, 0.0)
            ss = f_trial.sum()
            if ss < 1e-10:
                continue
            f_trial = f_trial / ss
            M_trial = fast_convolve(f_trial).max()
            if M_trial < best_M_new:
                best_M_new = M_trial
                best_t = t
        
        f = (1.0 - best_t) * f + best_t * b
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        M = fast_convolve(f).max()
        
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            no_improve = 0
            if verbose and iter_count % 10 == 0:
                print(f"  LP iter {iter_count}: M={best_M:.10f}, score={2*n*best_M:.6f}")
        else:
            no_improve += 1
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter {iter_count}: M={M:.10f}, best_M={best_M:.10f}, t={elapsed:.0f}s, no_imp={no_improve}")
    
    return best_f, best_M


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 2:
            break
    
    return best_a, best_obj


def optimize_subgrad(a0, time_limit):
    """Subgradient optimization with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    tau = 80.0
    lr = 0.04
    momentum = 0.82
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 400:
            momentum = min(0.95, momentum + 0.005)
            lr = max(0.003, lr * 0.97)
        elif no_improve > 200:
            tau = min(500.0, tau * 1.01)
            momentum = min(0.92, momentum + 0.002)
        else:
            tau = min(300.0, tau * 1.005)
        
        grad, conv = compute_grad(a, tau, top_k=min(150, n))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Remez algorithm
as the primary optimizer (solving a LP at each step to find the optimal
direction reducing the auto-convolution peak), combined with coordinate
descent polishing and subgradient refinement.
"""

import numpy as np
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, tau, top_k=100):
    """Compute smooth approximation of subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    return grad, conv


def build_conv_matrix(f):
    """Build matrix C such that (C @ b)[k] = conv(f,b)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        C[k, js:je + 1] = f[js:je + 1]
    return C


def solve_lp_direction(f, M, solver=None):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b or None."""
    if not HAS_CVXPY:
        return None
    n = len(f)
    C = build_conv_matrix(f)
    b = cp.Variable(n)
    constraints = [C @ b <= M, b >= 0]
    if solver is None:
        installed = cp.installed_solvers()
        if cp.CLARABEL in installed:
            solver = cp.CLARABEL
        elif cp.MOSEK in installed:
            solver = cp.MOSEK
        elif cp.ECOS in installed:
            solver = cp.ECOS
        elif cp.SCS in installed:
            solver = cp.SCS
        else:
            return None
    try:
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        prob.solve(solver=solver)
        if b.value is not None and np.all(np.isfinite(b.value)):
            return np.maximum(b.value, 0.0)
    except Exception:
        pass
    return None


def get_solver():
    """Determine which CVXPY solver to use."""
    if not HAS_CVXPY:
        return None
    installed = cp.installed_solvers()
    if cp.CLARABEL in installed:
        return cp.CLARABEL
    elif cp.MOSEK in installed:

    elif cp.ECOS in installed:

    return None


def lp_remez(f_init, time_limit, max_iters=500, verbose=True):
    """LP-based Remez algorithm to minimize max(conv(f,f))."""
    f = np.maximum(f_init.astype(np.float64), 0.0)

    no_improve = 0
    iter_count = 0

        if verbose:
            print("  No LP solver available, skipping LP phase.")
        return best_f, best_M

        best_M_new = M

def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}")

        if not improved and pass_count > 2:
            break

    return best_a, best_obj


def optimize_subgrad(a0, time_limit, seed=0):
    """Subgradient optimization with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    v = np.zeros(n)
    tau = 80.0
    lr = 0.04
    momentum = 0.82
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        if no_improve > 400:
            momentum = min(0.95, momentum + 0.005)
            lr = max(0.003, lr * 0.97)
        elif no_improve > 200:
            tau = min(500.0, tau * 1.01)
            momentum = min(0.92, momentum + 0.002)
        else:
            tau = min(300.0, tau * 1.005)

        grad, _ = compute_grad(a, tau, top_k=min(150, n))
        grad_centered = grad - grad.mean()

        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a + v, target_sum=1.0)
        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj:
            a_ext = project_simplex(a + 2.0 * v, target_sum=1.0)
            obj_ext = fast_convolve(a_ext).max()
            if obj_ext < obj_new:
                a_new = a_ext
                obj_new = obj_ext

        a = a_new

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            noise_scale = 0.003 + 0.004 * min(no_improve // 500, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.03
            tau = 60.0
            momentum = 0.82

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def pair_swap_search(a0, time_limit, seed=0):
    """Randomized pair-swap local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    support = np.where(a > 1e-8)[0]
    if len(support) == 0:
        support = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue

            for frac in [0.05, 0.1, 0.2, 0.3]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-based Remez algorithm (primary),
    coordinate descent, pair-swap search, and subgradient refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n = 768
        parent = np.ones(n) / n
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using default parent: n={n}, score={best_score:.6f}")

    n = len(parent)

    # === Phase 1: LP-based Remez (main driver) ===
    print(f"\n=== Phase 1: LP-based Remez (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 60:
        t_lp = min(rem, 1000)
        print(f"  LP time budget: {t_lp:.0f}s")
        f_opt, M_opt = lp_remez(parent, time_limit=t_lp, max_iters=2000)
        score = 2 * n * M_opt
        if score < best_score:
            best_score = score
            best_seq = f_opt.copy()
            print(f"  Phase 1 improved: {best_score:.6f}")
        else:
            print(f"  Phase 1: M_opt={M_opt:.8f}, score={score:.6f} (no improvement)")

    # === Phase 2: Coordinate descent polish ===
    print(f"\n=== Phase 2: CD Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 30 and best_seq is not None:
        t_cd = min(rem * 0.6, 180)
        print(f"  CD time: {t_cd:.0f}s")
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 2 improved: {best_score:.6f}")

    # === Phase 3: LP from CD result ===
    print(f"\n=== Phase 3: LP from CD result (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 60 and best_seq is not None:
        t_lp2 = min(rem * 0.5, 400)
        print(f"  LP2 time: {t_lp2:.0f}s")
        f_opt, M_opt = lp_remez(best_seq, time_limit=t_lp2, max_iters=1000)
        score = 2 * n * M_opt
        if score < best_score:
            best_score = score
            best_seq = f_opt.copy()
            print(f"  Phase 3 improved: {best_score:.6f}")

    # === Phase 4: Pair swap search ===
    print(f"\n=== Phase 4: Pair Swap (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_ps = min(rem * 0.4, 100)
            print(f"  PS time: {t_ps:.0f}s")
            a_opt, obj = pair_swap_search(best_seq, time_limit=t_ps, seed=seed + 1)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Phase 4 improved: {best_score:.6f}")

    # === Phase 5: Subgradient with perturbations ===
    print(f"\n=== Phase 5: Subgradient Perturb (best={best_score:.6f}) ===")
    if best_seq is not None:
        for p in range(4):
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_sg = min(rem / (4 - p), 80)

            if p == 0:
                noise = rng.standard_normal(n) * 0.003
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            elif p == 1:
                f_init = 0.92 * best_seq + 0.08 * (np.ones(n) / n)
                f_init = project_simplex(f_init, target_sum=1.0)
            elif p == 2:
                f_init = 0.95 * best_seq + 0.05 * best_seq[::-1]
                f_init = project_simplex(f_init, target_sum=1.0)
            else:
                noise = rng.standard_normal(n) * 0.01
                f_init = project_simplex(best_seq + noise, target_sum=1.0)

            a_opt, obj = optimize_subgrad(f_init, time_limit=t_sg, seed=seed + 100 + p)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  SG perturb {p} improved: {best_score:.6f}")

    # === Phase 6: Another LP round from best ===
    print(f"\n=== Phase 6: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 60 and best_seq is not None:
        t_lp3 = min(rem * 0.5, 200)
        print(f"  LP3 time: {t_lp3:.0f}s")
        f_opt, M_opt = lp_remez(best_seq, time_limit=t_lp3, max_iters=500, verbose=False)
        score = 2 * n * M_opt
        if score < best_score:
            best_score = score
            best_seq = f_opt.copy()
            print(f"  Phase 6 improved: {best_score:.6f}")

    # === Phase 7: Final CD ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_cd = min(rem * 0.7, 100)
            print(f"  Final CD time: {t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Phase 7 improved: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
