def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, T):
    """Compute gradient of smoothed log-sum-exp of conv(f,f)."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    N = 1 << (3 * n).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, conv


def build_lp_matrix(f, n):
    """Build the A_ub matrix for the LP direction finding."""
    A_ub = np.zeros((2 * n - 1, n))

        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    return A_ub


def lp_direction(f, M, n, A_ub_cache=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    if A_ub_cache is None:
        A_ub = build_lp_matrix(f, n)
    else:
        A_ub = A_ub_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    print(f"\n=== Phase 1: Smooth Opt ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t1 = min(rem * 0.12, 150)
        print(f"  Smooth n={len(best_seq)}, t={t1:.0f}s")
        a_opt, _ = smooth_optimize(best_seq, time_limit=t1, T_start=0.15, T_end=0.004, verbose=True)
        record(a_opt, "smooth_p1")

    # === Phase 2: Main LP-FW (core, ~900s) ===
    print(f"\n=== Phase 2: LP-FW Main ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t2 = min(rem * 0.52, 950)
        print(f"  LP-FW n={len(best_seq)}, t={t2:.0f}s")
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t2, verbose=True)
        record(a_opt, "lpfw_main")

    # === Phase 3: Second smooth pass (80s) ===
    print(f"\n=== Phase 3: Smooth Opt 2 ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t3 = min(rem * 0.06, 80)
        a_opt, _ = smooth_optimize(best_seq, time_limit=t3, T_start=0.04, T_end=0.002, verbose=True)
        record(a_opt, "smooth_p3")

    # === Phase 4: Targeted CD (60s) ===
    print(f"\n=== Phase 4: Targeted CD ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t4 = min(rem * 0.05, 60)
        a_opt, _ = targeted_cd(best_seq, time_limit=t4)
        record(a_opt, "cd_targeted")

    # === Phase 5: More LP-FW (250s) ===
    print(f"\n=== Phase 5: LP-FW 2 ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t5 = min(rem * 0.35, 250)
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t5, verbose=True)
        record(a_opt, "lpfw_2")

    # === Phase 6: Longer sequences ===
    print(f"\n=== Phase 6: Longer Sequences ===")
    for target_n in [1024, 1536, 2048]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        if len(best_seq) == target_n:
            continue
        f_interp = interpolate_seq(best_seq, target_n)
        t6 = min(rem * 0.3, 120)
        a_opt, _ = lpfw_optimize(f_interp, time_limit=t6)
        record(a_opt, f"lpfw_n{target_n}")

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        t7 = min(rem * 0.6, 100)
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t7, verbose=True)
        record(a_opt, "final_lpfw")

    rem = deadline - time.monotonic()
    if rem > 15 and best_seq is not None:
        t8 = min(rem * 0.5, 50)
        a_opt, _ = smooth_optimize(best_seq, time_limit=t8, T_start=0.02, T_end=0.001)
        record(a_opt, "final_smooth")

    rem = deadline - time.monotonic()
    if rem > 15 and best_seq is not None:
        t9 = min(rem * 0.5, 40)
        a_opt, _ = targeted_cd(best_seq, time_limit=t9)
        record(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = safe_evaluate(best_seq)

    # Final verification
    fs = safe_evaluate(best_seq)
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
