Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Adam optimizer with anti-symmetric perturbation to target the center peak
2. Adaptive log-sum-exp focusing (dynamic beta and top-k weighting)

4. Peak-shaving heuristic for direct reduction of max convolution value

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(v):
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def compute_gradient(a, beta, top_k=None):
    """
    Compute gradient of log-sum-exp approximation of max(conv(a,a)).
    Returns grad_a (gradient w.r.t. a) and the smooth objective value.
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    
    if top_k is not None and top_k < len(conv):
        # Focus on top-k peaks
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
        shifted = beta * (vals - vals.max())
        exp_s = np.exp(shifted)
        w_sparse = exp_s / exp_s.sum()
        # Only compute gradient from top-k
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(a, N)
        # For each peak index, accumulate contribution
        # grad_a[j] = 2 * sum_k w_k * a[k-j]
        # We can do this as: for each k in top_k, add w_k * a[k-j] for all j
        # This is like a weighted cross-correlation
        # Use FFT: create a sparse weight vector
        W = np.zeros(len(conv))
        W[idx] = w_sparse
        W = np.fft.rfft(W, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R
        obj = vals.max() + np.log(exp_s.sum()) / beta
    else:
        shifted = beta * (conv - conv_max)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R
        obj = conv_max + np.log(exp_s.sum()) / beta
    
    return obj, grad_a, conv

def softmax_grad(t, beta, top_k=None):
    """Compute gradient in t-space (for softmax parameterization)."""
    n = len(t)
    a = softmax_fn(t)
    obj, grad_a, conv = compute_gradient(a, beta, top_k)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj * 2 * n, grad_t * 2 * n  # Scale by 2n for the full objective

def adam_optimize(a0, time_limit, beta_start=20.0, beta_end=200.0, 
                   lr=0.01, top_k_frac=0.1, verbose=False):
    """Adam optimizer in softmax parameterization with adaptive beta."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    t = np.log(a + 1e-30)
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    best_obj = fast_convolve(a).max() * 2 * n  # Scale for actual objective (sum=1)
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        # Adaptive beta: increase over time
        progress = min(1.0, step / max(1, int(time_limit / 0.5)))
        beta = beta_start + (beta_end - beta_start) * progress**2
        
        # Adaptive top_k: focus more as we progress
        top_k = max(5, int(top_k_frac * (2 * n - 1) * (1.0 - 0.5 * progress)))
        
        obj, grad_t = softmax_grad(t, beta, top_k)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_t
        v = beta2 * v + (1 - beta2) * grad_t**2
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        
        t = t - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = softmax_fn(t)
        conv = fast_convolve(a_new)
        max_conv = conv.max()
        actual_obj = 2 * n * max_conv  # Since sum = 1
        
        if actual_obj < best_obj - 1e-10:
            best_obj = actual_obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and step % 100 == 0:
                print(f"    Adam step {step}: obj={best_obj:.8f}, beta={beta:.1f}")
        else:
            no_improve += 1
            if no_improve > 200:
                cur_lr *= 0.8
                if no_improve > 500:
                    # Restart with perturbation
                    rng = np.random.default_rng(step)
                    t = np.log(best_a + 1e-30) + rng.standard_normal(n) * 0.01
                    m = np.zeros(n)
                    v = np.zeros(n)
                    cur_lr = lr
                    no_improve = 0
                    step = 0
        
        cur_lr = max(cur_lr, 0.0005)
    
    return best_a, best_obj

def antisymmetric_perturb(a, strength=0.1, rng=None):
    """Add anti-symmetric perturbation to reduce center peak."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    # Generate random anti-symmetric vector
    p = rng.standard_normal(n)
    p = p - p[::-1]  # Make anti-symmetric: p[n-1-j] = -p[j]
    # Ensure center is 0 if n is odd
    if n % 2 == 1:
        p[n // 2] = 0
    p = p / (np.std(p) + 1e-10)
    a_new = a + strength * p
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

def peak_shave(a, n_shaves=50, verbose=False):
    """Directly shave peaks by reducing contributing elements."""
    n = len(a)
    a_work = a.copy()
    best_a = a_work.copy()
    best_obj = fast_convolve(a_work).max()
    rng = np.random.default_rng(123)
    
    for _ in range(n_shaves):
        conv = fast_convolve(a_work)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        
        # Find the pair that contributes most to this peak
        # conv[max_idx] = sum_j a[j] * a[max_idx - j]
        j_range = np.arange(max(0, max_idx - n + 1), min(n - 1, max_idx) + 1)
        if len(j_range) == 0:
            break
        products = a_work[j_range] * a_work[max_idx - j_range]
        j_star = j_range[np.argmax(products)]
        
        # Reduce a[j_star] slightly, redistribute to neighbors
        reduction = 0.001 * a_work[j_star]
        a_work[j_star] -= reduction
        # Redistribute to j_star-1 and j_star+1 (if they exist)
        if j_star > 0:
            a_work[j_star - 1] += reduction / 2
        if j_star < n - 1:
            a_work[j_star + 1] += reduction / 2
        
        a_work = np.maximum(a_work, 0)
        a_work = a_work / a_work.sum()
        
        conv_new = fast_convolve(a_work)
        if conv_new.max() < best_obj - 1e-12:
            best_obj = conv_new.max()
            best_a = a_work.copy()
    
    return best_a, best_obj * 2 * n  # Scale for objective (sum=1)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def coordinate_descent_step(a, n_trials=5):
    """Try to improve by adjusting individual coordinates."""
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
    
    # Try random coordinates
    for _ in range(n_trials):
        j = rng.integers(0, n)
        old_val = a[j]
        if old_val < 1e-10:
            continue
        
        for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]:
            new_val = old_val * mult
            if new_val < 0 or new_val > 0.99:
                continue
            scale = (1.0 - new_val) / (1.0 - old_val + 1e-15)
            a_trial = a.copy()
            a_trial[j] = new_val
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a_trial[mask] *= scale
            a_trial = a_trial / a_trial.sum()
            
            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_trial.copy()
    
    return best_a, best_obj * 2 * n

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_smooth_grad(a, beta, top_k=None):
    """
    Compute smooth log-sum-exp objective and gradient w.r.t. a.
    Returns (smooth_obj, grad_a, conv).
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    total_len = len(conv)

    if top_k is not None and top_k < total_len:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
        shifted = beta * (vals - vals.max())
        exp_s = np.exp(shifted)
        w_sparse = exp_s / exp_s.sum()
        W = np.zeros(total_len)
        W[idx] = w_sparse
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(a, N)
        Wf = np.fft.rfft(W, N)
        R = np.fft.irfft(Wf * np.conj(A), N)[:n]
        grad_a = 2.0 * R
        obj = vals.max() + np.log(exp_s.sum()) / beta
    else:
        shifted = beta * (conv - conv_max)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(a, N)
        Wf = np.fft.rfft(w, N)
        R = np.fft.irfft(Wf * np.conj(A), N)[:n]
        grad_a = 2.0 * R
        obj = conv_max + np.log(exp_s.sum()) / beta

    return obj, grad_a, conv


def adam_nesterov_optimize(a0, time_limit, beta_start=20.0, beta_end=300.0,
                           lr=0.008, top_k_frac=0.12, nesterov_mu=0.9,
                           verbose=False):
    """Adam with Nesterov momentum in softmax parameterization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    t = np.log(a + 1e-30)
    m = np.zeros(n)
    v = np.zeros(n)
    v_nest = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8

    best_obj = fast_convolve(a).max() * 2 * n
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    min_steps = int(time_limit * 2)

    while time.monotonic() - t0 < time_limit:
        step += 1

        progress = min(1.0, step / max(1, min_steps))
        beta = beta_start + (beta_end - beta_start) * progress ** 1.5
        top_k = max(4, int(top_k_frac * (2 * n - 1) * (1.0 - 0.6 * progress)))

        a_curr = softmax_fn(t)
        obj, grad_a, conv = compute_smooth_grad(a_curr, beta, top_k)
        dot = np.dot(a_curr, grad_a)
        grad_t = a_curr * (grad_a - dot)

        # Nesterov: look-ahead
        t_look = t - nesterov_mu * v_nest
        a_look = softmax_fn(t_look)
        obj_l, grad_a_l, _ = compute_smooth_grad(a_look, beta, top_k)
        dot_l = np.dot(a_look, grad_a_l)
        grad_t_l = a_look * (grad_a_l - dot_l)

        m = beta1 * m + (1 - beta1) * grad_t_l
        v = beta2 * v + (1 - beta2) * grad_t_l ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        v_nest = nesterov_mu * v_nest + cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        t = t - v_nest

        a_new = softmax_fn(t)
        max_conv = fast_convolve(a_new).max()
        actual_obj = 2 * n * max_conv

        if actual_obj < best_obj - 1e-10:
            best_obj = actual_obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and step % 200 == 0:
                print(f"    AdamN step {step}: obj={best_obj:.8f}, beta={beta:.1f}, tk={top_k}")
        else:
            no_improve += 1
            if no_improve > 300:
                cur_lr *= 0.75
            if no_improve > 800:
                rng = np.random.default_rng(step * 7 + 3)
                t = np.log(best_a + 1e-30) + rng.standard_normal(n) * 0.008
                v_nest = np.zeros(n)
                m = np.zeros(n)
                v = np.zeros(n)
                cur_lr = lr
                no_improve = 0
                step = 0

        cur_lr = max(cur_lr, 0.0003)

    return best_a, best_obj


def antisymmetric_perturb(a, strength=0.08, rng=None):
    """Add anti-symmetric perturbation to reduce center convolution peak."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    p = rng.standard_normal(n)
    p = 0.5 * (p - p[::-1])
    if n % 2 == 1:

    p_norm = np.linalg.norm(p)
    if p_norm > 1e-10:
        p = p / p_norm * strength
    a_new = a + p
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

def peak_shave(a, max_iters=3000, verbose=False):
    """Directly shave the peak by reducing contributing elements and redistributing."""
    n = len(a)
    a_work = a.copy()
    best_a = a_work.copy()
    best_obj = fast_convolve(a_work).max()
    rng = np.random.default_rng(42)

    for it in range(max_iters):
        conv = fast_convolve(a_work)
        max_val = conv.max()
        if max_val < best_obj:
            best_obj = max_val
            best_a = a_work.copy()

        if verbose and it % 500 == 0:
            print(f"    PeakShave iter {it}: max_conv={max_val:.8f}, best={best_obj:.8f}")

        # Find top 3 peak lags and try to reduce them
        top_lags = np.argsort(conv)[-3:]
        for max_idx in top_lags:
            max_idx = int(max_idx)
            j_lo = max(0, max_idx - n + 1)
            j_hi = min(n - 1, max_idx)
            if j_lo > j_hi:
                continue
            j_range = np.arange(j_lo, j_hi + 1)
            products = a_work[j_range] * a_work[max_idx - j_range]
            j_star = j_range[np.argmax(products)]

            # Reduce a[j_star] slightly
            reduction = 0.0005 * a_work[j_star]
            a_work[j_star] -= reduction

            # Redistribute to neighbors (preferring the one with lower value)
            neighbors = []
            if j_star > 0:
                neighbors.append(j_star - 1)
            if j_star < n - 1:
                neighbors.append(j_star + 1)
            if neighbors:
                for nb in neighbors:
                    a_work[nb] += reduction / len(neighbors)

        a_work = np.maximum(a_work, 0)
        a_work = a_work / a_work.sum()

        # Check progress
        conv_check = fast_convolve(a_work)
        if conv_check.max() < best_obj - 1e-12:
            best_obj = conv_check.max()
            best_a = a_work.copy()

    return best_a, best_obj * 2 * n


def coordinate_descent_round(a, n_rounds=500, verbose=False):
    """Coordinate descent: try to improve by adjusting individual coordinates."""
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)

    for rnd in range(n_rounds):
        improved = False
        for _ in range(20):
            j = int(rng.integers(0, n))
            old_val = a[j]
            if old_val < 1e-12:
                continue

            for mult in [0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.2]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()

                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break

        if verbose and rnd % 100 == 0:
            print(f"    CD round {rnd}: best_obj={best_obj:.8f}")
        if not improved and rnd > 50:
            break

    return best_a, best_obj * 2 * n


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def harmonic_perturb(a, rng, n_harmonics=3, max_amp=0.05):

    n = len(a)
    x = np.arange(n)
    pert = np.zeros(n)
    for h in range(1, n_harmonics + 1):

        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.001, max_amp / h)
        pert += amp * np.cos(2 * np.pi * freq * x / n + phase)
    a_new = a + pert
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.3, 120)
        print(f"  Interpolating to n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = adam_nesterov_optimize(
            f_interp, time_limit=t_alloc,
            beta_start=15.0, beta_end=200.0,
            lr=0.005, top_k_frac=0.1, nesterov_mu=0.9)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 4: Deep refinement on best ===
    print(f"\n=== Phase 4: Deep refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 4a: Fine Adam with sharp beta
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.4, 100)
            a_opt, _ = adam_nesterov_optimize(
                best_seq, time_limit=t_alloc,
                beta_start=50.0, beta_end=500.0,
                lr=0.002, top_k_frac=0.04, nesterov_mu=0.95)
            try_seq(a_opt, "deep_fine_adamn")

        # 4b: Peak shaving
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem * 0.3, 60)
            n_shaves = min(5000, int(t_alloc * 80))
            a_opt, _ = peak_shave(best_seq, max_iters=n_shaves)
            try_seq(a_opt, "peak_shave")

        # 4c: Coordinate descent
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem * 0.5, 60)
            n_rounds = min(2000, int(t_alloc * 30))
            a_opt, _ = coordinate_descent_round(best_seq, n_rounds=n_rounds)
            try_seq(a_opt, "coord_desc")

        # 4d: Anti-symmetric + fine optimization
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem * 0.6, 40)
            a_anti = antisymmetric_perturb(best_seq, strength=0.015, rng=rng)
            a_opt, _ = adam_nesterov_optimize(
                a_anti, time_limit=t_alloc,
                beta_start=100.0, beta_end=800.0,
                lr=0.001, top_k_frac=0.03, nesterov_mu=0.95)
            try_seq(a_opt, "deep_anti_fine")

        # 4e: Another peak shave from current best
        rem = deadline - time.monotonic() - 40
        if rem > 10 and best_seq is not None:
            t_alloc = min(rem * 0.7, 30)
            n_shaves = min(3000, int(t_alloc * 100))
            a_opt, _ = peak_shave(best_seq, max_iters=n_shaves)
            try_seq(a_opt, "peak_shave_2")

    # === Phase 5: Final polishing ===
    print(f"\n=== Phase 5: Final polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        for attempt in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            t_alloc = min(rem * 0.6, 35)

            if attempt % 3 == 0:
                # Very fine Adam
                a_opt, _ = adam_nesterov_optimize(
                    best_seq, time_limit=t_alloc,
                    beta_start=200.0, beta_end=1000.0,
                    lr=0.0008, top_k_frac=0.02, nesterov_mu=0.95)
                try_seq(a_opt, f"final_fine_{attempt}")
            elif attempt % 3 == 1:
                # Small perturbation + optimize
                a_pert = best_seq + rng.standard_normal(n_best) * 0.003
                a_pert = project_simplex(a_pert)
                a_opt, _ = adam_nesterov_optimize(
                    a_pert, time_limit=t_alloc,
                    beta_start=80.0, beta_end=400.0,
                    lr=0.002, top_k_frac=0.05, nesterov_mu=0.9)
                try_seq(a_opt, f"final_pert_{attempt}")
            else:
                # Peak shave + coordinate descent combo
                a_ps, _ = peak_shave(best_seq, max_iters=min(2000, int(t_alloc * 60)))
                a_cd, _ = coordinate_descent_round(a_ps, n_rounds=min(500, int(t_alloc * 10)))
                try_seq(a_cd, f"final_ps_cd_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
