"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Fast multiplicative weights / subgradient with adaptive step
2. LP-Frank-Wolfe with top-K constraint relaxation
3. Coordinate descent for peak-specific optimization
4. Multi-length exploration
5. Structured initializations (Fourier modes, windows)
6. Final high-precision polish
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, peak_pos):
    """Compute subgradient of max(auto_conv) at given peak position."""
    n = len(f)
    g = np.zeros(n)
    js = max(0, peak_pos - n + 1)
    je = min(n - 1, peak_pos)
    idx = np.arange(js, je + 1)
    g[idx] = 2.0 * f[peak_pos - idx]
    return g

def mw_subgrad_optimize(f0, time_limit, lr=0.3, verbose=False):
    """Multiplicative weights / subgradient with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    cur_lr = lr
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    x_prev = f.copy()
    g_prev = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peak_val = conv.max()
        peak_pos = int(np.argmax(conv))
        
        g = compute_subgradient(f, peak_pos)
        
        # BB step size (for subgradients, this is approximate)
        if g_prev is not None:
            s_vec = f - x_prev
            y_vec = g - g_prev
            sy = np.dot(s_vec, y_vec)
            if sy > 1e-14:
                bb_lr = np.dot(s_vec, s_vec) / sy
                bb_lr = max(1e-4, min(bb_lr, 1.0))
                cur_lr = 0.7 * cur_lr + 0.3 * bb_lr
        
        # Multiplicative update (mirror descent with KL)
        f_new = f * np.exp(-cur_lr * g)
        f_new = np.maximum(f_new, 1e-15)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / fs
        
        x_prev = f.copy()
        g_prev = g.copy()
        f = f_new
        
        new_peak = fast_convolve(f).max()
        if new_peak < best_obj - 1e-14:
            best_obj = new_peak
            best_f = f.copy()
            no_improve = 0
            cur_lr = min(cur_lr * 1.01, 2.0)
        else:
            no_improve += 1
            cur_lr *= 0.98
        
        if verbose and it % 500 == 0:
            print(f"      MW it={it} obj={best_obj:.10f} lr={cur_lr:.4f}")
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            cur_lr = max(lr * 0.5, 0.05)
            no_improve = 0
            x_prev = f.copy()
            g_prev = None
    
    return best_f, best_obj

def lp_fw_optimize(f0, time_limit, top_k=20, verbose=False):
    """LP-Frank-Wolfe with top-K constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top-K peaks for constraint relaxation
        k_eff = min(top_k, 2 * n - 1)
        peak_indices = np.argpartition(conv, -k_eff)[-k_eff:]
        
        # Build LP with only top-K constraints
        c = -np.ones(n)
        A_ub = np.zeros((len(peak_indices), n))
        b_ub = np.zeros(len(peak_indices))
        
        for row, k in enumerate(peak_indices):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            A_ub[row, idx] = f[k - idx]
            b_ub[row] = M
        
        bounds = [(0, None)] * n
        options = {'time_limit': 30, 'presolve': True}
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options=options)
        
        if not res.success or res.x is None:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 17 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
                M = fast_convolve(f).max()
            continue
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-12:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      LP-FW it={it} obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_f, best_obj

def coordinate_descent_peak(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
