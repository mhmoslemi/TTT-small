"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive subgradient descent
with peak focusing, Nesterov momentum, and multi-phase optimization.
Strategy: LP-FW for coarse optimization, then intensive SG polishing with
adaptive temperature, followed by multi-length search and micro-refinement.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.1
            if m_relax > 5.0:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            m_relax = 1.0
            f = f_new
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def sg_polish_v2(f0, time_limit, verbose=False):
    """
    Advanced subgradient polishing with:
    - Adaptive softmax temperature
    - Nesterov momentum
    - Backtracking line search
    - Periodic restarts with asymmetric perturbations
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_conv = fast_convolve(f)
    best_obj = best_conv.max()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Parameters
    lr = 0.005
    momentum = 0.9
    top_k = max(10, n // 8)
    temp = 0.00005
    temp_decay = 1.002
    lr_decay = 0.999
    no_improve = 0
    it = 0
    v = np.zeros(n)  # momentum

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()

        # Find top-k peaks with softmax weighting
        k_actual = min(top_k, len(conv))
        # Use a sliding window of peaks for smoothness
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        # Softmax weighting (temperature controls focus)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # d(c_k)/d(f_i) = f_{k-i} + f_{i-k}
            # For i in [js, je]: k-i is in [0, n-1] range
            # So d(c_k)/d(f_i) = f[k-i] + f[i-k] if i-k is valid
            f_ki = f[k - idx]  # f[k - i] for i in idx
            # f[i - k] is non-zero only when i >= k, i.e., i-k >= 0 and i-k < n
            ik_mask = (idx >= k) & (idx - k < n)
            f_ik = np.zeros(n)
            if ik_mask.any():
                f_ik[idx[ik_mask]] = f[idx[ik_mask] - k]
            contrib = 2.0 * w[i] * (f_ki + f_ik)
            sg[js:je+1] += contrib

        # Nesterov momentum update
        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)

        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-18:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        elif obj_new < cur_max * (1 + 1e-6):
            # Accept if not too much worse (for exploration)
            f = f_new
            no_improve += 1
        else:
            # Backtrack: reduce step
            f = f_new * 0.5 + f * 0.5
            f = project_simplex(f)
            no_improve += 1

        # Adapt parameters
        lr *= lr_decay
        temp *= temp_decay

        # Periodic restart
        if no_improve > 200:
            scale = 0.001 + 0.002 * min((no_improve - 200) // 100, 5)
            # Asymmetric perturbation
            asym = rng.random(n) - 0.5
            asym = asym - asym.mean()  # zero mean
            f = project_simplex(best_f + scale * (rng.standard_normal(n) + 0.3 * asym))
            v = np.zeros(n)
            lr = 0.003
            temp = 0.00003
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it} ({elapsed:.0f}s): best={best_obj:.12f}, lr={lr:.6f}, temp={temp:.8f}")

    return best_f, best_obj


def coordinate_descent_v2(f0, time_limit, verbose=False):
    """
    Coordinate descent with smart ordering (process coordinates
    in order of their impact on the peak).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Multipliers to try
    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_obj = cur_max
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-16:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                cur_max = best_j_obj
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

def compute_subgradient_topk(f, conv, top_k, temp):
    """Compute subgradient using top-k peaks with softmax weighting."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]

    scaled = (pv - pv.max()) / temp
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(k_actual)
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(k_actual):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        f_ki = f[k - idx]
        f_ik = np.zeros(je - js + 1)
        ik_mask = (idx >= k) & (idx - k < n)
        if ik_mask.any():
            f_ik[ik_mask] = f[idx[ik_mask] - k]
        sg[js:je + 1] += 2.0 * w[i] * (f_ki + f_ik)
    return sg

def coordinate_descent_v2(f0, time_limit, verbose=False):
    """Coordinate descent with adaptive multipliers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    multipliers = [0.6, 0.8, 0.95, 1.05, 1.2, 1.4]
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_max = fast_convolve(f).max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_obj = cur_max
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val >= 0.999 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-16:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                cur_max = best_j_obj
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            # Low frequency
            freq = rng.uniform(0.5, 3.0)
            noise += np.sin(2 * np.pi * freq * np.arange(n) / n) * scale * 0.5
        elif i % 3 == 1:
            # Asymmetric
            asym = np.linspace(0, 1, n) - 0.5
            noise += asym * scale * 2
        else:
            # Block
            block = max(1, n // 8)
            center = rng.integers(0, n)
            noise[max(0, center - block):min(n, center + block)] += scale

        f_pert = project_simplex(f_init + noise)
        a_opt, _ = sg_polish(f_pert, time_limit=time_per_start,
                             lr=0.003, mom=0.93, top_k=25, temp=0.00004)
        obj = fast_convolve(a_opt).max()
        if obj < best_obj:
            best_obj = obj
            best_f = a_opt.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-FW for coarse optimization,
    intensive subgradient polishing with focus mode, coordinate descent,
    multi-length search, and final micro-optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 512
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n={n_init}, score={best_score:.6f}")

    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW (best={best_score:.6f}) ===")
    n_parent = len(parent)
    t_remaining = deadline - time.monotonic()
    t1 = min(250, t_remaining * 0.18)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s, n={n_parent}")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 2: SG Polish (main workhorse) ===
    print(f"\n=== Phase 2: SG Polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t2 = min(rem * 0.35, 450)
        print(f"  Budget: {t2:.0f}s, n={n_best}")
        a_opt, _ = sg_polish(best_seq, time_limit=t2,
                             lr=0.004, mom=0.92, top_k=30, temp=0.00004,
                             verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 3: Focused peak optimization ===
    print(f"\n=== Phase 3: Focused Peak (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t3 = min(rem * 0.25, 200)
        print(f"  Budget: {t3:.0f}s")
        a_opt, _ = focused_peak_reset(best_seq, time_limit=t3)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in range(-48, 49, 4):
        nl = n_best + delta
        if 384 <= nl <= 1536 and nl != n_best:
            candidate_lengths.append(nl)

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic() - 150
        if rem < 15:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(30, rem * 0.08)
        if t_alloc < 8:
            continue
        a_opt, _ = sg_polish(f_interp, time_limit=t_alloc,
                             lr=0.003, mom=0.92, top_k=25, temp=0.00003)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (n={target_n}): {sc:.6f}")
            n_best = target_n
            candidate_lengths = [c for c in candidate_lengths if c != target_n]
            candidate_lengths.sort(key=lambda x: abs(x - n_best))

    # === Phase 5: Multi-start SG ===
    print(f"\n=== Phase 5: Multi-start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        n_starts = 5
        t_per = max(5, rem * 0.15 / n_starts)
        print(f"  {n_starts} starts, {t_per:.0f}s each")
        a_opt, _ = multi_start_sg(best_seq, n_starts=n_starts,
                                  time_per_start=t_per, seed=seed + 1)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t6 = min(rem * 0.5, 120)
        print(f"  Budget: {t6:.0f}s")
        a_opt, _ = coordinate_descent_v2(best_seq, time_limit=t6)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 7: Final SG focus ===
    print(f"\n=== Phase 7: Final Focus (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t7 = min(rem * 0.6, 60)
        a_opt, _ = sg_polish(best_seq, time_limit=t7,
                             lr=0.002, mom=0.94, top_k=15, temp=0.00002,
                             focus_mode=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 8: Micro-perturbation ===
    print(f"\n=== Phase 8: Micro (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(15):
        rem = deadline - time.monotonic()
        if rem < 2:
            break
        scale = 0.0001 + trial * 0.00008
        noise = rng.standard_normal(n_best) * scale
        if trial % 4 == 0:
            noise += (np.linspace(0, 1, n_best) - 0.5) * scale
        elif trial % 4 == 1:
            # Localized perturbation
            center = rng.integers(0, n_best)
            width = max(1, n_best // 12)
            noise[max(0, center - width):min(n_best, center + width)] *= 3
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [m{trial}]: {sc:.6f}")
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem2 * 0.3, 8),
                                     lr=0.001, mom=0.95, top_k=15,
                                     temp=0.00001, focus_mode=True)
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [m{trial}+sg]: {sc2:.6f}")

    # === Final: pointing peak reset ===
    rem = deadline - time.monotonic()
    if rem > 5:
        a_opt, _ = focused_peak_reset(best_seq, time_limit=min(rem * 0.8, 30))
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL FOCUS: {sc:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
