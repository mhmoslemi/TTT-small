                    best_obj = obj
                    best_v = float(v)

            if best_v != v_old:
                a[j] = best_v
                if best_obj < best_score:
                    best_score = best_obj
                improved = True
            else:
                a[j] = v_old

        if verbose and pass_count % 20 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

        if not improved:
            no_improve_count += 1
            if no_improve_count > 5:
                break
        else:
            no_improve_count = 0

        if pass_count > 3000:
            break

    return a, best_score


def coord_descent_fine(a0, time_limit, n_grid=20, verbose=False):
    """Finer coordinate descent for polishing (narrower range, more points)."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    size = 1 << (2 * n - 2).bit_length()
    conv_len = 2 * n - 1

    A_init = np.fft.rfft(a, size)
    b_init = np.fft.irfft(A_init * A_init, size)
    s_init = a.sum()
    best_score = 2.0 * n * b_init[:conv_len].max() / (s_init * s_init)

    t0 = time.monotonic()
    pass_count = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            v_old = a[j]
            if v_old < 1e-10:
                continue

            # Narrow range for fine polishing
            span = max(0.002, v_old * 0.15)
            lo = max(0.0, v_old - span)
            hi = min(1.0, v_old + span)

            grid = np.linspace(lo, hi, n_grid)

            s_cur = a.sum()
            A_cur = np.fft.rfft(a, size)
            b_cur = np.fft.irfft(A_cur * A_cur, size)
            cur_score = 2.0 * n * b_cur[:conv_len].max() / (s_cur * s_cur)

            best_v = v_old
            best_obj = cur_score

            for v in grid:
                if abs(v - v_old) < 1e-13:
                    continue
                a[j] = v
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = v_old
                    continue
                A_new = np.fft.rfft(a, size)
                b_new = np.fft.irfft(A_new * A_new, size)
                obj = 2.0 * n * b_new[:conv_len].max() / (s_new * s_new)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_v = float(v)

            if best_v != v_old:
                a[j] = best_v
                if best_obj < best_score:
                    best_score = best_obj
                improved = True
            else:
                a[j] = v_old

        if verbose and pass_count % 10 == 0:
            print(f"    Fine CD pass {pass_count}: score={best_score:.10f}")

        if not improved:
            no_improve_count += 1
            if no_improve_count > 3:
                break
        else:
            no_improve_count = 0

        if pass_count > 2000:
            break

    return a, best_score


def simulated_annealing(a0, time_limit, verbose=False):
    """Simulated annealing with block moves to escape local minima."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    size = 1 << (2 * n - 2).bit_length()
    conv_len = 2 * n - 1

    s = a.sum()
    A = np.fft.rfft(a, size)
    b = np.fft.irfft(A * A, size)
    cur_score = 2.0 * n * b[:conv_len].max() / (s * s)
    best_score = cur_score
    best_a = a.copy()

    T = 0.005
    T_min = 0.0001
    cooling = 0.998
    rng = np.random.default_rng(99)
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        step += 1

        # Choose move type
        move_type = rng.random()

        if move_type < 0.5:
            # Single coordinate perturbation
            j = rng.integers(0, n)
            scale_factor = rng.uniform(0.5, 1.5)
            a[j] = max(0.0, a[j] * scale_factor)
        elif move_type < 0.8:
            # Block move: reshape a small block
            block_size = rng.integers(3, min(15, n // 4))
            start = rng.integers(0, max(1, n - block_size))
            block = a[start:start + block_size].copy()
            # Randomly perturb the block
            block = block + rng.standard_normal(block_size) * block.mean() * 0.5
            a[start:start + block_size] = np.maximum(block, 0.0)
        else:
            # Distant transfer: move mass from one region to another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            transfer = a[i] * rng.uniform(0.05, 0.3)
            a[i] -= transfer
            a[j] += transfer
            a[i] = max(0.0, a[i])

        # Evaluate
        s_new = a.sum()
        if s_new < 1e-15:
            T *= cooling
            continue
        A_new = np.fft.rfft(a, size)
        b_new = np.fft.irfft(A_new * A_new, size)
        new_score = 2.0 * n * b_new[:conv_len].max() / (s_new * s_new)

        # Acceptance criterion
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                if verbose and step % 5000 == 0:
                    print(f"    SA step {step}: best={best_score:.10f}, T={T:.6f}")
        else:
            # Revert
            if move_type < 0.5:
                a[j] = a[j] / scale_factor if scale_factor > 0 else 0
            # Just restart from best for simplicity on reject of block/distant moves
            # Actually, let's keep it simple: on reject, we already have the old 'a'
            # Wait, we modified 'a' in place. We need to revert.
            # For simplicity, let's always keep a copy before the move.

        T *= cooling

        if verbose and step % 10000 == 0:
            print(f"    SA step {step}: cur={cur_score:.10f}, best={best_score:.10f}, T={T:.6f}")

    return best_a, best_score


def simulated_annealing_v2(a0, time_limit, verbose=False):
    """Improved simulated annealing with proper revert."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    size = 1 << (2 * n - 2).bit_length()
    conv_len = 2 * n - 1

    s = a.sum()
    A = np.fft.rfft(a, size)
    b = np.fft.irfft(A * A, size)
    cur_score = 2.0 * n * b[:conv_len].max() / (s * s)
    best_score = cur_score
    best_a = a.copy()

    T = 0.003
    T_min = 0.00005
    cooling = 0.9985
    rng = np.random.default_rng(777)
    t0 = time.monotonic()
    step = 0
    accept_count = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        step += 1

        a_before = a.copy()
        move_type = rng.random()

        if move_type < 0.4:
            # Perturb 1-3 random coordinates
            n_perturb = rng.integers(1, 4)
            indices = rng.choice(n, size=n_perturb, replace=False)
            for idx in indices:
                a[idx] = max(0.0, a[idx] * rng.uniform(0.6, 1.4))
        elif move_type < 0.7:
            # Transfer mass between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.2)
            transfer = a[i] * frac
            a[i] = max(0.0, a[i] - transfer)
            a[j] += transfer
        else:
            # Block reshape
            block_size = rng.integers(3, min(10, n // 5))
            start = rng.integers(0, max(1, n - block_size))
            block = a[start:start + block_size].copy()
            # Apply random smooth perturbation
            noise = rng.standard_normal(block_size)
            noise -= noise.mean()  # zero mean
            a[start:start + block_size] = np.maximum(block * (1.0 + noise * 0.3), 0.0)

        # Evaluate new state
        s_new = a.sum()
        if s_new < 1e-15:
            a = a_before
            T *= cooling
            continue
        A_new = np.fft.rfft(a, size)
        b_new = np.fft.irfft(A_new * A_new, size)
        new_score = 2.0 * n * b_new[:conv_len].max() / (s_new * s_new)

        # Acceptance
        delta = new_score - cur_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_score = new_score
            accept_count += 1
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
                if verbose and step % 5000 == 0:
                    print(f"    SA2 step {step}: best={best_score:.10f}, T={T:.6f}")
        else:
            a = a_before  # revert

        T *= cooling

        if verbose and step % 20000 == 0:
            print(f"    SA2 step {step}: cur={cur_score:.10f}, best={best_score:.10f}, T={T:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return np.maximum(f_interp, 0.0)


def exponential_sequence(n, alpha):
    a = np.exp(-alpha * np.arange(n))
    return normalize(a)


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac, center_frac=0.5):
    x = np.arange(n) - n * center_frac
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def triangular_sequence(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def smooth_random_sequence(n, rng, kernel_size=20):
    a = rng.random(n)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    return normalize(a)


def propose_candidate(budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Coordinate descent on diverse starts (including parent)
    Phase 2: Simulated annealing for global exploration
    Phase 3: Fine coordinate descent polishing
    Phase 4: Dimension search
    Phase 5: Final polish
    """
    rng = np.random.default_rng(42)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01 or len(a) == 0:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Generate starting points ===
    print("=== Generating starting points ===")
    starts = []

    # Parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    # Structured starts
    for n in [512, 768, 1024]:
        for alpha in [0.001, 0.002, 0.005, 0.01]:
            starts.append((exponential_sequence(n, alpha), f"exp_a{alpha}_n{n}"))
        starts.append((cosine_sequence(n), f"cos_n{n}"))
        for sf in [0.2, 0.3, 0.4]:
            starts.append((gaussian_sequence(n, sf), f"gauss_sf{sf}_n{n}"))
        starts.append((triangular_sequence(n), f"tri_n{n}"))
        starts.append((smooth_random_sequence(n, rng), f"rand_n{n}"))

    # Evaluate all starts
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    for i, (sc, a, label) in enumerate(start_scores[:8]):
        print(f"    Start #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # === Phase 1: Coordinate descent on top diverse starts ===
    print(f"\n=== Phase 1: CD on diverse starts (budget to {deadline - time.monotonic():.0f}s) ===")

    # Select diverse starts (different types/lengths)
    selected = []
    used_types = set()
    for sc, a, label in start_scores:
        if len(selected) >= 5:
            break
        # Extract type key
        if 'parent' in label:
            key = 'parent'
        elif 'exp' in label:
            key = f"exp_n{label.split('_n')[-1]}"
        elif 'cos' in label:
            key = f"cos_n{label.split('_n')[-1]}"
        elif 'gauss' in label:
            key = f"gauss_n{label.split('_n')[-1]}"
        elif 'tri' in label:
            key = f"tri_n{label.split('_n')[-1]}"
        elif 'rand' in label:
            key = f"rand_n{label.split('_n')[-1]}"
        else:
            key = label
        if key not in used_types:
            selected.append((a, label))
            used_types.add(key)

    # Ensure parent is in selected
    if 'parent' not in used_types:
        for sc, a, label in start_scores:
            if 'parent' in label:
                selected.append((a, label))
                used_types.add('parent')
                break

    n_selected = len(selected)
    phase1_budget = min((deadline - time.monotonic()) * 0.35, 500)
    t_per = phase1_budget / max(1, n_selected)

    for a_init, label in selected:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(t_per, rem)
        print(f"  CD [{label}]: budget={t_alloc:.0f}s")
        a_opt, sc_opt = coord_descent(a_init, time_limit=t_alloc, n_grid=12, verbose=True)
        try_seq(a_opt, label + "_cd")

    # === Phase 2: Simulated Annealing ===
    print(f"\n=== Phase 2: Simulated Annealing (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 300
        sa_budget = min(rem * 0.3, 250)
        if sa_budget > 30:
            print(f"  SA budget: {sa_budget:.0f}s")
            a_sa, sc_sa = simulated_annealing_v2(best_a, time_limit=sa_budget, verbose=True)
            try_seq(a_sa, "sa")
            # Also run CD from SA result
            rem2 = deadline - time.monotonic() - 200
            if rem2 > 30:
                a_opt, sc_opt = coord_descent(a_sa, time_limit=min(rem2 * 0.5, 100), n_grid=12)
                try_seq(a_opt, "sa_cd")

    # === Phase 3: Fine coordinate descent on best ===
    print(f"\n=== Phase 3: Fine CD (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 150
        fine_budget = min(rem * 0.3, 200)
        if fine_budget > 20:
            print(f"  Fine CD budget: {fine_budget:.0f}s")
            a_fine, sc_fine = coord_descent_fine(best_a, time_limit=fine_budget, n_grid=18, verbose=True)
            try_seq(a_fine, "fine_cd")

    # === Phase 4: Dimension search ===
    print(f"\n=== Phase 4: Dimension search (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        target_ns = []
        for tn in [384, 512, 640, 768, 896, 1024, 1280, 1536, 1024, 1280]:
            if tn != best_n and tn not in target_ns and 200 <= tn <= 2000:
                target_ns.append(tn)

        rem = deadline - time.monotonic() - 100
        per_dim = min(rem / max(1, len(target_ns)), 80)

        for tn in target_ns:
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(per_dim, rem * 0.7)
            a_interp = interpolate_sequence(best_a, tn)
            a_opt, sc = coord_descent(a_interp, time_limit=t_alloc, n_grid=10)
            try_seq(a_opt, f"dim_{tn}")

    # === Phase 5: Perturbation + re-optimize ===
    print(f"\n=== Phase 5: Perturbation restarts (best={best_score:.8f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        for trial in range(5):
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            # Perturb
            noise_scale = 0.01 * (1 + trial * 0.5)
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = np.maximum(best_a + noise, 0.0)
            a_pert = normalize(a_pert)

            t_alloc = min(rem * 0.5, 60)
            a_opt, sc = coord_descent(a_pert, time_limit=t_alloc, n_grid=12)
            try_seq(a_opt, f"perturb_{trial}_{noise_scale}")

    # === Phase 6: Final fine polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_final, sc_final = coord_descent_fine(best_a, time_limit=min(rem, 120), n_grid=20, verbose=True)
            try_seq(a_final, "final_polish")

    # === Verify with evaluate_sequence ===
    if best_a is None:
        best_a = cosine_sequence(768)

    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return final_list
