Minimizes 2*n*max(auto_conv)/(sum)^2 using:
1. Fast analytical subgradient computation (no FFT in the inner loop)
2. Nesterov-accelerated projected gradient with adaptive log-sum-exp smoothing

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peaks, weights):
    """
    Compute subgradient of max_k (f*f)_k at a, given active peaks and weights.
    Uses the formula: d(f*f)_k / d f_j = 2 * f_{k-j} (if 0 <= k-j < n)
    
    This is computed as: for each peak k with weight w_k,
    add w_k * 2 * flip_shift(a, k) to the gradient.
    
    The "flip_shift" of a by k gives a vector g where g_j = a_{k-j} (if valid).
    This is equivalent to: convolve the impulse train at peaks with 2*a (flipped).
    """
    n = len(a)
    sg = np.zeros(n)
    for k, w in zip(peaks, weights):
        # For each j, sg_j += 2 * w * a_{k-j} (if 0 <= k-j < n)
        # This is: sg += 2 * w * (a flipped, shifted by k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # sg[js:je+1] += 2 * w * a[k-je : k-js+1][::-1]
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * a[k - idx]
    return sg


def nesterov_projected_gradient(a, time_limit, lr=None, decay=0.9999,

    t_k = 1.0
    y = a.copy()  # Nesterov lookahead
    
    if lr is None:

        a = x_new

        # Adaptive learning rate

def parametric_power_law(n, beta, gamma):
    """f_i = (1 + gamma*i)^(-beta), normalized."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + gamma * i) ** (-beta)
    return normalize(a)


def parametric_exponential(n, alpha, asymmetry=0.0):
    """f_i = exp(-alpha*i + asymmetry*i*(n-i)/n^2), normalized."""
    i = np.arange(n, dtype=np.float64)
    t = i / n
    a = np.exp(-alpha * t + asymmetry * t * (1.0 - t))
    return normalize(a)


def parametric_rational(n, a1, a2):
    """f_i = 1/(a1 + a2*i), normalized."""
    i = np.arange(n, dtype=np.float64)
    # Ensure positivity
    vals = a1 + a2 * i
    vals = np.maximum(vals, 0.01)
    a = 1.0 / vals
    return normalize(a)


def parametric_two_exp(n, center, width, height2, center2=None, width2=None):
    """Sum of two exponentials."""
    i = np.arange(n, dtype=np.float64)
    if center2 is None:
        center2 = n - 1 - center
    if width2 is None:
        width2 = width
    a = height2 * np.exp(-((i - center) / width) ** 2) + np.exp(-((i - center2) / width2) ** 2)
    return normalize(a)

def pair_swap_local(a, time_limit, verbose=False):
    """Random pair transfer local search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Batch of swaps
        for _ in range(500):
            if time.monotonic() - t0 > time_limit:
                break
            
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            
            a_i, a_j = a[i], a[j]
            if a_j < 1e-12:
                continue
            
            # Try transferring fraction of a_j to a_i
            for f in [0.1, 0.3, 0.5]:
                delta = a_j * f
                new_a = a.copy()
                new_a[j] = a_j - delta
                new_a[i] = a_i + delta
                new_a = normalize(new_a)
                sc = compute_score(new_a)
                if sc < best_score - 1e-12:
                    a = new_a
                    best_score = sc
                    best_a = a.copy()
                    break
    
    return best_a, best_score

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def compute_score_normalized(a):
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max())


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fast(a, peaks, weights):
    n = len(a)
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        k = peaks[idx]
        w = weights[idx]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idxs = np.arange(js, je + 1)
        sg[idxs] += 2.0 * w * a[k - idxs]
    return sg


def subgradient_from_conv(a, conv, smooth, top_k):
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    pv_max = pv.max()
    exp_vals = np.exp((pv - pv_max) / smooth)
    w_sum = exp_vals.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        weights = exp_vals / w_sum
    sg = compute_subgradient_fast(a, peaks.astype(int), weights)
    return sg


def nesterov_optimize(a, time_limit, lr0=0.35, smooth_init=0.008,
                      smooth_decay=0.9997, top_k=40, reconvene_every=20,
                      verbose=False):
    n = len(a)
    a = normalize(a.copy())
    conv = fast_convolve(a)
    best_score = compute_score_normalized(a)
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0
    t_k = 1.0
    y = a.copy()
    lr = lr0
    smooth = smooth_init
    min_smooth = 0.0005

    while time.monotonic() - t0 < time_limit:
        step += 1

        if step == 1 or step % reconvene_every == 0:
            conv = fast_convolve(y)

        sg = subgradient_from_conv(y, conv, smooth, top_k)

        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new

        x_new = project_simplex(y - lr * sg)
        y = x_new + beta * (x_new - a)
        y = project_simplex(y)
        a = x_new
        t_k = t_new

        smooth = max(min_smooth, smooth * smooth_decay)

        if step % 100 == 0:

            if sc < best_score:
                best_score = sc

def dual_construct(n, lam_weights, epsilon=1e-8):
    N = 4 * n
    lam_padded = np.zeros(N)
    lam_padded[:len(lam_weights)] = lam_weights
    lam_fft = np.fft.fft(lam_padded)
    abs_lam = np.abs(lam_fft)
    phase = np.angle(lam_fft)
    lam_reg = (abs_lam + epsilon) * np.exp(1j * phase)
    f_fft = 1.0 / lam_reg
    f = np.fft.ifft(f_fft).real[:n]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s


def gen_dual_starts(n, num_samples=5):

        elif trial == 2:

def gen_power_law(n, beta, gamma):
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + gamma * i) ** (-beta)

def gen_exponential(n, alpha, asym):
    i = np.arange(n, dtype=np.float64)
    t = i / max(1, n - 1)
    a = np.exp(-alpha * t + asym * t * (1.0 - t))

    a = (1.0 + 0.5 * np.abs(i - c) / c) ** (-beta)

                    best_score = best_sc
                    best_a = a.copy()
                    improved = True

        if verbose and pass_count % 5 == 0:
            print(f"      CD pass {pass_count}: score={best_score:.10f}")
        if not improved and pass_count > 2:
            break
        if pass_count > 40:
            break

    return best_a, best_score


def pair_swap_refine(a, time_limit, verbose=False):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    rng = np.random.default_rng(555)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        for _ in range(300):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.integers(0, n, 2)
            if i == j or a[j] < 1e-6:
                continue
            for f in [0.1, 0.25, 0.5]:
                delta = a[j] * f
                new_a = a.copy()
                new_a[j] -= delta
                new_a[i] += delta
                new_a = normalize(new_a)
                sc = compute_score_normalized(new_a)
                if sc < best_score - 1e-13:
                    a = new_a
                    best_score = sc
                    best_a = a.copy()
                    break

    return best_a, best_score


def full_optimize(a, time_limit, verbose=False):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()

    # Stage 1: Nesterov
    t1 = min(time_limit * 0.35, 150)
    if time_limit - (time.monotonic() - t0) > t1 + 20:
        a1, sc1 = nesterov_optimize(a, time_limit=t1, lr0=0.35,
                                     smooth_init=0.008, top_k=max(20, n // 12),
                                     verbose=verbose)
        if sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            a = a1.copy()
        if verbose:
            print(f"        After Nesterov: {best_score:.10f}")

    # Stage 2: Second Nesterov with different params
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.25, 100)
    if rem > t2 + 15:
        a2, sc2 = nesterov_optimize(a, time_limit=t2, lr0=0.25,
                                     smooth_init=0.003, top_k=max(30, n // 8),
                                     verbose=verbose)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            a = a2.copy()
        if verbose:
            print(f"        After Nesterov2: {best_score:.10f}")

    # Stage 3: Coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.2, 80)
    if rem > t3 + 10:
        a3, sc3 = coord_descent_refine(a, time_limit=t3, verbose=verbose)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
            a = a3.copy()
        if verbose:
            print(f"        After CD: {best_score:.10f}")

    # Stage 4: Pair swap
    rem = time_limit - (time.monotonic() - t0)
    t4 = min(rem * 0.2, 60)
    if rem > t4 + 10:
        a4, sc4 = pair_swap_refine(a, time_limit=t4, verbose=verbose)
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
        if verbose:
            print(f"        After PS: {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Generate diverse starting points
    print("=== Generating starting points ===")
    starts = []

    # Parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent", len(parent)))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    n_list = [512, 768, 1024, 1280]

    for n in n_list:
        # Power law
        for beta in [1.5, 2.0, 2.5, 3.0, 4.0]:
            for gamma in [0.3, 0.7, 1.0, 1.5, 2.0]:
                starts.append((gen_power_law(n, beta, gamma), f"pl_b{beta}_g{gamma}_n{n}", n))

        # Symmetric power
        for beta in [2.0, 3.0, 4.0, 5.0]:
            starts.append((gen_symmetric_power(n, beta), f"sp_b{beta}_n{n}", n))

        # Exponential with asymmetry
        for alpha in [3.0, 5.0, 7.0, 10.0]:
            for asym in [-3.0, -1.0, 1.0, 3.0]:
                starts.append((gen_exponential(n, alpha, asym), f"ex_a{alpha}_s{asym}_n{n}", n))

        # Dual construction
        dual_starts = gen_dual_starts(n, num_samples=4)
        for di, ds in enumerate(dual_starts):
            starts.append((ds, f"dual_{di}_n{n}", n))

    # Evaluate all starts
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label, n in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label, n))
    start_scores.sort(key=lambda x: x[0])

    for i, (sc, a, label, n) in enumerate(start_scores[:8]):
        print(f"    Start #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # Phase 1: Optimize top diverse starts with Nesterov
    print(f"\n=== Phase 1: Nesterov on diverse starts (best={best_score:.8f}) ===")
    selected = []
    seen_types = set()
    for sc, a, label, n in start_scores:
        if len(selected) >= 6:
            break
        base = label.rsplit('_n', 1)[0] if '_n' in label else label
        key = base.split('_')[0] + str(n)
        if key not in seen_types:
            selected.append((a, label, sc))
            seen_types.add(key)

    # Ensure parent included
    parent_added = False
    for sc, a, label, n in start_scores:
        if label == "parent":
            selected.append((a, "parent", sc))
            parent_added = True
            break

    t_per = min((deadline - time.monotonic() - 300) / max(1, len(selected)), 220)

    for a_init, label, init_sc in selected:
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        t_alloc = min(t_per, rem)
        print(f"  Full opt [{label}] (init={init_sc:.8f}, t={t_alloc:.0f}s):")
        a_opt, sc_opt = full_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_full")

    # Phase 2: Simulated annealing on best
    print(f"\n=== Phase 2: Simulated annealing (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_sa = min(rem * 0.4, 120)
            print(f"  SA: t={t_sa:.0f}s")
            a_sa, sc_sa = simulated_annealing(best_a, time_limit=t_sa, t_start=0.005,
                                              t_end=0.0005, verbose=True)
            try_seq(a_sa, "sa")
            # Re-optimize after SA
            rem = deadline - time.monotonic() - 150
            if rem > 30:
                a_so, sc_so = full_optimize(a_sa, time_limit=min(rem, 100), verbose=True)
                try_seq(a_so, "sa_full")

    # Phase 3: Final refinement on best
    print(f"\n=== Phase 3: Final refinement (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_final = min(rem * 0.5, 120)
            print(f"  Final Nesterov: t={t_final:.0f}s")
            a_f, sc_f = nesterov_optimize(best_a, time_limit=t_final, lr0=0.2,
                                           smooth_init=0.002, top_k=max(40, len(best_a) // 6),
                                           verbose=True)
            try_seq(a_f, "final_ng")

            rem = deadline - time.monotonic() - 30
            if rem > 15:
                a_cd, sc_cd = coord_descent_refine(best_a, time_limit=min(rem, 60))
                try_seq(a_cd, "final_cd")

            rem = deadline - time.monotonic() - 20
            if rem > 10:
                a_ps, sc_ps = pair_swap_refine(best_a, time_limit=min(rem, 30))
                try_seq(a_ps, "final_ps")

    # Phase 4: Dimension exploration
    if best_a is not None and best_score < 1.506:
        print(f"\n=== Phase 4: Dimension exploration (best={best_score:.8f}) ===")
        best_n = len(best_a)
        target_ns = [best_n - 128, best_n + 128, best_n + 256, best_n // 2 + 128]
        for tn in target_ns:
            if tn < 200 or tn > 2500 or tn == best_n:
                continue
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            old_x = np.linspace(0, 1, best_n)
            new_x = np.linspace(0, 1, tn)
            a_interp = np.interp(new_x, old_x, best_a)
            a_interp = normalize(a_interp)
            t_alloc = min(rem * 0.4, 80)
            a_opt, sc = full_optimize(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"dim_{tn}")

    # Final verification
    if best_a is None:
        best_a = gen_power_law(768, 2.0, 1.0)
        best_score = compute_score(best_a)

    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
