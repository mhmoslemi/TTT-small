    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def peak_directed_search(a, time_limit, verbose=False):
    """
    Local search that specifically targets the convolution peaks.
    Identifies which coordinates contribute most to the peaks and reduces them.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    conv = fast_convolve(a)
    total_sum = a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * conv.max() / (total_sum * total_sum)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        total_sum = a.sum()
        current_max = conv.max()
        
        # Find peaks (positions in conv that are near the max)
        threshold = current_max * 0.95
        peak_positions = np.where(conv >= threshold)[0]
        
        # Compute "pressure" on each coordinate
        # Pressure on coordinate i = sum over peaks k of conv[k] * a[k-i] (contribution of a_i to peak k)
        pressure = np.zeros(n)
        for k in peak_positions:
            k_start = max(0, k - n + 1)
            k_end = min(n - 1, k)
            if k_start > k_end:
                continue
            weighted = conv[k] * a[k_start:k_end+1][::-1]
            pressure[k_start:k_end+1] += weighted
        
        # Reduce high-pressure coordinates
        top_pressure = np.argsort(pressure)[-n//4:]  # top 25%
        
        for j in top_pressure:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            
            # Try reducing
            for f in [0.5, 0.7, 0.85, 0.9]:
                nv = old_val * f
                delta = nv - old_val
                new_sum = total_sum + delta
                if new_sum < 1e-15:
                    continue
                
                k_s = j
                k_e = min(j + n, 2 * n - 1)
                oi = np.arange(k_s, k_e) - j
                new_seg = conv[k_s:k_e].copy() + 2.0 * delta * a[oi]
                if k_s <= 2*j < k_e:
                    dp = 2*j - k_s
                    new_seg[dp] = conv[2*j] + 2.0 * old_val * delta + delta * delta
                
                seg_max = float(new_seg.max())

                    conv[k_s:k_e] = new_seg
                    total_sum = new_sum
                    best_score = score
                    best_a = a.copy()
                    break  # Move to next coordinate
    
    return best_a, best_score

```python
def lp_best_response(f, M):
    """
    Find g >= 0 that maximizes sum(g) subject to (f*g)_k <= M for all k.
    Returns the optimal g.
    """
    from scipy.optimize import linprog
    n = len(f)
    
    # Build the convolution matrix C where C[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # (f*g)_k = sum_i C[k,i] * g[i]
    C = np.zeros((2*n-1, n))
    for i in range(n):
        # g[i] contributes to (f*g)_k for k = i, i+1, ..., i+n-1
        # The contribution is f[k-i]
        for k in range(i, min(i+n, 2*n-1)):
            C[k, i] = f[k-i]
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: C @ g <= M (component-wise)
    A_ub = C
    b_ub = np.full(2*n-1, M)
    
    # Variable bounds: 0 <= g_i
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return f.copy()  # fallback
```

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using:
1. Fast O(n) coordinate descent with incremental convolution updates
2. LP-based best response (Sinkhorn-like iteration for convolutions)
3. Peak-directed multiplicative updates
4. Multi-start with diverse initializations and length exploration
Key innovation: incremental convolution updates avoid O(n log n) FFT per step,
enabling thousands of coordinate descent passes within the time budget.

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_coord_descent(a, time_limit, verbose=False):
    """
    Fast coordinate descent using O(n) incremental convolution updates.
    No renormalization during optimization (score is scale-invariant).
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    conv = fast_convolve(a)
    total_sum = float(a.sum())
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max()) / (total_sum * total_sum)
    t0 = time.monotonic()
    pass_count = 0

    # Precompute affected ranges for each coordinate
    k_starts = np.arange(n)
    k_ends = np.minimum(np.arange(n) + n, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]

            # Generate candidate values
            if old_val > 1e-12:
                multipliers = np.array([0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0])
                candidates = old_val * multipliers
                candidates = np.append(candidates, 0.0)
            else:
                candidates = np.array([total_sum * f for f in [0.001, 0.005, 0.01, 0.05]])

            best_cand_score = best_score
            best_cand_nv = None
            best_cand_seg = None
            best_cand_sum = None

            ks = int(k_starts[j])
            ke = int(k_ends[j])
            other_idx = np.arange(ks, ke) - j
            current_conv_max = float(conv.max())

            for nv in candidates:
                nv = float(max(0.0, nv))
                delta = nv - old_val
                if abs(delta) < 1e-18:
                    continue
                new_sum = total_sum + delta
                if new_sum < 1e-15:
                    continue

                # Compute new conv segment
                new_seg = conv[ks:ke].copy() + 2.0 * delta * a[other_idx]
                # Fix diagonal
                diag_k = 2 * j
                if ks <= diag_k < ke:
                    dp = diag_k - ks
                    new_seg[dp] = conv[diag_k] + 2.0 * old_val * delta + delta * delta

                seg_max = float(new_seg.max())

                # Total max: if seg_max >= current global max, total = seg_max
                # Otherwise, the global max might be in the unchanged part = current_conv_max
                # (conservative upper bound)
                if seg_max >= current_conv_max:
                    total_max = seg_max
                else:
                    total_max = current_conv_max

                score = 2.0 * n * total_max / (new_sum * new_sum)

                if score < best_cand_score - 1e-14:
                    best_cand_score = score
                    best_cand_nv = nv
                    best_cand_seg = new_seg
                    best_cand_sum = new_sum

            if best_cand_nv is not None:
                a[j] = best_cand_nv
                conv[ks:ke] = best_cand_seg
                total_sum = best_cand_sum
                if best_cand_score < best_score:
                    best_score = best_cand_score
                    best_a = a.copy()
                    improved = True

        if verbose and pass_count % 50 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

        if not improved and pass_count > 2:
            break

    return best_a, best_score

def lp_best_response(f, M, max_time=10.0):
    """
    Find g >= 0 maximizing sum(g) subject to (f*g)_k <= M for all k.
    Uses scipy linprog with HiGHS solver.
    """
    n = len(f)
    
    # Build convolution matrix C where (C @ g)[k] = (f*g)[k]
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        end = min(i + n, 2 * n - 1)
        C[i:end, i] = f[:end - i]

    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)

    # Constraints: C @ g <= M
    A_ub = C
    b_ub = np.full(2 * n - 1, M)

    # Bounds: g_i >= 0
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max_time})
        if res.success:
            return res.x
    except Exception:
        pass
    return f.copy()


def sinkhorn_convolution(a, n_iters=20, verbose=False):
    """
    Alternating LP best-response iteration (Sinkhorn-like for convolutions).
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        conv = fast_convolve(a)
        M = float(conv.max())
        
        # Find best response g to a
        g = lp_best_response(a, M, max_time=5.0)
        g = np.maximum(g, 0.0)
        if g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        # Blend
        t_blend = 0.5
        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        if verbose:
            print(f"    Sinkhorn it {it}: score={best_score:.10f}")
        
        a = a_new
    
    return best_a, best_score


def peak_directed_update(a, time_limit, verbose=False):

    iteration = 0

        iteration += 1
        conv = fast_convolve(a)
        total_sum = float(a.sum())
        current_max = float(conv.max())
        
        # Find peak positions (top 20% of max)
        threshold = current_max * 0.9
        peak_positions = np.where(conv >= threshold)[0]
        
        if len(peak_positions) == 0:
            break
        
        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        for k in peak_positions:
            k_lo = max(0, k - n + 1)
            k_hi = min(n - 1, k)
            if k_lo > k_hi:
                continue
            # a[i] contributes to conv[k] via a[i]*a[k-i]
            # pressure[i] += conv[k] * a[k-i] for i in [k_lo, k_hi]
            idx = np.arange(k_lo, k_hi + 1)
            weight = a[k - idx]
            pressure[k_lo:k_hi + 1] += conv[k] * weight
        
        # Target high-pressure coordinates for reduction
        if pressure.max() < 1e-15:
            break
        
        n_target = max(1, n // 8)
        top_pressure_idx = np.argsort(pressure)[-n_target:]

            old_val = a[j]
            if old_val < 1e-14:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return np.maximum(f_interp, 0.0)


def geometric_sequence(n, r):
    """Geometric sequence."""
    a = r ** np.arange(n)
    return a  # don't normalize, score is scale-invariant


def cosine_sequence(n):
    """Sinusoidal shaped sequence."""
    x = np.linspace(0, np.pi, n)
    return np.sin(x) ** 2


def gaussian_sequence(n, sigma_frac=0.3):
    """Gaussian centered."""
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    return np.exp(-x ** 2 / (2 * sigma ** 2))


def triangular_sequence(n):
    """Triangular (concave) sequence."""
    x = np.linspace(0, 1, n)
    return 1.0 - np.abs(2.0 * x - 1.0)


def concave_poly_sequence(n, p=2.0):
    """Concave polynomial: (1 - (2x-1)^p) shape."""
    x = np.linspace(-1, 1, n)
    return np.maximum(1.0 - np.abs(x) ** p, 0.0)


def random_smooth(n, rng, k=20):
    """Random smoothed sequence."""
    a = rng.random(n)
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    return np.maximum(a, 0.0)

            rem = deadline - time.monotonic() - 40
            if rem > 10:
                a_opt2, sc2 = fine_coord_descent(a_opt, time_limit=min(rem, 20.0))
                try_seq(a_opt2, f"restart_{trial}_fine")

        # Final fine CD on absolute best
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            print(f"  Final fine CD: t={min(rem, 120.0):.0f}s")
            a_final, sc = fine_coord_descent(best_a, time_limit=min(rem, 120.0))
            try_seq(a_final, "final_cd")

        # One more peak-directed
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_pk, sc = peak_directed_update(best_a, time_limit=min(rem, 60.0))
            try_seq(a_pk, "final_peak")

        # Final micro-CD
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_m, sc = fine_coord_descent(best_a, time_limit=min(rem, 50.0))
            try_seq(a_m, "final_micro")

    # === Fallback ===
    if best_a is None:
        best_a = concave_poly_sequence(768, 2.0)
        best_score = compute_score(best_a)
        print(f"  Fallback: {best_score:.10f}")

    # Verify with actual evaluate_sequence
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_a]))
        if np.isfinite(final_score):
            best_score = final_score
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.10f}, n={len(best_a)}")
    return [float(x) for x in best_a]
