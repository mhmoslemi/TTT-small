Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-based Frank-Wolfe

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build sparse LP constraint matrix for (f*g)_k <= M."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        r = list(range(j, j + n))
        c = [j] * n
        v = list(f)
        rows.extend(r)
        cols.extend(c)
        vals.extend(v)
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(f, M, n, time_limit=60):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': time_limit, 'presolve': True}
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options=options)
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def peak_target_update(f, n_peaks=10, eta=0.02):
    """Multiplicative update targeting the highest convolution peaks."""
    conv = fast_convolve(f)
    n = len(f)
    k_actual = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    peak_vals = conv[peaks]
    peak_vals = peak_vals - conv.mean()
    peak_vals = np.maximum(peak_vals, 0)
    w = peak_vals / peak_vals.sum() if peak_vals.sum() > 1e-15 else np.ones(len(peaks)) / len(peaks)

    # Compute pressure on each position
    pressure = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        pressure[idx] += w[i] * f[k - idx]

    # Multiplicative update
    f_new = f * np.exp(-eta * pressure)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    f_new /= s
    return f_new


def peak_target_phase(f0, time_limit, n_peaks=15, eta_start=0.05):
    """Iterative peak-targeting with adaptive step size."""

def frequency_init(n, seed=0):
    """Initialize using frequency-domain thinking: flat PSD gives uniform,
    but we try a raised cosine in frequency to get smooth time-domain."""
    rng = np.random.default_rng(seed)
    # Try a Hann window
    t = np.arange(n)
    f = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def triangle_init(n):
    """Triangular (Bartlett) window initialization."""
    t = np.arange(n)
    mid = n / 2.0
    f = 1.0 - np.abs(t - mid) / mid
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian window initialization."""
    t = np.arange(n)
    mid = n / 2.0
    sigma = sigma_frac * n
    f = np.exp(-0.5 * ((t - mid) / sigma) ** 2)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent_fast(f0, time_limit):
    """Fast coordinate descent with multiplicative updates on single coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_obj = best_obj

        for j in range(0, n, max(1, n // 200)):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()

            for mult in [0.5, 0.7, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj
