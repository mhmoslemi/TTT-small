def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u - (cssv - target_sum) / rho) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho_last = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_last] - target_sum) / (rho_last + 1)
    return np.maximum(v - theta, 0.0)


def softmax_alpha(x):
    """Convert x to probability vector via softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def subgrad_optimize(a0, time_limit, verbose=False):
    """Hard-max subgradient optimization with momentum and adaptive restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    tau = 30.0
    lr = 0.06
    momentum = 0.80
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive parameters
        if no_improve > 500:
            lr = max(0.005, lr * 0.95)
            momentum = min(0.92, momentum + 0.003)
        elif no_improve > 200:
            tau = max(10.0, tau * 0.99)
            lr = min(0.10, lr * 1.003)
        else:
            tau = min(80.0, tau * 1.002)

        conv = fast_convolve(a)

        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = (pv - pv.max()) / tau
        w_peaks = np.exp(scaled)
        w_peaks /= w_peaks.sum()

        for i in range(len(peaks)):
            if w_peaks[i] < 1e-10:

            js = max(0, k - n + 1)
            je = min(n - 1, k)

            grad[idx] += w_peaks[i] * 2.0 * a[k - idx]

        a = a_new

            no_improve = 0

    x = np.log(np.maximum(a, 1e-15))

    best_a = a.copy()
    best_true_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    taus = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.2, 1.0]

    for ti, tau in enumerate(taus):
        if time.monotonic() - t0 >= time_limit:
            break
        remain = time_limit - (time.monotonic() - t0)
        if remain < 5:
            break

        max_iter = min(300, max(30, int(remain / 0.5)))

        def obj_fn(x):
            a_s = softmax_alpha(x)
            conv = fast_convolve(a_s)
            scaled = conv / tau
            sm = scaled.max()
            return tau * (sm + np.log(np.exp(scaled - sm).sum()))

        def grad_fn(x):
            a_s = softmax_alpha(x)
            conv = fast_convolve(a_s)
            scaled = conv / tau
            sm = scaled.max()
            exp_s = np.exp(scaled - sm)
            ms = exp_s.sum()
            if ms < 1e-300:
                w = np.zeros(len(conv))
                w[scaled.argmax()] = 1.0
            else:
                w = exp_s / ms
            C = np.dot(w, conv)
            S = compute_S_fast(a_s, w)
            return 2.0 * a_s * (S - C)

        try:
            result = minimize(obj_fn, x, jac=grad_fn, method='L-BFGS',
                              options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                       'gtol': 1e-8})
            x = result.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at tau={tau}: {e}")
            break

        a_cur = softmax_alpha(x)
        true_obj = fast_convolve(a_cur).max()

        if true_obj < best_true_obj - 1e-14:
            best_true_obj = true_obj
            best_a = a_cur.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS tau={tau:5.1f}: true_obj={best_true_obj:.10f}, msg={result.message[:40]}, t={elapsed:.0f}s")

    return best_a, best_true_obj


def iterative_lbfqsg(a0, time_limit, n_restarts=8, verbose=False):
    """Iterated L-BFGS with random perturbations to escape local minima."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for r in range(n_restarts):
        remain = time_limit - (time.monotonic() - t0)
        if remain < 20:
            break
        time_per = remain / (n_restarts - r)

        if r == 0:
            x = np.log(np.maximum(a, 1e-15))
        elif r == 1:
            noise = rng.standard_normal(n) * 0.008
            a_pert = project_simplex(a + noise, 1.0)
            x = np.log(np.maximum(a_pert, 1e-15))
        elif r == 2:
            # Reverse-mirror perturbation
            a_mir = 0.7 * a + 0.3 * a[::-1]
            a_mir = project_simplex(a_mir, 1.0)
            x = np.log(np.maximum(a_mir, 1e-15))
        elif r == 3:
            # Frequency-domain perturbation
            A = np.fft.rfft(a)
            phases = rng.uniform(-0.1, 0.1, len(A))
            A_pert = A * np.exp(1j * phases)
            a_pert = np.fft.irfft(A_pert, n * 2)[:n]
            a_pert = project_simplex(np.maximum(a_pert, 0), 1.0)
            x = np.log(np.maximum(a_pert, 1e-15))
        elif r == 4:
            noise = rng.standard_normal(n) * 0.015
            a_pert = project_simplex(a + noise, 1.0)
            x = np.log(np.maximum(a_pert, 1e-15))
        elif r == 5:
            # Blend with uniform
            a_blend = 0.85 * a + 0.15 * np.ones(n) / n
            x = np.log(np.maximum(a_blend, 1e-15))
        elif r == 6:
            noise = rng.standard_normal(n) * 0.025
            a_pert = project_simplex(a + noise, 1.0)
            x = np.log(np.maximum(a_pert, 1e-15))
        else:
            # Larger perturbation
            a_new = rng.exponential(1.0, n)
            a_blend = 0.5 * a + 0.5 * (a_new / a_new.sum())
            x = np.log(np.maximum(a_blend, 1e-15))

        a_res, obj_res = lbfqsg_continuation(softmax_alpha(x), time_limit=time_per, verbose=verbose and r < 3)
        if obj_res < best_obj - 1e-14:
            best_obj = obj_res
            best_a = a_res.copy()
            if verbose:
                print(f"    Restart {r}: improved to {best_obj:.10f}")
        # Update a to best found for next restart
        a = best_a.copy()

    return best_a, best_obj


def coordinate_descent_v2(a0, time_limit, verbose=False):
    """Improved coordinate descent with broader search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-12:
                continue

            old_val = a[j]
            old_remaining = 1.0 - old_val
            if old_remaining < 1e-10:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            # Wider range of trial values
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.1]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / old_remaining
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate():

        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score

    parent = None
    try:

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = flat_spectrum_init(768, rng)
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Generated parent: n={len(parent)}, score={p_score:.6f}", flush=True)

    n_parent = len(parent)
