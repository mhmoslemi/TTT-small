Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Nesterov-Adam in softmax parameterization with annealed log-sum-exp (primary)
2. L-BFGS with aggressive beta annealing (secondary)
3. LP-based Frank-Wolfe (polishing)
4. Coordinate descent and subgradient (final polish)
5. Multi-length exploration with interpolation
Key innovations: NAdam with lookahead, geometric restart scheduling,
and alternating gradient/LP phases for better exploration of non-convex landscape.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_grad_a(a, w):
    """Compute gradient of LSE objective w.r.t. a using cross-correlation via FFT."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    return 2.0 * R


def nadam_optimize(f0, time_limit, verbose=False):
    """
    Nesterov-Adam optimization in softmax parameterization with annealed beta,
    lookahead mechanism, and geometric restart scheduling.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    eps = 1e-30
    t = np.log(f + eps)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps_adam = 0.85, 0.999, 1e-8
    step = 0
    lr = 0.04

    # Lookahead
    slow_t = t.copy()
    la_k = 0.5
    la_period = 10

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    best_t = t.copy()

    t0 = time.monotonic()
    no_improve = 0
    restart_count = 0
    max_no_improve_base = 150

    while time.monotonic() - t0 < time_limit:
        step += 1
        a = softmax_fn(t)
        conv = fast_convolve(a)
        conv_max = conv.max()

        # Annealed beta: start low, increase aggressively
        beta = min(8.0 * (1.03 ** step), 800.0)

        # Log-sum-exp weights
        shifted = beta * (conv - conv_max)
        # Clamp for numerical stability
        shifted = np.clip(shifted, -500, 500)
        exp_shifted = np.exp(shifted)
        w_sum = exp_shifted.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_shifted / w_sum

        grad_a = compute_grad_a(a, w)

        # Convert to t-space gradient
        dot = np.dot(a, grad_a)
        grad_t = a * (grad_a - dot)

        # NAdam update
        m = beta1 * m + (1.0 - beta1) * grad_t
        v = beta2 * v + (1.0 - beta2) * grad_t ** 2
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        # Nesterov lookahead on momentum
        m_bar = beta1 * m_hat + m_hat / (1.0 - beta1 ** step)
        t = t - lr * m_bar / (np.sqrt(v_hat) + eps_adam)

        # Lookahead: periodically blend fast and slow weights
        if step % la_period == 0:
            slow_t = slow_t + la_k * (t - slow_t)
            t = slow_t

        a_new = softmax_fn(t)
        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = a_new.copy()
            best_t = t.copy()
            no_improve = 0
            if verbose and step % 200 == 0:
                print(f"    NAdam iter {step}: obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1

        # Learning rate decay
        lr *= 0.9997
        lr = max(lr, 0.001)

        # Geometric restart with increasing perturbation
        threshold = max_no_improve_base + 50 * restart_count
        if no_improve > threshold:
            restart_count += 1
            scale = 0.002 * (1.3 ** min(restart_count, 8))
            rng = np.random.default_rng(step * 7919 + restart_count * 104729)
            best_f_pert = best_f + rng.standard_normal(n) * scale
            best_f_pert = project_simplex(best_f_pert)
            t = np.log(best_f_pert + eps)
            slow_t = t.copy()
            m, v = np.zeros(n), np.zeros(n)
            step = 0
            lr = 0.02 + 0.01 * min(restart_count, 5)
            no_improve = 0
            if verbose and restart_count % 3 == 0:
                print(f"    NAdam restart {restart_count}: scale={scale:.5f}")

        # Time check
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

    return best_f, best_obj


def lbfqg_optimize(f0, time_limit, verbose=False):
    """L-BFGS on smooth log-sum-exp with aggressive beta schedule."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    eps = 1e-15
    t = np.log(f + eps)

    betas = [15.0, 30.0, 60.0, 120.0, 250.0, 500.0, 1000.0]
    t0_time = time.monotonic()
    total_time = time_limit

    for bi, beta in enumerate(betas):
        remaining = total_time - (time.monotonic() - t0_time)
        if remaining < 15:
            break
        # Allocate more time to later (higher beta) phases
        frac = 0.3 + 0.7 * (bi / len(betas))
        sub_time = remaining * (frac / (1.0 + 0.7 * bi / len(betas)))
        sub_time = max(sub_time, 10)

        try:
            res = minimize(
                _smooth_obj_grad_impl, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 500, 'gtol': 1e-12, 'maxls': 80,
                         'maxfun': 5000, 'ftol': 1e-15}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={beta} failed: {e}")

    return best_f, best_obj


def _smooth_obj_grad_impl(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = np.clip(beta * (conv - conv_max), -500, 500)
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_shifted / w_sum

    obj = conv_max + np.log(max(w_sum, 1e-300)) / beta

    grad_a = compute_grad_a(a, w)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 20.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.1, 10.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 10.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 60, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def peak_directed_descent(f0, time_limit):
    """
    Coordinate descent that targets the peak of the autocorrelation.
    For each peak position k*, redistributes mass between a[j] and a[j+k*]
    to reduce the peak while maintaining normalization.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_full_pass = 0

    while time.monotonic() - t0 < time_limit and no_full_pass < 50:
        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        improved = False

        if peak_k == 0:
            # Peak at lag 0: flatten large values
            threshold = f.mean() * 1.5
            large_idx = np.where(f > threshold)[0]
            for j in large_idx[:30]:
                # Try reducing f[j] and distributing to small elements
                small_idx = np.where(f < f.mean() * 0.5)[0]
                if len(small_idx) == 0:
                    continue
                target = small_idx[np.argmin(f[small_idx])]
                delta = f[j] * 0.1
                f_trial = f.copy()
                f_trial[j] -= delta
                f_trial[target] += delta
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
        else:
            # Peak at lag k*: reduce a[j]*a[j+k*] for all j
            k = peak_k
            for j in range(0, n - k, max(1, (n - k) // 50)):
                # Try moving mass from a[j] to a[j+k] or vice versa
                for frac in [0.05, 0.1, 0.15, 0.2]:
                    f_trial = f.copy()
                    # Move mass from larger to smaller
                    if f[j] > f[j + k]:
                        delta = f[j] * frac
                        f_trial[j] -= delta
                        f_trial[j + k] += delta
                    else:
                        delta = f[j + k] * frac
                        f_trial[j + k] -= delta
                        f_trial[j] += delta
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = f_trial / f_trial.sum()
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-14:
                        f = f_trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_full_pass += 1
            # Try a small random perturbation
            if no_full_pass > 3:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_full_pass = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.93, top_k=80, temp=0.0001):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.005

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0005)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use nearest for very small old_n, linear otherwise
    if old_n >= 4:
        f_interp = np.interp(new_x, old_x, f, kind='cubic')
    else:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: NAdam from parent and variants
    Phase 2: L-BFGS refinement
    Phase 3: LP-FW polishing
    Phase 4: NAdam fine-tuning
    Phase 5: Peak-directed descent
    Phase 6: Multi-length exploration
    Phase 7: Final polishing (SG + CD)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform 512")

    # === Phase 1: NAdam from parent (350s) ===
    print(f"\n=== Phase 1: NAdam from parent (n={len(parent)}) ===")
    rem = deadline - time.monotonic() - 800
    t1 = min(350, max(rem, 50))
    a_opt, obj1 = nadam_optimize(parent, time_limit=t1, verbose=True)
    try_seq(a_opt, "nadam_parent")

    # NAdam from reversed parent
    if best_seq is not None:
        # Also try small perturbation
        rem = deadline - time.monotonic() - 750
        if rem > 30:
            noise = rng.standard_normal(len(parent)) * 0.005
            f_pert = project_simplex(best_seq + noise)
            t1b = min(100, rem)
            a_opt, _ = nadam_optimize(f_pert, time_limit=t1b)
            try_seq(a_opt, "nadam_perturbed")

    # === Phase 2: L-BFGS refinement (350s) ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 700
    t2 = min(350, max(rem, 50))
    a_opt, obj2 = lbfqg_optimize(base, time_limit=t2, verbose=True)
    try_seq(a_opt, "lbfqg_main")

    # L-BFGS from perturbed
    rem = deadline - time.monotonic() - 600
    if rem > 20:
        noise = rng.standard_normal(len(base)) * 0.003
        f_pert = project_simplex(base + noise)
        t2b = min(100, rem)
        a_opt, _ = lbfqg_optimize(f_pert, time_limit=t2b)
        try_seq(a_opt, "lbfqg_pert")

    # === Phase 3: LP-FW polishing (250s) ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 500
    t3 = min(250, max(rem, 30))
    a_opt, obj3 = lp_fw_optimize(base, time_limit=t3, verbose=True)
    try_seq(a_opt, "lpfw_main")

    # === Phase 4: NAdam fine-tuning (200s) ===
    print(f"\n=== Phase 4: NAdam fine-tuning (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 400
    t4 = min(200, max(rem, 30))
    a_opt, _ = nadam_optimize(base, time_limit=t4, verbose=True)
    try_seq(a_opt, "nadam_fine")

    # === Phase 5: Peak-directed descent (150s) ===
    print(f"\n=== Phase 5: Peak-directed (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 300
    t5 = min(150, max(rem, 20))
    a_opt, _ = peak_directed_descent(base, time_limit=t5)
    try_seq(a_opt, "peak_directed")

    # L-BFGS after peak-directed
    rem = deadline - time.monotonic() - 250
    if rem > 20:
        a_opt, _ = lbfqg_optimize(a_opt, time_limit=min(60, rem))
        try_seq(a_opt, "lbfqg_after_peak")

    # === Phase 6: Multi-length exploration (200s) ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    # Try lengths near the parent
    target_lengths = []
    for dl in [-64, -32, 32, 64, 96, 128]:
        tl = n_base + dl
        if 128 <= tl <= 2048:
            target_lengths.append(tl)
    # Remove duplicates
    target_lengths = sorted(set(target_lengths))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.5, 60)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = nadam_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}")

        # Quick L-BFGS polish
        rem = deadline - time.monotonic() - 200
        if rem > 15:
            a_opt2, _ = lbfqg_optimize(a_opt, time_limit=min(30, rem))
            try_seq(a_opt2, f"len_{target_n}_lbfqg")

    # === Phase 7: Final polishing (remaining time) ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 7a: SG polish
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_alloc = min(rem * 0.5, 80)
            a_opt, _ = sg_polish(best_seq, time_limit=t_alloc,
                                 lr=0.001, mom=0.94,
                                 top_k=max(30, n_best // 6),
                                 temp=0.00005)
            try_seq(a_opt, "final_sg")

        # 7b: L-BFGS final
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem * 0.7, 60)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lbfqg")

        # 7c: NAdam final with very small lr
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.7, 40)
            a_opt, _ = nadam_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_nadam")

        # 7d: Peak-directed final
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = peak_directed_descent(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_peak")

        # 7e: One more L-BFGS from absolute best
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "absolute_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
