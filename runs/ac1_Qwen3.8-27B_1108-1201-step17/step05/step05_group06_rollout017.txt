- There exist multipliers μ_i >= 0 for a_i >= 0

∂(a*a)[k]/∂a_i = a[k-i] (with a[j] = 0 for j outside [0, n-1])

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B with log-sum-exp smoothing, subgradient methods,
diverse initializations, symmetrization, and length scaling.
"""

import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_fast_len(size)
    A = fft(a, N)
    return ifft(A * A, N).real[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_smooth_objective(a, T):
    """Compute log-sum-exp of conv(a,a)/T."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    return T * (scaled.max() + np.log(e.sum()))


def lse_smooth_grad(a, T):
    """Compute gradient of log-sum-exp of conv(a,a)/T."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    
    N = next_fast_len(3 * n - 1)
    W_fft = fft(w, N)
    A_fft = fft(a, N)
    C = ifft(W_fft * np.conj(A_fft), N).real[:n]
    grad = (2.0 / T) * C
    
    val = T * (scaled.max() + np.log(e.sum()))
    return grad, val


def lsbfgs_optimize(f0, time_limit, T_start=1.0, T_end=0.01, verbose=False):
    """Optimize using L-BFGS-B with log-sum-exp smoothing and annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    
    # Use a schedule of temperatures
    # For each temperature, run L-BFGS-B
    n_phases = 5
    T_schedule = np.geomspace(T_start, T_end, n_phases)
    
    for phase, T in enumerate(T_schedule):
        phase_time = time_limit * (0.4 ** (phase / n_phases))  # decreasing time
        phase_deadline = time.monotonic() + phase_time
        
        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-15:
                return 1e10, np.zeros_like(x)
            x = x / s
            grad, val = lse_smooth_grad(x, T)
            return val, grad
        
        x0 = f.copy()
        
        result = minimize(
            fun=lambda x: obj_and_grad(x)[0],
            x0=x0,
            jac=lambda x: obj_and_grad(x)[1],
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': 200, 'maxfun': 5000, 
                     'ftol': 1e-12, 'gtol': 1e-8,
                     'iprint': -1}
        )
        
        f_new = np.maximum(result.x, 0.0)
        s = f_new.sum()
        if s > 1e-15:
            f_new = f_new / s
        
        cur_obj = fast_convolve(f_new).max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_f = f_new.copy()
            if verbose:
                print(f"    Phase {phase} (T={T:.4f}): obj={best_obj:.10f} (iters={result.nit})")
        
        f = f_new
        
        if time.monotonic() > time.monotonic() + time_limit:
            break
    
    # Final evaluation
    final_obj = fast_convolve(best_f).max()
    return best_f, final_obj

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = next_fast_len(size)
    A = fft(a, N)
    return ifft(A * A, N).real[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
