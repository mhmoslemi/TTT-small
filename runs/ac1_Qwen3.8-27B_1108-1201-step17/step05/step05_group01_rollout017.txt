    n = len(a)
    conv = fast_convolve(a)
    f_p = np.sum(conv**p) ** (1.0/p)
    
    # Compute u_k = conv_k^(p-1)
    u = conv**(p-1)

    A = np.fft.rfft(a, N_fft)
