    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(f)
    return 2.0 * n * conv.max() / (s * s)


def smooth_grad(f, T):
    """
    Compute gradient of J(f) = 2*n*logsumexp(conv(f,f)/T) / (sum(f))^2.
    Returns (gradient, current_smooth_objective).
    """
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.zeros(n), float('inf')
    conv = fast_convolve(f)
    max_c = conv.max()
    shifted = (conv - max_c) / T
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    w_sum = exp_shifted.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_shifted / w_sum

    # Cross-correlation: g[j] = sum_h f[h] * w[j+h]
    fft_len = 1 << (2 * (2 * n - 1) - 1).bit_length()
    F = np.fft.rfft(f, fft_len)
    W_rev = w[::-1].copy()
    W = np.fft.rfft(W_rev, fft_len)
    corr = np.fft.irfft(F * W, fft_len)
    g = corr[2 * n - 2 - np.arange(2 * n - 1)]

    L = T * (max_c / T + max_shifted + np.log(w_sum))
    grad = (2.0 * n / (T * s * s)) * (2.0 * g) - (4.0 * n * L / (s * s * s)) * f
    obj_smooth = 2.0 * n * L / (s * s)
    return grad, obj_smooth


def adam_smooth_optimize(f0, time_limit, verbose=False):
    """Adam optimizer on smooth log-sum-exp objective with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    T = 0.001
    best_f = f.copy()
    best_obj = get_obj(f)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, _ = smooth_grad(f, T)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        obj = get_obj(f_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam iter {it} ({elapsed:.1f}s): obj={best_obj:.10f}")
        else:
            no_improve += 1

        f = f_new
        T = max(0.00005, T * 0.9995)
        lr = lr * 0.9998

        if no_improve > 200:
            if T < 0.0002:
                lr = min(lr * 3.0, 0.05)
                T = 0.0005
            else:
                T = max(0.0002, T * 1.5)
                lr = lr * 0.7
            no_improve = 0

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.1
            if m_relax > 6.0:
                m_relax = 1.0
                f = project_simplex(best_f + np.random.default_rng(it * 31).standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 60:
                break
            m_relax = min(m_relax * 1.03, 4.0)
            continue

        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo, hi = best_alpha - 0.1, best_alpha + 0.1
            lo = max(0.0, lo)
            hi = min(1.0, hi)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.003 * min(no_improve // 80, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def cd_polish(f0, time_limit):
    """Coordinate descent polishing with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = get_obj(f)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_obj = cur_max
            best_j_val = old_val
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_ramp(n, direction=1, curvature=1.0):
    """Create a ramp-shaped initialization."""
    x = np.linspace(0, 1, n)
    if direction == 1:
        f = x ** curvature
    else:
        f = (1 - x) ** curvature
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f


def make_bump(n, width=0.5, peak=1.0):
    """Create a bump-shaped initialization."""
    x = np.linspace(0, 1, n)
    center = 0.5
    f = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f


def make_cosine(n, power=1.0):
    """Raised cosine initialization."""
    x = np.linspace(0, np.pi, n)
    f = (0.5 + 0.5 * np.cos(x)) ** power
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse initializations, Adam smooth optimization,
    LP-FW polishing, coordinate descent, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_cosine(768)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Adam on parent ===
    print(f"\n=== Phase 1: Adam on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(200, rem * 0.2)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = adam_smooth_optimize(best_seq, time_limit=t1, verbose=True)
        sc = get_obj(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  After Adam: {sc:.8f}")
        # Evaluate with exact function
        sc_exact = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc_exact) and sc_exact < best_score:
            best_score = sc_exact
            best_seq = a_opt.copy()

    # === Phase 2: LP-FW on current best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(400, rem * 0.3)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        sc = get_obj(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  After LP-FW: {sc:.8f}")

    # === Phase 3: Diverse initializations at same length ===
    print(f"\n=== Phase 3: Diverse inits (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    inits = []

    # Monotone ramps (rearrangement insight)
    inits.append(("ramp_up", make_ramp(n_best, 1, 1.0)))
    inits.append(("ramp_up2", make_ramp(n_best, 1, 2.0)))
    inits.append(("ramp_down", make_ramp(n_best, -1, 1.0)))
    inits.append(("ramp_mix", make_ramp(n_best, 1, 0.5) * 0.3 + make_cosine(n_best) * 0.7))

    # Bumps with different widths
    inits.append(("bump_narrow", make_bump(n_best, 0.3)))
    inits.append(("bump_wide", make_bump(n_best, 0.7)))
    inits.append(("cosine2", make_cosine(n_best, 2.0)))
    inits.append(("cosine05", make_cosine(n_best, 0.5)))

    # Perturbed best
    for scale in [0.005, 0.02]:
        noise = rng.standard_normal(n_best) * scale
        inits.append((f"perturb_{scale}", project_simplex(best_seq + noise)))

    for label, f_init in inits:
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_alloc = min(80, rem * 0.15)
        if t_alloc < 10:
            continue
        # Quick Adam
        a_opt, _ = adam_smooth_optimize(f_init, time_limit=min(t_alloc * 0.6, 50))
        sc = get_obj(a_opt)
        if sc < best_score:
            # Polish with LP-FW
            rem2 = deadline - time.monotonic() - 100
            if rem2 > 20:
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=min(rem2 * 0.3, 60))
                sc2 = get_obj(a_opt2)
                if sc2 < sc:
                    a_opt = a_opt2
                    sc = sc2
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f}")

    # === Phase 4: Multi-length search ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    length_deltas = [16, -16, 32, -32, 64, -64, 128, -128, 256, -256]
    for delta in length_deltas:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        target_n = n_best + delta
        if target_n < 100 or target_n > 3000:
            continue
        if target_n == n_best:
            continue
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80, rem * 0.15)
        if t_alloc < 10:
            continue
        a_opt, _ = adam_smooth_optimize(f_interp, time_limit=t_alloc * 0.6)
        sc = get_obj(a_opt)
        if sc < best_score:
            rem2 = deadline - time.monotonic() - 100
            if rem2 > 20:
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=min(rem2 * 0.3, 50))
                sc2 = get_obj(a_opt2)
                if sc2 < sc:
                    a_opt = a_opt2
                    sc = sc2
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [n={target_n}]: {sc:.8f}")

    # === Phase 5: Deep LP-FW on best ===
    print(f"\n=== Phase 5: Deep LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t5 = min(rem * 0.5, 300)
        print(f"  Budget: {t5:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t5, verbose=True)
        sc = get_obj(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  After deep LP-FW: {sc:.8f}")

    # === Phase 6: CD polish ===
    print(f"\n=== Phase 6: CD Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t6 = min(rem * 0.4, 120)
        a_opt, _ = cd_polish(best_seq, time_limit=t6)
        sc = get_obj(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  After CD: {sc:.8f}")

    # === Phase 7: Final Adam + LP-FW cycle ===
    print(f"\n=== Phase 7: Final cycle (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        # Short Adam
        t_a = min(rem * 0.3, 60)
        a_opt, _ = adam_smooth_optimize(best_seq, time_limit=t_a)
        sc = get_obj(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
        # Short LP-FW
        rem2 = deadline - time.monotonic() - 15
        if rem2 > 10:
            a_opt2, _ = lp_fw_optimize(best_seq, time_limit=rem2 * 0.8)
            sc2 = get_obj(a_opt2)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a_opt2.copy()
                print(f"  Final LP-FW: {sc2:.8f}")

    # === Phase 8: Micro-search ===
    print(f"\n=== Phase 8: Micro-search (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    for trial in range(20):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scales = [0.0001, 0.0003, 0.0005, 0.001, 0.002, 0.005, 0.01]
        scale = scales[trial % len(scales)]
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = get_obj(f_pert)
        if sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  Micro {trial}: {sc:.10f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
