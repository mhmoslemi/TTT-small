import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Compute the objective value 2*n*max(conv)/(sum)^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_n = a / s
    conv = fast_convolve(a_n)
    return 2.0 * len(a_n) * conv.max()


def spectral_init(n, seed=0):
    """
    Create a non-negative sequence with approximately flat power spectrum.
    a[j] = c + sum_{k=1}^{K} alpha_k * cos(2*pi*k*j/n + phi_k)
    where c ensures non-negativity.
    """
    rng = np.random.default_rng(seed)
    K = min(n // 2, 20)
    alphas = np.arange(1, K + 1) ** (-0.7)
    alphas = alphas / alphas.sum()
    phis = rng.uniform(0, 2 * np.pi, K)
    j = np.arange(n)
    s = np.zeros(n)
    for k in range(K):
        s += alphas[k] * np.cos(2 * np.pi * (k + 1) * j / n + phis[k])
    c = max(0.01, -s.min() + 0.005)
    a = c + s
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def harmonic_init(n, seed=0):

    Designed to have low peak in auto-convolution.

def u_shape_init(n, alpha=1.5):

def solve_lp_direction(a, M, n):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 120, 'presolve': True})
    if res.success:
        return res.x
    return None

def mirror_descent(a0, time_limit, deadline, verbose=False, lr_init=0.004, top_k=15):
    """Mirror descent with multiplicative weight updates and adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 1e-14)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = 0.0005
    no_improve = 0
    it = 0

    while time.monotonic() < min(t0 + time_limit, deadline):
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        conv = fast_convolve(a)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        temp *= 1.008

        if no_improve > 200:
            rng = np.random.default_rng(it * 37 + 5)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.6
            no_improve = 0

        if lr < 1e-8:
            break

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, deadline, verbose=False):
    """Peak-targeted coordinate descent: directly reduces contributions to top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)

    no_improve = 0

    while time.monotonic() < min(t0 + time_limit, deadline):
        conv = fast_convolve(a)
        M = conv.max()
        improved = False

        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:][::-1]

        for k in peak_indices:
            if time.monotonic() >= min(t0 + time_limit, deadline):
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            j_max = idx[np.argmax(products)]

            if a[j_max] < 1e-12:
                continue

            best_trial = None
            best_trial_obj = M

            for frac in [0.5, 0.3, 0.15, 0.8, 1.2, 1.5]:
                old_val = a[j_max]
                new_val = old_val * frac
                if new_val < 0 or new_val > 0.5:
                    continue
                trial = a.copy()
                trial[j_max] = new_val
                diff_mass = old_val - new_val
                if abs(diff_mass) > 1e-15:
                    # Redistribute to elements NOT contributing to this peak
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[indices_range(k, n)] = False
                    if non_contrib.sum() > 0:
                        wts = a[non_contrib].copy()
                        wsum = wts.sum()
                        if wsum > 1e-15:
                            trial[non_contrib] += diff_mass * wts / wsum
                    else:
                        trial = a.copy()
                        trial[j_max] = new_val
                        trial /= trial.sum()
                        continue
                trial = np.maximum(trial, 0.0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-14:
                    best_trial_obj = obj
                    best_trial = trial

            if best_trial is not None:
                a = best_trial
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_a = a.copy()
                    best_obj_full = 2.0 * n * best_obj

                no_improve = 0
        else:

    return best_a, best_obj


def indices_range(k, n):
    """Get the range of indices that contribute to convolution position k."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    return np.arange(js, je + 1)


def interpolated_sequence(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, a)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        a_harm = harmonic_init(n_try, seed=seed + 100 + n_try)
        a_u = u_shape_init(n_try)
        a_gauss = gaussian_init(n_try, 0.25)
        a_flat = np.ones(n_try) / n_try
        for a_init in [a_spec, a_harm, a_u, a_gauss, a_flat]:
            sc = try_seq(a_init, f"init_n{n_try}")
            init_candidates.append((a_init / a_init.sum(), sc))

    if parent is not None:
        init_candidates.append((parent, best_score if np.isfinite(best_score) else float('inf')))

    # Sort by score, keep top 5
    init_candidates.sort(key=lambda x: x[1])
    top_inits = [ic[0] for ic in init_candidates[:5]]

    print(f"  Best init score: {init_candidates[0][1]:.8f}")

    # === Phase 2: LP-FW on top initializations ===
    print(f"\n=== Phase 2: LP-FW Optimization ===")
    for idx, a_init in enumerate(top_inits[:3]):
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        t_alloc = min(rem / max(1, 3 - idx), 400)
        n_init = len(a_init)
        print(f"  LP-FW on init {idx} (n={n_init}): t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(a_init, time_limit=t_alloc, deadline=deadline, verbose=True)
        try_seq(a_opt, f"lpfw_{idx}")

    # === Phase 3: Mirror descent on current best ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_alloc = min(rem, 200)
            print(f"  MD (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc, deadline=deadline,
                                        verbose=True, lr_init=0.003, top_k=20)
            try_seq(a_opt, "md")

    # === Phase 4: Peak coordinate descent ===
    print(f"\n=== Phase 4: Peak Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  PCD (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = peak_coordinate_descent(best_seq, time_limit=t_alloc, deadline=deadline)
            try_seq(a_opt, "pcd")

    # === Phase 5: Try interpolation to different lengths ===
    print(f"\n=== Phase 5: Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [base_n + 64, base_n - 64, 896, 1024]:
            if target_n <= 0 or target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 40:
                break
            f_interp = interpolated_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.01:  # Only optimize if promising
                t_alloc = min(rem, 100)
                a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, deadline=deadline)
                try_seq(a_opt, f"lpfw_interp_{target_n}")

    # === Phase 6: Diversification - LP-FW from perturbed best ===
    print(f"\n=== Phase 6: Diversification ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(rem / max(1, 4 - p), 100)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                # Blend with U-shape
                f_u = u_shape_init(n_best)
                f_init = 0.7 * best_seq + 0.3 * f_u
                f_init = project_simplex(f_init)
            elif p == 2:
                # Blend with spectral
                f_spec = spectral_init(n_best, seed=seed + 999)
                f_init = 0.6 * best_seq + 0.4 * f_spec
                f_init = project_simplex(f_init)
            else:
                # Coarse perturbation
                noise = rng.standard_normal(n_best) * 0.01
                f_init = np.maximum(best_seq + noise, 0)
                f_init = f_init / f_init.sum()

            print(f"  Div variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, deadline=deadline)
            try_seq(a_opt, f"div_{p}")

    # === Phase 7: Final MD polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Final MD: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc, deadline=deadline,
                                        verbose=True, lr_init=0.002, top_k=25)
            try_seq(a_opt, "final_md")

    # === Phase 8: Final PCD squeeze ===
    print(f"\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem, 30)
            a_opt, obj = peak_coordinate_descent(best_seq, time_limit=t_alloc, deadline=deadline)
            try_seq(a_opt, "final_pcd")

    # Fallback
    if best_seq is None:
        best_seq = gaussian_init(512, 0.3)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
