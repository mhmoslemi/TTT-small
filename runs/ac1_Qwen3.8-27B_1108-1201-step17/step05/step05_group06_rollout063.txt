def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix efficiently
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # For each k, the constraint is: sum_j f[j] * g[k-j] <= M
    # This is a linear constraint in g
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in constraints k where k-j is in [0, n-1]
        # i.e., k in [j, j+n-1]
        # And the coefficient is f[k-j]
        if j + n - 1 < 2 * n - 1:
            A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

"""
Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: diverse initializations, log-sum-exp smooth optimization with Adam,
exact subgradient descent with Nesterov momentum, mirror descent multiplicative
updates, LP-based Frank-Wolfe, and peak-focused coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, T):
    """Compute gradient of logsumexp(conv(f,f)/T) and the convolution."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()

    # grad[i] = (2/T) * sum_j f[j] * w[i+j]
    # This is (2/T) * cross_correlation(w, f)[i]
    N = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(W * np.conj(F), N)[:n]
    grad = (2.0 / T) * corr
    return grad, conv, w

    n = len(f)

    return best_f, best_obj


def eval_coord(f, j, val, n):
    """Evaluate max conv when f[j]=val and others scaled to maintain sum=1."""
    old_val = f[j]
    denom = 1.0 - old_val
    if denom < 1e-12:
        return 1e10
    scale = (1.0 - val) / denom
    f_trial = f.copy()
    f_trial[j] = val
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    f_trial[mask] *= scale
    return fast_convolve(f_trial).max()


def coordinate_descent_peak(f0, time_limit, verbose=False):
    """Coordinate descent focusing on peak-relevant coordinates with golden section."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    ORDER = 0.6180339887

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        passes += 1
        improved_in_pass = False

        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))

        # Peak-relevant coordinates
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        peak_coords = list(range(lo, hi + 1))

        # Add some random coordinates for exploration
        rng = np.random.default_rng(passes * 54321)
        other = [c for c in range(n) if c < lo or c > hi]
        n_rand = min(len(other), max(10, n // 8))
        if n_rand > 0 and other:
            rand_coords = list(rng.choice(other, size=n_rand, replace=False))
        else:
            rand_coords = []

        all_coords = peak_coords + rand_coords

        for j in all_coords:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            # Golden section search
            lo_g = 0.0
            hi_g = 0.9
            c_g = hi_g - ORDER * (hi_g - lo_g)
            d_g = lo_g + ORDER * (hi_g - lo_g)

            fc = eval_coord(f, j, c_g, n)
            fd = eval_coord(f, j, d_g, n)

            for _ in range(10):
                if fc < fd:
                    hi_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = hi_g - ORDER * (hi_g - lo_g)
                    fc = eval_coord(f, j, c_g, n)
                else:
                    lo_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = lo_g + ORDER * (hi_g - lo_g)
                    fd = eval_coord(f, j, d_g, n)

            opt_val = (lo_g + hi_g) / 2.0
            opt_obj = eval_coord(f, j, opt_val, n)

            if opt_obj < best_obj - 1e-13:
                scale = (1.0 - opt_val) / denom
                f[j] = opt_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved_in_pass = True

        if not improved_in_pass and passes > 4:
            break

    return best_f, best_obj


def lp_get_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g = lp_get_direction(f, M * m_relax, n)
        if g is None:
            m_relax = min(m_relax * 1.15, 5.0)
            if m_relax > 4.5:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.05, 3.0)
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_relax = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.1, 4.0)

        if no_improve > 60:
            rng_p = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_reversed_geometric(n, r):
    """Reversed geometric (increasing) sequence normalized to sum=1."""
    a = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_power_law(n, p):
    """Power-law decay: f[i] ~ (i+1)^(-p), normalized."""
    i = np.arange(n, dtype=np.float64) + 1
    a = i ** (-p)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.003
    return a / a.sum()


def make_asymmetric_bell(n, center_frac=0.3, width=0.15):
    """Asymmetric bell-shaped sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center_frac) / width) ** 2)
    a = a + 0.001
    return a / a.sum()


def make_ramp(n, power=1.0):
    """Power ramp: f[i] ~ (i+1)^power, normalized."""
    i = np.arange(n, dtype=np.float64) + 1
    a = i ** power
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimizer combining smooth log-sum-exp Adam optimization,
    Nesterov subgradient descent, mirror descent, LP Frank-Wolfe, and
    peak-focused coordinate descent across diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old_score:.6f}]")

    # === Phase 0: Diverse initializations ===
    print("=== Phase 0: Diverse Initializations ===")
    t0 = time.monotonic()

    n_base = n_parent
    for n_try in [n_base, 512, 1024]:
        for r in [0.96, 0.97, 0.98, 0.99, 0.995]:
            g = make_geometric(n_try, r)
            _try(g, f"geo_r{r}_n{n_try}")
            g2 = make_reversed_geometric(n_try, r)
            _try(g2, f"rgeo_r{r}_n{n_try}")

        for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            g = make_power_law(n_try, p)
            _try(g, f"pow_p{p}_n{n_try}")
            g2 = make_ramp(n_try, p)
            _try(g2, f"ramp_p{p}_n{n_try}")

        for pw in [1.0, 1.5, 2.0, 3.0, 4.0]:
            g = make_cosine(n_try, power=pw)
            _try(g, f"cos_p{pw}_n{n_try}")

        for cf in [0.2, 0.3, 0.4, 0.6, 0.7]:
            for w in [0.08, 0.12, 0.18, 0.25]:
                g = make_asymmetric_bell(n_try, cf, w)
                _try(g, f"bell_c{cf}_w{w}_n{n_try}")

    print(f"  Phase 0 done in {time.monotonic()-t0:.1f}s: best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    # === Phase 1: Smooth optimization ===
    print(f"\n=== Phase 1: Smooth Optimization (best={best_score:.6f}) ===")
    base = best_seq.copy()

    rem = deadline - time.monotonic() - 800
    if rem > 30:
        t_alloc = min(rem, 500)
        print(f"  Smooth opt n={len(base)}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize(base, time_limit=t_alloc, T_start=0.4, T_end=0.003, verbose=True)
        _try(a_opt, "smooth_p1")

    # Mixed initialization for diversity
    rem = deadline - time.monotonic() - 700
    if rem > 20 and best_seq is not None:
        n_cur = len(best_seq)
        f_init = 0.5 * best_seq + 0.5 * (np.ones(n_cur) / n_cur)
        f_init = project_simplex(f_init)
        t_alloc = min(rem, 150)
        a_opt, _ = smooth_optimize(f_init, time_limit=t_alloc, T_start=0.3, T_end=0.01)
        _try(a_opt, "smooth_mixed")

    # === Phase 2: Subgradient + Mirror Descent ===
    print(f"\n=== Phase 2: Subgradient/Mirror (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        t_alloc = min(rem, 200)
        print(f"  Nesterov subgrad n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_nesterov(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "nesterov_p2")

    rem = deadline - time.monotonic() - 400
    if rem > 20:
        t_alloc = min(rem, 150)
        print(f"  Mirror descent n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "mirror_p2")

    # === Phase 3: LP-FW ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t_alloc = min(rem, 180)
        print(f"  LP-FW n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "lpfw_p3")

    # LP-FW with perturbation
    rem = deadline - time.monotonic() - 250
    if rem > 15 and best_seq is not None:
        noise = rng.standard_normal(len(best_seq)) * 0.005
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem, 100)
        a_opt, _ = lpfw_optimize(f_pert, time_limit=t_alloc)
        _try(a_opt, "lpfw_pert")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 15:
        t_cd = min(rem, 200)
        print(f"  Peak CD n={len(best_seq)}, t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent_peak(best_seq, time_limit=t_cd, verbose=True)
        _try(a_opt, "cd_peak")

    # === Phase 5: Second round of smooth on improved best ===
    print(f"\n=== Phase 5: Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem, 120)
            a_opt, _ = smooth_optimize(best_seq, time_limit=t_alloc,
                                       T_start=0.05, T_end=0.002, verbose=True)
            _try(a_opt, "smooth_fine")

        rem = deadline - time.monotonic() - 80
        if rem > 10:
            t_alloc = min(rem, 80)
            a_opt, _ = subgradient_nesterov(best_seq, time_limit=t_alloc)
            _try(a_opt, "nesterov_fine")

        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_alloc = min(rem, 60)
            a_opt, _ = coordinate_descent_peak(best_seq, time_limit=t_alloc)
            _try(a_opt, "cd_fine")

        # Perturbations
        for _ in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 10:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.002
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 30)
            a_opt, _ = smooth_optimize(f_pert, time_limit=t_alloc,
                                       T_start=0.03, T_end=0.002)
            _try(a_opt, "perturb")

    # === Phase 6: Try longer sequences ===
    print(f"\n=== Phase 6: Longer Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [1024, 1536, 2048]:
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            f_interp = interpolate_seq(best_seq, target_n)
            _try(f_interp, f"interp_n{target_n}")

            t1 = min(rem * 0.4, 60)
            a_opt, _ = smooth_optimize(f_interp, time_limit=t1, T_start=0.2, T_end=0.02)
            _try(a_opt, f"smooth_n{target_n}")

            rem = deadline - time.monotonic() - 50
            if rem > 20:
                t2 = min(rem * 0.5, 50)
                a_opt, _ = coordinate_descent_peak(f_interp, time_limit=t2)
                _try(a_opt, f"cd_n{target_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        try:
            best_score = float(evaluate_sequence(best_seq.tolist()))
        except Exception:
            best_score = 2.0

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
