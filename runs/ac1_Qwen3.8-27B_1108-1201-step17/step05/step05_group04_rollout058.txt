def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    conv = fast_convolve(a)

import time

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(f, conv, top_k, temp):
    """Compute subgradient using top-k peaks with soft-max weights."""
    n = len(f)
    if top_k >= conv.size:
        idx = np.arange(conv.size)
        vals = conv.copy()
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:].copy()
        vals = conv[idx].copy()

    shifted = vals / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        k_star = int(idx[np.argmax(vals)])
        sg = np.zeros(n)
        lo = max(0, k_star - n + 1)

    w /= ws
    sg = np.zeros(n)

        hi = min(n - 1, kl)
        if lo > hi:
            continue
        seg = f[kl - hi : kl - lo + 1]
        seg = seg[::-1]
        sg[lo:hi + 1] += 2.0 * w[l] * seg
    return sg


def adam_optimize(f0, time_limit, lr=0.005, temp_init=0.008, temp_decay=0.9995,
                  top_k=50, restart_interval=300, restart_noise=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """Adam optimizer with simplex projection and noise restarts."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        cur_max = conv.max()

        sg = compute_subgradient_topk(f, conv, top_k, temp)

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-9)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-16:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 5000 == 0:
                print(f"    Adam iter {t}: obj={best_obj:.12f}")
        else:
            no_improve += 1

        if no_improve >= restart_interval and time.monotonic() - t0 < time_limit * 0.85:
            rng = np.random.default_rng(t * 13 + 7 + int(time.monotonic() * 1000) % 10000)
            ns = restart_noise * (1.0 + 0.5 * min(no_improve // restart_interval, 5))
            f = project_simplex(best_f + rng.standard_normal(n) * ns)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.015, lr_decay=0.9995,

def lp_fw_direction(a, M, n_constraints=8):
    """Solve LP to find Frank-Wolfe direction using top constraints."""
    n = len(a)
    conv = fast_convolve(a)

    if n_constraints >= conv.size:
        peak_idx = np.arange(conv.size)
    else:
        peak_idx = np.argpartition(conv, -n_constraints)[-n_constraints:].copy()

    # Build constraint matrix
    A_ub = np.zeros((len(peak_idx), n))
    for ci, k in enumerate(peak_idx):
        k = int(k)
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                A_ub[ci, i] = a[idx]

    c = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            b = np.asarray(res.x, dtype=np.float64)
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass

    return np.ones(n) / n

            f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
            f = project_simplex(f_new)
            temp = max(temp * temp_decay, 1e-9)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-16:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400 and time.monotonic() - t0 < time_limit * 0.8:
            rng = np.random.default_rng(t * 23 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.006)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.9, 0.004)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1.0 - np.abs(2.0 * x - 1.0)
    elif shape == '_cos':
        x = np.linspace(0, 1, n)
        f = 0.5 * (1.0 + np.cos(np.pi * (2.0 * x - 1.0)))
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=6):
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.3)
        a = rng.uniform(0.3, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_tapered_init(n, taper_power=2.0):
    """Create a tapered (raised cosine) initialization."""
    x = np.linspace(0, 1, n)
    f = np.sin(np.pi * x) ** taper_power
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_asymmetric_init(n, bias=0.3):
    """Create an asymmetric initialization (mass shifted to one side)."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - bias) / 0.25) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def iterated_local_search(f0, time_limit, n_restarts=None,
                          noise_base=0.02, noise_decay=0.95,
                          opt_time_ratio=0.7, verbose=False):
    """
    Iterated local search: optimize, perturb, re-optimize, keep best.
    """
    n = len(f0)
    f = f0.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    restart = 0
    noise = noise_base

    while time.monotonic() - t0 < time_limit:
        # Optimize current f
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        opt_time = min(rem * opt_time_ratio, 30.0)
        if opt_time < 3:
            break

        f_opt, obj = adam_optimize(f, time_limit=opt_time,
                                   lr=0.004, temp_init=0.005,
                                   temp_decay=0.9995, top_k=40,
                                   restart_interval=200,
                                   restart_noise=0.005)
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt.copy()
            if verbose:
                print(f"    ILS restart {restart}: obj={best_obj:.12f}")

        # Evaluate
        try:
            score = float(evaluate_sequence(best_f.tolist()))
            if np.isfinite(score) and verbose:
                print(f"    ILS eval: score={score:.8f}")
        except Exception:
            pass

        # Perturb
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        restart += 1
        rng = np.random.default_rng(1000 + restart * 37)
        noise *= noise_decay
        noise = max(noise, 0.003)
        f = project_simplex(best_f + rng.standard_normal(n) * noise)

    return best_f, best_obj
