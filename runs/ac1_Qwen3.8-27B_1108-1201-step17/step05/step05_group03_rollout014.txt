Actually, one more thought. Let me consider the problem from the perspective of "what makes the convolution small?".

The convolution c[k] = sum_i x[i] x[k-i]. This is the "overlap" between x and a shifted/reversed version of itself. To make the max overlap small, we want x to have low "self-similarity" at all lags.

For a non-negative sequence, this means: at any shift k, the product x[i] * x[k-i] should be small on average. This is achieved when x has a specific structure - e.g., a monotone sequence where the "large" part and the "small" part don't overlap much.

For a monotonically decreasing sequence x[0] > x[1] > ... > x[n-1]:
- c[0] = x[0]^2 (large, since x[0] is the biggest)
- c[1] = 2*x[0]*x[1] (also large)
- c[n-1] = sum_i x[i]*x[n-1-i] (this involves products of large and small terms)
- c[2n-2] = x[n-1]^2 (small)

The max will likely be at c[0] = x[0]^2 or c[1] = 2*x[0]*x[1]. To minimize this, we want x[0] to be small, which means the sequence should be relatively flat.

But if the sequence is flat, then c[n-1] = sum x[i]*x[n-1-i] ≈ n*x^2 (if x[i] ≈ 1/n), and the score is 2n*(1/n) = 2.

So there's a tradeoff: flat sequences have low peak at the "ends" but high peak at the "center". Asymmetric sequences can have low peak at the center but high peak at the "starts".

The optimal sequence balances these. It should be:
- Not too peaked (to keep c[0], c[1] small)
- Not too flat (to keep c[n-1] small)
- Asymmetric (so that c[n-1] = sum x[i]*x[n-1-i] < sum x[i]^2)

This suggests a shape like: moderate at the start, gradually decreasing, with a specific rate of decrease that optimally balances the peaks.

    """One step of mirror descent (multiplicative weights) using subgradient."""
    n = len(a)

    pv = conv[peaks]
    
    # Weights: softmax over peaks
    temp = max(1e-8, (pv.max() - pv.min()) * 0.3 + 1e-10)

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def run_subgradient(a, time_limit, lr0=0.05, decay=0.9998, top_k_start=20,
                    top_k_end=100, momentum=0.7):
    n = len(a)
    a = normalize(a.copy())
    conv = fast_autoconv(a)
    best_score = score_of(a)
    best_a = a.copy()
    sg_avg = np.zeros(n)
    t0 = time.monotonic()
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        progress = min(1.0, step / (time_limit * 30))
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        top_k = max(5, top_k)
        sg = compute_subgradient(a, conv, top_k)
        sg_avg = momentum * sg_avg + (1 - momentum) * sg
        lr = lr0 / (1.0 + 0.1 * step)
        a_new = a * np.exp(-lr * sg_avg)
        a_new = np.maximum(a_new, 1e-30)
        a = a_new / a_new.sum()
        conv = fast_autoconv(a)
        sc = 2.0 * n * conv.max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def coord_descent_1d(a, time_limit, rng=None):
    n = len(a)
    if rng is None:
        rng = np.random.RandomState(0)
    a = normalize(a.copy())
    best_score = score_of(a)
    best_a = a.copy()
    t0 = time.monotonic()
    stall = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                lo, hi = 0.0, 0.05
            else:
                lo = max(0.0, old_val * 0.2)
                hi = min(0.5, old_val * 5.0)
                if hi - lo < 1e-8:
                    lo = max(0.0, old_val - 0.02)
                    hi = min(0.5, old_val + 0.02)
            other_sum = 1.0 - old_val
            if other_sum < 1e-12:
                continue

            def obj(v):
                v = max(0.0, min(v, 0.99))
                new_a = a.copy()
                new_a[j] = v
                scale = (1.0 - v) / other_sum
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] *= scale
                if (new_a < 0).any():
                    return float('inf')
                conv = fast_autoconv(new_a)
                return conv.max()

            try:
                res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 25})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    scale = (1.0 - v_new) / other_sum
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    new_a[mask] *= scale
                    new_a = np.maximum(new_a, 0.0)
                    s = new_a.sum()
                    if s < 1e-12:
                        continue
                    new_a /= s
                    sc = 2.0 * n * fast_autoconv(new_a).max()
                    if sc < best_score - 1e-13:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        if not improved:
            stall += 1
            if stall > 3:
                break
            # Random perturbation to escape
            a = normalize(a + rng.standard_normal(n) * 0.005)
            best_a2, bs2 = a.copy(), score_of(a)
            if bs2 < best_score:
                best_score = bs2
                best_a = a.copy()
        else:
            stall = 0
    return best_a, best_score


def coord_descent_2d(a, time_limit, rng=None):
    n = len(a)
    if rng is None:
        rng = np.random.RandomState(1)
    a = normalize(a.copy())
    best_score = score_of(a)
    best_a = a.copy()
    t0 = time.monotonic()
    stall = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n // 2):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            vi, vj = a[i], a[j]
            if vi < 1e-12 and vj < 1e-12:
                continue
            other_sum = 1.0 - vi - vj
            if other_sum < 1e-10:
                continue

            def obj_2d(params):
                ui, uj = params
                ui = max(0.0, min(ui, 0.99))
                uj = max(0.0, min(uj, 0.99))
                if ui + uj > 0.999:
                    return float('inf')
                new_a = a.copy()
                new_a[i] = ui
                new_a[j] = uj
                scale = (1.0 - ui - uj) / other_sum
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask[j] = False
                new_a[mask] *= scale
                if (new_a < -1e-10).any():
                    return float('inf')
                new_a = np.maximum(new_a, 0.0)
                conv = fast_autoconv(new_a)
                return conv.max()

            x0 = np.array([vi, vj])
            try:
                res = minimize(obj_2d, x0, method='Nelder-Mead',
                               options={'maxiter': 50, 'xatol': 1e-8, 'fatol': 1e-10})
                if res.success:
                    vi_new, vj_new = np.maximum(res.x, 0)
                    if vi_new + vj_new > 0.999:
                        continue
                    new_a = a.copy()
                    new_a[i] = vi_new
                    new_a[j] = vj_new
                    scale = (1.0 - vi_new - vj_new) / other_sum
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    new_a[mask] *= scale
                    new_a = np.maximum(new_a, 0.0)
                    s = new_a.sum()
                    if s < 1e-12:
                        continue
                    new_a /= s
                    sc = 2.0 * n * fast_autoconv(new_a).max()
                    if sc < best_score - 1e-13:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        if not improved:
            stall += 1
            if stall > 5:
                break
            a = normalize(a + rng.standard_normal(n) * 0.003)
        else:
            stall = 0
    return best_a, best_score


def simulated_annealing(a, time_limit, rng=None):
    n = len(a)
    if rng is None:
        rng = np.random.RandomState(2)
    a = normalize(a.copy())
    best_score = score_of(a)
    best_a = a.copy()
    t0 = time.monotonic()
    T = 0.01
    T_min = 1e-6
    cooling = 0.9995
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        cur_score = best_score
        # Propose move
        move_type = rng.random()
        if move_type < 0.4:
            # Single coordinate change
            j = rng.randint(n)
            delta = rng.standard_normal() * 0.02
            new_a = a.copy()
            new_a[j] = max(0.0, new_a[j] + delta)
            new_a = normalize(new_a)
        elif move_type < 0.7:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            amt = rng.random() * min(a[i], a[j]) * 0.5
            new_a = a.copy()
            new_a[i] -= amt
            new_a[j] += amt
            new_a = normalize(new_a)
        elif move_type < 0.85:
            # Block perturbation
            j = rng.randint(0, n - 10)
            block_len = rng.randint(5, 20)
            new_a = a.copy()
            new_a[j:j + block_len] += rng.standard_normal(block_len) * 0.01
            new_a = normalize(new_a)
        else:
            # Large random restart (with blending)
            new_a = normalize(0.7 * a + 0.3 * rng.random(n))

        new_score = 2.0 * n * fast_autoconv(new_a).max()
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            a = new_a
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        T = max(T * cooling, T_min)
    return best_a, best_score


def optimize_multi_stage(a_init, time_limit, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    # Stage 1: Subgradient (40% time)
    rem = time_limit - (time.monotonic() - t0)
    t1 = rem * 0.35
    if t1 > 5:
        a_sg, sc_sg = run_subgradient(a, t1, lr0=0.04, decay=0.9997,
                                      top_k_start=15, top_k_end=80, momentum=0.6)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
            a = a_sg.copy()
        print(f"      After SG: {best_score:.10f}")

    # Stage 2: 1D coordinate descent (25% time)
    rem = time_limit - (time.monotonic() - t0)
    t2 = rem * 0.3
    if t2 > 5:
        a_cd, sc_cd = coord_descent_1d(a, t2, rng=rng)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
            a = a_cd.copy()
        print(f"      After CD1D: {best_score:.10f}")

    # Stage 3: Simulated annealing (20% time)
    rem = time_limit - (time.monotonic() - t0)
    t3 = rem * 0.25
    if t3 > 5:
        a_sa, sc_sa = simulated_annealing(a, t3, rng=rng)
        if sc_sa < best_score:
            best_score = sc_sa
            best_a = a_sa.copy()
            a = a_sa.copy()
        print(f"      After SA: {best_score:.10f}")

    # Stage 4: 2D coordinate descent (10% time)
    rem = time_limit - (time.monotonic() - t0)
    t4 = rem * 0.3
    if t4 > 5:
        a_cd2, sc_cd2 = coord_descent_2d(a, t4, rng=rng)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_a = a_cd2.copy()
            a = a_cd2.copy()
        print(f"      After CD2D: {best_score:.10f}")

    # Stage 5: Final 1D CD polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        a_final, sc_final = coord_descent_1d(a, rem, rng=rng)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
        print(f"      After final CD: {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining diverse parametric initialization,
    adaptive subgradient descent, block coordinate descent, and simulated
    annealing. Explores sequence lengths from 512 to 4096 with emphasis on
    asymmetric profiles that exploit the non-convex structure.
    """
    rng = np.random.RandomState(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a = a / s
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
        return sc

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            ps = try_seq(parent, "parent")
            print(f"  Parent score: {ps:.10f}")
    except Exception:
        pass

    # Phase 1: Generate and evaluate diverse initializations
    print("=== Phase 1: Diverse initialization ===")
    candidates = []

    n_values = [512, 768, 1024, 1536, 2048]

    for n in n_values:
        # Geometric
        for r in [0.995, 0.997, 0.998, 0.999, 0.9995]:
            candidates.append((init_geometric(n, r), f"geo_{r}_{n}"))
        # Exponential decay
        for alpha in [0.005, 0.01, 0.02, 0.05, 0.1]:
            candidates.append((init_exp_decay(n, alpha), f"exp_{alpha}_{n}"))
        # Power law
        for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            candidates.append((init_power(n, p), f"pow_{p}_{n}"))
        # Skew Gaussian
        for center in [0.1, 0.2, 0.3, 0.5]:
            for sigma in [0.1, 0.2, 0.3]:
                for skew in [0, 1.0, 2.0, 3.0]:
                    candidates.append((init_skew_gauss(n, center, sigma, skew),
                                       f"sg_{center}_{sigma}_{skew}_{n}"))
        # Two-phase
        for frac in [0.3, 0.5, 0.7]:
            for r1 in [0.99, 0.995, 0.999]:
                for r2 in [0.99, 0.995]:
                    candidates.append((init_two_phase(n, frac, r1, r2),
                                       f"2ph_{frac}_{r1}_{r2}_{n}"))
        # Random smooth
        for s in range(3):
            candidates.append((init_random_smooth(n, seed + s * 100, 0.3),
                               f"rnd_{s}_{n}"))

    # Evaluate all quickly
    print(f"  Evaluating {len(candidates)} candidates...")
    scored = []
    for a, label in candidates:
        sc = score_of(a)
        scored.append((sc, a, label))
    scored.sort(key=lambda x: x[0])

    print(f"  Top 5 initializations:")
    for i in range(min(5, len(scored))):
        sc, a, label = scored[i]
        print(f"    {i+1}. [{label}] score={sc:.10f}")
        try_seq(a, label)

    # Phase 2: Optimize top diverse candidates
    print(f"\n=== Phase 2: Local optimization ===")

    # Select diverse top candidates
    selected = []
    seen_types = set()
    for sc, a, label in scored:
        if len(selected) >= 8:
            break
        base = '_'.join(label.split('_')[:2])
        n_part = label.split('_')[-1]
        key = (base, n_part)
        if key not in seen_types:
            selected.append((a, label, sc))
            seen_types.add(key)

    # Also ensure parent is in the list if good
    if best_a is not None and len(selected) < 10:
        for sc, a, label in scored:
            if 'parent' in label:
                selected.append((a, label, sc))
                break

    n_sel = len(selected)
    time_per = (deadline - time.monotonic() - 200) / max(1, n_sel)
    time_per = min(time_per, 300)  # cap per candidate

    for idx, (a_init, label, sc_init) in enumerate(selected):
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break
        t_alloc = min(time_per, rem)
        print(f"\n  [{idx+1}/{n_sel}] Optimizing {label} (init={sc_init:.8f}, t={t_alloc:.0f}s)")
        a_opt, sc_opt = optimize_multi_stage(a_init, t_alloc,
                                             rng=np.random.RandomState(seed + idx))
        try_seq(a_opt, label + "_opt")

    # Phase 3: Dimension exploration on best
    print(f"\n=== Phase 3: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        target_ns = [384, 512, 640, 768, 896, 1024, 1280, 1536, 2048, 3072]
        for tn in target_ns:
            if tn == best_n:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            t_alloc = min((deadline - time.monotonic() - 60) * 0.15, 100)
            if t_alloc < 10:
                break
            a_interp = interp_to_n(best_a, tn)
            print(f"  Interpolating to n={tn} (t={t_alloc:.0f}s)")
            a_opt, sc = optimize_multi_stage(a_interp, t_alloc,
                                             rng=np.random.RandomState(seed + 500 + tn))
            try_seq(a_opt, f"dim_{tn}")

    # Phase 4: Aggressive refinement on global best
    print(f"\n=== Phase 4: Aggressive refinement (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            # Re-optimize with more time
            t_final = min(rem * 0.6, 300)
            print(f"  Final multi-stage (t={t_final:.0f}s)")
            a_opt, sc = optimize_multi_stage(best_a, t_final,
                                             rng=np.random.RandomState(seed + 999))
            try_seq(a_opt, "final_multi")

            rem = deadline - time.monotonic() - 20
            if rem > 10:
                # Perturbation + re-optimize
                n_best = len(best_a)
                for trial in range(4):
                    rem = deadline - time.monotonic() - 10
                    if rem < 8:
                        break
                    noise = rng.standard_normal(n_best) * 0.003
                    a_pert = project_simplex(best_a + noise)
                    t_sub = min(rem * 0.5, 60)
                    if t_sub < 5:
                        break
                    a_opt2, sc2 = optimize_multi_stage(a_pert, t_sub,
                                                       rng=np.random.RandomState(seed + 2000 + trial))
                    try_seq(a_opt2, f"perturb_{trial}")

            # Final 1D CD polish
            rem = deadline - time.monotonic() - 5
            if rem > 5:
                a_polish, sc_pol = coord_descent_1d(best_a, rem,
                                                     rng=np.random.RandomState(seed + 3000))
                try_seq(a_polish, "polish")

    # Fallback
    if best_a is None:
        best_a = init_geometric(768, 0.999)
        best_score = score_of(best_a)
        print(f"  Fallback used: score={best_score:.8f}")

    # Verify
    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Something went wrong, use a safe fallback
        best_a = init_geometric(512, 0.999)
        best_score = score_of(best_a)

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_a)})")
    return [float(x) for x in best_a]
