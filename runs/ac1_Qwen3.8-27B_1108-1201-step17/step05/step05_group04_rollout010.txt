Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based multiplicative weights,
direct subgradient polishing, structured local search, and multi-start.
"""

import time
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def convolve_with(a, b):
    """Linear convolution of a and b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)]) * (target_sum / max(v[0], 0.0) if v[0] > 0 else 1.0)
    v = v * target_sum  # scale to target sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_best_response(a, M_upper=None):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g and M.
    """
    n = len(a)
    # Build constraint matrix: for each k in 0..2n-2, sum_i a[i]*g[k-i] <= M
    # This is: row k of the matrix times [g, -M] <= 0
    # Or equivalently: A_k @ g - M <= 0
    # In linprog form: A_ub @ x <= b_ub, where x = [g_0,...,g_{n-1}, M]
    
    # A_k[i] = a[i] if 0 <= k-i < n, else 0
    # So A_k = a[max(0,k-n+1):min(k+1,n)] (reversed) interleaved... 
    # Actually, (a*g)[k] = sum_{i} a[i]*g[k-i] where 0<=i<n and 0<=k-i<n
    # = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i]*g[k-i]
    # Let j = k-i, then i = k-j, and j ranges from max(0, k-(n-1)) to min(k, n-1)
    # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[k-j]*g[j]
    
    # So the coefficient of g[j] in row k is a[k-j] if 0 <= k-j < n, else 0.
    
    num_constraints = 2 * n - 1
    # A_ub has shape (num_constraints, n+1): first n columns for g, last column for -M
    A_ub = np.zeros((num_constraints, n + 1))
    b_ub = np.zeros(num_constraints)
    
    for k in range(num_constraints):
        # Coefficients for g[j]: a[k-j] for j where 0 <= k-j < n
        # j ranges from max(0, k-(n-1)) to min(k, n-1)
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, n] = -1.0  # coefficient of M
        b_ub[k] = 0.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: g_i >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        M_opt = result.x[n]
        return g, M_opt
    else:
        return None, None

def lp_best_response_fast(a, M_init=None):
    """
    Faster LP best response using the Toeplitz structure.
    If M_init is given, check feasibility first (is there g with (a*g)[k] <= M_init?).
    """
    n = len(a)
    
    # Use the convolution structure to build the matrix more efficiently
    num_constraints = 2 * n - 1
    
    # Build the Toeplitz-like matrix T where T[k, j] = a[k-j] if valid
    # This is actually the "convolution matrix" of a
    T = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        indices = np.arange(j_min, j_max + 1)
        T[k, indices] = a[k - indices]
    
    A_ub = np.hstack([T, -np.ones((num_constraints, 1))])
    b_ub = np.zeros(num_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        M_opt = result.x[n]
        return g, M_opt
    return None, None

def subgradient_step(a, lr, n_top=5):
    """
    One subgradient step on the simplex for minimizing max_k (a*a)[k].
    Uses the top-n_top convolution values for a better subgradient.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Get top n_top values
    top_idx = np.argsort(conv)[-n_top:]
    top_vals = conv[top_idx]
    max_val = conv[top_idx[0]]  # sorted desc
    
    # Subgradient: for each j, sg[j] = sum over top k of (a*a)[k] coefficient
    # (a*a)[k] = sum_i a[i]*a[k-i], d/d a[j] = 2*a[k-j] if valid for the terms involving j
    # More precisely: d(a*a)[k] / d(a[j]) = a[k-j] + a[j-(k)] if both valid (careful with j=k-j)
    # Actually: (a*a)[k] = sum_i a[i]*a[k-i]
    # d/d(a[j]) = a[k-j] (from i=j term) + a[j-(k)] = a[k-j] (from k-i=j, i=k-j term)
    # If j = k-j (i.e., 2j = k), then it's 2*a[k-j].
    # Wait, let me be more careful.
    # (a*a)[k] = sum_{i: valid} a[i] * a[k-i]
    # The terms involving a[j] are:
    #   i = j: contributes a[j] * a[k-j] (if k-j is valid, i.e., 0 <= k-j < n)
    #   k-i = j, i.e., i = k-j: contributes a[k-j] * a[j] (if 0 <= k-j < n)
    # If j != k-j (i.e., j != k/2), these are two separate terms, each giving a[k-j].
    # If j == k-j (i.e., 2j = k), it's one term giving 2*a[j] = 2*a[k-j].
    # So d(a*a)[k]/d(a[j]) = a[k-j] if 0 <= k-j < n, else 0
    # (counting the case 2j=k as a[k-j] since it appears once in the sum... wait)
    
    # Let me just compute it numerically for safety.
    # Actually, for the subgradient of max, if the max is at unique k*, then:
    # sg = d(a*a)[k*] / da = (a reversed, truncated) essentially.
    # Specifically, d(a*a)[k*]/d(a[j]) = a[k*-j] if 0 <= k*-j < n, else 0.
    # (This is because (a*a)[k*] = sum_i a[i]*a[k*-i], and d/d(a[j]) picks out i=j giving a[k*-j],
    #  and i=k*-j giving a[j]*... wait, let me just be careful.)
    
    # (a*a)[k] as a function of a: it's a quadratic form.
    # (a*a)[k] = sum_{i=0}^{n-1} a[i] * a[k-i] * I[0 <= k-i < n]
    # = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
    
    # The gradient w.r.t. a[j] is:
    # = sum_{i: 0<=i<n, 0<=k-i<n} (delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j})
    # = a[k-j] * I[0<=k-j<n] + a[j] * I[0<=j<n]  -- wait this isn't right either.
    
    # Let me think again. f(a) = sum_i a[i] * a[k-i] (with validity conditions).
    # df/da[j] = sum_i (da[i]/da[j] * a[k-i] + a[i] * da[k-i]/da[j])
    # = sum_i (delta_{ij} * a[k-i] + a[i] * delta_{k-i,j})
    # = a[k-j] * I[0<=k-j<n] + a[j] * I[0<=j<n and k-j valid... ]
    
    # Hmm, I'm overcomplicating this. Let me just note that:
    # f(a) = sum_i a[i] * a[k-i]
    # This is a bilinear form: f = a^T Q_k a where (Q_k)_{i,j} = 1 if i+j=k (wait, that's not quite right for linear convolution).
    
    # Actually for linear convolution: (a*a)[k] = sum_{i+j=k} a[i]*a[j] where 0<=i<n, 0<=j<n.
    # So it's sum over pairs (i,j) with i+j=k.
    # If i != j: the pair (i,j) and (j,i) both contribute, giving 2*a[i]*a[j].
    # If i = j (i.e., 2i=k): only one term, giving a[i]^2.
    # So (a*a)[k] = sum_{i: 0<=i<n, 0<=k-i<n, i<k-i} 2*a[i]*a[k-i] + (a[k/2]^2 if k even and k/2 < n)
    
    # The gradient:
    # d/d(a[j]) [(a*a)[k]] = sum over valid i: (delta_{ij}*a[k-i] + a[i]*delta_{k-i,j})
    # = a[k-j] (from i=j, if k-j valid) + a[j] (from k-i=j, i.e., i=k-j, if k-j valid)
    # = 2*a[k-j] if 0<=k-j<n and j != k-j
    # = 2*a[j] if 2j=k (j = k/2)
    # = a[k-j] if 0<=k-j<n and j = k-j (which is the same as 2j=k)
    # Wait, I keep confusing myself.
    
    # Let me just use: d(a*a)[k] / d(a[j]) = a_rev[k-j] where a_rev = a reversed? No...
    
    # Simplest: (a*a)[k] = <a, S_k a> where S_k is the shift matrix.
    # S_k has (S_k a)[i] = a[k-i] if 0 <= k-i < n, else 0.
    # So (a*a)[k] = sum_i a[i] * (S_k a)[i] = a^T S_k a.
    # But S_k is symmetric (since (S_k a)[i] = a[k-i] and (S_k^T a)[i] = a[k-i] as well... 
    # wait, S_k is the "anti-diagonal" permutation of size depending on k).
    
    # Actually, S_k is symmetric: (S_k)_{i,j} = 1 if i+j=k and 0<=i<n and 0<=j<n.
    # This IS symmetric since i+j=k iff j+i=k.
    # So (a*a)[k] = a^T S_k a where S_k is symmetric.
    # Gradient = 2 * S_k * a.
    # (S_k a)[j] = sum_i (S_k)_{j,i} a[i] = a[k-j] if 0<=k-j<n, else 0.
    
    # So the gradient of (a*a)[k] w.r.t. a is: 2 * (S_k a)
    # where (S_k a)[j] = a[k-j] if 0<=k-j<n, else 0.
    
    # For a subgradient of max_k (a*a)[k], if argmax = {k_1, ..., k_m}:
    # sg = (1/m) * sum_{r=1}^{m} 2 * S_{k_r} * a
    
    sg = np.zeros(n)
    for k in top_idx:
        # S_k * a: result[j] = a[k-j] if 0 <= k-j < n
        j_range = np.arange(max(0, k - (n-1)), min(k, n-1) + 1)
        vals = a[k - j_range]
        sg[j_range] += 2 * vals / len(top_idx)
    
    # Projected gradient step
    a_new = project_simplex(a - lr * sg)
    return a_new, conv

def lp_mw_optimize(a0, time_limit, n_iterations_max=5000, use_running_avg=True):
    """
    LP-based multiplicative weights optimizer.
    At each step, find the best response g to current a, then update.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    # Running average
    avg_a = a.copy() if use_running_avg else None
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and iteration < n_iterations_max:
        iteration += 1
        
        # Solve LP for best response
        g, M_lp = lp_best_response_fast(a)
        
        if g is None:
            # LP failed, use subgradient step
            a, conv = subgradient_step(a, lr=0.01, n_top=5)
        else:
            # Update using running average
            if use_running_avg:
                t = iteration
                avg_a = ((t - 1) / t) * avg_a + (1.0 / t) * g
            else:
                # Direct update with decreasing step
                t_step = 1.0 / np.sqrt(iteration)
                a = (1 - t_step) * a + t_step * g
                a = np.maximum(a, 0)
                a = a / a.sum()
            
            if use_running_avg:
                a = avg_a.copy()
        
        # Evaluate
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-MW iter {iteration}: obj={best_obj:.10f}, t={elapsed:.1f}s")
        
        if no_improve > 200:
            # Reset with perturbation
            g_pert = g + np.random.default_rng(iteration).standard_normal(n) * 0.01
            g_pert = project_simplex(g_pert)
            if use_running_avg:
                avg_a = g_pert.copy()
                a = avg_a.copy()
            else:
                a = g_pert.copy()
            no_improve = 0
            iteration = 0
    
    return best_a, best_obj

def subgradient_optimize(a0, time_limit, lr_init=0.01, lr_decay=0.999, n_top=5):
    """Direct subgradient optimization on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    lr = lr_init
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        a, conv = subgradient_step(a, lr, n_top)
        
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= lr_decay
        if lr < 0.001:
            lr = 0.001
        
        if no_improve > 300:
            rng = np.random.default_rng(iteration * 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            lr = lr_init
            no_improve = 0
            iteration = 0
    
    return best_a, best_obj

def local_search(a0, time_limit, n_trials=100):
    """
    Local search: try zeroing coordinates, merging, splitting.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        # Try random perturbations
        for _ in range(min(n_trials, max(10, (time_limit - (time.monotonic()-t0)) * 5))):
            if time.monotonic() - t0 > time_limit:
                break
            
            trial = a.copy()
            pert_type = rng.integers(0, 4)
            
            if pert_type == 0:
                # Zero out a random coordinate
                idx = rng.integers(0, n)
                if trial[idx] > 0.01:
                    trial[idx] = 0
                    trial = trial / trial.sum()
            elif pert_type == 1:
                # Merge two adjacent coordinates
                idx = rng.integers(0, n - 1)
                total = trial[idx] + trial[idx+1]
                trial[idx] = total
                trial[idx+1] = 0
                trial = trial / trial.sum()
            elif pert_type == 2:
                # Random smooth perturbation
                sigma = rng.uniform(0.01, 0.05)
                neighbors = np.ones(2*int(sigma*n)+1) / (2*int(sigma*n)+1)
                noise = np.convolve(rng.standard_normal(n), neighbors, 'same')
                noise -= noise.mean()
                trial = trial + noise * 0.02
                trial = project_simplex(trial)
            else:
                # Exchange mass between two coordinates
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    delta = min(trial[i], trial[j]) * rng.uniform(0.1, 0.5)
                    trial[i] -= delta
                    trial[j] += delta
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()
            
            if trial.sum() < 0.01:
                continue
                
            conv_t = fast_convolve(trial)
            obj_t = conv_t.max()
            
            if obj_t
