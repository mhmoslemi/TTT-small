"""

3. Peak-directed subgradient with Nesterov momentum

Key innovations: continuous annealing schedule, adaptive peak focusing,

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5})
    if res.success:
        return res.x
    return None


def solve_lp_direction_fast(f, M, n, top_k=None):
    """Solve LP with only top-k active constraints (faster for large n)."""
    if top_k is None or top_k >= 2 * n - 1:
        return solve_lp_direction(f, M, n)
    conv_f = fast_convolve(f)
    active = np.argpartition(conv_f, -top_k)[-top_k:]
    c = -np.ones(n)
    A_ub = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A_ub[idx, j] = f[k - j]
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 3})
    if res.success:
        g = res.x
        full_conv = fast_convolve(g)
        if full_conv.max() <= M * 1.01:
            return g
        return None
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M, line search, and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        use_fast = (n > 600 and it % 3 == 0)
        if use_fast:
            g = solve_lp_direction_fast(f, M * m_exp, n, top_k=max(20, n // 4))
        else:
            g = solve_lp_direction(f, M * m_exp, n)

        if g is None:
            m_exp *= 1.12
            if m_exp > 8.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.06, 5.0)
            continue

        best_alpha = 0.0
        best_val = M

        alphas = [0.02, 0.04, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7 + int(time.monotonic() * 1000))
            scale = 0.002 + 0.002 * min(no_improve // 80, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def smooth_obj_and_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def lbfqg_optimize(f0, time_limit, verbose=False):
    """Optimize using L-BFGS with continuous adaptive beta annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t = np.log(f + 1e-15)
    t0_time = time.monotonic()

    beta_start = 3.0
    beta_end = 600.0
    n_phases = 12
    n_iters_per_phase = 150

    for phase in range(n_phases):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break

        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        sub_time = remaining / (n_phases - phase)

        try:
            res = minimize(
                smooth_obj_and_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': n_iters_per_phase, 'gtol': 1e-10,
                         'maxls': 40, 'maxfun': n_iters_per_phase * 3,
                         'ftol': 1e-16}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS phase {phase}, beta={beta:.1f}: obj={best_obj:.10f}")
        except Exception:
            break

    return best_f, best_obj


def nesterov_smooth_optimize(f0, time_limit, verbose=False):
    """Nesterov-accelerated gradient on smoothed objective with dynamical step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    k = 0
    a_prev = f.copy()
    a_curr = f.copy()
    t = np.log(f + 1e-15)
    t_prev = t.copy()

    beta = 10.0
    momentum = 0.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        k += 1
        mom = k / (k + 2.5)

        if k % 50 == 0:
            beta = min(beta * 1.15, 500.0)

        y = a_curr + mom * (a_curr - a_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        t_y = np.log(y)

        obj_y, grad_t = smooth_obj_and_grad(t_y, beta)
        a_y = softmax_fn(t_y)
        step = 1.0 / (beta * a_y.max() * 4.0)
        step = min(step, 0.01)

        t_new = t_y - step * grad_t
        a_new = softmax_fn(t_new)
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        obj_new = fast_convolve(a_new).max()
        a_prev = a_curr.copy()
        a_curr = a_new
        t_prev = t_y.copy()
        t = t_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = a_new.copy()
            no_improve = 0
            if verbose and k % 500 == 0:
                print(f"    Nesterov iter {k}: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(k * 17 + int(time.monotonic() * 100))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                a_curr = f.copy()
                a_prev = f.copy()
                t = np.log(f + 1e-15)
                no_improve = 0
                beta = max(5.0, beta * 0.5)

    return best_f, best_obj


def peak_subgradient_optimize(f0, time_limit, verbose=False):
    """Peak-directed subgradient with adaptive temperature and Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    mom = 0.9
    temp = 0.0001
    top_k = max(20, n // 6)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, temp={temp:.2e}")
        else:
            f = f_new
            no_improve += 1

        lr *= 0.9995
        temp *= 1.002
        mom = min(mom + 0.0003, 0.98)

        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + int(time.monotonic() * 100))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = max(lr * 0.7, 0.001)
            temp = max(temp * 0.5, 1e-6)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with multi-step line search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        sweep += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            mults = [0.5, 0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]
            for mult in mults:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial_sum = f_trial.sum()
                if f_trial_sum < 1e-10:
                    continue
                f_trial = f_trial / f_trial_sum

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    if old_n >= 4:
        from scipy.interpolate import PchipInterpolator
        try:
            interp = PchipInterpolator(old_x, f)
            f_interp = interp(new_x)
        except Exception:
            f_interp = np.interp(new_x, old_x, f)
    else:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def mwu_optimize(f0, time_limit, verbose=False):
    """Multiplicative Weight Update with smooth gradient and adaptive eta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta_ls = 50.0
    eta = 0.04
    momentum = 0.75
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta_ls * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        v = momentum * v + (1.0 - momentum) * grad_a

        log_f = np.log(f + 1e-30) - eta * v
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    MWU iter {it}: obj={best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1
            eta *= 0.98

        if no_improve > 250:
            rng = np.random.default_rng(it * 23 + int(time.monotonic() * 100))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            v = np.zeros(n)
            eta = 0.03
            beta_ls = max(20.0, beta_ls * 0.7)
            no_improve = 0

        eta = max(eta, 0.001)
        beta_ls = min(beta_ls * 1.001, 200.0)

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining L-BFGS annealing, LP-FW, Nesterov
    subgradient, MWU, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform 512")

    # === Phase 1: Quick L-BFGS annealing (250s) ===
    print("\n=== Phase 1: L-BFGS annealing ===")
    t1 = min(250, deadline - time.monotonic() - 1200)
    if t1 > 30:
        a_opt, _ = lbfqg_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lbfqg_phase1")

    # === Phase 2: LP-FW on best (400s) ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(400, deadline - time.monotonic() - 800)
    if t2 > 30:
        a_opt, _ = lpfw_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw_phase2")

    # === Phase 3: Nesterov smooth optimization (200s) ===
    print(f"\n=== Phase 3: Nesterov (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t3 = min(200, deadline - time.monotonic() - 500)
    if t3 > 20:
        a_opt, _ = nesterov_smooth_optimize(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "nesterov_phase3")

    # === Phase 4: Multi-length exploration (200s) ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    n_base = len(base)
    target_lengths = []
    for delta in [-128, -64, 64, 128, 256]:
        candidate_n = n_base + delta
        if 128 <= candidate_n <= 2048:
            target_lengths.append(candidate_n)
    target_lengths = sorted(set(target_lengths))

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.3, 60)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lbfqg_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}_lbfqg")

        rem2 = deadline - time.monotonic() - 200
        if rem2 > 20:
            a_opt2, _ = lpfw_optimize(a_opt, time_limit=min(rem2 * 0.5, 40))
            try_seq(a_opt2, f"len_{target_n}_lpfw")

    # === Phase 5: Deep optimization on best (300s) ===
    print(f"\n=== Phase 5: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 5a: LP-FW deep
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.4, 150)
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 5b: L-BFGS deep
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.4, 120)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # 5c: Peak subgradient
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.5, 100)
            a_opt, _ = peak_subgradient_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_sg")

        # 5d: MWU
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem * 0.6, 80)
            a_opt, _ = mwu_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_mwu")

    # === Phase 6: Coordinate descent + final polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem * 0.4, 80)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd_polish")

        # Final LP-FW burst
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(rem * 0.5, 60)
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

        # Final L-BFGS burst
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem * 0.6, 40)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lbfqg")

        # Diverse restarts
        for attempt in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break

            noise_scale = 0.002 + 0.002 * attempt
            noise = rng.standard_normal(n_best) * noise_scale
            f_start = project_simplex(best_seq + noise)

            if attempt % 2 == 0:
                t_alloc = min(rem * 0.6, 30)
                a_opt, _ = lpfw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"restart_lpfw_{attempt}")
            else:
                t_alloc = min(rem * 0.6, 25)
                a_opt, _ = lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"restart_lbfqg_{attempt}")

        # One last CD pass
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "final_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
