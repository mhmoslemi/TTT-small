"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using LP-based iterative flattening
(azaf-rotation), peak-targeted coordinate descent, and simulated annealing.
Key innovations: proper LP formulation for the inner problem, adaptive

import numpy as np
from scipy.optimize import linprog, minimize_scalar

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize_to_sum(a, target=1.0):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a) * target
    return a * (target / s)

def solve_lp_for_g(f, M):
    """Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    # (f*g)[k] = sum_i f[i] * g[i+k-n+1]... 
    # Actually, np.convolve(f, g)[k] = sum_i f[i] * g[k-i]
    # where the sum is over valid i.
    # For k = 0: f[0]*g[0]
    # For k = 1: f[0]*g[1] + f[1]*g[0]
    # ...
    # For k = n-1: f[0]*g[n-1] + f[1]*g[n-2] + ... + f[n-1]*g[0]
    # For k = n: f[1]*g[n-1] + ... + f[n-1]*g[0]
    # ...
    # For k = 2n-2: f[n-1]*g[n-1]
    
    # So constraint k: sum_i f[i] * g[k-i] <= M for valid i
    # i.e., sum_i f[i] * g[j] where j = k-i, i.e., j ranges over valid indices
    
    # Let's build the constraint matrix A_ub where A_ub @ g <= M * ones
    # A_ub[k, j] = coefficient of g[j] in constraint k
    # (f*g)[k] = sum_j f[k-j] * g[j] where 0 <= k-j < n and 0 <= j < n
    # So A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # Number of constraints: 2n-1
    # Number of variables: n
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]
    
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    # bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n) / n

def lp_iteration(f, t=0.3):
    """One LP iteration: solve for g, blend with f."""
    n = len(f)
    f = normalize_to_sum(f, 1.0)
    conv = fast_convolve(f)
    M = conv.max()
    
    g = solve_lp_for_g(f, M)
    g = normalize_to_sum(g, 1.0)
    
    f_new = (1 - t) * f + t * g
    f_new = normalize_to_sum(f_new, 1.0)
    return f_new

def multiplicative_weight_descent(a, time_limit, verbose=False):
    """
    Multiplicative weight descent targeting the auto-correlation peaks.
    At each step, compute the subgradient of max_k b[k] and update multiplicatively.
    """
    n = len(a)
    a = normalize_to_sum(a.copy(), 1.0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    
    # Learning rate schedule
    lr0 = 0.5
    decay = 0.999
    
    cur_lr = lr0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)
        max_conv = conv.max()
        
        # Find active lags (within tolerance of max)
        tol = max_conv * 0.01
        active = np.where(conv >= max_conv - tol)[0]
        
        # Compute subgradient: for each active lag k, add contribution to a
        # d(max b)/da[i] = sum_{k active} (1/|active|) * (a[k-i] + a[(k-i)] if valid)
        # More carefully: b[k] = sum_j a[j]*a[k-j]
        # db[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # But for i and k-i to be different, we get two terms
        
        sg = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                term = 0.0
                k_minus_i = k - i
                if 0 <= k_minus_i < n:
                    term += a[k_minus_i]
                i_minus_k = i - k
                if 0 <= i_minus_k < n and i_minus_k != k_minus_i:
                    term += a[i_minus_k]
                sg[i] += term / len(active)
        
        # Multiplicative update
        a = a * np.exp(-cur_lr * sg)
        a = np.maximum(a, 1e-15)
        a = normalize_to_sum(a, 1.0)
        
        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        
        cur_lr *= decay
        if cur_lr < 1e-4:
            cur_lr = lr0 * 0.1
        
        if verbose and step % 1000 == 0:
            print(f"    MWD step {step}: score={best_score:.10f}, lr={cur_lr:.6f}")
    
    return best_a, best_score

def lp_based_optimization(a, time_limit, verbose=False):
    """LP-based iterative optimization (the state-of-the-art approach)."""
    n = len(a)
    a = normalize_to_sum(a.copy(), 1.0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iteration = 0
    
    # Adaptive blending
    t_blend = 0.3
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        # Solve LP
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_lp_for_g(a, M)
        g = normalize_to_sum(g, 1.0)
        
        # Adaptive blending: if improvement is large, use bigger step
        f_new = (1 - t_blend) * a + t_blend * g
        f_new = normalize_to_sum(f_new, 1.0)
        
        sc = compute_score(f_new)
        
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
            a = f_new
            # Increase blending if improving
            t_blend = min(0.7, t_blend * 1.05)
        else:
            # Decrease blending
            t_blend = max(0.05, t_blend * 0.8)
            a = (1 - t_blend * 0.5) * a + t_blend * 0.5 * g
            a = normalize_to_sum(a, 1.0)
            sc2 = compute_score(a)
            if sc2 < best_score:
                best_score = sc2
                best_a = a.copy()
        
        if verbose and iteration % 20 == 0:
            print(f"    LP iter {iteration}: score={best_score:.10f}, t={t_blend:.3f}")
        
        # Check convergence
        if iteration > 200:
            break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize_to_sum(f_interp, 1.0)

def geometric_sequence(n, r):
    a = r ** np.arange(n)
    return normalize_to_sum(a, 1.0)

def reversed_geometric_sequence(n, r):
    a = r ** np.arange(n - 1, -1, -1)
    return normalize_to_sum(a, 1.0)

def linear_decreasing(n):
    a = np.linspace(1.0, 0.01, n)
    return normalize_to_sum(a, 1.0)

def bimodal_sequence(n, frac=0.3):
    """Two peaks at frac and 1-frac of the length."""
    a = np.zeros(n)
    for c in [int(n * frac), int(n * (1 - frac))]:
        for i in range(n):
            a[i] = max(a[i], np.exp(-(i - c) ** 2 / (n * 0.05)))
    return normalize_to_sum(a, 1.0)

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using LP-based iterative flattening,
multiplicative weight descent, simulated annealing, and refined coordinate
descent. Key innovations: efficient LP inner loop for auto-correlation
flattening, adaptive peak-tracking MWD, structured diverse starts including
anti-symmetric (monotone) profiles that exploit non-symmetry to reduce
center-lag auto-correlation below 1/n.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize_to_sum(a, target=1.0):
    """Normalize non-negative array to given sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a) * target
    return a * (target / s)


def build_lp_matrix(f):
    """Build A_ub matrix for LP: (f*g)[k] <= M constraints."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = f[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_for_g(f, M):
    """Solve LP: maximize sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    n = len(f)
    A_ub = build_lp_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return np.ones(n) / n


def lp_iteration(a, verbose=False):
    """Single LP iteration: solve for g, return blended result."""
    n = len(a)
    a = normalize_to_sum(a, 1.0)
    conv = fast_convolve(a)
    M = conv.max()
    g = solve_lp_for_g(a, M)
    g = normalize_to_sum(g, 1.0)
    return a, g, M


def lp_based_optimization(a, time_limit, verbose=False):
    """LP-based iterative flattening with adaptive blending."""
    n = len(a)
    a = normalize_to_sum(a.copy(), 1.0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iteration = 0
    t_blend = 0.3
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        a, g, M = lp_iteration(a)

        # Try multiple blend factors
        best_candidate = None
        best_cand_score = best_score
        for t in [t_blend, t_blend * 0.5, t_blend * 1.5, 0.1, 0.5]:
            t = min(0.9, max(0.05, t))
            f_new = (1 - t) * a + t * g
            f_new = normalize_to_sum(f_new, 1.0)
            sc = compute_score(f_new)
            if sc < best_cand_score:
                best_cand_score = sc
                best_candidate = f_new.copy()

        if best_candidate is not None:
            best_score = best_cand_score
            best_a = best_candidate.copy()
            a = best_candidate
            stagnation = 0
            t_blend = min(0.6, t_blend * 1.02)
        else:
            stagnation += 1
            t_blend = max(0.05, t_blend * 0.9)
            if stagnation > 30:
                # Restart from best with perturbation
                a = normalize_to_sum(best_a + np.random.standard_normal(n) * 0.005, 1.0)
                stagnation = 0
                t_blend = 0.3

        if verbose and iteration % 10 == 0:
            print(f"    LP iter {iteration}: score={best_score:.10f}, t={t_blend:.3f}, stag={stagnation}")

        if iteration > 500:
            break

    return best_a, best_score


def multiplicative_weight_descent(a, time_limit, verbose=False):
    """MWD with adaptive peak targeting."""
    n = len(a)
    a = normalize_to_sum(a.copy(), 1.0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    lr0 = 0.4
    decay = 0.9995
    cur_lr = lr0

    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)
        max_conv = conv.max()

        # Find active lags (top peaks)
        tol = max_conv * 0.03
        active = np.where(conv >= max_conv - tol)[0].astype(int)
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient
        sg = np.zeros(n)
        for k in active:
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            for i in range(i_min, i_max + 1):
                term = a[k - i]
                ik = i - k
                if 0 <= ik < n and ik != k - i:
                    term += a[ik]
                sg[i] += term
        sg /= len(active)

        # Normalize subgradient for stability
        sg_max = sg.max()
        if sg_max > 1e-15:
            sg = sg / sg_max

        # Multiplicative update
        a = a * np.exp(-cur_lr * sg)
        a = np.maximum(a, 1e-15)
        a = normalize_to_sum(a, 1.0)

        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        cur_lr *= decay
        if cur_lr < 1e-4:
            # Reset with perturbation from best
            cur_lr = lr0 * 0.3
            a = normalize_to_sum(best_a + np.random.standard_normal(n) * 0.003, 1.0)

        if verbose and step % 200 == 0:
            print(f"    MWD step {step}: score={best_score:.10f}, lr={cur_lr:.5f}, |A|={len(active)}")

    return best_a, best_score


def simulated_annealing(a, time_limit, verbose=False):
    """SA with peak-aware moves and block operations."""
    n = len(a)
    a = normalize_to_sum(a.copy(), 1.0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    T = 0.05
    cooling = 0.9997
    step = 0
    n_acc = 0
    n_rej = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and T > 1e-5:
        step += 1
        move_type = rng.random()

        if move_type < 0.35:
            # Pair transfer
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            delta = a[i] * rng.uniform(0.01, 0.3)
            delta = min(delta, a[i] * 0.8)
            if delta < 1e-15:
                continue
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        elif move_type < 0.55:
            # Block redistribution
            bs = int(rng.integers(3, min(15, n // 5)))
            start = int(rng.integers(0, n - bs))
            total = a[start:start + bs].sum()
            if total < 1e-15:
                continue
            new_block = rng.dirichlet(np.ones(bs) * 1.5) * total
            new_a = a.copy()
            new_a[start:start + bs] = new_block
        elif move_type < 0.8:
            # Peak-aware: reduce contribution to argmax lag
            conv = fast_convolve(a)
            k_peak = int(np.argmax(conv))
            i_min = max(0, k_peak - (n - 1))
            i_max = min(k_peak, n - 1)
            if i_min > i_max:
                continue
            i_vals = np.arange(i_min, i_max + 1)
            j_vals = k_peak - i_vals
            contribs = a[i_vals] * a[j_vals]
            tot_c = contribs.sum()
            if tot_c < 1e-15:
                continue
            probs = contribs / tot_c
            idx = int(rng.choice(len(i_vals), p=probs))
            i_s = int(i_vals[idx])
            j_s = int(j_vals[idx])
            # Reduce the larger, add to random
            bigger = i_s if a[i_s] >= a[j_s] else j_s
            delta = a[bigger] * rng.uniform(0.03, 0.15)
            new_a = a.copy()
            new_a[bigger] -= delta
            rand_idx = int(rng.integers(0, n))
            new_a[rand_idx] += delta
        else:
            # Anti-correlation move: move mass from a[i] to a[n-1-i]
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            if a[i] < 1e-15:
                continue
            delta = a[i] * rng.uniform(0.05, 0.3)
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = np.maximum(new_a, 0.0)
        new_a = normalize_to_sum(new_a, 1.0)
        new_sc = compute_score(new_a)
        cur_sc = compute_score(a)
        delta_sc = new_sc - cur_sc

        if delta_sc < 0 or (T > 1e-8 and rng.random() < math.exp(-delta_sc / T)):
            a = new_a
            n_acc += 1
            if new_sc < best_score:
                best_score = new_sc
                best_a = new_a.copy()
        else:
            n_rej += 1

        T *= cooling

        if verbose and step % 2000 == 0:
            print(f"    SA step {step}: T={T:.6f}, score={best_score:.10f}, acc/n={n_acc}/{n_acc+n_rej}")

    return best_a, best_score


def coordinate_descent_refined(a, time_limit, verbose=False):
    """CD with 1D golden-section and local spread moves."""
    n = len(a)
    a = normalize_to_sum(a.copy(), 1.0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]

            # Golden section on single coordinate
            lo = 0.0
            hi = min(0.3, max(old_val * 3.0, 0.01))

            def obj_cd(v):
                new_a = a.copy()
                new_a[j] = v
                new_a = np.maximum(new_a, 0)
                s = new_a.sum()
                if s < 1e-15:
                    return float('inf')
                new_a = new_a / s
                return fast_convolve(new_a).max()

            try:
                res = minimize_scalar(obj_cd, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-7, 'maxiter': 20})
                if res.success:
                    sc_new = obj_cd(res.x)
                    sc_old = obj_cd(old_val)
                    if sc_new < sc_old - 1e-10:
                        new_a = a.copy()
                        new_a[j] = res.x
                        new_a = np.maximum(new_a, 0)
                        new_a = normalize_to_sum(new_a, 1.0)
                        sc = compute_score(new_a)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best_a = new_a.copy()
                            a = new_a
                            improved = True
            except Exception:
                pass

            # Spread move: redistribute j's mass to neighbors
            if old_val > 1e-8 and not improved:
                for spread_n in [1, 2, 3]:
                    neighbors = [x for x in range(max(0, j - spread_n), min(n, j + spread_n + 1)) if x != j]
                    if not neighbors:
                        continue
                    mass = old_val * 0.4
                    new_a = a.copy()
                    new_a[j] = old_val - mass
                    per = mass / len(neighbors)
                    for nb in neighbors:
                        new_a[nb] += per
                    new_a = np.maximum(new_a, 0)
                    new_a = normalize_to_sum(new_a, 1.0)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = new_a.copy()
                        a = new_a
                        improved = True
                        break

        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")
        if not improved and pass_count > 3:
            break
        if pass_count > 80:
            break

    return best_a, best_score

    t0 = time.monotonic()

        if verbose:
            print(f"      MWD: {best_score:.10f}")

    # Stage 2: LP
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.35, 300)
    if rem > t2 + 20:
        a_l, sc_l = lp_based_optimization(a, time_limit=t2, verbose=verbose)
        if sc_l < best_score:

        if verbose:
            print(f"      LP: {best_score:.10f}")

    # Stage 3: SA

    t3 = min(rem * 0.25, 150)
    if rem > t3 + 20:
        a_s, sc_s = simulated_annealing(a, time_limit=t3, verbose=verbose)

def interpolate_sequence(f, new_n):
    """Resample sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize_to_sum(f_interp, 1.0)


def geometric_sequence(n, r):
    """Monotonically decreasing geometric sequence."""
    a = r ** np.arange(n, dtype=np.float64)
    return normalize_to_sum(a, 1.0)


def linear_decreasing(n):
    """Linearly decreasing sequence."""
    a = np.linspace(1.0, 0.01, n)
    return normalize_to_sum(a, 1.0)


def power_decreasing(n, p):
    """Power-law decreasing sequence."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return normalize_to_sum(a, 1.0)


def shifted_peak(n, c_frac, width):
    """Gaussian peak at c_frac * n with given width."""
    c = c_frac * n
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-(x - c) ** 2 / (2 * width ** 2))
    return normalize_to_sum(a, 1.0)


def two_peak_anti(n, gap_frac):
    """Two peaks separated, promoting anti-correlation at center lag."""
    a = np.zeros(n)
    c1 = 0.15 * n
    c2 = (1.0 - gap_frac) * n
    w = n * 0.08
    for i in range(n):
        a[i] = max(a[i], np.exp(-(i - c1) ** 2 / (2 * w ** 2)))
        a[i] = max(a[i], np.exp(-(i - c2) ** 2 / (2 * w ** 2)))
    return normalize_to_sum(a, 1.0)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-based flattening, MWD, SA, and CD.
    Explores diverse non-symmetric initializations to exploit the fact that
    non-symmetric sequences can achieve center-lag auto-correlation below 1/n.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Generate diverse starting points ===
    print("=== Generating starting points ===")
    starts = []

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize_to_sum(p, 1.0)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Non-symmetric sequences (key: these can beat score 2)
    for n in [512, 640, 768, 896, 1024]:
        for r in [0.995, 0.997, 0.998, 0.999, 0.9995]:
            starts.append((geometric_sequence(n, r), f"geom_n{n}_r{r}"))
        starts.append((linear_decreasing(n), f"lindec_n{n}"))
        for p in [1.5, 2.0, 2.5, 3.0]:
            starts.append((power_decreasing(n, p), f"pow_n{n}_p{p}"))
        # Off-center peaks
        for cf in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
            starts.append((shifted_peak(n, cf, n * 0.25), f"peak_n{n}_c{cf}"))
        # Two-peak anti
        for gf in [0.3, 0.5, 0.6]:
            starts.append((two_peak_anti(n, gf), f"anti_n{n}_g{gf}"))

    # Random smooth
    for n in [640, 768, 896]:
        a = rng.random(n)
        ks = max(3, int(n * 0.04))
        kernel = np.ones(ks) / ks
        a = np.convolve(a, kernel, mode='same')
        starts.append((normalize_to_sum(a, 1.0), f"rsm_n{n}"))

    # Evaluate all
    print(f"  Evaluating {len(starts)} starts...")
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    for i, (sc, a, label) in enumerate(start_scores[:10]):
        print(f"    #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # === Phase 1: Optimize top diverse starts ===
    print(f"\n=== Phase 1: Optimize (best={best_score:.8f}) ===")

    selected = []
    seen_types = set()
    for sc, a, label in start_scores:
        if len(selected) >= 5:
            break
        # Extract base type for diversity
        base = label.split('_n')[0] if '_n' in label else label
        if base not in seen_types:
            selected.append((a, label))
            seen_types.add(base)

    # Ensure parent included
    if not any(l == "parent" for _, l in selected):
        for sc, a, label in start_scores:
            if label == "parent":
                selected.append((a, label))
                break

    n_sel = len(selected)
    t_per = min((deadline - time.monotonic() - 250) / max(1, n_sel), 280)

    for a_init, label in selected:
        rem = deadline - time.monotonic() - 180
        if rem < 60:
            break
        t_alloc = min(t_per, rem)
        print(f"  Opt [{label}]: t={t_alloc:.0f}s")
        a_opt, sc_opt = optimize_full(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_opt")

    # === Phase 2: Length exploration ===
    print(f"\n=== Phase 2: Lengths (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        for tn in [512, 640, 768, 896, 1024, 1280]:
            if tn == best_n:
                continue
            rem = deadline - time.monotonic() - 200
            if rem < 50:
                break
            t_alloc = min(rem * 0.35, 120)
            a_interp = interpolate_sequence(best_a, tn)
            a_opt, sc = optimize_full(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"dim{tn}")

    # === Phase 3: Intensive final optimization ===
    print(f"\n=== Phase 3: Final (best={best_score:.8f}) ===")
    if best_a is not None:
        # LP intensive
        rem = deadline - time.monotonic() - 100
        if rem > 60:
            t_lp = min(rem * 0.4, 250)
            print(f"  LP: t={t_lp:.0f}s")
            a_l, sc_l = lp_based_optimization(best_a, time_limit=t_lp, verbose=True)
            try_seq(a_l, "finLP")

        # SA
        rem = deadline - time.monotonic() - 80
        if rem > 50:
            t_sa = min(rem * 0.35, 150)
            a_s, sc_s = simulated_annealing(best_a, time_limit=t_sa, verbose=True)
            try_seq(a_s, "finSA")

        # CD
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t_cd = min(rem * 0.7, 120)
            a_c, sc_c = coordinate_descent_refined(best_a, time_limit=t_cd, verbose=True)
            try_seq(a_c, "finCD")

    # === Phase 4: Perturbation restarts ===
    print(f"\n=== Phase 4: Perturb (best={best_score:.8f}) ===")
    if best_a is not None:
        nb = len(best_a)
        for trial in range(4):
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            scales = [0.008, 0.004, 0.015, 0.002]
            noise = rng.standard_normal(nb) * scales[trial]
            a_pert = normalize_to_sum(best_a + noise, 1.0)
            t_alloc = min(rem * 0.6, 80)
            a_opt, sc = optimize_full(a_pert, time_limit=t_alloc)
            try_seq(a_opt, f"pert{trial}")

    # === Final ===
    if best_a is None:
        best_a = geometric_sequence(768, 0.999)
        best_score = compute_score(best_a)

    final_list = [float(min(max(x, 0.0), 1000.0)) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: {best_score:.8f} (n={len(best_a)})")
    return final_list
