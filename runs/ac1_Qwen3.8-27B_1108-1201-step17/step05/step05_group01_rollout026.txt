"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA, KL mirror descent,
and structured perturbation search with adaptive smoothing.
Key improvements over previous approach:
1. FISTA (Nesterov acceleration) for faster smooth optimization
2. KL mirror descent for better simplex geometry handling
3. Structured sinusoidal perturbations for diverse exploration
4. Aggressive beta annealing with restarts
5. Multi-length interpolation with optimization
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_and_grad(a, beta):
    """Compute log-sum-exp smoothed objective and gradient."""
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    logsumexp = conv_max + (max_shifted + np.log(exp_shifted.sum())) / beta
    w = exp_shifted / exp_shifted.sum()
    
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * R
    return logsumexp, grad

    """FISTA with adaptive beta annealing on the smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:

def kl_mirror_descent(a0, time_limit, verbose=False):
    """KL mirror descent (equivalent to multiplicative updates) with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = 0.1
    momentum = 0.5
    v = np.zeros(n)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        conv_max = conv.max()
        
        # Temperature for the subgradient concentration
        temp = max(0.001, (conv_max - conv.min()) / (2 * n) * 50)
        
        # Soft subgradient: weight each position by its relative magnitude
        shifted = (conv - conv.min()) / temp
        max_s = shifted.max()
        w = np.exp(shifted - max_s)
        w /= w.sum()
        
        # Compute subgradient (cross-correlation)
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        sg = 2.0 * R
        
        # Momentum
        v = momentum * v + (1 - momentum) * sg
        
        # KL mirror descent update: a_new[i] = a[i] * exp(-eta * v[i])
        # This is equivalent to a_new = softmax(log(a) - eta * v)
        log_a = np.log(a)
        a_new = np.exp(log_a - eta * v)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 100 == 0:
                print(f"    KL-MD: {best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1
            a = a_new  # still move
            eta *= 0.98
            if no_improve > 200:
                # Restart with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                eta = 0.05
                no_improve = 0
        
        eta = max(eta, 0.001)
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_step(f, M_scale=1.0):
    """One LP-FW step."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max() * M_scale
    
    g = lp_direction(f, M, n)
    if g is None:
        return None
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    g = g / g_sum
    return g

        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, beta):
    """Compute log-sum-exp smoothed objective and gradient."""
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    lse = conv_max + (max_shifted + np.log(exp_shifted.sum())) / beta
    w = exp_shifted / exp_shifted.sum()
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * R
    return lse, grad


def fista_optimize(a0, time_limit, max_beta=4096, verbose=False):
    """FISTA with aggressive beta annealing on the smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta_schedule = [10, 20, 50, 100, 200, 400, 800, 1600, 3200, 4096]

    for bi, beta in enumerate(beta_schedule):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        # Allocate more time to higher betas
        if bi < len(beta_schedule) - 3:
            sub_time = remaining * 0.15
        else:
            sub_time = remaining / (len(beta_schedule) - bi)

        y = a.copy()
        v = a.copy()
        t_k = 1.0
        L = 8.0 * beta
        patience = 0

        for it in range(400):
            if time.monotonic() - t0 > time_limit - 5:
                break

            _, grad_y = smooth_obj_and_grad_fista(y, beta, n)

            # Backtracking line search
            step = 1.0 / L
            accepted = False
            for bt in range(20):
                a_new = y - step * grad_y
                a_new = project_simplex(a_new)
                obj_new_smooth, _ = smooth_obj_grad(a_new, beta)
                obj_y_smooth, _ = smooth_obj_grad(y, beta)
                if obj_new_smooth <= obj_y_smooth + 0.5 * step * np.dot(grad_y, a_new - y):
                    accepted = True
                    break
                L *= 1.5
                step = 1.0 / L

            if not accepted:
                a_new = project_simplex(y - (1.0 / L) * grad_y)

            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            v_new = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
            v_new = project_simplex(v_new)

            a = a_new
            y = v_new
            t_k = t_new

            actual_obj = fast_convolve(a).max()
            if actual_obj < best_obj - 1e-13:
                best_obj = actual_obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            if patience > 100:
                break

        if verbose:
            print(f"    FISTA beta={beta}: obj={best_obj:.10f}")

    return best_a, best_obj


def smooth_obj_and_grad_fista(y, beta, n):
    """Helper for FISTA: compute smoothed objective and gradient."""
    return smooth_obj_grad(y, beta)


def adam_logsoft_optimize(a0, time_limit, verbose=False):
    """Adam optimizer in log-softmax parameterization."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t = np.log(a + 1e-15)
    # Center t for numerical stability
    t = t - t.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    t0 = time.monotonic()
    step_count = 0

    # Anneal the smoothing parameter
    beta_start = 20.0
    beta_end = 2048.0
    total_planned = max(1, int(time_limit / 0.5))  # estimate total steps

    while time.monotonic() - t0 < time_limit - 5:
        step_count += 1
        progress = min(1.0, step_count / max(1, total_planned // 2))
        beta = beta_start * (beta_end / beta_start) ** progress

        a_curr = softmax_fn(t)

        # Compute gradient in t-space
        conv = fast_convolve(a_curr)
        conv_max = conv.max()
        shifted = beta * (conv - conv_max)
        max_sh = shifted.max()
        exp_sh = np.exp(shifted - max_sh)
        w = exp_sh / exp_sh.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(a_curr, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        # Gradient in t-space: grad_t[i] = a[i] * (grad_a[i] - sum(a*grad_a))
        dot = np.dot(a_curr, grad_a)
        grad_t = a_curr * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_t
        v = beta2 * v + (1 - beta2) * grad_t * grad_t
        m_hat = m / (1 - beta1 ** step_count)
        v_hat = v / (1 - beta2 ** step_count)
        t = t - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodically recenter t
        if step_count % 500 == 0:
            t = t - t.mean()

        # Check improvement
        if step_count % 10 == 0:
            a_check = softmax_fn(t)
            obj = fast_convolve(a_check).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_check.copy()
                if verbose and step_count % 500 == 0:
                    print(f"    Adam-LS step={step_count}, beta={beta:.0f}: {best_obj:.10f}")

        # Adaptive learning rate
        if step_count % 500 == 0:
            lr = max(0.001, lr * 0.85)

    return best_a, best_obj


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit - 5:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.08, 5.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 100, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def symmetric_fista(a0, time_limit, verbose=False):
    """FISTA optimization exploiting symmetry a[j] = a[n-1-j]."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    # Enforce symmetry
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_sym[i] = val
        a_sym[n - 1 - i] = val
    a_sym = a_sym / a_sym.sum()

    best_obj = fast_convolve(a_sym).max()
    best_a = a_sym.copy()

    t0 = time.monotonic()
    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000]

    for beta in beta_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        y = a_sym.copy()
        v = a_sym.copy()
        t_k = 1.0
        L = 8.0 * beta

        for it in range(200):
            if time.monotonic() - t0 > time_limit - 5:
                break

            _, grad_full = smooth_obj_grad(y, beta)
            # Symmetrize gradient
            grad_sym = np.zeros(n)
            for i in range(half):
                val = (grad_full[i] + grad_full[n - 1 - i]) / 2.0
                grad_sym[i] = val
                grad_sym[n - 1 - i] = val

            step = 1.0 / L
            a_new = y - step * grad_sym
            a_new = np.maximum(a_new, 0.0)
            # Enforce symmetry
            for i in range(half):
                val = (a_new[i] + a_new[n - 1 - i]) / 2.0
                a_new[i] = val
                a_new[n - 1 - i] = val
            a_new = a_new / a_new.sum()

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            v_new = a_new + ((t_k - 1.0) / t_new) * (a_new - a_sym)
            v_new = np.maximum(v_new, 0.0)
            for i in range(half):
                val = (v_new[i] + v_new[n - 1 - i]) / 2.0
                v_new[i] = val
                v_new[n - 1 - i] = val
            v_new = v_new / v_new.sum()

            a_sym = a_new
            y = v_new
            t_k = t_new

            actual_obj = fast_convolve(a_sym).max()
            if actual_obj < best_obj - 1e-13:
                best_obj = actual_obj
                best_a = a_sym.copy()

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_perturb_optimize(a0, time_limit, verbose=False):
    """Try structured sinusoidal and polynomial perturbations with optimization."""
    n = len(a0)
    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    x = np.arange(n, dtype=np.float64)
    frequencies = [1, 2, 3, 4, 5, 7, 8, 11, 13]
    amplitudes = [0.005, 0.01, 0.02, 0.04, 0.06]

    attempts = 0
    per_attempt_time = max(15, time_limit // 25)

    while time.monotonic() - t0 < time_limit - 10 and attempts < 25:
        attempts += 1
        f = int(rng.choice(frequencies))
        amp = float(rng.choice(amplitudes))
        phase = float(rng.uniform(0, 2 * np.pi))

        # Sinusoidal perturbation
        perturbation = amp * np.cos(2 * np.pi * f * x / n + phase)
        a_pert = a0 + perturbation
        a_pert = np.maximum(a_pert, 0)
        ss = a_pert.sum()
        if ss < 1e-10:
            continue
        a_pert = a_pert / ss

        # Quick FISTA
        a_opt, obj = fista_optimize(a_pert, time_limit=per_attempt_time, max_beta=1024, verbose=False)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_opt
            if verbose:
                print(f"    Structured f={f}, amp={amp:.3f}: {best_obj:.10f}")

        # Also try polynomial perturbation
        if attempts % 3 == 0:
            coeff = float(rng.normal(0, 0.01))
            xn = x / n - 0.5
            poly_pert = coeff * (4 * xn * xn - 1)
            a_pert2 = a0 + poly_pert
            a_pert2 = np.maximum(a_pert2, 0)
            ss2 = a_pert2.sum()
            if ss2 > 1e-10:
                a_pert2 = a_pert2 / ss2
                a_opt2, obj2 = fista_optimize(a_pert2, time_limit=per_attempt_time // 2, max_beta=512, verbose=False)
                if obj2 < best_obj - 1e-10:
                    best_obj = obj2
                    best_a = a_opt2
                    if verbose:
                        print(f"    Poly pert: {best_obj:.10f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining FISTA, Adam-LogSoftmax, LP-FW,
    symmetric optimization, structured perturbations, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0 or a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # === Phase 1: FISTA on parent (aggressive beta) ===
    print("\n=== Phase 1: FISTA on parent ===")
    rem = deadline - time.monotonic() - (budget_s * 0.55)
    t1 = min(rem, 350)
    if t1 > 30:
        a_opt, _ = fista_optimize(parent, time_limit=t1, max_beta=4096, verbose=True)
        try_seq(a_opt, "fista_main")

    # === Phase 2: Adam in log-softmax space ===
    print(f"\n=== Phase 2: Adam-LogSoftmax (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - (budget_s * 0.42)
    t2 = min(rem, 250)
    if t2 > 30:
        a_opt, _ = adam_logsoft_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "adam_ls")

    # === Phase 3: Symmetric optimization ===
    print(f"\n=== Phase 3: Symmetric FISTA (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - (budget_s * 0.32)
    t3 = min(rem, 200)
    if t3 > 20:
        a_opt, _ = symmetric_fista(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "symmetric")

    # === Phase 4: Structured perturbations ===
    print(f"\n=== Phase 4: Structured Perturbations (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - (budget_s * 0.22)
    t4 = min(rem, 250)
    if t4 > 30:
        a_opt, _ = structured_perturb_optimize(base, time_limit=t4, verbose=True)
        try_seq(a_opt, "structured")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = sorted(set([
        max(64, n_parent - 128),
        n_parent,
        n_parent + 64,
        n_parent + 128,
        n_parent + 256,
        n_parent * 3 // 2,
    ]))

    for target_n in target_lengths:
        if target_n < 64 or target_n > 5000:
            continue
        if target_n == n_parent:
            continue
        rem = deadline - time.monotonic() - (budget_s * 0.12)
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.35, 70)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = fista_optimize(f_interp, time_limit=t_alloc, max_beta=2048, verbose=False)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 6: LP-FW refinement ===
    print(f"\n=== Phase 6: LP-FW Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - (budget_s * 0.08)
        if rem > 30:
            t_alloc = min(rem, 200)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_refine")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate methods with small perturbations
        for attempt in range(10):

            if rem < 15:

                # FISTA from perturbed start
                noise = rng.standard_normal(n_best) * 0.002
                f_start = project_simplex(best_seq + noise)

                a_opt, _ = fista_optimize(f_start, time_limit=t_alloc, max_beta=2048, verbose=False)
                try_seq(a_opt, f"polish_fista_{attempt}")
            elif attempt % 3 == 1:
                # LP-FW from current best
                t_alloc = min(rem * 0.5, 50)
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, f"polish_lpfw_{attempt}")
            else:
                # Adam from current best
                t_alloc = min(rem * 0.5, 50)
                a_opt, _ = adam_logsoft_optimize(best_seq, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, f"polish_adam_{attempt}")

        # One final FISTA push with maximum beta
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            a_opt, _ = fista_optimize(best_seq, time_limit=min(rem, 40), max_beta=8192, verbose=False)
            try_seq(a_opt, "final_push")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
