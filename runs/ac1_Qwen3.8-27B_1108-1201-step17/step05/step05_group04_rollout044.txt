import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lp_direction(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    g = cp.Variable(n)
    
    # Objective: maximize sum(g)
    objective = cp.Maximize(cp.sum(g))
    
    # Constraints: g >= 0, conv(f, g)[k] <= M for all k
    constraints = [g >= 0]
    
    # Build the constraint matrix A_k such that (f*g)[k] = A_k @ g <= M
    # (f*g)[k] = sum_i f[i] * g[k-i] where 0 <= k-i < n
    # So (f*g)[k] = sum_{i: 0<=k-i<n} f[i] * g[k-i]
    # = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j]
    # Let j = k-i, so i = k-j
    # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j]
    
    for k in range(2*n - 1):
        # (f*g)[k] = sum_j f[k-j] * g[j] for valid j
        # j ranges from max(0, k-(n-1)) to min(n-1, k)
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        # coefficients for g[j]
        coeffs = np.zeros(n)
        for j in range(j_min, j_max + 1):
            coeffs[j] = f[k - j]
        constraints.append(coeffs @ g <= M)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL)
    
    if g.value is None:
        return None
    return np.maximum(g.value, 0)

def line_search(f, g, conv_f, conv_fg, conv_g):
    """Find best t in [0,1] for f_new = (1-t)*f + t*g."""
    def obj(t):
        conv_new = (1-t)**2 * conv_f + 2*t*(1-t) * conv_fg + t**2 * conv_g
        return np.max(conv_new)
    
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(obj, bounds=(0, 1), method='bounded')
    return res.x

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n + 1)
    new_x = np.linspace(0, 1, new_n + 1)
    f_padded = np.concatenate([f, [f[-1]]])
    f_interp = np.interp(new_x, old_x, f_padded)[:-1]
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        try_seq(f_opt, "p1_lp_topk")

    # === Phase 2: Subgradient refinement on best ===
    print(f"\n=== Phase 2: Subgradient refinement (best={best_score:.8f}) ===")
    t2 = min(200.0, (deadline - time.monotonic()) * 0.25)
    if best_seq is not None and t2 > 30:
        n_best = len(best_seq)
        f_opt, _ = subgradient_optimize(best_seq, time_limit=t2 * 0.6,
                                        temp_init=0.004, lr=0.002,
                                        top_k=max(20, n_best // 10))
        try_seq(f_opt, "p2_sg")

        remaining = t2 * 0.4
        if remaining > 20 and best_seq is not None:
            perturbed = best_seq + rng.standard_normal(len(best_seq)) * 0.003
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            f_opt, _ = subgradient_optimize(perturbed, time_limit=remaining,
                                            temp_init=0.006, lr=0.004,
                                            top_k=max(30, n_best // 8))
            try_seq(f_opt, "p2_sg_pert")

    # === Phase 3: LP on the refined best ===
    print(f"\n=== Phase 3: LP refinement (best={best_score:.8f}) ===")
    t3 = min(300.0, (deadline - time.monotonic()) * 0.35)
    if best_seq is not None and t3 > 60:
        f_opt, obj = lp_optimize(best_seq, time_limit=t3,
                                 top_k=max(40, len(best_seq) // 4), verbose=True)
        try_seq(f_opt, "p3_lp_refine")

    # === Phase 4: Try different lengths ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    t4 = min(200.0, (deadline - time.monotonic()) - 60)
    if best_seq is not None and t4 > 60:
        n_best = len(best_seq)
        target_lengths = [512, 640, 896, 1024, 1280]
        target_lengths = [t for t in target_lengths if t != n_best and 256 <= t <= 2048]
        target_lengths.sort(key=lambda x: abs(x - n_best))
        target_lengths = target_lengths[:2]

        for target_n in target_lengths:
            remaining = t4 - (time.monotonic() - (deadline - t4 - 60))  # rough
            remaining = min(80.0, (deadline - time.monotonic()) - 60)
            if remaining < 40:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            print(f"  Trying n={target_n}, time={remaining:.0f}s")

            # Quick LP on interpolated
            f_opt, _ = lp_optimize(f_interp, time_limit=remaining * 0.7,
                                   top_k=max(30, target_n // 5), verbose=False)
            try_seq(f_opt, f"p4_n{target_n}")

            # Subgradient polish
            remaining2 = min(30.0, (deadline - time.monotonic()) - 60)
            if remaining2 > 15:
                f_opt, _ = subgradient_optimize(f_opt if f_opt is not None else f_interp,
                                                time_limit=remaining2,
                                                temp_init=0.005, lr=0.003,
                                                top_k=max(20, target_n // 10))
                try_seq(f_opt, f"p4_n{target_n}_sg")

    # === Phase 5: Final fine LP on absolute best ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.8f}) ===")
    t5 = max(30.0, (deadline - time.monotonic()) - 30)
    if best_seq is not None and t5 > 30:
        f_opt, obj = lp_optimize(best_seq, time_limit=t5,
                                 top_k=max(30, len(best_seq) // 5), verbose=True)
        try_seq(f_opt, "p5_final_lp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
