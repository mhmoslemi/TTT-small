"""
Optimized step function search using exact subgradient computation via FFT,
Frank-Wolfe steps, coordinate descent on critical coordinates, and LP-based
direction finding. Key innovations: (1) exact subgradient of max-convolution
computed via cross-correlation FFT, (2) "anti-concentration" initializations
that exploit the pairing structure of convolution peaks, (3) adaptive
coordinate descent focusing on peak-contributing coordinates, (4) multi-length
exploration with progressive refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, S_indicator):
    """
    Compute subgradient of max_k conv_k at point a, where S_indicator is
    the indicator of the active set (length 2n-1).
    Uses FFT cross-correlation: grad[i] = 2 * sum_j a[j] * S[i+j]
    """
    n = len(a)
    conv_len = 2 * n - 1
    N = 1 << (n + conv_len - 1).bit_length()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    S_pad = np.zeros(N)
    S_pad[:conv_len] = S_indicator
    A = np.fft.rfft(a_pad)
    S = np.fft.rfft(S_pad)
    corr = np.fft.irfft(A * np.conj(S), N)
    return 2.0 * corr[:n]


def get_active_set(conv, threshold_frac=1.0):
    """Get indicator of active set (where conv is near max)."""
    M = conv.max()
    if M <= 0:
        return np.ones(len(conv))
    threshold = M * threshold_frac
    return (conv >= threshold).astype(np.float64)


def subgradient_fw_step(a, lr, threshold_frac=1.0):
    """One Frank-Wolfe step using subgradient direction."""
    n = len(a)
    conv = fast_convolve(a)
    S_ind = get_active_set(conv, threshold_frac)
    g = compute_subgradient(a, S_ind)
    # FW: move towards e_i where i = argmin g
    i_min = int(np.argmin(g))
    # Line search
    best_t = 0.0
    best_val = conv.max()
    for t in [lr * 0.1, lr * 0.3, lr * 0.5, lr * 0.7, lr]:
        a_new = (1 - t) * a + t * np.zeros(n)
        a_new[i_min] += t
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        val = fast_convolve(a_new).max()
        if val < best_val:
            best_val = val
            best_t = t
    if best_t > 1e-10:
        a_new = (1 - best_t) * a + best_t * np.zeros(n)
        a_new[i_min] += best_t
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        return a_new, best_val
    return a, best_val


def smoothed_subgradient_step(a, lr, T, threshold_frac=0.95):
    """Step using smoothed (softmax-weighted) subgradient."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    # Softmax weights over convolution values
    scaled = (conv - M) / T
    w = np.exp(scaled)
    w = w / w.sum()
    # Weighted subgradient
    N = 1 << (n + len(conv) - 1).bit_length()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    W_pad = np.zeros(N)
    W_pad[:len(conv)] = w
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(W_pad)
    corr = np.fft.irfft(A * np.conj(W), N)
    g = 2.0 * corr[:n]
    # Gradient step with projection
    a_new = a - lr * g
    a_new = project_simplex(a_new)
    return a_new, fast_convolve(a_new).max()


def coordinate_descent_critical(a, time_limit, top_k=50, verbose=False):
    """Coordinate descent focusing on coordinates that contribute most to peak."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find peak location(s)
        peak_idx = int(np.argmax(conv))
        
        # Find coordinates with highest contribution to peak
        # Contribution of a[i] to conv[peak_idx] is a[i]*a[peak_idx-i] (if valid)
        contributions = np.zeros(n)
        valid = (peak_idx >= np.arange(n)) & (peak_idx - np.arange(n) < n)
        contrib_vals = np.where(valid, a * a[peak_idx - np.arange(n)], 0.0)
        
        # Also consider near-peak convolutions
        top_conv_indices = np.argsort(conv)[::-1][:5]
        for ki in top_conv_indices:
            valid_k = (ki >= np.arange(n)) & (ki - np.arange(n) < n)
            contrib_vals += np.where(valid_k, a * a[ki - np.arange(n)], 0.0) * 0.5
        
        # Get top-k coordinates by contribution
        top_coords = np.argsort(contrib_vals)[::-1][:top_k]
        
        improved = False
        ORDER = 0.6180339887
        
        for j in top_coords:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            # Golden section search on a[j], scaling others proportionally
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            lo = 0.0
            hi = min(0.999, old_val * 3 + 0.1)
            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)
            
            def eval_at(x):
                scale = (1.0 - x) / denom
                a_trial = a.copy()
                a_trial[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                return fast_convolve(a_trial).max()
            
            fc = eval_at(c)
            fd = eval_at(d)
            
            for _ in range(10):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = eval_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = eval_at(d)
            
            opt_val = (lo + hi) / 2.0
            opt_obj = eval_at(opt_val)
            
            if opt_obj < best_obj - 1e-14:
                scale = (1.0 - opt_val) / denom
                a[j] = opt_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                best_obj = opt_obj
                best_a = a.copy()
                improved = True
        
        if not improved and passes > 2:
            break
    
    return best_a, best_obj


def lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = lp_direction(a, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = a + alpha * (g - a)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Refine with bisection
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = a + mid * (g - a)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            a_new = a + best_alpha * (g - a)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            conv = fast_convolve(a_new)
            M = conv.max()
            m_exp = 1.0
            a = a_new
            no_improve = 0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)
        
        if no_improve > 80:
            # Perturb and restart
            rng_p = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min(no_improve // 50, 5)
            noise = rng_p.standard_normal(n) * scale
            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_a, best_obj


def smooth_optimize(a0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Smooth log-sum-exp optimization with Adam, using exact FFT gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        T = T_start * (T_end / T_start) ** (elapsed / time_limit)
        T = max(T, T_end)
        lr = alpha * (0.5 ** (elapsed / (time_limit * 0.7)))
        
        # Compute smooth gradient
        conv = fast_convolve(a)
        M = conv.max()
        scaled = (conv - M) / T
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Cross-correlation for gradient
        N = 1 << (n + len(conv) - 1).bit_length()
        a_pad = np.zeros(N)
        a_pad[:n] = a
        W_pad = np.zeros(N)
        W_pad[:len(conv)] = w
        A = np.fft.rfft(a_pad)
        W = np.fft.rfft(W_pad)
        corr = np.fft.irfft(A * np.conj(W), N)
        grad = 2.0 * corr[:n]
        
        # Adam update
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)
        
        cur_obj = fast_convolve(a).max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Smooth iter {it}: obj={best_obj:.10f}, T={T:.5f}")
        else:
            no_improve += 1
        
        if no_improve > 2000:
            seed_val = int(elapsed * 1000) % 1000000
            rng_p = np.random.default_rng(seed_val)
            a = project_simplex(best_a + rng_p.standard_normal(n) * 0.003)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_a, best_obj


def fw_subgradient_optimize(a0, time_limit, threshold_frac=0.9, verbose=False):
    """Frank-Wolfe with exact subgradient, multiple step sizes."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    base_lr = 1.0 / np.sqrt(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        S_ind = get_active_set(conv, threshold_frac)
        g = compute_subgradient(a, S_ind)
        
        # FW step: move towards argmin gradient
        i_min = int(np.argmin(g))
        lr = base_lr / np.sqrt(1 + it * 0.01)
        
        best_t = 0.0
        best_val = M
        for t in [lr * 0.3, lr * 0.5, lr * 0.7, lr, lr * 1.5, lr * 2.0]:
            a_new = a.copy()
            a_new[i_min] += t
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_t = t
        
        if best_t > 1e-10:
            a_new = a.copy()
            a_new[i_min] += best_t
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            no_improve += 1
        
        if verbose and it % 200 == 0:
            print(f"    FW-SG iter {it}: obj={best_obj:.10f}")
        
        if no_improve > 300:
            rng_p = np.random.default_rng(it * 17 + 3)
            a = project_simplex(best_a + rng_p.standard_normal(n) * 0.004)
            conv = fast_convolve(a)
            no_improve = 0
    
    return best_a, best_obj


def make_anti_concentrate(n, edge_weight=2.0):
    """Sequence with high edges, low middle - anti-concentration pattern."""
    x = np.linspace(0, 1, n)
    # High at edges, low in middle
    a = (1.0 - np.sin(np.pi * x)) ** 2 + 0.01
    a = a / a.sum()
    return a


def make_alternating(n, ratio=0.3):
    """Alternating high/low pattern."""
    a = np.ones(n)
    a[::2] = 1.0
    a[1::2] = ratio
    a = a / a.sum()
    return a


def make_bimodal(n, center_spread=0.2):
    """Bimodal: peaks near edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / center_spread) ** 2) + np.exp(-((x - 0.9) / center_spread) ** 2) + 0.005
    a = a / a.sum()
    return a


def make_geometric(n, r):
    """Geometric sequence."""
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def make_cosine(n, power=2.0):
    """Cosine window."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_optimize(f_pert, time_limit=min(rem, 20),
                                       T_start=0.02, T_end=0.003)
            _try(a_opt, "perturb")
        
        # Reversed
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            f_rev = best_seq[::-1].copy()
            a_opt, _ = smooth_optimize(f_rev, time_limit=min(rem, 15),
                                       T_start=0.02, T_end=0.003)
            _try(a_opt, "reversed")
        
        # Blend with anti-concentrate
        rem = deadline - time.monotonic() - 8
        if rem > 6:
            f_anti = make_anti_concentrate(n_best, 2.0)
            f_blend = 0.85 * best_seq + 0.15 * f_anti
            f_blend = project_simplex(f_blend)
            a_opt, _ = smooth_optimize(f_blend, time_limit=min(rem, 10),
                                       T_start=0.015, T_end=0.002)
            _try(a_opt, "blend_anti")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
