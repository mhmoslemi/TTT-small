def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute the objective 2*n*max(conv)/sum^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_n = a / s
    return 2.0 * len(a_n) * fast_convolve(a_n).max()


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.3):

    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n

def make_cosine_taper(n, plateau_frac=0.6):
    """Cosine-tapered rectangle with flat plateau in the middle."""
    a = np.zeros(n)
    plateau = int(n * plateau_frac)
    start = (n - plateau) // 2
    for i in range(start, start + plateau):
        a[i] = 1.0
    taper = (n - plateau) // 2
    for i in range(taper):
        a[start - 1 - i] = 0.5 * (1 + np.cos(np.pi * (i + 1) / (taper + 1)))
        a[start + plateau + i] = 0.5 * (1 + np.cos(np.pi * (i + 1) / (taper + 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
