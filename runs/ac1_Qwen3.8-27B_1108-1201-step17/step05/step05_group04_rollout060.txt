```python
def active_set_direction(x, R, delta=0.01, max_active=50):
    """Find a descent direction using active-set LP."""
    n = len(x)
    R_max = R.max()
    # Active set: k values where R(k) is close to max
    # R has length 2n-1, indexed 0 to 2n-2, corresponding to lags -(n-1) to n-1
    active = np.where(R >= R_max - delta * R_max)[0]
    if len(active) > max_active:
        # Keep only the top max_active
        vals = R[active]
        top_idx = np.argsort(vals)[::-1][:max_active]
        active = active[top_idx]
    
    # Compute gradients g_k for each active k
    # R(k) = sum_i x_i x_{i+k} for lag k (where lag k means R index = n-1+k)
    # dR(k)/dx_j = x_{j+k} + x_{j-k} (with x_i = 0 for i < 0 or i >= n)
    
    # For lag k (can be negative), the gradient at position j is:
    # if j+k is in [0, n-1]: add x_{j+k}
    # if j-k is in [0, n-1]: add x_{j-k}
    # But we need to be careful about double-counting when k=0: dR(0)/dx_j = 2*x_j
    
    g = np.zeros((len(active), n))
    for ai, ri in enumerate(active):
        k = ri - (n - 1)  # actual lag
        for j in range(n):
            val = 0.0
            jp = j + k
            jm = j - k
            if 0 <= jp < n:
                val += x[jp]
            if 0 <= jm < n and jm != jp:  # avoid double count
                val += x[jm]
            elif k == 0:
                val = 2.0 * x[j]
            g[ai, j] = val
    
    # LP: minimize M
    # s.t. g @ d <= M (for each active constraint)
    #      sum(d) = 0
    #      d_i >= 0 for i where x_i < 1e-10 (boundary)
    
    # Use scipy linprog
    # Variables: [d_0, ..., d_{n-1}, M]
    # Minimize: [0, ..., 0, 1]
    
    # Constraints:
    # g @ d - M <= 0  =>  [g, -1] @ [d, M] <= 0  (|S| inequalities)
    # sum(d) = 0  =>  [1,...,1, 0] @ [d, M] = 0  (1 equality)
    # d_i >= 0 for boundary: -d_i <= 0  =>  [-e_i, 0] @ [d, M] <= 0
    
    boundary = x < 1e-10
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ineq = np.zeros((len(active) + boundary.sum(), n + 1))
    b_ineq = np.zeros(len(active) + boundary.sum())
    
    # Active constraints: g @ d - M <= 0
    A_ineq[:len(active), :n] = g
    A_ineq[:len(active), -1] = -1.0
    
    # Boundary constraints: -d_i <= 0
    for bi, j in enumerate(np.where(boundary)[0]):
        row = len(active) + bi
        A_ineq[row, j] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Solve LP
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, method='highs')
    
    if res.success:
        d = res.x[:n]
        M_val = res.x[-1]
        return d, M_val
    else:
        return None, None
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated smoothed gradient (log-sum-exp)
2. Active-set LP direction refinement
3. Coordinate descent polishing
4. Length exploration

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv_gradient(f):
    """Compute gradients of all R(k) with respect to f.
    Returns array of shape (2n-1, n) where grad[k, j] = dR(k)/df_j.
    """
    n = len(f)

    # R(n-1+k) in the conv array = sum_i f[i] * f[i+k]
    # dR(k)/df_j = f[j+k] + f[j-k] (with f[i]=0 for i<0 or i>=n)

def compute_gradient_at_lags(f, lags):
    """Compute dR(k)/df for specific lags.
    lags: array of lag values (can be negative)
    Returns: array of shape (len(lags), n)
    """
    n = len(f)
    grad = np.zeros((len(lags), n))
    for i, k in enumerate(lags):
        # dR(k)/df[j] = f[j+k] (if valid) + f[j-k] (if valid)
        # For k=0: dR(0)/df[j] = 2*f[j]
        if k == 0:
            grad[i] = 2.0 * f
        else:
            # f[j+k] for j = 0 to n-1-k (if k > 0) or n-1+k to n-1 (if k < 0)
            # f[j-k] for j = k to n-1 (if k > 0) or 0 to n-1+k (if k < 0)
            if k > 0:
                # f[j+k] is valid for j = 0, ..., n-1-k
                grad[i, :n-k] += f[k:]
                # f[j-k] is valid for j = k, ..., n-1
                grad[i, k:] += f[:n-k]

                # f[j-k] = f[j+k_abs] is valid for j = 0, ..., n-1-k_abs
                grad[i, :n-k_abs] += f[k_abs:]
    return grad

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_gradient(f, alpha):
    """Compute gradient of log-sum-exp smoothed autocorrelation objective.
    
    Returns (gradient, softmax_weights, R_array).
    """
    n = len(f)
    R = fast_convolve(f)
    # Softmax weights
    R_shifted = R / alpha
    R_shifted -= R_shifted.max()
    s = np.exp(R_shifted)
    s_sum = s.sum()
    if s_sum < 1e-300:
        s = np.zeros(2 * n - 1)
        s[np.argmax(R)] = 1.0
    else:
        s /= s_sum

    # Compute gradient using only significant softmax values
    s_threshold = s.max() * 1e-8
    significant = np.where(s > s_threshold)[0]

    A = np.zeros(n)
    B = np.zeros(n)
    for k in significant:
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                A[:n - lag] += s[k] * f[lag:]
                B[lag:] += s[k] * f[:n - lag]
        else:
            al = -lag
            al = min(al, n - 1)
            A[al:] += s[k] * f[:n - al]
            B[:n - al] += s[k] * f[al:]

    grad = (A + B) / alpha
    return grad, s, R


def nesterov_optimize(f0, time_limit, alpha_init=0.005, alpha_decay=0.9995,
                      alpha_min=0.0005, lr_init=0.02, verbose=False):
    """Nesterov-accelerated gradient descent on log-sum-exp smoothed objective."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    alpha = alpha_init
    lr = lr_init
    t_nest = 1.0

    best_f = f.copy()
    best_conv = fast_convolve(f)
    best_obj = best_conv.max()

    f_prev = f.copy()
    y = f.copy()  # Nesterov momentum point

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov extrapolation
        beta_nest = (t_nest - 1.0) / (t_nest + 1.0)
        y = f + beta_nest * (f - f_prev)
        f_prev = f.copy()
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = f.copy()
        else:
            y = y / y_sum

        # Compute gradient at y
        grad, _, _ = compute_softmax_gradient(y, alpha)

        # Gradient step + projection
        f_new = project_simplex(y - lr * grad)

        # Check objective
        conv = fast_convolve(f_new)

            no_improve += 1

        f = f_new
        t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))

        # Adapt parameters

def active_set_direction(x, R, delta_frac=0.01, max_active=40):
    """Find descent direction via active-set LP.
    
    Solves: min M s.t. d·g_k <= M for k in S, sum(d)=0, d_i>=0 for boundary.
    Returns (direction, M_value) or (None, None) if no descent direction.
    """
    n = len(x)
    R_max = R.max()
    if R_max < 1e-15:
        return None, None

    # Active set: indices where R is close to max
    threshold = R_max - delta_frac * R_max
    active_idx = np.where(R >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(R)])
    if len(active_idx) > max_active:
        # Keep top max_active
        vals = R[active_idx]
        top = np.argsort(vals)[::-1][:max_active]
        active_idx = active_idx[top]

    # Compute gradients g_k for each active lag
    lags = active_idx - (n - 1)
    g = np.zeros((len(active_idx), n))
    for i, k in enumerate(lags):
        if k == 0:
            g[i] = 2.0 * x
        elif k > 0:
            g[i, :n - k] += x[k:]
            g[i, k:] += x[:n - k]
        else:
            al = -k
            g[i, al:] += x[:n - al]
            g[i, :n - al] += x[al:]

    # Boundary mask (where x is ~0)
    boundary = x < 1e-12
    n_boundary = boundary.sum()

    # LP: variables [d_0,...,d_{n-1}, M], minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_ineq = len(active_idx) + n_boundary
    A_ub = np.zeros((n_ineq, n + 1))
    b_ub = np.zeros(n_ineq)

    # Active constraints: g @ d - M <= 0
    A_ub[:len(active_idx), :n] = g
    A_ub[:len(active_idx), -1] = -1.0

    # Boundary: -d_i <= 0 for i in boundary
    if n_boundary > 0:
        boundary_indices = np.where(boundary)[0]
        for bi, j in enumerate(boundary_indices):
            A_ub[len(active_idx) + bi, j] = -1.0

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True})
        if res.success and res.x[-1] < -1e-10:
            d = res.x[:n]
            return d, res.x[-1]
        elif res.success:
            return None, res.x[-1]
        else:
            return None, None
    except Exception:
        return None, None

    best_obj = R.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = f[j]

            # Try multiplicative adjustments
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = f[j] * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                # Renormalize others
                others_sum = f_sum - f[j]
                if others_sum < 1e-15:
                    continue
                f_trial[np.arange(n) != j] *= (f_sum - new_val) / others_sum
                f_trial = np.maximum(f_trial, 0)
                f_trial_sum = f_trial.sum()
                if f_trial_sum < 1e-15:
                    continue
                f_trial /= f_trial_sum

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != f[j]:
                f[j] = best_j_val
                others_sum = f_sum - f[j]
                if others_sum > 1e-15:
                    f[np.arange(n) != j] *= (f_sum - best_j_val) / others_sum
                f = np.maximum(f, 0)
                f /= f.sum()

                cur_obj = fast_convolve(f).max()
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n + 1)
    old_x[0] = -0.5 / old_n
    old_x[-1] = 1.0 + 0.5 / old_n
    f_padded = np.concatenate([[0], f, [0]])
    new_x = np.linspace(old_x[0], old_x[-1], new_n + 2)
    f_interp = np.interp(new_x[1:-1], old_x, f_padded)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_diverse_inits(n, rng):
    """Create diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Gaussian
    for sigma in [0.1, 0.15, 0.2, 0.3]:
        f = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        f = np.maximum(f, 1e-10)
        inits.append((f'gauss_{sigma}', f / f.sum()))

    # Bump (sin^2)
    f = np.sin(np.pi * x) ** 2
    f = np.maximum(f, 1e-10)
    inits.append(('bump', f / f.sum()))

    # U-shaped (cos^2)
    f = np.cos(np.pi * x) ** 2
    f = np.maximum(f, 1e-10)
    inits.append(('u_shape', f / f.sum()))

    # Edge-concentrated
    f = np.exp(-((x - 0.5) ** 2) / (2 * 0.05 ** 2)) + 0.5
    f = np.maximum(f, 1e-10)
    inits.append(('mixed', f / f.sum()))

    # Random smooth
    for _ in range(3):
        f = np.zeros(n)
        for _c in range(4):
            c = rng.uniform(0.1, 0.9)
            s = rng.uniform(0.05, 0.3)
            a = rng.uniform(0.5, 2.0)
            f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
        f = np.maximum(f, 1e-10)
        inits.append(('rand_smooth', f / f.sum()))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Nesterov smoothed gradient from diverse inits
    Phase 2: Active-set LP refinement
    Phase 3: Coordinate descent polish
    Phase 4: Length exploration
    Phase 5: Final deep refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform init n=768")

    n_parent = len(parent)

    # === Phase 1: Nesterov from diverse inits ===
    print(f"\n=== Phase 1: Nesterov smoothed gradient (n={n_parent}) ===")
    t_phase1_end = time.monotonic() + 300

    inits = make_diverse_inits(n_parent, rng)
    # Add parent and perturbed parent
    inits.append(('parent', parent.copy()))
    inits.append(('parent_pert', project_simplex(parent + rng.standard_normal(n_parent) * 0.01)))

    phase1_best = None
    phase1_best_obj = float('inf')

    for name, init in inits:
        rem = t_phase1_end - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.12, 30)
        if t_alloc < 5:
            break

        print(f"  Nesterov init '{name}', t={t_alloc:.0f}s")
        a_opt, obj = nesterov_optimize(init, time_limit=t_alloc,
                                       alpha_init=0.004, alpha_decay=0.9995,
                                       alpha_min=0.0005, lr_init=0.015)
        if obj < phase1_best_obj:
            phase1_best_obj = obj
            phase1_best = a_opt.copy()
        try_seq(a_opt, f"p1_{name}")

    if phase1_best is not None:
        try_seq(phase1_best, "p1_best")

    base = best_seq if best_seq is not None else parent

    # === Phase 2: Active-set LP refinement ===
    print(f"\n=== Phase 2: Active-set LP refinement (best={best_score:.8f}) ===")
    t_phase2_end = time.monotonic() + 500
    n_base = len(base)

    # Multiple active-set runs with different parameters
    as_configs = [
        {'delta_frac': 0.01, 'max_active': 30},
        {'delta_frac': 0.005, 'max_active': 20},
        {'delta_frac': 0.02, 'max_active': 50},
        {'delta_frac': 0.003, 'max_active': 15},
    ]

    for ci, cfg in enumerate(as_configs):
        rem = t_phase2_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.4, 100)
        print(f"  AS config {ci} (delta={cfg['delta_frac']}, max_act={cfg['max_active']}), t={t_alloc:.0f}s")
        a_opt, obj = active_set_refine(base, time_limit=t_alloc,
                                       delta_frac=cfg['delta_frac'],
                                       max_active=cfg['max_active'])
        try_seq(a_opt, f"p2_as_{ci}")
        base = best_seq if best_seq is not None else base

    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: Coordinate descent (best={best_score:.8f}) ===")
    t_phase3_end = time.monotonic() + 200
    if best_seq is not None:
        rem = t_phase3_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.7, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p3_cd")

            # CD from perturbed start
            rem = t_phase3_end - time.monotonic()
            if rem > 20:
                n_b = len(best_seq)
                perturbed = project_simplex(best_seq + rng.standard_normal(n_b) * 0.003)
                a_opt, obj = coordinate_descent_polish(perturbed, time_limit=min(rem * 0.5, 50))
                try_seq(a_opt, "p3_cd_pert")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + 350
    n_best = len(best_seq) if best_seq is not None else n_parent

    # Try various lengths
    candidate_lengths = [384, 512, 640, 768, 896, 1024, 1152, 1280, 1536]
    candidate_lengths = [t for t in candidate_lengths if t != n_best and 256 <= t <= 2048]
    candidate_lengths.sort(key=lambda x: abs(x - n_best))
    candidate_lengths = candidate_lengths[:5]

    for target_n in candidate_lengths:
        rem = t_phase4_end - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_sequence(best_seq, target_n)

        # Nesterov on interpolated
        t_alloc1 = min(rem * 0.35, 60)
        print(f"  n={target_n}: Nesterov t={t_alloc1:.0f}s")
        a_opt, obj = nesterov_optimize(f_interp, time_limit=t_alloc1,
                                       alpha_init=0.004, alpha_decay=0.9995,
                                       alpha_min=0.0005, lr_init=0.015)
        try_seq(a_opt, f"p4_n{target_n}_nest")

        # Active-set on interpolated
        rem = t_phase4_end - time.monotonic()
        if rem > 30:
            t_alloc2 = min(rem * 0.4, 80)
            a_opt, obj = active_set_refine(a_opt, time_limit=t_alloc2,
                                           delta_frac=0.01, max_active=30)
            try_seq(a_opt, f"p4_n{target_n}_as")

    # === Phase 5: Final deep refinement ===
    print(f"\n=== Phase 5: Final deep refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_final = len(best_seq)
        rem = deadline - time.monotonic() - 20

        # Alternating Nesterov + AS with decreasing alpha
        if rem > 60:
            t_nest = min(rem * 0.3, 60)
            print(f"  Final Nesterov (low alpha), t={t_nest:.0f}s")
            a_opt, obj = nesterov_optimize(best_seq, time_limit=t_nest,
                                           alpha_init=0.002, alpha_decay=0.9998,
                                           alpha_min=0.0002, lr_init=0.008)
            try_seq(a_opt, "p5_fine_nest")

        rem = deadline - time.monotonic() - 15
        if rem > 40:
            t_as = min(rem * 0.4, 60)
            print(f"  Final AS (tight), t={t_as:.0f}s")
            a_opt, obj = active_set_refine(best_seq, time_limit=t_as,
                                           delta_frac=0.003, max_active=20)
            try_seq(a_opt, "p5_fine_as")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for pi in range(4):
            rem = deadline - time.monotonic() - 8
            if rem < 10:
                break
            noise = rng.standard_normal(n_final) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, obj = active_set_refine(f_pert, time_limit=min(rem * 0.5, 20),
                                           delta_frac=0.005, max_active=25)
            try_seq(a_opt, f"p5_pert_{pi}")

        # Final CD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=min(rem * 0.8, 30))
            try_seq(a_opt, "p5_cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
