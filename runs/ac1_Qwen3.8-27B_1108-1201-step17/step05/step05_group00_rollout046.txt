from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_primal(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def solve_lp_dual(f, M, n):
    """Solve dual LP: min M*sum(y) s.t. (f^T y)_j >= 1 for all j, y >= 0.
    Then recover primal from complementary slackness (approximate)."""
    m = 2 * n - 1
    c = M * np.ones(m)
    # Constraint: sum_k y_k * f_{k-j} >= 1 for each j

    A_ub = np.zeros((n, m))
    for j in range(n):
        for k in range(j, min(j + n, m)):
            A_ub[j, k] = -f[k - j]
    b_ub = -np.ones(n)
    bounds = [(0, None)] * m
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    return res

            lo = max(0.0, best_alpha - 0.06)
            hi = min(1.0, best_alpha + 0.06)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new
            m_exp = m_init_factor

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 8.0)

        if no_improve > 150:
            break

    return best_f, best_obj


def logsumexp_mirror_descent(f0, time_limit, verbose=False, beta_start=50.0, beta_end=500.0,
                              lr_init=0.01, lr_end=0.0001):
    """
    Mirror descent minimizing the log-sum-exp smooth approximation of max_k (a*a)_k.
    Uses multiplicative weight updates (negative entropy mirror).
    """
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)
    m = 2 * n - 1

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    total_time = time_limit
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < total_time:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / total_time)

        # Adaptive beta and learning rate
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_init * (lr_end / lr_init) ** progress

        conv = fast_convolve(f)
        M = conv.max()

        # Log-sum-exp weights
        scaled = beta * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient of logsumexp: d/d a_j = 2 * sum_k w_k * a_{k-j} (for valid k-j)
        grad = np.zeros(n)
        for k in range(m):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[k] * f[k - idx]

        # Multiplicative weight update (mirror descent with neg entropy)
        f = f * np.exp(-lr * grad)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        new_obj = fast_convolve(f).max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    LSE-MD iter {it}: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1

        # Periodic restart from best with small perturbation
        if no_improve > 200:
            rng = np.random.default_rng(it * 104729 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 1e-15)
            f = f / f.sum()
            no_improve = 0

        if no_improve > 500:
            break

    return best_f, best_obj


def peak_coordinate_descent(f0, time_limit, verbose=False):
    """Peak-targeted coordinate descent: reduces contributions to the top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Target top peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            j_max = int(idx[np.argmax(products)])

            if f[j_max] < 1e-10:
                continue

            # Try scaling down f[j_max] and redistributing
            best_trial = None
            best_trial_obj = M

            for frac in [0.5, 0.6, 0.7, 0.8, 0.9, 1.1, 1.3]:
                new_val = f[j_max] * frac
                if new_val < 0 or new_val > 0.5:
                    continue
                trial = f.copy()
                trial[j_max] = new_val
                diff = f[j_max] - new_val

                if abs(diff) > 1e-15:
                    # Redistribute to elements NOT contributing to this peak
                    mask = np.ones(n, dtype=bool)
                    mask[js:je + 1] = False
                    if mask.sum() > 0:
                        target = f[mask]
                        tw = target.sum()
                        if tw > 1e-15:
                            trial[mask] += diff * target / tw
                        else:
                            trial[mask] += diff / mask.sum()
                    else:
                        continue

                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-14:
                    best_trial_obj = obj
                    best_trial = trial

            if best_trial is not None:
                f = best_trial
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Try global peak reduction: scale down all contributors to top peak
                k_top = int(np.argmax(fast_convolve(f)))
                js = max(0, k_top - n + 1)
                je = min(n - 1, k_top)

                trial = f.copy()
                trial[js:je + 1] *= 0.9
                mask = np.ones(n, dtype=bool)
                mask[js:je + 1] = False
                diff = f.sum() - trial.sum()
                if mask.sum() > 0:
                    nc = f[mask]
                    nc_sum = nc.sum()
                    if nc_sum > 1e-15:
                        trial[mask] += diff * nc / nc_sum
                    else:
                        trial[mask] += diff / mask.sum()
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

    return best_f, best_obj


def full_coordinate_descent(f0, time_limit, verbose=False):
    """Full coordinate descent: try adjusting each coordinate individually."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(0, n, max(1, n // 200)):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.7, 0.85, 1.0, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = f.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss

                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and pass_count > 2:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_taper(n, power=2):
    """Cosine-tapered window."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2.0) ** (2 * power)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_fourier_flat(n):
    """Inverse FFT of a flat-top window in frequency domain."""
    # Create a flat-top window in frequency
    freq_n = 1 << (2 * n - 1).bit_length()
    freq_vals = np.zeros(freq_n // 2 + 1)
    # Flat top with smooth edges
    center = freq_n // 4
    width = freq_n // 8
    for k in range(freq_n // 2 + 1):
        d = abs(k - center)
        if d < width:
            freq_vals[k] = np.cos(np.pi * d / (2 * width)) ** 2
    # Make it real-symmetric
    freq_full = np.zeros(freq_n)
    freq_full[:freq_n // 2 + 1] = freq_vals
    for k in range(1, freq_n // 2):
        freq_full[freq_n - k] = freq_vals[k]
    a = np.fft.irfft(freq_full, freq_n)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - (n - 1) / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = (1.0 - t ** 4)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
