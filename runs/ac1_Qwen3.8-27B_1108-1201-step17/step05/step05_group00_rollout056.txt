def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_symmetric(a):
    """Enforce palindromic symmetry and normalize."""
    n = len(a)
    s = (a + a[::-1]) / 2.0
    s = np.maximum(s, 0)
    ss = s.sum()
    if ss > 1e-15:
        s = s / ss
    return s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_direction_symmetric(f, M, n):
    """LP with symmetry constraint: g[j] = g[n-1-j]."""
    half = (n + 1) // 2
    c = -np.ones(half)
    
    # Build constraint matrix for symmetric variables
    # g[j] = g[n-1-j], so we parametrize by first half
    A_ub = np.zeros((2 * n - 1, half))
    for j in range(half):
        col_j = np.zeros(n)
        col_j[j] = 1.0
        if j != n - 1 - j:
            col_j[n - 1 - j] = 1.0
        # Convolve with f
        conv_col = np.convolve(f, col_j)[:2*n-1]
        A_ub[:, j] = conv_col
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * half
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g_half = res.x
        g = np.zeros(n)
        for j in range(half):
            g[j] = g_half[j]
            g[n - 1 - j] = g_half[j]
        return g
    return None

def freq_domain_peakhunter(f0, time_limit, verbose=False):
    """
    Frequency-domain peak reduction.
    Works by modifying the power spectrum to reduce autocorrelation peaks,
    then projecting back to non-negative time domain.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # FFT size
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    t0 = time.monotonic()
    alpha = 0.3
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Get top peaks
        top_k = min(8, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]
        peak_vals = conv[peak_indices]
        
        # Softmax weights
        scale = 5.0 / max(peak_vals.max() - peak_vals.min(), 1e-10)
        w = np.exp(scale * (peak_vals - peak_vals.max()))
        w /= w.sum()

        # Compute the "peaking direction" in frequency domain
        # r[k] = (1/N) sum_j P[j] cos(2*pi*j*k/N)
        # To reduce r[k*], decrease P[j] where cos(2*pi*j*k*/N) > 0
        A = np.fft.rfft(f, N)
        P = np.abs(A)**2
        phases = np.angle(A)

        # Compute combined peaking direction
        J = np.arange(len(P))
        peaking_dir = np.zeros(len(P))
        for i in range(top_k):
            k = int(peak_indices[i])
            peaking_dir += w[i] * np.cos(2.0 * np.pi * J * k / N)
        
        # Don't modify DC component
        peaking_dir[0] = 0.0

        # Update power spectrum
        P_new = P - alpha * peaking_dir
        P_new = np.maximum(P_new, 0.0)
        P_new[0] = P[0]  # Keep DC fixed

        # Reconstruct
        A_new = np.sqrt(P_new) * np.exp(1j * phases)
        f_new = np.fft.irfft(A_new, N)[:n]
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = best_f.copy()
            alpha *= 0.5
            continue
        f_new = f_new / ss

        # Check improvement
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < M:
            f = f_new
            alpha = min(alpha * 1.02, 2.0)
        else:
            alpha *= 0.7

        if verbose and it % 100 == 0:
            print(f"    FreqPeak iter {it}: obj={best_obj:.10f}")

        if no_improve > 200:
            # Perturb and restart
            rng = np.random.default_rng(it * 13 + 5)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 0)
            f = f / f.sum()
            no_improve = 0
            alpha = 0.3

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, verbose=False):
    """
    Subgradient descent with top-peaks weighting and adaptive step size.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    step = 0.05
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = min(6, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]
        peak_vals = conv[peak_indices]
        
        scale = 8.0 / max(peak_vals.max() - peak_vals.min(), 1e-10)
        w = np.exp(scale * (peak_vals - peak_vals.max()))
        w /= w.sum()

        # Subgradient
        sg = np.zeros(n)
        for i in range(top_k):
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Step with projection
        f_candidate = f - step * sg
        f_new = project_simplex(f_candidate)

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new <= M * 1.001:
            f = f_new
            step = min(step * 1.03, 0.3)
        else:
            step *= 0.75
            f_new = project_simplex(f - step * sg)
            f = f_new

        if no_improve > 80:
            rng = np.random.default_rng(it * 19 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
            step = 0.05

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-FW with symmetry, frequency-domain peak
    reduction, subgradient descent, and multi-length exploration.
    """

    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(512, 0.28)

    # === Phase 1: Diverse initializations ===
    print("=== Phase 1: Initializations ===")
    init_candidates = []
    for n_try in [384, 512, 640, 768, 896]:
        for sf in [0.22, 0.28, 0.35]:
            init_candidates.append(make_gaussian(n_try, sf))
        init_candidates.append(make_bell(n_try, 0.5))
        init_candidates.append(make_bell(n_try, 0.7))

    if parent is not None:
        init_candidates.append(parent)
        init_candidates.append(make_symmetric(parent))

    best_init = None
    best_init_obj = float('inf')
    for a_init in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm
        try_seq(a_norm, f"init_n{len(a_norm)}")

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: LP-FW with symmetry on best init ===
    print(f"\n=== Phase 2: LP-FW symmetric (n={len(best_init)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350)
    if t_alloc > 30:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(best_init, time_limit=t_alloc, verbose=True, use_symmetry=True)
        try_seq(a_opt, "lpfw_sym")

    # === Phase 3: Frequency-domain peak reduction ===
    print(f"\n=== Phase 3: Frequency-domain ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.15, 180)

    # === Phase 4: LP-FW without symmetry (explore asymmetric solutions) ===
    print(f"\n=== Phase 4: LP-FW asymmetric ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 250)

    print(f"\n=== Phase 7: Deep LP-FW variants ===")

    # === Phase 8: Final frequency-domain polish ===
    print(f"\n=== Phase 8: Final freq polish ===")
    if best_seq is not None:

    print(f"\n=== Phase 9: Final LP-FW ===")

    print(f"\n=== Phase 10: Final subgrad ===")

        if rem > 15:
            t_alloc = min(rem * 0.6, 50)

            a_opt, obj = subgradient_optimize(best_seq, time_limit=t_alloc)

    if best_seq is None:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))

        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
