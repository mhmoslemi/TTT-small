import numpy as np
import scipy.optimize as optimize
from typing import List

def evaluate_sequence(sequence: List[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    total = np.sum(sequence)
    if total < 0.01:
        return np.inf
    convolved = np.convolve(sequence, sequence)
    max_val = np.max(convolved)
    return float(2 * n * max_val / (total ** 2))

def fast_convolve(a):
    """Fast full convolution using FFT."""
    n = len(a)
    fft_size = 1 << (len(a)-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:2*n - 1]

def interpolate_sequence(a, new_n):
    """Interpolate to a new length while preserving normalization."""
    n_old = len(a)
    s = a.sum() / n_old  # Normalization factor
    a_normalized = a / s
    old_x = np.linspace(0, 1, n_old)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, a_normalized)
    f_interp = f_interp / f_interp.sum()  # Renormalize
    return f_interp

    s = a.sum()

    a_normalized = a / s
    best_a = a_normalized.copy()
    best_obj = np.max(fast_convolve(best_a))
    
    t0 = time.monotonic()
    if time_budget <= 0:
        return a_normalized, best_obj
        
    for _ in range(num_steps):
        current_time = time.monotonic()
        if current_time - t0 > time_budget:
            break
            
        conv = fast_convolve(a_normalized)
        peak_indices = np.argpartition(conv, -num_peaks)[-num_peaks:]
        valid_peaks = [k for k in peak_indices if conv[k] > best_obj * 0.95]
        
        if not valid_peaks:
            break
            
        for peak in valid_peaks:
            js = max(0, peak - n + 1)
            je = min(n-1, peak)
            idx = np.arange(js, je+1)
            contributions = a_normalized[idx] * a_normalized[k-idx]
            max_contrib_index = np.argmax(contributions)
            i = idx[max_contrib_index]
            j = peak - i
            
            trial_direction = None
            # Try reducing either i or j

                if target < 0 or target >= n:
                    continue

                if old_val < 1e-10:
                    continue

                w = exp_T_z(a_normalized, lbfgs_T) / old_val  # exp(T*z) from your softmax gradient
                
                factor = 0.95
                new_val = old_val * factor
                a_trial = a_normalized.copy()
                a_trial[target] = new_val
                diff = old_val - new_val
                if diff > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False

                trial_conv = fast_convolve(a_trial)
