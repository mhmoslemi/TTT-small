"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on smoothed objective,
Nesterov subgradient optimization, rapid correlation-based gradient computation,
coordinate descent, and frequency-domain phase annealing.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_conv_gradient(a, w):
    """Compute sum_k w_k * a[k-i] for all i, using FFT.
    
    Parameters:
        a: array of length n (non-negative, sum to 1)
        w: array of length 2n-1 (weights, e.g., softmax of conv)
    
    Returns:
        grad: array of length n where grad[i] = sum_k w[k] * a[k-i]
    """
    n = len(a)
    w_len = len(w)  # should be 2n-1
    # G_j = sum_m a_m * w_{m+j} for j=0..n-1
    # Using: G_j = (a_rev * w)[n-1+j] where a_rev = reversed a
    a_rev = a[::-1]
    size = n + w_len - 1  # = n + 2n-1 - 1 = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(A * W, fft_size)
    # G_j = conv[n-1+j] for j=0..n-1
    return conv[n - 1: 2 * n - 1]


def softmax_obj_and_grad(x, tau=50.0):
    """Compute smoothed objective and gradient in softmax parameterization.
    
    L(x) = 2n/tau * log(sum_k exp(tau * c_k)) where a = softmax(x), c = conv(a,a)
    """
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    S = a.sum()
    a = a / S
    n = len(a)
    
    c = fast_convolve(a)
    max_c = c.max()
    shifted = tau * (c - max_c)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    w = exp_shifted / exp_shifted.sum()
    
    lse = max_c + (max_shifted + np.log(exp_shifted.sum())) / tau
    obj = 2.0 * n * lse
    
    # Gradient: dL/da_i = 2n * sum_k w_k * a[k-i]
    grad_a = compute_conv_gradient(a, w)
    grad_a *= 2.0 * n
    
    # Chain rule through softmax: dL/dx_j = a_j * (grad_a_j - dot(a, grad_a))
    H = np.dot(a, grad_a)
    grad_x = a * (grad_a - H)
    
    return obj, grad_x


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    a = rng.exponential(1.0, n)
    return a / a.sum()


def triangular_init(n):
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return a / a.sum()


def gaussian_init(n):
    x = np.linspace(-3, 3, n)
    a = np.exp(-x**2 / 2)
    return a / a.sum()


def phase_anneal(n, rng, n_iter=300, cooling=0.999):
    """Frequency-domain phase annealing with non-negativity projection."""
    best_a = None
    best_obj = np.inf
    temp = 1.0
    
    # Initialize
    A = np.fft.fft(flat_spectrum_init(n, rng))
    
    for it in range(n_iter):
        temp *= cooling
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            A = np.fft.fft(rng.exponential(1.0, n))
            continue
        a = a / s
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Update: add random phase perturbation scaled by temperature
        A_new = np.fft.fft(a)
        phase_pert = rng.standard_normal(n) * temp * 0.1
        new_phases = np.angle(A_new) + phase_pert
        new_mags = np.abs(A_new)
        
        # Blend mags towards flat
        target_mag = np.ones(n) * np.sqrt(new_mags.mean() ** 2)
        mags_blend = 0.7 * new_mags + 0.3 * target_mag
        
        A = mags_blend * np.exp(1j * new_phases)
        # Ensure A[0] is real and positive (DC component)
        A[0] = np.abs(A[0])
        if n % 2 == 0:
            A[n // 2] = np.abs(A[n // 2])
        # Ensure conjugate symmetry
        for k in range(1, (n // 2) + 1):
            if k == n - k:
                continue
            avg = (A[k] + np.conj(A[n - k])) / 2
            A[k] = avg
            A[n - k] = np.conj(avg)
    
    return best_a


def l_bfgs_optimize(a0, tau=50.0, max_iter=200, time_limit=100.0):
    """Optimize using L-BFGS on the smoothed objective."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    # Initialize x from a: x_i = log(a_i) (with guard for zeros)
    x0 = np.where(a > 1e-15, np.log(a + 1e-300), -5.0)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    def fun(x):
        if time.monotonic() - t0 > time_limit:
            return np.inf
        obj, _ = softmax_obj_and_grad(x, tau)
        return obj
    
    def jac(x):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(len(x))
        _, g = softmax_obj_and_grad(x, tau)
        return g
    
    try:
        result = minimize(fun, x0, jac=jac, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'maxfun': 500,
                                  'ftol': 1e-10, 'gtol': 1e-8,
                                  'iprint': -1})
    except Exception:
        return best_a, best_obj
    
    if result.x is not None and time.monotonic() - t0 < time_limit:
        x_best = result.x - result.x.max()
        a_new = np.exp(x_best)
        a_new = a_new / a_new.sum()
        obj_new = 2.0 * n * fast_convolve(a_new).max()
        if obj_new < best_obj:
            best_a = a_new
            best_obj = obj_new
    
    return best_a, best_obj


def nesterov_subgrad(a0, time_limit=100.0, tau=80.0, lr=0.05, momentum=0.85):
    """Subgradient optimization with Nesterov momentum."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(y)
        max_c = c.max()
        shifted = tau * (c - max_c)
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        
        grad = compute_conv_gradient(y, w)
        # Center for simplex
        grad = grad - grad.mean()
        
        v = momentum * v - lr * grad
        a_new = project_simplex(y + v, target_sum=1.0)
        y = a_new  # In Nesterov, y is the lookahead point

def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent with multiplicative updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
