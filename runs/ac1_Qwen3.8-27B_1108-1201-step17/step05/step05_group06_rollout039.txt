"""

Uses improved log-sum-exp smooth optimization, peak-aware coordinate descent,

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective_from_conv(conv):
    """Compute objective from convolution, assuming sum(a)=1."""
    n = (len(conv) + 1) // 2
    return 2.0 * n * np.max(conv)

def smooth_objective_and_grad(f, T):
    """
    Compute smooth objective (log-sum-exp of conv/T) and gradient.
    f is normalized to sum=1.
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (np.log(e.sum()))  # This is the smooth approximation to max(conv)
    
    # grad_i = (2/T) * sum_j f[j] * w[i+j] for i=0..n-1
    # w has length 2n-1, indexed 0 to 2n-2
    # For gradient wrt f[i]: d(conv[k])/df[i] = f[k-i] for valid k-i
    # So grad[i] = (2/T) * sum_k w[k] * f[k-i] (for valid k-i)
    # = (2/T) * cross_correlation(w, f)[i]
    
    # cross_correlation(w, f)[i] = sum_k w[k] * f[k-i]
    # This is the same as: for each i, sum over k of w[k]*f[k-i] where 0<=k-i<n
    
    # Using FFT: cross_correlation = ifft(fft(w) * conj(fft(f)))
    N = 1 << (2 * n).bit_length()
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(F), N)
    grad = (2.0 / T) * corr[:n]
    
    return lse_val, grad, conv, w

def smooth_optimize_v2(f0, time_limit, T_start=0.1, T_end=0.001, verbose=False):
    """
    Improved smooth optimization with:
    - Better temperature schedule (slow decay)
    - Backtracking line search
    - Periodic restarts
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_conv_max = fast_convolve(f).max()
    best_lse = best_conv_max  # for tracking
    
    # Adam states
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    # Hyperparameters
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_iters = []
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        
        # Temperature schedule: fast initial decay, then slow
        if progress < 0.3:
            T = T_start * (0.3 / T_start) ** (progress / 0.3)
        else:
            T = 0.3 * (T_end / 0.3) ** ((progress - 0.3) / 0.7)
        T = max(T, T_end)
        
        # Learning rate schedule
        lr = alpha * (1.0 - 0.5 * progress)
        
        lse_val, grad, conv, w = smooth_objective_and_grad(f, T)
        
        # Adam update
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        cur_max = conv_new.max()
        
        if cur_max < best_conv_max - 1e-13:
            best_conv_max = cur_max
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"  Smooth v2 iter {it}: max_conv={best_conv_max:.10f}, T={T:.5f}")
        else:
            no_improve += 1
        
        f = f_new
        
        # Restart if stuck
        if no_improve > 2000:
            seed_val = int(elapsed * 1000 + it) % 1000000
            rng_p = np.random.default_rng(seed_val)
            scale = min(0.005 + 0.002 * (no_improve // 500), 0.02)
            f = project_simplex(best_f + rng_p.standard_normal(n) * scale)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_f, best_conv_max

def peak_cd(f0, time_limit, verbose=False):
    """
    Peak-aware coordinate descent:
    For each coordinate, find the value that minimizes the max convolution.
    Uses the fact that changing one coordinate changes conv in a simple way.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_max = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        conv = fast_convolve(f)
        cur_max = conv.max()
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            # When we change f[j] from old_val to new_val (keeping sum=1),
            # the other elements scale by (1-new_val)/(1-old_val)
            # conv changes as:
            # conv[k] = sum_i f[i]*f[k-i]
            # The terms involving j: f[j]*f[k-j] + f[k-j]*f[j] - f[j]*f[j] (if k=2j)
            # Actually: conv[k] includes f[j]*f[k-j] if 0<=k-j<n, and f[k-j]*f[j] if 0<=j<n
            # For k = j + m: term is f[j]*f[m] (from i=j) and f[m]*f[j] (from i=m, if m != j)
            
            # Simpler: just evaluate directly
            # Try a few values of new_val
            best_new_val = old_val
            best_local_max = cur_max
            
            for frac in [0.5, 0.7, 0.8, 0.9, 0.95, 1.1, 1.2, 1.5, 2.0]:
                new_val = old_val * frac
                if new_val < 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / (1.0 - old_val) if old_val < 1.0 else 1.0
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                local_max = fast_convolve(f_trial).max()
                if local_max < best_local_max - 1e-14:
                    best_local_max = local_max
                    best_new_val = new_val
            
            if best_new_val != old_val:
                new_val = best_new_val
                scale = (1.0 - new_val) / (1.0 - old_val) if old_val < 1.0 else 1.0
                f[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f /= f.sum()
                if best_local_max < best_max - 1e-14:
                    best_max = best_local_max
                    best_f = f.copy()
                    improved = True
        
        if not improved and pass_count > 5:
            break
    
    return best_f, best_max

def lbf_smooth_optimize(f0, time_limit, verbose=False):
    """
    L-BFGS based optimization of the smooth objective.
    Uses scipy's L-BFGS-B with boundary conditions.
    """
    f_start = np.maximum(f0.astype(np.float64), 0.0)
    s = f_start.sum()
    if s < 1e-15:
        f_start = np.ones(len(f0)) / len(f0)
    else:
        f_start = f_start / s
    n = len(f_start)
    
    best_f = f_start.copy()
    best_max = fast_convolve(f_start).max()
    
    T = 0.02  # fixed moderate temperature for L-BFGS
    
    def loss_and_grad(x):
        x = np.maximum(x, 0)
        x = x / x.sum()
        lse, grad, conv, w = smooth_objective_and_grad(x, T)
        # We need the gradient wrt the original x (before normalization)
        # Since we normalize inside, the effective gradient is:
        # grad_normalized = grad - (sum(grad) * x) (tangent plane correction)
        grad_tangent = grad - np.sum(grad * x) * x
        return lse, grad_tangent
    
    t0 = time.monotonic()
    bounds = [(0, None)] * n
    
    # Run L-BFGS-B with time limit
    # Since scipy doesn't support time limits directly, we'll run for a fixed number of iterations
    n_iter = min(max(50, int(time_limit / 0.5)), 200)
    
    try:
        res = minimize(
            fun=loss_and_grad,
            x0=f_start,
            jac=True,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': n_iter, 'maxfun': n_iter * 2, 'disp': False}
        )
        x_opt = np.maximum(res.x, 0)
        x_opt = x_opt / x_opt.sum()
        conv_opt = fast_convolve(x_opt)
        if conv_opt.max() < best_max:
            best_max = conv_opt.max()
            best_f = x_opt.copy()
    except Exception as e:
        if verbose:
            print(f"  L-BFGS error: {e}")
    
    return best_f, best_max

def make_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Geometric (various ratios)
    for r in [0.97, 0.98, 0.99, 0.995]:
        a = r ** np.arange(n, dtype=np.float64)
        inits.append((f'geo_{r}', a / a.sum()))
    
    # Symmetric geometric
    for r in [0.97, 0.98, 0.99]:
        half = (n + 1) // 2
        asc = r ** np.arange(half, dtype=np.float64)
        desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
        a = np.concatenate([asc, desc])
        if len(a) < n:
            a = np.concatenate([a, np.ones(n - len(a))])
        inits.append((f'symgeo_{r}', a / a.sum()))
    
    # Cosine windows
    for pw in [1.0, 1.5, 2.0, 3.0, 4.0, 6.0]:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** pw + 0.01
        inits.append((f'cos_{pw}', a / a.sum()))
    
    # Raised cosine
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.01
    inits.append(('raised_cos', a / a.sum()))
    
    # Modulated: (1 + alpha*cos) * window
    for alpha in [0.3, 0.5, 0.7]:
        for period in [n // 3, n // 5, n // 7]:
            if period < 2:
                continue
            x = np.linspace(0, 1, n)
            window = np.sin(np.pi * x) ** 2 + 0.05
            mod = 1 + alpha * np.cos(2 * np.pi * x * period * 2 / n)
            a = np.maximum(window * mod, 0)
            if a.sum() > 0:
                inits.append((f'mod_a{alpha}_p{period}', a / a.sum()))
    
    # Random smooth (Gaussian process-like)
    for _ in range(5):
        freqs = rng.integers(1, 5, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        amps = rng.uniform(0.5, 2.0, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for freq, phase, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0)
        if a.sum() > 0:
            inits.append(('random_smooth', a / a.sum()))
    
    return inits

def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
