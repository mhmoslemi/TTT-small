import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def cross_corr_weighted(w, a):
    """Compute 2*sum_k w[k]*a[k-m] for m=0..n-1 using FFT."""
    n = len(a)
    N = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    return 2.0 * R


def fista_optimize(f0, time_limit, verbose=False):
    """FISTA with annealed log-sum-exp smoothing on the simplex."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    y = a.copy()
    t_k = 1.0
    beta = 20.0
    step = 0.3 / n
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_y = fast_convolve(y)
        max_y = conv_y.max()
        shifted = beta * (conv_y - max_y)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()

        grad = cross_corr_weighted(w, y)

        z = y - step * grad
        a_new = project_simplex(z)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        t_k = t_new
        a = a_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    FISTA iter {it}: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1

        if no_improve > 50:
            beta *= 1.4
            step *= 0.85
            no_improve = 0
        elif no_improve > 200:
            step *= 0.7
            no_improve = 0
            if step < 1e-6 / n:
                break

        if step > 0.5 / n:
            step = 0.5 / n

    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns g normalized."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, tight_M=True):
    """LP-based Frank-Wolfe with adaptive M, golden-section line search, and smart restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        target_M = M * m_exp if tight_M else M * 1.5

        g = solve_lp_direction(f, target_M, n)
        if g is None:
            m_exp = min(m_exp * 1.2, 8.0)
            if m_exp > 8.0:
                m_exp = 1.0
            continue

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                scale = 0.002 + 0.004 * min(no_improve // 80, 4)
                noise = rng.standard_normal(n) * scale
                f = project_simplex(best_f + noise)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            m_exp = min(m_exp * 1.05, 6.0)
            continue

        # Golden section line search
        phi = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c1 = lo + (1.0 - phi) * (hi - lo)
        c2 = lo + phi * (hi - lo)

        ft1 = np.maximum(f + c1 * (g - f), 0.0)
        ft1 = ft1 / ft1.sum()
        val1 = fast_convolve(ft1).max()

        ft2 = np.maximum(f + c2 * (g - f), 0.0)
        ft2 = ft2 / ft2.sum()
        val2 = fast_convolve(ft2).max()

        for _ in range(25):
            if val1 < val2:
                hi = c2
                c2 = c1
                val2 = val1
                c1 = lo + (1.0 - phi) * (hi - lo)
                ft1 = np.maximum(f + c1 * (g - f), 0.0)
                ft1 = ft1 / ft1.sum()
                val1 = fast_convolve(ft1).max()
            else:
                lo = c1
                c1 = c2
                val1 = val2
                c2 = lo + phi * (hi - lo)
                ft2 = np.maximum(f + c2 * (g - f), 0.0)
                ft2 = ft2 / ft2.sum()
                val2 = fast_convolve(ft2).max()

        best_alpha = (lo + hi) / 2.0
        best_val = min(val1, val2)

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        M = conv.max()
        m_exp = 1.0
        f = f_new

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 100:
                scale = 0.003 + 0.003 * min(no_improve // 100, 5)
                noise = rng.standard_normal(n) * scale
                f = project_simplex(best_f + noise)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0

    return best_f, best_obj


def spectral_smooth(f0, time_limit, verbose=False):
    """Spectral peak-flattening: iteratively modify |A(f)|^2 to flatten auto-convolution."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    gamma = 0.1
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        conv_max = conv.max()
        if conv_max < best_obj - 1e-12:
            best_obj = conv_max
            best_a = a.copy()

        # Get the "power spectrum" |A(f)|^2
        N = 1 << (2 * n).bit_length()
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2

        # Identify peaks in the auto-convolution
        k_peak = np.argmax(conv)

        # Modify a to reduce the peak: reduce mass at positions contributing to peak
        # The peak at k_peak is sum_j a[j]*a[k_peak-j]
        # Gradient w.r.t. a[m] is 2*a[k_peak-m]
        k_actual = k_peak
        js = max(0, k_actual - n + 1)
        je = min(n - 1, k_actual)
        contrib = np.zeros(n)
        idx = np.arange(js, je + 1)
        contrib[idx] = 2.0 * a[k_actual - idx]

        # Normalize gradient
        c_norm = contrib.sum()
        if c_norm < 1e-15:
            break
        contrib = contrib / c_norm

        # Update: move mass from high-contrib positions to low-contrib positions
        a_new = a - gamma * contrib
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        # Also try the reverse direction (massive positions get more mass from
        # non-contributing positions)
        a_new2 = a + gamma * (1.0 / n - contrib)
        a_new2 = np.maximum(a_new2, 0.0)
        a_new2 = a_new2 / a_new2.sum()

        obj1 = fast_convolve(a_new).max()
        obj2 = fast_convolve(a_new2).max()

        if obj1 < best_obj:
            a = a_new
        elif obj2 < best_obj:
            a = a_new2
        else:
            gamma *= 0.9
            if gamma < 1e-5:
                break

    return best_a, best_obj


def lbfqg_optimize(f0, time_limit, verbose=False):
    """L-BFGS on smooth log-sum-exp objective in softmax parameterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)
    betas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    t0_time = time.monotonic()

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)
        max_iter = max(50, int(sub_time * 100))

        try:
            res = minimize(
                _smooth_obj_grad_impl, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-10, 'maxls': 40,
                         'maxfun': max_iter * 5}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            break

    return best_f, best_obj


def _smooth_obj_grad_impl(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta
    grad_a = cross_corr_weighted(w, a)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.75, 0.9, 1.05, 1.2, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_starts(base, n, rng, num_starts=5):

def block_lp_fw(f0, time_limit, block_size=64, verbose=False):
    """Block coordinate LP-FW: optimize blocks of coordinates alternately."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    num_blocks = (n + block_size - 1) // block_size
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        for b in range(num_blocks):
            if time.monotonic() - t0 > time_limit - 10:
                break
            start = b * block_size
            end = min(start + block_size, n)
            size = end - start

            if size < 2:
                continue

            # Solve LP for this block only
            f_block = f[start:end]
            # Constraints: for each k, sum_j f[j]*g[k-j] <= M
            # For block variables g[start:end], the constraint at k is:
            # sum_{j in block} f[j]*g_block[k-j] + sum_{j not in block} f[j]*g[k-j] <= M
            # => sum_{j in block} f[j]*g_block[k-j] <= M - sum_{j not in block} f[j]*g[k-j]

            conv_full = fast_convolve(f)
            M = conv_full.max()

            # For each constraint k, the "remaining" after fixing non-block variables:
            # residual[k] = M - conv_full[k] + sum_{j in block} f[j]*f[k-j]
            # (we subtract the block's contribution from the current conv and add back M)
            block_contrib = np.zeros(2 * n - 1)
            for j in range(start, end):
                block_contrib[j:j + n] += f[j] * f[:n]
                # Actually this is getting complicated. Let me simplify.
            # Instead, just do a simple local search on the block.
            best_block = f_block.copy()
            best_block_obj = M

            for trial_scale in [0.5, 0.8, 1.2, 1.5]:
                for shift in [-0.02, -0.01, 0.01, 0.02]:
                    block_trial = f_block * trial_scale + shift
                    block_trial = np.maximum(block_trial, 0)
                    bs = block_trial.sum()
                    if bs < 1e-10:
                        continue
                    # Rescale to maintain total sum
                    old_sum = f_block.sum()
                    if old_sum < 1e-10:
                        continue
                    scale_factor = 1.0 - (bs - old_sum) / (1.0 - old_sum + 1e-15)
                    if scale_factor <= 0:
                        continue

                    f_trial = f.copy()
                    f_trial[start:end] = block_trial
                    f_trial[:start] *= scale_factor
                    f_trial[end:] *= scale_factor
                    f_trial = f_trial / f_trial.sum()
                    obj = fast_convolve(f_trial).max()
                    if obj < best_block_obj - 1e-14:
                        best_block_obj = obj
                        best_block = block_trial
                        f = f_trial

            if best_block_obj < best_obj - 1e-14:
                best_obj = best_block_obj
                best_f = f.copy()
                improved = True

        if not improved and it > 5:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining FISTA, LP-FW, spectral smoothing,
    L-BFGS, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)
    print(f"\nStarting optimization. Parent n={n_parent}, score={best_score:.6f}")

    # === Phase 1: FISTA quick improvement (150s) ===
    print("\n=== Phase 1: FISTA Quick ===")
    t1 = min(150, deadline - time.monotonic() - 1200)
    if t1 > 30:
        a_opt, _ = fista_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "fista_quick")

    # === Phase 2: LP-FW on best (350s) ===
    print(f"\n=== Phase 2: LP-FW Main (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(350, deadline - time.monotonic() - 1000)
    if t2 > 40:
        a_opt, _ = lp_fw_optimize(base, time_limit=t2, verbose=True, tight_M=True)
        try_seq(a_opt, "lpfw_main")

    # === Phase 3: L-BFGS (250s) ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t3 = min(250, deadline - time.monotonic() - 700)
    if t3 > 30:
        a_opt, _ = lbfqg_optimize(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "lbfqg_main")

        # L-BFGS from diverse starts
        t3b = min(100, deadline - time.monotonic() - 500)
        if t3b > 30:
            for si, start in enumerate(diverse_starts(base, len(base), rng, num_starts=3)):
                rem = deadline - time.monotonic() - 400
                if rem < 20:
                    break
                a_opt, _ = lbfqg_optimize(start, time_limit=min(rem * 0.3, 40))
                try_seq(a_opt, f"lbfqg_divers_{si}")

    # === Phase 4: Spectral smoothing (100s) ===
    print(f"\n=== Phase 4: Spectral (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t4 = min(100, deadline - time.monotonic() - 400)
    if t4 > 20:
        a_opt, _ = spectral_smooth(base, time_limit=t4)
        try_seq(a_opt, "spectral")

    # === Phase 5: Multi-length exploration (250s) ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = sorted(set([n_parent // 2, int(n_parent * 0.8), n_parent,
                                  int(n_parent * 1.2), n_parent * 2,
                                  512, 640, 896, 1024, 1280, 1536, 1792]))
    target_lengths = [n for n in target_lengths if 256 <= n <= 2048]
    target_lengths = sorted(target_lengths)

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        # Quick FISTA to warm up
        t_fista = min(rem * 0.2, 20)
        if t_fista > 5:
            f_interp, _ = fista_optimize(f_interp, time_limit=t_fista)
        # LP-FW
        t_lpfw = min(rem * 0.5, 40)
        if t_lpfw > 10:
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_lpfw)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 6: Deep optimization on best (350s) ===
    print(f"\n=== Phase 6: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 6a: LP-FW deep
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_alloc = min(rem * 0.5, 180)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True, tight_M=True)
            try_seq(a_opt, "deep_lpfw")

        # 6b: FISTA deep
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.5, 100)
            a_opt, _ = fista_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fista")

        # 6c: L-BFGS deep
        rem = deadline - time.monotonic() - 120
        if rem > 25:
            t_alloc = min(rem * 0.6, 100)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # 6d: Coordinate descent
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem * 0.5, 60)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_cd")

        # 6e: Spectral deep
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem * 0.6, 40)
            a_opt, _ = spectral_smooth(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_spectral")

    # === Phase 7: Final polishing (remaining) ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        for attempt in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break

            strategy = attempt % 4
            if strategy == 0:
                # LP-FW with perturbation
                noise = rng.standard_normal(n_best) * 0.005
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.6, 60)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            elif strategy == 1:
                # FISTA
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = fista_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_fista_{attempt}")
            elif strategy == 2:
                # L-BFGS
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_lbfqg_{attempt}")
            else:
                # Coordinate descent
                t_alloc = min(rem * 0.6, 40)
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_cd_{attempt}")

        # Very final: one more LP-FW on absolute best
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 40), tight_M=True)
            try_seq(a_opt, "absolute_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
